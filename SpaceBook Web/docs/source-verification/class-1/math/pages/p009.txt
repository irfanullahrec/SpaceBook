# Counting || گنتی || شمېرل
p Let's count the objects and read. || آئیں چیزیں گنیں اور پڑھیں۔ || راځئ شیان وشمېرو او ولولو.
cnt "0" "zero" || nothing — 0 — zero || کچھ نہیں — ۰ — صفر || هېڅ نه — ۰ — صفر
cnt watermelon:1 "1" "one" || 1 watermelon — 1 — one || ۱ تربوز — ۱ — ایک || ۱ هندوانه — ۱ — یو
cnt mango:2 "2" "two" || 2 mangoes — 2 — two || ۲ آم — ۲ — دو || ۲ امونه — ۲ — دوه
cnt candy:3 "3" "three" || 3 sweets — 3 — three || ۳ ٹافیاں — ۳ — تین || ۳ خوږې — ۳ — درې
cnt jelly:4 "4" "four" || 4 jellies — 4 — four || ۴ جیلیاں — ۴ — چار || ۴ جېلي — ۴ — څلور
cnt tart:5 "5" "five" || 5 tarts — 5 — five || ۵ ٹارٹ — ۵ — پانچ || ۵ ټارټونه — ۵ — پنځه
cnt pumpkin:6 "6" "six" || 6 pumpkins — 6 — six || ۶ کدو — ۶ — چھ || ۶ کدوګان — ۶ — شپږ
cnt date:7 "7" "seven" || 7 dates — 7 — seven || ۷ کھجوریں — ۷ — سات || ۷ خرما — ۷ — اووه
cnt donut:8 "8" "eight" || 8 doughnuts — 8 — eight || ۸ ڈونٹ — ۸ — آٹھ || ۸ ډونټونه — ۸ — اته
cnt orange:9 "9" "nine" || 9 oranges — 9 — nine || ۹ مالٹے — ۹ — نو || ۹ مالټې — ۹ — نهه
box teach | Teaching Point || تدریسی نکتہ || د تدریس ټکی
p Paste a wall chart (zero to nine) on the wall. Then ask students to read aloud. || دیوار پر (صفر سے نو تک) چارٹ لگائیں۔ پھر طلبہ سے بلند آواز میں پڑھنے کو کہیں۔ || په دېوال د (صفر نه تر نهو پورې) چارټ ولګوئ. بیا له زده کوونکو وغواړئ چې په لوړ غږ یې ولولي.
endbox
