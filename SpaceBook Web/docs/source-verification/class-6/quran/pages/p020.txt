p **وَلَا يَدْخُلُوْنَ الْجَنَّةَ حَتّٰى يَلِجَ الْجَمَلُ فِيْ سَمِّ الْخِيَاطِ ؕ وَكَذٰلِكَ نَجْزِي الْمُجْرِمِيْنَ ۝٤٠** || **وَلَا يَدْخُلُوْنَ الْجَنَّةَ حَتّٰى يَلِجَ الْجَمَلُ فِيْ سَمِّ الْخِيَاطِ ؕ وَكَذٰلِكَ نَجْزِي الْمُجْرِمِيْنَ ۝٤٠** || **وَلَا يَدْخُلُوْنَ الْجَنَّةَ حَتّٰى يَلِجَ الْجَمَلُ فِيْ سَمِّ الْخِيَاطِ ؕ وَكَذٰلِكَ نَجْزِي الْمُجْرِمِيْنَ ۝٤٠**
p ترجمہ: اور نہ ہی وہ جنت میں داخل ہوں گے یہاں تک کہ اونٹ سوئی کے ناکہ میں داخل ہو جائے اور ہم مجرموں کو اسی طرح بدلہ دیتے ہیں۔ || nor will they enter Paradise until a camel passes through the eye of a needle. Thus do We recompense the criminals. || او نه به جنت ته ننوځي تر دې چې اوښ د ستنې په سوري کې ننوځي؛ او موږ مجرمانو ته همداسې بدله ورکوو.
p **لَهُمْ مِّنْ جَهَنَّمَ مِهَادٌ وَّمِنْ فَوْقِهِمْ غَوَاشٍ ؕ وَكَذٰلِكَ نَجْزِي الظّٰلِمِيْنَ ۝٤١ وَالَّذِيْنَ اٰمَنُوْا** || **لَهُمْ مِّنْ جَهَنَّمَ مِهَادٌ وَّمِنْ فَوْقِهِمْ غَوَاشٍ ؕ وَكَذٰلِكَ نَجْزِي الظّٰلِمِيْنَ ۝٤١ وَالَّذِيْنَ اٰمَنُوْا** || **لَهُمْ مِّنْ جَهَنَّمَ مِهَادٌ وَّمِنْ فَوْقِهِمْ غَوَاشٍ ؕ وَكَذٰلِكَ نَجْزِي الظّٰلِمِيْنَ ۝٤١ وَالَّذِيْنَ اٰمَنُوْا**
p ترجمہ: اُن کے لیے جہنم کا بچھونا ہو گا اور اُن کے اوپر (اُسی کا) اوڑھنا ہو گا اور ہم ظالموں کو اِسی طرح بدلہ دیتے ہیں۔ اور جو ایمان لائے || They will have a bed of Hell and coverings (of it) over them. Thus do We recompense the wrongdoers. But those who believe || د دوی لپاره به د دوزخ بستره وي او پاس به یې (د هغه) پوښ وي؛ او ظالمانو ته همداسې بدله ورکوو. او څوک چې ایمان راوړي
p **وَعَمِلُوا الصّٰلِحٰتِ لَا نُكَلِّفُ نَفْسًا اِلَّا وُسْعَهَآ ؗ اُولٰٓئِكَ اَصْحٰبُ الْجَنَّةِ ۚ هُمْ فِيْهَا خٰلِدُوْنَ ۝٤٢** || **وَعَمِلُوا الصّٰلِحٰتِ لَا نُكَلِّفُ نَفْسًا اِلَّا وُسْعَهَآ ؗ اُولٰٓئِكَ اَصْحٰبُ الْجَنَّةِ ۚ هُمْ فِيْهَا خٰلِدُوْنَ ۝٤٢** || **وَعَمِلُوا الصّٰلِحٰتِ لَا نُكَلِّفُ نَفْسًا اِلَّا وُسْعَهَآ ؗ اُولٰٓئِكَ اَصْحٰبُ الْجَنَّةِ ۚ هُمْ فِيْهَا خٰلِدُوْنَ ۝٤٢**
p ترجمہ: اور انھوں نے نیک اعمال کیے ہم کسی شخص پر اس کی طاقت سے زیادہ بوجھ نہیں ڈالتے یہی لوگ جنت والے ہیں وہ اس میں ہمیشہ رہنے والے ہیں۔ || and do good deeds — We burden no soul beyond its capacity — they are the people of Paradise; they will remain in it forever. || او نېک عملونه وکړي — موږ پر هېچا تر وس زیات بار نه اچوو — همدوی د جنت خاوندان دي، تل به په کې وي.
p **وَنَزَعْنَا مَا فِيْ صُدُوْرِهِمْ مِّنْ غِلٍّ تَجْرِيْ مِنْ تَحْتِهِمُ الْاَنْهٰرُ ۚ وَقَالُوا الْحَمْدُ لِلّٰهِ** || **وَنَزَعْنَا مَا فِيْ صُدُوْرِهِمْ مِّنْ غِلٍّ تَجْرِيْ مِنْ تَحْتِهِمُ الْاَنْهٰرُ ۚ وَقَالُوا الْحَمْدُ لِلّٰهِ** || **وَنَزَعْنَا مَا فِيْ صُدُوْرِهِمْ مِّنْ غِلٍّ تَجْرِيْ مِنْ تَحْتِهِمُ الْاَنْهٰرُ ۚ وَقَالُوا الْحَمْدُ لِلّٰهِ**
p ترجمہ: اور اُن کے سینوں میں جو رنجش تھی وہ ہم نکال لیں گے اُن (محلات) کے نیچے نہریں بہتی ہوں گی اور وہ کہیں گے کہ تمام تعریفیں اللہ کے لیے ہیں || And We will remove whatever ill-feeling is in their hearts; rivers will flow beneath them (their palaces). And they will say, "All praise is for Allah, || او د دوی په سینو کې چې کینه وه، هغه به وباسو؛ د (ماڼیو) لاندې به یې ویالې بهېږي، او وايي به: "ټوله ستاینه د الله لپاره ده
p **الَّذِيْ هَدٰىنَا لِهٰذَا ۟ وَمَا كُنَّا لِنَهْتَدِيَ لَوْلَآ اَنْ هَدٰىنَا اللّٰهُ ۚ** || **الَّذِيْ هَدٰىنَا لِهٰذَا ۟ وَمَا كُنَّا لِنَهْتَدِيَ لَوْلَآ اَنْ هَدٰىنَا اللّٰهُ ۚ** || **الَّذِيْ هَدٰىنَا لِهٰذَا ۟ وَمَا كُنَّا لِنَهْتَدِيَ لَوْلَآ اَنْ هَدٰىنَا اللّٰهُ ۚ**
p ترجمہ: جس نے ہمیں اِس (جنت کے مقام) کی راہ دکھائی اور ہم راہ پانے والے نہ ہوتے اگر اللہ ہمیں ہدایت نہ عطا فرماتا || who guided us to this (place in Paradise); we would never have been guided had Allah not guided us. || چې دې (د جنت ځای) ته یې لار راوښوده؛ او موږ به هېڅکله لار نه وای موندلې که الله هدایت نه وای راکړی.
p **لَقَدْ جَآءَتْ رُسُلُ رَبِّنَا بِالْحَقِّ ؕ وَنُوْدُوْٓا اَنْ تِلْكُمُ الْجَنَّةُ اُوْرِثْتُمُوْهَا** || **لَقَدْ جَآءَتْ رُسُلُ رَبِّنَا بِالْحَقِّ ؕ وَنُوْدُوْٓا اَنْ تِلْكُمُ الْجَنَّةُ اُوْرِثْتُمُوْهَا** || **لَقَدْ جَآءَتْ رُسُلُ رَبِّنَا بِالْحَقِّ ؕ وَنُوْدُوْٓا اَنْ تِلْكُمُ الْجَنَّةُ اُوْرِثْتُمُوْهَا**
p ترجمہ: یقیناً ہمارے رب کے رسول حق کے ساتھ آئے تھے اور انھیں پکار کر کہا جائے گا کہ یہی وہ جنت ہے جس کے تم وارث بنائے گئے ہو || The messengers of our Lord certainly came with the truth." And it will be called out to them: "This is the Paradise you have been made to inherit || بېشکه زموږ د رب رسولان په حق راغلي وو." او غږ به ورته وشي: "همدا هغه جنت دی چې تاسو یې وارثان شوي یئ
p **بِمَا كُنْتُمْ تَعْمَلُوْنَ ۝٤٣ وَنَادٰٓى اَصْحٰبُ الْجَنَّةِ اَصْحٰبَ النَّارِ اَنْ قَدْ وَجَدْنَا مَا وَعَدَنَا رَبُّنَا حَقًّا** || **بِمَا كُنْتُمْ تَعْمَلُوْنَ ۝٤٣ وَنَادٰٓى اَصْحٰبُ الْجَنَّةِ اَصْحٰبَ النَّارِ اَنْ قَدْ وَجَدْنَا مَا وَعَدَنَا رَبُّنَا حَقًّا** || **بِمَا كُنْتُمْ تَعْمَلُوْنَ ۝٤٣ وَنَادٰٓى اَصْحٰبُ الْجَنَّةِ اَصْحٰبَ النَّارِ اَنْ قَدْ وَجَدْنَا مَا وَعَدَنَا رَبُّنَا حَقًّا**
p ترجمہ: اُن اعمال کی وجہ سے جو تم کیا کرتے تھے۔ اور جنت والے آگ والوں کو پکاریں گے کہ یقیناً ہمارے رب نے ہم سے جو وعدہ کیا تھا ہم نے اُس کو سچا پا لیا || because of what you used to do." And the people of Paradise will call to the people of the Fire: "We have found what our Lord promised us to be true. || د هغو کړنو له امله چې کولې مو." او جنتیان به دوزخیانو ته غږ وکړي: "زموږ رب چې څه ژمنه کړې وه، موږ ریښتیا وموندله.
p **فَهَلْ وَجَدْتُّمْ مَّا وَعَدَ رَبُّكُمْ حَقًّا ؕ قَالُوْا نَعَمْ ۚ فَاَذَّنَ مُؤَذِّنٌۢ بَيْنَهُمْ** || **فَهَلْ وَجَدْتُّمْ مَّا وَعَدَ رَبُّكُمْ حَقًّا ؕ قَالُوْا نَعَمْ ۚ فَاَذَّنَ مُؤَذِّنٌۢ بَيْنَهُمْ** || **فَهَلْ وَجَدْتُّمْ مَّا وَعَدَ رَبُّكُمْ حَقًّا ؕ قَالُوْا نَعَمْ ۚ فَاَذَّنَ مُؤَذِّنٌۢ بَيْنَهُمْ**
p ترجمہ: تو کیا تم نے بھی اُس وعدے کو سچا پایا جو تمھارے رب نے کیا تھا وہ کہیں گے ہاں! پھر اُن کے درمیان ایک اعلان کرنے والا اعلان کرے گا || Have you also found what your Lord promised to be true?" They will say, "Yes!" Then a caller will announce among them: || ایا تاسو هم د خپل رب ژمنه ریښتیا وموندله؟" وايي به: "هو!" بیا به یو اعلان کوونکی د دوی تر منځ اعلان وکړي:
p **اَنْ لَّعْنَةُ اللّٰهِ عَلَى الظّٰلِمِيْنَ ۝٤٤ الَّذِيْنَ يَصُدُّوْنَ عَنْ سَبِيْلِ اللّٰهِ وَيَبْغُوْنَهَا عِوَجًا ۚ وَهُمْ بِالْاٰخِرَةِ كٰفِرُوْنَ ۝٤٥** || **اَنْ لَّعْنَةُ اللّٰهِ عَلَى الظّٰلِمِيْنَ ۝٤٤ الَّذِيْنَ يَصُدُّوْنَ عَنْ سَبِيْلِ اللّٰهِ وَيَبْغُوْنَهَا عِوَجًا ۚ وَهُمْ بِالْاٰخِرَةِ كٰفِرُوْنَ ۝٤٥** || **اَنْ لَّعْنَةُ اللّٰهِ عَلَى الظّٰلِمِيْنَ ۝٤٤ الَّذِيْنَ يَصُدُّوْنَ عَنْ سَبِيْلِ اللّٰهِ وَيَبْغُوْنَهَا عِوَجًا ۚ وَهُمْ بِالْاٰخِرَةِ كٰفِرُوْنَ ۝٤٥**
p ترجمہ: کہ ظالموں پر اللہ کی لعنت ہے۔ وہ جو اللہ کی راہ سے روکتے تھے اور اُس میں ٹیڑھ تلاش کرتے تھے اور وہ آخرت کا انکار کرنے والے تھے۔ || "The curse of Allah is on the wrongdoers, who used to turn (people) away from the path of Allah and seek to make it crooked, and they denied the Hereafter." || "پر ظالمانو د الله لعنت دی، هغه چې له الله لارې یې منع کول او په کې یې کوږوالی لټاوه او آخرت یې نه مانه."
p **وَبَيْنَهُمَا حِجَابٌ ۚ وَعَلَى الْاَعْرَافِ رِجَالٌ يَّعْرِفُوْنَ كُلًّاۢ بِسِيْمٰىهُمْ ۚ** || **وَبَيْنَهُمَا حِجَابٌ ۚ وَعَلَى الْاَعْرَافِ رِجَالٌ يَّعْرِفُوْنَ كُلًّاۢ بِسِيْمٰىهُمْ ۚ** || **وَبَيْنَهُمَا حِجَابٌ ۚ وَعَلَى الْاَعْرَافِ رِجَالٌ يَّعْرِفُوْنَ كُلًّاۢ بِسِيْمٰىهُمْ ۚ**
p ترجمہ: اور ان دونوں (جنتیوں اور جہنمیوں) کے درمیان پردہ ہو گا اور اعراف پر کچھ لوگ ہوں گے جو سب کو اُن کی صورتوں سے پہچانتے ہوں گے || Between the two (the people of Paradise and Hell) there will be a barrier, and on al-A'raf there will be men who recognise everyone by their marks. || د دواړو (جنتیانو او دوزخیانو) تر منځ به پرده وي، او پر اعراف به ځینې کسان وي چې ټول به په نښو پېژني.
p **وَنَادَوْا اَصْحٰبَ الْجَنَّةِ اَنْ سَلٰمٌ عَلَيْكُمْ ۟ لَمْ يَدْخُلُوْهَا وَهُمْ يَطْمَعُوْنَ ۝٤٦** || **وَنَادَوْا اَصْحٰبَ الْجَنَّةِ اَنْ سَلٰمٌ عَلَيْكُمْ ۟ لَمْ يَدْخُلُوْهَا وَهُمْ يَطْمَعُوْنَ ۝٤٦** || **وَنَادَوْا اَصْحٰبَ الْجَنَّةِ اَنْ سَلٰمٌ عَلَيْكُمْ ۟ لَمْ يَدْخُلُوْهَا وَهُمْ يَطْمَعُوْنَ ۝٤٦**
p ترجمہ: اور وہ جنت والوں کو پکاریں گے کہ تم پر سلامتی ہو وہ ابھی اس (جنت) میں داخل نہیں ہوئے ہوں گے اور وہ اُس کی امید رکھتے ہوں گے۔ || They will call to the people of Paradise, "Peace be upon you." They will not yet have entered it, though they hope to. || جنتیانو ته به غږ وکړي: "سلام دې وي پر تاسو." لا به (جنت ته) نه وي ننوتي، خو هیله به یې لري.
p **وَاِذَا صُرِفَتْ اَبْصَارُهُمْ تِلْقَآءَ اَصْحٰبِ النَّارِ ۙ قَالُوْا رَبَّنَا لَا تَجْعَلْنَا مَعَ الْقَوْمِ الظّٰلِمِيْنَ ۝٤٧** || **وَاِذَا صُرِفَتْ اَبْصَارُهُمْ تِلْقَآءَ اَصْحٰبِ النَّارِ ۙ قَالُوْا رَبَّنَا لَا تَجْعَلْنَا مَعَ الْقَوْمِ الظّٰلِمِيْنَ ۝٤٧** || **وَاِذَا صُرِفَتْ اَبْصَارُهُمْ تِلْقَآءَ اَصْحٰبِ النَّارِ ۙ قَالُوْا رَبَّنَا لَا تَجْعَلْنَا مَعَ الْقَوْمِ الظّٰلِمِيْنَ ۝٤٧**
p ترجمہ: اور جب اُن کی نگاہیں آگ والوں کی طرف پھیری جائیں گی (تو) وہ عرض کریں گے اے ہمارے ربّ! ہمیں ظالم قوم کے ساتھ شامل نہ فرما۔ (رکوع 5) || And when their eyes are turned towards the people of the Fire, they will say, "Our Lord! Do not place us with the wrongdoing people." (End of ruku' 5) || او کله چې سترګې یې دوزخیانو ته واړول شي، وايي به: "ای زموږ ربه! موږ له ظالم قوم سره مه شاملوه." (پنځم رکوع)
