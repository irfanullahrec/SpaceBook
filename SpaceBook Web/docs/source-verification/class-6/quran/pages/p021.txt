p **وَنَادٰٓى اَصْحٰبُ الْاَعْرَافِ رِجَالًا يَّعْرِفُوْنَهُمْ بِسِيْمٰىهُمْ قَالُوْا مَآ اَغْنٰى عَنْكُمْ جَمْعُكُمْ** || **وَنَادٰٓى اَصْحٰبُ الْاَعْرَافِ رِجَالًا يَّعْرِفُوْنَهُمْ بِسِيْمٰىهُمْ قَالُوْا مَآ اَغْنٰى عَنْكُمْ جَمْعُكُمْ** || **وَنَادٰٓى اَصْحٰبُ الْاَعْرَافِ رِجَالًا يَّعْرِفُوْنَهُمْ بِسِيْمٰىهُمْ قَالُوْا مَآ اَغْنٰى عَنْكُمْ جَمْعُكُمْ**

p **وَمَا كُنْتُمْ تَسْتَكْبِرُوْنَ ۝٤٨ اَهٰٓؤُلَآءِ الَّذِيْنَ اَقْسَمْتُمْ** || **وَمَا كُنْتُمْ تَسْتَكْبِرُوْنَ ۝٤٨ اَهٰٓؤُلَآءِ الَّذِيْنَ اَقْسَمْتُمْ** || **وَمَا كُنْتُمْ تَسْتَكْبِرُوْنَ ۝٤٨ اَهٰٓؤُلَآءِ الَّذِيْنَ اَقْسَمْتُمْ**
p ترجمہ: اور نہ اس (مال و اسباب) نے جس پر تم تکبر کیا کرتے تھے۔ کیا یہ (جنتی) ہیں جن کے بارے میں تم قسمیں اُٹھاتے تھے || nor did that (wealth) of which you were arrogant. Are these (the people of Paradise) the ones about whom you swore || او نه هغه (مال) چې کبر مو پرې کاوه. ایا دا (جنتیان) هغه دي چې تاسو قسمونه خوړل
p **لَا يَنَالُهُمُ اللّٰهُ بِرَحْمَةٍ ؕ اُدْخُلُوا الْجَنَّةَ لَا خَوْفٌ عَلَيْكُمْ** || **لَا يَنَالُهُمُ اللّٰهُ بِرَحْمَةٍ ؕ اُدْخُلُوا الْجَنَّةَ لَا خَوْفٌ عَلَيْكُمْ** || **لَا يَنَالُهُمُ اللّٰهُ بِرَحْمَةٍ ؕ اُدْخُلُوا الْجَنَّةَ لَا خَوْفٌ عَلَيْكُمْ**
p ترجمہ: کہ اللہ اُن پر اپنی رحمت نہیں کرے گا (انھیں یہ فرما دیا گیا) کہ جنت میں داخل ہو جاؤ تم پر نہ کوئی خوف ہے || that Allah would never extend His mercy to them? (They have been told:) 'Enter Paradise; no fear will be on you, || چې الله به پرې رحمت ونه کړي؟ (هغوی ته وویل شول:) جنت ته ننوځئ، پر تاسو به نه وېره وي
p **وَلَآ اَنْتُمْ تَحْزَنُوْنَ ۝٤٩ وَنَادٰٓى اَصْحٰبُ النَّارِ اَصْحٰبَ الْجَنَّةِ اَنْ اَفِيْضُوْا عَلَيْنَا مِنَ الْمَآءِ** || **وَلَآ اَنْتُمْ تَحْزَنُوْنَ ۝٤٩ وَنَادٰٓى اَصْحٰبُ النَّارِ اَصْحٰبَ الْجَنَّةِ اَنْ اَفِيْضُوْا عَلَيْنَا مِنَ الْمَآءِ** || **وَلَآ اَنْتُمْ تَحْزَنُوْنَ ۝٤٩ وَنَادٰٓى اَصْحٰبُ النَّارِ اَصْحٰبَ الْجَنَّةِ اَنْ اَفِيْضُوْا عَلَيْنَا مِنَ الْمَآءِ**
p ترجمہ: اور نہ تم غمگین ہو گے۔ اور آگ والے جنت والوں کو پکاریں گے کہ تھوڑا سا پانی ہم پر بہا دو || nor will you grieve.'" And the people of the Fire will call to the people of Paradise, "Pour some water on us, || او نه به غمجن شئ." او دوزخیان به جنتیانو ته غږ وکړي: "لږ اوبه راباندې توی کړئ
p **اَوْ مِمَّا رَزَقَكُمُ اللّٰهُ ؕ قَالُوْٓا اِنَّ اللّٰهَ حَرَّمَهُمَا عَلَى الْكٰفِرِيْنَ ۝٥٠** || **اَوْ مِمَّا رَزَقَكُمُ اللّٰهُ ؕ قَالُوْٓا اِنَّ اللّٰهَ حَرَّمَهُمَا عَلَى الْكٰفِرِيْنَ ۝٥٠** || **اَوْ مِمَّا رَزَقَكُمُ اللّٰهُ ؕ قَالُوْٓا اِنَّ اللّٰهَ حَرَّمَهُمَا عَلَى الْكٰفِرِيْنَ ۝٥٠**
p ترجمہ: یا اُس رزق میں سے (کچھ دے دو) جو اللہ نے تمھیں دیا ہے (جنتی) کہیں گے کہ بے شک اللہ نے یہ دونوں چیزیں کافروں پر حرام کر دی ہیں۔ || or (give us) some of what Allah has provided you." (The people of Paradise) will say, "Allah has forbidden both to the disbelievers, || یا له هغه رزق (څه راکړئ) چې الله درکړی." (جنتیان) به ووايي: "بېشکه الله دا دواړه پر کافرانو حرام کړي دي،
p **الَّذِيْنَ اتَّخَذُوْا دِيْنَهُمْ لَهْوًا وَّلَعِبًا وَّغَرَّتْهُمُ الْحَيٰوةُ الدُّنْيَا ۚ فَالْيَوْمَ نَنْسٰىهُمْ** || **الَّذِيْنَ اتَّخَذُوْا دِيْنَهُمْ لَهْوًا وَّلَعِبًا وَّغَرَّتْهُمُ الْحَيٰوةُ الدُّنْيَا ۚ فَالْيَوْمَ نَنْسٰىهُمْ** || **الَّذِيْنَ اتَّخَذُوْا دِيْنَهُمْ لَهْوًا وَّلَعِبًا وَّغَرَّتْهُمُ الْحَيٰوةُ الدُّنْيَا ۚ فَالْيَوْمَ نَنْسٰىهُمْ**
p ترجمہ: جنھوں نے اپنے دین کو کھیل اور تماشا بنا لیا تھا اور دنیا کی زندگی نے انھیں دھوکا میں ڈال دیا تھا تو آج ہم انھیں (اسی طرح) نظر انداز کر دیں گے || who took their religion as amusement and play, and whom the life of this world deceived." So today We will ignore them || هغو چې خپل دین یې لوبه او تماشه ګرځولی و او د دنیا ژوند غولولي وو." نو نن به موږ (همداسې) هغوی هېر کړو
p **كَمَا نَسُوْا لِقَآءَ يَوْمِهِمْ هٰذَا ۙ وَمَا كَانُوْا بِاٰيٰتِنَا يَجْحَدُوْنَ ۝٥١ وَلَقَدْ جِئْنٰهُمْ بِكِتٰبٍ** || **كَمَا نَسُوْا لِقَآءَ يَوْمِهِمْ هٰذَا ۙ وَمَا كَانُوْا بِاٰيٰتِنَا يَجْحَدُوْنَ ۝٥١ وَلَقَدْ جِئْنٰهُمْ بِكِتٰبٍ** || **كَمَا نَسُوْا لِقَآءَ يَوْمِهِمْ هٰذَا ۙ وَمَا كَانُوْا بِاٰيٰتِنَا يَجْحَدُوْنَ ۝٥١ وَلَقَدْ جِئْنٰهُمْ بِكِتٰبٍ**
p ترجمہ: جیسے انھوں نے اِس دن کی ملاقات کو بُھلا دیا تھا اور جس طرح وہ ہماری آیات کا انکار کیا کرتے تھے۔ اور یقیناً ہم اُن کے پاس ایک کتاب لائے || as they forgot the meeting of this Day of theirs, and as they used to deny Our verses. And We have certainly brought them a Book || لکه څنګه چې دوی د دې ورځې ملاقات هېر کړی و او زموږ آیتونه یې نه منل. او بېشکه موږ دوی ته یو کتاب راوړ
p **فَصَّلْنٰهُ عَلٰى عِلْمٍ هُدًى وَّرَحْمَةً لِّقَوْمٍ يُّؤْمِنُوْنَ ۝٥٢ هَلْ يَنْظُرُوْنَ اِلَّا تَاْوِيْلَهٗ ؕ** || **فَصَّلْنٰهُ عَلٰى عِلْمٍ هُدًى وَّرَحْمَةً لِّقَوْمٍ يُّؤْمِنُوْنَ ۝٥٢ هَلْ يَنْظُرُوْنَ اِلَّا تَاْوِيْلَهٗ ؕ** || **فَصَّلْنٰهُ عَلٰى عِلْمٍ هُدًى وَّرَحْمَةً لِّقَوْمٍ يُّؤْمِنُوْنَ ۝٥٢ هَلْ يَنْظُرُوْنَ اِلَّا تَاْوِيْلَهٗ ؕ**
p ترجمہ: جسے ہم نے اپنے علم کی بنیاد پر واضح کر دیا (وہ) ہدایت اور رحمت ہے مومنوں کے لیے۔ کیا وہ انجام کا انتظار کر رہے ہیں || which We have explained in detail with knowledge — guidance and mercy for people who believe. Are they waiting only for its outcome? || چې په خپل علم مو تفصیل کړی، (هغه) د مومنانو لپاره هدایت او رحمت دی. ایا دوی یوازې د هغه انجام ته سترګې په لار دي؟
p **يَوْمَ يَاْتِيْ تَاْوِيْلُهٗ يَقُوْلُ الَّذِيْنَ نَسُوْهُ مِنْ قَبْلُ قَدْ جَآءَتْ رُسُلُ رَبِّنَا بِالْحَقِّ ۚ** || **يَوْمَ يَاْتِيْ تَاْوِيْلُهٗ يَقُوْلُ الَّذِيْنَ نَسُوْهُ مِنْ قَبْلُ قَدْ جَآءَتْ رُسُلُ رَبِّنَا بِالْحَقِّ ۚ** || **يَوْمَ يَاْتِيْ تَاْوِيْلُهٗ يَقُوْلُ الَّذِيْنَ نَسُوْهُ مِنْ قَبْلُ قَدْ جَآءَتْ رُسُلُ رَبِّنَا بِالْحَقِّ ۚ**
p ترجمہ: جس دن اس کا انجام سامنے آ جائے گا جنھوں نے اسے پہلے بُھلا رکھا تھا وہ کہیں گے یقیناً ہمارے رب کے رسول حق لائے تھے || On the Day its outcome arrives, those who had forgotten it before will say, "The messengers of our Lord did bring the truth. || په هغه ورځ چې انجام یې راشي، هغه چې مخکې یې هېر کړی و، وايي به: "بېشکه زموږ د رب رسولان حق راوړی و.
p **فَهَلْ لَّنَا مِنْ شُفَعَآءَ فَيَشْفَعُوْا لَنَآ اَوْ نُرَدُّ فَنَعْمَلَ غَيْرَ الَّذِيْ كُنَّا نَعْمَلُ ؕ** || **فَهَلْ لَّنَا مِنْ شُفَعَآءَ فَيَشْفَعُوْا لَنَآ اَوْ نُرَدُّ فَنَعْمَلَ غَيْرَ الَّذِيْ كُنَّا نَعْمَلُ ؕ** || **فَهَلْ لَّنَا مِنْ شُفَعَآءَ فَيَشْفَعُوْا لَنَآ اَوْ نُرَدُّ فَنَعْمَلَ غَيْرَ الَّذِيْ كُنَّا نَعْمَلُ ؕ**
p ترجمہ: تو کیا (آج) ہمارے کوئی سفارشی ہیں جو ہمارے لیے سفارش کریں یا ہمیں واپس بھیج دیا جائے تاکہ ہم اُس کے برعکس عمل کریں جو ہم کیا کرتے تھے || Are there any intercessors to intercede for us (today)? Or could we be sent back to do other than what we used to do?" || نو ایا (نن) کوم شفاعت کوونکي لرو چې زموږ شفاعت وکړي؟ یا بېرته ولېږل شو چې له پخوانیو کړنو خلاف عمل وکړو؟"
p **قَدْ خَسِرُوْٓا اَنْفُسَهُمْ وَضَلَّ عَنْهُمْ مَّا كَانُوْا يَفْتَرُوْنَ ۝٥٣** || **قَدْ خَسِرُوْٓا اَنْفُسَهُمْ وَضَلَّ عَنْهُمْ مَّا كَانُوْا يَفْتَرُوْنَ ۝٥٣** || **قَدْ خَسِرُوْٓا اَنْفُسَهُمْ وَضَلَّ عَنْهُمْ مَّا كَانُوْا يَفْتَرُوْنَ ۝٥٣**
p ترجمہ: یقیناً انھوں نے اپنے آپ کو خسارے میں ڈال دیا اور ان سے گم ہو گئے جو وہ جھوٹ باندھا کرتے تھے۔ (رکوع 6) || They have truly ruined themselves, and what they used to invent has vanished from them.
