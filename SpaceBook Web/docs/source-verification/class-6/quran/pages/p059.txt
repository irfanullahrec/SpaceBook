p **وَاِنَّكُمْ اِذًا لَّمِنَ الْمُقَرَّبِيْنَ ۝٤٢ قَالَ لَهُمْ مُّوْسٰٓى اَلْقُوْا مَآ اَنْتُمْ مُّلْقُوْنَ ۝٤٣** || **وَاِنَّكُمْ اِذًا لَّمِنَ الْمُقَرَّبِيْنَ ۝٤٢ قَالَ لَهُمْ مُّوْسٰٓى اَلْقُوْا مَآ اَنْتُمْ مُّلْقُوْنَ ۝٤٣** || **وَاِنَّكُمْ اِذًا لَّمِنَ الْمُقَرَّبِيْنَ ۝٤٢ قَالَ لَهُمْ مُّوْسٰٓى اَلْقُوْا مَآ اَنْتُمْ مُّلْقُوْنَ ۝٤٣**
p ترجمہ: اور تم اس صورت میں یقیناً قربت والوں میں سے ہو جاؤ گے۔ موسیٰ (علیہ السلام) نے ان سے فرمایا ڈالو جو کچھ تم ڈالنے والے ہو۔ || and you will then surely be among those brought near." Musa said to them, "Throw what you are going to throw." || او په هغه صورت کې به خامخا له نږدې کسانو شئ." موسی (علیه السلام) ورته وویل: "هغه وغورځوئ چې غورځوئ یې."
p **فَاَلْقَوْا حِبَالَهُمْ وَعِصِيَّهُمْ وَقَالُوْا بِعِزَّةِ فِرْعَوْنَ اِنَّا لَنَحْنُ الْغٰلِبُوْنَ ۝٤٤** || **فَاَلْقَوْا حِبَالَهُمْ وَعِصِيَّهُمْ وَقَالُوْا بِعِزَّةِ فِرْعَوْنَ اِنَّا لَنَحْنُ الْغٰلِبُوْنَ ۝٤٤** || **فَاَلْقَوْا حِبَالَهُمْ وَعِصِيَّهُمْ وَقَالُوْا بِعِزَّةِ فِرْعَوْنَ اِنَّا لَنَحْنُ الْغٰلِبُوْنَ ۝٤٤**
p ترجمہ: تو انھوں نے اپنی رسیاں اور اپنی لاٹھیاں (زمین پر) ڈالیں اور وہ کہنے لگے فرعون کی عزت کی قسم! یقیناً ہم ضرور غالب ہوں گے۔ || So they threw their ropes and staffs, and said, "By the might of Pharaoh, we will surely be the winners." || نو هغوی خپل رسۍ او لرګي (پر ځمکه) وغورځول او ویې ویل: "د فرعون په عزت قسم! بېشکه موږ به خامخا بریالي شو."
p **فَاَلْقٰى مُوْسٰى عَصَاهُ فَاِذَا هِيَ تَلْقَفُ مَا يَاْفِكُوْنَ ۝٤٥ فَاُلْقِيَ السَّحَرَةُ سٰجِدِيْنَ ۝٤٦** || **فَاَلْقٰى مُوْسٰى عَصَاهُ فَاِذَا هِيَ تَلْقَفُ مَا يَاْفِكُوْنَ ۝٤٥ فَاُلْقِيَ السَّحَرَةُ سٰجِدِيْنَ ۝٤٦** || **فَاَلْقٰى مُوْسٰى عَصَاهُ فَاِذَا هِيَ تَلْقَفُ مَا يَاْفِكُوْنَ ۝٤٥ فَاُلْقِيَ السَّحَرَةُ سٰجِدِيْنَ ۝٤٦**
p ترجمہ: پھر موسیٰ (علیہ السلام) نے اپنا عصا ڈالا تو وہ اچانک نگلنے لگا ان جھوٹی چیزوں کو جو انھوں نے بنائی تھیں۔ تو جادوگر سجدے میں گر پڑے۔ || Then Musa threw his staff, and at once it swallowed up what they had falsely made. And the magicians fell down in prostration. || بیا موسی (علیه السلام) امسا وغورځوله، ناڅاپه یې هغه دروغجن شیان تېر کړل چې جوړ کړي یې وو. نو جادوګر په سجده کې ولوېدل.
p **قَالُوْٓا اٰمَنَّا بِرَبِّ الْعٰلَمِيْنَ ۝٤٧ رَبِّ مُوْسٰى وَهٰرُوْنَ ۝٤٨** || **قَالُوْٓا اٰمَنَّا بِرَبِّ الْعٰلَمِيْنَ ۝٤٧ رَبِّ مُوْسٰى وَهٰرُوْنَ ۝٤٨** || **قَالُوْٓا اٰمَنَّا بِرَبِّ الْعٰلَمِيْنَ ۝٤٧ رَبِّ مُوْسٰى وَهٰرُوْنَ ۝٤٨**
p ترجمہ: وہ کہنے لگے ہم سارے جہانوں کے رب پر ایمان لائے۔ جو موسیٰ (علیہ السلام) اور ہارون (علیہ السلام) کا رب ہے۔ || They said, "We believe in the Lord of all the worlds, the Lord of Musa and Harun." || ویې ویل: "موږ د ټولو نړیو پر رب ایمان راوړ، چې د موسی او هارون (علیهما السلام) رب دی."
p **قَالَ اٰمَنْتُمْ لَهٗ قَبْلَ اَنْ اٰذَنَ لَكُمْ ۚ اِنَّهٗ لَكَبِيْرُكُمُ** || **قَالَ اٰمَنْتُمْ لَهٗ قَبْلَ اَنْ اٰذَنَ لَكُمْ ۚ اِنَّهٗ لَكَبِيْرُكُمُ** || **قَالَ اٰمَنْتُمْ لَهٗ قَبْلَ اَنْ اٰذَنَ لَكُمْ ۚ اِنَّهٗ لَكَبِيْرُكُمُ**
p ترجمہ: (فرعون نے) کہا تم اس پر ایمان لے آئے ہو اس سے پہلے کہ میں تمھیں اجازت دیتا، یقیناً وہ تمھارا بڑا ہے || (Pharaoh) said, "You believed in him before I gave you permission? He is surely your chief || (فرعون) وویل: "ایا پرې ایمان راوړ مخکې له دې چې زه اجازه درکړم؟ بېشکه هغه ستاسو مشر دی
p **الَّذِيْ عَلَّمَكُمُ السِّحْرَ ۚ فَلَسَوْفَ تَعْلَمُوْنَ ؕ لَاُقَطِّعَنَّ اَيْدِيَكُمْ وَاَرْجُلَكُمْ مِّنْ خِلَافٍ** || **الَّذِيْ عَلَّمَكُمُ السِّحْرَ ۚ فَلَسَوْفَ تَعْلَمُوْنَ ؕ لَاُقَطِّعَنَّ اَيْدِيَكُمْ وَاَرْجُلَكُمْ مِّنْ خِلَافٍ** || **الَّذِيْ عَلَّمَكُمُ السِّحْرَ ۚ فَلَسَوْفَ تَعْلَمُوْنَ ؕ لَاُقَطِّعَنَّ اَيْدِيَكُمْ وَاَرْجُلَكُمْ مِّنْ خِلَافٍ**
p ترجمہ: جس نے تمھیں جادو سکھایا ہے تو عنقریب تم جان لو گے میں ضرور تمھارے ہاتھ اور تمھارے پاؤں کاٹوں گا مخالف سمت سے || who taught you magic. You will soon know! I will surely cut off your hands and feet on opposite sides || چې جادو یې در زده کړی؛ ژر به پوه شئ! خامخا به ستاسو لاسونه او پښې له مخالف اړخه پرې کړم
p **وَّلَاُصَلِّبَنَّكُمْ اَجْمَعِيْنَ ۝٤٩ قَالُوْا لَا ضَيْرَ ؗ اِنَّآ اِلٰى رَبِّنَا مُنْقَلِبُوْنَ ۝٥٠** || **وَّلَاُصَلِّبَنَّكُمْ اَجْمَعِيْنَ ۝٤٩ قَالُوْا لَا ضَيْرَ ؗ اِنَّآ اِلٰى رَبِّنَا مُنْقَلِبُوْنَ ۝٥٠** || **وَّلَاُصَلِّبَنَّكُمْ اَجْمَعِيْنَ ۝٤٩ قَالُوْا لَا ضَيْرَ ؗ اِنَّآ اِلٰى رَبِّنَا مُنْقَلِبُوْنَ ۝٥٠**
p ترجمہ: اور ضرور تم سب کو سولی پر لٹکا دوں گا۔ انھوں نے کہا کوئی نقصان نہیں بے شک ہم اپنے رب کے پاس پہنچنے والے ہیں۔ || and crucify you all." They said, "No harm! We will surely return to our Lord. || او خامخا به مو ټول په دار ځړوم." ویې ویل: "هېڅ زیان نشته؛ بېشکه موږ خپل رب ته رسېدونکي یو.
p **اِنَّا نَطْمَعُ اَنْ يَّغْفِرَ لَنَا رَبُّنَا خَطٰيٰنَآ اَنْ كُنَّآ اَوَّلَ الْمُؤْمِنِيْنَ ۝٥١** || **اِنَّا نَطْمَعُ اَنْ يَّغْفِرَ لَنَا رَبُّنَا خَطٰيٰنَآ اَنْ كُنَّآ اَوَّلَ الْمُؤْمِنِيْنَ ۝٥١** || **اِنَّا نَطْمَعُ اَنْ يَّغْفِرَ لَنَا رَبُّنَا خَطٰيٰنَآ اَنْ كُنَّآ اَوَّلَ الْمُؤْمِنِيْنَ ۝٥١**
p ترجمہ: بے شک ہم اُمید رکھتے ہیں کہ ہمارا رب ہماری خطائیں بخش دے گا اس لیے کہ ہم سب سے پہلے ایمان لانے والے ہیں۔ (رکوع 3) || We hope our Lord will forgive us our sins, because we are the first of the believers." (End of ruku' 3) || بېشکه هیله لرو چې زموږ رب به زموږ خطاوې وبښي، ځکه چې موږ تر ټولو لومړي مومنان یو." (درېیم رکوع)
p **وَاَوْحَيْنَآ اِلٰى مُوْسٰٓى اَنْ اَسْرِ بِعِبَادِيْٓ اِنَّكُمْ مُّتَّبَعُوْنَ ۝٥٢** || **وَاَوْحَيْنَآ اِلٰى مُوْسٰٓى اَنْ اَسْرِ بِعِبَادِيْٓ اِنَّكُمْ مُّتَّبَعُوْنَ ۝٥٢** || **وَاَوْحَيْنَآ اِلٰى مُوْسٰٓى اَنْ اَسْرِ بِعِبَادِيْٓ اِنَّكُمْ مُّتَّبَعُوْنَ ۝٥٢**
p ترجمہ: اور ہم نے موسیٰ (علیہ السلام) کی طرف وحی بھیجی کہ میرے بندوں کو راتوں رات لے جاؤ بے شک تمھارا پیچھا کیا جائے گا۔ || And We revealed to Musa, "Travel by night with My servants; you will surely be pursued." || او موږ موسی (علیه السلام) ته وحی وکړه چې "زما بندګان د شپې بوځه؛ بېشکه پسې به درشي."
p **فَاَرْسَلَ فِرْعَوْنُ فِي الْمَدَآئِنِ حٰشِرِيْنَ ۝٥٣ اِنَّ هٰٓؤُلَآءِ لَشِرْذِمَةٌ قَلِيْلُوْنَ ۝٥٤** || **فَاَرْسَلَ فِرْعَوْنُ فِي الْمَدَآئِنِ حٰشِرِيْنَ ۝٥٣ اِنَّ هٰٓؤُلَآءِ لَشِرْذِمَةٌ قَلِيْلُوْنَ ۝٥٤** || **فَاَرْسَلَ فِرْعَوْنُ فِي الْمَدَآئِنِ حٰشِرِيْنَ ۝٥٣ اِنَّ هٰٓؤُلَآءِ لَشِرْذِمَةٌ قَلِيْلُوْنَ ۝٥٤**
p ترجمہ: پھر فرعون نے شہروں میں جمع کرنے والے بھیج دیے۔ (اور کہا کہ) بے شک یہ لوگ ضرور چھوٹی سی جماعت ہیں۔ || Then Pharaoh sent gatherers to the cities, (saying,) "These are but a small band, || بیا فرعون ښارونو ته راټولوونکي ولېږل. (او ویې ویل:) "بېشکه دا خلک یوه وړه ډله ده
p **وَاِنَّهُمْ لَنَا لَغَآئِظُوْنَ ۝٥٥ وَاِنَّا لَجَمِيْعٌ حٰذِرُوْنَ ۝٥٦ فَاَخْرَجْنٰهُمْ مِّنْ جَنّٰتٍ وَّعُيُوْنٍ ۝٥٧** || **وَاِنَّهُمْ لَنَا لَغَآئِظُوْنَ ۝٥٥ وَاِنَّا لَجَمِيْعٌ حٰذِرُوْنَ ۝٥٦ فَاَخْرَجْنٰهُمْ مِّنْ جَنّٰتٍ وَّعُيُوْنٍ ۝٥٧** || **وَاِنَّهُمْ لَنَا لَغَآئِظُوْنَ ۝٥٥ وَاِنَّا لَجَمِيْعٌ حٰذِرُوْنَ ۝٥٦ فَاَخْرَجْنٰهُمْ مِّنْ جَنّٰتٍ وَّعُيُوْنٍ ۝٥٧**
