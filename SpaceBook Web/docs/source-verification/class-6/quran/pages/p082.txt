# سبق نمبر 11: سُوْرَةُ الصّٰٓفّٰتِ || Lesson 11: Surah As-Saffat || یوولسم لوست: سورة الصافات
p آیات: 83 تا 182 || Verses 83 to 182 || آیتونه: 83 تر 182
p **وَاِنَّ مِنْ شِيْعَتِهٖ لَاِبْرٰهِيْمَ ۝٨٣ اِذْ جَآءَ رَبَّهٗ بِقَلْبٍ سَلِيْمٍ ۝٨٤** || **وَاِنَّ مِنْ شِيْعَتِهٖ لَاِبْرٰهِيْمَ ۝٨٣ اِذْ جَآءَ رَبَّهٗ بِقَلْبٍ سَلِيْمٍ ۝٨٤** || **وَاِنَّ مِنْ شِيْعَتِهٖ لَاِبْرٰهِيْمَ ۝٨٣ اِذْ جَآءَ رَبَّهٗ بِقَلْبٍ سَلِيْمٍ ۝٨٤**
p ترجمہ: اور بے شک انھی کے گروہ میں سے ضرور ابراہیم (علیہ السلام بھی) تھے۔ جب وہ اپنے رب کے پاس پاک دل لے کر آئے۔ || And among his (Nuh's) followers was Ibrahim, when he came to his Lord with a sound (pure) heart, || او بېشکه د هغه (نوح) له ډلې ابراهیم (علیه السلام هم) و، کله چې خپل رب ته یې پاک زړه راوړ،
p **اِذْ قَالَ لِاَبِيْهِ وَقَوْمِهٖ مَاذَا تَعْبُدُوْنَ ۝٨٥ اَئِفْكًا اٰلِهَةً دُوْنَ اللّٰهِ تُرِيْدُوْنَ ۝٨٦** || **اِذْ قَالَ لِاَبِيْهِ وَقَوْمِهٖ مَاذَا تَعْبُدُوْنَ ۝٨٥ اَئِفْكًا اٰلِهَةً دُوْنَ اللّٰهِ تُرِيْدُوْنَ ۝٨٦** || **اِذْ قَالَ لِاَبِيْهِ وَقَوْمِهٖ مَاذَا تَعْبُدُوْنَ ۝٨٥ اَئِفْكًا اٰلِهَةً دُوْنَ اللّٰهِ تُرِيْدُوْنَ ۝٨٦**
p ترجمہ: جب انھوں نے اپنے باپ اور اپنی قوم سے فرمایا کہ تم کن چیزوں کو پوجتے ہو؟ کیا تم اللہ کو چھوڑ کر گھڑے ہوئے معبودوں کو چاہتے ہو؟ || when he said to his father and his people, "What do you worship? Do you want false gods instead of Allah? || کله چې خپل پلار او قوم ته یې وویل: "د څه شي عبادت کوئ؟ ایا له الله پرته جوړ کړي معبودان غواړئ؟
p **فَمَا ظَنُّكُمْ بِرَبِّ الْعٰلَمِيْنَ ۝٨٧ فَنَظَرَ نَظْرَةً فِي النُّجُوْمِ ۝٨٨ فَقَالَ اِنِّيْ سَقِيْمٌ ۝٨٩** || **فَمَا ظَنُّكُمْ بِرَبِّ الْعٰلَمِيْنَ ۝٨٧ فَنَظَرَ نَظْرَةً فِي النُّجُوْمِ ۝٨٨ فَقَالَ اِنِّيْ سَقِيْمٌ ۝٨٩** || **فَمَا ظَنُّكُمْ بِرَبِّ الْعٰلَمِيْنَ ۝٨٧ فَنَظَرَ نَظْرَةً فِي النُّجُوْمِ ۝٨٨ فَقَالَ اِنِّيْ سَقِيْمٌ ۝٨٩**
p ترجمہ: تو تمام جہانوں کے رب (کے بارے) میں تمھارا کیا خیال ہے؟ پھر (ایک مرتبہ ابراہیم علیہ السلام نے) ستاروں کی طرف ایک نظر ڈالی۔ پھر فرمایا میں بیمار ہوں۔ || What then do you think of the Lord of all the worlds?" Then he cast a glance at the stars and said, "I am unwell." || نو د ټولو نړیو د رب په اړه ستاسو څه ګومان دی؟" بیا (ابراهیم) یو ځل ستورو ته وکتل او ویې ویل: "زه ناروغه یم."
p **فَتَوَلَّوْا عَنْهُ مُدْبِرِيْنَ ۝٩٠ فَرَاغَ اِلٰٓى اٰلِهَتِهِمْ فَقَالَ اَلَا تَاْكُلُوْنَ ۝٩١** || **فَتَوَلَّوْا عَنْهُ مُدْبِرِيْنَ ۝٩٠ فَرَاغَ اِلٰٓى اٰلِهَتِهِمْ فَقَالَ اَلَا تَاْكُلُوْنَ ۝٩١** || **فَتَوَلَّوْا عَنْهُ مُدْبِرِيْنَ ۝٩٠ فَرَاغَ اِلٰٓى اٰلِهَتِهِمْ فَقَالَ اَلَا تَاْكُلُوْنَ ۝٩١**
p ترجمہ: تو وہ ان سے پیٹھ پھیر کر پلٹ گئے۔ پھر (ابراہیم علیہ السلام) چپکے سے ان کے معبودوں کی طرف گئے پھر انھوں نے فرمایا کیا تم کھاتے نہیں؟ || So they turned away from him and left. Then he went quietly to their gods and said, "Will you not eat? || نو هغوی شا ورته کړه او لاړل. بیا (ابراهیم) پټ د دوی معبودانو ته ورغی او ویې ویل: "ایا نه خورئ؟
p **مَا لَكُمْ لَا تَنْطِقُوْنَ ۝٩٢ فَرَاغَ عَلَيْهِمْ ضَرْبًا ۢ بِالْيَمِيْنِ ۝٩٣ فَاَقْبَلُوْٓا اِلَيْهِ يَزِفُّوْنَ ۝٩٤** || **مَا لَكُمْ لَا تَنْطِقُوْنَ ۝٩٢ فَرَاغَ عَلَيْهِمْ ضَرْبًا ۢ بِالْيَمِيْنِ ۝٩٣ فَاَقْبَلُوْٓا اِلَيْهِ يَزِفُّوْنَ ۝٩٤** || **مَا لَكُمْ لَا تَنْطِقُوْنَ ۝٩٢ فَرَاغَ عَلَيْهِمْ ضَرْبًا ۢ بِالْيَمِيْنِ ۝٩٣ فَاَقْبَلُوْٓا اِلَيْهِ يَزِفُّوْنَ ۝٩٤**
p ترجمہ: تمھیں کیا ہوا ہے کہ تم بولتے نہیں ہو؟ پھر ان کو دائیں ہاتھ سے مارنا (اور توڑنا) شروع کر دیا۔ پھر (قوم کے) لوگ ان کے پاس دوڑتے ہوئے آئے۔ || What is wrong with you that you do not speak?" Then he turned on them, striking (and breaking them) with his right hand. Then (the people) came rushing to him. || څه درته شوي چې خبرې نه کوئ؟" بیا یې په ښي لاس په وهلو (او ماتولو) پیل وکړ. بیا (د قوم) خلک ورته منډې وهونکي راغلل.
p **قَالَ اَتَعْبُدُوْنَ مَا تَنْحِتُوْنَ ۝٩٥ وَاللّٰهُ خَلَقَكُمْ وَمَا تَعْمَلُوْنَ ۝٩٦** || **قَالَ اَتَعْبُدُوْنَ مَا تَنْحِتُوْنَ ۝٩٥ وَاللّٰهُ خَلَقَكُمْ وَمَا تَعْمَلُوْنَ ۝٩٦** || **قَالَ اَتَعْبُدُوْنَ مَا تَنْحِتُوْنَ ۝٩٥ وَاللّٰهُ خَلَقَكُمْ وَمَا تَعْمَلُوْنَ ۝٩٦**
p ترجمہ: (ابراہیم علیہ السلام نے) فرمایا کیا تم ایسی چیزوں کو پوجتے ہو جنھیں تم خود تراشتے ہو؟ حالاں کہ اللہ نے تمھیں پیدا فرمایا اور ان (کاموں) کو (بھی) جو تم کرتے ہو۔ || (Ibrahim) said, "Do you worship what you yourselves carve, when Allah created you and what you make?" || (ابراهیم) وویل: "ایا د هغو شیانو عبادت کوئ چې پخپله یې تراشئ؟ حال دا چې الله تاسو او ستاسو کارونه (هم) پیدا کړي."
p **قَالُوا ابْنُوْا لَهٗ بُنْيَانًا فَاَلْقُوْهُ فِي الْجَحِيْمِ ۝٩٧** || **قَالُوا ابْنُوْا لَهٗ بُنْيَانًا فَاَلْقُوْهُ فِي الْجَحِيْمِ ۝٩٧** || **قَالُوا ابْنُوْا لَهٗ بُنْيَانًا فَاَلْقُوْهُ فِي الْجَحِيْمِ ۝٩٧**
p ترجمہ: (لوگوں نے) کہا اِن کے لیے ایک عمارت بناؤ پھر انھیں بھڑکتی آگ میں ڈال دو۔ || They said, "Build a structure for him and throw him into the blazing fire." || (خلکو) وویل: "د ده لپاره یوه ودانۍ جوړه کړئ او په بلېدونکي اور کې یې وغورځوئ."
p **فَاَرَادُوْا بِهٖ كَيْدًا فَجَعَلْنٰهُمُ الْاَسْفَلِيْنَ ۝٩٨** || **فَاَرَادُوْا بِهٖ كَيْدًا فَجَعَلْنٰهُمُ الْاَسْفَلِيْنَ ۝٩٨** || **فَاَرَادُوْا بِهٖ كَيْدًا فَجَعَلْنٰهُمُ الْاَسْفَلِيْنَ ۝٩٨**
p ترجمہ: لہٰذا انھوں نے ابراہیم (علیہ السلام) کے ساتھ ایک چال چلنے کا ارادہ کیا تو ہم نے اُن لوگوں کو سب سے نیچا کر دکھایا۔ || They intended a plot against him, but We made them the lowest. || نو هغوی د ابراهیم (علیه السلام) پر ضد د چل اراده وکړه، خو موږ هغوی تر ټولو ټیټ کړل.
p **وَقَالَ اِنِّيْ ذَاهِبٌ اِلٰى رَبِّيْ سَيَهْدِيْنِ ۝٩٩** || **وَقَالَ اِنِّيْ ذَاهِبٌ اِلٰى رَبِّيْ سَيَهْدِيْنِ ۝٩٩** || **وَقَالَ اِنِّيْ ذَاهِبٌ اِلٰى رَبِّيْ سَيَهْدِيْنِ ۝٩٩**
p ترجمہ: اور (ابراہیم علیہ السلام نے) فرمایا میں اپنے رب کی طرف جانے والا ہوں وہی میری راہ نمائی فرمائے گا۔ || And (Ibrahim) said, "I am going to my Lord; He will guide me. || او (ابراهیم) وویل: "زه خپل رب ته ځم، همغه به مې لارښوونه وکړي.
p **رَبِّ هَبْ لِيْ مِنَ الصّٰلِحِيْنَ ۝١٠٠ فَبَشَّرْنٰهُ بِغُلٰمٍ حَلِيْمٍ ۝١٠١** || **رَبِّ هَبْ لِيْ مِنَ الصّٰلِحِيْنَ ۝١٠٠ فَبَشَّرْنٰهُ بِغُلٰمٍ حَلِيْمٍ ۝١٠١** || **رَبِّ هَبْ لِيْ مِنَ الصّٰلِحِيْنَ ۝١٠٠ فَبَشَّرْنٰهُ بِغُلٰمٍ حَلِيْمٍ ۝١٠١**
p ترجمہ: (پھر انھوں نے عرض کیا) اے میرے رب! مجھے صالحین میں سے (بیٹا) عطا فرما۔ تو ہم نے انھیں ایک تحمل والے لڑکے (اسماعیل علیہ السلام) کی خوش خبری دی۔ || My Lord! Grant me (a son) from among the righteous." So We gave him the good news of a forbearing boy (Isma'il). || ای زما ربه! له صالحانو ماته (زوی) راکړه." نو موږ ورته د یوه زغمناک هلک (اسماعیل علیه السلام) زېری ورکړ.
p **فَلَمَّا بَلَغَ مَعَهُ السَّعْيَ قَالَ يٰبُنَيَّ اِنِّيْٓ اَرٰى فِي الْمَنَامِ** || **فَلَمَّا بَلَغَ مَعَهُ السَّعْيَ قَالَ يٰبُنَيَّ اِنِّيْٓ اَرٰى فِي الْمَنَامِ** || **فَلَمَّا بَلَغَ مَعَهُ السَّعْيَ قَالَ يٰبُنَيَّ اِنِّيْٓ اَرٰى فِي الْمَنَامِ**
