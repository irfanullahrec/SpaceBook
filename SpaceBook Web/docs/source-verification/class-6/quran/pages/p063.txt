# سبق نمبر 8: سُوْرَةُ الشُّعَرَآءِ || Lesson 8: Surah Ash-Shu'ara || اتم لوست: سورة الشعراء
p آیات: 69 تا 159 || Verses 69 to 159 || آیتونه: 69 تر 159
p **وَاتْلُ عَلَيْهِمْ نَبَاَ اِبْرٰهِيْمَ ۝٦٩ اِذْ قَالَ لِاَبِيْهِ وَقَوْمِهٖ مَا تَعْبُدُوْنَ ۝٧٠** || **وَاتْلُ عَلَيْهِمْ نَبَاَ اِبْرٰهِيْمَ ۝٦٩ اِذْ قَالَ لِاَبِيْهِ وَقَوْمِهٖ مَا تَعْبُدُوْنَ ۝٧٠** || **وَاتْلُ عَلَيْهِمْ نَبَاَ اِبْرٰهِيْمَ ۝٦٩ اِذْ قَالَ لِاَبِيْهِ وَقَوْمِهٖ مَا تَعْبُدُوْنَ ۝٧٠**
p ترجمہ: اور آپ (ﷺ) ان پر تلاوت کیجیے ابراہیم (علیہ السلام) کا قصہ۔ جب انھوں نے اپنے باپ اور اپنی قوم سے فرمایا تم کس چیز کی عبادت کرتے ہو؟ || And recite to them (ﷺ) the story of Ibrahim (peace be upon him), when he said to his father and his people, "What do you worship?" || او ته (ﷺ) پرې د ابراهیم (علیه السلام) کیسه ولوله، کله چې خپل پلار او قوم ته یې وویل: "د څه عبادت کوئ؟"
p **قَالُوْا نَعْبُدُ اَصْنَامًا فَنَظَلُّ لَهَا عٰكِفِيْنَ ۝٧١ قَالَ** || **قَالُوْا نَعْبُدُ اَصْنَامًا فَنَظَلُّ لَهَا عٰكِفِيْنَ ۝٧١ قَالَ** || **قَالُوْا نَعْبُدُ اَصْنَامًا فَنَظَلُّ لَهَا عٰكِفِيْنَ ۝٧١ قَالَ**
p ترجمہ: وہ کہنے لگے کہ ہم بتوں کی پوجا کرتے ہیں پھر ہم تو انھی کی پوجا پر جمے بیٹھے رہتے ہیں۔ (ابراہیم علیہ السلام نے) فرمایا || They said, "We worship idols and remain devoted to them." (Ibrahim) said, || ویې ویل: "موږ د بتانو عبادت کوو او د هغو په عبادت ټینګ ناست یو." (ابراهیم) وویل:
p **هَلْ يَسْمَعُوْنَكُمْ اِذْ تَدْعُوْنَ ۝٧٢ اَوْ يَنْفَعُوْنَكُمْ اَوْ يَضُرُّوْنَ ۝٧٣ قَالُوْا** || **هَلْ يَسْمَعُوْنَكُمْ اِذْ تَدْعُوْنَ ۝٧٢ اَوْ يَنْفَعُوْنَكُمْ اَوْ يَضُرُّوْنَ ۝٧٣ قَالُوْا** || **هَلْ يَسْمَعُوْنَكُمْ اِذْ تَدْعُوْنَ ۝٧٢ اَوْ يَنْفَعُوْنَكُمْ اَوْ يَضُرُّوْنَ ۝٧٣ قَالُوْا**
p ترجمہ: کیا وہ تمھاری (بات) سنتے ہیں جب تم (انھیں) پکارتے ہو؟ یا وہ تمھیں نفع یا نقصان پہنچاتے ہیں؟ وہ کہنے لگے || "Do they hear you when you call (on them)? Or do they benefit or harm you?" They said, || "ایا ستاسو (خبره) اوري کله چې یې بولئ؟ یا تاسو ته ګټه یا زیان رسوي؟" ویې ویل:
p **بَلْ وَجَدْنَآ اٰبَآءَنَا كَذٰلِكَ يَفْعَلُوْنَ ۝٧٤ قَالَ اَفَرَءَيْتُمْ مَّا كُنْتُمْ تَعْبُدُوْنَ ۝٧٥** || **بَلْ وَجَدْنَآ اٰبَآءَنَا كَذٰلِكَ يَفْعَلُوْنَ ۝٧٤ قَالَ اَفَرَءَيْتُمْ مَّا كُنْتُمْ تَعْبُدُوْنَ ۝٧٥** || **بَلْ وَجَدْنَآ اٰبَآءَنَا كَذٰلِكَ يَفْعَلُوْنَ ۝٧٤ قَالَ اَفَرَءَيْتُمْ مَّا كُنْتُمْ تَعْبُدُوْنَ ۝٧٥**
p ترجمہ: بلکہ ہم نے اپنے باپ دادا کو پایا تھا کہ وہ ایسا ہی کرتے تھے۔ (ابراہیم علیہ السلام نے) فرمایا کیا تم نے دیکھا (غور کیا) جن کی تم پوجا کرتے ہو؟ || "Rather, we found our forefathers doing so." (Ibrahim) said, "Have you considered what you have been worshipping — || "بلکې خپل پلرونه مو همداسې کول موندلي." (ابراهیم) وویل: "ایا مو ولیدل (فکر مو وکړ) چې د چا عبادت کوئ —
p **اَنْتُمْ وَاٰبَآؤُكُمُ الْاَقْدَمُوْنَ ۝٧٦ فَاِنَّهُمْ عَدُوٌّ لِّيْٓ اِلَّا رَبَّ الْعٰلَمِيْنَ ۝٧٧ الَّذِيْ خَلَقَنِيْ** || **اَنْتُمْ وَاٰبَآؤُكُمُ الْاَقْدَمُوْنَ ۝٧٦ فَاِنَّهُمْ عَدُوٌّ لِّيْٓ اِلَّا رَبَّ الْعٰلَمِيْنَ ۝٧٧ الَّذِيْ خَلَقَنِيْ** || **اَنْتُمْ وَاٰبَآؤُكُمُ الْاَقْدَمُوْنَ ۝٧٦ فَاِنَّهُمْ عَدُوٌّ لِّيْٓ اِلَّا رَبَّ الْعٰلَمِيْنَ ۝٧٧ الَّذِيْ خَلَقَنِيْ**
p ترجمہ: تم بھی اور تمھارے پہلے باپ دادا بھی۔ تو بے شک وہ میرے دشمن ہیں سوائے رب العالمین کے۔ وہ جس نے مجھے پیدا فرمایا || you and your ancient forefathers? They are enemies to me, except the Lord of all the worlds, who created me || تاسو او ستاسو پخواني پلرونه؟ بېشکه هغه زما دښمنان دي پرته له رب العالمین، هغه چې زه یې پیدا کړم
p **فَهُوَ يَهْدِيْنِ ۝٧٨ وَالَّذِيْ هُوَ يُطْعِمُنِيْ وَيَسْقِيْنِ ۝٧٩ وَاِذَا مَرِضْتُ فَهُوَ يَشْفِيْنِ ۝٨٠** || **فَهُوَ يَهْدِيْنِ ۝٧٨ وَالَّذِيْ هُوَ يُطْعِمُنِيْ وَيَسْقِيْنِ ۝٧٩ وَاِذَا مَرِضْتُ فَهُوَ يَشْفِيْنِ ۝٨٠** || **فَهُوَ يَهْدِيْنِ ۝٧٨ وَالَّذِيْ هُوَ يُطْعِمُنِيْ وَيَسْقِيْنِ ۝٧٩ وَاِذَا مَرِضْتُ فَهُوَ يَشْفِيْنِ ۝٨٠**
p ترجمہ: تو وہی میری راہ نمائی فرماتا ہے۔ اور وہی مجھے کھلاتا اور پلاتا ہے۔ اور جب میں بیمار ہو جاتا ہوں تو وہی مجھے شفا دیتا ہے۔ || and He guides me; who feeds me and gives me drink; and when I fall ill, He heals me; || او همغه مې لارښوونه کوي؛ او همغه مې خوړوي او څښوي؛ او کله چې ناروغ شم، همغه شفا راکوي؛
p **وَالَّذِيْ يُمِيْتُنِيْ ثُمَّ يُحْيِيْنِ ۝٨١ وَالَّذِيْٓ اَطْمَعُ اَنْ يَّغْفِرَ لِيْ خَطِيْٓــَٔتِيْ يَوْمَ الدِّيْنِ ۝٨٢** || **وَالَّذِيْ يُمِيْتُنِيْ ثُمَّ يُحْيِيْنِ ۝٨١ وَالَّذِيْٓ اَطْمَعُ اَنْ يَّغْفِرَ لِيْ خَطِيْٓــَٔتِيْ يَوْمَ الدِّيْنِ ۝٨٢** || **وَالَّذِيْ يُمِيْتُنِيْ ثُمَّ يُحْيِيْنِ ۝٨١ وَالَّذِيْٓ اَطْمَعُ اَنْ يَّغْفِرَ لِيْ خَطِيْٓــَٔتِيْ يَوْمَ الدِّيْنِ ۝٨٢**
p ترجمہ: اور جو مجھے موت دے گا پھر مجھے زندہ فرمائے گا۔ اور وہ جس سے میں امید رکھتا ہوں کہ وہ میری خطائیں بخش دے گا بدلے کے دن۔ || who will cause me to die and then give me life; and who, I hope, will forgive my faults on the Day of Recompense. || او همغه چې مړ به مې کړي بیا به مې ژوندی کړي؛ او همغه چې هیله ترې لرم چې د بدلې په ورځ به مې خطاوې وبښي.
p **رَبِّ هَبْ لِيْ حُكْمًا وَّاَلْحِقْنِيْ بِالصّٰلِحِيْنَ ۝٨٣ وَاجْعَلْ لِّيْ لِسَانَ صِدْقٍ فِي الْاٰخِرِيْنَ ۝٨٤** || **رَبِّ هَبْ لِيْ حُكْمًا وَّاَلْحِقْنِيْ بِالصّٰلِحِيْنَ ۝٨٣ وَاجْعَلْ لِّيْ لِسَانَ صِدْقٍ فِي الْاٰخِرِيْنَ ۝٨٤** || **رَبِّ هَبْ لِيْ حُكْمًا وَّاَلْحِقْنِيْ بِالصّٰلِحِيْنَ ۝٨٣ وَاجْعَلْ لِّيْ لِسَانَ صِدْقٍ فِي الْاٰخِرِيْنَ ۝٨٤**
p ترجمہ: اے میرے رب! مجھے حکمت عطا فرما اور مجھے صالحین میں شامل فرما۔ اور میرا ذکرِ خیر بعد میں آنے والوں میں (جاری) فرما۔ || My Lord! Grant me wisdom and join me with the righteous, and grant me an honourable mention among later generations. || ای زما ربه! حکمت راکړه او له صالحانو سره مې یوځای کړه؛ او په راتلونکو نسلونو کې مې ښه یاد (جاري) کړه.
p **وَاجْعَلْنِيْ مِنْ وَّرَثَةِ جَنَّةِ النَّعِيْمِ ۝٨٥ وَاغْفِرْ لِاَبِيْٓ اِنَّهٗ كَانَ مِنَ الضَّآلِّيْنَ ۝٨٦** || **وَاجْعَلْنِيْ مِنْ وَّرَثَةِ جَنَّةِ النَّعِيْمِ ۝٨٥ وَاغْفِرْ لِاَبِيْٓ اِنَّهٗ كَانَ مِنَ الضَّآلِّيْنَ ۝٨٦** || **وَاجْعَلْنِيْ مِنْ وَّرَثَةِ جَنَّةِ النَّعِيْمِ ۝٨٥ وَاغْفِرْ لِاَبِيْٓ اِنَّهٗ كَانَ مِنَ الضَّآلِّيْنَ ۝٨٦**
p ترجمہ: اور مجھے بنا دے نعمتوں والی جنت کے وارثوں میں سے۔ اور میرے باپ کو بخش دے بے شک وہ گم راہوں میں سے ہے۔ || Make me one of the inheritors of the Garden of Bliss, and forgive my father; he was among the misguided. || او د نعمتونو د جنت له وارثانو مې کړه؛ او زما پلار وبښه، بېشکه هغه له ګمراهانو دی.
p **وَلَا تُخْزِنِيْ يَوْمَ يُبْعَثُوْنَ ۝٨٧ يَوْمَ لَا يَنْفَعُ مَالٌ وَّلَا بَنُوْنَ ۝٨٨** || **وَلَا تُخْزِنِيْ يَوْمَ يُبْعَثُوْنَ ۝٨٧ يَوْمَ لَا يَنْفَعُ مَالٌ وَّلَا بَنُوْنَ ۝٨٨** || **وَلَا تُخْزِنِيْ يَوْمَ يُبْعَثُوْنَ ۝٨٧ يَوْمَ لَا يَنْفَعُ مَالٌ وَّلَا بَنُوْنَ ۝٨٨**
p ترجمہ: اور مجھے رسوا نہ کرنا جس دن لوگ دوبارہ اٹھائے جائیں گے۔ جس دن نہ مال فائدہ دے گا اور نہ بیٹے (اولاد)۔ || And do not disgrace me on the Day they are raised — the Day when neither wealth nor sons will help, || او په هغه ورځ مې مه رسوا کوه چې خلک بیا راپورته کېږي؛ په هغه ورځ چې نه مال ګټه ورکوي او نه زامن،
p **اِلَّا مَنْ اَتَى اللّٰهَ بِقَلْبٍ سَلِيْمٍ ۝٨٩ وَاُزْلِفَتِ الْجَنَّةُ لِلْمُتَّقِيْنَ ۝٩٠** || **اِلَّا مَنْ اَتَى اللّٰهَ بِقَلْبٍ سَلِيْمٍ ۝٨٩ وَاُزْلِفَتِ الْجَنَّةُ لِلْمُتَّقِيْنَ ۝٩٠** || **اِلَّا مَنْ اَتَى اللّٰهَ بِقَلْبٍ سَلِيْمٍ ۝٨٩ وَاُزْلِفَتِ الْجَنَّةُ لِلْمُتَّقِيْنَ ۝٩٠**
p ترجمہ: مگر جو اللہ کے پاس آیا سلامتی والے (پاک) دل کے ساتھ (وہی کامیاب ہو گا)۔ اور جنت قریب کر دی جائے گی پرہیزگاروں کے لیے۔ || except one who comes to Allah with a sound (pure) heart." And Paradise will be brought near to the righteous, || مګر هغه چې الله ته په سالم (پاک) زړه ورشي (همغه بریالی دی)." او جنت به پرهېزګارانو ته نږدې کړل شي،
