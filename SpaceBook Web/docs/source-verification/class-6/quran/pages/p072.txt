p **وَاِنَّهٗ لَفِيْ زُبُرِ الْاَوَّلِيْنَ ۝١٩٦ اَوَلَمْ يَكُنْ لَّهُمْ اٰيَةً** || **وَاِنَّهٗ لَفِيْ زُبُرِ الْاَوَّلِيْنَ ۝١٩٦ اَوَلَمْ يَكُنْ لَّهُمْ اٰيَةً** || **وَاِنَّهٗ لَفِيْ زُبُرِ الْاَوَّلِيْنَ ۝١٩٦ اَوَلَمْ يَكُنْ لَّهُمْ اٰيَةً**
p ترجمہ: اور بے شک اس (قرآن) کا ذکر پہلی کتابوں میں بھی موجود ہے۔ کیا ان کے لیے یہ نشانی کافی نہیں || And it is (mentioned) in the scriptures of earlier peoples. Is it not a sufficient sign for them || او بېشکه د دې (قرآن) یادونه په پخوانیو کتابونو کې هم شته. ایا دا نښه ورته بس نه ده
p **اَنْ يَّعْلَمَهٗ عُلَمٰٓؤُا بَنِيْٓ اِسْرَآءِيْلَ ۝١٩٧ وَلَوْ نَزَّلْنٰهُ عَلٰى بَعْضِ الْاَعْجَمِيْنَ ۝١٩٨** || **اَنْ يَّعْلَمَهٗ عُلَمٰٓؤُا بَنِيْٓ اِسْرَآءِيْلَ ۝١٩٧ وَلَوْ نَزَّلْنٰهُ عَلٰى بَعْضِ الْاَعْجَمِيْنَ ۝١٩٨** || **اَنْ يَّعْلَمَهٗ عُلَمٰٓؤُا بَنِيْٓ اِسْرَآءِيْلَ ۝١٩٧ وَلَوْ نَزَّلْنٰهُ عَلٰى بَعْضِ الْاَعْجَمِيْنَ ۝١٩٨**
p ترجمہ: کہ اسے بنی اسرائیل کے علماء بھی جانتے ہیں؟ اور اگر ہم اس (قرآن) کو نازل فرماتے کسی عجمی (غیر عربی) پر || that the scholars of the Children of Israel know it? And had We sent it down to a non-Arab, || چې د بني اسرائیلو علماء یې هم پېژني؟ او که موږ دا (قرآن) پر یو عجمي (غیر عرب) نازل کړی وای
p **فَقَرَاَهٗ عَلَيْهِمْ مَّا كَانُوْا بِهٖ مُؤْمِنِيْنَ ۝١٩٩ كَذٰلِكَ سَلَكْنٰهُ فِيْ قُلُوْبِ الْمُجْرِمِيْنَ ۝٢٠٠** || **فَقَرَاَهٗ عَلَيْهِمْ مَّا كَانُوْا بِهٖ مُؤْمِنِيْنَ ۝١٩٩ كَذٰلِكَ سَلَكْنٰهُ فِيْ قُلُوْبِ الْمُجْرِمِيْنَ ۝٢٠٠** || **فَقَرَاَهٗ عَلَيْهِمْ مَّا كَانُوْا بِهٖ مُؤْمِنِيْنَ ۝١٩٩ كَذٰلِكَ سَلَكْنٰهُ فِيْ قُلُوْبِ الْمُجْرِمِيْنَ ۝٢٠٠**
p ترجمہ: پھر وہ اسے ان کے سامنے پڑھتا تو بھی وہ ایمان نہ لاتے۔ اس طرح ہم نے اس (کے انکار) کو مجرموں کے دلوں میں داخل کر دیا ہے۔ || and he had recited it to them, they still would not have believed in it. Thus have We inserted (denial of) it into the hearts of the criminals: || او هغه ورته لوستی وای، بیا به هم ایمان نه وای راوړی. همداسې موږ د دې (انکار) د مجرمانو په زړونو کې ننه ایستلی.
p **لَا يُؤْمِنُوْنَ بِهٖ حَتّٰى يَرَوُا الْعَذَابَ الْاَلِيْمَ ۝٢٠١ فَيَاْتِيَهُمْ بَغْتَةً وَّهُمْ لَا يَشْعُرُوْنَ ۝٢٠٢ فَيَقُوْلُوْا** || **لَا يُؤْمِنُوْنَ بِهٖ حَتّٰى يَرَوُا الْعَذَابَ الْاَلِيْمَ ۝٢٠١ فَيَاْتِيَهُمْ بَغْتَةً وَّهُمْ لَا يَشْعُرُوْنَ ۝٢٠٢ فَيَقُوْلُوْا** || **لَا يُؤْمِنُوْنَ بِهٖ حَتّٰى يَرَوُا الْعَذَابَ الْاَلِيْمَ ۝٢٠١ فَيَاْتِيَهُمْ بَغْتَةً وَّهُمْ لَا يَشْعُرُوْنَ ۝٢٠٢ فَيَقُوْلُوْا**
p ترجمہ: وہ اس پر ایمان نہیں لائیں گے یہاں تک کہ دردناک عذاب دیکھ لیں۔ پھر وہ ان پر اچانک آ جائے گا اور انھیں خبر بھی نہیں ہو گی۔ تو وہ کہیں گے || they will not believe in it until they see the painful punishment, which will come upon them suddenly while they do not perceive. Then they will say, || ایمان به پرې رانه وړي تر دې چې دردناک عذاب ووینې. بیا به ناڅاپه پرې راشي او خبر به هم نه وي. نو وايي به:
p **هَلْ نَحْنُ مُنْظَرُوْنَ ۝٢٠٣ اَفَبِعَذَابِنَا يَسْتَعْجِلُوْنَ ۝٢٠٤ اَفَرَءَيْتَ اِنْ مَّتَّعْنٰهُمْ سِنِيْنَ ۝٢٠٥** || **هَلْ نَحْنُ مُنْظَرُوْنَ ۝٢٠٣ اَفَبِعَذَابِنَا يَسْتَعْجِلُوْنَ ۝٢٠٤ اَفَرَءَيْتَ اِنْ مَّتَّعْنٰهُمْ سِنِيْنَ ۝٢٠٥** || **هَلْ نَحْنُ مُنْظَرُوْنَ ۝٢٠٣ اَفَبِعَذَابِنَا يَسْتَعْجِلُوْنَ ۝٢٠٤ اَفَرَءَيْتَ اِنْ مَّتَّعْنٰهُمْ سِنِيْنَ ۝٢٠٥**
p ترجمہ: کیا ہمیں مہلت مل سکتی ہے؟ تو کیا وہ ہمارے عذاب کی جلدی کرتے ہیں؟ تو بھلا آپ دیکھیے اگر ہم انھیں کئی سال فائدہ دیں۔ || "Can we be given respite?" Do they seek to hasten Our punishment? Have you considered: if We let them enjoy (life) for years, || "ایا موږ ته مهلت راکول کېږي؟" ایا زموږ عذاب ته بېړه کوي؟ نو ووینه که موږ څو کاله ګټه ورکړو
p **ثُمَّ جَآءَهُمْ مَّا كَانُوْا يُوْعَدُوْنَ ۝٢٠٦ مَآ اَغْنٰى عَنْهُمْ مَّا كَانُوْا يُمَتَّعُوْنَ ۝٢٠٧** || **ثُمَّ جَآءَهُمْ مَّا كَانُوْا يُوْعَدُوْنَ ۝٢٠٦ مَآ اَغْنٰى عَنْهُمْ مَّا كَانُوْا يُمَتَّعُوْنَ ۝٢٠٧** || **ثُمَّ جَآءَهُمْ مَّا كَانُوْا يُوْعَدُوْنَ ۝٢٠٦ مَآ اَغْنٰى عَنْهُمْ مَّا كَانُوْا يُمَتَّعُوْنَ ۝٢٠٧**
p ترجمہ: پھر اُن پر وہ (عذاب) آ جائے جس کا ان سے وعدہ کیا جاتا ہے۔ تو وہ (سامانِ زندگی) ان کے کسی کام نہ آئے گا جو انھیں دیا گیا تھا۔ || then there comes to them what they were promised, what they had been given to enjoy will not avail them. || بیا هغه (عذاب) ورشي چې ژمنه ورسره شوې، نو هغه (د ژوند سامان) چې ورکړل شوی و، هېڅ ګټه به ورته ونه رسوي.
p **وَمَآ اَهْلَكْنَا مِنْ قَرْيَةٍ اِلَّا لَهَا مُنْذِرُوْنَ ۝٢٠٨ ڰ ذِكْرٰى ۛ وَمَا كُنَّا ظٰلِمِيْنَ ۝٢٠٩** || **وَمَآ اَهْلَكْنَا مِنْ قَرْيَةٍ اِلَّا لَهَا مُنْذِرُوْنَ ۝٢٠٨ ڰ ذِكْرٰى ۛ وَمَا كُنَّا ظٰلِمِيْنَ ۝٢٠٩** || **وَمَآ اَهْلَكْنَا مِنْ قَرْيَةٍ اِلَّا لَهَا مُنْذِرُوْنَ ۝٢٠٨ ڰ ذِكْرٰى ۛ وَمَا كُنَّا ظٰلِمِيْنَ ۝٢٠٩**
p ترجمہ: اور ہم نے کسی ایسی بستی (کے لوگوں) کو ہلاک نہیں کیا جسے ڈرانے والے نہ (آئے) ہوں۔ (انھیں) نصیحت کرنے کے لیے اور ہم ظالم نہیں ہیں۔ || We never destroyed a town without its having warners, as a reminder; and We were never unjust. || موږ داسې هېڅ کلی نه دی هلاک کړی چې وېروونکي ورته نه وي راغلي؛ د نصیحت لپاره؛ او موږ ظالمان نه یو.
p **وَمَا تَنَزَّلَتْ بِهِ الشَّيٰطِيْنُ ۝٢١٠ وَمَا يَنْۢبَغِيْ لَهُمْ وَمَا يَسْتَطِيْعُوْنَ ۝٢١١ اِنَّهُمْ عَنِ السَّمْعِ لَمَعْزُوْلُوْنَ ۝٢١٢** || **وَمَا تَنَزَّلَتْ بِهِ الشَّيٰطِيْنُ ۝٢١٠ وَمَا يَنْۢبَغِيْ لَهُمْ وَمَا يَسْتَطِيْعُوْنَ ۝٢١١ اِنَّهُمْ عَنِ السَّمْعِ لَمَعْزُوْلُوْنَ ۝٢١٢** || **وَمَا تَنَزَّلَتْ بِهِ الشَّيٰطِيْنُ ۝٢١٠ وَمَا يَنْۢبَغِيْ لَهُمْ وَمَا يَسْتَطِيْعُوْنَ ۝٢١١ اِنَّهُمْ عَنِ السَّمْعِ لَمَعْزُوْلُوْنَ ۝٢١٢**
p ترجمہ: اور اس (قرآن) کو شیاطین لے کر نازل نہیں ہوئے۔ اور نہ وہ اس قابل ہیں اور نہ وہ ایسا کر سکتے ہیں۔ یقیناً وہ تو سننے کی جگہ سے بھی دور کر دیے گئے ہیں۔ || The devils did not bring it (the Quran) down; it is not fitting for them, nor are they able. They are kept far from even hearing (it). || او دا (قرآن) شیطانانو نه دی راوړی؛ نه یې وړ دي او نه یې کولای شي. بېشکه هغوی د اورېدو له ځایه هم لرې کړل شوي.
p **فَلَا تَدْعُ مَعَ اللّٰهِ اِلٰهًا اٰخَرَ فَتَكُوْنَ مِنَ الْمُعَذَّبِيْنَ ۝٢١٣ وَاَنْذِرْ عَشِيْرَتَكَ الْاَقْرَبِيْنَ ۝٢١٤** || **فَلَا تَدْعُ مَعَ اللّٰهِ اِلٰهًا اٰخَرَ فَتَكُوْنَ مِنَ الْمُعَذَّبِيْنَ ۝٢١٣ وَاَنْذِرْ عَشِيْرَتَكَ الْاَقْرَبِيْنَ ۝٢١٤** || **فَلَا تَدْعُ مَعَ اللّٰهِ اِلٰهًا اٰخَرَ فَتَكُوْنَ مِنَ الْمُعَذَّبِيْنَ ۝٢١٣ وَاَنْذِرْ عَشِيْرَتَكَ الْاَقْرَبِيْنَ ۝٢١٤**
p ترجمہ: تو تم اللہ کے ساتھ کسی اور معبود کو مت پکارو ورنہ تم بھی عذاب پانے والوں میں سے ہو جاؤ گے۔ اور اپنے قریبی رشتہ داروں کو ڈرایے۔ || So do not call upon another god besides Allah, or you will be among the punished. And warn your nearest relatives. || نو له الله سره بل معبود مه بوله، ګنې ته به هم له عذاب شویو شې. او خپل نږدې خپلوان وډاروه.
p **وَاخْفِضْ جَنَاحَكَ لِمَنِ اتَّبَعَكَ مِنَ الْمُؤْمِنِيْنَ ۝٢١٥ فَاِنْ عَصَوْكَ فَقُلْ** || **وَاخْفِضْ جَنَاحَكَ لِمَنِ اتَّبَعَكَ مِنَ الْمُؤْمِنِيْنَ ۝٢١٥ فَاِنْ عَصَوْكَ فَقُلْ** || **وَاخْفِضْ جَنَاحَكَ لِمَنِ اتَّبَعَكَ مِنَ الْمُؤْمِنِيْنَ ۝٢١٥ فَاِنْ عَصَوْكَ فَقُلْ**
p ترجمہ: اور اپنا (رحمت کا) بازو جھکائیے ان کے لیے جو ایمان والوں میں سے آپ کی پیروی کریں۔ پھر اگر وہ آپ کی نافرمانی کریں تو آپ (ﷺ) فرما دیجیے || And lower your wing (in mercy) to the believers who follow you. But if they disobey you, say, || او د (رحمت) وزر دې د هغو مومنانو لپاره ټیټ کړه چې ستا پیروي کوي. که نافرماني دې وکړي، ووایه (ﷺ):
p **اِنِّيْ بَرِيْٓءٌ مِّمَّا تَعْمَلُوْنَ ۝٢١٦ وَتَوَكَّلْ عَلَى الْعَزِيْزِ الرَّحِيْمِ ۝٢١٧** || **اِنِّيْ بَرِيْٓءٌ مِّمَّا تَعْمَلُوْنَ ۝٢١٦ وَتَوَكَّلْ عَلَى الْعَزِيْزِ الرَّحِيْمِ ۝٢١٧** || **اِنِّيْ بَرِيْٓءٌ مِّمَّا تَعْمَلُوْنَ ۝٢١٦ وَتَوَكَّلْ عَلَى الْعَزِيْزِ الرَّحِيْمِ ۝٢١٧**
p ترجمہ: کہ بے شک میں بیزار ہوں اس سے جو تم کرتے ہو۔ اور آپ (ﷺ) بھروسہ کیجیے اس پر جو بہت غالب نہایت رحم فرمانے والا ہے۔ || "I am free of what you do." And put your trust (ﷺ) in the Almighty, the Most Merciful, || "بېشکه زه له هغه بېزار یم چې تاسو یې کوئ." او (ﷺ) پر هغه توکل وکړه چې ډېر غالب، ډېر مهربان دی،
p **الَّذِيْ يَرٰىكَ حِيْنَ تَقُوْمُ ۝٢١٨** || **الَّذِيْ يَرٰىكَ حِيْنَ تَقُوْمُ ۝٢١٨** || **الَّذِيْ يَرٰىكَ حِيْنَ تَقُوْمُ ۝٢١٨**
