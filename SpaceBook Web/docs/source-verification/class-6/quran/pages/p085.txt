p **وَّهُمْ شٰهِدُوْنَ ۝١٥٠ اَلَآ اِنَّهُمْ مِّنْ اِفْكِهِمْ لَيَقُوْلُوْنَ ۝١٥١ وَلَدَ اللّٰهُ ۙ وَاِنَّهُمْ لَكٰذِبُوْنَ ۝١٥٢** || **وَّهُمْ شٰهِدُوْنَ ۝١٥٠ اَلَآ اِنَّهُمْ مِّنْ اِفْكِهِمْ لَيَقُوْلُوْنَ ۝١٥١ وَلَدَ اللّٰهُ ۙ وَاِنَّهُمْ لَكٰذِبُوْنَ ۝١٥٢** || **وَّهُمْ شٰهِدُوْنَ ۝١٥٠ اَلَآ اِنَّهُمْ مِّنْ اِفْكِهِمْ لَيَقُوْلُوْنَ ۝١٥١ وَلَدَ اللّٰهُ ۙ وَاِنَّهُمْ لَكٰذِبُوْنَ ۝١٥٢**
p ترجمہ: اور وہ (اس وقت) موجود تھے؟ آگاہ ہو جاؤ بے شک یہ ضرور اپنی جھوٹ بنائی ہوئی (بات) کہتے ہیں۔ کہ اللہ کی اولاد ہے اور بے شک وہ ضرور جھوٹے ہیں۔ || while they were witnesses? Truly, it is out of their own invention that they say, "Allah has begotten (children)," and they are surely liars. || او دوی (هغه وخت) حاضر وو؟ خبر شئ! بېشکه دوی خامخا خپله جوړه کړې (خبره) کوي چې د الله اولاد دی، او بېشکه دوی دروغجن دي.
p **اَصْطَفَى الْبَنَاتِ عَلَى الْبَنِيْنَ ۝١٥٣ مَا لَكُمْ ۫ كَيْفَ تَحْكُمُوْنَ ۝١٥٤ اَفَلَا تَذَكَّرُوْنَ ۝١٥٥** || **اَصْطَفَى الْبَنَاتِ عَلَى الْبَنِيْنَ ۝١٥٣ مَا لَكُمْ ۫ كَيْفَ تَحْكُمُوْنَ ۝١٥٤ اَفَلَا تَذَكَّرُوْنَ ۝١٥٥** || **اَصْطَفَى الْبَنَاتِ عَلَى الْبَنِيْنَ ۝١٥٣ مَا لَكُمْ ۫ كَيْفَ تَحْكُمُوْنَ ۝١٥٤ اَفَلَا تَذَكَّرُوْنَ ۝١٥٥**
p ترجمہ: کیا اس (اللہ) نے بیٹوں کے بجائے بیٹیوں کو پسند فرمایا؟ تمھیں کیا ہوا ہے؟ تم کیسا فیصلہ کرتے ہو۔ تو کیا تم نصیحت حاصل نہیں کرتے؟ || Has He chosen daughters over sons? What is wrong with you? How do you judge? Will you not take heed? || ایا هغه (الله) د زامنو پر ځای لوڼې غوره کړې؟ څه درته شوي؟ څنګه فیصله کوئ؟ ایا نصیحت نه اخلئ؟
p **اَمْ لَكُمْ سُلْطٰنٌ مُّبِيْنٌ ۝١٥٦ فَاْتُوْا بِكِتٰبِكُمْ اِنْ كُنْتُمْ صٰدِقِيْنَ ۝١٥٧ وَجَعَلُوْا بَيْنَهٗ وَبَيْنَ الْجِنَّةِ نَسَبًا ؕ** || **اَمْ لَكُمْ سُلْطٰنٌ مُّبِيْنٌ ۝١٥٦ فَاْتُوْا بِكِتٰبِكُمْ اِنْ كُنْتُمْ صٰدِقِيْنَ ۝١٥٧ وَجَعَلُوْا بَيْنَهٗ وَبَيْنَ الْجِنَّةِ نَسَبًا ؕ** || **اَمْ لَكُمْ سُلْطٰنٌ مُّبِيْنٌ ۝١٥٦ فَاْتُوْا بِكِتٰبِكُمْ اِنْ كُنْتُمْ صٰدِقِيْنَ ۝١٥٧ وَجَعَلُوْا بَيْنَهٗ وَبَيْنَ الْجِنَّةِ نَسَبًا ؕ**
p ترجمہ: یا تمھارے پاس کوئی واضح دلیل ہے؟ تو تم اپنی کتاب پیش کرو اگر تم سچے ہو۔ اور انھوں نے اس (اللہ) کے اور جنات کے درمیان رشتہ قائم کر دیا || Or do you have a clear authority? Then bring your book, if you are truthful. And they have set up a kinship between Him and the jinn, || یا کوم ښکاره دلیل لرئ؟ نو خپل کتاب راوړئ که رښتیني یئ. او د هغه (الله) او پېریانو تر منځ یې نسب جوړ کړ
p **وَلَقَدْ عَلِمَتِ الْجِنَّةُ اِنَّهُمْ لَمُحْضَرُوْنَ ۝١٥٨ سُبْحٰنَ اللّٰهِ عَمَّا يَصِفُوْنَ ۝١٥٩** || **وَلَقَدْ عَلِمَتِ الْجِنَّةُ اِنَّهُمْ لَمُحْضَرُوْنَ ۝١٥٨ سُبْحٰنَ اللّٰهِ عَمَّا يَصِفُوْنَ ۝١٥٩** || **وَلَقَدْ عَلِمَتِ الْجِنَّةُ اِنَّهُمْ لَمُحْضَرُوْنَ ۝١٥٨ سُبْحٰنَ اللّٰهِ عَمَّا يَصِفُوْنَ ۝١٥٩**
p ترجمہ: حالاں کہ جنات کو معلوم ہے کہ بے شک وہ (اللہ کے سامنے) ضرور حاضر کیے جائیں گے۔ اللہ ان باتوں سے پاک ہے جو یہ بیان کرتے ہیں۔ || though the jinn know that they will surely be brought (before Him). Glory be to Allah above what they describe! || حال دا چې پېریان پوهېږي چې دوی به خامخا (د الله پر وړاندې) حاضر کړل شي. الله له هغو خبرو پاک دی چې دوی یې بیانوي.
p **اِلَّا عِبَادَ اللّٰهِ الْمُخْلَصِيْنَ ۝١٦٠ فَاِنَّكُمْ وَمَا تَعْبُدُوْنَ ۝١٦١** || **اِلَّا عِبَادَ اللّٰهِ الْمُخْلَصِيْنَ ۝١٦٠ فَاِنَّكُمْ وَمَا تَعْبُدُوْنَ ۝١٦١** || **اِلَّا عِبَادَ اللّٰهِ الْمُخْلَصِيْنَ ۝١٦٠ فَاِنَّكُمْ وَمَا تَعْبُدُوْنَ ۝١٦١**
p ترجمہ: سوائے اللہ کے خالص کیے ہوئے بندوں کے (جو ایسی باتیں نہیں کرتے وہ عذاب سے محفوظ رہیں گے)۔ تو بے شک تم اور جنھیں تم پوجتے ہو۔ || Except Allah's chosen servants (who say no such thing and will be safe). So you and what you worship || پرته د الله له خالص کړل شویو بندګانو (چې داسې خبرې نه کوي، خوندي به وي). نو بېشکه تاسو او هغه چې عبادت یې کوئ
p **مَآ اَنْتُمْ عَلَيْهِ بِفٰتِنِيْنَ ۝١٦٢ اِلَّا مَنْ هُوَ صَالِ الْجَحِيْمِ ۝١٦٣** || **مَآ اَنْتُمْ عَلَيْهِ بِفٰتِنِيْنَ ۝١٦٢ اِلَّا مَنْ هُوَ صَالِ الْجَحِيْمِ ۝١٦٣** || **مَآ اَنْتُمْ عَلَيْهِ بِفٰتِنِيْنَ ۝١٦٢ اِلَّا مَنْ هُوَ صَالِ الْجَحِيْمِ ۝١٦٣**
p ترجمہ: تم سب اُس (اللہ) کے خلاف (کسی کو) گم راہ نہیں کر سکتے۔ سوائے اس کے جس کو جہنم میں داخل ہونا ہے۔ || cannot lead anyone astray against Him, except one who is destined to burn in Hell. || تاسو ټول د هغه (الله) پر ضد (هېڅوک) نه شئ ګمراه کولای، پرته له هغه چې دوزخ ته ننوتونکی دی.
p **وَمَا مِنَّآ اِلَّا لَهٗ مَقَامٌ مَّعْلُوْمٌ ۝١٦٤ وَّاِنَّا لَنَحْنُ الصَّآفُّوْنَ ۝١٦٥** || **وَمَا مِنَّآ اِلَّا لَهٗ مَقَامٌ مَّعْلُوْمٌ ۝١٦٤ وَّاِنَّا لَنَحْنُ الصَّآفُّوْنَ ۝١٦٥** || **وَمَا مِنَّآ اِلَّا لَهٗ مَقَامٌ مَّعْلُوْمٌ ۝١٦٤ وَّاِنَّا لَنَحْنُ الصَّآفُّوْنَ ۝١٦٥**
p ترجمہ: اور ہم (فرشتوں) میں سے بھی ہر ایک کا مقام مقرر ہے۔ اور بے شک ہم ضرور صف باندھے کھڑے رہنے والے ہیں۔ || (The angels say:) "There is none of us but has a known station. We are those who stand in rows, || او له موږ (پرښتو) هر یو ټاکلی مقام لري. او بېشکه موږ خامخا په لیکو ولاړ یو،
p **وَاِنَّا لَنَحْنُ الْمُسَبِّحُوْنَ ۝١٦٦ وَاِنْ كَانُوْا لَيَقُوْلُوْنَ ۝١٦٧ لَوْ اَنَّ عِنْدَنَا ذِكْرًا مِّنَ الْاَوَّلِيْنَ ۝١٦٨** || **وَاِنَّا لَنَحْنُ الْمُسَبِّحُوْنَ ۝١٦٦ وَاِنْ كَانُوْا لَيَقُوْلُوْنَ ۝١٦٧ لَوْ اَنَّ عِنْدَنَا ذِكْرًا مِّنَ الْاَوَّلِيْنَ ۝١٦٨** || **وَاِنَّا لَنَحْنُ الْمُسَبِّحُوْنَ ۝١٦٦ وَاِنْ كَانُوْا لَيَقُوْلُوْنَ ۝١٦٧ لَوْ اَنَّ عِنْدَنَا ذِكْرًا مِّنَ الْاَوَّلِيْنَ ۝١٦٨**
p ترجمہ: اور ضرور ہم (اللہ کی) پاکی بیان کرنے والے ہیں۔ اور وہ (کافر) تو ضرور کہا کرتے تھے۔ اگر ہمارے پاس پچھلے لوگوں کی کوئی نصیحت (کی کتاب) ہوتی۔ || and we are those who glorify (Allah)." And they (the disbelievers) used to say, "If only we had a reminder (book) from the earlier peoples, || او بېشکه موږ د (الله) تسبیح ویونکي یو." او هغوی (کافرانو) به خامخا ویل: "که موږ سره د پخوانیو کوم نصیحت (کتاب) وای
p **لَكُنَّا عِبَادَ اللّٰهِ الْمُخْلَصِيْنَ ۝١٦٩ فَكَفَرُوْا بِهٖ فَسَوْفَ يَعْلَمُوْنَ ۝١٧٠ وَلَقَدْ سَبَقَتْ كَلِمَتُنَا** || **لَكُنَّا عِبَادَ اللّٰهِ الْمُخْلَصِيْنَ ۝١٦٩ فَكَفَرُوْا بِهٖ فَسَوْفَ يَعْلَمُوْنَ ۝١٧٠ وَلَقَدْ سَبَقَتْ كَلِمَتُنَا** || **لَكُنَّا عِبَادَ اللّٰهِ الْمُخْلَصِيْنَ ۝١٦٩ فَكَفَرُوْا بِهٖ فَسَوْفَ يَعْلَمُوْنَ ۝١٧٠ وَلَقَدْ سَبَقَتْ كَلِمَتُنَا**
p ترجمہ: تو یقیناً ہم اللہ کے خالص کیے ہوئے بندے ہوتے۔ پھر انھوں نے اس (قرآن) کا انکار کر دیا تو عنقریب وہ جان لیں گے۔ اور یقیناً ہماری بات پہلے طے ہو چکی ہے || we would surely have been Allah's chosen servants." Yet they disbelieved in it (the Quran); soon they will know. Our word has already gone forth || خامخا به موږ د الله خالص کړل شوي بندګان وای." خو هغوی په هغه (قرآن) کفر وکړ؛ ژر به پوه شي. او بېشکه زموږ خبره مخکې ټاکل شوې
p **لِعِبَادِنَا الْمُرْسَلِيْنَ ۝١٧١ اِنَّهُمْ لَهُمُ الْمَنْصُوْرُوْنَ ۝١٧٢ وَاِنَّ جُنْدَنَا لَهُمُ الْغٰلِبُوْنَ ۝١٧٣** || **لِعِبَادِنَا الْمُرْسَلِيْنَ ۝١٧١ اِنَّهُمْ لَهُمُ الْمَنْصُوْرُوْنَ ۝١٧٢ وَاِنَّ جُنْدَنَا لَهُمُ الْغٰلِبُوْنَ ۝١٧٣** || **لِعِبَادِنَا الْمُرْسَلِيْنَ ۝١٧١ اِنَّهُمْ لَهُمُ الْمَنْصُوْرُوْنَ ۝١٧٢ وَاِنَّ جُنْدَنَا لَهُمُ الْغٰلِبُوْنَ ۝١٧٣**
p ترجمہ: ہمارے بھیجے ہوئے بندوں (رسولوں) کے لیے۔ بے شک یہ وہ ہیں جن کی ضرور مدد کی جائے گی۔ اور بے شک ہمارا لشکر ہی ضرور غالب ہو کر رہے گا۔ || to Our servants, the messengers: that they will surely be helped, and that Our army will surely prevail. || زموږ د لېږل شویو بندګانو (رسولانو) لپاره: بېشکه همدوی دي چې خامخا به مرسته یې کېږي، او بېشکه زموږ لښکر به خامخا غالب وي.
p **فَتَوَلَّ عَنْهُمْ حَتّٰى حِيْنٍ ۝١٧٤ وَّاَبْصِرْهُمْ فَسَوْفَ يُبْصِرُوْنَ ۝١٧٥ اَفَبِعَذَابِنَا يَسْتَعْجِلُوْنَ ۝١٧٦** || **فَتَوَلَّ عَنْهُمْ حَتّٰى حِيْنٍ ۝١٧٤ وَّاَبْصِرْهُمْ فَسَوْفَ يُبْصِرُوْنَ ۝١٧٥ اَفَبِعَذَابِنَا يَسْتَعْجِلُوْنَ ۝١٧٦** || **فَتَوَلَّ عَنْهُمْ حَتّٰى حِيْنٍ ۝١٧٤ وَّاَبْصِرْهُمْ فَسَوْفَ يُبْصِرُوْنَ ۝١٧٥ اَفَبِعَذَابِنَا يَسْتَعْجِلُوْنَ ۝١٧٦**
p ترجمہ: لہٰذا آپ ان سے کچھ وقت تک منہ موڑ لیجیے۔ اور آپ انھیں دیکھتے رہیے پھر عنقریب یہ خود بھی دیکھ لیں گے۔ تو کیا وہ ہمارے عذاب کو جلدی مانگ رہے ہیں؟ || So turn away from them for a while, and watch them — they too will soon see. Do they seek to hasten Our punishment? || نو ته (ﷺ) تر یوه وخته ترې مخ واړوه، او ورته ګوره، ژر به پخپله هم ووینې. ایا زموږ عذاب ته بېړه کوي؟
p **فَاِذَا نَزَلَ بِسَاحَتِهِمْ فَسَآءَ صَبَاحُ الْمُنْذَرِيْنَ ۝١٧٧** || **فَاِذَا نَزَلَ بِسَاحَتِهِمْ فَسَآءَ صَبَاحُ الْمُنْذَرِيْنَ ۝١٧٧** || **فَاِذَا نَزَلَ بِسَاحَتِهِمْ فَسَآءَ صَبَاحُ الْمُنْذَرِيْنَ ۝١٧٧**
p ترجمہ: پھر جب وہ (عذاب) ان کے صحن میں اتر آئے گا وہ بہت بُری صبح ہو گی (عذاب سے) ڈرائے جانے والوں پر۔ || When it descends upon their courtyard, evil will be the morning of those who were warned. || کله چې هغه (عذاب) د دوی انګړ ته راکوز شي، د وېرول شویو سهار به ډېر بد وي.

p ترجمہ: اور آپ ان سے کچھ وقت تک منہ موڑ لیجیے۔ اور آپ دیکھتے رہیے پھر عنقریب یہ خود بھی دیکھ لیں گے۔ پاک ہے آپ کا رب جو عزت کا مالک ہے || And turn away from them for a while, and watch — they too will soon see.
