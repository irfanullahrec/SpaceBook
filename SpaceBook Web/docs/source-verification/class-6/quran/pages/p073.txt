p **وَتَقَلُّبَكَ فِي السّٰجِدِيْنَ ۝٢١٩ اِنَّهٗ هُوَ السَّمِيْعُ الْعَلِيْمُ ۝٢٢٠** || **وَتَقَلُّبَكَ فِي السّٰجِدِيْنَ ۝٢١٩ اِنَّهٗ هُوَ السَّمِيْعُ الْعَلِيْمُ ۝٢٢٠** || **وَتَقَلُّبَكَ فِي السّٰجِدِيْنَ ۝٢١٩ اِنَّهٗ هُوَ السَّمِيْعُ الْعَلِيْمُ ۝٢٢٠**
p ترجمہ: اور سجدہ کرنے والوں میں آپ (ﷺ) کی نشست و برخاست کو (بھی دیکھتا ہے)۔ بے شک وہ خوب سننے والا خوب جاننے والا ہے۔ || and (sees) your movements (ﷺ) among those who prostrate. He is the All-Hearing, the All-Knowing. || او د سجده کوونکو په منځ کې ستا (ﷺ) ناستې او پاڅون (هم ویني). بېشکه هغه ښه اورېدونکی، ښه پوه دی.
p **هَلْ اُنَبِّئُكُمْ عَلٰى مَنْ تَنَزَّلُ الشَّيٰطِيْنُ ۝٢٢١ تَنَزَّلُ عَلٰى كُلِّ اَفَّاكٍ اَثِيْمٍ ۝٢٢٢** || **هَلْ اُنَبِّئُكُمْ عَلٰى مَنْ تَنَزَّلُ الشَّيٰطِيْنُ ۝٢٢١ تَنَزَّلُ عَلٰى كُلِّ اَفَّاكٍ اَثِيْمٍ ۝٢٢٢** || **هَلْ اُنَبِّئُكُمْ عَلٰى مَنْ تَنَزَّلُ الشَّيٰطِيْنُ ۝٢٢١ تَنَزَّلُ عَلٰى كُلِّ اَفَّاكٍ اَثِيْمٍ ۝٢٢٢**
p ترجمہ: کیا میں تمھیں بتاؤں کہ شیاطین کس پر اترتے ہیں؟ وہ ہر جھوٹے گناہ گار پر اترتے ہیں۔ || Shall I tell you upon whom the devils descend? They descend upon every sinful liar. || ایا درته ووایم چې شیطانان پر چا کوزېږي؟ پر هر دروغجن ګناهګار کوزېږي.
p **يُّلْقُوْنَ السَّمْعَ وَاَكْثَرُهُمْ كٰذِبُوْنَ ۝٢٢٣ وَالشُّعَرَآءُ يَتَّبِعُهُمُ الْغَاوٗنَ ۝٢٢٤** || **يُّلْقُوْنَ السَّمْعَ وَاَكْثَرُهُمْ كٰذِبُوْنَ ۝٢٢٣ وَالشُّعَرَآءُ يَتَّبِعُهُمُ الْغَاوٗنَ ۝٢٢٤** || **يُّلْقُوْنَ السَّمْعَ وَاَكْثَرُهُمْ كٰذِبُوْنَ ۝٢٢٣ وَالشُّعَرَآءُ يَتَّبِعُهُمُ الْغَاوٗنَ ۝٢٢٤**
p ترجمہ: وہ سنی سنائی باتیں (کانوں میں) ڈال دیتے ہیں اور ان میں سے اکثر جھوٹے ہوتے ہیں۔ اور شاعروں کی پیروی گم راہ لوگ ہی کرتے ہیں۔ || They pass on what they hear, and most of them are liars. And as for the poets — only the misguided follow them. || هغوی اورېدلې خبرې (غوږونو ته) اچوي او ډېری یې دروغجن دي. او د شاعرانو پیروي یوازې ګمراهان کوي.
p **اَلَمْ تَرَ اَنَّهُمْ فِيْ كُلِّ وَادٍ يَّهِيْمُوْنَ ۝٢٢٥ وَاَنَّهُمْ يَقُوْلُوْنَ مَا لَا يَفْعَلُوْنَ ۝٢٢٦ اِلَّا الَّذِيْنَ اٰمَنُوْا** || **اَلَمْ تَرَ اَنَّهُمْ فِيْ كُلِّ وَادٍ يَّهِيْمُوْنَ ۝٢٢٥ وَاَنَّهُمْ يَقُوْلُوْنَ مَا لَا يَفْعَلُوْنَ ۝٢٢٦ اِلَّا الَّذِيْنَ اٰمَنُوْا** || **اَلَمْ تَرَ اَنَّهُمْ فِيْ كُلِّ وَادٍ يَّهِيْمُوْنَ ۝٢٢٥ وَاَنَّهُمْ يَقُوْلُوْنَ مَا لَا يَفْعَلُوْنَ ۝٢٢٦ اِلَّا الَّذِيْنَ اٰمَنُوْا**
p ترجمہ: کیا آپ نے نہیں دیکھا کہ بے شک وہ ہر وادی میں بھٹکتے پھرتے ہیں؟ اور وہ ایسی باتیں کہتے ہیں جو کرتے نہیں۔ سوائے ان (شاعروں) کے جو ایمان لائے || Do you not see that they wander aimlessly in every valley, and that they say what they do not do? Except those (poets) who believe, || ایا نه وینې چې هغوی په هر دره کې سرګردانه ګرځي؟ او هغه خبرې کوي چې نه یې کوي. پرته له هغو (شاعرانو) چې ایمان یې راوړ
p **وَعَمِلُوا الصّٰلِحٰتِ وَذَكَرُوا اللّٰهَ كَثِيْرًا وَّانْتَصَرُوْا مِنْۢ بَعْدِ مَا ظُلِمُوْا ؕ** || **وَعَمِلُوا الصّٰلِحٰتِ وَذَكَرُوا اللّٰهَ كَثِيْرًا وَّانْتَصَرُوْا مِنْۢ بَعْدِ مَا ظُلِمُوْا ؕ** || **وَعَمِلُوا الصّٰلِحٰتِ وَذَكَرُوا اللّٰهَ كَثِيْرًا وَّانْتَصَرُوْا مِنْۢ بَعْدِ مَا ظُلِمُوْا ؕ**
p ترجمہ: اور انھوں نے نیک اعمال کیے اور اللہ کو کثرت سے یاد کرتے رہے اور جنھوں نے اپنے اوپر ظلم ہونے کے بعد بدلہ لیا || do good deeds, remember Allah often, and defend themselves after being wronged. || او نېک عملونه یې وکړل، الله یې ډېر یاد کړ، او پر ځان له ظلم وروسته یې بدله واخیسته.
p **وَسَيَعْلَمُ الَّذِيْنَ ظَلَمُوْٓا اَيَّ مُنْقَلَبٍ يَّنْقَلِبُوْنَ ۝٢٢٧** || **وَسَيَعْلَمُ الَّذِيْنَ ظَلَمُوْٓا اَيَّ مُنْقَلَبٍ يَّنْقَلِبُوْنَ ۝٢٢٧** || **وَسَيَعْلَمُ الَّذِيْنَ ظَلَمُوْٓا اَيَّ مُنْقَلَبٍ يَّنْقَلِبُوْنَ ۝٢٢٧**
p ترجمہ: اور وہ عنقریب جان لیں گے جنھوں نے ظلم کیا کہ وہ کون سی پلٹنے کی جگہ واپس جائیں گے۔ (رکوع 11) || And the wrongdoers will soon know to what (evil) return they will return. (End of ruku' 11) || او ظالمان به ژر پوه شي چې کوم ځای ته به ورګرځي. (یوولسم رکوع)

## علم و عمل کی باتیں || Points of Knowledge and Practice || د علم او عمل خبرې
- ہمیں قرآنِ مجید سمجھنے کے لیے عربی زبان سیکھنی چاہیے۔ || We should learn Arabic to understand the Holy Quran. || د قرآن د پوهېدو لپاره باید عربي زده کړو.
- اللہ تعالیٰ پہلے انبیاء کرام علیہم السلام کے ذریعے قوموں کو بُرے انجام سے ڈراتا ہے پھر کفر و نافرمانی پر ڈٹے رہنے والوں کو ہلاک کر دیتا ہے۔ || Allah first warns nations of a bad end through the prophets, then destroys those who persist in disbelief and disobedience. || الله لومړی د پیغمبرانو له لارې قومونه له بد انجامه وېروي، بیا په کفر ټینګ پاتې کېدونکي هلاکوي.
- ہمیں اپنے رشتہ داروں کو اللہ تعالیٰ کی نافرمانی اور عذاب سے ڈرانا چاہیے۔ || We should warn our relatives against disobeying Allah and His punishment. || باید خپل خپلوان د الله له نافرمانۍ او عذابه وېرو.
- ہمیں ایمان والوں سے شفقت اور تواضع سے پیش آنا چاہیے۔ || We should treat the believers with kindness and humility. || له مومنانو سره باید په شفقت او تواضع چلند وکړو.
- شیطان جھوٹے اور گناہ گار لوگوں کا ساتھی ہوتا ہے۔ || Satan is the companion of liars and sinners. || شیطان د دروغجنو او ګناهګارو ملګری وي.
- جھوٹے شاعروں کی پیروی کرنے والے عموماً گم راہ لوگ ہوتے ہیں۔ || Those who follow lying poets are usually misguided people. || د دروغجنو شاعرانو پیروان عموماً ګمراهان وي.
- ایمان، عملِ صالح، ذکرِ الٰہی اور دین کا دفاع، اچھے شاعروں کی صفات ہیں۔ || Faith, good deeds, remembrance of Allah and defending the faith are the qualities of good poets. || ایمان، نېک عمل، د الله ذکر او د دین دفاع د ښو شاعرانو صفتونه دي.
