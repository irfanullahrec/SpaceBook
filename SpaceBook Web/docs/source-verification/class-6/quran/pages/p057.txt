p **كَمْ اَنْۢبَتْنَا فِيْهَا مِنْ كُلِّ زَوْجٍ كَرِيْمٍ ۝٧ اِنَّ فِيْ ذٰلِكَ لَاٰيَةً ؕ وَمَا كَانَ اَكْثَرُهُمْ مُّؤْمِنِيْنَ ۝٨** || **كَمْ اَنْۢبَتْنَا فِيْهَا مِنْ كُلِّ زَوْجٍ كَرِيْمٍ ۝٧ اِنَّ فِيْ ذٰلِكَ لَاٰيَةً ؕ وَمَا كَانَ اَكْثَرُهُمْ مُّؤْمِنِيْنَ ۝٨** || **كَمْ اَنْۢبَتْنَا فِيْهَا مِنْ كُلِّ زَوْجٍ كَرِيْمٍ ۝٧ اِنَّ فِيْ ذٰلِكَ لَاٰيَةً ؕ وَمَا كَانَ اَكْثَرُهُمْ مُّؤْمِنِيْنَ ۝٨**
p ترجمہ: کہ ہم نے اِس میں کس طرح عمدہ چیزیں اُگائی ہیں؟ یقیناً اس میں ایک نشانی ہے اور ان میں سے اکثر ایمان لانے والے نہیں۔ || how many excellent kinds (of plants) We have grown in it? In that is surely a sign, yet most of them do not believe. || چې موږ په کې څومره ښه جوړې جوړې (بوټي) زرغون کړي؟ بېشکه په دې کې نښه ده، خو ډېری یې ایمان نه راوړي.
p **وَاِنَّ رَبَّكَ لَهُوَ الْعَزِيْزُ الرَّحِيْمُ ۝٩ وَاِذْ نَادٰى رَبُّكَ مُوْسٰٓى** || **وَاِنَّ رَبَّكَ لَهُوَ الْعَزِيْزُ الرَّحِيْمُ ۝٩ وَاِذْ نَادٰى رَبُّكَ مُوْسٰٓى** || **وَاِنَّ رَبَّكَ لَهُوَ الْعَزِيْزُ الرَّحِيْمُ ۝٩ وَاِذْ نَادٰى رَبُّكَ مُوْسٰٓى**
p ترجمہ: اور بے شک آپ کا رب ہی بہت غالب نہایت رحم فرمانے والا ہے۔ (رکوع 1) اور جب آپ کے رب نے موسیٰ (علیہ السلام) کو ندا دی || And your Lord is the Almighty, the Most Merciful. (End of ruku' 1) And when your Lord called Musa: || او بېشکه ستا رب همغه ډېر غالب، ډېر مهربان دی. (لومړی رکوع) او کله چې ستا رب موسی (علیه السلام) ته غږ وکړ:
p **اَنِ ائْتِ الْقَوْمَ الظّٰلِمِيْنَ ۝١٠ قَوْمَ فِرْعَوْنَ ؕ اَلَا يَتَّقُوْنَ ۝١١ قَالَ رَبِّ** || **اَنِ ائْتِ الْقَوْمَ الظّٰلِمِيْنَ ۝١٠ قَوْمَ فِرْعَوْنَ ؕ اَلَا يَتَّقُوْنَ ۝١١ قَالَ رَبِّ** || **اَنِ ائْتِ الْقَوْمَ الظّٰلِمِيْنَ ۝١٠ قَوْمَ فِرْعَوْنَ ؕ اَلَا يَتَّقُوْنَ ۝١١ قَالَ رَبِّ**
p ترجمہ: کہ تم ظالم قوم کے پاس جاؤ۔ (یعنی) قومِ فرعون کے پاس کیا وہ لوگ (اللہ کی نافرمانی سے) ڈرتے نہیں؟ (موسیٰ علیہ السلام نے) عرض کیا اے میرے رب! || "Go to the wrongdoing people — the people of Pharaoh. Will they not fear (disobeying Allah)?" (Musa) said, "My Lord! || "ظالم قوم ته ورشه — د فرعون قوم ته؛ ایا (د الله له نافرمانۍ) نه وېرېږي؟" (موسی) عرض وکړ: "ای زما ربه!
p **اِنِّيْٓ اَخَافُ اَنْ يُّكَذِّبُوْنِ ۝١٢ وَيَضِيْقُ صَدْرِيْ وَلَا يَنْطَلِقُ لِسَانِيْ** || **اِنِّيْٓ اَخَافُ اَنْ يُّكَذِّبُوْنِ ۝١٢ وَيَضِيْقُ صَدْرِيْ وَلَا يَنْطَلِقُ لِسَانِيْ** || **اِنِّيْٓ اَخَافُ اَنْ يُّكَذِّبُوْنِ ۝١٢ وَيَضِيْقُ صَدْرِيْ وَلَا يَنْطَلِقُ لِسَانِيْ**
p ترجمہ: میں ڈرتا ہوں کہ وہ مجھے جھٹلا دیں گے۔ اور میرا سینہ تنگ ہوتا ہے اور میری زبان (خوب روانی سے) نہیں چلتی || I fear they will deny me, and my chest tightens and my tongue does not speak (fluently), || زه وېرېږم چې دروغجن به مې وګڼي، او سینه مې تنګېږي او ژبه مې (روانه) نه چلېږي
p **فَاَرْسِلْ اِلٰى هٰرُوْنَ ۝١٣ وَلَهُمْ عَلَيَّ ذَنْۢبٌ فَاَخَافُ اَنْ يَّقْتُلُوْنِ ۝١٤** || **فَاَرْسِلْ اِلٰى هٰرُوْنَ ۝١٣ وَلَهُمْ عَلَيَّ ذَنْۢبٌ فَاَخَافُ اَنْ يَّقْتُلُوْنِ ۝١٤** || **فَاَرْسِلْ اِلٰى هٰرُوْنَ ۝١٣ وَلَهُمْ عَلَيَّ ذَنْۢبٌ فَاَخَافُ اَنْ يَّقْتُلُوْنِ ۝١٤**
p ترجمہ: پس تو ہارون (علیہ السلام) کی طرف بھی (نبوت) بھیج دے۔ اور ان کا میرے اوپر ایک (قتل کا) الزام بھی ہے تو میں ڈرتا ہوں کہ وہ مجھے قتل (نہ) کر دیں۔ || so send (prophethood) to Harun as well. And they have a charge (of killing) against me, so I fear they will kill me." || نو هارون (علیه السلام) ته هم (نبوت) ولېږه. او زما پر ضد یې د (وژنې) تور هم شته، نو وېرېږم چې ونه وژني مې."
p **قَالَ كَلَّا ۚ فَاذْهَبَا بِاٰيٰتِنَآ اِنَّا مَعَكُمْ مُّسْتَمِعُوْنَ ۝١٥** || **قَالَ كَلَّا ۚ فَاذْهَبَا بِاٰيٰتِنَآ اِنَّا مَعَكُمْ مُّسْتَمِعُوْنَ ۝١٥** || **قَالَ كَلَّا ۚ فَاذْهَبَا بِاٰيٰتِنَآ اِنَّا مَعَكُمْ مُّسْتَمِعُوْنَ ۝١٥**
p ترجمہ: (اللہ نے) فرمایا ہرگز (ایسا) نہیں (ہو گا)! لہٰذا تم دونوں ہماری نشانیاں لے کر جاؤ بے شک ہم تمھارے ساتھ سننے والے ہیں۔ || (Allah) said, "Never (will that happen)! Go, both of you, with Our signs; We are with you, listening. || (الله) وفرمایل: "هېڅکله (داسې) نه (کېږي)! نو دواړه زموږ له نښو سره ورشئ؛ بېشکه موږ له تاسو سره اورېدونکي یو.
p **فَاْتِيَا فِرْعَوْنَ فَقُوْلَآ اِنَّا رَسُوْلُ رَبِّ الْعٰلَمِيْنَ ۝١٦ اَنْ اَرْسِلْ مَعَنَا بَنِيْٓ اِسْرَآءِيْلَ ۝١٧** || **فَاْتِيَا فِرْعَوْنَ فَقُوْلَآ اِنَّا رَسُوْلُ رَبِّ الْعٰلَمِيْنَ ۝١٦ اَنْ اَرْسِلْ مَعَنَا بَنِيْٓ اِسْرَآءِيْلَ ۝١٧** || **فَاْتِيَا فِرْعَوْنَ فَقُوْلَآ اِنَّا رَسُوْلُ رَبِّ الْعٰلَمِيْنَ ۝١٦ اَنْ اَرْسِلْ مَعَنَا بَنِيْٓ اِسْرَآءِيْلَ ۝١٧**
p ترجمہ: لہٰذا تم دونوں فرعون کے پاس جاؤ پھر کہو کہ ہم تمام جہانوں کے رب کے بھیجے ہوئے ہیں۔ کہ تم بنی اسرائیل کو ہمارے ساتھ بھیج دو۔ || Go to Pharaoh and say, 'We are messengers of the Lord of all the worlds: let the Children of Israel go with us.'" || نو دواړه فرعون ته ورشئ او ووایئ: 'موږ د ټولو نړیو د رب لېږل شوي یو؛ چې بني اسرائیل له موږ سره ولېږې.'"
p **قَالَ اَلَمْ نُرَبِّكَ فِيْنَا وَلِيْدًا وَّلَبِثْتَ فِيْنَا مِنْ عُمُرِكَ سِنِيْنَ ۝١٨** || **قَالَ اَلَمْ نُرَبِّكَ فِيْنَا وَلِيْدًا وَّلَبِثْتَ فِيْنَا مِنْ عُمُرِكَ سِنِيْنَ ۝١٨** || **قَالَ اَلَمْ نُرَبِّكَ فِيْنَا وَلِيْدًا وَّلَبِثْتَ فِيْنَا مِنْ عُمُرِكَ سِنِيْنَ ۝١٨**
p ترجمہ: (فرعون نے) کہا کیا ہم نے تمھیں بچپن میں اپنے پاس نہیں پالا تھا؟ اور تم نے ہمارے پاس اپنی عمر کے کئی سال گزارے۔ || (Pharaoh) said, "Did we not raise you among us as a child, and did you not spend many years of your life with us? || (فرعون) وویل: "ایا موږ ته په ماشومتوب کې له ځان سره نه یې پاللی؟ او ستا د عمر ډېر کلونه له موږ سره تېر شول.
p **وَفَعَلْتَ فَعْلَتَكَ الَّتِيْ فَعَلْتَ وَاَنْتَ مِنَ الْكٰفِرِيْنَ ۝١٩ قَالَ فَعَلْتُهَآ اِذًا وَّاَنَا مِنَ الضَّآلِّيْنَ ۝٢٠** || **وَفَعَلْتَ فَعْلَتَكَ الَّتِيْ فَعَلْتَ وَاَنْتَ مِنَ الْكٰفِرِيْنَ ۝١٩ قَالَ فَعَلْتُهَآ اِذًا وَّاَنَا مِنَ الضَّآلِّيْنَ ۝٢٠** || **وَفَعَلْتَ فَعْلَتَكَ الَّتِيْ فَعَلْتَ وَاَنْتَ مِنَ الْكٰفِرِيْنَ ۝١٩ قَالَ فَعَلْتُهَآ اِذًا وَّاَنَا مِنَ الضَّآلِّيْنَ ۝٢٠**
p ترجمہ: اور (پھر) تم نے کر ڈالا اپنا کام جو تم نے کیا اور تم تو ناشکروں میں سے ہو۔ (موسیٰ علیہ السلام نے) فرمایا میں نے وہ اس حال میں کیا جب کہ مجھے پتہ نہیں تھا۔ || And you did that deed of yours which you did, and you are one of the ungrateful." (Musa) said, "I did it when I was unaware. || او (بیا) هغه کار دې وکړ چې دې وکړ، او ته له ناشکرانو یې." (موسی) وویل: "ما هغه هغه وخت وکړ چې خبر نه وم.
p **فَفَرَرْتُ مِنْكُمْ لَمَّا خِفْتُكُمْ فَوَهَبَ لِيْ رَبِّيْ حُكْمًا وَّجَعَلَنِيْ مِنَ الْمُرْسَلِيْنَ ۝٢١** || **فَفَرَرْتُ مِنْكُمْ لَمَّا خِفْتُكُمْ فَوَهَبَ لِيْ رَبِّيْ حُكْمًا وَّجَعَلَنِيْ مِنَ الْمُرْسَلِيْنَ ۝٢١** || **فَفَرَرْتُ مِنْكُمْ لَمَّا خِفْتُكُمْ فَوَهَبَ لِيْ رَبِّيْ حُكْمًا وَّجَعَلَنِيْ مِنَ الْمُرْسَلِيْنَ ۝٢١**
p ترجمہ: تو میں تمھارے پاس سے چلا گیا جب کہ مجھے تم سے خوف ہوا پھر میرے رب نے مجھے حکم (یعنی نبوت اور علم) عطا فرمایا اور مجھے رسولوں میں سے بنایا۔ || So I fled from you when I feared you; then my Lord granted me wisdom (prophethood and knowledge) and made me one of the messengers. || نو کله چې له تاسو ووېرېدم، درنه لاړم؛ بیا زما رب ماته حکم (یعنې نبوت او علم) راکړ او له رسولانو یې کړم.
p **وَتِلْكَ نِعْمَةٌ تَمُنُّهَا عَلَيَّ اَنْ عَبَّدْتَّ بَنِيْٓ اِسْرَآءِيْلَ ۝٢٢ قَالَ فِرْعَوْنُ وَمَا رَبُّ الْعٰلَمِيْنَ ۝٢٣** || **وَتِلْكَ نِعْمَةٌ تَمُنُّهَا عَلَيَّ اَنْ عَبَّدْتَّ بَنِيْٓ اِسْرَآءِيْلَ ۝٢٢ قَالَ فِرْعَوْنُ وَمَا رَبُّ الْعٰلَمِيْنَ ۝٢٣** || **وَتِلْكَ نِعْمَةٌ تَمُنُّهَا عَلَيَّ اَنْ عَبَّدْتَّ بَنِيْٓ اِسْرَآءِيْلَ ۝٢٢ قَالَ فِرْعَوْنُ وَمَا رَبُّ الْعٰلَمِيْنَ ۝٢٣**
p ترجمہ: اور یہ کوئی احسان ہے جو تم مجھ پر جتا رہے ہو (جب) کہ تم نے بنی اسرائیل کو غلام بنا رکھا ہے۔ فرعون نے کہا اور (یہ) تمام جہانوں کا رب کیا ہے؟ || And is that a favour you remind me of — that you enslaved the Children of Israel?" Pharaoh said, "And what is the 'Lord of all the worlds'?" || او دا څه احسان دی چې پر ما یې منت کوې (حال دا) چې بني اسرائیل دې غلامان کړي؟" فرعون وویل: "او (دا) د ټولو نړیو رب څه دی؟"
p **قَالَ رَبُّ السَّمٰوٰتِ وَالْاَرْضِ وَمَا بَيْنَهُمَا ؕ اِنْ كُنْتُمْ مُّوْقِنِيْنَ ۝٢٤ قَالَ** || **قَالَ رَبُّ السَّمٰوٰتِ وَالْاَرْضِ وَمَا بَيْنَهُمَا ؕ اِنْ كُنْتُمْ مُّوْقِنِيْنَ ۝٢٤ قَالَ** || **قَالَ رَبُّ السَّمٰوٰتِ وَالْاَرْضِ وَمَا بَيْنَهُمَا ؕ اِنْ كُنْتُمْ مُّوْقِنِيْنَ ۝٢٤ قَالَ**
p ترجمہ: (موسیٰ علیہ السلام نے) فرمایا (وہ) رب ہے آسمانوں کا اور زمینوں کا اور جو کچھ ان دونوں کے درمیان ہے اگر تم یقین کرنے والے ہو۔ (فرعون نے) کہا || (Musa) said, "The Lord of the heavens and the earth and all that is between them, if you would be certain." (Pharaoh) said || (موسی) وویل: "د اسمانونو، ځمکې او د دواړو تر منځ د هر څه رب، که یقین لرئ." (فرعون) وویل
