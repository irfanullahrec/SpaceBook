p **اَلَّذِيْنَ يَتَّبِعُوْنَ الرَّسُوْلَ النَّبِيَّ الْاُمِّيَّ الَّذِيْ يَجِدُوْنَهٗ مَكْتُوْبًا عِنْدَهُمْ فِي التَّوْرٰىةِ وَالْاِنْجِيْلِ ۫** || **اَلَّذِيْنَ يَتَّبِعُوْنَ الرَّسُوْلَ النَّبِيَّ الْاُمِّيَّ الَّذِيْ يَجِدُوْنَهٗ مَكْتُوْبًا عِنْدَهُمْ فِي التَّوْرٰىةِ وَالْاِنْجِيْلِ ۫** || **اَلَّذِيْنَ يَتَّبِعُوْنَ الرَّسُوْلَ النَّبِيَّ الْاُمِّيَّ الَّذِيْ يَجِدُوْنَهٗ مَكْتُوْبًا عِنْدَهُمْ فِي التَّوْرٰىةِ وَالْاِنْجِيْلِ ۫**
p ترجمہ: جو لوگ اس رسول (ﷺ) کی پیروی کرتے ہیں جو نبی اُمّی (ﷺ) ہیں جنھیں وہ اپنے پاس لکھا ہوا پاتے ہیں تورات و انجیل میں || those who follow the Messenger ﷺ, the unlettered Prophet ﷺ, whom they find written with them in the Torah and the Gospel. || هغه چې د دې رسول ﷺ پیروي کوي چې امي نبي ﷺ دی، چې په تورات او انجیل کې یې له ځان سره لیکلی مومي.
p **يَاْمُرُهُمْ بِالْمَعْرُوْفِ وَيَنْهٰىهُمْ عَنِ الْمُنْكَرِ** || **يَاْمُرُهُمْ بِالْمَعْرُوْفِ وَيَنْهٰىهُمْ عَنِ الْمُنْكَرِ** || **يَاْمُرُهُمْ بِالْمَعْرُوْفِ وَيَنْهٰىهُمْ عَنِ الْمُنْكَرِ**
p ترجمہ: وہ (نبی) انھیں اچھے کاموں کا حکم دیتے ہیں اور انھیں بُرائیوں سے روکتے ہیں || He (the Prophet) commands them what is right and forbids them what is wrong, || هغه (نبي) ښو کارونو ته امر کوي او له بدیو منع کوي
p **وَيُحِلُّ لَهُمُ الطَّيِّبٰتِ وَيُحَرِّمُ عَلَيْهِمُ الْخَبٰٓئِثَ** || **وَيُحِلُّ لَهُمُ الطَّيِّبٰتِ وَيُحَرِّمُ عَلَيْهِمُ الْخَبٰٓئِثَ** || **وَيُحِلُّ لَهُمُ الطَّيِّبٰتِ وَيُحَرِّمُ عَلَيْهِمُ الْخَبٰٓئِثَ**
p ترجمہ: اور ان کے لیے پاکیزہ چیزوں کو حلال قرار دیتے ہیں اور ناپاک چیزوں کو ان پر حرام کرتے ہیں || makes good things lawful for them and forbids them impure things, || او پاک شیان ورته حلالوي او ناپاک پرې حراموي
p **وَيَضَعُ عَنْهُمْ اِصْرَهُمْ وَالْاَغْلٰلَ الَّتِيْ كَانَتْ عَلَيْهِمْ ؕ فَالَّذِيْنَ اٰمَنُوْا بِهٖ وَعَزَّرُوْهُ** || **وَيَضَعُ عَنْهُمْ اِصْرَهُمْ وَالْاَغْلٰلَ الَّتِيْ كَانَتْ عَلَيْهِمْ ؕ فَالَّذِيْنَ اٰمَنُوْا بِهٖ وَعَزَّرُوْهُ** || **وَيَضَعُ عَنْهُمْ اِصْرَهُمْ وَالْاَغْلٰلَ الَّتِيْ كَانَتْ عَلَيْهِمْ ؕ فَالَّذِيْنَ اٰمَنُوْا بِهٖ وَعَزَّرُوْهُ**
p ترجمہ: اور ان سے ان کے بوجھ اتارتے ہیں اور وہ طوق (بھی) جو ان پر پڑے ہوئے تھے پس جو لوگ ان پر ایمان لائے اور جنھوں نے ان کی تعظیم کی || and relieves them of their burdens and the shackles that were upon them. So those who believe in him, honour him, || او بارونه او هغه زنځیرونه ترې لرې کوي چې پرې وو. نو چا چې پرې ایمان راوړ او درناوی یې وکړ
p **وَنَصَرُوْهُ وَاتَّبَعُوا النُّوْرَ الَّذِيْٓ اُنْزِلَ مَعَهٗٓ ۙ اُولٰٓئِكَ هُمُ الْمُفْلِحُوْنَ ۝١٥٧ قُلْ يٰٓاَيُّهَا النَّاسُ** || **وَنَصَرُوْهُ وَاتَّبَعُوا النُّوْرَ الَّذِيْٓ اُنْزِلَ مَعَهٗٓ ۙ اُولٰٓئِكَ هُمُ الْمُفْلِحُوْنَ ۝١٥٧ قُلْ يٰٓاَيُّهَا النَّاسُ** || **وَنَصَرُوْهُ وَاتَّبَعُوا النُّوْرَ الَّذِيْٓ اُنْزِلَ مَعَهٗٓ ۙ اُولٰٓئِكَ هُمُ الْمُفْلِحُوْنَ ۝١٥٧ قُلْ يٰٓاَيُّهَا النَّاسُ**
p ترجمہ: اور ان کی مدد کی اور اس نور (قرآن) کی پیروی کی جو ان کے ساتھ اتارا گیا ہے یہی لوگ کامیاب ہونے والے ہیں۔ (رکوع 19) آپ (ﷺ) فرما دیجیے اے لوگو! || support him and follow the light (the Quran) sent down with him — they are the successful. (End of ruku' 19) Say (ﷺ): "O people! || او مرسته یې وکړه او د هغه نور (قرآن) پیروي یې وکړه چې ورسره نازل شوی، همدوی بریالي دي. (نولسم رکوع) ته (ﷺ) ووایه: "ای خلکو!
p **اِنِّيْ رَسُوْلُ اللّٰهِ اِلَيْكُمْ جَمِيْعَا ِۨ الَّذِيْ لَهٗ مُلْكُ السَّمٰوٰتِ وَالْاَرْضِ ۚ لَآ اِلٰهَ اِلَّا هُوَ** || **اِنِّيْ رَسُوْلُ اللّٰهِ اِلَيْكُمْ جَمِيْعَا ِۨ الَّذِيْ لَهٗ مُلْكُ السَّمٰوٰتِ وَالْاَرْضِ ۚ لَآ اِلٰهَ اِلَّا هُوَ** || **اِنِّيْ رَسُوْلُ اللّٰهِ اِلَيْكُمْ جَمِيْعَا ِۨ الَّذِيْ لَهٗ مُلْكُ السَّمٰوٰتِ وَالْاَرْضِ ۚ لَآ اِلٰهَ اِلَّا هُوَ**
p ترجمہ: بے شک میں تم سب کی طرف اللہ کا رسول ہوں وہ (اللہ) جس کے لیے آسمانوں اور زمینوں کی بادشاہت ہے اس (اللہ) کے سوا کوئی معبود نہیں || I am the Messenger of Allah to you all — (Allah) to whom belongs the kingdom of the heavens and the earth. There is no god but He; || بېشکه زه تاسو ټولو ته د الله رسول یم — هغه (الله) چې د اسمانونو او ځمکې پاچاهي یې ده؛ له هغه پرته بل معبود نشته؛
p **يُحْيٖ وَيُمِيْتُ ۪ فَاٰمِنُوْا بِاللّٰهِ وَرَسُوْلِهِ النَّبِيِّ الْاُمِّيِّ الَّذِيْ يُؤْمِنُ بِاللّٰهِ وَكَلِمٰتِهٖ** || **يُحْيٖ وَيُمِيْتُ ۪ فَاٰمِنُوْا بِاللّٰهِ وَرَسُوْلِهِ النَّبِيِّ الْاُمِّيِّ الَّذِيْ يُؤْمِنُ بِاللّٰهِ وَكَلِمٰتِهٖ** || **يُحْيٖ وَيُمِيْتُ ۪ فَاٰمِنُوْا بِاللّٰهِ وَرَسُوْلِهِ النَّبِيِّ الْاُمِّيِّ الَّذِيْ يُؤْمِنُ بِاللّٰهِ وَكَلِمٰتِهٖ**
p ترجمہ: وہ زندگی اور موت دیتا ہے تو ایمان لاؤ اللہ پر اور اس کے رسول پر جو نبی اُمّی (ﷺ) ہیں جو (خود بھی) اللہ اور اس کے کلمات پر ایمان رکھتے ہیں || He gives life and causes death. So believe in Allah and His Messenger, the unlettered Prophet ﷺ, who (himself) believes in Allah and His words, || هغه ژوند او مرګ ورکوي؛ نو پر الله او د هغه پر رسول، امي نبي ﷺ، ایمان راوړئ چې (پخپله هم) پر الله او د هغه پر کلماتو ایمان لري
p **وَاتَّبِعُوْهُ لَعَلَّكُمْ تَهْتَدُوْنَ ۝١٥٨ وَمِنْ قَوْمِ مُوْسٰٓى اُمَّةٌ يَّهْدُوْنَ بِالْحَقِّ** || **وَاتَّبِعُوْهُ لَعَلَّكُمْ تَهْتَدُوْنَ ۝١٥٨ وَمِنْ قَوْمِ مُوْسٰٓى اُمَّةٌ يَّهْدُوْنَ بِالْحَقِّ** || **وَاتَّبِعُوْهُ لَعَلَّكُمْ تَهْتَدُوْنَ ۝١٥٨ وَمِنْ قَوْمِ مُوْسٰٓى اُمَّةٌ يَّهْدُوْنَ بِالْحَقِّ**
p ترجمہ: اور ان کی پیروی کرو تاکہ تم ہدایت پاؤ۔ اور موسیٰ (علیہ السلام) کی قوم میں ایک گروہ (ایسے لوگوں کا بھی) ہے جو حق کے ساتھ راہ نمائی کرتے ہیں || and follow him so that you may be guided." And among the people of Musa is a community that guides by the truth || او پیروي یې وکړئ چې هدایت ومومئ." او د موسی (علیه السلام) په قوم کې یوه ډله شته چې په حق لارښوونه کوي
p **وَبِهٖ يَعْدِلُوْنَ ۝١٥٩ وَقَطَّعْنٰهُمُ اثْنَتَيْ عَشْرَةَ اَسْبَاطًا اُمَمًا ؕ** || **وَبِهٖ يَعْدِلُوْنَ ۝١٥٩ وَقَطَّعْنٰهُمُ اثْنَتَيْ عَشْرَةَ اَسْبَاطًا اُمَمًا ؕ** || **وَبِهٖ يَعْدِلُوْنَ ۝١٥٩ وَقَطَّعْنٰهُمُ اثْنَتَيْ عَشْرَةَ اَسْبَاطًا اُمَمًا ؕ**
p ترجمہ: اور اسی کے مطابق انصاف کرتے ہیں۔ اور ہم نے ان کو بارہ خاندانوں میں تقسیم کر کے الگ الگ جماعتیں بنا دیں || and by it establishes justice. And We divided them into twelve tribes (descended from Ya'qub) as communities. || او په همدې انصاف کوي. او موږ هغوی په دولسو کورنیو وېشلي، جلا جلا ډلې مو کړې.
p **وَاَوْحَيْنَآ اِلٰى مُوْسٰٓى اِذِ اسْتَسْقٰىهُ قَوْمُهٗٓ اَنِ اضْرِبْ بِّعَصَاكَ الْحَجَرَ ۚ** || **وَاَوْحَيْنَآ اِلٰى مُوْسٰٓى اِذِ اسْتَسْقٰىهُ قَوْمُهٗٓ اَنِ اضْرِبْ بِّعَصَاكَ الْحَجَرَ ۚ** || **وَاَوْحَيْنَآ اِلٰى مُوْسٰٓى اِذِ اسْتَسْقٰىهُ قَوْمُهٗٓ اَنِ اضْرِبْ بِّعَصَاكَ الْحَجَرَ ۚ**
p ترجمہ: اور جب موسیٰ (علیہ السلام) سے ان کی قوم نے پانی مانگا تو ہم نے ان کی طرف وحی بھیجی کہ اپنا عصا پتھر پر ماریے || And when his people asked Musa for water, We inspired him, "Strike the rock with your staff," || او کله چې د موسی (علیه السلام) قوم ترې اوبه وغوښتې، ورته مو وحی وکړه چې امسا دې پر ډبره ووهه
p **فَانْۢبَجَسَتْ مِنْهُ اثْنَتَا عَشْرَةَ عَيْنًا ؕ قَدْ عَلِمَ كُلُّ اُنَاسٍ مَّشْرَبَهُمْ ؕ وَظَلَّلْنَا عَلَيْهِمُ الْغَمَامَ** || **فَانْۢبَجَسَتْ مِنْهُ اثْنَتَا عَشْرَةَ عَيْنًا ؕ قَدْ عَلِمَ كُلُّ اُنَاسٍ مَّشْرَبَهُمْ ؕ وَظَلَّلْنَا عَلَيْهِمُ الْغَمَامَ** || **فَانْۢبَجَسَتْ مِنْهُ اثْنَتَا عَشْرَةَ عَيْنًا ؕ قَدْ عَلِمَ كُلُّ اُنَاسٍ مَّشْرَبَهُمْ ؕ وَظَلَّلْنَا عَلَيْهِمُ الْغَمَامَ**
p ترجمہ: تو اس میں سے بارہ چشمے پھوٹ پڑے ہر قبیلہ نے اپنے پانی پینے کی جگہ جان لی اور ہم نے ان پر بادلوں کا سایہ کیا || and twelve springs gushed out of it; each tribe knew its drinking place. And We shaded them with clouds || نو ترې دولس چینې وبهېدې؛ هرې قبیلې خپل د اوبو ځای وپېژانده؛ او موږ پرې د ورېځو سیوری وکړ
p **وَاَنْزَلْنَا عَلَيْهِمُ الْمَنَّ وَالسَّلْوٰى ؕ كُلُوْا مِنْ طَيِّبٰتِ مَا رَزَقْنٰكُمْ ؕ وَمَا ظَلَمُوْنَا** || **وَاَنْزَلْنَا عَلَيْهِمُ الْمَنَّ وَالسَّلْوٰى ؕ كُلُوْا مِنْ طَيِّبٰتِ مَا رَزَقْنٰكُمْ ؕ وَمَا ظَلَمُوْنَا** || **وَاَنْزَلْنَا عَلَيْهِمُ الْمَنَّ وَالسَّلْوٰى ؕ كُلُوْا مِنْ طَيِّبٰتِ مَا رَزَقْنٰكُمْ ؕ وَمَا ظَلَمُوْنَا**
p ترجمہ: اور ان پر مَنّ اور سَلویٰ اُتارا (ان سے فرمایا کہ) تم ان پاکیزہ چیزوں میں سے کھاؤ جو ہم نے تمھیں عطا فرمائی ہیں اور انھوں نے ہمارا نقصان نہ کیا || and sent down upon them manna and quails: "Eat of the good things We have provided you." They did not wrong Us, || او پرې من او سلوی نازل کړل: "له هغو پاکو شیانو وخورئ چې درکړي مو دي." هغوی موږ ته زیان ونه رساوه
