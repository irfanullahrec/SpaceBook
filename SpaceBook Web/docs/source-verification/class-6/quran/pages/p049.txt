p **اُولٰٓئِكَ هُمُ الْغٰفِلُوْنَ ۝١٧٩ وَلِلّٰهِ الْاَسْمَآءُ الْحُسْنٰى فَادْعُوْهُ بِهَا ۪** || **اُولٰٓئِكَ هُمُ الْغٰفِلُوْنَ ۝١٧٩ وَلِلّٰهِ الْاَسْمَآءُ الْحُسْنٰى فَادْعُوْهُ بِهَا ۪** || **اُولٰٓئِكَ هُمُ الْغٰفِلُوْنَ ۝١٧٩ وَلِلّٰهِ الْاَسْمَآءُ الْحُسْنٰى فَادْعُوْهُ بِهَا ۪**
p ترجمہ: یہی لوگ غافل ہیں۔ اور اللہ ہی کے لیے تمام اچھے نام ہیں لہٰذا اسے ان (ناموں) کے ساتھ پکارو || They are the heedless. And to Allah belong the most beautiful names, so call upon Him by them, || همدوی غافلان دي. او ټول ښکلي نومونه د الله دي؛ نو په همدې (نومونو) یې وبولئ
p **وَذَرُوا الَّذِيْنَ يُلْحِدُوْنَ فِيْٓ اَسْمَآئِهٖ ؕ سَيُجْزَوْنَ مَا كَانُوْا يَعْمَلُوْنَ ۝١٨٠** || **وَذَرُوا الَّذِيْنَ يُلْحِدُوْنَ فِيْٓ اَسْمَآئِهٖ ؕ سَيُجْزَوْنَ مَا كَانُوْا يَعْمَلُوْنَ ۝١٨٠** || **وَذَرُوا الَّذِيْنَ يُلْحِدُوْنَ فِيْٓ اَسْمَآئِهٖ ؕ سَيُجْزَوْنَ مَا كَانُوْا يَعْمَلُوْنَ ۝١٨٠**
p ترجمہ: اور ان لوگوں کو چھوڑ دو جو اس کے ناموں میں ٹیڑھ اختیار کرتے ہیں عنقریب انھیں اس کا بدلہ دیا جائے گا جو وہ کیا کرتے تھے۔ || and leave those who distort His names; they will be repaid for what they used to do. || او هغه پرېږدئ چې د هغه په نومونو کې کوږوالی کوي؛ ژر به یې د کړنو بدله ورکړل شي.
p **وَمِمَّنْ خَلَقْنَآ اُمَّةٌ يَّهْدُوْنَ بِالْحَقِّ وَبِهٖ يَعْدِلُوْنَ ۝١٨١** || **وَمِمَّنْ خَلَقْنَآ اُمَّةٌ يَّهْدُوْنَ بِالْحَقِّ وَبِهٖ يَعْدِلُوْنَ ۝١٨١** || **وَمِمَّنْ خَلَقْنَآ اُمَّةٌ يَّهْدُوْنَ بِالْحَقِّ وَبِهٖ يَعْدِلُوْنَ ۝١٨١**
p ترجمہ: اور جو لوگ ہم نے پیدا کیے ان میں سے ایک گروہ ایسا بھی ہے جو حق کے مطابق راہ نمائی کرتا ہے اور اسی کے مطابق انصاف کرتا ہے۔ (رکوع 22) || Among those We have created is a community that guides by the truth and by it establishes justice. (End of ruku' 22) || او له هغو چې موږ پیدا کړي، یوه ډله شته چې په حق لارښوونه کوي او په همدې انصاف کوي. (دوه ویشتم رکوع)
p **وَالَّذِيْنَ كَذَّبُوْا بِاٰيٰتِنَا سَنَسْتَدْرِجُهُمْ** || **وَالَّذِيْنَ كَذَّبُوْا بِاٰيٰتِنَا سَنَسْتَدْرِجُهُمْ** || **وَالَّذِيْنَ كَذَّبُوْا بِاٰيٰتِنَا سَنَسْتَدْرِجُهُمْ**
p ترجمہ: اور جن لوگوں نے ہماری آیتوں کو جھٹلایا ہم آہستہ آہستہ انھیں (عذاب کی طرف اس طرح) لے جائیں گے || As for those who deny Our signs, We will draw them on little by little (towards punishment) || او چا چې زموږ آیتونه دروغ ګڼل، ورو ورو به یې (عذاب ته داسې) بوځو
p **مِّنْ حَيْثُ لَا يَعْلَمُوْنَ ۝١٨٢ وَاُمْلِيْ لَهُمْ ۚ اِنَّ كَيْدِيْ مَتِيْنٌ ۝١٨٣** || **مِّنْ حَيْثُ لَا يَعْلَمُوْنَ ۝١٨٢ وَاُمْلِيْ لَهُمْ ۚ اِنَّ كَيْدِيْ مَتِيْنٌ ۝١٨٣** || **مِّنْ حَيْثُ لَا يَعْلَمُوْنَ ۝١٨٢ وَاُمْلِيْ لَهُمْ ۚ اِنَّ كَيْدِيْ مَتِيْنٌ ۝١٨٣**
p ترجمہ: کہ انھیں خبر بھی نہ ہو گی۔ اور میں انھیں مہلت دوں گا بے شک میری تدبیر بہت مضبوط ہے۔ || from where they do not know. And I give them respite; My plan is firm. || چې خبر به هم نه شي. او مهلت به ورکړم؛ بېشکه زما تدبیر ډېر ټینګ دی.
p **اَوَلَمْ يَتَفَكَّرُوْا ۫ مَا بِصَاحِبِهِمْ مِّنْ جِنَّةٍ ؕ اِنْ هُوَ اِلَّا نَذِيْرٌ مُّبِيْنٌ ۝١٨٤** || **اَوَلَمْ يَتَفَكَّرُوْا ۫ مَا بِصَاحِبِهِمْ مِّنْ جِنَّةٍ ؕ اِنْ هُوَ اِلَّا نَذِيْرٌ مُّبِيْنٌ ۝١٨٤** || **اَوَلَمْ يَتَفَكَّرُوْا ۫ مَا بِصَاحِبِهِمْ مِّنْ جِنَّةٍ ؕ اِنْ هُوَ اِلَّا نَذِيْرٌ مُّبِيْنٌ ۝١٨٤**
p ترجمہ: کیا انھوں نے غور نہیں کیا کہ ان کے صاحب کو کوئی جنون نہیں ہے وہ تو واضح طور پر ڈرانے والے ہیں۔ || Have they not reflected that there is no madness in their companion? He is only a clear warner. || ایا یې فکر نه دی کړی چې په ملګري یې لېونتوب نشته؟ هغه یوازې ښکاره وېروونکی دی.
p **اَوَلَمْ يَنْظُرُوْا فِيْ مَلَكُوْتِ السَّمٰوٰتِ وَالْاَرْضِ وَمَا خَلَقَ اللّٰهُ مِنْ شَيْءٍ ۙ** || **اَوَلَمْ يَنْظُرُوْا فِيْ مَلَكُوْتِ السَّمٰوٰتِ وَالْاَرْضِ وَمَا خَلَقَ اللّٰهُ مِنْ شَيْءٍ ۙ** || **اَوَلَمْ يَنْظُرُوْا فِيْ مَلَكُوْتِ السَّمٰوٰتِ وَالْاَرْضِ وَمَا خَلَقَ اللّٰهُ مِنْ شَيْءٍ ۙ**
p ترجمہ: کیا ان لوگوں نے غور نہیں کیا آسمانوں اور زمینوں کی بادشاہت میں اور جو شے بھی اللہ نے پیدا فرمائی ہے || Have they not looked into the kingdom of the heavens and the earth and whatever Allah has created, || ایا دوی د اسمانونو او ځمکې په پاچاهۍ او په هر هغه شي کې فکر نه دی کړی چې الله پیدا کړی
p **وَّاَنْ عَسٰٓى اَنْ يَّكُوْنَ قَدِ اقْتَرَبَ اَجَلُهُمْ ۚ فَبِاَيِّ حَدِيْثٍۭ بَعْدَهٗ يُؤْمِنُوْنَ ۝١٨٥** || **وَّاَنْ عَسٰٓى اَنْ يَّكُوْنَ قَدِ اقْتَرَبَ اَجَلُهُمْ ۚ فَبِاَيِّ حَدِيْثٍۭ بَعْدَهٗ يُؤْمِنُوْنَ ۝١٨٥** || **وَّاَنْ عَسٰٓى اَنْ يَّكُوْنَ قَدِ اقْتَرَبَ اَجَلُهُمْ ۚ فَبِاَيِّ حَدِيْثٍۭ بَعْدَهٗ يُؤْمِنُوْنَ ۝١٨٥**
p ترجمہ: اور (نہ اس بات پر غور کیا) کہ ہو سکتا ہے کہ ان کا (مقررہ) وقت قریب آ پہنچا ہو پھر اس (قرآن) کے بعد وہ کس بات پر ایمان لائیں گے؟ || and that perhaps their term has drawn near? In what message after this (Quran) will they believe? || او (په دې یې فکر نه دی کړی) چې کېدای شي (ټاکلی) وخت یې نږدې شوی وي؟ نو له دې (قرآن) وروسته به په کومه خبره ایمان راوړي؟
p **مَنْ يُّضْلِلِ اللّٰهُ فَلَا هَادِيَ لَهٗ ؕ وَيَذَرُهُمْ فِيْ طُغْيَانِهِمْ يَعْمَهُوْنَ ۝١٨٦** || **مَنْ يُّضْلِلِ اللّٰهُ فَلَا هَادِيَ لَهٗ ؕ وَيَذَرُهُمْ فِيْ طُغْيَانِهِمْ يَعْمَهُوْنَ ۝١٨٦** || **مَنْ يُّضْلِلِ اللّٰهُ فَلَا هَادِيَ لَهٗ ؕ وَيَذَرُهُمْ فِيْ طُغْيَانِهِمْ يَعْمَهُوْنَ ۝١٨٦**
p ترجمہ: جسے اللہ گم راہ کر دے تو اسے کوئی ہدایت دینے والا نہیں اور (اللہ) انھیں چھوڑ دیتا ہے تاکہ وہ اپنی سرکشی میں بھٹکتے رہیں۔ || Whomever Allah lets go astray has no guide, and He leaves them wandering blindly in their transgression. || چا ته چې الله ګمراه کړي، هېڅ لارښود نه لري؛ او (الله) پرېږدي یې چې په خپله سرکشۍ کې سرګردانه وي.
p **يَسْـَٔلُوْنَكَ عَنِ السَّاعَةِ اَيَّانَ مُرْسٰىهَا ؕ قُلْ اِنَّمَا عِلْمُهَا عِنْدَ رَبِّيْ ۚ** || **يَسْـَٔلُوْنَكَ عَنِ السَّاعَةِ اَيَّانَ مُرْسٰىهَا ؕ قُلْ اِنَّمَا عِلْمُهَا عِنْدَ رَبِّيْ ۚ** || **يَسْـَٔلُوْنَكَ عَنِ السَّاعَةِ اَيَّانَ مُرْسٰىهَا ؕ قُلْ اِنَّمَا عِلْمُهَا عِنْدَ رَبِّيْ ۚ**
p ترجمہ: وہ آپ سے قیامت کے بارے میں سوال کرتے ہیں کہ وہ کب قائم ہو گی؟ آپ (ﷺ) فرما دیجیے اس کا علم صرف میرے رب ہی کے پاس ہے || They ask you about the Hour: when will it come? Say (ﷺ): "Its knowledge is only with my Lord; || له تا د قیامت په اړه پوښتي چې کله به قایمېږي؟ ته (ﷺ) ووایه: "د هغې علم یوازې زما رب سره دی؛
p **لَا يُجَلِّيْهَا لِوَقْتِهَآ اِلَّا هُوَ ۘ ثَقُلَتْ فِي السَّمٰوٰتِ وَالْاَرْضِ ؕ لَا تَاْتِيْكُمْ اِلَّا بَغْتَةً ؕ** || **لَا يُجَلِّيْهَا لِوَقْتِهَآ اِلَّا هُوَ ۘ ثَقُلَتْ فِي السَّمٰوٰتِ وَالْاَرْضِ ؕ لَا تَاْتِيْكُمْ اِلَّا بَغْتَةً ؕ** || **لَا يُجَلِّيْهَا لِوَقْتِهَآ اِلَّا هُوَ ۘ ثَقُلَتْ فِي السَّمٰوٰتِ وَالْاَرْضِ ؕ لَا تَاْتِيْكُمْ اِلَّا بَغْتَةً ؕ**
p ترجمہ: وہی اسے اس کے وقت پر ظاہر فرمائے گا وہ (قیامت) آسمانوں اور زمینوں میں ایک بھاری بات ہے وہ تم پر اچانک آ جائے گی || none but He will reveal it at its time. It weighs heavily on the heavens and the earth; it will come upon you only suddenly." || یوازې هغه به یې په خپل وخت ښکاره کړي؛ (قیامت) په اسمانونو او ځمکه کې درنه خبره ده؛ ناڅاپه به درشي."
p **يَسْـَٔلُوْنَكَ كَاَنَّكَ حَفِيٌّ عَنْهَا ؕ قُلْ اِنَّمَا عِلْمُهَا عِنْدَ اللّٰهِ وَلٰكِنَّ اَكْثَرَ النَّاسِ لَا يَعْلَمُوْنَ ۝١٨٧** || **يَسْـَٔلُوْنَكَ كَاَنَّكَ حَفِيٌّ عَنْهَا ؕ قُلْ اِنَّمَا عِلْمُهَا عِنْدَ اللّٰهِ وَلٰكِنَّ اَكْثَرَ النَّاسِ لَا يَعْلَمُوْنَ ۝١٨٧** || **يَسْـَٔلُوْنَكَ كَاَنَّكَ حَفِيٌّ عَنْهَا ؕ قُلْ اِنَّمَا عِلْمُهَا عِنْدَ اللّٰهِ وَلٰكِنَّ اَكْثَرَ النَّاسِ لَا يَعْلَمُوْنَ ۝١٨٧**
p ترجمہ: وہ آپ سے (اس طرح) پوچھتے ہیں گویا آپ اس کی تلاش میں ہیں آپ (ﷺ) فرما دیجیے اس کا علم تو اللہ ہی کے پاس ہے لیکن اکثر لوگ نہیں جانتے۔ || They ask you as if you were eagerly searching for it. Say (ﷺ): "Its knowledge is with Allah alone, but most people do not know." || داسې پوښتنه درنه کوي لکه چې ته یې په لټه کې یې. ووایه (ﷺ): "د هغې علم یوازې الله سره دی، خو ډېری خلک نه پوهېږي."
