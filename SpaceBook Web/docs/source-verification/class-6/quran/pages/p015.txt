p **وَلَا تَتَّبِعُوْا مِنْ دُوْنِهٖٓ اَوْلِيَآءَ ؕ قَلِيْلًا مَّا تَذَكَّرُوْنَ ۝٣ وَكَمْ مِّنْ قَرْيَةٍ اَهْلَكْنٰهَا** || **وَلَا تَتَّبِعُوْا مِنْ دُوْنِهٖٓ اَوْلِيَآءَ ؕ قَلِيْلًا مَّا تَذَكَّرُوْنَ ۝٣ وَكَمْ مِّنْ قَرْيَةٍ اَهْلَكْنٰهَا** || **وَلَا تَتَّبِعُوْا مِنْ دُوْنِهٖٓ اَوْلِيَآءَ ؕ قَلِيْلًا مَّا تَذَكَّرُوْنَ ۝٣ وَكَمْ مِّنْ قَرْيَةٍ اَهْلَكْنٰهَا**
p ترجمہ: اور اس کو چھوڑ کر دوسرے حاکموں کی پیروی نہ کرو تم لوگ بہت ہی کم نصیحت حاصل کرتے ہو۔ اور کتنی ہی بستیاں تھیں ہم نے انھیں برباد کر دیا || and do not follow other masters besides Him; how little you take heed! And how many towns We destroyed! || او له هغه پرته د نورو واکمنانو پیروي مه کوئ؛ تاسو ډېر لږ نصیحت اخلئ. او څومره کلي وو چې موږ تباه کړل
p **فَجَآءَهَا بَاْسُنَا بَيَاتًا اَوْ هُمْ قَآئِلُوْنَ ۝٤ فَمَا كَانَ دَعْوٰىهُمْ اِذْ جَآءَهُمْ بَاْسُنَآ** || **فَجَآءَهَا بَاْسُنَا بَيَاتًا اَوْ هُمْ قَآئِلُوْنَ ۝٤ فَمَا كَانَ دَعْوٰىهُمْ اِذْ جَآءَهُمْ بَاْسُنَآ** || **فَجَآءَهَا بَاْسُنَا بَيَاتًا اَوْ هُمْ قَآئِلُوْنَ ۝٤ فَمَا كَانَ دَعْوٰىهُمْ اِذْ جَآءَهُمْ بَاْسُنَآ**
p ترجمہ: تو رات کے وقت اُن پر ہمارا عذاب آیا یا جب وہ دوپہر کو آرام کر رہے تھے۔ تو جب اُن پر ہمارا عذاب آیا تو اُن کی یہی پکار تھی || Our punishment came upon them by night, or while they were resting at midday. And when Our punishment came upon them, their only cry was || زموږ عذاب پرې د شپې راغی، یا کله چې د غرمې دمه کوله. کله چې زموږ عذاب راغی، چیغه یې یوازې دا وه
p **اِلَّآ اَنْ قَالُوْٓا اِنَّا كُنَّا ظٰلِمِيْنَ ۝٥ فَلَنَسْـَٔلَنَّ الَّذِيْنَ اُرْسِلَ اِلَيْهِمْ** || **اِلَّآ اَنْ قَالُوْٓا اِنَّا كُنَّا ظٰلِمِيْنَ ۝٥ فَلَنَسْـَٔلَنَّ الَّذِيْنَ اُرْسِلَ اِلَيْهِمْ** || **اِلَّآ اَنْ قَالُوْٓا اِنَّا كُنَّا ظٰلِمِيْنَ ۝٥ فَلَنَسْـَٔلَنَّ الَّذِيْنَ اُرْسِلَ اِلَيْهِمْ**
p ترجمہ: کہ وہ کہنے لگے بے شک ہم ہی ظالم تھے۔ تو ہم اُن سے ضرور پوچھیں گے جن کی طرف رسول بھیجے گئے || that they said, "Indeed, we were wrongdoers." We will surely question those to whom messengers were sent || چې ویې ویل: "بېشکه موږ ظالمان وو." موږ به خامخا له هغو وپوښتو چې رسولان ورته لېږل شوي وو
p **وَلَنَسْـَٔلَنَّ الْمُرْسَلِيْنَ ۝٦ فَلَنَقُصَّنَّ عَلَيْهِمْ بِعِلْمٍ** || **وَلَنَسْـَٔلَنَّ الْمُرْسَلِيْنَ ۝٦ فَلَنَقُصَّنَّ عَلَيْهِمْ بِعِلْمٍ** || **وَلَنَسْـَٔلَنَّ الْمُرْسَلِيْنَ ۝٦ فَلَنَقُصَّنَّ عَلَيْهِمْ بِعِلْمٍ**
p ترجمہ: اور ہم رسولوں سے (بھی) ضرور پوچھیں گے۔ پھر ہم ضرور اپنے علم سے انھیں سب کچھ بیان کر دیں گے || and We will surely question the messengers (too). Then We will surely relate everything to them with knowledge || او له رسولانو به (هم) خامخا وپوښتو. بیا به خامخا په خپل علم ټول څه ورته بیان کړو
p **وَّمَا كُنَّا غَآئِبِيْنَ ۝٧ وَالْوَزْنُ يَوْمَئِذِ ِۨالْحَقُّ ۚ** || **وَّمَا كُنَّا غَآئِبِيْنَ ۝٧ وَالْوَزْنُ يَوْمَئِذِ ِۨالْحَقُّ ۚ** || **وَّمَا كُنَّا غَآئِبِيْنَ ۝٧ وَالْوَزْنُ يَوْمَئِذِ ِۨالْحَقُّ ۚ**
p ترجمہ: اور ہم غائب نہیں تھے۔ اور (اعمال کا) تولنا اُس روز برحق ہے || for We were never absent. And the weighing (of deeds) on that Day is true || او موږ غایب نه وو. او (د اعمالو) تلل په هغه ورځ حق دي
p **فَمَنْ ثَقُلَتْ مَوَازِيْنُهٗ فَاُولٰٓئِكَ هُمُ الْمُفْلِحُوْنَ ۝٨ وَمَنْ خَفَّتْ مَوَازِيْنُهٗ** || **فَمَنْ ثَقُلَتْ مَوَازِيْنُهٗ فَاُولٰٓئِكَ هُمُ الْمُفْلِحُوْنَ ۝٨ وَمَنْ خَفَّتْ مَوَازِيْنُهٗ** || **فَمَنْ ثَقُلَتْ مَوَازِيْنُهٗ فَاُولٰٓئِكَ هُمُ الْمُفْلِحُوْنَ ۝٨ وَمَنْ خَفَّتْ مَوَازِيْنُهٗ**
p ترجمہ: لہٰذا جن کے (نیک اعمال کے) پلڑے بھاری ہوں گے تو وہی لوگ کامیاب ہونے والے ہیں۔ اور جن کے (نیک اعمال کے) پلڑے ہلکے ہوں گے || So those whose scales (of good deeds) are heavy — they are the successful. And those whose scales (of good deeds) are light || نو د چا (د نېکو اعمالو) پلې چې درنې وي، همدوی بریالي دي. او د چا پلې چې سپکې وي
p **فَاُولٰٓئِكَ الَّذِيْنَ خَسِرُوْٓا اَنْفُسَهُمْ بِمَا كَانُوْا بِاٰيٰتِنَا يَظْلِمُوْنَ ۝٩** || **فَاُولٰٓئِكَ الَّذِيْنَ خَسِرُوْٓا اَنْفُسَهُمْ بِمَا كَانُوْا بِاٰيٰتِنَا يَظْلِمُوْنَ ۝٩** || **فَاُولٰٓئِكَ الَّذِيْنَ خَسِرُوْٓا اَنْفُسَهُمْ بِمَا كَانُوْا بِاٰيٰتِنَا يَظْلِمُوْنَ ۝٩**
p ترجمہ: تو یہ وہ لوگ ہیں جنھوں نے اپنے آپ کو خسارے میں ڈالا اس وجہ سے کہ وہ ہماری آیات کے ساتھ ظلم کیا کرتے تھے۔ || those are the ones who have ruined themselves, because they used to wrong Our verses. || همدوی هغه دي چې ځانونه یې په تاوان کې واچول، ځکه چې زموږ له آیتونو سره یې ظلم کاوه.
p **وَلَقَدْ مَكَّنّٰكُمْ فِي الْاَرْضِ وَجَعَلْنَا لَكُمْ فِيْهَا مَعَايِشَ ؕ قَلِيْلًا مَّا تَشْكُرُوْنَ ۝١٠** || **وَلَقَدْ مَكَّنّٰكُمْ فِي الْاَرْضِ وَجَعَلْنَا لَكُمْ فِيْهَا مَعَايِشَ ؕ قَلِيْلًا مَّا تَشْكُرُوْنَ ۝١٠** || **وَلَقَدْ مَكَّنّٰكُمْ فِي الْاَرْضِ وَجَعَلْنَا لَكُمْ فِيْهَا مَعَايِشَ ؕ قَلِيْلًا مَّا تَشْكُرُوْنَ ۝١٠**
p ترجمہ: اور یقیناً ہم نے ہی تمھیں زمین میں آباد کیا اور اِس میں ہم نے تمھارے لیے سامانِ زندگی فراہم کیا (مگر) تم کم ہی شکر کرتے ہو۔ (رکوع 1) || And We have indeed settled you on the earth and provided you with the means of life in it; (yet) little do you give thanks. (End of ruku' 1) || او بېشکه موږ تاسو په ځمکه کې مېشت کړئ او ستاسو لپاره مو په کې د ژوند سامان برابر کړ؛ (خو) تاسو لږ شکر کوئ. (لومړی رکوع)
p **وَلَقَدْ خَلَقْنٰكُمْ ثُمَّ صَوَّرْنٰكُمْ ثُمَّ قُلْنَا لِلْمَلٰٓئِكَةِ اسْجُدُوْا لِاٰدَمَ ۖ** || **وَلَقَدْ خَلَقْنٰكُمْ ثُمَّ صَوَّرْنٰكُمْ ثُمَّ قُلْنَا لِلْمَلٰٓئِكَةِ اسْجُدُوْا لِاٰدَمَ ۖ** || **وَلَقَدْ خَلَقْنٰكُمْ ثُمَّ صَوَّرْنٰكُمْ ثُمَّ قُلْنَا لِلْمَلٰٓئِكَةِ اسْجُدُوْا لِاٰدَمَ ۖ**
p ترجمہ: اور یقیناً ہم نے تمھیں پیدا فرمایا پھر ہم نے تمھاری صورتیں بنائیں پھر ہم نے فرشتوں سے فرمایا تم سب آدم (علیہ السلام) کو سجدہ کرو || And We created you, then We shaped you, then We said to the angels, "Prostrate yourselves before Adam (peace be upon him)" || او بېشکه موږ تاسو پیدا کړئ، بیا مو بڼې درکړې، بیا مو پرښتو ته وویل: "آدم (علیه السلام) ته سجده وکړئ"
p **فَسَجَدُوْٓا اِلَّآ اِبْلِيْسَ ؕ لَمْ يَكُنْ مِّنَ السّٰجِدِيْنَ ۝١١ قَالَ مَا مَنَعَكَ اَلَّا تَسْجُدَ** || **فَسَجَدُوْٓا اِلَّآ اِبْلِيْسَ ؕ لَمْ يَكُنْ مِّنَ السّٰجِدِيْنَ ۝١١ قَالَ مَا مَنَعَكَ اَلَّا تَسْجُدَ** || **فَسَجَدُوْٓا اِلَّآ اِبْلِيْسَ ؕ لَمْ يَكُنْ مِّنَ السّٰجِدِيْنَ ۝١١ قَالَ مَا مَنَعَكَ اَلَّا تَسْجُدَ**
p ترجمہ: تو ابلیس کے سوا اُن سب نے سجدہ کیا وہ سجدہ کرنے والوں میں سے نہ ہوا۔ (اللہ نے) فرمایا کس چیز نے تجھے منع کیا کہ تو سجدہ نہ کرے || So they all prostrated, except Iblis; he was not among those who prostrated. p **اِذْ اَمَرْتُكَ ؕ قَالَ اَنَا خَيْرٌ مِّنْهُ ۚ خَلَقْتَنِيْ مِنْ نَّارٍ وَّخَلَقْتَهٗ مِنْ طِيْنٍ ۝١٢** || **اِذْ اَمَرْتُكَ ؕ قَالَ اَنَا خَيْرٌ مِّنْهُ ۚ خَلَقْتَنِيْ مِنْ نَّارٍ وَّخَلَقْتَهٗ مِنْ طِيْنٍ ۝١٢** || **اِذْ اَمَرْتُكَ ؕ قَالَ اَنَا خَيْرٌ مِّنْهُ ۚ خَلَقْتَنِيْ مِنْ نَّارٍ وَّخَلَقْتَهٗ مِنْ طِيْنٍ ۝١٢**
