p **مَنْ يَّسُوْمُهُمْ سُوْٓءَ الْعَذَابِ ؕ اِنَّ رَبَّكَ لَسَرِيْعُ الْعِقَابِ ۖ** || **مَنْ يَّسُوْمُهُمْ سُوْٓءَ الْعَذَابِ ؕ اِنَّ رَبَّكَ لَسَرِيْعُ الْعِقَابِ ۖ** || **مَنْ يَّسُوْمُهُمْ سُوْٓءَ الْعَذَابِ ؕ اِنَّ رَبَّكَ لَسَرِيْعُ الْعِقَابِ ۖ**
p ترجمہ: جو انھیں بُرا عذاب چکھائیں گے بے شک آپ کا رب ضرور جلد سزا دینے والا ہے || those who would inflict on them the worst punishment. Your Lord is swift in retribution, || چې بد عذاب به ورڅکوي؛ بېشکه ستا رب ژر سزا ورکوونکی دی
p **وَاِنَّهٗ لَغَفُوْرٌ رَّحِيْمٌ ۝١٦٧ وَقَطَّعْنٰهُمْ فِي الْاَرْضِ اُمَمًا ۚ مِنْهُمُ الصّٰلِحُوْنَ** || **وَاِنَّهٗ لَغَفُوْرٌ رَّحِيْمٌ ۝١٦٧ وَقَطَّعْنٰهُمْ فِي الْاَرْضِ اُمَمًا ۚ مِنْهُمُ الصّٰلِحُوْنَ** || **وَاِنَّهٗ لَغَفُوْرٌ رَّحِيْمٌ ۝١٦٧ وَقَطَّعْنٰهُمْ فِي الْاَرْضِ اُمَمًا ۚ مِنْهُمُ الصّٰلِحُوْنَ**
p ترجمہ: اور یقیناً وہ بڑا ہی بخشنے والا نہایت رحم فرمانے والا ہے۔ اور ہم نے انھیں زمین میں مختلف گروہوں میں تقسیم کر دیا ان میں سے کچھ نیک تھے || and He is Most Forgiving, Most Merciful. And We divided them on the earth into communities; some of them were righteous || او بېشکه هغه ډېر بښونکی او مهربان دی. او موږ هغوی په ځمکه کې په بېلابېلو ډلو ووېشل؛ ځینې یې نېکان وو
p **وَمِنْهُمْ دُوْنَ ذٰلِكَ ؗ وَبَلَوْنٰهُمْ بِالْحَسَنٰتِ وَالسَّيِّاٰتِ لَعَلَّهُمْ يَرْجِعُوْنَ ۝١٦٨** || **وَمِنْهُمْ دُوْنَ ذٰلِكَ ؗ وَبَلَوْنٰهُمْ بِالْحَسَنٰتِ وَالسَّيِّاٰتِ لَعَلَّهُمْ يَرْجِعُوْنَ ۝١٦٨** || **وَمِنْهُمْ دُوْنَ ذٰلِكَ ؗ وَبَلَوْنٰهُمْ بِالْحَسَنٰتِ وَالسَّيِّاٰتِ لَعَلَّهُمْ يَرْجِعُوْنَ ۝١٦٨**
p ترجمہ: اور کچھ اس کے علاوہ تھے اور ہم نے اچھے اور بُرے حالات کے ساتھ ان کی آزمائش کی تاکہ وہ (اللہ کی طرف) رجوع کریں۔ || and some were otherwise. And We tested them with good times and bad, so that they might return (to Allah). || او ځینې پرته له دې وو؛ او موږ په ښو او بدو حالاتو وازمویل چې (الله ته) رجوع وکړي.
p **فَخَلَفَ مِنْۢ بَعْدِهِمْ خَلْفٌ وَّرِثُوا الْكِتٰبَ يَاْخُذُوْنَ عَرَضَ هٰذَا الْاَدْنٰى وَيَقُوْلُوْنَ سَيُغْفَرُ لَنَا ۚ** || **فَخَلَفَ مِنْۢ بَعْدِهِمْ خَلْفٌ وَّرِثُوا الْكِتٰبَ يَاْخُذُوْنَ عَرَضَ هٰذَا الْاَدْنٰى وَيَقُوْلُوْنَ سَيُغْفَرُ لَنَا ۚ** || **فَخَلَفَ مِنْۢ بَعْدِهِمْ خَلْفٌ وَّرِثُوا الْكِتٰبَ يَاْخُذُوْنَ عَرَضَ هٰذَا الْاَدْنٰى وَيَقُوْلُوْنَ سَيُغْفَرُ لَنَا ۚ**
p ترجمہ: پھر ان کے بعد ایسے نالائق جانشین ہوئے جو کتاب کے وارث بنے وہ اس حقیر دنیا کا سامان ہی لیتے ہیں اور کہتے ہیں کہ ہمیں ضرور بخش دیا جائے گا || Then after them came successors who inherited the Book; they take the goods of this lowly world and say, "We will be forgiven." || بیا له دوی وروسته داسې نالایق ځایناستي راغلل چې د کتاب وارثان شول؛ د دې سپکې دنیا سامان اخلي او وايي: "خامخا به وبښل شو."
p **وَاِنْ يَّاْتِهِمْ عَرَضٌ مِّثْلُهٗ يَاْخُذُوْهُ ؕ اَلَمْ يُؤْخَذْ عَلَيْهِمْ مِّيْثَاقُ الْكِتٰبِ** || **وَاِنْ يَّاْتِهِمْ عَرَضٌ مِّثْلُهٗ يَاْخُذُوْهُ ؕ اَلَمْ يُؤْخَذْ عَلَيْهِمْ مِّيْثَاقُ الْكِتٰبِ** || **وَاِنْ يَّاْتِهِمْ عَرَضٌ مِّثْلُهٗ يَاْخُذُوْهُ ؕ اَلَمْ يُؤْخَذْ عَلَيْهِمْ مِّيْثَاقُ الْكِتٰبِ**
p ترجمہ: اور اس کے بعد اگر ان کے پاس اس جیسا اور سامان آ جائے تو اسے بھی لے لیں کیا ان سے تورات میں عہد نہیں لیا گیا تھا || And if similar goods come to them, they take them (too). Was not the covenant of the Book (the Torah) taken from them, || او که بیا ورته همداسې سامان راشي، هغه هم اخلي. ایا په تورات کې ترې ژمنه نه وه اخیستل شوې
p **اَنْ لَّا يَقُوْلُوْا عَلَى اللّٰهِ اِلَّا الْحَقَّ وَدَرَسُوْا مَا فِيْهِ ؕ** || **اَنْ لَّا يَقُوْلُوْا عَلَى اللّٰهِ اِلَّا الْحَقَّ وَدَرَسُوْا مَا فِيْهِ ؕ** || **اَنْ لَّا يَقُوْلُوْا عَلَى اللّٰهِ اِلَّا الْحَقَّ وَدَرَسُوْا مَا فِيْهِ ؕ**
p ترجمہ: کہ وہ اللہ کے بارے میں حق کے سوا کچھ اور نہ کہیں گے؟ اور انھوں نے پڑھ لیا ہے جو کچھ اس میں ہے || that they would say nothing about Allah but the truth? And they have studied what is in it. || چې د الله په اړه به پرته له حقه څه نه وايي؟ او هغه چې په کې دي، لوستي یې دي.
p **وَالدَّارُ الْاٰخِرَةُ خَيْرٌ لِّلَّذِيْنَ يَتَّقُوْنَ ؕ اَفَلَا تَعْقِلُوْنَ ۝١٦٩** || **وَالدَّارُ الْاٰخِرَةُ خَيْرٌ لِّلَّذِيْنَ يَتَّقُوْنَ ؕ اَفَلَا تَعْقِلُوْنَ ۝١٦٩** || **وَالدَّارُ الْاٰخِرَةُ خَيْرٌ لِّلَّذِيْنَ يَتَّقُوْنَ ؕ اَفَلَا تَعْقِلُوْنَ ۝١٦٩**
p ترجمہ: اور آخرت کا گھر ان لوگوں کے لیے بہتر ہے جو ڈرتے ہیں کیا تم عقل سے کام نہیں لیتے؟ || And the home of the Hereafter is better for those who are mindful. Will you not then understand? || او د آخرت کور د هغو لپاره غوره دی چې وېرېږي؛ ایا عقل نه کاروئ؟
p **وَالَّذِيْنَ يُمَسِّكُوْنَ بِالْكِتٰبِ وَاَقَامُوا الصَّلٰوةَ ؕ اِنَّا لَا نُضِيْعُ اَجْرَ الْمُصْلِحِيْنَ ۝١٧٠** || **وَالَّذِيْنَ يُمَسِّكُوْنَ بِالْكِتٰبِ وَاَقَامُوا الصَّلٰوةَ ؕ اِنَّا لَا نُضِيْعُ اَجْرَ الْمُصْلِحِيْنَ ۝١٧٠** || **وَالَّذِيْنَ يُمَسِّكُوْنَ بِالْكِتٰبِ وَاَقَامُوا الصَّلٰوةَ ؕ اِنَّا لَا نُضِيْعُ اَجْرَ الْمُصْلِحِيْنَ ۝١٧٠**
p ترجمہ: اور جو لوگ (اللہ کی) کتاب کو مضبوطی سے پکڑتے ہیں اور نماز قائم کرتے ہیں (تو) بے شک ہم اصلاح کرنے والوں کا اجر ضائع نہیں کرتے۔ || As for those who hold fast to the Book (of Allah) and establish prayer — We never waste the reward of those who set things right. || او هغه چې د (الله) کتاب په ټینګه نیسي او لمونځ قایموي — بېشکه موږ د سمونکو اجر نه ضایع کوو.
p **وَاِذْ نَتَقْنَا الْجَبَلَ فَوْقَهُمْ كَاَنَّهٗ ظُلَّةٌ وَّظَنُّوْٓا اَنَّهٗ وَاقِعٌۢ بِهِمْ ۚ** || **وَاِذْ نَتَقْنَا الْجَبَلَ فَوْقَهُمْ كَاَنَّهٗ ظُلَّةٌ وَّظَنُّوْٓا اَنَّهٗ وَاقِعٌۢ بِهِمْ ۚ** || **وَاِذْ نَتَقْنَا الْجَبَلَ فَوْقَهُمْ كَاَنَّهٗ ظُلَّةٌ وَّظَنُّوْٓا اَنَّهٗ وَاقِعٌۢ بِهِمْ ۚ**
p ترجمہ: اور (یاد کیجیے) جب ہم نے پہاڑ کو ان کے اوپر اٹھایا گویا وہ ایک سائبان ہو اور انھوں نے خیال کیا کہ وہ ان پر گرنے والا ہے || And (remember) when We raised the mountain above them as if it were a canopy, and they thought it would fall on them: || او (یاد کړه) کله چې موږ غر پر دوی پورته کړ لکه چې سیوری وي، او ویې ګڼل چې پرې لوېږي:
p **خُذُوْا مَآ اٰتَيْنٰكُمْ بِقُوَّةٍ وَّاذْكُرُوْا مَا فِيْهِ لَعَلَّكُمْ تَتَّقُوْنَ ۝١٧١** || **خُذُوْا مَآ اٰتَيْنٰكُمْ بِقُوَّةٍ وَّاذْكُرُوْا مَا فِيْهِ لَعَلَّكُمْ تَتَّقُوْنَ ۝١٧١** || **خُذُوْا مَآ اٰتَيْنٰكُمْ بِقُوَّةٍ وَّاذْكُرُوْا مَا فِيْهِ لَعَلَّكُمْ تَتَّقُوْنَ ۝١٧١**
p ترجمہ: (ہم نے کہا) جو ہم نے تمھیں (تورات) عطا فرمائی ہے اسے مضبوطی سے پکڑو اور جو کچھ اس میں ہے اسے یاد رکھو تاکہ تم بچ جاؤ۔ (رکوع 21) || "Hold firmly to what We have given you (the Torah) and remember what is in it, so that you may be saved." (End of ruku' 21) || (موږ وویل:) "هغه چې درکړی مو دی (تورات) په ټینګه یې ونیسئ او څه چې په کې دي، یاد یې ساتئ چې ځان وساتئ." (یوویشتم رکوع)
p **وَاِذْ اَخَذَ رَبُّكَ مِنْۢ بَنِيْٓ اٰدَمَ مِنْ ظُهُوْرِهِمْ ذُرِّيَّتَهُمْ وَاَشْهَدَهُمْ عَلٰٓى اَنْفُسِهِمْ ۚ** || **وَاِذْ اَخَذَ رَبُّكَ مِنْۢ بَنِيْٓ اٰدَمَ مِنْ ظُهُوْرِهِمْ ذُرِّيَّتَهُمْ وَاَشْهَدَهُمْ عَلٰٓى اَنْفُسِهِمْ ۚ** || **وَاِذْ اَخَذَ رَبُّكَ مِنْۢ بَنِيْٓ اٰدَمَ مِنْ ظُهُوْرِهِمْ ذُرِّيَّتَهُمْ وَاَشْهَدَهُمْ عَلٰٓى اَنْفُسِهِمْ ۚ**
p ترجمہ: اور جب آپ کے رب نے بنی آدم کی پشتوں سے ان کی نسل کو نکالا اور انھیں خود ان کی جانوں پر گواہ بنایا || And when your Lord brought forth from the loins of the children of Adam their descendants and made them testify about themselves, || او کله چې ستا رب د آدم د اولاد له شاګانو نسل یې راوویست او پر ځانونو یې شاهدان کړل:
p **اَلَسْتُ بِرَبِّكُمْ ؕ قَالُوْا بَلٰى ۛۚ شَهِدْنَا ۛۚ اَنْ تَقُوْلُوْا يَوْمَ الْقِيٰمَةِ** || **اَلَسْتُ بِرَبِّكُمْ ؕ قَالُوْا بَلٰى ۛۚ شَهِدْنَا ۛۚ اَنْ تَقُوْلُوْا يَوْمَ الْقِيٰمَةِ** || **اَلَسْتُ بِرَبِّكُمْ ؕ قَالُوْا بَلٰى ۛۚ شَهِدْنَا ۛۚ اَنْ تَقُوْلُوْا يَوْمَ الْقِيٰمَةِ**
p ترجمہ: (اور پوچھا) کیا میں تمھارا رب نہیں ہوں؟ انھوں نے جواب دیا کیوں نہیں! ہم (اس واقعے کے) گواہ ہیں (ایسا نہ ہو) کہ تم قیامت کے دن کہو || (asking,) "Am I not your Lord?" They said, "Yes indeed! We testify." (This) lest you say on the Day of Resurrection, || (ویې پوښتل:) "ایا زه ستاسو رب نه یم؟" ویې ویل: "ولې نه! موږ شاهدان یو." (داسې نه چې) د قیامت په ورځ ووایئ
