p **اَلَآ اِنَّمَا طٰٓئِرُهُمْ عِنْدَ اللّٰهِ وَلٰكِنَّ اَكْثَرَهُمْ لَا يَعْلَمُوْنَ ۝١٣١ وَقَالُوْا** || **اَلَآ اِنَّمَا طٰٓئِرُهُمْ عِنْدَ اللّٰهِ وَلٰكِنَّ اَكْثَرَهُمْ لَا يَعْلَمُوْنَ ۝١٣١ وَقَالُوْا** || **اَلَآ اِنَّمَا طٰٓئِرُهُمْ عِنْدَ اللّٰهِ وَلٰكِنَّ اَكْثَرَهُمْ لَا يَعْلَمُوْنَ ۝١٣١ وَقَالُوْا**
p ترجمہ: آگاہ رہو اللہ کے نزدیک اُن ہی کی نحوست (مقدر) ہے لیکن اُن میں سے اکثر لوگ نہیں جانتے۔ اور وہ لوگ کہنے لگے || Surely their (evil) fortune is with Allah, but most of them do not know. And they said, || خبر شئ! د دوی نحوست (برخلیک) د الله سره دی، خو ډېری یې نه پوهېږي. او ویې ویل:
p **مَهْمَا تَاْتِنَا بِهٖ مِنْ اٰيَةٍ لِّتَسْحَرَنَا بِهَا ۙ فَمَا نَحْنُ لَكَ بِمُؤْمِنِيْنَ ۝١٣٢** || **مَهْمَا تَاْتِنَا بِهٖ مِنْ اٰيَةٍ لِّتَسْحَرَنَا بِهَا ۙ فَمَا نَحْنُ لَكَ بِمُؤْمِنِيْنَ ۝١٣٢** || **مَهْمَا تَاْتِنَا بِهٖ مِنْ اٰيَةٍ لِّتَسْحَرَنَا بِهَا ۙ فَمَا نَحْنُ لَكَ بِمُؤْمِنِيْنَ ۝١٣٢**
p ترجمہ: تم ہمارے پاس جو بھی نشانی لاؤ تاکہ تم اس کے ذریعے ہم پر جادو کرو تو بھی ہم تم پر ایمان لانے والے نہیں ہیں۔ || "Whatever sign you bring us to bewitch us with, we will not believe in you." || "هر نښه چې راوړې چې پرې جادو راباندې وکړې، بیا هم پر تا ایمان نه راوړو."
p **فَاَرْسَلْنَا عَلَيْهِمُ الطُّوْفَانَ وَالْجَرَادَ وَالْقُمَّلَ وَالضَّفَادِعَ وَالدَّمَ اٰيٰتٍ مُّفَصَّلٰتٍ ۫ فَاسْتَكْبَرُوْا** || **فَاَرْسَلْنَا عَلَيْهِمُ الطُّوْفَانَ وَالْجَرَادَ وَالْقُمَّلَ وَالضَّفَادِعَ وَالدَّمَ اٰيٰتٍ مُّفَصَّلٰتٍ ۫ فَاسْتَكْبَرُوْا** || **فَاَرْسَلْنَا عَلَيْهِمُ الطُّوْفَانَ وَالْجَرَادَ وَالْقُمَّلَ وَالضَّفَادِعَ وَالدَّمَ اٰيٰتٍ مُّفَصَّلٰتٍ ۫ فَاسْتَكْبَرُوْا**
p ترجمہ: پھر ہم نے اُن پر طوفان اور ٹڈیاں اور جوئیں اور مینڈک اور خون (یہ) الگ الگ نشانیاں (بطور عذاب) بھیجیں تو انھوں نے تکبر کیا || So We sent upon them the flood, locusts, lice, frogs and blood as distinct signs (of punishment), but they were arrogant || نو موږ پرې توپان، ملخان، سپږې، چونګښې او وینه — جلا جلا نښې (د عذاب په توګه) — ولېږلې؛ خو هغوی کبر وکړ
p **وَكَانُوْا قَوْمًا مُّجْرِمِيْنَ ۝١٣٣ وَلَمَّا وَقَعَ عَلَيْهِمُ الرِّجْزُ قَالُوْا يٰمُوْسَى ادْعُ لَنَا رَبَّكَ** || **وَكَانُوْا قَوْمًا مُّجْرِمِيْنَ ۝١٣٣ وَلَمَّا وَقَعَ عَلَيْهِمُ الرِّجْزُ قَالُوْا يٰمُوْسَى ادْعُ لَنَا رَبَّكَ** || **وَكَانُوْا قَوْمًا مُّجْرِمِيْنَ ۝١٣٣ وَلَمَّا وَقَعَ عَلَيْهِمُ الرِّجْزُ قَالُوْا يٰمُوْسَى ادْعُ لَنَا رَبَّكَ**
p ترجمہ: اور وہ مجرم قوم تھی۔ اور جب اُن پر عذاب واقع ہوتا تو کہتے کہ اے موسیٰ! آپ دعا فرمائیں ہمارے لیے اپنے رب سے || and were a criminal people. Whenever the punishment fell upon them they said, "O Musa! Pray to your Lord for us || او مجرم قوم و. کله چې به عذاب پرې راغی، ویل به یې: "ای موسی! زموږ لپاره له خپل رب دعا وکړه
p **بِمَا عَهِدَ عِنْدَكَ ۚ لَئِنْ كَشَفْتَ عَنَّا الرِّجْزَ لَنُؤْمِنَنَّ لَكَ** || **بِمَا عَهِدَ عِنْدَكَ ۚ لَئِنْ كَشَفْتَ عَنَّا الرِّجْزَ لَنُؤْمِنَنَّ لَكَ** || **بِمَا عَهِدَ عِنْدَكَ ۚ لَئِنْ كَشَفْتَ عَنَّا الرِّجْزَ لَنُؤْمِنَنَّ لَكَ**
p ترجمہ: اس کے ذریعے سے جو اُس نے آپ سے عہد فرما رکھا ہے اگر آپ ہم سے اس عذاب کو ہٹا دیں گے تو ہم ضرور آپ پر ایمان لائیں گے || by the covenant He has made with you. If you remove the punishment from us, we will surely believe in you || د هغې ژمنې په وسیله چې له تا سره یې کړې؛ که دا عذاب له موږ لرې کړې، خامخا به پر تا ایمان راوړو
p **وَلَنُرْسِلَنَّ مَعَكَ بَنِيْٓ اِسْرَآءِيْلَ ۝١٣٤ فَلَمَّا كَشَفْنَا عَنْهُمُ الرِّجْزَ** || **وَلَنُرْسِلَنَّ مَعَكَ بَنِيْٓ اِسْرَآءِيْلَ ۝١٣٤ فَلَمَّا كَشَفْنَا عَنْهُمُ الرِّجْزَ** || **وَلَنُرْسِلَنَّ مَعَكَ بَنِيْٓ اِسْرَآءِيْلَ ۝١٣٤ فَلَمَّا كَشَفْنَا عَنْهُمُ الرِّجْزَ**
p ترجمہ: اور ہم ضرور آپ کے ساتھ بنی اسرائیل کو بھیج دیں گے۔ پھر جب ہم اُن سے عذاب ہٹا دیتے || and will surely send the Children of Israel with you." But whenever We removed the punishment from them || او بني اسرائیل به خامخا درسره ولېږو." خو کله چې به موږ عذاب ترې لرې کړ
p **اِلٰٓى اَجَلٍ هُمْ بٰلِغُوْهُ اِذَا هُمْ يَنْكُثُوْنَ ۝١٣٥ فَانْتَقَمْنَا مِنْهُمْ** || **اِلٰٓى اَجَلٍ هُمْ بٰلِغُوْهُ اِذَا هُمْ يَنْكُثُوْنَ ۝١٣٥ فَانْتَقَمْنَا مِنْهُمْ** || **اِلٰٓى اَجَلٍ هُمْ بٰلِغُوْهُ اِذَا هُمْ يَنْكُثُوْنَ ۝١٣٥ فَانْتَقَمْنَا مِنْهُمْ**
p ترجمہ: ایک مدت تک کے لیے جس کو وہ پہنچنے والے ہوتے تب وہ عہد توڑ دیتے۔ پھر ہم نے اُن سے بدلہ لیا || for a term they were to reach, at once they broke their word. So We took retribution from them || تر هغې مودې چې رسېدونکي وو، سمدستي به یې ژمنه ماته کړه. نو موږ ترې غچ واخیست
p **فَاَغْرَقْنٰهُمْ فِي الْيَمِّ بِاَنَّهُمْ كَذَّبُوْا بِاٰيٰتِنَا وَكَانُوْا عَنْهَا غٰفِلِيْنَ ۝١٣٦** || **فَاَغْرَقْنٰهُمْ فِي الْيَمِّ بِاَنَّهُمْ كَذَّبُوْا بِاٰيٰتِنَا وَكَانُوْا عَنْهَا غٰفِلِيْنَ ۝١٣٦** || **فَاَغْرَقْنٰهُمْ فِي الْيَمِّ بِاَنَّهُمْ كَذَّبُوْا بِاٰيٰتِنَا وَكَانُوْا عَنْهَا غٰفِلِيْنَ ۝١٣٦**
p ترجمہ: تو ہم نے انھیں سمندر میں غرق کر دیا اس وجہ سے کہ انھوں نے ہماری آیتوں کو جھٹلایا اور وہ اُن سے غافل تھے۔ || and drowned them in the sea, because they denied Our signs and were heedless of them. || او په سمندر کې مو ډوب کړل، ځکه چې زموږ آیتونه یې دروغ ګڼل او ترې غافل وو.
p **وَاَوْرَثْنَا الْقَوْمَ الَّذِيْنَ كَانُوْا يُسْتَضْعَفُوْنَ مَشَارِقَ الْاَرْضِ وَمَغَارِبَهَا الَّتِيْ بٰرَكْنَا فِيْهَا ؕ** || **وَاَوْرَثْنَا الْقَوْمَ الَّذِيْنَ كَانُوْا يُسْتَضْعَفُوْنَ مَشَارِقَ الْاَرْضِ وَمَغَارِبَهَا الَّتِيْ بٰرَكْنَا فِيْهَا ؕ** || **وَاَوْرَثْنَا الْقَوْمَ الَّذِيْنَ كَانُوْا يُسْتَضْعَفُوْنَ مَشَارِقَ الْاَرْضِ وَمَغَارِبَهَا الَّتِيْ بٰرَكْنَا فِيْهَا ؕ**
p ترجمہ: اور ہم نے اس قوم کو وارث بنا دیا جو کمزور کر دی گئی تھی سرزمین مشرق و مغرب کا جس میں ہم نے برکت رکھی تھی || And We made the people who had been oppressed inherit the east and west of the land which We had blessed. || او هغه قوم چې کمزوری کړل شوی و، د هغې ځمکې د ختیځ او لوېدیځ وارث مو کړ چې برکت مو پکې ایښی و.
p **وَتَمَّتْ كَلِمَتُ رَبِّكَ الْحُسْنٰى عَلٰى بَنِيْٓ اِسْرَآءِيْلَ ۙ بِمَا صَبَرُوْا ؕ** || **وَتَمَّتْ كَلِمَتُ رَبِّكَ الْحُسْنٰى عَلٰى بَنِيْٓ اِسْرَآءِيْلَ ۙ بِمَا صَبَرُوْا ؕ** || **وَتَمَّتْ كَلِمَتُ رَبِّكَ الْحُسْنٰى عَلٰى بَنِيْٓ اِسْرَآءِيْلَ ۙ بِمَا صَبَرُوْا ؕ**
p ترجمہ: اور آپ کے رب کا اچھا وعدہ پورا ہو گیا بنی اسرائیل کے بارے میں اس وجہ سے کہ انھوں نے صبر کیا تھا || And the good word (promise) of your Lord was fulfilled for the Children of Israel because they were patient. || او ستا د رب ښه ژمنه د بني اسرائیلو په اړه پوره شوه، ځکه چې صبر یې کړی و.
p **وَدَمَّرْنَا مَا كَانَ يَصْنَعُ فِرْعَوْنُ وَقَوْمُهٗ وَمَا كَانُوْا يَعْرِشُوْنَ ۝١٣٧** || **وَدَمَّرْنَا مَا كَانَ يَصْنَعُ فِرْعَوْنُ وَقَوْمُهٗ وَمَا كَانُوْا يَعْرِشُوْنَ ۝١٣٧** || **وَدَمَّرْنَا مَا كَانَ يَصْنَعُ فِرْعَوْنُ وَقَوْمُهٗ وَمَا كَانُوْا يَعْرِشُوْنَ ۝١٣٧**
p ترجمہ: اور ہم نے ان (محلات) کو تباہ کر دیا جو فرعون اور اس کی قوم نے بنا رکھے تھے اور (انگور کے باغات کو بھی) جو وہ بلندی پر چڑھاتے تھے۔ || And We destroyed what Pharaoh and his people had built and what (vineyards) they had raised up. || او هغه (ماڼۍ) مو ورانې کړې چې فرعون او قوم یې جوړې کړې وې، او هغه (د انګورو باغونه) چې لوړول یې.
p **وَجٰوَزْنَا بِبَنِيْٓ اِسْرَآءِيْلَ الْبَحْرَ فَاَتَوْا عَلٰى قَوْمٍ يَّعْكُفُوْنَ عَلٰٓى اَصْنَامٍ لَّهُمْ ۚ قَالُوْا** || **وَجٰوَزْنَا بِبَنِيْٓ اِسْرَآءِيْلَ الْبَحْرَ فَاَتَوْا عَلٰى قَوْمٍ يَّعْكُفُوْنَ عَلٰٓى اَصْنَامٍ لَّهُمْ ۚ قَالُوْا** || **وَجٰوَزْنَا بِبَنِيْٓ اِسْرَآءِيْلَ الْبَحْرَ فَاَتَوْا عَلٰى قَوْمٍ يَّعْكُفُوْنَ عَلٰٓى اَصْنَامٍ لَّهُمْ ۚ قَالُوْا**
p ترجمہ: اور ہم نے بنی اسرائیل کو سمندر کے پار کرایا تو وہ (ایسی) قوم کے پاس پہنچے جو اپنے بتوں کی پوجا میں لگے ہوئے تھے انھوں نے کہا || And We brought the Children of Israel across the sea, and they came upon a people devoted to their idols. They said, || او بني اسرائیل مو له سمندر واړول، نو داسې قوم ته ورسېدل چې د خپلو بتانو په عبادت بوخت وو. ویې ویل:
