p **فَاخْرُجْ اِنَّكَ مِنَ الصّٰغِرِيْنَ ۝١٣ قَالَ اَنْظِرْنِيْٓ اِلٰى يَوْمِ** || **فَاخْرُجْ اِنَّكَ مِنَ الصّٰغِرِيْنَ ۝١٣ قَالَ اَنْظِرْنِيْٓ اِلٰى يَوْمِ** || **فَاخْرُجْ اِنَّكَ مِنَ الصّٰغِرِيْنَ ۝١٣ قَالَ اَنْظِرْنِيْٓ اِلٰى يَوْمِ**
p ترجمہ: لہٰذا تو نکل جا بے شک تُو ذلیلوں میں سے ہے۔ (ابلیس نے) کہا تو مجھے (زندہ رہنے کی) مہلت دے اُس دن تک کے لیے || So get out; you are surely among the humiliated." (Iblis) said, "Then grant me respite (to live) until the Day || نو ووځه، بېشکه ته له ذلیلانو یې." (ابلیس) وویل: "نو ما ته (د ژوند) مهلت راکړه تر هغې ورځې
p **يُبْعَثُوْنَ ۝١٤ قَالَ اِنَّكَ مِنَ الْمُنْظَرِيْنَ ۝١٥** || **يُبْعَثُوْنَ ۝١٤ قَالَ اِنَّكَ مِنَ الْمُنْظَرِيْنَ ۝١٥** || **يُبْعَثُوْنَ ۝١٤ قَالَ اِنَّكَ مِنَ الْمُنْظَرِيْنَ ۝١٥**
p ترجمہ: جس دن لوگ (قبروں سے) اُٹھائے جائیں گے۔ (اللہ نے) فرمایا بے شک تو مہلت دیے جانے والوں میں سے ہے۔ || when people will be raised (from their graves)." (Allah) said, "You are surely among those given respite." || چې خلک به (له قبرونو) راپورته کېږي." (الله) وفرمایل: "بېشکه ته له مهلت ورکړل شویو یې."
p **قَالَ فَبِمَآ اَغْوَيْتَنِيْ لَاَقْعُدَنَّ لَهُمْ صِرَاطَكَ الْمُسْتَقِيْمَ ۝١٦** || **قَالَ فَبِمَآ اَغْوَيْتَنِيْ لَاَقْعُدَنَّ لَهُمْ صِرَاطَكَ الْمُسْتَقِيْمَ ۝١٦** || **قَالَ فَبِمَآ اَغْوَيْتَنِيْ لَاَقْعُدَنَّ لَهُمْ صِرَاطَكَ الْمُسْتَقِيْمَ ۝١٦**
p ترجمہ: (ابلیس نے) کہا تو (مجھے قسم ہے) جیسا کہ تُو نے مجھے گم راہ کر دیا میں (بھی) ان کی تاک میں ضرور بیٹھوں گا تیرے سیدھے راستے پر۔ || (Iblis) said, "Then (I swear), because You have led me astray, I will surely lie in wait for them on Your straight path. || (ابلیس) وویل: "نو (قسم دی) لکه څنګه چې تا زه ګمراه کړم، زه به (هم) خامخا ستا پر سمه لار د دوی په څار کې کېنم.
p **ثُمَّ لَاٰتِيَنَّهُمْ مِّنْۢ بَيْنِ اَيْدِيْهِمْ وَمِنْ خَلْفِهِمْ وَعَنْ اَيْمَانِهِمْ وَعَنْ شَمَآئِلِهِمْ ؕ** || **ثُمَّ لَاٰتِيَنَّهُمْ مِّنْۢ بَيْنِ اَيْدِيْهِمْ وَمِنْ خَلْفِهِمْ وَعَنْ اَيْمَانِهِمْ وَعَنْ شَمَآئِلِهِمْ ؕ** || **ثُمَّ لَاٰتِيَنَّهُمْ مِّنْۢ بَيْنِ اَيْدِيْهِمْ وَمِنْ خَلْفِهِمْ وَعَنْ اَيْمَانِهِمْ وَعَنْ شَمَآئِلِهِمْ ؕ**
p ترجمہ: پھر میں ضرور اُن کے پاس آؤں گا اُن کے سامنے سے اور اُن کے پیچھے سے اور اُن کے دائیں سے اور اُن کے بائیں سے || Then I will surely come at them from before them and from behind them, from their right and from their left, || بیا به خامخا ورځم، له مخې، له شا، له ښي او له کیڼ اړخه
p **وَلَا تَجِدُ اَكْثَرَهُمْ شٰكِرِيْنَ ۝١٧ قَالَ اخْرُجْ مِنْهَا مَذْءُوْمًا مَّدْحُوْرًا ؕ** || **وَلَا تَجِدُ اَكْثَرَهُمْ شٰكِرِيْنَ ۝١٧ قَالَ اخْرُجْ مِنْهَا مَذْءُوْمًا مَّدْحُوْرًا ؕ** || **وَلَا تَجِدُ اَكْثَرَهُمْ شٰكِرِيْنَ ۝١٧ قَالَ اخْرُجْ مِنْهَا مَذْءُوْمًا مَّدْحُوْرًا ؕ**
p ترجمہ: اور تُو اُن کی اکثریت کو شکر کرنے والا نہیں پائے گا۔ (اللہ نے) فرمایا تُو یہاں سے ذلیل و رُسوا ہو کر نکل جا || and You will not find most of them grateful." (Allah) said, "Get out of here, disgraced and banished. || او ډېری به یې شکر کوونکي ونه مومې." (الله) وفرمایل: "له دې ځایه ذلیل او رټل شوی ووځه.
p **لَمَنْ تَبِعَكَ مِنْهُمْ لَاَمْلَئَنَّ جَهَنَّمَ مِنْكُمْ اَجْمَعِيْنَ ۝١٨** || **لَمَنْ تَبِعَكَ مِنْهُمْ لَاَمْلَئَنَّ جَهَنَّمَ مِنْكُمْ اَجْمَعِيْنَ ۝١٨** || **لَمَنْ تَبِعَكَ مِنْهُمْ لَاَمْلَئَنَّ جَهَنَّمَ مِنْكُمْ اَجْمَعِيْنَ ۝١٨**
p ترجمہ: یقیناً اِن میں سے جو تیری پیروی کرے گا (تو وہ تیرا ساتھی ہوگا) (اور) میں تم سب سے ضرور جہنم کو بھر دوں گا۔ || Whoever of them follows you (will be your companion), and I will surely fill Hell with all of you." || له دوی څوک چې ستا پیروي وکړي (ستا ملګری به وي)، او زه به خامخا دوزخ له تاسو ټولو ډک کړم."
p **وَيٰٓاٰدَمُ اسْكُنْ اَنْتَ وَزَوْجُكَ الْجَنَّةَ فَكُلَا مِنْ حَيْثُ شِئْتُمَا** || **وَيٰٓاٰدَمُ اسْكُنْ اَنْتَ وَزَوْجُكَ الْجَنَّةَ فَكُلَا مِنْ حَيْثُ شِئْتُمَا** || **وَيٰٓاٰدَمُ اسْكُنْ اَنْتَ وَزَوْجُكَ الْجَنَّةَ فَكُلَا مِنْ حَيْثُ شِئْتُمَا**
p ترجمہ: اور اے آدم! تم اور تمھاری بیوی جنت میں رہو پھر تم دونوں کھاؤ جہاں سے تم دونوں چاہو || "And O Adam, dwell, you and your wife, in Paradise, and eat, both of you, from wherever you wish, || "او ای آدمه! ته او مېرمن دې په جنت کې اوسېږئ او له هر ځایه چې مو خوښه وي، وخورئ
p **وَلَا تَقْرَبَا هٰذِهِ الشَّجَرَةَ فَتَكُوْنَا مِنَ الظّٰلِمِيْنَ ۝١٩ فَوَسْوَسَ لَهُمَا الشَّيْطٰنُ** || **وَلَا تَقْرَبَا هٰذِهِ الشَّجَرَةَ فَتَكُوْنَا مِنَ الظّٰلِمِيْنَ ۝١٩ فَوَسْوَسَ لَهُمَا الشَّيْطٰنُ** || **وَلَا تَقْرَبَا هٰذِهِ الشَّجَرَةَ فَتَكُوْنَا مِنَ الظّٰلِمِيْنَ ۝١٩ فَوَسْوَسَ لَهُمَا الشَّيْطٰنُ**
p ترجمہ: اور تم دونوں اس درخت کے قریب نہ جانا ورنہ تم دونوں نقصان کاروں میں سے ہو جاؤ گے۔ پھر شیطان نے اُن دونوں (کے دلوں) میں وسوسہ ڈالا || but do not go near this tree, or you will both be among the wrongdoers." Then Satan whispered to them (in their hearts) || خو دې ونې ته مه ورنږدې کېږئ، ګنې له تاوانیانو به شئ." بیا شیطان د دواړو (په زړونو) کې وسوسه واچوله
p **لِيُبْدِيَ لَهُمَا مَا وٗرِيَ عَنْهُمَا مِنْ سَوْاٰتِهِمَا وَقَالَ** || **لِيُبْدِيَ لَهُمَا مَا وٗرِيَ عَنْهُمَا مِنْ سَوْاٰتِهِمَا وَقَالَ** || **لِيُبْدِيَ لَهُمَا مَا وٗرِيَ عَنْهُمَا مِنْ سَوْاٰتِهِمَا وَقَالَ**
p ترجمہ: تاکہ وہ اُن دونوں پر ظاہر کر دے اُن کے شرم کے مقامات جو اُن سے چھپائے گئے تھے اور (ابلیس نے) کہا || to expose to them their private parts that had been hidden from them, and (Iblis) said, || چې د دواړو هغه د شرم ځایونه ورته ښکاره کړي چې ترې پټ وو، او (ابلیس) وویل:
p **مَا نَهٰىكُمَا رَبُّكُمَا عَنْ هٰذِهِ الشَّجَرَةِ اِلَّآ اَنْ تَكُوْنَا مَلَكَيْنِ** || **مَا نَهٰىكُمَا رَبُّكُمَا عَنْ هٰذِهِ الشَّجَرَةِ اِلَّآ اَنْ تَكُوْنَا مَلَكَيْنِ** || **مَا نَهٰىكُمَا رَبُّكُمَا عَنْ هٰذِهِ الشَّجَرَةِ اِلَّآ اَنْ تَكُوْنَا مَلَكَيْنِ**
p ترجمہ: (اے آدم و حوا) تمھارے رب نے تم دونوں کو صرف اس لیے اِس درخت سے روکا کہ کہیں تم دونوں فرشتے نہ بن جاؤ || "(O Adam and Hawwa,) your Lord forbade you this tree only lest you become angels || "(ای آدمه او حوا!) ستاسو رب تاسو له دې ونې یوازې ځکه منع کړئ چې پرښتې نه شئ
p **اَوْ تَكُوْنَا مِنَ الْخٰلِدِيْنَ ۝٢٠ وَقَاسَمَهُمَآ اِنِّيْ لَكُمَا لَمِنَ النّٰصِحِيْنَ ۝٢١** || **اَوْ تَكُوْنَا مِنَ الْخٰلِدِيْنَ ۝٢٠ وَقَاسَمَهُمَآ اِنِّيْ لَكُمَا لَمِنَ النّٰصِحِيْنَ ۝٢١** || **اَوْ تَكُوْنَا مِنَ الْخٰلِدِيْنَ ۝٢٠ وَقَاسَمَهُمَآ اِنِّيْ لَكُمَا لَمِنَ النّٰصِحِيْنَ ۝٢١**
p ترجمہ: یا ہمیشہ رہنے والوں میں سے نہ ہو جاؤ۔ اور اس نے اُن دونوں سے قسم کھائی یقیناً میں تمھارے خیر خواہوں میں سے ہوں۔ || or become immortal." And he swore to them, "I am truly one of your sincere advisers." || یا تل پاتې کېدونکي نه شئ." او دواړو ته یې قسم وخوړ: "بېشکه زه ستاسو له خیرخواهانو یم."
p **فَدَلّٰىهُمَا بِغُرُوْرٍ ۚ فَلَمَّا ذَاقَا الشَّجَرَةَ** || **فَدَلّٰىهُمَا بِغُرُوْرٍ ۚ فَلَمَّا ذَاقَا الشَّجَرَةَ** || **فَدَلّٰىهُمَا بِغُرُوْرٍ ۚ فَلَمَّا ذَاقَا الشَّجَرَةَ**
p ترجمہ: تو اُس نے اُن دونوں کو دھوکا سے (پھل کھانے پر) مائل کر لیا پھر جب اُن دونوں نے اُس درخت (کے پھل) کو چکھا || So he lured them by deception (to eat the fruit). Then, when they tasted (the fruit of) the tree, || نو دواړه یې په دوکه (مېوې خوړلو ته) مایل کړل. بیا چې دواړو د هغې ونې (مېوه) وڅکله
