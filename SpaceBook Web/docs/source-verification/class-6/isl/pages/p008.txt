p **فَوَسَطْنَ بِهٖ جَمْعًا ﴿٥﴾ اِنَّ الْاِنْسَانَ لِرَبِّهٖ لَكَنُوْدٌ ﴿٦﴾ وَاِنَّهٗ عَلٰى ذٰلِكَ لَشَهِيْدٌ ﴿٧﴾ وَاِنَّهٗ لِحُبِّ الْخَيْرِ لَشَدِيْدٌ ﴿٨﴾ اَفَلَا يَعْلَمُ اِذَا بُعْثِرَ مَا فِي الْقُبُوْرِ ﴿٩﴾ وَحُصِّلَ مَا فِي الصُّدُوْرِ ﴿١٠﴾ اِنَّ رَبَّهُمْ بِهِمْ يَوْمَئِذٍ لَّخَبِيْرٌ ﴿١١﴾** || **فَوَسَطْنَ بِهٖ جَمْعًا ﴿٥﴾ اِنَّ الْاِنْسَانَ لِرَبِّهٖ لَكَنُوْدٌ ﴿٦﴾ وَاِنَّهٗ عَلٰى ذٰلِكَ لَشَهِيْدٌ ﴿٧﴾ وَاِنَّهٗ لِحُبِّ الْخَيْرِ لَشَدِيْدٌ ﴿٨﴾ اَفَلَا يَعْلَمُ اِذَا بُعْثِرَ مَا فِي الْقُبُوْرِ ﴿٩﴾ وَحُصِّلَ مَا فِي الصُّدُوْرِ ﴿١٠﴾ اِنَّ رَبَّهُمْ بِهِمْ يَوْمَئِذٍ لَّخَبِيْرٌ ﴿١١﴾** || **فَوَسَطْنَ بِهٖ جَمْعًا ﴿٥﴾ اِنَّ الْاِنْسَانَ لِرَبِّهٖ لَكَنُوْدٌ ﴿٦﴾ وَاِنَّهٗ عَلٰى ذٰلِكَ لَشَهِيْدٌ ﴿٧﴾ وَاِنَّهٗ لِحُبِّ الْخَيْرِ لَشَدِيْدٌ ﴿٨﴾ اَفَلَا يَعْلَمُ اِذَا بُعْثِرَ مَا فِي الْقُبُوْرِ ﴿٩﴾ وَحُصِّلَ مَا فِي الصُّدُوْرِ ﴿١٠﴾ اِنَّ رَبَّهُمْ بِهِمْ يَوْمَئِذٍ لَّخَبِيْرٌ ﴿١١﴾**
p *مفہوم:* پھر اسی وقت (دشمن کے) لشکر کے بیچ جا گھستے ہیں (5)۔ بے شک انسان اپنے رب کا بڑا ناشکرا ہے (6)، اور وہ خود اس پر گواہ ہے (7)، اور بے شک وہ مال کی محبت میں بہت سخت ہے (8)۔ کیا وہ نہیں جانتا کہ جب قبروں میں جو کچھ ہے نکال لیا جائے گا (9)، اور سینوں میں جو کچھ ہے ظاہر کر دیا جائے گا (10)، بے شک ان کا رب اس دن ان (کے حال) سے پوری طرح باخبر ہوگا (11)۔ || *Meaning:* then plunging with it into the midst of a host (5). Surely man is most ungrateful to his Lord (6), and he himself is a witness to that (7), and he is intense in his love of wealth (8). Does he not know — when what is in the graves is brought out (9), and what is in the hearts is laid bare (10) — that their Lord on that Day will be fully aware of them? (11) || *مانا:* بیا په هماغه وخت د (دښمن) لښکر منځ ته ننوځي (5). بېشکه انسان د خپل رب ډېر ناشکره دی (6)، او هغه پخپله پر دې شاهد دی (7)، او بېشکه هغه د مال په مینه کې ډېر سخت دی (8). ایا نه پوهېږي چې کله په قبرونو کې څه دي راوایستل شي (9)، او په سینو کې څه دي ښکاره شي (10)، بېشکه د هغوی رب به په هغه ورځ د هغوی (له حاله) پوره خبر وي (11).
### سُوْرَةُ الْقَارِعَةِ (مَكِّيَّة) || Surah Al-Qari'ah (Makki) || سورة القارعه (مکي)
p **بِسْمِ اللّٰهِ الرَّحْمٰنِ الرَّحِيْمِ** || **بِسْمِ اللّٰهِ الرَّحْمٰنِ الرَّحِيْمِ** || **بِسْمِ اللّٰهِ الرَّحْمٰنِ الرَّحِيْمِ**
p **اَلْقَارِعَةُ ﴿١﴾ مَا الْقَارِعَةُ ﴿٢﴾ وَمَا اَدْرٰىكَ مَا الْقَارِعَةُ ﴿٣﴾ يَوْمَ يَكُوْنُ النَّاسُ كَالْفَرَاشِ الْمَبْثُوْثِ ﴿٤﴾ وَتَكُوْنُ الْجِبَالُ كَالْعِهْنِ الْمَنْفُوْشِ ﴿٥﴾ فَاَمَّا مَنْ ثَقُلَتْ مَوَازِيْنُهٗ ﴿٦﴾ فَهُوَ فِيْ عِيْشَةٍ رَّاضِيَةٍ ﴿٧﴾ وَاَمَّا مَنْ خَفَّتْ مَوَازِيْنُهٗ ﴿٨﴾ فَاُمُّهٗ هَاوِيَةٌ ﴿٩﴾ وَمَا اَدْرٰىكَ مَا هِيَهْ ﴿١٠﴾ نَارٌ حَامِيَةٌ ﴿١١﴾** || **اَلْقَارِعَةُ ﴿١﴾ مَا الْقَارِعَةُ ﴿٢﴾ وَمَا اَدْرٰىكَ مَا الْقَارِعَةُ ﴿٣﴾ يَوْمَ يَكُوْنُ النَّاسُ كَالْفَرَاشِ الْمَبْثُوْثِ ﴿٤﴾ وَتَكُوْنُ الْجِبَالُ كَالْعِهْنِ الْمَنْفُوْشِ ﴿٥﴾ فَاَمَّا مَنْ ثَقُلَتْ مَوَازِيْنُهٗ ﴿٦﴾ فَهُوَ فِيْ عِيْشَةٍ رَّاضِيَةٍ ﴿٧﴾ وَاَمَّا مَنْ خَفَّتْ مَوَازِيْنُهٗ ﴿٨﴾ فَاُمُّهٗ هَاوِيَةٌ ﴿٩﴾ وَمَا اَدْرٰىكَ مَا هِيَهْ ﴿١٠﴾ نَارٌ حَامِيَةٌ ﴿١١﴾** || **اَلْقَارِعَةُ ﴿١﴾ مَا الْقَارِعَةُ ﴿٢﴾ وَمَا اَدْرٰىكَ مَا الْقَارِعَةُ ﴿٣﴾ يَوْمَ يَكُوْنُ النَّاسُ كَالْفَرَاشِ الْمَبْثُوْثِ ﴿٤﴾ وَتَكُوْنُ الْجِبَالُ كَالْعِهْنِ الْمَنْفُوْشِ ﴿٥﴾ فَاَمَّا مَنْ ثَقُلَتْ مَوَازِيْنُهٗ ﴿٦﴾ فَهُوَ فِيْ عِيْشَةٍ رَّاضِيَةٍ ﴿٧﴾ وَاَمَّا مَنْ خَفَّتْ مَوَازِيْنُهٗ ﴿٨﴾ فَاُمُّهٗ هَاوِيَةٌ ﴿٩﴾ وَمَا اَدْرٰىكَ مَا هِيَهْ ﴿١٠﴾ نَارٌ حَامِيَةٌ ﴿١١﴾**
p *مفہوم:* کھڑکھڑانے والی (1)! کیا ہے وہ کھڑکھڑانے والی؟ (2) اور تمھیں کیا معلوم کہ کھڑکھڑانے والی کیا ہے؟ (3) جس دن لوگ بکھرے ہوئے پروانوں کی طرح ہوں گے (4)، اور پہاڑ دھنکی ہوئی رنگین اون کی طرح ہو جائیں گے (5)۔ پھر جس کے (نیکیوں کے) پلڑے بھاری ہوں گے (6)، وہ پسندیدہ زندگی میں ہوگا (7)، اور جس کے پلڑے ہلکے ہوں گے (8)، اس کا ٹھکانا ہاویہ (گہرا گڑھا) ہوگا (9)۔ اور تمھیں کیا معلوم وہ کیا ہے؟ (10) دہکتی ہوئی آگ (11)۔ || *Meaning:* The Striking Calamity (1)! What is the Striking Calamity? (2) And what will make you know what the Striking Calamity is? (3) The Day when people will be like scattered moths (4), and the mountains will be like carded wool (5). Then as for him whose scales (of good deeds) are heavy (6), he will be in a pleasing life (7). But as for him whose scales are light (8), his abode will be the Pit (Hawiyah) (9). And what will make you know what that is? (10) A blazing Fire (11). || *مانا:* ټکوونکې پېښه (1)! څه ده هغه ټکوونکې پېښه؟ (2) او ته څه پوهېږې چې ټکوونکې پېښه څه ده؟ (3) هغه ورځ چې خلک به د خپرو شوو پتنګانو په څېر وي (4)، او غرونه به د ټکول شوې رنګینې وړۍ په څېر شي (5). نو د چا چې (د نېکیو) تلې درنې شوې (6)، هغه به په خوښ ژوند کې وي (7)، او د چا چې تلې سپکې شوې (8)، ځای به یې هاویه (ژوره کنده) وي (9). او ته څه پوهېږې چې هغه څه ده؟ (10) بله اور (11).
### سُوْرَةُ التَّكَاثُرِ (مَكِّيَّة) || Surah At-Takathur (Makki) || سورة التکاثر (مکي)
p **بِسْمِ اللّٰهِ الرَّحْمٰنِ الرَّحِيْمِ** || **بِسْمِ اللّٰهِ الرَّحْمٰنِ الرَّحِيْمِ** || **بِسْمِ اللّٰهِ الرَّحْمٰنِ الرَّحِيْمِ**
p **اَلْهٰىكُمُ التَّكَاثُرُ ﴿١﴾ حَتّٰى زُرْتُمُ الْمَقَابِرَ ﴿٢﴾ كَلَّا سَوْفَ تَعْلَمُوْنَ ﴿٣﴾ ثُمَّ كَلَّا سَوْفَ تَعْلَمُوْنَ ﴿٤﴾ كَلَّا لَوْ تَعْلَمُوْنَ عِلْمَ الْيَقِيْنِ ﴿٥﴾ لَتَرَوُنَّ الْجَحِيْمَ ﴿٦﴾ ثُمَّ لَتَرَوُنَّهَا عَيْنَ الْيَقِيْنِ ﴿٧﴾ ثُمَّ لَتُسْئَلُنَّ يَوْمَئِذٍ عَنِ النَّعِيْمِ ﴿٨﴾** || **اَلْهٰىكُمُ التَّكَاثُرُ ﴿١﴾ حَتّٰى زُرْتُمُ الْمَقَابِرَ ﴿٢﴾ كَلَّا سَوْفَ تَعْلَمُوْنَ ﴿٣﴾ ثُمَّ كَلَّا سَوْفَ تَعْلَمُوْنَ ﴿٤﴾ كَلَّا لَوْ تَعْلَمُوْنَ عِلْمَ الْيَقِيْنِ ﴿٥﴾ لَتَرَوُنَّ الْجَحِيْمَ ﴿٦﴾ ثُمَّ لَتَرَوُنَّهَا عَيْنَ الْيَقِيْنِ ﴿٧﴾ ثُمَّ لَتُسْئَلُنَّ يَوْمَئِذٍ عَنِ النَّعِيْمِ ﴿٨﴾** || **اَلْهٰىكُمُ التَّكَاثُرُ ﴿١﴾ حَتّٰى زُرْتُمُ الْمَقَابِرَ ﴿٢﴾ كَلَّا سَوْفَ تَعْلَمُوْنَ ﴿٣﴾ ثُمَّ كَلَّا سَوْفَ تَعْلَمُوْنَ ﴿٤﴾ كَلَّا لَوْ تَعْلَمُوْنَ عِلْمَ الْيَقِيْنِ ﴿٥﴾ لَتَرَوُنَّ الْجَحِيْمَ ﴿٦﴾ ثُمَّ لَتَرَوُنَّهَا عَيْنَ الْيَقِيْنِ ﴿٧﴾ ثُمَّ لَتُسْئَلُنَّ يَوْمَئِذٍ عَنِ النَّعِيْمِ ﴿٨﴾**
p *مفہوم:* تمھیں (مال و اولاد کی) کثرت کی چاہت نے غافل کر دیا (1)، یہاں تک کہ تم قبروں تک جا پہنچے (2)۔ ہرگز نہیں! تمھیں جلد معلوم ہو جائے گا (3)، پھر ہرگز نہیں! تمھیں جلد معلوم ہو جائے گا (4)۔ ہرگز نہیں! کاش تم یقینی علم سے جانتے (5)۔ تم ضرور دوزخ کو دیکھو گے (6)، پھر تم اسے یقین کی آنکھ سے ضرور دیکھو گے (7)، پھر اس دن تم سے نعمتوں کے بارے میں ضرور پوچھا جائے گا (8)۔ || *Meaning:* Competing for more (wealth and children) distracted you (1), until you reached the graves (2). No! You will soon know (3). Again, no! You will soon know (4). No! If only you knew with certain knowledge (5) — you will surely see Hellfire (6); then you will surely see it with the eye of certainty (7); then on that Day you will surely be asked about the blessings (8). || *مانا:* د (مال او اولاد) د ډېرښت لېوالتیا تاسو غافل کړئ (1)، تر دې چې قبرونو ته ورسېدئ (2). هېڅکله نه! ژر به پوه شئ (3)، بیا هېڅکله نه! ژر به پوه شئ (4). هېڅکله نه! که چېرې مو په یقیني علم پوهېدلای (5). تاسو به خامخا دوزخ وګورئ (6)، بیا به یې خامخا د یقین په سترګه وګورئ (7)، بیا به په هغه ورځ خامخا د نعمتونو په اړه وپوښتل شئ (8).
### سُوْرَةُ الْعَصْرِ (مَكِّيَّة) || Surah Al-Asr (Makki) || سورة العصر (مکي)
p **بِسْمِ اللّٰهِ الرَّحْمٰنِ الرَّحِيْمِ** || **بِسْمِ اللّٰهِ الرَّحْمٰنِ الرَّحِيْمِ** || **بِسْمِ اللّٰهِ الرَّحْمٰنِ الرَّحِيْمِ**
p **وَالْعَصْرِ ﴿١﴾ اِنَّ الْاِنْسَانَ لَفِيْ خُسْرٍ ﴿٢﴾ اِلَّا الَّذِيْنَ اٰمَنُوْا وَعَمِلُوا الصّٰلِحٰتِ وَتَوَاصَوْا بِالْحَقِّ ۙ وَتَوَاصَوْا بِالصَّبْرِ ﴿٣﴾** || **وَالْعَصْرِ ﴿١﴾ اِنَّ الْاِنْسَانَ لَفِيْ خُسْرٍ ﴿٢﴾ اِلَّا الَّذِيْنَ اٰمَنُوْا وَعَمِلُوا الصّٰلِحٰتِ وَتَوَاصَوْا بِالْحَقِّ ۙ وَتَوَاصَوْا بِالصَّبْرِ ﴿٣﴾** || **وَالْعَصْرِ ﴿١﴾ اِنَّ الْاِنْسَانَ لَفِيْ خُسْرٍ ﴿٢﴾ اِلَّا الَّذِيْنَ اٰمَنُوْا وَعَمِلُوا الصّٰلِحٰتِ وَتَوَاصَوْا بِالْحَقِّ ۙ وَتَوَاصَوْا بِالصَّبْرِ ﴿٣﴾**
p *مفہوم:* زمانے کی قسم (1)! بے شک انسان خسارے میں ہے (2)، سوائے ان لوگوں کے جو ایمان لائے اور نیک عمل کیے اور ایک دوسرے کو حق کی تلقین کی اور ایک دوسرے کو صبر کی تلقین کی (3)۔ || *Meaning:* By Time (1)! Surely mankind is in loss (2), except those who believe, do good deeds, urge one another to the truth, and urge one another to patience (3). || *مانا:* په زمانې قسم (1)! بېشکه انسان په تاوان کې دی (2)، پرته له هغو چې ایمان یې راوړ او نېک عملونه یې وکړل او یو بل ته یې د حق سپارښتنه وکړه او یو بل ته یې د صبر سپارښتنه وکړه (3).
