p لَعَلَّهُمْ يَرْشُدُونَ ۝ أُحِلَّ لَكُمْ لَيْلَةَ الصِّيَامِ الرَّفَثُ إِلَىٰ نِسَائِكُمْ ۚ هُنَّ لِبَاسٌ لَّكُمْ وَأَنتُمْ لِبَاسٌ لَّهُنَّ ۗ تاکہ وہ ہدایت پر آجائیں۔ تمھارے لیے روزہ کی رات میں اپنی بیویوں کے پاس جانا حلال کر دیا گیا وہ تمھارے لیے لباس ہیں اور تم اُن کے لیے لباس ہو || لَعَلَّهُمْ يَرْشُدُونَ ۝ أُحِلَّ لَكُمْ لَيْلَةَ الصِّيَامِ الرَّفَثُ إِلَىٰ نِسَائِكُمْ ۚ هُنَّ لِبَاسٌ لَّكُمْ وَأَنتُمْ لِبَاسٌ لَّهُنَّ ۗ that they may be (rightly) guided. It has been made permissible for you the night preceding fasting to go to your wives. They are clothing for you and you are clothing for them || لَعَلَّهُمْ يَرْشُدُونَ ۝ أُحِلَّ لَكُمْ لَيْلَةَ الصِّيَامِ الرَّفَثُ إِلَىٰ نِسَائِكُمْ ۚ هُنَّ لِبَاسٌ لَّكُمْ وَأَنتُمْ لِبَاسٌ لَّهُنَّ ۗ ترڅو هغوی هدایت ومومي. ستاسو لپاره د روژې په شپه خپلو ښځو ته تلل حلال شوي هغوی ستاسو لپاره جامې دي او تاسو د هغوی لپاره جامې یاست
p عَلِمَ اللَّهُ أَنَّكُمْ كُنتُمْ تَخْتَانُونَ أَنفُسَكُمْ فَتَابَ عَلَيْكُمْ اللہ کو علم تھا کہ تم (اُن کے پاس جا کر) اپنے آپ سے خیانت کیا کرتے تھے تو اس نے تم پر نظر کرم فرمائی || عَلِمَ اللَّهُ أَنَّكُمْ كُنتُمْ تَخْتَانُونَ أَنفُسَكُمْ فَتَابَ عَلَيْكُمْ Allah knew that you used to deceive yourselves, so He accepted your repentance || عَلِمَ اللَّهُ أَنَّكُمْ كُنتُمْ تَخْتَانُونَ أَنفُسَكُمْ فَتَابَ عَلَيْكُمْ الله پوهېده چې تاسو (هغوی ته تلو سره) له خپل ځان سره خیانت کاوه نو هغه پر تاسو مهرباني وکړه
p وَعَفَا عَنكُمْ ۖ فَالْآنَ بَاشِرُوهُنَّ وَابْتَغُوا مَا كَتَبَ اللَّهُ لَكُمْ ۚ وَكُلُوا وَاشْرَبُوا اور تمھیں معاف فرما دیا تو اب تم ان سے صحبت اختیار کرو اور تلاش کرو جو اللہ نے تمھارے لیے (نصیب میں) لکھ دیا ہے اور کھاؤ اور پیو || وَعَفَا عَنكُمْ ۖ فَالْآنَ بَاشِرُوهُنَّ وَابْتَغُوا مَا كَتَبَ اللَّهُ لَكُمْ ۚ وَكُلُوا وَاشْرَبُوا and pardoned you. So now, have relations with them and seek that which Allah has decreed for you. And eat and drink || وَعَفَا عَنكُمْ ۖ فَالْآنَ بَاشِرُوهُنَّ وَابْتَغُوا مَا كَتَبَ اللَّهُ لَكُمْ ۚ وَكُلُوا وَاشْرَبُوا او تاسو ته یې بخښنه وکړه نو اوس تاسو له هغوی سره ملګرتیا وکړئ او هغه ولټوئ چې الله ستاسو لپاره (په برخه کې) لیکلی او وخورئ او وڅښئ
p حَتَّىٰ يَتَبَيَّنَ لَكُمُ الْخَيْطُ الْأَبْيَضُ مِنَ الْخَيْطِ الْأَسْوَدِ مِنَ الْفَجْرِ ۖ ثُمَّ أَتِمُّوا الصِّيَامَ إِلَى اللَّيْلِ ۚ یہاں تک کہ تمھارے لیے نمایاں ہو جائے (صبح کی) سفید دھاری (رات کی) سیاہ دھاری سے فجر کے وقت پھر رات تک روزہ پورا کرو || حَتَّىٰ يَتَبَيَّنَ لَكُمُ الْخَيْطُ الْأَبْيَضُ مِنَ الْخَيْطِ الْأَسْوَدِ مِنَ الْفَجْرِ ۖ ثُمَّ أَتِمُّوا الصِّيَامَ إِلَى اللَّيْلِ ۚ until the white thread of dawn becomes distinct to you from the black thread (of night). Then complete the fast until the night || حَتَّىٰ يَتَبَيَّنَ لَكُمُ الْخَيْطُ الْأَبْيَضُ مِنَ الْخَيْطِ الْأَسْوَدِ مِنَ الْفَجْرِ ۖ ثُمَّ أَتِمُّوا الصِّيَامَ إِلَى اللَّيْلِ ۚ تر هغه چې ستاسو لپاره (د سهار) سپینه تار (د شپې) تور تار له فجره څرګند شي بیا تر شپې پورې روژه پوره کړئ
p وَلَا تُبَاشِرُوهُنَّ وَأَنتُمْ عَاكِفُونَ فِي الْمَسَاجِدِ ۗ تِلْكَ حُدُودُ اللَّهِ فَلَا تَقْرَبُوهَا ۗ اور بیویوں سے صحبت اختیار نہ کرو جب تم مسجدوں میں اعتکاف کی حالت میں ہو یہ اللہ کی (طے کردہ) حدود ہیں تو ان کے قریب بھی مت جاؤ || وَلَا تُبَاشِرُوهُنَّ وَأَنتُمْ عَاكِفُونَ فِي الْمَسَاجِدِ ۗ تِلْكَ حُدُودُ اللَّهِ فَلَا تَقْرَبُوهَا ۗ And do not have relations with them while you are staying for worship in the mosques. These are the limits of Allah, so do not approach them || وَلَا تُبَاشِرُوهُنَّ وَأَنتُمْ عَاكِفُونَ فِي الْمَسَاجِدِ ۗ تِلْكَ حُدُودُ اللَّهِ فَلَا تَقْرَبُوهَا ۗ او له ښځو سره ملګرتیا مه کوئ کله چې تاسو په جوماتونو کې په اعتکاف کې یاست دا د الله (ټاکل شوې) حدود دي نو هغو ته نژدې هم مه ځئ
p كَذَٰلِكَ يُبَيِّنُ اللَّهُ آيَاتِهِ لِلنَّاسِ لَعَلَّهُمْ يَتَّقُونَ ۝ وَلَا تَأْكُلُوا أَمْوَالَكُم بَيْنَكُم بِالْبَاطِلِ اسی طرح اللہ اپنی آیات لوگوں کے لیے کھول کھول کر بیان فرماتا ہے تاکہ وہ پرہیز گار بنیں۔ اور آپس میں ایک دوسرے کا مال ناجائز طریقہ سے نہ کھاؤ || كَذَٰلِكَ يُبَيِّنُ اللَّهُ آيَاتِهِ لِلنَّاسِ لَعَلَّهُمْ يَتَّقُونَ ۝ وَلَا تَأْكُلُوا أَمْوَالَكُم بَيْنَكُم بِالْبَاطِلِ Thus does Allah make clear His verses to the people that they may become righteous. And do not consume one another’s wealth unjustly || كَذَٰلِكَ يُبَيِّنُ اللَّهُ آيَاتِهِ لِلنَّاسِ لَعَلَّهُمْ يَتَّقُونَ ۝ وَلَا تَأْكُلُوا أَمْوَالَكُم بَيْنَكُم بِالْبَاطِلِ همداسې الله خپل آیتونه د خلکو لپاره په ښه ډول بیانوي ترڅو هغوی پرهېزګاره شي. او په خپل منځ کې یو د بل مال په ناروا طریقه مه خورئ
p وَتُدْلُوا بِهَا إِلَى الْحُكَّامِ لِتَأْكُلُوا فَرِيقًا مِّنْ أَمْوَالِ النَّاسِ بِالْإِثْمِ وَأَنتُمْ تَعْلَمُونَ ۝ اور (نہ ہی) اِس (مال) کو (بطورِ رشوت) حکام تک پہنچاؤ کہ تم جانتے بوجھتے لوگوں کے مال کا کچھ حصہ گناہ کے ساتھ کھاؤ۔ || وَتُدْلُوا بِهَا إِلَى الْحُكَّامِ لِتَأْكُلُوا فَرِيقًا مِّنْ أَمْوَالِ النَّاسِ بِالْإِثْمِ وَأَنتُمْ تَعْلَمُونَ ۝ nor bribe with it the rulers in order that (they might aid) you to consume a portion of the wealth of the people in sin, while you know || وَتُدْلُوا بِهَا إِلَى الْحُكَّامِ لِتَأْكُلُوا فَرِيقًا مِّنْ أَمْوَالِ النَّاسِ بِالْإِثْمِ وَأَنتُمْ تَعْلَمُونَ ۝ او (نه هم) دا (مال) د رشوت په توګه حاکمانو ته مه رسوئ چې تاسو په پوهه سره د خلکو د مال یوه برخه په ګناه سره وخورئ
p يَسْأَلُونَكَ عَنِ الْأَهِلَّةِ ۖ قُلْ وہ آپ (ﷺ) سے نئے چاندوں (کے گھٹنے اور بڑھنے) کے بارے میں سوال کرتے ہیں آپ (ﷺ) فرما دیجیے || يَسْأَلُونَكَ عَنِ الْأَهِلَّةِ ۖ قُلْ They ask you, (O Muhammad), about the new moons. Say || يَسْأَلُونَكَ عَنِ الْأَهِلَّةِ ۖ قُلْ هغوی له تا (ﷺ) نه د نوو سپوږمیو (د کمېدو او زیاتېدو) په اړه پوښتنه کوي ته (ﷺ) ووایه
p هِيَ مَوَاقِيتُ لِلنَّاسِ وَالْحَجِّ ۗ وَلَيْسَ الْبِرُّ بِأَن تَأْتُوا الْبُيُوتَ مِن ظُهُورِهَا کہ یہ لوگوں کے (معاملات) اور حج (کے اوقات) کے تعین کا ذریعہ ہے اور یہ نیکی نہیں ہے کہ تم گھروں میں ان کے پچھواڑے سے داخل ہو || هِيَ مَوَاقِيتُ لِلنَّاسِ وَالْحَجِّ ۗ وَلَيْسَ الْبِرُّ بِأَن تَأْتُوا الْبُيُوتَ مِن ظُهُورِهَا “They are measures of time for the people and for Hajj.” And it is not righteousness to enter houses from the back || هِيَ مَوَاقِيتُ لِلنَّاسِ وَالْحَجِّ ۗ وَلَيْسَ الْبِرُّ بِأَن تَأْتُوا الْبُيُوتَ مِن ظُهُورِهَا چې دا د خلکو (معاملاتو) او د حج (وختونو) ټاکلو وسیله ده او دا نېکي نه ده چې تاسو کورونو ته د هغو له شا نه ننوځئ
p وَلَٰكِنَّ الْبِرَّ مَنِ اتَّقَىٰ ۗ وَأْتُوا الْبُيُوتَ مِنْ أَبْوَابِهَا ۚ وَاتَّقُوا اللَّهَ لَعَلَّكُمْ تُفْلِحُونَ ۝ بلکہ نیکی تو اُس کی ہے جس نے پرہیز گاری اختیار کی اور گھروں میں اُن کے دروازوں سے آیا کرو اور اللہ (کی نافرمانی) سے ڈرو تاکہ تم فلاح پا سکو۔ || وَلَٰكِنَّ الْبِرَّ مَنِ اتَّقَىٰ ۗ وَأْتُوا الْبُيُوتَ مِنْ أَبْوَابِهَا ۚ وَاتَّقُوا اللَّهَ لَعَلَّكُمْ تُفْلِحُونَ ۝ but righteousness is (in) one who fears Allah. And enter houses from their doors. And fear Allah that you may succeed || وَلَٰكِنَّ الْبِرَّ مَنِ اتَّقَىٰ ۗ وَأْتُوا الْبُيُوتَ مِنْ أَبْوَابِهَا ۚ وَاتَّقُوا اللَّهَ لَعَلَّكُمْ تُفْلِحُونَ ۝ بلکې نېکي د هغه چا ده چې پرهېزګاري یې خپله کړې او کورونو ته د هغو له دروازو نه ننوځئ او له الله (د نافرمانۍ) نه ووېرېږئ ترڅو تاسو فلاح ومومئ
p وَقَاتِلُوا فِي سَبِيلِ اللَّهِ الَّذِينَ يُقَاتِلُونَكُمْ وَلَا تَعْتَدُوا ۚ إِنَّ اللَّهَ لَا يُحِبُّ الْمُعْتَدِينَ ۝ اور اللہ کی راہ میں (اُن سے) لڑو جو تم سے لڑتے ہیں اور زیادتی نہ کرنا بے شک اللہ زیادتی کرنے والوں کو پسند نہیں کرتا۔ || وَقَاتِلُوا فِي سَبِيلِ اللَّهِ الَّذِينَ يُقَاتِلُونَكُمْ وَلَا تَعْتَدُوا ۚ إِنَّ اللَّهَ لَا يُحِبُّ الْمُعْتَدِينَ ۝ And fight in the way of Allah those who fight you but do not transgress. Indeed Allah does not like transgressors || وَقَاتِلُوا فِي سَبِيلِ اللَّهِ الَّذِينَ يُقَاتِلُونَكُمْ وَلَا تَعْتَدُوا ۚ إِنَّ اللَّهَ لَا يُحِبُّ الْمُعْتَدِينَ ۝ او د الله په لاره کې (له هغو سره) وجنګېږئ چې له تاسو سره جنګېږي او زیاتی مه کوئ بېشکه الله زیاتی کوونکي نه خوښوي
p وَاقْتُلُوهُمْ حَيْثُ ثَقِفْتُمُوهُمْ وَأَخْرِجُوهُم مِّنْ حَيْثُ أَخْرَجُوكُم اور ان (کافروں) کو قتل کرو جہاں کہیں بھی تم انھیں پاؤ اور انھیں (وہاں سے) نکالو جہاں سے انھوں نے تمھیں نکالا ہے || وَاقْتُلُوهُمْ حَيْثُ ثَقِفْتُمُوهُمْ وَأَخْرِجُوهُم مِّنْ حَيْثُ أَخْرَجُوكُم And kill them wherever you overtake them and expel them from wherever they have expelled you || وَاقْتُلُوهُمْ حَيْثُ ثَقِفْتُمُوهُمْ وَأَخْرِجُوهُم مِّنْ حَيْثُ أَخْرَجُوكُم او هغوی (کافران) ووژنئ چېرته چې هغوی ومومئ او هغوی (له هغه ځایه) وباسئ چېرته چې هغوی تاسو ایستلي دي
