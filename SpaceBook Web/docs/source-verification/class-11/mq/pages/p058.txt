p أَوْ تَسْرِيحٌ بِإِحْسَانٍ ۗ وَلَا يَحِلُّ لَكُمْ أَن تَأْخُذُوا مِمَّا آتَيْتُمُوهُنَّ شَيْئًا یا خوش اسلوبی سے چھوڑ دیا جائے اور تمھارے لیے جائز نہیں کہ تم انھیں جو کچھ دے چکے ہو اُس میں سے کچھ بھی واپس لو || أَوْ تَسْرِيحٌ بِإِحْسَانٍ ۗ وَلَا يَحِلُّ لَكُمْ أَن تَأْخُذُوا مِمَّا آتَيْتُمُوهُنَّ شَيْئًا or release (her) with good treatment. And it is not lawful for you to take anything of what you have given them || أَوْ تَسْرِيحٌ بِإِحْسَانٍ ۗ وَلَا يَحِلُّ لَكُمْ أَن تَأْخُذُوا مِمَّا آتَيْتُمُوهُنَّ شَيْئًا یا په ښه اسلوب پرېښودل شي او ستاسو لپاره روا نه ده چې هغوی ته چې تاسو څه ورکړي له هغو نه څه بېرته واخلئ
p إِلَّا أَن يَخَافَا أَلَّا يُقِيمَا حُدُودَ اللَّهِ ۖ فَإِنْ خِفْتُمْ سوائے اس کے کہ وہ دونوں ڈریں کہ وہ اللہ کی حدود قائم نہ رکھ سکیں گے تو اگر تم خوف محسوس کرو || إِلَّا أَن يَخَافَا أَلَّا يُقِيمَا حُدُودَ اللَّهِ ۖ فَإِنْ خِفْتُمْ except when both fear that they will not be able to keep (within) the limits of Allah. But if you fear || إِلَّا أَن يَخَافَا أَلَّا يُقِيمَا حُدُودَ اللَّهِ ۖ فَإِنْ خِفْتُمْ پرته له دې چې دواړه وويرېږي چې هغوی به د الله حدود نه شي ساتلی نو که تاسو ویره احساس کړئ
p أَلَّا يُقِيمَا حُدُودَ اللَّهِ فَلَا جُنَاحَ عَلَيْهِمَا فِيمَا افْتَدَتْ بِهِ ۗ کہ وہ دونوں اللہ کی حدود کو قائم نہ رکھ سکیں گے تو پھر اُن دونوں پر کوئی گناہ نہیں ہے کہ وہ (عورت جو کہ اپنے آپ کو چھڑانے کے لیے) کچھ معاوضہ دے || أَلَّا يُقِيمَا حُدُودَ اللَّهِ فَلَا جُنَاحَ عَلَيْهِمَا فِيمَا افْتَدَتْ بِهِ ۗ that they will not keep (within) the limits of Allah, then there is no blame upon either of them concerning that by which she ransoms herself || أَلَّا يُقِيمَا حُدُودَ اللَّهِ فَلَا جُنَاحَ عَلَيْهِمَا فِيمَا افْتَدَتْ بِهِ ۗ چې هغوی دواړه به د الله حدود ونه شي ساتلی نو بیا پر هغوی دواړو کومه ګناه نشته چې هغه (ښځه چې خپل ځان خلاصولو لپاره) څه بدل ورکړي
p تِلْكَ حُدُودُ اللَّهِ فَلَا تَعْتَدُوهَا ۚ وَمَن يَتَعَدَّ حُدُودَ اللَّهِ یہ اللہ کی حدود ہیں تو ان سے تجاوز مت کرو اور جس نے اللہ کی حدود سے تجاوز کیا || تِلْكَ حُدُودُ اللَّهِ فَلَا تَعْتَدُوهَا ۚ وَمَن يَتَعَدَّ حُدُودَ اللَّهِ These are the limits (set by) Allah, so do not transgress them. And whoever transgresses the limits of Allah || تِلْكَ حُدُودُ اللَّهِ فَلَا تَعْتَدُوهَا ۚ وَمَن يَتَعَدَّ حُدُودَ اللَّهِ دا د الله حدود دي نو له هغو نه تېرېدل مه کوئ او چا چې د الله له حدودو نه تېرېدل وکړل
p فَأُولَٰئِكَ هُمُ الظَّالِمُونَ ۝ فَإِن طَلَّقَهَا فَلَا تَحِلُّ لَهُ تو وہی لوگ ظالم ہیں۔ پھر اگر وہ (شوہر) اُسے (تیسری بار) طلاق دے دے تو اب وہ عورت اُس کے لیے حلال نہیں || فَأُولَٰئِكَ هُمُ الظَّالِمُونَ ۝ فَإِن طَلَّقَهَا فَلَا تَحِلُّ لَهُ then it is those who are the wrongdoers. And if he divorces her (a third time), then she is not lawful to him || فَأُولَٰئِكَ هُمُ الظَّالِمُونَ ۝ فَإِن طَلَّقَهَا فَلَا تَحِلُّ لَهُ نو همدا کسان ظالمان دي. بیا که هغه (میړه) هغې ته (دریم ځل) طلاق ورکړي نو اوس هغه ښځه د هغه لپاره حلاله نه ده
p مِن بَعْدُ حَتَّىٰ تَنكِحَ زَوْجًا غَيْرَهُ ۗ فَإِن طَلَّقَهَا فَلَا جُنَاحَ عَلَيْهِمَا جب تک کہ وہ اُس کے سوا کسی دوسرے سے نکاح نہ کرے پھر اگر وہ (دوسرا شوہر) بھی اُسے طلاق دے دے تو اُن دونوں پر کوئی گناہ نہیں ہے || مِن بَعْدُ حَتَّىٰ تَنكِحَ زَوْجًا غَيْرَهُ ۗ فَإِن طَلَّقَهَا فَلَا جُنَاحَ عَلَيْهِمَا after that, until she marries a husband other than him. And if the latter husband divorces her, there is no blame upon the two of them || مِن بَعْدُ حَتَّىٰ تَنكِحَ زَوْجًا غَيْرَهُ ۗ فَإِن طَلَّقَهَا فَلَا جُنَاحَ عَلَيْهِمَا له دې وروسته تر هغه چې هغه د هغه له پرته بل سره نکاح ونکړي بیا که هغه (دویم میړه) هم هغې ته طلاق ورکړي نو پر دوی دواړو کومه ګناه نشته
p أَن يَتَرَاجَعَا إِن ظَنَّا أَن يُقِيمَا حُدُودَ اللَّهِ ۗ وَتِلْكَ حُدُودُ اللَّهِ کہ وہ (نیا نکاح کر کے) باہم رجوع کر لیں بشرطیکہ وہ خیال رکھتے ہوں کہ وہ اللہ کی حدود کو قائم رکھ سکیں گے اور یہ اللہ کی حدود ہیں || أَن يَتَرَاجَعَا إِن ظَنَّا أَن يُقِيمَا حُدُودَ اللَّهِ ۗ وَتِلْكَ حُدُودُ اللَّهِ to return to each other if they think that they can keep (within) the limits of Allah. And those are the limits of Allah || أَن يَتَرَاجَعَا إِن ظَنَّا أَن يُقِيمَا حُدُودَ اللَّهِ ۗ وَتِلْكَ حُدُودُ اللَّهِ چې هغوی (نوې نکاح کولو سره) یو بل ته رجوع وکړي په دې شرط چې هغوی انګېري چې د الله حدود به وساتلای شي او دا د الله حدود دي
p يُبَيِّنُهَا لِقَوْمٍ يَعْلَمُونَ ۝ وَإِذَا طَلَّقْتُمُ النِّسَاءَ فَبَلَغْنَ أَجَلَهُنَّ جنھیں وہ واضح فرماتا ہے اُن کے لیے جو سمجھ رکھتے ہیں۔ اور جب تم عورتوں کو (ایک یا دو بار) طلاق دے دو پھر وہ اپنی عدت پوری کرنے پر آئیں || يُبَيِّنُهَا لِقَوْمٍ يَعْلَمُونَ ۝ وَإِذَا طَلَّقْتُمُ النِّسَاءَ فَبَلَغْنَ أَجَلَهُنَّ which He makes clear to a people who know. And when you divorce women and they have (nearly) fulfilled their term || يُبَيِّنُهَا لِقَوْمٍ يَعْلَمُونَ ۝ وَإِذَا طَلَّقْتُمُ النِّسَاءَ فَبَلَغْنَ أَجَلَهُنَّ چې هغه د هغو خلکو لپاره څرګندوي چې پوهه لري. او کله چې تاسو ښځو ته (یو یا دوه ځله) طلاق ورکړئ بیا هغوی خپل عدت پوره کولو ته ورسېږي
p فَأَمْسِكُوهُنَّ بِمَعْرُوفٍ أَوْ سَرِّحُوهُنَّ بِمَعْرُوفٍ ۚ وَلَا تُمْسِكُوهُنَّ ضِرَارًا لِّتَعْتَدُوا ۚ وَمَن يَفْعَلْ ذَٰلِكَ تو انھیں بھلائی کے ساتھ روک لو یا بھلائی کے ساتھ چھوڑ دو اور انھیں ستانے کی خاطر زیادتی کرنے کے لیے نہ روکے رکھو اور جس کسی نے ایسا کیا || فَأَمْسِكُوهُنَّ بِمَعْرُوفٍ أَوْ سَرِّحُوهُنَّ بِمَعْرُوفٍ ۚ وَلَا تُمْسِكُوهُنَّ ضِرَارًا لِّتَعْتَدُوا ۚ وَمَن يَفْعَلْ ذَٰلِكَ then retain them according to acceptable terms or release them according to acceptable terms. And do not retain them to harm them and transgress. And whoever does that || فَأَمْسِكُوهُنَّ بِمَعْرُوفٍ أَوْ سَرِّحُوهُنَّ بِمَعْرُوفٍ ۚ وَلَا تُمْسِكُوهُنَّ ضِرَارًا لِّتَعْتَدُوا ۚ وَمَن يَفْعَلْ ذَٰلِكَ نو هغوی په ښه توګه وساتئ یا په ښه توګه پرېږدئ او هغوی د ځورولو او زیاتي کولو لپاره مه ساتئ او چا چې داسې وکړل
p فَقَدْ ظَلَمَ نَفْسَهُ ۚ وَلَا تَتَّخِذُوا آيَاتِ اللَّهِ هُزُوًا ۚ وَاذْكُرُوا نِعْمَتَ اللَّهِ عَلَيْكُمْ تو اُس نے اپنی ہی جان پر ظلم کیا اور اللہ کی آیات کا مذاق نہ اڑاؤ اور اپنے اوپر اللہ کی نعمت کو یاد کرو || فَقَدْ ظَلَمَ نَفْسَهُ ۚ وَلَا تَتَّخِذُوا آيَاتِ اللَّهِ هُزُوًا ۚ وَاذْكُرُوا نِعْمَتَ اللَّهِ عَلَيْكُمْ has certainly wronged himself. And do not take the verses of Allah in jest. And remember the favour of Allah upon you || فَقَدْ ظَلَمَ نَفْسَهُ ۚ وَلَا تَتَّخِذُوا آيَاتِ اللَّهِ هُزُوًا ۚ وَاذْكُرُوا نِعْمَتَ اللَّهِ عَلَيْكُمْ نو هغه پر خپل ځان ظلم وکړ او د الله آیتونه ملنډې مه کوئ او پر خپل ځان د الله نعمت یاد کړئ
p وَمَا أَنزَلَ عَلَيْكُم مِّنَ الْكِتَابِ وَالْحِكْمَةِ يَعِظُكُم بِهِ ۚ اور جو اللہ نے تم پر نازل فرمایا ہے کتاب اور حکمت میں سے (اسے بھی یاد رکھو) وہ اس کے ذریعہ تمھیں نصیحت فرماتا ہے || وَمَا أَنزَلَ عَلَيْكُم مِّنَ الْكِتَابِ وَالْحِكْمَةِ يَعِظُكُم بِهِ ۚ and what has been revealed to you of the Book and wisdom by which He instructs you || وَمَا أَنزَلَ عَلَيْكُم مِّنَ الْكِتَابِ وَالْحِكْمَةِ يَعِظُكُم بِهِ ۚ او هغه چې الله پر تاسو نازل کړی له کتاب او حکمت نه (هغه هم یاد کړئ) هغه په هغې سره تاسو ته نصیحت کوي
p وَاتَّقُوا اللَّهَ وَاعْلَمُوا أَنَّ اللَّهَ بِكُلِّ شَيْءٍ عَلِيمٌ ۝ اور اللہ (کی نافرمانی) سے ڈرو اور جان لو بے شک اللہ ہر شے کو خوب جاننے والا ہے۔ || وَاتَّقُوا اللَّهَ وَاعْلَمُوا أَنَّ اللَّهَ بِكُلِّ شَيْءٍ عَلِيمٌ ۝ And fear Allah and know that Allah is, of all things, Knowing || وَاتَّقُوا اللَّهَ وَاعْلَمُوا أَنَّ اللَّهَ بِكُلِّ شَيْءٍ عَلِيمٌ ۝ او له الله (د نافرمانۍ) نه ووېرېږئ او پوه شئ چې بېشکه الله هر څه ښه پېژني
