p إِنَّمَا ٱلْمُؤْمِنُونَ ٱلَّذِينَ ءَامَنُوا۟ بِٱللَّهِ وَرَسُولِهِۦ وَإِذَا كَانُوا۟ مَعَهُۥ عَلَىٰٓ أَمْرٍۢ جَامِعٍۢ لَّمْ يَذْهَبُوا۟ حَتَّىٰ يَسْتَـْٔذِنُوهُ ۚ إِنَّ ٱلَّذِينَ يَسْتَـْٔذِنُونَكَ أُو۟لَٰٓئِكَ ٱلَّذِينَ يُؤْمِنُونَ بِٱللَّهِ وَرَسُولِهِۦ ۚ فَإِذَا ٱسْتَـْٔذَنُوكَ لِبَعْضِ شَأْنِهِمْ فَأْذَن لِّمَن شِئْتَ مِنْهُمْ وَٱسْتَغْفِرْ لَهُمُ ٱللَّهَ ۚ إِنَّ ٱللَّهَ غَفُورٌۭ رَّحِيمٌۭ ﴿٦٢﴾ بے شک مومن تو وہی ہیں جو اللہ اور اُس کے رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) پر ایمان لائے اور جب وہ ان (رسول اللہ صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کے ساتھ کسی اجتماعی کام پر (حاضر) ہوتے ہیں تو وہ نہیں جاتے جب تک وہ ان (نبی صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) سے اجازت نہ لے لیں بے شک جو آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) سے اجازت مانگتے ہیں یہی وہ لوگ ہیں جو اللہ اور اُس کے رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) پر ایمان رکھتے ہیں تو جب وہ آپ سے اپنے کسی کام کے لیے اجازت مانگیں تو اُن میں سے جسے آپ چاہیں اجازت دے دیں اور اُن کے لیے اللہ سے مغفرت مانگیں بے شک اللہ بڑا ہی بخشنے والا نہایت رحم فرمانے والا ہے۔ || إِنَّمَا ٱلْمُؤْمِنُونَ ٱلَّذِينَ ءَامَنُوا۟ بِٱللَّهِ وَرَسُولِهِۦ وَإِذَا كَانُوا۟ مَعَهُۥ عَلَىٰٓ أَمْرٍۢ جَامِعٍۢ لَّمْ يَذْهَبُوا۟ حَتَّىٰ يَسْتَـْٔذِنُوهُ ۚ إِنَّ ٱلَّذِينَ يَسْتَـْٔذِنُونَكَ أُو۟لَٰٓئِكَ ٱلَّذِينَ يُؤْمِنُونَ بِٱللَّهِ وَرَسُولِهِۦ ۚ فَإِذَا ٱسْتَـْٔذَنُوكَ لِبَعْضِ شَأْنِهِمْ فَأْذَن لِّمَن شِئْتَ مِنْهُمْ وَٱسْتَغْفِرْ لَهُمُ ٱللَّهَ ۚ إِنَّ ٱللَّهَ غَفُورٌۭ رَّحِيمٌۭ ﴿٦٢﴾ They only are the true believers who believe in Allah and His messenger and, when they are with him on some common errand, go not away until they have asked leave of him. Lo! those who ask leave of thee, those are they who believe in Allah and His messenger. So, if they ask thy leave for some affair of theirs, give leave to whom thou wilt of them, and ask for them forgiveness of Allah. Lo! Allah is Forgiving, Merciful. (24:62) || إِنَّمَا ٱلْمُؤْمِنُونَ ٱلَّذِينَ ءَامَنُوا۟ بِٱللَّهِ وَرَسُولِهِۦ وَإِذَا كَانُوا۟ مَعَهُۥ عَلَىٰٓ أَمْرٍۢ جَامِعٍۢ لَّمْ يَذْهَبُوا۟ حَتَّىٰ يَسْتَـْٔذِنُوهُ ۚ إِنَّ ٱلَّذِينَ يَسْتَـْٔذِنُونَكَ أُو۟لَٰٓئِكَ ٱلَّذِينَ يُؤْمِنُونَ بِٱللَّهِ وَرَسُولِهِۦ ۚ فَإِذَا ٱسْتَـْٔذَنُوكَ لِبَعْضِ شَأْنِهِمْ فَأْذَن لِّمَن شِئْتَ مِنْهُمْ وَٱسْتَغْفِرْ لَهُمُ ٱللَّهَ ۚ إِنَّ ٱللَّهَ غَفُورٌۭ رَّحِيمٌۭ ﴿٦٢﴾ بې‌شکه مؤمنان همغه دي چې پر الله او د هغه پر رسول (صلی الله علیه وعلی آله واصحابه وسلم) یې ایمان راوړی، او کله چې له هغه (رسول الله صلی الله علیه وعلی آله واصحابه وسلم) سره په کوم ټولیز کار کې (حاضر) وي، نو نه ځي تر څو له هغه (نبي صلی الله علیه وعلی آله واصحابه وسلم) اجازه وانه خلي؛ بې‌شکه هغه کسان چې له تاسو (صلی الله علیه وعلی آله واصحابه وسلم) اجازه غواړي، همدا هغه خلک دي چې پر الله او د هغه پر رسول (صلی الله علیه وعلی آله واصحابه وسلم) ایمان لري؛ نو کله چې له تاسو د خپل کوم کار لپاره اجازه وغواړي، نو له هغوی چې چا ته وغواړئ اجازه ورکړئ، او د هغوی لپاره له الله بښنه وغواړئ؛ بې‌شکه الله ډېر بښونکی، بې‌حده مهربان دی.
p لَّا تَجْعَلُوا۟ دُعَآءَ ٱلرَّسُولِ بَيْنَكُمْ كَدُعَآءِ بَعْضِكُم بَعْضًۭا ۚ قَدْ يَعْلَمُ ٱللَّهُ ٱلَّذِينَ يَتَسَلَّلُونَ مِنكُمْ لِوَاذًۭا ۚ فَلْيَحْذَرِ ٱلَّذِينَ يُخَالِفُونَ عَنْ أَمْرِهِۦٓ أَن تُصِيبَهُمْ فِتْنَةٌ أَوْ يُصِيبَهُمْ عَذَابٌ أَلِيمٌ ﴿٦٣﴾ (مومنو!) تم اپنے درمیان رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کے بلانے کو ایسا (معمولی) نہ بنالو جیسے تم آپس میں ایک دوسرے کو بلایا کرتے ہو یقیناً اللہ تم میں سے اُن لوگوں کو (خوب) جانتا ہے جو ایک دوسرے کی آڑ میں چپکے سے کھسک جاتے ہیں پس وہ لوگ ڈریں جو رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کے حکم کے خلاف کر رہے ہیں کہ کہیں ان پر (دنیا میں) کوئی آفت نہ آ پڑے یا انھیں (آخرت میں) کوئی دردناک عذاب نہ آ پکڑے۔ || لَّا تَجْعَلُوا۟ دُعَآءَ ٱلرَّسُولِ بَيْنَكُمْ كَدُعَآءِ بَعْضِكُم بَعْضًۭا ۚ قَدْ يَعْلَمُ ٱللَّهُ ٱلَّذِينَ يَتَسَلَّلُونَ مِنكُمْ لِوَاذًۭا ۚ فَلْيَحْذَرِ ٱلَّذِينَ يُخَالِفُونَ عَنْ أَمْرِهِۦٓ أَن تُصِيبَهُمْ فِتْنَةٌ أَوْ يُصِيبَهُمْ عَذَابٌ أَلِيمٌ ﴿٦٣﴾ Make not the calling of the messenger among you as your calling one of another. Allah knoweth those of you who steal away, hiding themselves. And let those who conspire to evade orders beware lest grief or painful punishment befall them. (24:63) || لَّا تَجْعَلُوا۟ دُعَآءَ ٱلرَّسُولِ بَيْنَكُمْ كَدُعَآءِ بَعْضِكُم بَعْضًۭا ۚ قَدْ يَعْلَمُ ٱللَّهُ ٱلَّذِينَ يَتَسَلَّلُونَ مِنكُمْ لِوَاذًۭا ۚ فَلْيَحْذَرِ ٱلَّذِينَ يُخَالِفُونَ عَنْ أَمْرِهِۦٓ أَن تُصِيبَهُمْ فِتْنَةٌ أَوْ يُصِيبَهُمْ عَذَابٌ أَلِيمٌ ﴿٦٣﴾ (مؤمنانو!) د خپل منځ کې د رسول (صلی الله علیه وعلی آله واصحابه وسلم) بلنه داسې (عادي) مه ګرځوئ لکه تاسو چې یو بل سره رابولئ؛ یقیناً الله له تاسو هغه کسان (ښه) پېژني چې د یو بل تر شا په پټه ووځي؛ نو هغه کسان دې ووېرېږي چې د رسول (صلی الله علیه وعلی آله واصحابه وسلم) د حکم خلاف کوي، چې هسې نه پر هغوی (په دنیا کې) کوم مصیبت راشي یا هغوی (په آخرت کې) دردناک عذاب ونیسي.
p أَلَآ إِنَّ لِلَّهِ مَا فِى ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ ۖ قَدْ يَعْلَمُ مَآ أَنتُمْ عَلَيْهِ وَيَوْمَ يُرْجَعُونَ إِلَيْهِ فَيُنَبِّئُهُم بِمَا عَمِلُوا۟ ۗ وَٱللَّهُ بِكُلِّ شَىْءٍ عَلِيمٌۢ ﴿٦٤﴾ آگاہ ہو جاؤ! بے شک اللہ ہی کا ہے جو کچھ آسمانوں اور زمینوں میں ہے وہ یقیناً جانتا ہے جس حال پر تم ہو اور جس دن لوگ اس کی طرف لوٹائے جائیں گے تو وہ ان کو بتا دے گا جو کچھ وہ کیا کرتے تھے اور اللہ ہر شے کو خوب جاننے والا ہے۔ || أَلَآ إِنَّ لِلَّهِ مَا فِى ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ ۖ قَدْ يَعْلَمُ مَآ أَنتُمْ عَلَيْهِ وَيَوْمَ يُرْجَعُونَ إِلَيْهِ فَيُنَبِّئُهُم بِمَا عَمِلُوا۟ ۗ وَٱللَّهُ بِكُلِّ شَىْءٍ عَلِيمٌۢ ﴿٦٤﴾ Lo! verily unto Allah belongeth whatsoever is in the heavens and the earth. He knoweth your condition. And (He knoweth) the Day when they are returned unto Him so that He may inform them of what they did. Allah is Knower of all things. (24:64) || أَلَآ إِنَّ لِلَّهِ مَا فِى ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ ۖ قَدْ يَعْلَمُ مَآ أَنتُمْ عَلَيْهِ وَيَوْمَ يُرْجَعُونَ إِلَيْهِ فَيُنَبِّئُهُم بِمَا عَمِلُوا۟ ۗ وَٱللَّهُ بِكُلِّ شَىْءٍ عَلِيمٌۢ ﴿٦٤﴾ خبر اوسئ! بې‌شکه چې څه په آسمانونو او ځمکو کې دي یوازې د الله دي؛ هغه یقیناً پوهېږي چې تاسو پر کوم حال یاست؛ او په کومه ورځ چې خلک هغه ته ور وګرځول شي، نو هغوی ته به ووايي چې څه به یې کول؛ او الله په هر شي ښه پوه دی.
## علم و عمل کی باتیں || Matters of knowledge and action || د علم او عمل خبرې
- آسمان اور زمین کی تمام مخلوقات کو ہدایت کا نور پہنچانے والا صرف اللہ تعالیٰ ہے۔ اللہ تعالیٰ کے نورِ ہدایت کی تمثیل چراغ سے اس لیے دی گئی ہے کہ جیسے چراغ ہمیشہ اندھیرے کے بیچوں بیچ روشنی پیدا کرتا ہے۔ اسی طرح ہدایت کا نور گمراہی کے اندھیروں میں حق کا راستہ دکھاتا ہے۔ || Allah Almighty alone is the One who conveys the light of guidance to all creatures of the heavens and the earth. The light of the guidance of Allah Almighty has been likened to a lamp because, just as a lamp always creates light in the midst of darkness, in the same way the light of guidance shows the way of truth in the darkness of the wayfarer. || د آسمان او ځمکې ټولو مخلوقاتو ته د هدایت نور رسوونکی یوازې الله تعالی دی. د الله تعالی د هدایت نور ته د څراغ مثال ځکه ورکړل شوی چې لکه څنګه څراغ تل د تیارې په منځ کې رڼا پیدا کوي، همداسې د هدایت نور د ګمراهۍ په تیارو کې د حق لار ښيي.
- کفار کے اعمال کی تمثیل سراب کی طرح ہے جس کی کوئی حقیقت نہیں یعنی جس طرح سراب پر پہنچ کر پیاسے آدمی کی پریشانی اور تھکاوٹ میں اضافہ ہوتا ہے اسی طرح قیامت کے دن کفار کا حال ہوگا۔ || The deeds of the disbelievers have been likened to a mirage, which has no reality; that is, just as on reaching a mirage a thirsty man's distress and fatigue increase, in the same way will be the condition of the disbelievers on the Day of Resurrection. || د کافرانو د عملونو مثال د سراب په څېر دی چې هېڅ حقیقت نه لري؛ یعنې لکه څنګه چې سراب ته په رسېدو د تږي سړي پرېشاني او ستړیا زیاتېږي، همداسې به د قیامت په ورځ د کافرانو حال وي.
- بچوں کی تربیت ابتدائی عمر سے کرنا ضروری ہے بلوغت سے قبل ہی سے ان کو پردے کے معاملات اور حیا کے پہلوؤں سے آگاہ کرنا چاہیے۔ بچوں کی تربیت میں نامناسب مناظر سے احتراز بہت اہمیت کا حامل ہے۔ || The upbringing of children should be done from an early age; even before puberty they should be made aware of matters of purdah and aspects of modesty. Avoiding inappropriate scenes in the upbringing of children is of great importance. || د ماشومانو روزنه له لومړني عمر څخه اړینه ده؛ له بلوغ مخکې باید د پردې له چارو او د حیا له اړخونو خبر کړای شي. د ماشومانو په روزنه کې له ناوړه منظرو ډډه ډېر اهمیت لري.
