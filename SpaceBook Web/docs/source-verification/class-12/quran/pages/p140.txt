p وَٱلَّذِينَ يُحَآجُّونَ فِى ٱللَّهِ مِنۢ بَعْدِ مَا ٱسْتُجِيبَ لَهُۥ حُجَّتُهُمْ دَاحِضَةٌ عِندَ رَبِّهِمْ وَعَلَيْهِمْ غَضَبٌۭ وَلَهُمْ عَذَابٌۭ شَدِيدٌ ﴿١٦﴾ اور جو لوگ اللہ کے بارے میں جھگڑتے ہیں اس کے بعد کہ اس (کی دعوت) کو قبول کر لیا گیا ہے ان کا جھگڑا ان کے رب کے یہاں باطل ہے اور ان پر (اللہ کا) غضب ہے اور ان کے لیے سخت عذاب ہے۔ || وَٱلَّذِينَ يُحَآجُّونَ فِى ٱللَّهِ مِنۢ بَعْدِ مَا ٱسْتُجِيبَ لَهُۥ حُجَّتُهُمْ دَاحِضَةٌ عِندَ رَبِّهِمْ وَعَلَيْهِمْ غَضَبٌۭ وَلَهُمْ عَذَابٌۭ شَدِيدٌ ﴿١٦﴾ And those who argue concerning Allah after He hath been acknowledged, their argument hath no weight with their Lord, and wrath is upon them and theirs will be an awful doom. (42:16) || وَٱلَّذِينَ يُحَآجُّونَ فِى ٱللَّهِ مِنۢ بَعْدِ مَا ٱسْتُجِيبَ لَهُۥ حُجَّتُهُمْ دَاحِضَةٌ عِندَ رَبِّهِمْ وَعَلَيْهِمْ غَضَبٌۭ وَلَهُمْ عَذَابٌۭ شَدِيدٌ ﴿١٦﴾ او هغه کسان چې د الله په اړه شخړه کوي وروسته له دې چې د هغه (بلنه) ومنل شوه، د هغوی شخړه د هغوی د رب په وړاندې باطله ده، پر هغوی د (الله) غضب دی او د هغوی لپاره سخت عذاب دی.
p ٱللَّهُ ٱلَّذِىٓ أَنزَلَ ٱلْكِتَٰبَ بِٱلْحَقِّ وَٱلْمِيزَانَ ۗ وَمَا يُدْرِيكَ لَعَلَّ ٱلسَّاعَةَ قَرِيبٌۭ ﴿١٧﴾ اللہ ہی ہے جس نے حق کے ساتھ کتاب کو نازل فرمایا اور میزان (کو بھی) اور تمھیں کیا معلوم شاید قیامت قریب ہی ہو۔ || ٱللَّهُ ٱلَّذِىٓ أَنزَلَ ٱلْكِتَٰبَ بِٱلْحَقِّ وَٱلْمِيزَانَ ۗ وَمَا يُدْرِيكَ لَعَلَّ ٱلسَّاعَةَ قَرِيبٌۭ ﴿١٧﴾ Allah it is Who hath revealed the Scripture with truth, and the Balance. How canst thou know? It may be that the Hour is nigh. (42:17) || ٱللَّهُ ٱلَّذِىٓ أَنزَلَ ٱلْكِتَٰبَ بِٱلْحَقِّ وَٱلْمِيزَانَ ۗ وَمَا يُدْرِيكَ لَعَلَّ ٱلسَّاعَةَ قَرِيبٌۭ ﴿١٧﴾ یوازې الله دی چې کتاب یې په حق سره نازل کړ او میزان (هم)؛ او ته څه پوهېږې، کېدای شي قیامت نږدې وي.
p يَسْتَعْجِلُ بِهَا ٱلَّذِينَ لَا يُؤْمِنُونَ بِهَا ۖ وَٱلَّذِينَ ءَامَنُوا۟ مُشْفِقُونَ مِنْهَا وَيَعْلَمُونَ أَنَّهَا ٱلْحَقُّ ۗ أَلَآ إِنَّ ٱلَّذِينَ يُمَارُونَ فِى ٱلسَّاعَةِ لَفِى ضَلَٰلٍۭ بَعِيدٍ ﴿١٨﴾ جو لوگ اس پر ایمان نہیں رکھتے وہ اس کے لیے جلدی مچاتے ہیں اور جو اس پر ایمان رکھتے ہیں وہ اس سے ڈرتے ہیں اور وہ جانتے ہیں کہ وہ حق ہے آگاہ ہو جاؤ بے شک جو لوگ قیامت کے بارے میں جھگڑتے ہیں (وہ) ضرور دور کی گمراہی میں پڑے ہوئے ہیں۔ || يَسْتَعْجِلُ بِهَا ٱلَّذِينَ لَا يُؤْمِنُونَ بِهَا ۖ وَٱلَّذِينَ ءَامَنُوا۟ مُشْفِقُونَ مِنْهَا وَيَعْلَمُونَ أَنَّهَا ٱلْحَقُّ ۗ أَلَآ إِنَّ ٱلَّذِينَ يُمَارُونَ فِى ٱلسَّاعَةِ لَفِى ضَلَٰلٍۭ بَعِيدٍ ﴿١٨﴾ Those who believe not therein seek to hasten it, while those who believe are fearful of it and know that it is the Truth. Are not they who dispute, in doubt concerning the Hour, far astray? (42:18) || يَسْتَعْجِلُ بِهَا ٱلَّذِينَ لَا يُؤْمِنُونَ بِهَا ۖ وَٱلَّذِينَ ءَامَنُوا۟ مُشْفِقُونَ مِنْهَا وَيَعْلَمُونَ أَنَّهَا ٱلْحَقُّ ۗ أَلَآ إِنَّ ٱلَّذِينَ يُمَارُونَ فِى ٱلسَّاعَةِ لَفِى ضَلَٰلٍۭ بَعِيدٍ ﴿١٨﴾ هغه کسان چې پرې ایمان نه لري، د هغې لپاره بېړه کوي، او هغه کسان چې پرې ایمان لري، ترې وېرېږي او پوهېږي چې هغه حق دی؛ خبر اوسئ! بې‌شکه هغه کسان چې د قیامت په اړه شخړه کوي، (هغوی) خامخا په لرې ګمراهۍ کې پراته دي.
p ٱللَّهُ لَطِيفٌۢ بِعِبَادِهِۦ يَرْزُقُ مَن يَشَآءُ ۖ وَهُوَ ٱلْقَوِىُّ ٱلْعَزِيزُ ﴿١٩﴾ اللہ اپنے بندوں پر بہت مہربان ہے وہ جسے چاہتا ہے رزق عطا فرماتا ہے اور وہی بہت قوت والا بہت غالب ہے۔ || ٱللَّهُ لَطِيفٌۢ بِعِبَادِهِۦ يَرْزُقُ مَن يَشَآءُ ۖ وَهُوَ ٱلْقَوِىُّ ٱلْعَزِيزُ ﴿١٩﴾ Allah is gracious unto His slaves. He provideth for whom He will. And He is the Strong, the Mighty. (42:19) || ٱللَّهُ لَطِيفٌۢ بِعِبَادِهِۦ يَرْزُقُ مَن يَشَآءُ ۖ وَهُوَ ٱلْقَوِىُّ ٱلْعَزِيزُ ﴿١٩﴾ الله پر خپلو بندګانو ډېر مهربان دی؛ چا ته چې وغواړي رزق ورکوي؛ او همغه ډېر ځواکمن، ډېر غالب دی.
p مَن كَانَ يُرِيدُ حَرْثَ ٱلْءَاخِرَةِ نَزِدْ لَهُۥ فِى حَرْثِهِۦ ۖ وَمَن كَانَ يُرِيدُ حَرْثَ ٱلدُّنْيَا نُؤْتِهِۦ مِنْهَا وَمَا لَهُۥ فِى ٱلْءَاخِرَةِ مِن نَّصِيبٍ ﴿٢٠﴾ جو شخص آخرت کی کھیتی چاہتا ہے ہم اس کے لیے اس کی کھیتی میں اضافہ فرما دیتے ہیں اور جو شخص دنیا کی کھیتی چاہتا ہے (تو) ہم اسے اس میں سے کچھ عطا کر دیتے ہیں اور آخرت میں اس کے لیے کوئی حصہ نہیں۔ || مَن كَانَ يُرِيدُ حَرْثَ ٱلْءَاخِرَةِ نَزِدْ لَهُۥ فِى حَرْثِهِۦ ۖ وَمَن كَانَ يُرِيدُ حَرْثَ ٱلدُّنْيَا نُؤْتِهِۦ مِنْهَا وَمَا لَهُۥ فِى ٱلْءَاخِرَةِ مِن نَّصِيبٍ ﴿٢٠﴾ Whoso desireth the harvest of the Hereafter, We give him increase in its harvest. And whoso desireth the harvest of the world, We give him thereof, and he hath no portion in the Hereafter. (42:20) || مَن كَانَ يُرِيدُ حَرْثَ ٱلْءَاخِرَةِ نَزِدْ لَهُۥ فِى حَرْثِهِۦ ۖ وَمَن كَانَ يُرِيدُ حَرْثَ ٱلدُّنْيَا نُؤْتِهِۦ مِنْهَا وَمَا لَهُۥ فِى ٱلْءَاخِرَةِ مِن نَّصِيبٍ ﴿٢٠﴾ څوک چې د آخرت کښت غواړي، موږ یې د هغه لپاره په کښت کې زیاتوالی کوو؛ او څوک چې د دنیا کښت غواړي، (نو) موږ یې له هغې څه ورکوو، او په آخرت کې یې هېڅ برخه نشته.
p أَمْ لَهُمْ شُرَكَٰٓؤُا۟ شَرَعُوا۟ لَهُم مِّنَ ٱلدِّينِ مَا لَمْ يَأْذَنۢ بِهِ ٱللَّهُ ۚ وَلَوْلَا كَلِمَةُ ٱلْفَصْلِ لَقُضِىَ بَيْنَهُمْ ۗ وَإِنَّ ٱلظَّٰلِمِينَ لَهُمْ عَذَابٌ أَلِيمٌۭ ﴿٢١﴾ کیا ان کے کچھ شریک ہیں جنھوں نے ان کے لیے ایسا دین مقرر کیا ہے جس کا اللہ نے حکم نہیں دیا؟ اور اگر فیصلہ کی بات طے نہ ہو چکی ہوتی تو ان کے درمیان (کب کا) فیصلہ کر دیا جاتا اور بے شک ظالموں کے لیے دردناک عذاب ہے۔ || أَمْ لَهُمْ شُرَكَٰٓؤُا۟ شَرَعُوا۟ لَهُم مِّنَ ٱلدِّينِ مَا لَمْ يَأْذَنۢ بِهِ ٱللَّهُ ۚ وَلَوْلَا كَلِمَةُ ٱلْفَصْلِ لَقُضِىَ بَيْنَهُمْ ۗ وَإِنَّ ٱلظَّٰلِمِينَ لَهُمْ عَذَابٌ أَلِيمٌۭ ﴿٢١﴾ Or have they partners (of Allah) who have made lawful for them in religion that which Allah allowed not? And but for a decisive word (gone forth already), it would have been judged between them. Lo! for wrong-doers is a painful doom. (42:21) || أَمْ لَهُمْ شُرَكَٰٓؤُا۟ شَرَعُوا۟ لَهُم مِّنَ ٱلدِّينِ مَا لَمْ يَأْذَنۢ بِهِ ٱللَّهُ ۚ وَلَوْلَا كَلِمَةُ ٱلْفَصْلِ لَقُضِىَ بَيْنَهُمْ ۗ وَإِنَّ ٱلظَّٰلِمِينَ لَهُمْ عَذَابٌ أَلِيمٌۭ ﴿٢١﴾ ایا د هغوی ځینې شریکان دي چې د هغوی لپاره یې داسې دین ټاکلی چې الله یې حکم نه دی کړی؟ او که د پرېکړې خبره ټاکل شوې نه وای، نو د هغوی ترمنځ به (کله ناکله) پرېکړه شوې وای؛ او بې‌شکه د ظالمانو لپاره دردناک عذاب دی.
p تَرَى ٱلظَّٰلِمِينَ مُشْفِقِينَ مِمَّا كَسَبُوا۟ وَهُوَ وَاقِعٌۢ بِهِمْ ۗ وَٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّٰلِحَٰتِ فِى رَوْضَاتِ ٱلْجَنَّاتِ ۖ لَهُم مَّا يَشَآءُونَ عِندَ رَبِّهِمْ ۚ ذَٰلِكَ هُوَ ٱلْفَضْلُ ٱلْكَبِيرُ ﴿٢٢﴾ آپ ظالموں کو دیکھیں گے وہ اپنے کیے ہوئے (اعمال) سے ڈر رہے ہوں گے اور وہ (عذاب) ان پر واقع ہو کر رہے گا اور جو لوگ ایمان لائے اور انھوں نے نیک اعمال کیے (وہ) جنتوں کے باغوں میں ہوں گے ان کے لیے ان کے رب کے پاس وہ سب کچھ ہوگا جو وہ چاہیں گے یہی بہت بڑا فضل ہے۔ || تَرَى ٱلظَّٰلِمِينَ مُشْفِقِينَ مِمَّا كَسَبُوا۟ وَهُوَ وَاقِعٌۢ بِهِمْ ۗ وَٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّٰلِحَٰتِ فِى رَوْضَاتِ ٱلْجَنَّاتِ ۖ لَهُم مَّا يَشَآءُونَ عِندَ رَبِّهِمْ ۚ ذَٰلِكَ هُوَ ٱلْفَضْلُ ٱلْكَبِيرُ ﴿٢٢﴾ Thou seest the wrong-doers fearful of that which they have earned, and it will surely befall them, while those who believe and do good works (will be) in flowering meadows of the Gardens, having what they wish from their Lord. This is the great preferment. (42:22) || تَرَى ٱلظَّٰلِمِينَ مُشْفِقِينَ مِمَّا كَسَبُوا۟ وَهُوَ وَاقِعٌۢ بِهِمْ ۗ وَٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّٰلِحَٰتِ فِى رَوْضَاتِ ٱلْجَنَّاتِ ۖ لَهُم مَّا يَشَآءُونَ عِندَ رَبِّهِمْ ۚ ذَٰلِكَ هُوَ ٱلْفَضْلُ ٱلْكَبِيرُ ﴿٢٢﴾ تاسو به ظالمان ووینئ چې له خپلو کړو (عملونو) وېرېږي، او هغه (عذاب) به خامخا پرې واقع کېږي؛ او هغه کسان چې ایمان یې راوړ او نېک عملونه یې وکړل، (هغوی) به د جنتونو په باغونو کې وي؛ د هغوی لپاره به د هغوی د رب سره هر هغه څه وي چې وې غواړي؛ همدا ډېر لوی فضل دی.
