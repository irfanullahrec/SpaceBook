p فَهَلْ عَسَيْتُمْ إِن تَوَلَّيْتُمْ أَن تُفْسِدُوا۟ فِى ٱلْأَرْضِ وَتُقَطِّعُوٓا۟ أَرْحَامَكُمْ ﴿٢٢﴾ پھر تم سے کیا توقع رکھی جائے اگر تم حاکم بن جاؤ؟ یہی کہ تم زمین میں فساد مچاؤ گے اور اپنی قرابت داریاں کاٹنے لگو گے۔ || فَهَلْ عَسَيْتُمْ إِن تَوَلَّيْتُمْ أَن تُفْسِدُوا۟ فِى ٱلْأَرْضِ وَتُقَطِّعُوٓا۟ أَرْحَامَكُمْ ﴿٢٢﴾ Would ye then, if ye were given the command, work corruption in the land and sever your ties of kinship? (47:22) || فَهَلْ عَسَيْتُمْ إِن تَوَلَّيْتُمْ أَن تُفْسِدُوا۟ فِى ٱلْأَرْضِ وَتُقَطِّعُوٓا۟ أَرْحَامَكُمْ ﴿٢٢﴾ بیا له تاسو څه هیله وکړای شي که حاکمان شئ؟ همدا چې په ځمکه کې به فساد جوړ کړئ او خپلې خپلولۍ به پرې کړئ.
p أُو۟لَٰٓئِكَ ٱلَّذِينَ لَعَنَهُمُ ٱللَّهُ فَأَصَمَّهُمْ وَأَعْمَىٰٓ أَبْصَٰرَهُمْ ﴿٢٣﴾ یہ وہ لوگ ہیں اللہ نے اُن پر لعنت کر دی چنانچہ انھیں بہرا کر دیا اور ان کی آنکھوں کو اندھا کر دیا۔ || أُو۟لَٰٓئِكَ ٱلَّذِينَ لَعَنَهُمُ ٱللَّهُ فَأَصَمَّهُمْ وَأَعْمَىٰٓ أَبْصَٰرَهُمْ ﴿٢٣﴾ Such are they whom Allah curseth so that He deafeneth them and maketh blind their eyes. (47:23) || أُو۟لَٰٓئِكَ ٱلَّذِينَ لَعَنَهُمُ ٱللَّهُ فَأَصَمَّهُمْ وَأَعْمَىٰٓ أَبْصَٰرَهُمْ ﴿٢٣﴾ دا هغه خلک دي چې الله پرې لعنت کړی، نو کاڼه یې کړل او سترګې یې ړندې کړې.
p أَفَلَا يَتَدَبَّرُونَ ٱلْقُرْءَانَ أَمْ عَلَىٰ قُلُوبٍ أَقْفَالُهَآ ﴿٢٤﴾ تو کیا وہ قرآن میں غور نہیں کرتے یا ان کے دلوں پر تالے (لگے ہوئے) ہیں۔ || أَفَلَا يَتَدَبَّرُونَ ٱلْقُرْءَانَ أَمْ عَلَىٰ قُلُوبٍ أَقْفَالُهَآ ﴿٢٤﴾ Will they then not meditate on the Qur'an, or are there locks on the hearts? (47:24) || أَفَلَا يَتَدَبَّرُونَ ٱلْقُرْءَانَ أَمْ عَلَىٰ قُلُوبٍ أَقْفَالُهَآ ﴿٢٤﴾ نو ایا هغوی په قرآن کې فکر نه کوي، که د هغوی پر زړونو قلفونه (لګېدلي) دي؟
p إِنَّ ٱلَّذِينَ ٱرْتَدُّوا۟ عَلَىٰٓ أَدْبَٰرِهِم مِّنۢ بَعْدِ مَا تَبَيَّنَ لَهُمُ ٱلْهُدَى ۙ ٱلشَّيْطَٰنُ سَوَّلَ لَهُمْ وَأَمْلَىٰ لَهُمْ ﴿٢٥﴾ بے شک جو لوگ اپنی پیٹھ پھیر کر پیچھے (کفر کی طرف) لوٹ گئے بعد اس کے کہ ان کے لیے ہدایت خوب واضح ہو چکی ہے شیطان نے انھیں فریب دیا ہے اور انھیں لمبی زندگی کی امید دلائی ہے۔ || إِنَّ ٱلَّذِينَ ٱرْتَدُّوا۟ عَلَىٰٓ أَدْبَٰرِهِم مِّنۢ بَعْدِ مَا تَبَيَّنَ لَهُمُ ٱلْهُدَى ۙ ٱلشَّيْطَٰنُ سَوَّلَ لَهُمْ وَأَمْلَىٰ لَهُمْ ﴿٢٥﴾ Lo! those who turn back after the guidance hath been manifested unto them, Satan hath seduced them, and He giveth them the rein. (47:25) || إِنَّ ٱلَّذِينَ ٱرْتَدُّوا۟ عَلَىٰٓ أَدْبَٰرِهِم مِّنۢ بَعْدِ مَا تَبَيَّنَ لَهُمُ ٱلْهُدَى ۙ ٱلشَّيْطَٰنُ سَوَّلَ لَهُمْ وَأَمْلَىٰ لَهُمْ ﴿٢٥﴾ بې‌شکه هغه کسان چې شا یې واړوله او بېرته (کفر ته) وګرځېدل وروسته له دې چې هدایت ورته ښه روښانه شوی و، شیطان هغوی غولولي او د اوږد ژوند هیله یې ورکړې.
p ذَٰلِكَ بِأَنَّهُمْ قَالُوا۟ لِلَّذِينَ كَرِهُوا۟ مَا نَزَّلَ ٱللَّهُ سَنُطِيعُكُمْ فِى بَعْضِ ٱلْأَمْرِ ۖ وَٱللَّهُ يَعْلَمُ إِسْرَارَهُمْ ﴿٢٦﴾ یہ اس لیے کہ (ان منافقین نے) ایسے لوگوں کے لیے کہا جنھیں وہ ناپسند ہوا جو اللہ نے نازل کیا ہے کہ ہم بعض معاملات میں تمھاری بات مانیں گے اور اللہ اُن کی پوشیدہ باتوں کو جانتا ہے۔ || ذَٰلِكَ بِأَنَّهُمْ قَالُوا۟ لِلَّذِينَ كَرِهُوا۟ مَا نَزَّلَ ٱللَّهُ سَنُطِيعُكُمْ فِى بَعْضِ ٱلْأَمْرِ ۖ وَٱللَّهُ يَعْلَمُ إِسْرَارَهُمْ ﴿٢٦﴾ That is because they say unto those who hate what Allah hath revealed: We will obey you in some matters; and Allah knoweth their secret talk. (47:26) || ذَٰلِكَ بِأَنَّهُمْ قَالُوا۟ لِلَّذِينَ كَرِهُوا۟ مَا نَزَّلَ ٱللَّهُ سَنُطِيعُكُمْ فِى بَعْضِ ٱلْأَمْرِ ۖ وَٱللَّهُ يَعْلَمُ إِسْرَارَهُمْ ﴿٢٦﴾ دا ځکه چې (دې منافقانو) هغو خلکو ته وویل چې هغه څه یې ناخوښ وو چې الله نازل کړي: موږ به په ځینو چارو کې ستاسو خبره ومنو؛ او الله د هغوی په پټو خبرو پوهېږي.
p فَكَيْفَ إِذَا تَوَفَّتْهُمُ ٱلْمَلَٰٓئِكَةُ يَضْرِبُونَ وُجُوهَهُمْ وَأَدْبَٰرَهُمْ ﴿٢٧﴾ تو کیا حال ہوگا جب فرشتے ان کی جان نکالیں گے (تو) وہ ماریں گے ان کے چہروں پر اور ان کی پیٹھوں پر۔ || فَكَيْفَ إِذَا تَوَفَّتْهُمُ ٱلْمَلَٰٓئِكَةُ يَضْرِبُونَ وُجُوهَهُمْ وَأَدْبَٰرَهُمْ ﴿٢٧﴾ Then how (will it be with them) when the angels gather them, smiting their faces and their backs! (47:27) || فَكَيْفَ إِذَا تَوَفَّتْهُمُ ٱلْمَلَٰٓئِكَةُ يَضْرِبُونَ وُجُوهَهُمْ وَأَدْبَٰرَهُمْ ﴿٢٧﴾ نو حال به یې څه وي کله چې پرښتې یې ساه وباسي، (نو) د هغوی پر مخونو او شاګانو به وهي.
p ذَٰلِكَ بِأَنَّهُمُ ٱتَّبَعُوا۟ مَآ أَسْخَطَ ٱللَّهَ وَكَرِهُوا۟ رِضْوَٰنَهُۥ فَأَحْبَطَ أَعْمَٰلَهُمْ ﴿٢٨﴾ یہ اس لیے کہ انھوں نے اس چیز کی پیروی کی جس نے اللہ کو ناراض کر دیا اور انھیں اس کی رضامندی ناپسند ہوئی تو اُس (اللہ) نے ان کے اعمال اکارت کر دیے۔ || ذَٰلِكَ بِأَنَّهُمُ ٱتَّبَعُوا۟ مَآ أَسْخَطَ ٱللَّهَ وَكَرِهُوا۟ رِضْوَٰنَهُۥ فَأَحْبَطَ أَعْمَٰلَهُمْ ﴿٢٨﴾ That will be because they followed that which angereth Allah, and hated that which pleaseth Him. Therefor He hath made their actions vain. (47:28) || ذَٰلِكَ بِأَنَّهُمُ ٱتَّبَعُوا۟ مَآ أَسْخَطَ ٱللَّهَ وَكَرِهُوا۟ رِضْوَٰنَهُۥ فَأَحْبَطَ أَعْمَٰلَهُمْ ﴿٢٨﴾ دا ځکه چې هغوی د هغه شي پیروي وکړه چې الله یې ناراضه کړ، او د هغه رضا یې ناخوښه شوه، نو هغه (الله) د هغوی عملونه برباد کړل.
p أَمْ حَسِبَ ٱلَّذِينَ فِى قُلُوبِهِم مَّرَضٌ أَن لَّن يُخْرِجَ ٱللَّهُ أَضْغَٰنَهُمْ ﴿٢٩﴾ کیا ان لوگوں نے یہ خیال کیا جن کے دلوں میں (نفاق کی) بیماری ہے کہ اللہ اُن کے چھپے ہوئے کینے ہرگز ظاہر نہیں کرے گا۔ || أَمْ حَسِبَ ٱلَّذِينَ فِى قُلُوبِهِم مَّرَضٌ أَن لَّن يُخْرِجَ ٱللَّهُ أَضْغَٰنَهُمْ ﴿٢٩﴾ Or do those in whose hearts is a disease deem that Allah will not bring to light their (secret) hates? (47:29) || أَمْ حَسِبَ ٱلَّذِينَ فِى قُلُوبِهِم مَّرَضٌ أَن لَّن يُخْرِجَ ٱللَّهُ أَضْغَٰنَهُمْ ﴿٢٩﴾ ایا هغو کسانو چې په زړونو کې یې د (نفاق) ناروغي ده، دا ګومان کړی چې الله به د هغوی پټې کینې هېڅکله ښکاره نه کړي؟
p وَلَوْ نَشَآءُ لَأَرَيْنَٰكَهُمْ فَلَعَرَفْتَهُم بِسِيمَٰهُمْ ۚ وَلَتَعْرِفَنَّهُمْ فِى لَحْنِ ٱلْقَوْلِ ۚ وَٱللَّهُ يَعْلَمُ أَعْمَٰلَكُمْ ﴿٣٠﴾ اور اگر ہم چاہیں تو آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کو وہ لوگ ضرور دکھا دیں پھر یقیناً آپ انھیں ان کے چہروں سے پہچان لیں اور آپ انھیں اندازِ کلام سے (بھی) ضرور پہچان لیں گے اور اللہ تمھارے اعمال جانتا ہے۔ || وَلَوْ نَشَآءُ لَأَرَيْنَٰكَهُمْ فَلَعَرَفْتَهُم بِسِيمَٰهُمْ ۚ وَلَتَعْرِفَنَّهُمْ فِى لَحْنِ ٱلْقَوْلِ ۚ وَٱللَّهُ يَعْلَمُ أَعْمَٰلَكُمْ ﴿٣٠﴾ And if We would, We could show them unto thee (Muhammad) so that thou shouldst know them surely by their marks. And thou shalt know them by the burden of their talk. And Allah knoweth your deeds. (47:30) || وَلَوْ نَشَآءُ لَأَرَيْنَٰكَهُمْ فَلَعَرَفْتَهُم بِسِيمَٰهُمْ ۚ وَلَتَعْرِفَنَّهُمْ فِى لَحْنِ ٱلْقَوْلِ ۚ وَٱللَّهُ يَعْلَمُ أَعْمَٰلَكُمْ ﴿٣٠﴾ او که موږ وغواړو، نو تاسو (صلی الله علیه وعلی آله واصحابه وسلم) ته به هغه خلک خامخا وښیو، بیا به یې یقیناً د هغوی له مخونو وپېژنئ، او د خبرو له انداز به یې (هم) خامخا وپېژنئ؛ او الله ستاسو په عملونو پوهېږي.
p وَلَنَبْلُوَنَّكُمْ حَتَّىٰ نَعْلَمَ ٱلْمُجَٰهِدِينَ مِنكُمْ وَٱلصَّٰبِرِينَ وَنَبْلُوَا۟ أَخْبَارَكُمْ ﴿٣١﴾ اور ہم تمھیں ضرور آزمائیں گے یہاں تک کہ ہم ظاہر کر دیں تم میں سے جہاد کرنے والوں اور صبر کرنے والوں کو اور ہم جانچ لیں تمھارے حالات۔ || وَلَنَبْلُوَنَّكُمْ حَتَّىٰ نَعْلَمَ ٱلْمُجَٰهِدِينَ مِنكُمْ وَٱلصَّٰبِرِينَ وَنَبْلُوَا۟ أَخْبَارَكُمْ ﴿٣١﴾ And verily We shall try you till We know those of you who strive hard (for the cause of Allah) and the steadfast, and till We test your record. (47:31) || وَلَنَبْلُوَنَّكُمْ حَتَّىٰ نَعْلَمَ ٱلْمُجَٰهِدِينَ مِنكُمْ وَٱلصَّٰبِرِينَ وَنَبْلُوَا۟ أَخْبَارَكُمْ ﴿٣١﴾ او موږ به تاسو خامخا وازمویو تر دې چې له تاسو جهاد کوونکي او صبر کوونکي ښکاره کړو، او ستاسو حالات وڅېړو.
