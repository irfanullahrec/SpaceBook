p وَإِذْ أَوْحَيْتُ إِلَى ٱلْحَوَارِيِّۦنَ أَنْ ءَامِنُوا۟ بِى وَبِرَسُولِى قَالُوٓا۟ ءَامَنَّا وَٱشْهَدْ بِأَنَّنَا مُسْلِمُونَ ﴿١١١﴾ اور جب میں نے حواریوں کے دل میں (یہ) ڈالا کہ مجھ پر اور میرے رسول (عیسیٰ علیہ السلام) پر ایمان لاؤ تو انھوں نے کہا کہ ہم ایمان لائے اور (اے اللہ!) تو گواہ رہ کہ بے شک ہم فرماں بردار ہیں۔ || وَإِذْ أَوْحَيْتُ إِلَى ٱلْحَوَارِيِّۦنَ أَنْ ءَامِنُوا۟ بِى وَبِرَسُولِى قَالُوٓا۟ ءَامَنَّا وَٱشْهَدْ بِأَنَّنَا مُسْلِمُونَ ﴿١١١﴾ And when I inspired the disciples, (saying): Believe in Me and in My messenger, they said: We believe. Bear witness that we have surrendered (unto Thee) "we are muslims". (5:111) || وَإِذْ أَوْحَيْتُ إِلَى ٱلْحَوَارِيِّۦنَ أَنْ ءَامِنُوا۟ بِى وَبِرَسُولِى قَالُوٓا۟ ءَامَنَّا وَٱشْهَدْ بِأَنَّنَا مُسْلِمُونَ ﴿١١١﴾ او کله چې ما د حواریانو په زړه کې (دا) واچول چې پر ما او زما پر رسول (عیسی علیه السلام) ایمان راوړئ، نو هغوی وویل: «موږ ایمان راوړ، او (ای الله!) شاهد اوسه چې بې‌شکه موږ فرمانبردار یو.»
p إِذْ قَالَ ٱلْحَوَارِيُّونَ يَٰعِيسَى ٱبْنَ مَرْيَمَ هَلْ يَسْتَطِيعُ رَبُّكَ أَن يُنَزِّلَ عَلَيْنَا مَآئِدَةًۭ مِّنَ ٱلسَّمَآءِ ۖ قَالَ ٱتَّقُوا۟ ٱللَّهَ إِن كُنتُم مُّؤْمِنِينَ ﴿١١٢﴾ جب حواریوں نے کہا تھا اے عیسیٰ ابنِ مریم! کیا آپ کا رب ایسا کر سکتا ہے کہ ہم پر آسمان سے (کھانے کا) ایک خوان اتار دے عیسیٰ (علیہ السلام) نے فرمایا اللہ (کی نافرمانی) سے ڈرو اگر تم مومن ہو۔ || إِذْ قَالَ ٱلْحَوَارِيُّونَ يَٰعِيسَى ٱبْنَ مَرْيَمَ هَلْ يَسْتَطِيعُ رَبُّكَ أَن يُنَزِّلَ عَلَيْنَا مَآئِدَةًۭ مِّنَ ٱلسَّمَآءِ ۖ قَالَ ٱتَّقُوا۟ ٱللَّهَ إِن كُنتُم مُّؤْمِنِينَ ﴿١١٢﴾ When the disciples said: O Jesus, son of Mary! Is thy Lord able to send down for us a table spread with food from heaven? He said: Observe your duty to Allah, if ye are true believers. (5:112) || إِذْ قَالَ ٱلْحَوَارِيُّونَ يَٰعِيسَى ٱبْنَ مَرْيَمَ هَلْ يَسْتَطِيعُ رَبُّكَ أَن يُنَزِّلَ عَلَيْنَا مَآئِدَةًۭ مِّنَ ٱلسَّمَآءِ ۖ قَالَ ٱتَّقُوا۟ ٱللَّهَ إِن كُنتُم مُّؤْمِنِينَ ﴿١١٢﴾ کله چې حواریانو ویلي وو: «ای عیسی ابن مریم! ایا ستا رب دا کولی شي چې پر موږ له اسمانه د (خوړو) یو دسترخوان راکوز کړي؟» عیسی (علیه السلام) وویل: «له الله (د نافرمانۍ) ووېرېږئ که مؤمنان یاست.»
p قَالُوا۟ نُرِيدُ أَن نَّأْكُلَ مِنْهَا وَتَطْمَئِنَّ قُلُوبُنَا وَنَعْلَمَ أَن قَدْ صَدَقْتَنَا وَنَكُونَ عَلَيْهَا مِنَ ٱلشَّٰهِدِينَ ﴿١١٣﴾ انھوں نے کہا ہم تو چاہتے ہیں کہ اُس میں سے کھائیں اور ہمارے دل (پوری طرح) مطمئن ہو جائیں اور ہم (مزید یقین کے ساتھ) جان لیں کہ آپ نے ہم سے سچ کہا تھا اور ہم اس پر گواہی دینے والوں میں سے ہو جائیں۔ || قَالُوا۟ نُرِيدُ أَن نَّأْكُلَ مِنْهَا وَتَطْمَئِنَّ قُلُوبُنَا وَنَعْلَمَ أَن قَدْ صَدَقْتَنَا وَنَكُونَ عَلَيْهَا مِنَ ٱلشَّٰهِدِينَ ﴿١١٣﴾ (They said:) We wish to eat thereof, that we may satisfy our hearts and know that thou hast spoken truth to us, and that thereof we may be witnesses. (5:113) || قَالُوا۟ نُرِيدُ أَن نَّأْكُلَ مِنْهَا وَتَطْمَئِنَّ قُلُوبُنَا وَنَعْلَمَ أَن قَدْ صَدَقْتَنَا وَنَكُونَ عَلَيْهَا مِنَ ٱلشَّٰهِدِينَ ﴿١١٣﴾ هغوی وویل: «موږ خو غواړو چې له هغه وخورو، زړونه مو (بشپړ) ډاډه شي، او (په لا یقین) پوه شو چې تا موږ ته رښتیا ویلي وو، او پر هغه له شاهدي ورکوونکو شو.»
p قَالَ عِيسَى ٱبْنُ مَرْيَمَ ٱللَّهُمَّ رَبَّنَآ أَنزِلْ عَلَيْنَا مَآئِدَةًۭ مِّنَ ٱلسَّمَآءِ تَكُونُ لَنَا عِيدًۭا لِّأَوَّلِنَا وَءَاخِرِنَا وَءَايَةًۭ مِّنكَ ۖ وَٱرْزُقْنَا وَأَنتَ خَيْرُ ٱلرَّٰزِقِينَ ﴿١١٤﴾ عیسیٰ ابنِ مریم (علیہ السلام) نے عرض کیا اے اللہ! اے ہمارے رب! ہم پر آسمان سے (کھانے کا) خوان اتار دے تاکہ وہ (دن) ہمارے اگلوں اور پچھلوں کے لیے عید ہو جائے اور تیری طرف سے ایک نشانی بن جائے اور ہمیں رزق عطا فرما اور تو بہترین رزق دینے والا ہے۔ || قَالَ عِيسَى ٱبْنُ مَرْيَمَ ٱللَّهُمَّ رَبَّنَآ أَنزِلْ عَلَيْنَا مَآئِدَةًۭ مِّنَ ٱلسَّمَآءِ تَكُونُ لَنَا عِيدًۭا لِّأَوَّلِنَا وَءَاخِرِنَا وَءَايَةًۭ مِّنكَ ۖ وَٱرْزُقْنَا وَأَنتَ خَيْرُ ٱلرَّٰزِقِينَ ﴿١١٤﴾ Jesus, son of Mary, said: O Allah, Lord of us! Send down for us a table spread with food from heaven, that it may be a feast for us, for the first of us and for the last of us, and a sign from Thee. Give us sustenance, for Thou art the Best of Sustainers. (5:114) || قَالَ عِيسَى ٱبْنُ مَرْيَمَ ٱللَّهُمَّ رَبَّنَآ أَنزِلْ عَلَيْنَا مَآئِدَةًۭ مِّنَ ٱلسَّمَآءِ تَكُونُ لَنَا عِيدًۭا لِّأَوَّلِنَا وَءَاخِرِنَا وَءَايَةًۭ مِّنكَ ۖ وَٱرْزُقْنَا وَأَنتَ خَيْرُ ٱلرَّٰزِقِينَ ﴿١١٤﴾ عیسی ابن مریم (علیه السلام) عرض وکړ: «ای الله! ای زموږ ربه! پر موږ له اسمانه د (خوړو) دسترخوان راکوز کړه، تر څو هغه (ورځ) زموږ د مخکېنیو او وروستیو لپاره اختر شي او ستا له لوري یوه نښه شي؛ او موږ ته رزق راکړه، او ته غوره رزق ورکوونکی یې.»
p قَالَ ٱللَّهُ إِنِّى مُنَزِّلُهَا عَلَيْكُمْ ۖ فَمَن يَكْفُرْ بَعْدُ مِنكُمْ فَإِنِّىٓ أُعَذِّبُهُۥ عَذَابًۭا لَّآ أُعَذِّبُهُۥٓ أَحَدًۭا مِّنَ ٱلْعَٰلَمِينَ ﴿١١٥﴾ اللہ نے فرمایا بے شک میں تم پر اس (خوان) کو نازل کرنے والا ہوں لیکن اس کے بعد تم میں سے جس نے کفر کیا تو بے شک میں اُسے ایسا عذاب دوں گا کہ تمام جہانوں میں سے ایسا عذاب کسی کو بھی نہیں دوں گا۔ || قَالَ ٱللَّهُ إِنِّى مُنَزِّلُهَا عَلَيْكُمْ ۖ فَمَن يَكْفُرْ بَعْدُ مِنكُمْ فَإِنِّىٓ أُعَذِّبُهُۥ عَذَابًۭا لَّآ أُعَذِّبُهُۥٓ أَحَدًۭا مِّنَ ٱلْعَٰلَمِينَ ﴿١١٥﴾ Allah said: Lo! I send it down for you. And whoso disbelieveth of you afterward, him surely will I punish with a punishment wherewith I have not punished any of (My) creatures. (5:115) || قَالَ ٱللَّهُ إِنِّى مُنَزِّلُهَا عَلَيْكُمْ ۖ فَمَن يَكْفُرْ بَعْدُ مِنكُمْ فَإِنِّىٓ أُعَذِّبُهُۥ عَذَابًۭا لَّآ أُعَذِّبُهُۥٓ أَحَدًۭا مِّنَ ٱلْعَٰلَمِينَ ﴿١١٥﴾ الله وفرمایل: «بې‌شکه زه دا (دسترخوان) پر تاسو نازلوونکی یم؛ خو له دې وروسته له تاسو چا چې کفر وکړ، نو بې‌شکه زه به هغه ته داسې عذاب ورکړم چې د ټولو جهانونو هېچا ته به داسې عذاب ورنه کړم.»
p وَإِذْ قَالَ ٱللَّهُ يَٰعِيسَى ٱبْنَ مَرْيَمَ ءَأَنتَ قُلْتَ لِلنَّاسِ ٱتَّخِذُونِى وَأُمِّىَ إِلَٰهَيْنِ مِن دُونِ ٱللَّهِ ۖ قَالَ سُبْحَٰنَكَ مَا يَكُونُ لِىٓ أَنْ أَقُولَ مَا لَيْسَ لِى بِحَقٍّ ۚ إِن كُنتُ قُلْتُهُۥ فَقَدْ عَلِمْتَهُۥ ۚ تَعْلَمُ مَا فِى نَفْسِى وَلَآ أَعْلَمُ مَا فِى نَفْسِكَ ۚ إِنَّكَ أَنتَ عَلَّٰمُ ٱلْغُيُوبِ ﴿١١٦﴾ اور جب اللہ فرمائے گا اے عیسیٰ ابنِ مریم! کیا تم نے لوگوں سے کہا تھا کہ مجھے اور میری والدہ کو اللہ کے سوا دو معبود بنا لو عیسیٰ (علیہ السلام) عرض کریں گے تُو (ہر عیب سے) پاک ہے مجھے یہ لائق نہیں تھا کہ میں ایسی بات کہوں جس کا مجھے کوئی حق ہی نہیں اگر میں نے ایسی بات کہی ہوتی تو تُو اسے ضرور جانتا تُو جانتا ہے جو میرے دل میں پوشیدہ ہے اور میں تیری پوشیدہ باتوں کو نہیں جانتا بے شک تو غیب کی ساری باتوں کو خوب جاننے والا ہے۔ || وَإِذْ قَالَ ٱللَّهُ يَٰعِيسَى ٱبْنَ مَرْيَمَ ءَأَنتَ قُلْتَ لِلنَّاسِ ٱتَّخِذُونِى وَأُمِّىَ إِلَٰهَيْنِ مِن دُونِ ٱللَّهِ ۖ قَالَ سُبْحَٰنَكَ مَا يَكُونُ لِىٓ أَنْ أَقُولَ مَا لَيْسَ لِى بِحَقٍّ ۚ إِن كُنتُ قُلْتُهُۥ فَقَدْ عَلِمْتَهُۥ ۚ تَعْلَمُ مَا فِى نَفْسِى وَلَآ أَعْلَمُ مَا فِى نَفْسِكَ ۚ إِنَّكَ أَنتَ عَلَّٰمُ ٱلْغُيُوبِ ﴿١١٦﴾ And when Allah saith: O Jesus, son of Mary! Didst thou say unto mankind: Take me and my mother for two gods beside Allah? he saith: Be glorified! It was not mine to utter that to which I had no right. If I used to say it, then Thou knewest it. Thou knowest what is in my mind, and I know not what is in Thy Mind. Lo! Thou, only Thou, art the Knower of Things Hidden? (5:116) || وَإِذْ قَالَ ٱللَّهُ يَٰعِيسَى ٱبْنَ مَرْيَمَ ءَأَنتَ قُلْتَ لِلنَّاسِ ٱتَّخِذُونِى وَأُمِّىَ إِلَٰهَيْنِ مِن دُونِ ٱللَّهِ ۖ قَالَ سُبْحَٰنَكَ مَا يَكُونُ لِىٓ أَنْ أَقُولَ مَا لَيْسَ لِى بِحَقٍّ ۚ إِن كُنتُ قُلْتُهُۥ فَقَدْ عَلِمْتَهُۥ ۚ تَعْلَمُ مَا فِى نَفْسِى وَلَآ أَعْلَمُ مَا فِى نَفْسِكَ ۚ إِنَّكَ أَنتَ عَلَّٰمُ ٱلْغُيُوبِ ﴿١١٦﴾ او کله چې الله ووایي: «ای عیسی ابن مریم! ایا تا خلکو ته ویلي وو چې ما او زما مور له الله پرته دوه معبودان ونیسئ؟» عیسی (علیه السلام) به عرض وکړي: «ته (له هر عیبه) پاک یې؛ ما ته دا نه ښایېده چې داسې خبره ووایم چې هېڅ حق یې نه لرم؛ که ما داسې خبره کړې وای، نو ته به خامخا پرې پوهېدې؛ ته پوهېږې چې زما په زړه کې څه پټ دي، او زه ستا په پټو خبرو نه پوهېږم؛ بې‌شکه ته د غیب په ټولو خبرو ښه پوه یې.»
