p يَعِدُهُمْ وَيُمَنِّيهِمْ ۖ وَمَا يَعِدُهُمُ ٱلشَّيْطَٰنُ إِلَّا غُرُورًا ﴿١٢٠﴾ وہ (شیطان) اُن سے وعدہ کرتا ہے اور انھیں امیدیں دلاتا ہے اور شیطان اُن سے صرف دھوکہ کا وعدہ کرتا ہے۔ || يَعِدُهُمْ وَيُمَنِّيهِمْ ۖ وَمَا يَعِدُهُمُ ٱلشَّيْطَٰنُ إِلَّا غُرُورًا ﴿١٢٠﴾ He promiseth them and stirreth up desires in them, and Satan promiseth them only to beguile. (4:120) || يَعِدُهُمْ وَيُمَنِّيهِمْ ۖ وَمَا يَعِدُهُمُ ٱلشَّيْطَٰنُ إِلَّا غُرُورًا ﴿١٢٠﴾ هغه (شیطان) له هغوی سره ژمنې کوي او هیلې ورکوي، او شیطان له هغوی سره یوازې د دوکې ژمنه کوي.
p أُو۟لَٰٓئِكَ مَأْوَىٰهُمْ جَهَنَّمُ وَلَا يَجِدُونَ عَنْهَا مَحِيصًۭا ﴿١٢١﴾ یہ وہ لوگ ہیں جن کا ٹھکانا جہنم ہے اور وہ اس (جہنم) سے بچ نکلنے کی کوئی جگہ نہیں پائیں گے۔ || أُو۟لَٰٓئِكَ مَأْوَىٰهُمْ جَهَنَّمُ وَلَا يَجِدُونَ عَنْهَا مَحِيصًۭا ﴿١٢١﴾ For such, their habitation will be hell, and they will find no refuge therefrom. (4:121) || أُو۟لَٰٓئِكَ مَأْوَىٰهُمْ جَهَنَّمُ وَلَا يَجِدُونَ عَنْهَا مَحِيصًۭا ﴿١٢١﴾ دا هغه خلک دي چې ټیکانه یې دوزخ دی، او له هغه (دوزخ) د تېښتې هېڅ ځای به ونه مومي.
p وَٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّٰلِحَٰتِ سَنُدْخِلُهُمْ جَنَّٰتٍۢ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَٰرُ خَٰلِدِينَ فِيهَآ أَبَدًۭا ۖ وَعْدَ ٱللَّهِ حَقًّۭا ۚ وَمَنْ أَصْدَقُ مِنَ ٱللَّهِ قِيلًۭا ﴿١٢٢﴾ اور جو لوگ ایمان لائے اور انھوں نے نیک عمل کیے ہم عنقریب انھیں ایسے باغات میں داخل فرمائیں گے جن کے نیچے نہریں بہتی ہیں وہ ان میں ہمیشہ ہمیشہ رہیں گے اللہ کا وعدہ سچا ہے اور کون ہے جو اللہ سے زیادہ بات کا سچا ہو؟ || وَٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّٰلِحَٰتِ سَنُدْخِلُهُمْ جَنَّٰتٍۢ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَٰرُ خَٰلِدِينَ فِيهَآ أَبَدًۭا ۖ وَعْدَ ٱللَّهِ حَقًّۭا ۚ وَمَنْ أَصْدَقُ مِنَ ٱللَّهِ قِيلًۭا ﴿١٢٢﴾ But as for those who believe and do good works We shall bring them into Gardens underneath which rivers flow, wherein they will abide for ever. It is a promise from Allah in truth; and who can be more truthful than Allah in utterance? (4:122) || وَٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّٰلِحَٰتِ سَنُدْخِلُهُمْ جَنَّٰتٍۢ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَٰرُ خَٰلِدِينَ فِيهَآ أَبَدًۭا ۖ وَعْدَ ٱللَّهِ حَقًّۭا ۚ وَمَنْ أَصْدَقُ مِنَ ٱللَّهِ قِيلًۭا ﴿١٢٢﴾ او هغه کسان چې ایمان یې راوړ او نېک عملونه یې وکړل، ژر به یې داسې باغونو ته ننباسو چې لاندې یې ویالې بهېږي؛ هغوی به پکې تل تر تله وي؛ د الله ژمنه رښتیا ده؛ او څوک دی چې له الله په خبره کې رښتینی وي؟
p لَّيْسَ بِأَمَانِيِّكُمْ وَلَآ أَمَانِىِّ أَهْلِ ٱلْكِتَٰبِ ۗ مَن يَعْمَلْ سُوٓءًۭا يُجْزَ بِهِۦ وَلَا يَجِدْ لَهُۥ مِن دُونِ ٱللَّهِ وَلِيًّۭا وَلَا نَصِيرًۭا ﴿١٢٣﴾ (نجات کا انحصار) نہ تمھاری آرزوؤں پر ہے اور نہ اہلِ کتاب کی آرزوؤں پر جو بُرا کام کرے گا اُسے اُس کا بدلہ دیا جائے گا اور وہ اپنے لیے اللہ کے سوا نہ ہی کوئی حمایتی پائے گا اور نہ ہی کوئی مدد گار۔ || لَّيْسَ بِأَمَانِيِّكُمْ وَلَآ أَمَانِىِّ أَهْلِ ٱلْكِتَٰبِ ۗ مَن يَعْمَلْ سُوٓءًۭا يُجْزَ بِهِۦ وَلَا يَجِدْ لَهُۥ مِن دُونِ ٱللَّهِ وَلِيًّۭا وَلَا نَصِيرًۭا ﴿١٢٣﴾ It will not be in accordance with your desires, nor the desires of the People of the Scripture. He who doeth wrong will have the recompense thereof, and will not find against Allah any protecting friend or helper. (4:123) || لَّيْسَ بِأَمَانِيِّكُمْ وَلَآ أَمَانِىِّ أَهْلِ ٱلْكِتَٰبِ ۗ مَن يَعْمَلْ سُوٓءًۭا يُجْزَ بِهِۦ وَلَا يَجِدْ لَهُۥ مِن دُونِ ٱللَّهِ وَلِيًّۭا وَلَا نَصِيرًۭا ﴿١٢٣﴾ (د نجات تکیه) نه ستاسو پر هیلو ده او نه د اهل کتابو پر هیلو؛ څوک چې بد کار وکړي، بدله به یې ورکړل شي، او هغه به ځان ته له الله پرته نه کوم ملاتړی ومومي او نه مرستندوی.
p وَمَن يَعْمَلْ مِنَ ٱلصَّٰلِحَٰتِ مِن ذَكَرٍ أَوْ أُنثَىٰ وَهُوَ مُؤْمِنٌۭ فَأُو۟لَٰٓئِكَ يَدْخُلُونَ ٱلْجَنَّةَ وَلَا يُظْلَمُونَ نَقِيرًۭا ﴿١٢٤﴾ اور جس نے نیک عمل کیا خواہ وہ مرد ہو یا عورت اور وہ مومن ہو تو یہ لوگ جنت میں داخل ہوں گے اور اُن پر ایک تل کے برابر ظلم نہیں کیا جائے گا۔ || وَمَن يَعْمَلْ مِنَ ٱلصَّٰلِحَٰتِ مِن ذَكَرٍ أَوْ أُنثَىٰ وَهُوَ مُؤْمِنٌۭ فَأُو۟لَٰٓئِكَ يَدْخُلُونَ ٱلْجَنَّةَ وَلَا يُظْلَمُونَ نَقِيرًۭا ﴿١٢٤﴾ And whoso doeth good works, whether of male or female, and he (or she) is a believer, such will enter paradise and they will not be wronged the dint in a date-stone. (4:124) || وَمَن يَعْمَلْ مِنَ ٱلصَّٰلِحَٰتِ مِن ذَكَرٍ أَوْ أُنثَىٰ وَهُوَ مُؤْمِنٌۭ فَأُو۟لَٰٓئِكَ يَدْخُلُونَ ٱلْجَنَّةَ وَلَا يُظْلَمُونَ نَقِيرًۭا ﴿١٢٤﴾ او چا چې نېک عمل وکړ — که نارینه وي که ښځه — او مؤمن وي، نو دا خلک به جنت ته ننوځي او پر هغوی به د کونجي د غوټې په اندازه ظلم ونه شي.
p وَمَنْ أَحْسَنُ دِينًۭا مِّمَّنْ أَسْلَمَ وَجْهَهُۥ لِلَّهِ وَهُوَ مُحْسِنٌۭ وَٱتَّبَعَ مِلَّةَ إِبْرَٰهِيمَ حَنِيفًۭا ۗ وَٱتَّخَذَ ٱللَّهُ إِبْرَٰهِيمَ خَلِيلًۭا ﴿١٢٥﴾ اور دین کے لحاظ سے کون اس شخص سے بہتر ہو سکتا ہے جس نے اپنا رخ اللہ کے سامنے جھکا دیا اور وہ نیک بھی ہو اور اُس نے ابراہیم (علیہ السلام) کی ملّت کی پیروی کی جو (ہر باطل سے جدا اللہ کے لیے) یکسو تھے اور اللہ نے ابراہیم (علیہ السلام) کو (خاص) دوست بنا لیا۔ || وَمَنْ أَحْسَنُ دِينًۭا مِّمَّنْ أَسْلَمَ وَجْهَهُۥ لِلَّهِ وَهُوَ مُحْسِنٌۭ وَٱتَّبَعَ مِلَّةَ إِبْرَٰهِيمَ حَنِيفًۭا ۗ وَٱتَّخَذَ ٱللَّهُ إِبْرَٰهِيمَ خَلِيلًۭا ﴿١٢٥﴾ Who is better in religion than he who surrendereth his purpose to Allah while doing good (to men) and followeth the tradition of Abraham, the upright? Allah (Himself) chose Abraham for friend. (4:125) || وَمَنْ أَحْسَنُ دِينًۭا مِّمَّنْ أَسْلَمَ وَجْهَهُۥ لِلَّهِ وَهُوَ مُحْسِنٌۭ وَٱتَّبَعَ مِلَّةَ إِبْرَٰهِيمَ حَنِيفًۭا ۗ وَٱتَّخَذَ ٱللَّهُ إِبْرَٰهِيمَ خَلِيلًۭا ﴿١٢٥﴾ او د دین له پلوه له هغه چا څوک غوره کېدای شي چې خپل مخ یې الله ته ټیټ کړی، نېکوکار هم وي او د ابراهیم (علیه السلام) د ملت پیروي یې کړې وي چې (له هر باطله جلا، الله ته) یکسو و؟ او الله ابراهیم (علیه السلام) (ځانګړی) دوست نیولی و.
p وَلِلَّهِ مَا فِى ٱلسَّمَٰوَٰتِ وَمَا فِى ٱلْأَرْضِ ۚ وَكَانَ ٱللَّهُ بِكُلِّ شَىْءٍۢ مُّحِيطًۭا ﴿١٢٦﴾ اور اللہ کے لیے ہے جو کچھ آسمانوں میں ہے اور جو کچھ زمینوں میں ہے اور اللہ ہر چیز کا احاطہ کرنے والا ہے۔ || وَلِلَّهِ مَا فِى ٱلسَّمَٰوَٰتِ وَمَا فِى ٱلْأَرْضِ ۚ وَكَانَ ٱللَّهُ بِكُلِّ شَىْءٍۢ مُّحِيطًۭا ﴿١٢٦﴾ Unto Allah belongeth whatsoever is in the heavens and whatsoever is in the earth. Allah ever surroundeth all things. (4:126) || وَلِلَّهِ مَا فِى ٱلسَّمَٰوَٰتِ وَمَا فِى ٱلْأَرْضِ ۚ وَكَانَ ٱللَّهُ بِكُلِّ شَىْءٍۢ مُّحِيطًۭا ﴿١٢٦﴾ او د الله دي هغه څه چې په آسمانونو کې دي او هغه څه چې په ځمکه کې دي؛ او الله پر هر شي احاطه لرونکی دی.
