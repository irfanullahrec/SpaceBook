p إِنَّ ٱلَّذِينَ كَفَرُوا۟ وَصَدُّوا۟ عَن سَبِيلِ ٱللَّهِ وَشَآقُّوا۟ ٱلرَّسُولَ مِنۢ بَعْدِ مَا تَبَيَّنَ لَهُمُ ٱلْهُدَىٰ لَن يَضُرُّوا۟ ٱللَّهَ شَيْـًۭٔا وَسَيُحْبِطُ أَعْمَٰلَهُمْ ﴿٣٢﴾ بے شک جن لوگوں نے کفر کیا اور اللہ کی راہ سے روکا اور رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کی مخالفت کی بعد اس کے کہ اُن کے لیے ہدایت خوب واضح ہو چکی ہے وہ ہرگز اللہ کو کوئی نقصان نہیں پہنچا سکیں گے اور عنقریب وہ (اللہ) اُن کے اعمال اکارت کر دے گا۔ || إِنَّ ٱلَّذِينَ كَفَرُوا۟ وَصَدُّوا۟ عَن سَبِيلِ ٱللَّهِ وَشَآقُّوا۟ ٱلرَّسُولَ مِنۢ بَعْدِ مَا تَبَيَّنَ لَهُمُ ٱلْهُدَىٰ لَن يَضُرُّوا۟ ٱللَّهَ شَيْـًۭٔا وَسَيُحْبِطُ أَعْمَٰلَهُمْ ﴿٣٢﴾ Lo! those who disbelieve and turn from the way of Allah and oppose the messenger after the guidance hath been manifested unto them, they hurt Allah not a jot, and He will make their actions fruitless. (47:32) || إِنَّ ٱلَّذِينَ كَفَرُوا۟ وَصَدُّوا۟ عَن سَبِيلِ ٱللَّهِ وَشَآقُّوا۟ ٱلرَّسُولَ مِنۢ بَعْدِ مَا تَبَيَّنَ لَهُمُ ٱلْهُدَىٰ لَن يَضُرُّوا۟ ٱللَّهَ شَيْـًۭٔا وَسَيُحْبِطُ أَعْمَٰلَهُمْ ﴿٣٢﴾ بې‌شکه هغه کسان چې کفر یې وکړ، د الله له لارې یې منع کړل، او د رسول (صلی الله علیه وعلی آله واصحابه وسلم) مخالفت یې وکړ وروسته له دې چې هدایت ورته ښه روښانه شوی و، هغوی به الله ته هېڅکله هېڅ زیان ونه رسوي، او ژر به هغه (الله) د هغوی عملونه برباد کړي.
p ۞ يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓا۟ أَطِيعُوا۟ ٱللَّهَ وَأَطِيعُوا۟ ٱلرَّسُولَ وَلَا تُبْطِلُوٓا۟ أَعْمَٰلَكُمْ ﴿٣٣﴾ اے ایمان والو! اللہ کی اطاعت کرو اور رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کی اطاعت کرو اور اپنے اعمال ضائع مت کرو۔ || ۞ يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓا۟ أَطِيعُوا۟ ٱللَّهَ وَأَطِيعُوا۟ ٱلرَّسُولَ وَلَا تُبْطِلُوٓا۟ أَعْمَٰلَكُمْ ﴿٣٣﴾ O ye who believe! Obey Allah and obey the messenger, and render not your actions vain. (47:33) || ۞ يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓا۟ أَطِيعُوا۟ ٱللَّهَ وَأَطِيعُوا۟ ٱلرَّسُولَ وَلَا تُبْطِلُوٓا۟ أَعْمَٰلَكُمْ ﴿٣٣﴾ ای مؤمنانو! د الله اطاعت وکړئ او د رسول (صلی الله علیه وعلی آله واصحابه وسلم) اطاعت وکړئ، او خپل عملونه مه ضایع کوئ.
p إِنَّ ٱلَّذِينَ كَفَرُوا۟ وَصَدُّوا۟ عَن سَبِيلِ ٱللَّهِ ثُمَّ مَاتُوا۟ وَهُمْ كُفَّارٌۭ فَلَن يَغْفِرَ ٱللَّهُ لَهُمْ ﴿٣٤﴾ بے شک جن لوگوں نے کفر کیا اور (لوگوں کو) اللہ کی راہ سے روکا پھر اس حال میں مر گئے کہ وہ کافر تھے تو اللہ ہرگز اُن کو نہیں بخشے گا۔ || إِنَّ ٱلَّذِينَ كَفَرُوا۟ وَصَدُّوا۟ عَن سَبِيلِ ٱللَّهِ ثُمَّ مَاتُوا۟ وَهُمْ كُفَّارٌۭ فَلَن يَغْفِرَ ٱللَّهُ لَهُمْ ﴿٣٤﴾ Lo! those who disbelieve and turn from the way of Allah and then die disbelievers, Allah surely will not pardon them. (47:34) || إِنَّ ٱلَّذِينَ كَفَرُوا۟ وَصَدُّوا۟ عَن سَبِيلِ ٱللَّهِ ثُمَّ مَاتُوا۟ وَهُمْ كُفَّارٌۭ فَلَن يَغْفِرَ ٱللَّهُ لَهُمْ ﴿٣٤﴾ بې‌شکه هغه کسان چې کفر یې وکړ او (خلک) یې د الله له لارې منع کړل، بیا په داسې حال مړه شول چې کافران وو، نو الله به یې هېڅکله ونه بښي.
p فَلَا تَهِنُوا۟ وَتَدْعُوٓا۟ إِلَى ٱلسَّلْمِ وَأَنتُمُ ٱلْأَعْلَوْنَ وَٱللَّهُ مَعَكُمْ وَلَن يَتِرَكُمْ أَعْمَٰلَكُمْ ﴿٣٥﴾ تو (مسلمانو!) ہمت نہ ہارو اور نہ صلح کی طرف بلاؤ اور تم ہی غالب رہو گے اور اللہ تمھارے ساتھ ہے اور وہ ہرگز تمھارے اعمال ضائع نہیں کرے گا۔ || فَلَا تَهِنُوا۟ وَتَدْعُوٓا۟ إِلَى ٱلسَّلْمِ وَأَنتُمُ ٱلْأَعْلَوْنَ وَٱللَّهُ مَعَكُمْ وَلَن يَتِرَكُمْ أَعْمَٰلَكُمْ ﴿٣٥﴾ So do not falter and cry out for peace when ye (will be) the uppermost, and Allah is with you, and He will not grudge (the reward of) your actions. (47:35) || فَلَا تَهِنُوا۟ وَتَدْعُوٓا۟ إِلَى ٱلسَّلْمِ وَأَنتُمُ ٱلْأَعْلَوْنَ وَٱللَّهُ مَعَكُمْ وَلَن يَتِرَكُمْ أَعْمَٰلَكُمْ ﴿٣٥﴾ نو (ای مسلمانانو!) همت مه بایلئ او سولې ته مه رابولئ، او تاسو به غالب یاست، او الله له تاسو سره دی، او هغه به ستاسو عملونه هېڅکله ضایع نه کړي.
p إِنَّمَا ٱلْحَيَوٰةُ ٱلدُّنْيَا لَعِبٌۭ وَلَهْوٌۭ ۚ وَإِن تُؤْمِنُوا۟ وَتَتَّقُوا۟ يُؤْتِكُمْ أُجُورَكُمْ وَلَا يَسْـَٔلْكُمْ أَمْوَٰلَكُمْ ﴿٣٦﴾ دنیا کی زندگی تو صرف کھیل اور تماشا ہے اور اگر تم ایمان لاؤ اور اللہ (کی نافرمانی) سے ڈرو، تو وہ تمھیں تمھارے اجر عطا فرمائے گا اور تم سے تمھارا (سارا) مال طلب نہیں کرے گا۔ || إِنَّمَا ٱلْحَيَوٰةُ ٱلدُّنْيَا لَعِبٌۭ وَلَهْوٌۭ ۚ وَإِن تُؤْمِنُوا۟ وَتَتَّقُوا۟ يُؤْتِكُمْ أُجُورَكُمْ وَلَا يَسْـَٔلْكُمْ أَمْوَٰلَكُمْ ﴿٣٦﴾ The life of the world is but a sport and a pastime. And if ye believe and ward off (evil). He will give you your wages, and will not ask of you your wordly wealth. (47:36) || إِنَّمَا ٱلْحَيَوٰةُ ٱلدُّنْيَا لَعِبٌۭ وَلَهْوٌۭ ۚ وَإِن تُؤْمِنُوا۟ وَتَتَّقُوا۟ يُؤْتِكُمْ أُجُورَكُمْ وَلَا يَسْـَٔلْكُمْ أَمْوَٰلَكُمْ ﴿٣٦﴾ د دنیا ژوند خو یوازې لوبه او تماشه ده؛ او که ایمان راوړئ او له الله (د نافرمانۍ) ووېرېږئ، نو هغه به تاسو ته ستاسو اجرونه درکړي او له تاسو به ستاسو (ټول) مال ونه غواړي.
p إِن يَسْـَٔلْكُمُوهَا فَيُحْفِكُمْ تَبْخَلُوا۟ وَيُخْرِجْ أَضْغَٰنَكُمْ ﴿٣٧﴾ اگر وہ تم سے (سارا) مال طلب کرے پھر تم سے اصرار کرے تو تم بخل کرو گے اور وہ تمھارے (دلوں کے) کھوٹ ظاہر کر دے گا۔ || إِن يَسْـَٔلْكُمُوهَا فَيُحْفِكُمْ تَبْخَلُوا۟ وَيُخْرِجْ أَضْغَٰنَكُمْ ﴿٣٧﴾ If He should ask it of you and importune you, ye would hoard it, and He would bring to light your (secret) hates. (47:37) || إِن يَسْـَٔلْكُمُوهَا فَيُحْفِكُمْ تَبْخَلُوا۟ وَيُخْرِجْ أَضْغَٰنَكُمْ ﴿٣٧﴾ که هغه له تاسو (ټول) مال وغواړي، بیا پرې ټینګار وکړي، نو تاسو به بخل وکړئ، او هغه به ستاسو د (زړونو) کینې ښکاره کړي.
p هَٰٓأَنتُمْ هَٰٓؤُلَآءِ تُدْعَوْنَ لِتُنفِقُوا۟ فِى سَبِيلِ ٱللَّهِ فَمِنكُم مَّن يَبْخَلُ ۖ وَمَن يَبْخَلْ فَإِنَّمَا يَبْخَلُ عَن نَّفْسِهِۦ ۚ وَٱللَّهُ ٱلْغَنِىُّ وَأَنتُمُ ٱلْفُقَرَآءُ ۚ وَإِن تَتَوَلَّوْا۟ يَسْتَبْدِلْ قَوْمًا غَيْرَكُمْ ثُمَّ لَا يَكُونُوٓا۟ أَمْثَٰلَكُم ﴿٣٨﴾ سنو! تم وہ لوگ ہو جنھیں اللہ کی راہ میں خرچ کے لیے بلایا جاتا ہے تو تم میں سے کچھ وہ ہیں جو بخل کرتے ہیں اور جو کوئی بھی بخل کرتا ہے تو بے شک وہ اپنے آپ ہی سے بخل کرتا ہے اور اللہ بے نیاز ہے اور تم محتاج ہو اور اگر تم (حکمِ الٰہی سے) منہ پھیرو گے تو وہ تمھاری جگہ بدل کر دوسری قوم کو لے آئے گا پھر وہ تمھارے جیسے نہ ہوں گے۔ || هَٰٓأَنتُمْ هَٰٓؤُلَآءِ تُدْعَوْنَ لِتُنفِقُوا۟ فِى سَبِيلِ ٱللَّهِ فَمِنكُم مَّن يَبْخَلُ ۖ وَمَن يَبْخَلْ فَإِنَّمَا يَبْخَلُ عَن نَّفْسِهِۦ ۚ وَٱللَّهُ ٱلْغَنِىُّ وَأَنتُمُ ٱلْفُقَرَآءُ ۚ وَإِن تَتَوَلَّوْا۟ يَسْتَبْدِلْ قَوْمًا غَيْرَكُمْ ثُمَّ لَا يَكُونُوٓا۟ أَمْثَٰلَكُم ﴿٣٨﴾ Lo! ye are those who are called to spend in the way of Allah, yet among you there are some who hoard. And as for him who hoardeth, he hoardeth only from his soul. And Allah is the Rich, and ye are the poor. And if ye turn away He will exchange you for some other folk, and they will not be the likes of you. (47:38) || هَٰٓأَنتُمْ هَٰٓؤُلَآءِ تُدْعَوْنَ لِتُنفِقُوا۟ فِى سَبِيلِ ٱللَّهِ فَمِنكُم مَّن يَبْخَلُ ۖ وَمَن يَبْخَلْ فَإِنَّمَا يَبْخَلُ عَن نَّفْسِهِۦ ۚ وَٱللَّهُ ٱلْغَنِىُّ وَأَنتُمُ ٱلْفُقَرَآءُ ۚ وَإِن تَتَوَلَّوْا۟ يَسْتَبْدِلْ قَوْمًا غَيْرَكُمْ ثُمَّ لَا يَكُونُوٓا۟ أَمْثَٰلَكُم ﴿٣٨﴾ واورئ! تاسو هغه خلک یاست چې د الله په لار کې د لګښت لپاره رابلل کېږئ، نو له تاسو ځینې هغه دي چې بخل کوي؛ او څوک چې بخل کوي، نو بې‌شکه له خپل ځان سره بخل کوي؛ او الله بې‌نیازه دی او تاسو محتاج یاست؛ او که (له الهي حکم) مخ واړوئ، نو ستاسو پر ځای به بل قوم راولي، بیا به هغوی ستاسو په څېر نه وي.
## علم و عمل کی باتیں || Matters of knowledge and action || د علم او عمل خبرې
- جو اللہ تعالیٰ کی راہ میں شہید ہو جاتے ہیں ان کی کوشش اور اعمال ضائع نہیں ہوں گے۔ اللہ تعالیٰ انھیں کامیابی کی منزل تک پہنچائے گا اور انھیں نعمتوں سے بھری جنت میں داخل فرمائے گا۔ || Those who are martyred in the way of Allah Almighty, their effort and deeds will not be wasted. Allah Almighty will bring them to the destination of success and will admit them into the Paradise full of blessings. || هغه کسان چې د الله تعالی په لار کې شهیدان شي، د هغوی هڅه او عملونه به ضایع نه شي؛ الله تعالی به یې د بریا تر منزل ورسوي او په نعمتونو ډک جنت ته به یې ننباسي.
- مسلمان اللہ تعالیٰ کے دین کی مدد کریں گے تو اللہ تعالیٰ ان کی مدد کرے گا، ان کے حوصلے بڑھائے گا اور انھیں ثابت قدم رہنے کی توفیق عطا فرمائے گا۔ || If the Muslims help the religion of Allah Almighty, Allah Almighty will help them, raise their courage and grant them the ability to remain steadfast. || که مسلمانان د الله تعالی د دین مرسته وکړي، نو الله تعالی به یې مرسته وکړي، حوصلې به یې لوړې کړي او د ثابت قدم پاتې کېدو توفیق به ورکړي.
