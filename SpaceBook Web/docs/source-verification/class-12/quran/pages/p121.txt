p سُنَّةَ ٱللَّهِ فِى ٱلَّذِينَ خَلَوْا۟ مِن قَبْلُ ۖ وَلَن تَجِدَ لِسُنَّةِ ٱللَّهِ تَبْدِيلًۭا ﴿٦٢﴾ یہی اللہ کی سنت رہی ہے اُن کے بارے میں جو پہلے گزر چکے ہیں اور آپ اللہ کی سنت میں ہرگز کوئی تبدیلی نہیں پائیں گے۔ || سُنَّةَ ٱللَّهِ فِى ٱلَّذِينَ خَلَوْا۟ مِن قَبْلُ ۖ وَلَن تَجِدَ لِسُنَّةِ ٱللَّهِ تَبْدِيلًۭا ﴿٦٢﴾ That was the way of Allah in the case of those who passed away of old; thou wilt not find for the way of Allah aught of power to change. (33:62) || سُنَّةَ ٱللَّهِ فِى ٱلَّذِينَ خَلَوْا۟ مِن قَبْلُ ۖ وَلَن تَجِدَ لِسُنَّةِ ٱللَّهِ تَبْدِيلًۭا ﴿٦٢﴾ همدا د الله سنت و د هغو کسانو په اړه چې مخکې تېر شوي، او تاسو به د الله په سنت کې هېڅکله کوم بدلون ونه مومئ.
p يَسْـَٔلُكَ ٱلنَّاسُ عَنِ ٱلسَّاعَةِ ۖ قُلْ إِنَّمَا عِلْمُهَا عِندَ ٱللَّهِ ۚ وَمَا يُدْرِيكَ لَعَلَّ ٱلسَّاعَةَ تَكُونُ قَرِيبًا ﴿٦٣﴾ لوگ آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) سے قیامت کے بارے میں سوال کرتے ہیں آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) فرما دیجیے بے شک اُس کا علم تو اللہ ہی کے پاس ہے اور (اے مخاطب!) تمھیں کیا معلوم شاید کہ قیامت قریب ہی ہو۔ || يَسْـَٔلُكَ ٱلنَّاسُ عَنِ ٱلسَّاعَةِ ۖ قُلْ إِنَّمَا عِلْمُهَا عِندَ ٱللَّهِ ۚ وَمَا يُدْرِيكَ لَعَلَّ ٱلسَّاعَةَ تَكُونُ قَرِيبًا ﴿٦٣﴾ Men ask thee of the Hour. Say: The knowledge of it is with Allah only. What can convey (the knowledge) unto thee? It may be that the Hour is nigh. (33:63) || يَسْـَٔلُكَ ٱلنَّاسُ عَنِ ٱلسَّاعَةِ ۖ قُلْ إِنَّمَا عِلْمُهَا عِندَ ٱللَّهِ ۚ وَمَا يُدْرِيكَ لَعَلَّ ٱلسَّاعَةَ تَكُونُ قَرِيبًا ﴿٦٣﴾ خلک له تاسو (صلی الله علیه وعلی آله واصحابه وسلم) د قیامت په اړه پوښتنه کوي؛ تاسو (صلی الله علیه وعلی آله واصحابه وسلم) ووایئ: بې‌شکه د هغې علم خو یوازې له الله سره دی؛ او (ای مخاطبه!) ته څه پوهېږې، کېدای شي قیامت نږدې وي.
p إِنَّ ٱللَّهَ لَعَنَ ٱلْكَٰفِرِينَ وَأَعَدَّ لَهُمْ سَعِيرًا ﴿٦٤﴾ بے شک اللہ نے کافروں پر لعنت کی ہے اور ان کے لیے بھڑکتی آگ کا عذاب تیار کر رکھا ہے۔ || إِنَّ ٱللَّهَ لَعَنَ ٱلْكَٰفِرِينَ وَأَعَدَّ لَهُمْ سَعِيرًا ﴿٦٤﴾ Lo! Allah hath cursed the disbelievers, and hath prepared for them a flaming fire, (33:64) || إِنَّ ٱللَّهَ لَعَنَ ٱلْكَٰفِرِينَ وَأَعَدَّ لَهُمْ سَعِيرًا ﴿٦٤﴾ بې‌شکه الله پر کافرانو لعنت کړی، او د هغوی لپاره یې د بل اور عذاب تیار کړی دی.
p خَٰلِدِينَ فِيهَآ أَبَدًۭا ۖ لَّا يَجِدُونَ وَلِيًّۭا وَلَا نَصِيرًۭا ﴿٦٥﴾ وہ اُس میں ہمیشہ ہمیشہ رہیں گے وہ نہ کوئی حمایت کرنے والا پائیں گے اور نہ کوئی مددگار۔ || خَٰلِدِينَ فِيهَآ أَبَدًۭا ۖ لَّا يَجِدُونَ وَلِيًّۭا وَلَا نَصِيرًۭا ﴿٦٥﴾ Wherein they will abide for ever. They will find (then) no protecting friend nor helper. (33:65) || خَٰلِدِينَ فِيهَآ أَبَدًۭا ۖ لَّا يَجِدُونَ وَلِيًّۭا وَلَا نَصِيرًۭا ﴿٦٥﴾ هغوی به په هغه کې تل تر تله پاتې وي؛ نه به کوم حمایتي ومومي او نه کوم مرستندوی.
p يَوْمَ تُقَلَّبُ وُجُوهُهُمْ فِى ٱلنَّارِ يَقُولُونَ يَٰلَيْتَنَآ أَطَعْنَا ٱللَّهَ وَأَطَعْنَا ٱلرَّسُولَا۠ ﴿٦٦﴾ جس دن مجرموں کے چہرے (دوزخ کی) آگ میں الٹ پلٹ کیے جائیں گے تو وہ کہیں گے اے کاش! ہم نے اللہ کی اطاعت کی ہوتی اور رسول کی اطاعت کی ہوتی۔ || يَوْمَ تُقَلَّبُ وُجُوهُهُمْ فِى ٱلنَّارِ يَقُولُونَ يَٰلَيْتَنَآ أَطَعْنَا ٱللَّهَ وَأَطَعْنَا ٱلرَّسُولَا۠ ﴿٦٦﴾ On the day when their faces are turned over in the Fire, they say: Oh, would that we had obeyed Allah and had obeyed His messenger! (33:66) || يَوْمَ تُقَلَّبُ وُجُوهُهُمْ فِى ٱلنَّارِ يَقُولُونَ يَٰلَيْتَنَآ أَطَعْنَا ٱللَّهَ وَأَطَعْنَا ٱلرَّسُولَا۠ ﴿٦٦﴾ په کومه ورځ چې د مجرمانو مخونه د (دوزخ) په اور کې اړول راړول کېږي، نو وایي به: کاش! موږ د الله اطاعت کړی وای او د رسول اطاعت مو کړی وای.
p وَقَالُوا۟ رَبَّنَآ إِنَّآ أَطَعْنَا سَادَتَنَا وَكُبَرَآءَنَا فَأَضَلُّونَا ٱلسَّبِيلَا۠ ﴿٦٧﴾ اور وہ کہیں گے اے ہمارے رب! بے شک ہم نے اپنے سرداروں اور اپنے بڑوں کا کہنا مانا تو انھوں نے ہمیں راستہ سے بھٹکا دیا۔ || وَقَالُوا۟ رَبَّنَآ إِنَّآ أَطَعْنَا سَادَتَنَا وَكُبَرَآءَنَا فَأَضَلُّونَا ٱلسَّبِيلَا۠ ﴿٦٧﴾ And they say: Our Lord! Lo! we obeyed our princes and great men, and they misled us from the Way. (33:67) || وَقَالُوا۟ رَبَّنَآ إِنَّآ أَطَعْنَا سَادَتَنَا وَكُبَرَآءَنَا فَأَضَلُّونَا ٱلسَّبِيلَا۠ ﴿٦٧﴾ او وایي به: ای زموږ ربه! بې‌شکه موږ د خپلو سردارانو او خپلو مشرانو خبره ومنله، نو هغوی موږ له لارې بې‌لارې کړو.
p رَبَّنَآ ءَاتِهِمْ ضِعْفَيْنِ مِنَ ٱلْعَذَابِ وَٱلْعَنْهُمْ لَعْنًۭا كَبِيرًۭا ﴿٦٨﴾ اے ہمارے رب! انھیں (آگ کا) دُگنا عذاب دے اور ان پر بہت بڑی لعنت بھیج۔ || رَبَّنَآ ءَاتِهِمْ ضِعْفَيْنِ مِنَ ٱلْعَذَابِ وَٱلْعَنْهُمْ لَعْنًۭا كَبِيرًۭا ﴿٦٨﴾ Our Lord! Oh, give them double torment and curse them with a mighty curse. (33:68) || رَبَّنَآ ءَاتِهِمْ ضِعْفَيْنِ مِنَ ٱلْعَذَابِ وَٱلْعَنْهُمْ لَعْنًۭا كَبِيرًۭا ﴿٦٨﴾ ای زموږ ربه! هغوی ته د (اور) دوه چنده عذاب ورکړه او پر هغوی ډېره لویه لعنت ولېږه.
p يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ لَا تَكُونُوا۟ كَٱلَّذِينَ ءَاذَوْا۟ مُوسَىٰ فَبَرَّأَهُ ٱللَّهُ مِمَّا قَالُوا۟ ۚ وَكَانَ عِندَ ٱللَّهِ وَجِيهًۭا ﴿٦٩﴾ اے ایمان والو! اُن کی طرح نہ ہو جاؤ جنھوں نے موسیٰ (علیہ السلام) کو اذیت پہنچائی تو اللہ نے ان (موسیٰ علیہ السلام) کو اُس سے بری کر دیا جو ان لوگوں نے (بطور الزام) کہا تھا اور وہ اللہ کے نزدیک بڑے مرتبے والے تھے۔ || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ لَا تَكُونُوا۟ كَٱلَّذِينَ ءَاذَوْا۟ مُوسَىٰ فَبَرَّأَهُ ٱللَّهُ مِمَّا قَالُوا۟ ۚ وَكَانَ عِندَ ٱللَّهِ وَجِيهًۭا ﴿٦٩﴾ O ye who believe! Be not as those who slandered Moses, but Allah proved his innocence of that which they alleged, and he was well esteemed in Allah's sight. (33:69) || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ لَا تَكُونُوا۟ كَٱلَّذِينَ ءَاذَوْا۟ مُوسَىٰ فَبَرَّأَهُ ٱللَّهُ مِمَّا قَالُوا۟ ۚ وَكَانَ عِندَ ٱللَّهِ وَجِيهًۭا ﴿٦٩﴾ ای مؤمنانو! د هغو کسانو په څېر مه کېږئ چې موسی (علیه السلام) ته یې اذیت ورساوه، نو الله هغه (موسی علیه السلام) له هغه څه بري کړ چې هغوی (د تور په توګه) ویلي وو؛ او هغه د الله په وړاندې د لوی مقام خاوند و.
p يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ ٱتَّقُوا۟ ٱللَّهَ وَقُولُوا۟ قَوْلًۭا سَدِيدًۭا ﴿٧٠﴾ اے ایمان والو! اللہ (کی نافرمانی) سے ڈرو اور بالکل سیدھی (سچی) بات کہو۔ || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ ٱتَّقُوا۟ ٱللَّهَ وَقُولُوا۟ قَوْلًۭا سَدِيدًۭا ﴿٧٠﴾ O ye who believe! Guard your duty to Allah, and speak words straight to the point; (33:70) || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ ٱتَّقُوا۟ ٱللَّهَ وَقُولُوا۟ قَوْلًۭا سَدِيدًۭا ﴿٧٠﴾ ای مؤمنانو! له الله (د نافرمانۍ) ووېرېږئ او بېخي سمه (رښتینې) خبره کوئ.
p يُصْلِحْ لَكُمْ أَعْمَٰلَكُمْ وَيَغْفِرْ لَكُمْ ذُنُوبَكُمْ ۗ وَمَن يُطِعِ ٱللَّهَ وَرَسُولَهُۥ فَقَدْ فَازَ فَوْزًا عَظِيمًا ﴿٧١﴾ وہ (اللہ) تمھارے لیے تمھارے اعمال کی اصلاح فرما دے گا اور تمھارے لیے تمھارے گناہوں کی مغفرت فرما دے گا اور جو اللہ اور اُس کے رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کی اطاعت کرے تو یقیناً اُس نے بہت عظیم کامیابی حاصل کی۔ || يُصْلِحْ لَكُمْ أَعْمَٰلَكُمْ وَيَغْفِرْ لَكُمْ ذُنُوبَكُمْ ۗ وَمَن يُطِعِ ٱللَّهَ وَرَسُولَهُۥ فَقَدْ فَازَ فَوْزًا عَظِيمًا ﴿٧١﴾ He will adjust your works for you and will forgive you your sins. Whosoever obeyeth Allah and His messenger, he verily hath gained a signal victory. (33:71) || يُصْلِحْ لَكُمْ أَعْمَٰلَكُمْ وَيَغْفِرْ لَكُمْ ذُنُوبَكُمْ ۗ وَمَن يُطِعِ ٱللَّهَ وَرَسُولَهُۥ فَقَدْ فَازَ فَوْزًا عَظِيمًا ﴿٧١﴾ هغه (الله) به ستاسو لپاره ستاسو عملونه سم کړي او ستاسو لپاره به ستاسو ګناهونه وبښي؛ او څوک چې د الله او د هغه رسول (صلی الله علیه وعلی آله واصحابه وسلم) اطاعت وکړي، نو یقیناً هغه ډېره لویه بریا ترلاسه کړه.
