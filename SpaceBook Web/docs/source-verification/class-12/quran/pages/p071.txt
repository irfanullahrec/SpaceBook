p وَيَقُولُ ٱلَّذِينَ ءَامَنُوٓا۟ أَهَٰٓؤُلَآءِ ٱلَّذِينَ أَقْسَمُوا۟ بِٱللَّهِ جَهْدَ أَيْمَٰنِهِمْ ۙ إِنَّهُمْ لَمَعَكُمْ ۚ حَبِطَتْ أَعْمَٰلُهُمْ فَأَصْبَحُوا۟ خَٰسِرِينَ ﴿٥٣﴾ اور (اُس وقت) جو لوگ ایمان لائے کہیں گے کیا یہی وہ لوگ ہیں جنھوں نے اللہ کی بڑی پختہ قسمیں کھائی تھیں کہ وہ یقیناً تمھارے ساتھ ہیں پس ان کے اعمال برباد ہو گئے پھر وہ خسارہ اٹھانے والوں میں سے ہو گئے۔ || وَيَقُولُ ٱلَّذِينَ ءَامَنُوٓا۟ أَهَٰٓؤُلَآءِ ٱلَّذِينَ أَقْسَمُوا۟ بِٱللَّهِ جَهْدَ أَيْمَٰنِهِمْ ۙ إِنَّهُمْ لَمَعَكُمْ ۚ حَبِطَتْ أَعْمَٰلُهُمْ فَأَصْبَحُوا۟ خَٰسِرِينَ ﴿٥٣﴾ Then will the believers say (unto the people of the Scripture): are these they who swore by Allah their most binding oaths that they were surely with you? Their works have failed, and they have become the losers. (5:53) || وَيَقُولُ ٱلَّذِينَ ءَامَنُوٓا۟ أَهَٰٓؤُلَآءِ ٱلَّذِينَ أَقْسَمُوا۟ بِٱللَّهِ جَهْدَ أَيْمَٰنِهِمْ ۙ إِنَّهُمْ لَمَعَكُمْ ۚ حَبِطَتْ أَعْمَٰلُهُمْ فَأَصْبَحُوا۟ خَٰسِرِينَ ﴿٥٣﴾ او (په هغه وخت کې) هغه کسان چې ایمان یې راوړی، وایي به: «ایا همدا هغه خلک دي چې په الله یې ډېر ټینګ قسمونه خوړلي وو چې یقیناً له تاسو سره دي؟» نو عملونه یې برباد شول، بیا له زیانکارانو شول.
p يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ مَن يَرْتَدَّ مِنكُمْ عَن دِينِهِۦ فَسَوْفَ يَأْتِى ٱللَّهُ بِقَوْمٍۢ يُحِبُّهُمْ وَيُحِبُّونَهُۥٓ أَذِلَّةٍ عَلَى ٱلْمُؤْمِنِينَ أَعِزَّةٍ عَلَى ٱلْكَٰفِرِينَ يُجَٰهِدُونَ فِى سَبِيلِ ٱللَّهِ وَلَا يَخَافُونَ لَوْمَةَ لَآئِمٍۢ ۚ ذَٰلِكَ فَضْلُ ٱللَّهِ يُؤْتِيهِ مَن يَشَآءُ ۚ وَٱللَّهُ وَٰسِعٌ عَلِيمٌ ﴿٥٤﴾ اے ایمان والو! جو شخص تم میں سے اپنے دین سے پھر جائے تو اللہ جلد ایک ایسی قوم کو لے آئے گا کہ اللہ ان سے محبت کرے گا اور وہ اُس سے محبت کریں گے وہ مومنوں پر نرم ہوں گے کافروں پر سخت ہوں گے اللہ کی راہ میں جہاد کریں گے اور کسی ملامت کرنے والے کی ملامت سے نہیں ڈریں گے یہ اللہ کا فضل ہے وہ اسے عطا فرماتا ہے جسے چاہتا ہے اور اللہ بڑی وسعت والا خوب جاننے والا ہے۔ || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ مَن يَرْتَدَّ مِنكُمْ عَن دِينِهِۦ فَسَوْفَ يَأْتِى ٱللَّهُ بِقَوْمٍۢ يُحِبُّهُمْ وَيُحِبُّونَهُۥٓ أَذِلَّةٍ عَلَى ٱلْمُؤْمِنِينَ أَعِزَّةٍ عَلَى ٱلْكَٰفِرِينَ يُجَٰهِدُونَ فِى سَبِيلِ ٱللَّهِ وَلَا يَخَافُونَ لَوْمَةَ لَآئِمٍۢ ۚ ذَٰلِكَ فَضْلُ ٱللَّهِ يُؤْتِيهِ مَن يَشَآءُ ۚ وَٱللَّهُ وَٰسِعٌ عَلِيمٌ ﴿٥٤﴾ O ye who believe! Whoso of you becometh a renegade from his religion, (know that in his stead) Allah will bring a people whom He loveth and who love Him, humble toward believers, stern toward disbelievers, striving in the way of Allah, and fearing not the blame of any blamer. Such is the grace of Allah which He giveth unto whom He will. Allah is All-Embracing, All-Knowing. (5:54) || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ مَن يَرْتَدَّ مِنكُمْ عَن دِينِهِۦ فَسَوْفَ يَأْتِى ٱللَّهُ بِقَوْمٍۢ يُحِبُّهُمْ وَيُحِبُّونَهُۥٓ أَذِلَّةٍ عَلَى ٱلْمُؤْمِنِينَ أَعِزَّةٍ عَلَى ٱلْكَٰفِرِينَ يُجَٰهِدُونَ فِى سَبِيلِ ٱللَّهِ وَلَا يَخَافُونَ لَوْمَةَ لَآئِمٍۢ ۚ ذَٰلِكَ فَضْلُ ٱللَّهِ يُؤْتِيهِ مَن يَشَآءُ ۚ وَٱللَّهُ وَٰسِعٌ عَلِيمٌ ﴿٥٤﴾ ای مؤمنانو! له تاسو څخه چې څوک له خپل دینه واوړي، نو الله به ژر داسې قوم راولي چې الله به ورسره مینه کوي او هغوی به ورسره مینه کوي؛ پر مؤمنانو به نرم وي، پر کافرانو به سخت وي، د الله په لار کې به جهاد کوي او د هېڅ ملامتوونکي له ملامته به نه وېرېږي؛ دا د الله فضل دی، چا ته چې وغواړي ورکوي یې؛ او الله ډېر پراخ، ښه پوه دی.
p إِنَّمَا وَلِيُّكُمُ ٱللَّهُ وَرَسُولُهُۥ وَٱلَّذِينَ ءَامَنُوا۟ ٱلَّذِينَ يُقِيمُونَ ٱلصَّلَوٰةَ وَيُؤْتُونَ ٱلزَّكَوٰةَ وَهُمْ رَٰكِعُونَ ﴿٥٥﴾ بے شک تمھارے دوست تو اللہ اور اُس کے رسول ہیں اور وہ ایمان والے ہیں جو نماز قائم کرتے ہیں اور زکوٰۃ ادا کرتے ہیں اور وہ (ہر حال میں اللہ کے حضور) جھکنے والے ہیں۔ || إِنَّمَا وَلِيُّكُمُ ٱللَّهُ وَرَسُولُهُۥ وَٱلَّذِينَ ءَامَنُوا۟ ٱلَّذِينَ يُقِيمُونَ ٱلصَّلَوٰةَ وَيُؤْتُونَ ٱلزَّكَوٰةَ وَهُمْ رَٰكِعُونَ ﴿٥٥﴾ Your guardian can be only Allah; and His messenger and those who believe, who establish worship and pay the poordue, and bow down (in prayer). (5:55) || إِنَّمَا وَلِيُّكُمُ ٱللَّهُ وَرَسُولُهُۥ وَٱلَّذِينَ ءَامَنُوا۟ ٱلَّذِينَ يُقِيمُونَ ٱلصَّلَوٰةَ وَيُؤْتُونَ ٱلزَّكَوٰةَ وَهُمْ رَٰكِعُونَ ﴿٥٥﴾ بې‌شکه ستاسو دوستان خو الله او د هغه رسول دي، او هغه مؤمنان چې لمونځ قایموي، زکات ورکوي او (په هر حال کې د الله په وړاندې) ټیټېدونکي دي.
p وَمَن يَتَوَلَّ ٱللَّهَ وَرَسُولَهُۥ وَٱلَّذِينَ ءَامَنُوا۟ فَإِنَّ حِزْبَ ٱللَّهِ هُمُ ٱلْغَٰلِبُونَ ﴿٥٦﴾ اور جو اللہ اور اُس کے رسول اور ایمان والوں کو دوست بنائے تو بے شک اللہ کی جماعت ہی غالب آنے والی ہے۔ || وَمَن يَتَوَلَّ ٱللَّهَ وَرَسُولَهُۥ وَٱلَّذِينَ ءَامَنُوا۟ فَإِنَّ حِزْبَ ٱللَّهِ هُمُ ٱلْغَٰلِبُونَ ﴿٥٦﴾ And whoso taketh Allah and His messenger and those who believe for guardian (will know that), lo! the party of Allah, they are the victorious. (5:56) || وَمَن يَتَوَلَّ ٱللَّهَ وَرَسُولَهُۥ وَٱلَّذِينَ ءَامَنُوا۟ فَإِنَّ حِزْبَ ٱللَّهِ هُمُ ٱلْغَٰلِبُونَ ﴿٥٦﴾ او څوک چې الله، د هغه رسول او مؤمنان دوستان ونیسي، نو بې‌شکه د الله ډله ده چې غالبه کېدونکې ده.
p يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ لَا تَتَّخِذُوا۟ ٱلَّذِينَ ٱتَّخَذُوا۟ دِينَكُمْ هُزُوًۭا وَلَعِبًۭا مِّنَ ٱلَّذِينَ أُوتُوا۟ ٱلْكِتَٰبَ مِن قَبْلِكُمْ وَٱلْكُفَّارَ أَوْلِيَآءَ ۚ وَٱتَّقُوا۟ ٱللَّهَ إِن كُنتُم مُّؤْمِنِينَ ﴿٥٧﴾ اے ایمان والو! تم (دوست) نہ بناؤ ان لوگوں کو جنھوں نے تمھارے دین کو مذاق اور کھیل بنا رکھا ہے جنھیں تم سے پہلے کتاب دی گئی اور کافروں کو دوست (نہ بناؤ) اور اللہ (کی نافرمانی) سے ڈرو اگر تم مومن ہو۔ || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ لَا تَتَّخِذُوا۟ ٱلَّذِينَ ٱتَّخَذُوا۟ دِينَكُمْ هُزُوًۭا وَلَعِبًۭا مِّنَ ٱلَّذِينَ أُوتُوا۟ ٱلْكِتَٰبَ مِن قَبْلِكُمْ وَٱلْكُفَّارَ أَوْلِيَآءَ ۚ وَٱتَّقُوا۟ ٱللَّهَ إِن كُنتُم مُّؤْمِنِينَ ﴿٥٧﴾ O Ye who believe! Choose not for guardians such of those who received the Scripture before you, and of the disbelievers, as make a jest and sport of your religion. But keep your duty to Allah if ye are true believers. (5:57) || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ لَا تَتَّخِذُوا۟ ٱلَّذِينَ ٱتَّخَذُوا۟ دِينَكُمْ هُزُوًۭا وَلَعِبًۭا مِّنَ ٱلَّذِينَ أُوتُوا۟ ٱلْكِتَٰبَ مِن قَبْلِكُمْ وَٱلْكُفَّارَ أَوْلِيَآءَ ۚ وَٱتَّقُوا۟ ٱللَّهَ إِن كُنتُم مُّؤْمِنِينَ ﴿٥٧﴾ ای مؤمنانو! هغه کسان (دوستان) مه نیسئ چې ستاسو دین یې ټوکه او لوبه ګرځولی — له هغو چې له تاسو مخکې کتاب ورکړل شوی — او کافران (هم) دوستان (مه نیسئ)؛ او له الله (د نافرمانۍ) ووېرېږئ که مؤمنان یاست.
p وَإِذَا نَادَيْتُمْ إِلَى ٱلصَّلَوٰةِ ٱتَّخَذُوهَا هُزُوًۭا وَلَعِبًۭا ۚ ذَٰلِكَ بِأَنَّهُمْ قَوْمٌۭ لَّا يَعْقِلُونَ ﴿٥٨﴾ اور جب تم (اذان دیتے ہوئے) نماز کے لیے (لوگوں کو) پکارتے ہو وہ اس (پکار) کو مذاق اور کھیل بناتے ہیں اس لیے کہ یہ وہ لوگ ہیں جو کچھ نہیں سمجھتے۔ || وَإِذَا نَادَيْتُمْ إِلَى ٱلصَّلَوٰةِ ٱتَّخَذُوهَا هُزُوًۭا وَلَعِبًۭا ۚ ذَٰلِكَ بِأَنَّهُمْ قَوْمٌۭ لَّا يَعْقِلُونَ ﴿٥٨﴾ And when ye call to prayer they take it for a jest and sport. That is because they are a folk who understand not. (5:58) || وَإِذَا نَادَيْتُمْ إِلَى ٱلصَّلَوٰةِ ٱتَّخَذُوهَا هُزُوًۭا وَلَعِبًۭا ۚ ذَٰلِكَ بِأَنَّهُمْ قَوْمٌۭ لَّا يَعْقِلُونَ ﴿٥٨﴾ او کله چې (په اذان سره) لمانځه ته (خلک) رابلئ، هغوی دا (بلنه) ټوکه او لوبه ګرځوي؛ دا ځکه چې دا هغه خلک دي چې هېڅ نه پوهېږي.
