p وَٱلْخَٰمِسَةَ أَنَّ غَضَبَ ٱللَّهِ عَلَيْهَآ إِن كَانَ مِنَ ٱلصَّٰدِقِينَ ﴿٩﴾ اور پانچویں بار یہ (کہے) کہ اللہ کا غضب ہو اُس (عورت) پر اگر وہ (شوہر) سچا ہے۔ || وَٱلْخَٰمِسَةَ أَنَّ غَضَبَ ٱللَّهِ عَلَيْهَآ إِن كَانَ مِنَ ٱلصَّٰدِقِينَ ﴿٩﴾ And a fifth (time) that the wrath of Allah be upon her if he speaketh truth. (24:9) || وَٱلْخَٰمِسَةَ أَنَّ غَضَبَ ٱللَّهِ عَلَيْهَآ إِن كَانَ مِنَ ٱلصَّٰدِقِينَ ﴿٩﴾ او پنځم ځل دې (ووایي) چې پر هغې (ښځې) دې د الله غضب وي که هغه (مېړه) رښتینی وي.
p وَلَوْلَا فَضْلُ ٱللَّهِ عَلَيْكُمْ وَرَحْمَتُهُۥ وَأَنَّ ٱللَّهَ تَوَّابٌ حَكِيمٌ ﴿١٠﴾ اور اگر تم پر اللہ کا فضل اور اُس کی رحمت نہ ہوتی (تو تم بڑی مشکل میں پڑ جاتے) اور بے شک اللہ بہت زیادہ توبہ قبول فرمانے والا بڑی حکمت والا ہے۔ || وَلَوْلَا فَضْلُ ٱللَّهِ عَلَيْكُمْ وَرَحْمَتُهُۥ وَأَنَّ ٱللَّهَ تَوَّابٌ حَكِيمٌ ﴿١٠﴾ And had it not been for the grace of Allah and His mercy unto you, and that Allah is Clement, Wise, (ye had been undone). (24:10) || وَلَوْلَا فَضْلُ ٱللَّهِ عَلَيْكُمْ وَرَحْمَتُهُۥ وَأَنَّ ٱللَّهَ تَوَّابٌ حَكِيمٌ ﴿١٠﴾ او که پر تاسو د الله فضل او د هغه رحمت نه وای (نو په لویه ستونزه کې به لوېدلي وئ)؛ او بې‌شکه الله ډېر توبه قبلوونکی، ډېر حکمتناک دی.
p إِنَّ ٱلَّذِينَ جَآءُو بِٱلْإِفْكِ عُصْبَةٌۭ مِّنكُمْ ۚ لَا تَحْسَبُوهُ شَرًّۭا لَّكُم ۖ بَلْ هُوَ خَيْرٌۭ لَّكُمْ ۚ لِكُلِّ ٱمْرِئٍۢ مِّنْهُم مَّا ٱكْتَسَبَ مِنَ ٱلْإِثْمِ ۚ وَٱلَّذِى تَوَلَّىٰ كِبْرَهُۥ مِنْهُمْ لَهُۥ عَذَابٌ عَظِيمٌۭ ﴿١١﴾ بے شک جن لوگوں نے (عائشہ صدیقہ رضی اللہ تعالیٰ عنہا پر) بہتان لگایا (وہ) تمھارے اندر کا (ہی) ایک ٹولہ ہے تم اس (بہتان کے واقعہ) کو اپنے حق میں برا مت سمجھو بلکہ وہ تمھارے حق میں بہتر (ہی) ہے اُن میں سے ہر ایک نے جتنا گناہ کمایا اُس کے لیے اتنا ہی حصہ ہے اور اُن میں سے جس نے اس (بہتان) میں سب سے زیادہ حصہ لیا اُس کے لیے بڑا عذاب ہے۔ || إِنَّ ٱلَّذِينَ جَآءُو بِٱلْإِفْكِ عُصْبَةٌۭ مِّنكُمْ ۚ لَا تَحْسَبُوهُ شَرًّۭا لَّكُم ۖ بَلْ هُوَ خَيْرٌۭ لَّكُمْ ۚ لِكُلِّ ٱمْرِئٍۢ مِّنْهُم مَّا ٱكْتَسَبَ مِنَ ٱلْإِثْمِ ۚ وَٱلَّذِى تَوَلَّىٰ كِبْرَهُۥ مِنْهُمْ لَهُۥ عَذَابٌ عَظِيمٌۭ ﴿١١﴾ Lo! they who spread the slander are a gang among you. Deem it not a bad thing for you; nay, it is good for you. Unto every man of them (will be paid) that which he hath earned of the sin; and as for him among them who had the greater share therein, his will be an awful doom. (24:11) || إِنَّ ٱلَّذِينَ جَآءُو بِٱلْإِفْكِ عُصْبَةٌۭ مِّنكُمْ ۚ لَا تَحْسَبُوهُ شَرًّۭا لَّكُم ۖ بَلْ هُوَ خَيْرٌۭ لَّكُمْ ۚ لِكُلِّ ٱمْرِئٍۢ مِّنْهُم مَّا ٱكْتَسَبَ مِنَ ٱلْإِثْمِ ۚ وَٱلَّذِى تَوَلَّىٰ كِبْرَهُۥ مِنْهُمْ لَهُۥ عَذَابٌ عَظِيمٌۭ ﴿١١﴾ بې‌شکه هغو کسانو چې (پر عایشه صدیقه رضي الله تعالی عنها) بهتان ولګاوه، (هغوی) ستاسو له منځه یوه ډله ده؛ دا (د بهتان پېښه) د ځان لپاره بده مه ګڼئ، بلکې هغه ستاسو لپاره غوره (ده)؛ له هغوی هر یوه ته د هغه ګناه په اندازه برخه ده چې ګټلې یې ده؛ او له هغوی څخه چا چې په دې (بهتان) کې تر ټولو زیاته برخه واخیسته، د هغه لپاره لوی عذاب دی.
p لَّوْلَآ إِذْ سَمِعْتُمُوهُ ظَنَّ ٱلْمُؤْمِنُونَ وَٱلْمُؤْمِنَٰتُ بِأَنفُسِهِمْ خَيْرًۭا وَقَالُوا۟ هَٰذَآ إِفْكٌۭ مُّبِينٌۭ ﴿١٢﴾ (ایسا) کیوں نہ ہوا کہ جب تم نے اس (بہتان) کو سنا تھا تو مومن مردوں اور مومن عورتوں نے اپنوں کے بارے میں نیک گمان کیا ہوتا اور کہا ہوتا کہ یہ تو بالکل واضح بہتان ہے۔ || لَّوْلَآ إِذْ سَمِعْتُمُوهُ ظَنَّ ٱلْمُؤْمِنُونَ وَٱلْمُؤْمِنَٰتُ بِأَنفُسِهِمْ خَيْرًۭا وَقَالُوا۟ هَٰذَآ إِفْكٌۭ مُّبِينٌۭ ﴿١٢﴾ Why did not the believers, men and women, when ye heard it, think good of their own own folk, and say: It is a manifest untruth? (24:12) || لَّوْلَآ إِذْ سَمِعْتُمُوهُ ظَنَّ ٱلْمُؤْمِنُونَ وَٱلْمُؤْمِنَٰتُ بِأَنفُسِهِمْ خَيْرًۭا وَقَالُوا۟ هَٰذَآ إِفْكٌۭ مُّبِينٌۭ ﴿١٢﴾ (داسې) ولې ونه شول چې کله تاسو دا (بهتان) واورېد، مؤمنو نارینه وو او مؤمنو ښځو د خپلو په اړه نېک ګومان کړی وای او ویلي یې وای: «دا خو بېخي ښکاره بهتان دی»؟
p لَّوْلَا جَآءُو عَلَيْهِ بِأَرْبَعَةِ شُهَدَآءَ ۚ فَإِذْ لَمْ يَأْتُوا۟ بِٱلشُّهَدَآءِ فَأُو۟لَٰٓئِكَ عِندَ ٱللَّهِ هُمُ ٱلْكَٰذِبُونَ ﴿١٣﴾ وہ (بہتان لگانے والے) اس پر چار گواہ کیوں نہ لائے پھر جب وہ گواہ نہیں لا سکے تو (جان لو) یہی لوگ اللہ کے نزدیک جھوٹے ہیں۔ || لَّوْلَا جَآءُو عَلَيْهِ بِأَرْبَعَةِ شُهَدَآءَ ۚ فَإِذْ لَمْ يَأْتُوا۟ بِٱلشُّهَدَآءِ فَأُو۟لَٰٓئِكَ عِندَ ٱللَّهِ هُمُ ٱلْكَٰذِبُونَ ﴿١٣﴾ Why did they not produce four witnesses? Since they produce not witnesses, they verily are liars in the sight of Allah. (24:13) || لَّوْلَا جَآءُو عَلَيْهِ بِأَرْبَعَةِ شُهَدَآءَ ۚ فَإِذْ لَمْ يَأْتُوا۟ بِٱلشُّهَدَآءِ فَأُو۟لَٰٓئِكَ عِندَ ٱللَّهِ هُمُ ٱلْكَٰذِبُونَ ﴿١٣﴾ هغوی (بهتان لګوونکو) پرې څلور شاهدان ولې رانه وړل؟ بیا چې شاهدان یې رانه وړای شول، نو (پوه شئ) همدا خلک د الله په نزد درواغجن دي.
p وَلَوْلَا فَضْلُ ٱللَّهِ عَلَيْكُمْ وَرَحْمَتُهُۥ فِى ٱلدُّنْيَا وَٱلْءَاخِرَةِ لَمَسَّكُمْ فِى مَآ أَفَضْتُمْ فِيهِ عَذَابٌ عَظِيمٌ ﴿١٤﴾ اور اگر تم پر دنیا اور آخرت میں اللہ کا فضل اور اُس کی رحمت نہ ہوتی تو جس (تہمت کے) چرچے میں تم پڑ گئے تھے اس پر تمھیں بڑا عذاب پہنچتا۔ || وَلَوْلَا فَضْلُ ٱللَّهِ عَلَيْكُمْ وَرَحْمَتُهُۥ فِى ٱلدُّنْيَا وَٱلْءَاخِرَةِ لَمَسَّكُمْ فِى مَآ أَفَضْتُمْ فِيهِ عَذَابٌ عَظِيمٌ ﴿١٤﴾ Had it not been for the grace of Allah and His mercy unto you in the world and the Hereafter an awful doom had overtaken you for that whereof ye murmured. (24:14) || وَلَوْلَا فَضْلُ ٱللَّهِ عَلَيْكُمْ وَرَحْمَتُهُۥ فِى ٱلدُّنْيَا وَٱلْءَاخِرَةِ لَمَسَّكُمْ فِى مَآ أَفَضْتُمْ فِيهِ عَذَابٌ عَظِيمٌ ﴿١٤﴾ او که پر تاسو په دنیا او آخرت کې د الله فضل او د هغه رحمت نه وای، نو د هغه (تور) په خبرو کې چې تاسو لوېدلي وئ، تاسو ته به لوی عذاب رسېدلی و.
p إِذْ تَلَقَّوْنَهُۥ بِأَلْسِنَتِكُمْ وَتَقُولُونَ بِأَفْوَاهِكُم مَّا لَيْسَ لَكُم بِهِۦ عِلْمٌۭ وَتَحْسَبُونَهُۥ هَيِّنًۭا وَهُوَ عِندَ ٱللَّهِ عَظِيمٌۭ ﴿١٥﴾ (جب تم ایک دوسرے سے) اس (بہتان) کو اپنی زبانوں سے نقل کر رہے تھے اور اپنے منہ سے وہ کچھ کہتے تھے جس کا تمھیں کوئی علم ہی نہ تھا اور تم اس کو معمولی بات سمجھ رہے تھے حالاں کہ وہ اللہ کے نزدیک بہت بڑی بات تھی۔ || إِذْ تَلَقَّوْنَهُۥ بِأَلْسِنَتِكُمْ وَتَقُولُونَ بِأَفْوَاهِكُم مَّا لَيْسَ لَكُم بِهِۦ عِلْمٌۭ وَتَحْسَبُونَهُۥ هَيِّنًۭا وَهُوَ عِندَ ٱللَّهِ عَظِيمٌۭ ﴿١٥﴾ When ye welcomed it with your tongues, and uttered with your mouths that whereof ye had no knowledge, ye counted it a trifle. In the sight of Allah it is very great. (24:15) || إِذْ تَلَقَّوْنَهُۥ بِأَلْسِنَتِكُمْ وَتَقُولُونَ بِأَفْوَاهِكُم مَّا لَيْسَ لَكُم بِهِۦ عِلْمٌۭ وَتَحْسَبُونَهُۥ هَيِّنًۭا وَهُوَ عِندَ ٱللَّهِ عَظِيمٌۭ ﴿١٥﴾ (کله چې تاسو یو له بله) دا (بهتان) په خپلو ژبو نقلاوه او په خپلو خولو مو هغه څه ویل چې تاسو ته پرې هېڅ علم نه و، او تاسو یې عادي خبره ګڼله، حال دا چې د الله په نزد ډېره لویه خبره وه.
p وَلَوْلَآ إِذْ سَمِعْتُمُوهُ قُلْتُم مَّا يَكُونُ لَنَآ أَن نَّتَكَلَّمَ بِهَٰذَا سُبْحَٰنَكَ هَٰذَا بُهْتَٰنٌ عَظِيمٌۭ ﴿١٦﴾ اور ایسا کیوں نہ ہوا جب تم نے اس (بہتان) کو سنا تھا تو تم کہہ دیتے کہ ہمیں لائق نہیں کہ ہم ایسی بات منہ پر لائیں (اے اللہ!) تُو پاک ہے یہ تو بہت بڑا بہتان ہے۔ || وَلَوْلَآ إِذْ سَمِعْتُمُوهُ قُلْتُم مَّا يَكُونُ لَنَآ أَن نَّتَكَلَّمَ بِهَٰذَا سُبْحَٰنَكَ هَٰذَا بُهْتَٰنٌ عَظِيمٌۭ ﴿١٦﴾ Wherefor, when ye heard it, said ye not: It is not for us to speak of this. Glory be to Thee (O Allah)! This is awful calumny. (24:16) || وَلَوْلَآ إِذْ سَمِعْتُمُوهُ قُلْتُم مَّا يَكُونُ لَنَآ أَن نَّتَكَلَّمَ بِهَٰذَا سُبْحَٰنَكَ هَٰذَا بُهْتَٰنٌ عَظِيمٌۭ ﴿١٦﴾ او داسې ولې ونه شول چې کله تاسو دا (بهتان) واورېد، ویلي مو وای: «موږ ته نه ښایي چې داسې خبره پر خوله راوړو؛ (ای الله!) ته پاک یې، دا خو ډېر لوی بهتان دی»؟
