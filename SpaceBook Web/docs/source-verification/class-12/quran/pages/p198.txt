p ۞ عَسَى ٱللَّهُ أَن يَجْعَلَ بَيْنَكُمْ وَبَيْنَ ٱلَّذِينَ عَادَيْتُم مِّنْهُم مَّوَدَّةًۭ ۚ وَٱللَّهُ قَدِيرٌۭ ۚ وَٱللَّهُ غَفُورٌۭ رَّحِيمٌۭ ﴿٧﴾ قریب ہے کہ اللہ تمھارے اور ان کے درمیان جن سے تم دشمنی رکھتے ہو محبت پیدا کر دے اور اللہ بہت قدرت رکھنے والا ہے اور اللہ بڑا ہی بخشنے والا نہایت رحم فرمانے والا ہے۔ || ۞ عَسَى ٱللَّهُ أَن يَجْعَلَ بَيْنَكُمْ وَبَيْنَ ٱلَّذِينَ عَادَيْتُم مِّنْهُم مَّوَدَّةًۭ ۚ وَٱللَّهُ قَدِيرٌۭ ۚ وَٱللَّهُ غَفُورٌۭ رَّحِيمٌۭ ﴿٧﴾ It may be that Allah will ordain love between you and those of them with whom ye are at enmity. Allah is Mighty, and Allah is Forgiving, Merciful. (60:7) || ۞ عَسَى ٱللَّهُ أَن يَجْعَلَ بَيْنَكُمْ وَبَيْنَ ٱلَّذِينَ عَادَيْتُم مِّنْهُم مَّوَدَّةًۭ ۚ وَٱللَّهُ قَدِيرٌۭ ۚ وَٱللَّهُ غَفُورٌۭ رَّحِيمٌۭ ﴿٧﴾ نږدې ده چې الله ستاسو او د هغو کسانو ترمنځ چې تاسو ورسره دښمني لرئ، مینه پیدا کړي؛ او الله ډېر قدرتمند دی، او الله ډېر بښونکی، بې‌حده مهربان دی.
p لَّا يَنْهَىٰكُمُ ٱللَّهُ عَنِ ٱلَّذِينَ لَمْ يُقَٰتِلُوكُمْ فِى ٱلدِّينِ وَلَمْ يُخْرِجُوكُم مِّن دِيَٰرِكُمْ أَن تَبَرُّوهُمْ وَتُقْسِطُوٓا۟ إِلَيْهِمْ ۚ إِنَّ ٱللَّهَ يُحِبُّ ٱلْمُقْسِطِينَ ﴿٨﴾ اللہ تمھیں (اس بات سے) منع نہیں کرتا کہ جن لوگوں نے دین (کے معاملے) میں تم سے جنگ نہیں کی اور نہ تمھیں تمھارے گھروں سے نکالا ہے کہ تم ان سے بھلائی کا سلوک کرو اور ان سے انصاف کا برتاؤ کرو بے شک اللہ انصاف کرنے والوں کو پسند فرماتا ہے۔ || لَّا يَنْهَىٰكُمُ ٱللَّهُ عَنِ ٱلَّذِينَ لَمْ يُقَٰتِلُوكُمْ فِى ٱلدِّينِ وَلَمْ يُخْرِجُوكُم مِّن دِيَٰرِكُمْ أَن تَبَرُّوهُمْ وَتُقْسِطُوٓا۟ إِلَيْهِمْ ۚ إِنَّ ٱللَّهَ يُحِبُّ ٱلْمُقْسِطِينَ ﴿٨﴾ Allah forbiddeth you not those who warred not against you on account of religion and drove you not out from your homes, that ye should show them kindness and deal justly with them. Lo! Allah loveth the just dealers. (60:8) || لَّا يَنْهَىٰكُمُ ٱللَّهُ عَنِ ٱلَّذِينَ لَمْ يُقَٰتِلُوكُمْ فِى ٱلدِّينِ وَلَمْ يُخْرِجُوكُم مِّن دِيَٰرِكُمْ أَن تَبَرُّوهُمْ وَتُقْسِطُوٓا۟ إِلَيْهِمْ ۚ إِنَّ ٱللَّهَ يُحِبُّ ٱلْمُقْسِطِينَ ﴿٨﴾ الله تاسو له دې (خبرې) منع نه کوي چې له هغو خلکو سره چې د دین (په چاره) کې یې له تاسو سره جګړه نه ده کړې او نه یې تاسو له خپلو کورونو ایستلي یاست، ښه چلند وکړئ او ورسره انصاف وکړئ؛ بې‌شکه الله انصاف کوونکي خوښوي.
p إِنَّمَا يَنْهَىٰكُمُ ٱللَّهُ عَنِ ٱلَّذِينَ قَٰتَلُوكُمْ فِى ٱلدِّينِ وَأَخْرَجُوكُم مِّن دِيَٰرِكُمْ وَظَٰهَرُوا۟ عَلَىٰٓ إِخْرَاجِكُمْ أَن تَوَلَّوْهُمْ ۚ وَمَن يَتَوَلَّهُمْ فَأُو۟لَٰٓئِكَ هُمُ ٱلظَّٰلِمُونَ ﴿٩﴾ بے شک اللہ تمھیں ان لوگوں سے منع کرتا ہے جنھوں نے تم سے دین (کے معاملے) میں جنگ کی اور تمھیں تمھارے گھروں سے نکالا اور تمھارے نکالنے میں (تمھارے دشمنوں کی) مدد کی کہ تم ان سے دوستی کرو اور جو ان سے دوستی کرے گا تو وہی لوگ ظالم ہیں۔ || إِنَّمَا يَنْهَىٰكُمُ ٱللَّهُ عَنِ ٱلَّذِينَ قَٰتَلُوكُمْ فِى ٱلدِّينِ وَأَخْرَجُوكُم مِّن دِيَٰرِكُمْ وَظَٰهَرُوا۟ عَلَىٰٓ إِخْرَاجِكُمْ أَن تَوَلَّوْهُمْ ۚ وَمَن يَتَوَلَّهُمْ فَأُو۟لَٰٓئِكَ هُمُ ٱلظَّٰلِمُونَ ﴿٩﴾ Allah forbiddeth you only those who warred against you on account of religion and have driven you out from your homes and helped to drive you out, that ye make friends of them. Whosoever maketh friends of them - (All) such are wrong-doers. (60:9) || إِنَّمَا يَنْهَىٰكُمُ ٱللَّهُ عَنِ ٱلَّذِينَ قَٰتَلُوكُمْ فِى ٱلدِّينِ وَأَخْرَجُوكُم مِّن دِيَٰرِكُمْ وَظَٰهَرُوا۟ عَلَىٰٓ إِخْرَاجِكُمْ أَن تَوَلَّوْهُمْ ۚ وَمَن يَتَوَلَّهُمْ فَأُو۟لَٰٓئِكَ هُمُ ٱلظَّٰلِمُونَ ﴿٩﴾ بې‌شکه الله تاسو له هغو خلکو منع کوي چې د دین (په چاره) کې یې له تاسو سره جګړه وکړه، تاسو یې له خپلو کورونو وویستلئ، او ستاسو په ایستلو کې یې (ستاسو د دښمنانو) مرسته وکړه، چې له هغوی سره دوستي وکړئ؛ او څوک چې له هغوی سره دوستي وکړي، نو همدوی ظالمان دي.
p يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓا۟ إِذَا جَآءَكُمُ ٱلْمُؤْمِنَٰتُ مُهَٰجِرَٰتٍۢ فَٱمْتَحِنُوهُنَّ ۖ ٱللَّهُ أَعْلَمُ بِإِيمَٰنِهِنَّ ۖ فَإِنْ عَلِمْتُمُوهُنَّ مُؤْمِنَٰتٍۢ فَلَا تَرْجِعُوهُنَّ إِلَى ٱلْكُفَّارِ ۖ لَا هُنَّ حِلٌّۭ لَّهُمْ وَلَا هُمْ يَحِلُّونَ لَهُنَّ ۖ وَءَاتُوهُم مَّآ أَنفَقُوا۟ ۚ وَلَا جُنَاحَ عَلَيْكُمْ أَن تَنكِحُوهُنَّ إِذَآ ءَاتَيْتُمُوهُنَّ أُجُورَهُنَّ ۚ وَلَا تُمْسِكُوا۟ بِعِصَمِ ٱلْكَوَافِرِ وَسْـَٔلُوا۟ مَآ أَنفَقْتُمْ وَلْيَسْـَٔلُوا۟ مَآ أَنفَقُوا۟ ۚ ذَٰلِكُمْ حُكْمُ ٱللَّهِ ۖ يَحْكُمُ بَيْنَكُمْ ۚ وَٱللَّهُ عَلِيمٌ حَكِيمٌۭ ﴿١٠﴾ اے ایمان والو! جب تمھارے پاس ایمان والی عورتیں ہجرت کر کے آئیں تو تم ان کا امتحان کر لیا کرو اللہ ان کے ایمان کو خوب جانتا ہے پھر اگر تمھیں معلوم ہو جائے کہ وہ ایمان والیاں ہیں تو انھیں کافروں کی طرف واپس نہ بھیجو نہ یہ (مومنات) ان (کافروں) کے لیے حلال ہیں اور نہ وہ (کفار) ان (مومنات) کے لیے حلال ہیں اور ان (کافر شوہروں) کو ادا کر دو جو (مہر) انھوں نے خرچ کیا ہے اور تم پر کوئی گناہ نہیں کہ تم ان (عورتوں) سے نکاح کر لو جب تم انھیں ان کے مہر ادا کر دو اور (اے مسلمانو! تم بھی) کافر عورتوں کو اپنے نکاح میں نہ روکے رکھو اور جو (مہر) تم نے خرچ کیا تھا وہ (کافروں سے) طلب کر لو اور وہ (بھی تم سے) مانگ لیں جو (مہر) انھوں نے خرچ کیا ہے یہ اللہ کا حکم ہے وہ تمھارے درمیان فیصلہ فرماتا ہے اور اللہ خوب جاننے والا بڑی حکمت والا ہے۔ || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓا۟ إِذَا جَآءَكُمُ ٱلْمُؤْمِنَٰتُ مُهَٰجِرَٰتٍۢ فَٱمْتَحِنُوهُنَّ ۖ ٱللَّهُ أَعْلَمُ بِإِيمَٰنِهِنَّ ۖ فَإِنْ عَلِمْتُمُوهُنَّ مُؤْمِنَٰتٍۢ فَلَا تَرْجِعُوهُنَّ إِلَى ٱلْكُفَّارِ ۖ لَا هُنَّ حِلٌّۭ لَّهُمْ وَلَا هُمْ يَحِلُّونَ لَهُنَّ ۖ وَءَاتُوهُم مَّآ أَنفَقُوا۟ ۚ وَلَا جُنَاحَ عَلَيْكُمْ أَن تَنكِحُوهُنَّ إِذَآ ءَاتَيْتُمُوهُنَّ أُجُورَهُنَّ ۚ وَلَا تُمْسِكُوا۟ بِعِصَمِ ٱلْكَوَافِرِ وَسْـَٔلُوا۟ مَآ أَنفَقْتُمْ وَلْيَسْـَٔلُوا۟ مَآ أَنفَقُوا۟ ۚ ذَٰلِكُمْ حُكْمُ ٱللَّهِ ۖ يَحْكُمُ بَيْنَكُمْ ۚ وَٱللَّهُ عَلِيمٌ حَكِيمٌۭ ﴿١٠﴾ O ye who believe! When believing women come unto you as fugitives, examine them. Allah is Best Aware of their faith. Then, if ye know them for true believers, send them not back unto the disbelievers. They are not lawful for them (the disbelievers), nor are they (the disbelievers) lawful for them. And give them (the disbelievers) that which they have spent (upon them). And it is no sin for you to marry such women when ye have given them their dues. And hold not to the ties of disbelieving women; and ask for (the return of) that which ye have spent; and let them (the disbelievers) ask for that which they have spent. That is the judgment of Allah. He judgeth between you. Allah is Knower, Wise. (60:10) || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓا۟ إِذَا جَآءَكُمُ ٱلْمُؤْمِنَٰتُ مُهَٰجِرَٰتٍۢ فَٱمْتَحِنُوهُنَّ ۖ ٱللَّهُ أَعْلَمُ بِإِيمَٰنِهِنَّ ۖ فَإِنْ عَلِمْتُمُوهُنَّ مُؤْمِنَٰتٍۢ فَلَا تَرْجِعُوهُنَّ إِلَى ٱلْكُفَّارِ ۖ لَا هُنَّ حِلٌّۭ لَّهُمْ وَلَا هُمْ يَحِلُّونَ لَهُنَّ ۖ وَءَاتُوهُم مَّآ أَنفَقُوا۟ ۚ وَلَا جُنَاحَ عَلَيْكُمْ أَن تَنكِحُوهُنَّ إِذَآ ءَاتَيْتُمُوهُنَّ أُجُورَهُنَّ ۚ وَلَا تُمْسِكُوا۟ بِعِصَمِ ٱلْكَوَافِرِ وَسْـَٔلُوا۟ مَآ أَنفَقْتُمْ وَلْيَسْـَٔلُوا۟ مَآ أَنفَقُوا۟ ۚ ذَٰلِكُمْ حُكْمُ ٱللَّهِ ۖ يَحْكُمُ بَيْنَكُمْ ۚ وَٱللَّهُ عَلِيمٌ حَكِيمٌۭ ﴿١٠﴾ ای مؤمنانو! کله چې تاسو ته مؤمنې ښځې هجرت وکړي او راشي، نو د هغوی ازموینه وکړئ؛ الله د هغوی په ایمان ښه پوهېږي؛ بیا که تاسو ته معلومه شي چې هغوی مؤمنې دي، نو کافرانو ته یې بېرته مه لېږئ؛ نه دا (مؤمنې) د هغو (کافرانو) لپاره حلالې دي او نه هغوی (کافران) د دې (مؤمنو) لپاره حلال دي؛ او هغو (کافرو مېړونو) ته هغه (مهر) ورکړئ چې هغوی لګولی؛ او پر تاسو څه ګناه نشته چې له دې (ښځو) سره نکاح وکړئ کله چې د هغوی مهرونه ورکړئ؛ او (ای مسلمانانو! تاسو هم) کافرې ښځې په خپله نکاح کې مه ساتئ؛ او کوم (مهر) چې تاسو لګولی و، هغه (له کافرانو) وغواړئ، او هغوی (هم له تاسو) هغه وغواړي چې (مهر) یې لګولی؛ دا د الله حکم دی، هغه ستاسو ترمنځ پرېکړه کوي؛ او الله ښه پوه، ډېر حکمتناک دی.
