p هُوَ ٱلَّذِى خَلَقَكُمْ فَمِنكُمْ كَافِرٌۭ وَمِنكُم مُّؤْمِنٌۭ ۚ وَٱللَّهُ بِمَا تَعْمَلُونَ بَصِيرٌ ﴿٢﴾ وہی ہے جس نے تمھیں پیدا فرمایا پھر تم میں سے کوئی کافر ہے اور تم میں سے کوئی مومن ہے اور جو تم کر رہے ہو اللہ اُسے خوب دیکھنے والا ہے۔ || هُوَ ٱلَّذِى خَلَقَكُمْ فَمِنكُمْ كَافِرٌۭ وَمِنكُم مُّؤْمِنٌۭ ۚ وَٱللَّهُ بِمَا تَعْمَلُونَ بَصِيرٌ ﴿٢﴾ He it is Who created you, but one of you is a disbeliever and one of you is a believer, and Allah is Seer of what ye do. (64:2) || هُوَ ٱلَّذِى خَلَقَكُمْ فَمِنكُمْ كَافِرٌۭ وَمِنكُم مُّؤْمِنٌۭ ۚ وَٱللَّهُ بِمَا تَعْمَلُونَ بَصِيرٌ ﴿٢﴾ همغه دی چې تاسو یې پیدا کړئ، بیا له تاسو ځینې کافران دي او ځینې مؤمنان؛ او څه چې تاسو کوئ، الله یې ښه لیدونکی دی.
p خَلَقَ ٱلسَّمَٰوَٰتِ وَٱلْأَرْضَ بِٱلْحَقِّ وَصَوَّرَكُمْ فَأَحْسَنَ صُوَرَكُمْ ۖ وَإِلَيْهِ ٱلْمَصِيرُ ﴿٣﴾ اُس نے آسمانوں اور زمینوں کو حق کے ساتھ پیدا فرمایا اور اُس نے تمھاری صورتیں بنائیں تو تمھاری بہترین صورتیں بنائیں اور اسی کی طرف پلٹ کر جانا ہے۔ || خَلَقَ ٱلسَّمَٰوَٰتِ وَٱلْأَرْضَ بِٱلْحَقِّ وَصَوَّرَكُمْ فَأَحْسَنَ صُوَرَكُمْ ۖ وَإِلَيْهِ ٱلْمَصِيرُ ﴿٣﴾ He created the heavens and the earth with truth, and He shaped you and made good your shapes, and unto Him is the journeying. (64:3) || خَلَقَ ٱلسَّمَٰوَٰتِ وَٱلْأَرْضَ بِٱلْحَقِّ وَصَوَّرَكُمْ فَأَحْسَنَ صُوَرَكُمْ ۖ وَإِلَيْهِ ٱلْمَصِيرُ ﴿٣﴾ هغه آسمانونه او ځمکې په حق سره پیدا کړې، او ستاسو بڼې یې جوړې کړې، نو ستاسو بڼې یې ښې جوړې کړې؛ او هماغه ته بېرته ورتلل دي.
p يَعْلَمُ مَا فِى ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ وَيَعْلَمُ مَا تُسِرُّونَ وَمَا تُعْلِنُونَ ۚ وَٱللَّهُ عَلِيمٌۢ بِذَاتِ ٱلصُّدُورِ ﴿٤﴾ وہ جانتا ہے جو کچھ آسمانوں اور زمینوں میں ہے اور وہ جانتا ہے جو کچھ تم چھپاتے ہو اور جو (کچھ) تم ظاہر کرتے ہو اور اللہ سینوں کے رازوں کو بھی خوب جاننے والا ہے۔ || يَعْلَمُ مَا فِى ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ وَيَعْلَمُ مَا تُسِرُّونَ وَمَا تُعْلِنُونَ ۚ وَٱللَّهُ عَلِيمٌۢ بِذَاتِ ٱلصُّدُورِ ﴿٤﴾ He knoweth all that is in the heavens and the earth, and He knoweth what ye conceal and what ye publish. And Allah is Aware of what is in the breasts (of men). (64:4) || يَعْلَمُ مَا فِى ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ وَيَعْلَمُ مَا تُسِرُّونَ وَمَا تُعْلِنُونَ ۚ وَٱللَّهُ عَلِيمٌۢ بِذَاتِ ٱلصُّدُورِ ﴿٤﴾ هغه پوهېږي څه چې په آسمانونو او ځمکو کې دي، او پوهېږي څه چې تاسو پټوئ او څه چې ښکاره کوئ؛ او الله د سینو په رازونو هم ښه پوه دی.
p أَلَمْ يَأْتِكُمْ نَبَؤُا۟ ٱلَّذِينَ كَفَرُوا۟ مِن قَبْلُ فَذَاقُوا۟ وَبَالَ أَمْرِهِمْ وَلَهُمْ عَذَابٌ أَلِيمٌۭ ﴿٥﴾ کیا تمھارے پاس اُن لوگوں کی خبر نہیں آئی جنھوں نے (تم سے) پہلے کفر کیا تو انھوں نے (دنیا میں) اپنے کیے کا وبال چکھا اور (آخرت میں) ان کے لیے دردناک عذاب ہے۔ || أَلَمْ يَأْتِكُمْ نَبَؤُا۟ ٱلَّذِينَ كَفَرُوا۟ مِن قَبْلُ فَذَاقُوا۟ وَبَالَ أَمْرِهِمْ وَلَهُمْ عَذَابٌ أَلِيمٌۭ ﴿٥﴾ Hath not the story reached you of those who disbelieved of old and so did taste the ill-effects of their conduct, and theirs will be a painful doom. (64:5) || أَلَمْ يَأْتِكُمْ نَبَؤُا۟ ٱلَّذِينَ كَفَرُوا۟ مِن قَبْلُ فَذَاقُوا۟ وَبَالَ أَمْرِهِمْ وَلَهُمْ عَذَابٌ أَلِيمٌۭ ﴿٥﴾ ایا تاسو ته د هغو خلکو خبر نه دی راغلی چې (له تاسو) مخکې یې کفر وکړ، نو هغوی (په دنیا کې) د خپلو کړنو وبال وڅکه؟ او (په آخرت کې) د هغوی لپاره دردناک عذاب دی.
p ذَٰلِكَ بِأَنَّهُۥ كَانَت تَّأْتِيهِمْ رُسُلُهُم بِٱلْبَيِّنَٰتِ فَقَالُوٓا۟ أَبَشَرٌۭ يَهْدُونَنَا فَكَفَرُوا۟ وَتَوَلَّوا۟ ۚ وَّٱسْتَغْنَى ٱللَّهُ ۚ وَٱللَّهُ غَنِىٌّ حَمِيدٌۭ ﴿٦﴾ یہ اس لیے ہے کہ ان کے پاس اُن کے رسول واضح نشانیاں لے کر آتے تھے تو انھوں نے کہا کیا (ہم جیسے) بشر ہمیں ہدایت کریں گے؟ تو انھوں نے کفر کیا اور (حق سے) منہ موڑا اور اللہ نے (بھی ان سے) بے نیازی برتی اور اللہ بے نیاز ہر تعریف کے لائق ہے۔ || ذَٰلِكَ بِأَنَّهُۥ كَانَت تَّأْتِيهِمْ رُسُلُهُم بِٱلْبَيِّنَٰتِ فَقَالُوٓا۟ أَبَشَرٌۭ يَهْدُونَنَا فَكَفَرُوا۟ وَتَوَلَّوا۟ ۚ وَّٱسْتَغْنَى ٱللَّهُ ۚ وَٱللَّهُ غَنِىٌّ حَمِيدٌۭ ﴿٦﴾ That was because their messengers (from Allah) kept coming unto them with clear proofs (of Allah's Sovereignty), but they said: Shall mere mortals guide us? So they disbelieved and turned away, and Allah was independent (of them). Allah is Absolute, Owner of Praise. (64:6) || ذَٰلِكَ بِأَنَّهُۥ كَانَت تَّأْتِيهِمْ رُسُلُهُم بِٱلْبَيِّنَٰتِ فَقَالُوٓا۟ أَبَشَرٌۭ يَهْدُونَنَا فَكَفَرُوا۟ وَتَوَلَّوا۟ ۚ وَّٱسْتَغْنَى ٱللَّهُ ۚ وَٱللَّهُ غَنِىٌّ حَمِيدٌۭ ﴿٦﴾ دا ځکه چې هغوی ته به د هغوی رسولان له روښانه نښو سره راتلل، نو هغوی ویل: ایا (زموږ په څېر) بشر به موږ ته هدایت کوي؟ نو کفر یې وکړ او (له حقه) یې مخ واړاوه؛ او الله (هم ترې) بې‌پروايي وکړه؛ او الله بې‌نیازه، د هرې ستاینې وړ دی.
p زَعَمَ ٱلَّذِينَ كَفَرُوٓا۟ أَن لَّن يُبْعَثُوا۟ ۚ قُلْ بَلَىٰ وَرَبِّى لَتُبْعَثُنَّ ثُمَّ لَتُنَبَّؤُنَّ بِمَا عَمِلْتُمْ ۚ وَذَٰلِكَ عَلَى ٱللَّهِ يَسِيرٌۭ ﴿٧﴾ کافر لوگ یہ گمان کرتے ہیں کہ وہ ہرگز (مرنے کے بعد) نہیں اٹھائے جائیں گے آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کہہ دیجیے کیوں نہیں! میرے رب کی قسم! تم ضرور اٹھائے جاؤ گے پھر تمھیں ضرور بتایا جائے گا جو کچھ تم نے کیا اور یہ اللہ پر بہت آسان ہے۔ || زَعَمَ ٱلَّذِينَ كَفَرُوٓا۟ أَن لَّن يُبْعَثُوا۟ ۚ قُلْ بَلَىٰ وَرَبِّى لَتُبْعَثُنَّ ثُمَّ لَتُنَبَّؤُنَّ بِمَا عَمِلْتُمْ ۚ وَذَٰلِكَ عَلَى ٱللَّهِ يَسِيرٌۭ ﴿٧﴾ Those who disbelieve assert that they will not be raised again. Say (unto them, O Muhammad): Yea, verily, by my Lord! ye will be raised again and then ye will be informed of what ye did; and that is easy for Allah. (64:7) || زَعَمَ ٱلَّذِينَ كَفَرُوٓا۟ أَن لَّن يُبْعَثُوا۟ ۚ قُلْ بَلَىٰ وَرَبِّى لَتُبْعَثُنَّ ثُمَّ لَتُنَبَّؤُنَّ بِمَا عَمِلْتُمْ ۚ وَذَٰلِكَ عَلَى ٱللَّهِ يَسِيرٌۭ ﴿٧﴾ کافران ګومان کوي چې هغوی به هېڅکله (له مرګ وروسته) را ونه پاڅول شي؛ تاسو (صلی الله علیه وعلی آله واصحابه وسلم) ووایئ: ولې نه! زما د رب په قسم! تاسو به خامخا راپاڅول شئ، بیا به خامخا درته وویل شي چې څه مو کړي؛ او دا پر الله ډېر اسان دی.
p فَـَٔامِنُوا۟ بِٱللَّهِ وَرَسُولِهِۦ وَٱلنُّورِ ٱلَّذِىٓ أَنزَلْنَا ۚ وَٱللَّهُ بِمَا تَعْمَلُونَ خَبِيرٌۭ ﴿٨﴾ تو ایمان لاؤ اللہ پر اور اس کے رسول پر اور اس نور پر جو ہم نے نازل کیا ہے اور اللہ اُس سے خوب باخبر ہے جو تم کرتے ہو۔ || فَـَٔامِنُوا۟ بِٱللَّهِ وَرَسُولِهِۦ وَٱلنُّورِ ٱلَّذِىٓ أَنزَلْنَا ۚ وَٱللَّهُ بِمَا تَعْمَلُونَ خَبِيرٌۭ ﴿٨﴾ So believe in Allah and His messenger and the light which We have revealed. And Allah is Informed of what ye do. (64:8) || فَـَٔامِنُوا۟ بِٱللَّهِ وَرَسُولِهِۦ وَٱلنُّورِ ٱلَّذِىٓ أَنزَلْنَا ۚ وَٱللَّهُ بِمَا تَعْمَلُونَ خَبِيرٌۭ ﴿٨﴾ نو پر الله، د هغه پر رسول او پر هغه نور ایمان راوړئ چې موږ نازل کړی؛ او الله په هغه څه ښه خبر دی چې تاسو یې کوئ.
p يَوْمَ يَجْمَعُكُمْ لِيَوْمِ ٱلْجَمْعِ ۖ ذَٰلِكَ يَوْمُ ٱلتَّغَابُنِ ۗ وَمَن يُؤْمِنۢ بِٱللَّهِ وَيَعْمَلْ صَٰلِحًۭا يُكَفِّرْ عَنْهُ سَيِّـَٔاتِهِۦ وَيُدْخِلْهُ جَنَّٰتٍۢ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَٰرُ خَٰلِدِينَ فِيهَآ أَبَدًۭا ۚ ذَٰلِكَ ٱلْفَوْزُ ٱلْعَظِيمُ ﴿٩﴾ جس دن وہ تمھیں اکٹھا کرنے کے دن جمع فرمائے گا (وہی) ہار جیت کا دن ہے اور جو اللہ پر ایمان لائے اور نیک عمل کرے اللہ اُس سے اُس کی برائیاں دور کر دے گا اور اُسے ایسے باغات میں داخل فرمائے گا جن کے نیچے نہریں بہتی ہیں وہ اس میں ہمیشہ رہیں گے یہ بہت بڑی کامیابی ہے۔ || يَوْمَ يَجْمَعُكُمْ لِيَوْمِ ٱلْجَمْعِ ۖ ذَٰلِكَ يَوْمُ ٱلتَّغَابُنِ ۗ وَمَن يُؤْمِنۢ بِٱللَّهِ وَيَعْمَلْ صَٰلِحًۭا يُكَفِّرْ عَنْهُ سَيِّـَٔاتِهِۦ وَيُدْخِلْهُ جَنَّٰتٍۢ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَٰرُ خَٰلِدِينَ فِيهَآ أَبَدًۭا ۚ ذَٰلِكَ ٱلْفَوْزُ ٱلْعَظِيمُ ﴿٩﴾ The day when He shall gather you unto the Day of Assembling, that will be a day of mutual disillusion. And whoso believeth in Allah and doeth right, He will remit from him his evil deeds and will bring him unto Gardens underneath which rivers flow, therein to abide for ever. That is the supreme triumph. (64:9) || يَوْمَ يَجْمَعُكُمْ لِيَوْمِ ٱلْجَمْعِ ۖ ذَٰلِكَ يَوْمُ ٱلتَّغَابُنِ ۗ وَمَن يُؤْمِنۢ بِٱللَّهِ وَيَعْمَلْ صَٰلِحًۭا يُكَفِّرْ عَنْهُ سَيِّـَٔاتِهِۦ وَيُدْخِلْهُ جَنَّٰتٍۢ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَٰرُ خَٰلِدِينَ فِيهَآ أَبَدًۭا ۚ ذَٰلِكَ ٱلْفَوْزُ ٱلْعَظِيمُ ﴿٩﴾ په کومه ورځ چې هغه تاسو د راټولېدو په ورځ راټول کړي، (همغه) د ګټې او بایلنې ورځ ده؛ او څوک چې پر الله ایمان راوړي او نېک عمل وکړي، الله به ترې بدۍ لرې کړي او داسې باغونو ته به یې ننباسي چې لاندې یې ویالې بهېږي، هغوی به پکې تل پاتې وي؛ دا ډېره لویه بریا ده.
