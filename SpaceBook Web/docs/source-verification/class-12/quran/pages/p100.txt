p وَٱللَّهُ خَلَقَ كُلَّ دَآبَّةٍۢ مِّن مَّآءٍۢ ۖ فَمِنْهُم مَّن يَمْشِى عَلَىٰ بَطْنِهِۦ وَمِنْهُم مَّن يَمْشِى عَلَىٰ رِجْلَيْنِ وَمِنْهُم مَّن يَمْشِى عَلَىٰٓ أَرْبَعٍۢ ۚ يَخْلُقُ ٱللَّهُ مَا يَشَآءُ ۚ إِنَّ ٱللَّهَ عَلَىٰ كُلِّ شَىْءٍۢ قَدِيرٌۭ ﴿٤٥﴾ اور اللہ نے زمین پر چلنے والے ہر جاندار کو پانی سے پیدا فرمایا اُن میں سے کوئی اپنے پیٹ کے بل چلتا پھرتا ہے اور اُن میں سے کوئی دو ٹانگوں پر چلتا پھرتا ہے اور اُن میں سے کوئی چار پاؤں پر چلتا پھرتا ہے اللہ جو چاہتا ہے پیدا فرماتا ہے۔ بے شک اللہ ہر چیز پر پوری طرح قادر ہے۔ || وَٱللَّهُ خَلَقَ كُلَّ دَآبَّةٍۢ مِّن مَّآءٍۢ ۖ فَمِنْهُم مَّن يَمْشِى عَلَىٰ بَطْنِهِۦ وَمِنْهُم مَّن يَمْشِى عَلَىٰ رِجْلَيْنِ وَمِنْهُم مَّن يَمْشِى عَلَىٰٓ أَرْبَعٍۢ ۚ يَخْلُقُ ٱللَّهُ مَا يَشَآءُ ۚ إِنَّ ٱللَّهَ عَلَىٰ كُلِّ شَىْءٍۢ قَدِيرٌۭ ﴿٤٥﴾ Allah hath created every animal of water. Of them is (a kind) that goeth upon its belly and (a kind) that goeth upon two legs and (a kind) that goeth upon four. Allah createth what He will. Lo! Allah is Able to do all things. (24:45) || وَٱللَّهُ خَلَقَ كُلَّ دَآبَّةٍۢ مِّن مَّآءٍۢ ۖ فَمِنْهُم مَّن يَمْشِى عَلَىٰ بَطْنِهِۦ وَمِنْهُم مَّن يَمْشِى عَلَىٰ رِجْلَيْنِ وَمِنْهُم مَّن يَمْشِى عَلَىٰٓ أَرْبَعٍۢ ۚ يَخْلُقُ ٱللَّهُ مَا يَشَآءُ ۚ إِنَّ ٱللَّهَ عَلَىٰ كُلِّ شَىْءٍۢ قَدِيرٌۭ ﴿٤٥﴾ او الله په ځمکه هر ګرځېدونکی ژوندی له اوبو پیدا کړی؛ له هغوی ځینې پر خپله خېټه ګرځي، ځینې پر دوو پښو ګرځي، او ځینې پر څلورو پښو ګرځي؛ الله چې څه وغواړي پیدا کوي یې؛ بې‌شکه الله پر هر شي پوره قادر دی.
p لَّقَدْ أَنزَلْنَآ ءَايَٰتٍۢ مُّبَيِّنَٰتٍۢ ۚ وَٱللَّهُ يَهْدِى مَن يَشَآءُ إِلَىٰ صِرَٰطٍۢ مُّسْتَقِيمٍۢ ﴿٤٦﴾ یقیناً ہم نے واضح آیتیں نازل کیں اور اللہ جسے چاہتا ہے سیدھے راستے کی طرف ہدایت عطا فرماتا ہے۔ || لَّقَدْ أَنزَلْنَآ ءَايَٰتٍۢ مُّبَيِّنَٰتٍۢ ۚ وَٱللَّهُ يَهْدِى مَن يَشَآءُ إِلَىٰ صِرَٰطٍۢ مُّسْتَقِيمٍۢ ﴿٤٦﴾ Verily We have sent down revelations and explained them. Allah guideth whom He will unto a straight path. (24:46) || لَّقَدْ أَنزَلْنَآ ءَايَٰتٍۢ مُّبَيِّنَٰتٍۢ ۚ وَٱللَّهُ يَهْدِى مَن يَشَآءُ إِلَىٰ صِرَٰطٍۢ مُّسْتَقِيمٍۢ ﴿٤٦﴾ یقیناً موږ روښانه آیتونه نازل کړل، او الله چې چا ته وغواړي سمې لارې ته یې لارښوونه کوي.
p وَيَقُولُونَ ءَامَنَّا بِٱللَّهِ وَبِٱلرَّسُولِ وَأَطَعْنَا ثُمَّ يَتَوَلَّىٰ فَرِيقٌۭ مِّنْهُم مِّنۢ بَعْدِ ذَٰلِكَ ۚ وَمَآ أُو۟لَٰٓئِكَ بِٱلْمُؤْمِنِينَ ﴿٤٧﴾ اور وہ کہتے ہیں ہم اللہ پر اور رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) پر ایمان لائے اور ہم نے اطاعت کی پھر اس کے بعد اُن میں سے ایک گروہ (اس دعوے کے بعد) منہ موڑ لیتا ہے اور ایسے لوگ (حقیقت میں) مومن نہیں ہیں۔ || وَيَقُولُونَ ءَامَنَّا بِٱللَّهِ وَبِٱلرَّسُولِ وَأَطَعْنَا ثُمَّ يَتَوَلَّىٰ فَرِيقٌۭ مِّنْهُم مِّنۢ بَعْدِ ذَٰلِكَ ۚ وَمَآ أُو۟لَٰٓئِكَ بِٱلْمُؤْمِنِينَ ﴿٤٧﴾ And they say: We believe in Allah and the messenger, and we obey; then after that a faction of them turn away. Such are not believers. (24:47) || وَيَقُولُونَ ءَامَنَّا بِٱللَّهِ وَبِٱلرَّسُولِ وَأَطَعْنَا ثُمَّ يَتَوَلَّىٰ فَرِيقٌۭ مِّنْهُم مِّنۢ بَعْدِ ذَٰلِكَ ۚ وَمَآ أُو۟لَٰٓئِكَ بِٱلْمُؤْمِنِينَ ﴿٤٧﴾ او هغوی وايي: موږ پر الله او پر رسول (صلی الله علیه وعلی آله واصحابه وسلم) ایمان راوړ او اطاعت مو وکړ؛ بیا له دې وروسته له هغوی یوه ډله (له دې دعوې وروسته) مخ اړوي؛ او دا ډول خلک (په حقیقت کې) مؤمنان نه دي.
p وَإِذَا دُعُوٓا۟ إِلَى ٱللَّهِ وَرَسُولِهِۦ لِيَحْكُمَ بَيْنَهُمْ إِذَا فَرِيقٌۭ مِّنْهُم مُّعْرِضُونَ ﴿٤٨﴾ اور جب انھیں اللہ اور اُس کے رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کی طرف بلایا جاتا ہے تاکہ وہ اُن کے درمیان (جھگڑوں کا) فیصلہ فرما دیں تب ان میں سے ایک گروہ روگردانی کرنے والا ہوتا ہے۔ || وَإِذَا دُعُوٓا۟ إِلَى ٱللَّهِ وَرَسُولِهِۦ لِيَحْكُمَ بَيْنَهُمْ إِذَا فَرِيقٌۭ مِّنْهُم مُّعْرِضُونَ ﴿٤٨﴾ And when they appeal unto Allah and His messenger to judge between them, lo! a faction of them are averse; (24:48) || وَإِذَا دُعُوٓا۟ إِلَى ٱللَّهِ وَرَسُولِهِۦ لِيَحْكُمَ بَيْنَهُمْ إِذَا فَرِيقٌۭ مِّنْهُم مُّعْرِضُونَ ﴿٤٨﴾ او کله چې هغوی الله او د هغه رسول (صلی الله علیه وعلی آله واصحابه وسلم) ته راوبلل شي تر څو د هغوی ترمنځ (د شخړو) پرېکړه وکړي، نو له هغوی یوه ډله مخ اړوونکې وي.
p وَإِن يَكُن لَّهُمُ ٱلْحَقُّ يَأْتُوٓا۟ إِلَيْهِ مُذْعِنِينَ ﴿٤٩﴾ اور اگر فیصلہ اُن کے حق میں ہو تو وہ (رسول صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کے بڑے فرماں بردار بن کر چلے آتے ہیں۔ || وَإِن يَكُن لَّهُمُ ٱلْحَقُّ يَأْتُوٓا۟ إِلَيْهِ مُذْعِنِينَ ﴿٤٩﴾ But if right had been with them they would have come unto him willingly. (24:49) || وَإِن يَكُن لَّهُمُ ٱلْحَقُّ يَأْتُوٓا۟ إِلَيْهِ مُذْعِنِينَ ﴿٤٩﴾ او که پرېکړه د هغوی په ګټه وي، نو د (رسول صلی الله علیه وعلی آله واصحابه وسلم) لوی فرمانبردار جوړ شي او راشي.
p أَفِى قُلُوبِهِم مَّرَضٌ أَمِ ٱرْتَابُوٓا۟ أَمْ يَخَافُونَ أَن يَحِيفَ ٱللَّهُ عَلَيْهِمْ وَرَسُولُهُۥ ۚ بَلْ أُو۟لَٰٓئِكَ هُمُ ٱلظَّٰلِمُونَ ﴿٥٠﴾ کیا اُن کے دلوں میں کوئی (نفاق کی) بیماری ہے یا وہ شک میں پڑے ہوئے ہیں یا انھیں اندیشہ ہے کہ اللہ اور اُس کا رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) ان پر ظلم کریں گے (نہیں) بلکہ وہی لوگ ظالم ہیں۔ || أَفِى قُلُوبِهِم مَّرَضٌ أَمِ ٱرْتَابُوٓا۟ أَمْ يَخَافُونَ أَن يَحِيفَ ٱللَّهُ عَلَيْهِمْ وَرَسُولُهُۥ ۚ بَلْ أُو۟لَٰٓئِكَ هُمُ ٱلظَّٰلِمُونَ ﴿٥٠﴾ Is there in their hearts a disease, or have they doubts, or fear they lest Allah and His messenger should wrong them in judgment? Nay, but such are evil-doers. (24:50) || أَفِى قُلُوبِهِم مَّرَضٌ أَمِ ٱرْتَابُوٓا۟ أَمْ يَخَافُونَ أَن يَحِيفَ ٱللَّهُ عَلَيْهِمْ وَرَسُولُهُۥ ۚ بَلْ أُو۟لَٰٓئِكَ هُمُ ٱلظَّٰلِمُونَ ﴿٥٠﴾ ایا د هغوی په زړونو کې د (نفاق) کومه ناروغي ده، یا په شک کې لوېدلي، یا وېرېږي چې الله او د هغه رسول (صلی الله علیه وعلی آله واصحابه وسلم) به پرې ظلم وکړي؟ (نه)، بلکې همدوی ظالمان دي.
p إِنَّمَا كَانَ قَوْلَ ٱلْمُؤْمِنِينَ إِذَا دُعُوٓا۟ إِلَى ٱللَّهِ وَرَسُولِهِۦ لِيَحْكُمَ بَيْنَهُمْ أَن يَقُولُوا۟ سَمِعْنَا وَأَطَعْنَا ۚ وَأُو۟لَٰٓئِكَ هُمُ ٱلْمُفْلِحُونَ ﴿٥١﴾ بے شک مومنوں کی بات تو یہی ہوتی ہے کہ جب انھیں اللہ اور اُس کے رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کی طرف بلایا جائے تاکہ وہ اُن کے درمیان فیصلہ فرمائیں کہ وہ کہیں ہم نے (احکام کو) سنا اور ہم نے (ان کی) اطاعت کی اور یہی لوگ فلاح پانے والے ہیں۔ || إِنَّمَا كَانَ قَوْلَ ٱلْمُؤْمِنِينَ إِذَا دُعُوٓا۟ إِلَى ٱللَّهِ وَرَسُولِهِۦ لِيَحْكُمَ بَيْنَهُمْ أَن يَقُولُوا۟ سَمِعْنَا وَأَطَعْنَا ۚ وَأُو۟لَٰٓئِكَ هُمُ ٱلْمُفْلِحُونَ ﴿٥١﴾ The saying of (all true) believers when they appeal unto Allah and His messenger to judge between them is only that they say: We hear and we obey. And such are the successful. (24:51) || إِنَّمَا كَانَ قَوْلَ ٱلْمُؤْمِنِينَ إِذَا دُعُوٓا۟ إِلَى ٱللَّهِ وَرَسُولِهِۦ لِيَحْكُمَ بَيْنَهُمْ أَن يَقُولُوا۟ سَمِعْنَا وَأَطَعْنَا ۚ وَأُو۟لَٰٓئِكَ هُمُ ٱلْمُفْلِحُونَ ﴿٥١﴾ بې‌شکه د مؤمنانو خبره همدا وي چې کله الله او د هغه رسول (صلی الله علیه وعلی آله واصحابه وسلم) ته راوبلل شي تر څو د هغوی ترمنځ پرېکړه وکړي، نو وايي: موږ (حکمونه) واورېدل او (د هغو) اطاعت مو وکړ؛ او همدا خلک بریالي دي.
p وَمَن يُطِعِ ٱللَّهَ وَرَسُولَهُۥ وَيَخْشَ ٱللَّهَ وَيَتَّقْهِ فَأُو۟لَٰٓئِكَ هُمُ ٱلْفَآئِزُونَ ﴿٥٢﴾ اور جو اللہ اور اُس کے رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کی اطاعت کرے اور اللہ سے ڈرے اور اُس (کی نافرمانی) سے بچے تو یہی لوگ کامیاب ہونے والے ہیں۔ || وَمَن يُطِعِ ٱللَّهَ وَرَسُولَهُۥ وَيَخْشَ ٱللَّهَ وَيَتَّقْهِ فَأُو۟لَٰٓئِكَ هُمُ ٱلْفَآئِزُونَ ﴿٥٢﴾ He who obeyeth Allah and His messenger, and feareth Allah, and keepeth duty (unto Him): such indeed are the victorious. (24:52) || وَمَن يُطِعِ ٱللَّهَ وَرَسُولَهُۥ وَيَخْشَ ٱللَّهَ وَيَتَّقْهِ فَأُو۟لَٰٓئِكَ هُمُ ٱلْفَآئِزُونَ ﴿٥٢﴾ او څوک چې د الله او د هغه رسول (صلی الله علیه وعلی آله واصحابه وسلم) اطاعت وکړي، له الله ووېرېږي او د هغه له (نافرمانۍ) ځان وساتي، نو همدا خلک کامیاب دي.
