p إِنَّ ٱلَّذِينَ تَوَفَّىٰهُمُ ٱلْمَلَٰٓئِكَةُ ظَالِمِىٓ أَنفُسِهِمْ قَالُوا۟ فِيمَ كُنتُمْ ۖ قَالُوا۟ كُنَّا مُسْتَضْعَفِينَ فِى ٱلْأَرْضِ ۚ قَالُوٓا۟ أَلَمْ تَكُنْ أَرْضُ ٱللَّهِ وَٰسِعَةًۭ فَتُهَاجِرُوا۟ فِيهَا ۚ فَأُو۟لَٰٓئِكَ مَأْوَىٰهُمْ جَهَنَّمُ ۖ وَسَآءَتْ مَصِيرًا ﴿٩٧﴾ بے شک جن لوگوں کی روح فرشتے اس حال میں قبض کرتے ہیں کہ وہ اپنی جانوں پر ظلم کرنے والے تھے فرشتے پوچھتے ہیں کہ تم لوگ کس حال میں تھے؟ وہ کہتے ہیں ہم تو زمین میں بے بس تھے فرشتے کہتے ہیں کیا اللہ کی زمین وسیع نہ تھی کہ تم اُس میں ہجرت کر جاتے پس یہ لوگ ہیں جن کا ٹھکانا جہنم ہے اور وہ بہت ہی بُرا ٹھکانا ہے۔ || إِنَّ ٱلَّذِينَ تَوَفَّىٰهُمُ ٱلْمَلَٰٓئِكَةُ ظَالِمِىٓ أَنفُسِهِمْ قَالُوا۟ فِيمَ كُنتُمْ ۖ قَالُوا۟ كُنَّا مُسْتَضْعَفِينَ فِى ٱلْأَرْضِ ۚ قَالُوٓا۟ أَلَمْ تَكُنْ أَرْضُ ٱللَّهِ وَٰسِعَةًۭ فَتُهَاجِرُوا۟ فِيهَا ۚ فَأُو۟لَٰٓئِكَ مَأْوَىٰهُمْ جَهَنَّمُ ۖ وَسَآءَتْ مَصِيرًا ﴿٩٧﴾ Lo! as for those whom the angels take (in death) while they wrong themselves, (the angels) will ask: In what were ye engaged? They will say: We were oppressed in the land. (The angels) will say: Was not Allah's earth spacious that ye could have migrated therein? As for such, their habitation will be hell, an evil journey's end; (4:97) || إِنَّ ٱلَّذِينَ تَوَفَّىٰهُمُ ٱلْمَلَٰٓئِكَةُ ظَالِمِىٓ أَنفُسِهِمْ قَالُوا۟ فِيمَ كُنتُمْ ۖ قَالُوا۟ كُنَّا مُسْتَضْعَفِينَ فِى ٱلْأَرْضِ ۚ قَالُوٓا۟ أَلَمْ تَكُنْ أَرْضُ ٱللَّهِ وَٰسِعَةًۭ فَتُهَاجِرُوا۟ فِيهَا ۚ فَأُو۟لَٰٓئِكَ مَأْوَىٰهُمْ جَهَنَّمُ ۖ وَسَآءَتْ مَصِيرًا ﴿٩٧﴾ بې‌شکه هغه کسان چې پرښتې یې روح په داسې حال کې قبضوي چې پر خپلو ځانونو یې ظلم کړی و، پرښتې ترې پوښتي: «تاسو په څه حال کې وئ؟» هغوی وایي: «موږ خو په ځمکه کې بې‌وسه وو.» پرښتې وایي: «ایا د الله ځمکه پراخه نه وه چې تاسو پکې هجرت کړی وای؟» نو دا هغه خلک دي چې ټیکانه یې دوزخ دی، او هغه ډېر بد ټیکانه دی.
p إِلَّا ٱلْمُسْتَضْعَفِينَ مِنَ ٱلرِّجَالِ وَٱلنِّسَآءِ وَٱلْوِلْدَٰنِ لَا يَسْتَطِيعُونَ حِيلَةًۭ وَلَا يَهْتَدُونَ سَبِيلًۭا ﴿٩٨﴾ مگر وہ جو (واقعی) بے بس ہیں مردوں اور عورتوں اور بچوں میں سے جو نہ (ہجرت کی) کسی تدبیر پر قدرت رکھتے ہیں اور نہ (وہاں سے نکلنے کا) کوئی راستہ پاتے ہیں۔ || إِلَّا ٱلْمُسْتَضْعَفِينَ مِنَ ٱلرِّجَالِ وَٱلنِّسَآءِ وَٱلْوِلْدَٰنِ لَا يَسْتَطِيعُونَ حِيلَةًۭ وَلَا يَهْتَدُونَ سَبِيلًۭا ﴿٩٨﴾ Except the feeble among men, and the women, and the children, who are unable to devise a plan and are not shown a way. (4:98) || إِلَّا ٱلْمُسْتَضْعَفِينَ مِنَ ٱلرِّجَالِ وَٱلنِّسَآءِ وَٱلْوِلْدَٰنِ لَا يَسْتَطِيعُونَ حِيلَةًۭ وَلَا يَهْتَدُونَ سَبِيلًۭا ﴿٩٨﴾ مګر هغه نارینه، ښځې او ماشومان چې (رښتیا) بې‌وسه دي، چې نه د (هجرت) کومې تدبیر توان لري او نه (له هغه ځایه د وتلو) کومه لاره مومي.
p فَأُو۟لَٰٓئِكَ عَسَى ٱللَّهُ أَن يَعْفُوَ عَنْهُمْ ۚ وَكَانَ ٱللَّهُ عَفُوًّا غَفُورًۭا ﴿٩٩﴾ تو یہ وہ لوگ ہیں (جن کے بارے میں) امید ہے کہ اللہ اُن سے درگزر فرمائے گا اور اللہ بہت معاف فرمانے والا بڑا ہی بخشنے والا ہے۔ || فَأُو۟لَٰٓئِكَ عَسَى ٱللَّهُ أَن يَعْفُوَ عَنْهُمْ ۚ وَكَانَ ٱللَّهُ عَفُوًّا غَفُورًۭا ﴿٩٩﴾ As for such, it may be that Allah will pardon them. Allah is ever Clement, Forgiving. (4:99) || فَأُو۟لَٰٓئِكَ عَسَى ٱللَّهُ أَن يَعْفُوَ عَنْهُمْ ۚ وَكَانَ ٱللَّهُ عَفُوًّا غَفُورًۭا ﴿٩٩﴾ نو دا هغه خلک دي چې (د هغوی په اړه) هیله ده چې الله به ترې تېر شي؛ او الله ډېر معاف کوونکی، ډېر بښونکی دی.
p ۞ وَمَن يُهَاجِرْ فِى سَبِيلِ ٱللَّهِ يَجِدْ فِى ٱلْأَرْضِ مُرَٰغَمًۭا كَثِيرًۭا وَسَعَةًۭ ۚ وَمَن يَخْرُجْ مِنۢ بَيْتِهِۦ مُهَاجِرًا إِلَى ٱللَّهِ وَرَسُولِهِۦ ثُمَّ يُدْرِكْهُ ٱلْمَوْتُ فَقَدْ وَقَعَ أَجْرُهُۥ عَلَى ٱللَّهِ ۗ وَكَانَ ٱللَّهُ غَفُورًۭا رَّحِيمًۭا ﴿١٠٠﴾ اور جو کوئی اللہ کی راہ میں ہجرت کرے وہ زمین میں پناہ کے لیے بہت سی جگہیں اور وسعت (رزق) پائے گا اور جو کوئی اپنے گھر سے اللہ اور اُس کے رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کی طرف ہجرت کرتے ہوئے نکلے پھر اُسے موت آ پکڑے تو اُس کا اجر اللہ (کے ذمہ کرم) پر ہے اور اللہ بہت بخشنے والا نہایت رحم فرمانے والا ہے۔ || ۞ وَمَن يُهَاجِرْ فِى سَبِيلِ ٱللَّهِ يَجِدْ فِى ٱلْأَرْضِ مُرَٰغَمًۭا كَثِيرًۭا وَسَعَةًۭ ۚ وَمَن يَخْرُجْ مِنۢ بَيْتِهِۦ مُهَاجِرًا إِلَى ٱللَّهِ وَرَسُولِهِۦ ثُمَّ يُدْرِكْهُ ٱلْمَوْتُ فَقَدْ وَقَعَ أَجْرُهُۥ عَلَى ٱللَّهِ ۗ وَكَانَ ٱللَّهُ غَفُورًۭا رَّحِيمًۭا ﴿١٠٠﴾ Whoso migrateth for the cause of Allah will find much refuge and abundance in the earth, and whoso forsaketh his home, a fugitive unto Allah and His messenger, and death overtaketh him, his reward is then incumbent on Allah. Allah is ever Forgiving, Merciful. (4:100) || ۞ وَمَن يُهَاجِرْ فِى سَبِيلِ ٱللَّهِ يَجِدْ فِى ٱلْأَرْضِ مُرَٰغَمًۭا كَثِيرًۭا وَسَعَةًۭ ۚ وَمَن يَخْرُجْ مِنۢ بَيْتِهِۦ مُهَاجِرًا إِلَى ٱللَّهِ وَرَسُولِهِۦ ثُمَّ يُدْرِكْهُ ٱلْمَوْتُ فَقَدْ وَقَعَ أَجْرُهُۥ عَلَى ٱللَّهِ ۗ وَكَانَ ٱللَّهُ غَفُورًۭا رَّحِيمًۭا ﴿١٠٠﴾ او څوک چې د الله په لار کې هجرت وکړي، په ځمکه کې به د پناه لپاره ډېر ځایونه او پراخي (د رزق) ومومي؛ او څوک چې له خپل کوره د الله او د هغه د رسول (صلی الله علیه وعلی آله واصحابه وسلم) لوري ته د هجرت په نیت ووځي، بیا یې مرګ ونیسي، نو د هغه اجر پر الله (د هغه د کرم پر ذمه) دی؛ او الله ډېر بښونکی، بې‌حده مهربان دی.
## علم و عمل کی باتیں || Matters of knowledge and action || د علم او عمل خبرې
- مومن استقامت کا پہاڑ ہوتا ہے۔ ظاہری شکست کی صورت میں بھی اخروی کامیابی اس کے حوصلوں کو بلند رکھتی ہے۔ موت کے خوف اور عارضی دنیوی فائدہ کی خاطر قتال فی سبیل اللہ سے جی نہیں چرانا چاہیے۔ دنیا کا فائدہ حقیر ہے جب کہ آخرت کا گھر پرہیزگاروں کے لیے بہترین ہے۔ || A believer is a mountain of steadfastness. Even in the case of apparent defeat, success in the Hereafter keeps his courage high. Out of fear of death and for temporary worldly benefit one should not shirk fighting in the way of Allah. The benefit of the world is insignificant, whereas the home of the Hereafter is best for the God-fearing. || مؤمن د استقامت غر وي؛ د ظاهري ماتې په صورت کې هم د آخرت بریا یې هوډ لوړ ساتي. د مرګ له وېرې او د لنډمهاله نړیوالې ګټې لپاره باید د الله په لار کې له جګړې زړه ونه ښوئي؛ د دنیا ګټه کوچنۍ ده، په داسې حال کې چې د آخرت کور پرهېزګارانو ته غوره دی.
- کسی خبر اور بالخصوص دور رس نتائج رکھنے والی بات کی تشہیر سے پہلے اچھی طرح تحقیق کر لینی چاہیے اور اس خبر کو اس معاملہ سے متعلق ذمہ دار لوگوں تک پہنچانا چاہیے۔ || Before spreading any news, and especially a matter with far-reaching consequences, one should verify it well, and the news should be conveyed to the people responsible for the matter. || د کوم خبر، په ځانګړي ډول د لرې‌رسو پایلو لرونکې خبرې، له خپرولو مخکې باید ښه تحقیق وشي، او دا خبر باید د هغې چارې مسؤلو کسانو ته ورسول شي.
- جائز کام کی سفارش اور نیکی کی ترغیب ثواب کا کام ہے جبکہ ناجائز کام کی سفارش اور برائی کی ترغیب گناہ کا کام ہے۔ || Recommending a permissible matter and encouraging goodness is a deed of reward, whereas recommending an impermissible matter and encouraging evil is a deed of sin. || د روا کار سپارښتنه او پر نېکۍ هڅونه د ثواب کار دی، په داسې حال کې چې د ناروا کار سپارښتنه او پر بدۍ هڅونه د ګناه کار دی.
- مسلمانوں کو باہمی الفت و محبت کی تلقین کی گئی ہے کہ جب کوئی شخص سلام کرے تو اس کو سلام کا جواب ضرور دیا جائے اور وہ جواب سلام کرنے والے کے سلام سے بہتر ہو یا کم از کم اس جیسا ہی ہو۔ || Muslims have been instructed in mutual affection and love: when someone greets with salam, the reply to his salam must be given, and the reply should be better than the salam of the one greeting, or at least like it. || مسلمانانو ته د منځني الفت او مینې سپارښتنه شوې: کله چې څوک سلام واچوي، د هغه د سلام ځواب دې خامخا ورکړل شي، او هغه ځواب دې د سلام اچوونکي له سلامه غوره یا لږ تر لږه د هغه په څېر وي.
