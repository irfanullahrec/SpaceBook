p يَٰٓأَهْلَ ٱلْكِتَٰبِ قَدْ جَآءَكُمْ رَسُولُنَا يُبَيِّنُ لَكُمْ كَثِيرًۭا مِّمَّا كُنتُمْ تُخْفُونَ مِنَ ٱلْكِتَٰبِ وَيَعْفُوا۟ عَن كَثِيرٍۢ ۚ قَدْ جَآءَكُم مِّنَ ٱللَّهِ نُورٌۭ وَكِتَٰبٌۭ مُّبِينٌۭ ﴿١٥﴾ اے اہلِ کتاب! تمھارے پاس ہمارے (یہ) رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) آ چکے ہیں جو تم پر کتاب میں سے بہت سی وہ باتیں کھول کھول کر بیان فرماتے ہیں جنھیں تم چھپا رہے تھے اور بہت سی باتوں سے درگزر فرماتے ہیں یقیناً اللہ کی طرف سے تمھارے پاس آیا نور اور روشن کتاب۔ || يَٰٓأَهْلَ ٱلْكِتَٰبِ قَدْ جَآءَكُمْ رَسُولُنَا يُبَيِّنُ لَكُمْ كَثِيرًۭا مِّمَّا كُنتُمْ تُخْفُونَ مِنَ ٱلْكِتَٰبِ وَيَعْفُوا۟ عَن كَثِيرٍۢ ۚ قَدْ جَآءَكُم مِّنَ ٱللَّهِ نُورٌۭ وَكِتَٰبٌۭ مُّبِينٌۭ ﴿١٥﴾ O People of the Scripture! Now hath Our messenger come unto you, expounding unto you much of that which ye used to hide in the Scripture, and forgiving much. now hath come unto you light from Allah and plain Scripture, (5:15) || يَٰٓأَهْلَ ٱلْكِتَٰبِ قَدْ جَآءَكُمْ رَسُولُنَا يُبَيِّنُ لَكُمْ كَثِيرًۭا مِّمَّا كُنتُمْ تُخْفُونَ مِنَ ٱلْكِتَٰبِ وَيَعْفُوا۟ عَن كَثِيرٍۢ ۚ قَدْ جَآءَكُم مِّنَ ٱللَّهِ نُورٌۭ وَكِتَٰبٌۭ مُّبِينٌۭ ﴿١٥﴾ ای اهل کتابو! تاسو ته زموږ (دا) رسول (صلی الله علیه وعلی آله واصحابه وسلم) راغلی چې تاسو ته د کتاب ډېرې هغه خبرې ښه روښانه بیانوي چې تاسو یې پټولې، او له ډېرو خبرو تېرېږي؛ یقیناً تاسو ته د الله له لوري نور او روښانه کتاب راغلی.
p يَهْدِى بِهِ ٱللَّهُ مَنِ ٱتَّبَعَ رِضْوَٰنَهُۥ سُبُلَ ٱلسَّلَٰمِ وَيُخْرِجُهُم مِّنَ ٱلظُّلُمَٰتِ إِلَى ٱلنُّورِ بِإِذْنِهِۦ وَيَهْدِيهِمْ إِلَىٰ صِرَٰطٍۢ مُّسْتَقِيمٍۢ ﴿١٦﴾ جس کے ذریعہ اللہ ان لوگوں کو سلامتی کی راہیں دکھاتا ہے جو اس کی خوشنودی کے طالب ہیں اور وہ اُن کو اپنی توفیق سے اندھیروں سے روشنی کی طرف نکال لاتا ہے اور انھیں سیدھے راستہ کی طرف ہدایت عطا فرماتا ہے۔ || يَهْدِى بِهِ ٱللَّهُ مَنِ ٱتَّبَعَ رِضْوَٰنَهُۥ سُبُلَ ٱلسَّلَٰمِ وَيُخْرِجُهُم مِّنَ ٱلظُّلُمَٰتِ إِلَى ٱلنُّورِ بِإِذْنِهِۦ وَيَهْدِيهِمْ إِلَىٰ صِرَٰطٍۢ مُّسْتَقِيمٍۢ ﴿١٦﴾ Whereby Allah guideth him who seeketh His good pleasure unto paths of peace. He bringeth them out of darkness unto light by His decree, and guideth them unto a straight path. (5:16) || يَهْدِى بِهِ ٱللَّهُ مَنِ ٱتَّبَعَ رِضْوَٰنَهُۥ سُبُلَ ٱلسَّلَٰمِ وَيُخْرِجُهُم مِّنَ ٱلظُّلُمَٰتِ إِلَى ٱلنُّورِ بِإِذْنِهِۦ وَيَهْدِيهِمْ إِلَىٰ صِرَٰطٍۢ مُّسْتَقِيمٍۢ ﴿١٦﴾ چې د هغه په وسیله الله هغو خلکو ته د سلامتۍ لارې ښیي چې د هغه د خوښۍ غوښتونکي دي، او په خپل توفیق یې له تیارو رڼا ته راباسي، او سمې لارې ته یې هدایت کوي.
p لَّقَدْ كَفَرَ ٱلَّذِينَ قَالُوٓا۟ إِنَّ ٱللَّهَ هُوَ ٱلْمَسِيحُ ٱبْنُ مَرْيَمَ ۚ قُلْ فَمَن يَمْلِكُ مِنَ ٱللَّهِ شَيْـًٔا إِنْ أَرَادَ أَن يُهْلِكَ ٱلْمَسِيحَ ٱبْنَ مَرْيَمَ وَأُمَّهُۥ وَمَن فِى ٱلْأَرْضِ جَمِيعًۭا ۗ وَلِلَّهِ مُلْكُ ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ وَمَا بَيْنَهُمَا ۚ يَخْلُقُ مَا يَشَآءُ ۚ وَٱللَّهُ عَلَىٰ كُلِّ شَىْءٍۢ قَدِيرٌۭ ﴿١٧﴾ یقیناً اُن لوگوں نے کفر کیا جنھوں نے کہا بے شک اللہ مسیح ابنِ مریم (علیہ السلام) ہی ہے آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) فرما دیجیے کون ہے جو اللہ کے سامنے کچھ بھی اختیار رکھتا ہے اگر اللہ ارادہ کرے کہ مسیح ابنِ مریم (علیہ السلام) اور اُن کی والدہ اور سب زمین والوں کو ہلاک کر دے (تو اللہ کو کوئی نہیں روک سکتا) اور اللہ ہی کے لیے آسمانوں اور زمینوں کی بادشاہت ہے اور اُس کی بھی جو کچھ ان دونوں کے درمیان ہے وہ جو چاہتا ہے پیدا فرما دیتا ہے اور اللہ ہر چیز پر قدرت رکھنے والا ہے۔ || لَّقَدْ كَفَرَ ٱلَّذِينَ قَالُوٓا۟ إِنَّ ٱللَّهَ هُوَ ٱلْمَسِيحُ ٱبْنُ مَرْيَمَ ۚ قُلْ فَمَن يَمْلِكُ مِنَ ٱللَّهِ شَيْـًٔا إِنْ أَرَادَ أَن يُهْلِكَ ٱلْمَسِيحَ ٱبْنَ مَرْيَمَ وَأُمَّهُۥ وَمَن فِى ٱلْأَرْضِ جَمِيعًۭا ۗ وَلِلَّهِ مُلْكُ ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ وَمَا بَيْنَهُمَا ۚ يَخْلُقُ مَا يَشَآءُ ۚ وَٱللَّهُ عَلَىٰ كُلِّ شَىْءٍۢ قَدِيرٌۭ ﴿١٧﴾ They indeed have disbelieved who say: Lo! Allah is the Messiah, son of Mary. Say: Who then can do aught against Allah, if He had willed to destroy the Messiah son of Mary, and his mother and everyone on earth? Allah's is the Sovereignty of the heavens and the earth and all that is between them. He createth what He will. And Allah is Able to do all things. (5:17) || لَّقَدْ كَفَرَ ٱلَّذِينَ قَالُوٓا۟ إِنَّ ٱللَّهَ هُوَ ٱلْمَسِيحُ ٱبْنُ مَرْيَمَ ۚ قُلْ فَمَن يَمْلِكُ مِنَ ٱللَّهِ شَيْـًٔا إِنْ أَرَادَ أَن يُهْلِكَ ٱلْمَسِيحَ ٱبْنَ مَرْيَمَ وَأُمَّهُۥ وَمَن فِى ٱلْأَرْضِ جَمِيعًۭا ۗ وَلِلَّهِ مُلْكُ ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ وَمَا بَيْنَهُمَا ۚ يَخْلُقُ مَا يَشَآءُ ۚ وَٱللَّهُ عَلَىٰ كُلِّ شَىْءٍۢ قَدِيرٌۭ ﴿١٧﴾ یقیناً هغو کسانو کفر وکړ چې ویل یې: «بې‌شکه الله همدا مسیح ابن مریم (علیه السلام) دی.» تاسو (صلی الله علیه وعلی آله واصحابه وسلم) ووایئ: څوک دی چې د الله په وړاندې هېڅ واک ولري، که الله اراده وکړي چې مسیح ابن مریم (علیه السلام)، د هغه مور او د ځمکې ټول اوسېدونکي هلاک کړي؟ (نو الله هېڅوک نه شي منع کولی)؛ او د الله ده د آسمانونو او ځمکې پاچاهي او د هغه څه چې د دواړو ترمنځ دي؛ هغه چې څه غواړي پیدا کوي، او الله پر هر شي قدرت لرونکی دی.
p وَقَالَتِ ٱلْيَهُودُ وَٱلنَّصَٰرَىٰ نَحْنُ أَبْنَٰٓؤُا۟ ٱللَّهِ وَأَحِبَّٰٓؤُهُۥ ۚ قُلْ فَلِمَ يُعَذِّبُكُم بِذُنُوبِكُم ۖ بَلْ أَنتُم بَشَرٌۭ مِّمَّنْ خَلَقَ ۚ يَغْفِرُ لِمَن يَشَآءُ وَيُعَذِّبُ مَن يَشَآءُ ۚ وَلِلَّهِ مُلْكُ ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ وَمَا بَيْنَهُمَا ۖ وَإِلَيْهِ ٱلْمَصِيرُ ﴿١٨﴾ اور یہود اور نصاریٰ نے کہا کہ ہم اللہ کے بیٹے ہیں اور اُس کے بڑے محبوب ہیں آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) فرما دیجیے کہ پھر کیوں وہ تمھیں تمھارے گناہوں کی وجہ سے عذاب دیتا ہے بلکہ تم بھی بشر ہو اُس (مخلوق) میں سے جسے اُس نے پیدا فرمایا وہ جسے چاہے گا معاف فرما دے گا اور جسے چاہے گا عذاب دے گا اور اللہ ہی کے لیے آسمانوں اور زمینوں کی بادشاہت ہے اور اُس کی بھی جو کچھ ان دونوں کے درمیان ہے اور اسی کی طرف پلٹنا ہے۔ || وَقَالَتِ ٱلْيَهُودُ وَٱلنَّصَٰرَىٰ نَحْنُ أَبْنَٰٓؤُا۟ ٱللَّهِ وَأَحِبَّٰٓؤُهُۥ ۚ قُلْ فَلِمَ يُعَذِّبُكُم بِذُنُوبِكُم ۖ بَلْ أَنتُم بَشَرٌۭ مِّمَّنْ خَلَقَ ۚ يَغْفِرُ لِمَن يَشَآءُ وَيُعَذِّبُ مَن يَشَآءُ ۚ وَلِلَّهِ مُلْكُ ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ وَمَا بَيْنَهُمَا ۖ وَإِلَيْهِ ٱلْمَصِيرُ ﴿١٨﴾ The Jews and Christians say: We are sons of Allah and His loved ones. Say: Why then doth He chastise you for your sins? Nay, ye are but mortals of His creating. He forgiveth whom He will, and chastiseth whom He will. Allah's is the Sovereignty of the heavens and the earth and all that is between them, and unto Him is the journeying. (5:18) || وَقَالَتِ ٱلْيَهُودُ وَٱلنَّصَٰرَىٰ نَحْنُ أَبْنَٰٓؤُا۟ ٱللَّهِ وَأَحِبَّٰٓؤُهُۥ ۚ قُلْ فَلِمَ يُعَذِّبُكُم بِذُنُوبِكُم ۖ بَلْ أَنتُم بَشَرٌۭ مِّمَّنْ خَلَقَ ۚ يَغْفِرُ لِمَن يَشَآءُ وَيُعَذِّبُ مَن يَشَآءُ ۚ وَلِلَّهِ مُلْكُ ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ وَمَا بَيْنَهُمَا ۖ وَإِلَيْهِ ٱلْمَصِيرُ ﴿١٨﴾ او یهودو او نصاراوو وویل: «موږ د الله زامن او د هغه ډېر ګران یو.» تاسو (صلی الله علیه وعلی آله واصحابه وسلم) ووایئ: نو بیا ولې تاسو ستاسو د ګناهونو له امله عذابوي؟ بلکې تاسو هم د هغه له پیدا کړي (مخلوق) څخه انسانان یاست؛ چا ته چې وغواړي، بښي یې، او چا ته چې وغواړي، عذابوي یې؛ او د الله ده د آسمانونو او ځمکې پاچاهي او د هغه څه چې د دواړو ترمنځ دي، او هغه ته بېرته ورتلل دي.
