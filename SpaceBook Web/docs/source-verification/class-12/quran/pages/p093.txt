p ٱلْخَبِيثَٰتُ لِلْخَبِيثِينَ وَٱلْخَبِيثُونَ لِلْخَبِيثَٰتِ ۖ وَٱلطَّيِّبَٰتُ لِلطَّيِّبِينَ وَٱلطَّيِّبُونَ لِلطَّيِّبَٰتِ ۚ أُو۟لَٰٓئِكَ مُبَرَّءُونَ مِمَّا يَقُولُونَ ۖ لَهُم مَّغْفِرَةٌۭ وَرِزْقٌۭ كَرِيمٌۭ ﴿٢٦﴾ ناپاک عورتیں، ناپاک مردوں کے لائق ہیں اور ناپاک مرد ناپاک عورتوں کے لائق ہیں اور پاکیزہ عورتیں پاکیزہ مردوں کے لائق ہیں اور پاکیزہ مرد پاکیزہ عورتوں کے لائق ہیں یہ لوگ بَری ہیں اُن (تہمت والی) باتوں سے جو (ناپاک) لوگ بیان کر رہے ہیں ان (پاکیزہ افراد) کے لیے مغفرت اور عزت والا رزق ہے۔ || ٱلْخَبِيثَٰتُ لِلْخَبِيثِينَ وَٱلْخَبِيثُونَ لِلْخَبِيثَٰتِ ۖ وَٱلطَّيِّبَٰتُ لِلطَّيِّبِينَ وَٱلطَّيِّبُونَ لِلطَّيِّبَٰتِ ۚ أُو۟لَٰٓئِكَ مُبَرَّءُونَ مِمَّا يَقُولُونَ ۖ لَهُم مَّغْفِرَةٌۭ وَرِزْقٌۭ كَرِيمٌۭ ﴿٢٦﴾ Vile women are for vile men, and vile men for vile women. Good women are for good men, and good men for good women; such are innocent of that which people say: For them is pardon and a bountiful provision. (24:26) || ٱلْخَبِيثَٰتُ لِلْخَبِيثِينَ وَٱلْخَبِيثُونَ لِلْخَبِيثَٰتِ ۖ وَٱلطَّيِّبَٰتُ لِلطَّيِّبِينَ وَٱلطَّيِّبُونَ لِلطَّيِّبَٰتِ ۚ أُو۟لَٰٓئِكَ مُبَرَّءُونَ مِمَّا يَقُولُونَ ۖ لَهُم مَّغْفِرَةٌۭ وَرِزْقٌۭ كَرِيمٌۭ ﴿٢٦﴾ ناپاکې ښځې د ناپاکو نارینه وو وړ دي او ناپاک نارینه د ناپاکو ښځو وړ دي؛ او پاکې ښځې د پاکو نارینه وو وړ دي او پاک نارینه د پاکو ښځو وړ دي؛ دا خلک له هغو (د تور) خبرو بري دي چې (ناپاک) خلک یې بیانوي؛ د دې (پاکو کسانو) لپاره بښنه او د عزت رزق دی.
p يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ لَا تَدْخُلُوا۟ بُيُوتًا غَيْرَ بُيُوتِكُمْ حَتَّىٰ تَسْتَأْنِسُوا۟ وَتُسَلِّمُوا۟ عَلَىٰٓ أَهْلِهَا ۚ ذَٰلِكُمْ خَيْرٌۭ لَّكُمْ لَعَلَّكُمْ تَذَكَّرُونَ ﴿٢٧﴾ اے ایمان والو! اپنے گھروں کے سوا دوسرے گھروں میں داخل نہ ہونا یہاں تک کہ تم اجازت (نہ) لے لو اور گھر والوں کو سلام (نہ) کر لو یہ تمھارے لیے بہتر ہے تاکہ تم نصیحت حاصل کرو۔ || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ لَا تَدْخُلُوا۟ بُيُوتًا غَيْرَ بُيُوتِكُمْ حَتَّىٰ تَسْتَأْنِسُوا۟ وَتُسَلِّمُوا۟ عَلَىٰٓ أَهْلِهَا ۚ ذَٰلِكُمْ خَيْرٌۭ لَّكُمْ لَعَلَّكُمْ تَذَكَّرُونَ ﴿٢٧﴾ O ye who believe! Enter not houses other than your own without first announcing your presence and invoking peace upon the folk thereof. That is better for you, that ye may be heedful. (24:27) || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ لَا تَدْخُلُوا۟ بُيُوتًا غَيْرَ بُيُوتِكُمْ حَتَّىٰ تَسْتَأْنِسُوا۟ وَتُسَلِّمُوا۟ عَلَىٰٓ أَهْلِهَا ۚ ذَٰلِكُمْ خَيْرٌۭ لَّكُمْ لَعَلَّكُمْ تَذَكَّرُونَ ﴿٢٧﴾ ای مؤمنانو! له خپلو کورونو پرته نورو کورونو ته مه ننوځئ تر څو چې اجازه وانه خلئ او د کور خلکو ته سلام وانه چوئ؛ دا ستاسو لپاره غوره دی تر څو نصیحت واخلئ.
p فَإِن لَّمْ تَجِدُوا۟ فِيهَآ أَحَدًۭا فَلَا تَدْخُلُوهَا حَتَّىٰ يُؤْذَنَ لَكُمْ ۖ وَإِن قِيلَ لَكُمُ ٱرْجِعُوا۟ فَٱرْجِعُوا۟ ۖ هُوَ أَزْكَىٰ لَكُمْ ۚ وَٱللَّهُ بِمَا تَعْمَلُونَ عَلِيمٌۭ ﴿٢٨﴾ پھر اگر تم ان (گھروں) میں کسی کو موجود نہ پاؤ تو اُن میں مت داخل ہو یہاں تک کہ تمھیں اجازت دی جائے اور اگر تم سے کہا جائے کہ واپس چلے جاؤ تو تم واپس چلے جاؤ یہ تمھارے لیے پاکیزہ ترین بات ہے اور اللہ ان کاموں کو جو تم کرتے ہو خوب جانتا ہے۔ || فَإِن لَّمْ تَجِدُوا۟ فِيهَآ أَحَدًۭا فَلَا تَدْخُلُوهَا حَتَّىٰ يُؤْذَنَ لَكُمْ ۖ وَإِن قِيلَ لَكُمُ ٱرْجِعُوا۟ فَٱرْجِعُوا۟ ۖ هُوَ أَزْكَىٰ لَكُمْ ۚ وَٱللَّهُ بِمَا تَعْمَلُونَ عَلِيمٌۭ ﴿٢٨﴾ And if ye find no-one therein, still enter not until permission hath been given. And if it be said unto you: Go away again, then go away, for it is purer for you. Allah knoweth what ye do. (24:28) || فَإِن لَّمْ تَجِدُوا۟ فِيهَآ أَحَدًۭا فَلَا تَدْخُلُوهَا حَتَّىٰ يُؤْذَنَ لَكُمْ ۖ وَإِن قِيلَ لَكُمُ ٱرْجِعُوا۟ فَٱرْجِعُوا۟ ۖ هُوَ أَزْكَىٰ لَكُمْ ۚ وَٱللَّهُ بِمَا تَعْمَلُونَ عَلِيمٌۭ ﴿٢٨﴾ بیا که په هغو (کورونو) کې څوک ونه مومئ، نو ورته مه ننوځئ تر څو چې اجازه درکړل شي؛ او که درته وویل شي چې «بېرته لاړ شئ»، نو بېرته لاړ شئ؛ دا ستاسو لپاره تر ټولو پاکه خبره ده؛ او الله په هغو کارونو ښه پوه دی چې تاسو یې کوئ.
p لَّيْسَ عَلَيْكُمْ جُنَاحٌ أَن تَدْخُلُوا۟ بُيُوتًا غَيْرَ مَسْكُونَةٍۢ فِيهَا مَتَٰعٌۭ لَّكُمْ ۚ وَٱللَّهُ يَعْلَمُ مَا تُبْدُونَ وَمَا تَكْتُمُونَ ﴿٢٩﴾ تم پر کوئی گناہ نہیں ہے کہ تم ایسے گھروں میں (اجازت کے بغیر) داخل ہو، جن میں کوئی رہائش پذیر نہ ہو جن میں تمھارا کوئی سامان ہو اور اللہ سب جانتا ہے جو کچھ تم ظاہر کرتے ہو اور جو کچھ تم چھپاتے ہو۔ || لَّيْسَ عَلَيْكُمْ جُنَاحٌ أَن تَدْخُلُوا۟ بُيُوتًا غَيْرَ مَسْكُونَةٍۢ فِيهَا مَتَٰعٌۭ لَّكُمْ ۚ وَٱللَّهُ يَعْلَمُ مَا تُبْدُونَ وَمَا تَكْتُمُونَ ﴿٢٩﴾ (It is) no sin for you to enter uninhabited houses wherein is comfort for you. Allah knoweth what ye proclaim and what ye hide. (24:29) || لَّيْسَ عَلَيْكُمْ جُنَاحٌ أَن تَدْخُلُوا۟ بُيُوتًا غَيْرَ مَسْكُونَةٍۢ فِيهَا مَتَٰعٌۭ لَّكُمْ ۚ وَٱللَّهُ يَعْلَمُ مَا تُبْدُونَ وَمَا تَكْتُمُونَ ﴿٢٩﴾ پر تاسو هېڅ ګناه نشته چې داسې کورونو ته (له اجازې پرته) ننوځئ چې پکې څوک نه اوسېږي او ستاسو کوم سامان پکې وي؛ او الله ټول پوهېږي چې تاسو څه ښکاره کوئ او څه پټوئ.
p قُل لِّلْمُؤْمِنِينَ يَغُضُّوا۟ مِنْ أَبْصَٰرِهِمْ وَيَحْفَظُوا۟ فُرُوجَهُمْ ۚ ذَٰلِكَ أَزْكَىٰ لَهُمْ ۗ إِنَّ ٱللَّهَ خَبِيرٌۢ بِمَا يَصْنَعُونَ ﴿٣٠﴾ آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) فرما دیجیے مومن مردوں سے کہ اپنی نگاہیں نیچی رکھیں اور اپنی شرم گاہوں کی حفاظت کریں یہ ان کے لیے پاکیزہ ترین بات ہے بے شک اللہ ان کاموں سے خوب واقف ہے جو وہ کر رہے ہیں۔ || قُل لِّلْمُؤْمِنِينَ يَغُضُّوا۟ مِنْ أَبْصَٰرِهِمْ وَيَحْفَظُوا۟ فُرُوجَهُمْ ۚ ذَٰلِكَ أَزْكَىٰ لَهُمْ ۗ إِنَّ ٱللَّهَ خَبِيرٌۢ بِمَا يَصْنَعُونَ ﴿٣٠﴾ Tell the believing men to lower their gaze and be modest. That is purer for them. Lo! Allah is aware of what they do. (24:30) || قُل لِّلْمُؤْمِنِينَ يَغُضُّوا۟ مِنْ أَبْصَٰرِهِمْ وَيَحْفَظُوا۟ فُرُوجَهُمْ ۚ ذَٰلِكَ أَزْكَىٰ لَهُمْ ۗ إِنَّ ٱللَّهَ خَبِيرٌۢ بِمَا يَصْنَعُونَ ﴿٣٠﴾ تاسو (صلی الله علیه وعلی آله واصحابه وسلم) مؤمنو نارینه وو ته ووایئ چې خپلې سترګې ټیټې وساتي او د خپلو شرمګاهونو ساتنه وکړي؛ دا د هغوی لپاره تر ټولو پاکه خبره ده؛ بې‌شکه الله له هغو کارونو ښه خبر دی چې هغوی یې کوي.
