p ۞ وَأَقْسَمُوا۟ بِٱللَّهِ جَهْدَ أَيْمَٰنِهِمْ لَئِنْ أَمَرْتَهُمْ لَيَخْرُجُنَّ ۖ قُل لَّا تُقْسِمُوا۟ ۖ طَاعَةٌۭ مَّعْرُوفَةٌ ۚ إِنَّ ٱللَّهَ خَبِيرٌۢ بِمَا تَعْمَلُونَ ﴿٥٣﴾ اور یہ (منافقین) اللہ کی سخت قسمیں کھاتے تھے کہ اگر آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) انھیں حکم دیں تو وہ (جہاد کے لیے) ضرور نکلیں گے آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) فرما دیجیے قسمیں مت کھاؤ (تمھاری) اطاعت (خوب) معلوم ہے بے شک اللہ خوب خبردار ہے اُس سے جو تم کرتے ہو۔ || ۞ وَأَقْسَمُوا۟ بِٱللَّهِ جَهْدَ أَيْمَٰنِهِمْ لَئِنْ أَمَرْتَهُمْ لَيَخْرُجُنَّ ۖ قُل لَّا تُقْسِمُوا۟ ۖ طَاعَةٌۭ مَّعْرُوفَةٌ ۚ إِنَّ ٱللَّهَ خَبِيرٌۢ بِمَا تَعْمَلُونَ ﴿٥٣﴾ They swear by Allah solemnly that, if thou order them, they will go forth. Say: Swear not; known obedience (is better). Lo! Allah is Informed of what ye do. (24:53) || ۞ وَأَقْسَمُوا۟ بِٱللَّهِ جَهْدَ أَيْمَٰنِهِمْ لَئِنْ أَمَرْتَهُمْ لَيَخْرُجُنَّ ۖ قُل لَّا تُقْسِمُوا۟ ۖ طَاعَةٌۭ مَّعْرُوفَةٌ ۚ إِنَّ ٱللَّهَ خَبِيرٌۢ بِمَا تَعْمَلُونَ ﴿٥٣﴾ او دې (منافقانو) د الله سخت قسمونه خوړل چې که تاسو (صلی الله علیه وعلی آله واصحابه وسلم) هغوی ته حکم وکړئ، نو خامخا به (جهاد ته) ووځي؛ تاسو (صلی الله علیه وعلی آله واصحابه وسلم) ووایئ: قسمونه مه خورئ، (ستاسو) اطاعت (ښه) معلوم دی؛ بې‌شکه الله په هغه څه ښه خبر دی چې تاسو یې کوئ.
p قُلْ أَطِيعُوا۟ ٱللَّهَ وَأَطِيعُوا۟ ٱلرَّسُولَ ۖ فَإِن تَوَلَّوْا۟ فَإِنَّمَا عَلَيْهِ مَا حُمِّلَ وَعَلَيْكُم مَّا حُمِّلْتُمْ ۖ وَإِن تُطِيعُوهُ تَهْتَدُوا۟ ۚ وَمَا عَلَى ٱلرَّسُولِ إِلَّا ٱلْبَلَٰغُ ٱلْمُبِينُ ﴿٥٤﴾ آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) فرما دیجیے کہ اطاعت کرو اللہ کی اور اطاعت کرو رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کی پھر اگر تم نے منہ موڑ لیا تو بے شک رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کے ذمہ وہی کچھ ہے جو اُن پر لازم کیا گیا اور تمھارے ذمہ وہ ہے جو تم پر لازم کیا گیا اور اگر تم اُن کی اطاعت کرو گے تو ہدایت پا جاؤ گے اور رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کے ذمہ تو صرف واضح طور پر (حق بات) پہنچا دینا ہے۔ || قُلْ أَطِيعُوا۟ ٱللَّهَ وَأَطِيعُوا۟ ٱلرَّسُولَ ۖ فَإِن تَوَلَّوْا۟ فَإِنَّمَا عَلَيْهِ مَا حُمِّلَ وَعَلَيْكُم مَّا حُمِّلْتُمْ ۖ وَإِن تُطِيعُوهُ تَهْتَدُوا۟ ۚ وَمَا عَلَى ٱلرَّسُولِ إِلَّا ٱلْبَلَٰغُ ٱلْمُبِينُ ﴿٥٤﴾ Say: Obey Allah and obey the messenger. But if ye turn away, then (it is) for him (to do) only that wherewith he hath been charged, and for you (to do) only that wherewith ye have been charged. If ye obey him, ye will go aright. But the messenger hath no other charge than to convey (the message) plainly. (24:54) || قُلْ أَطِيعُوا۟ ٱللَّهَ وَأَطِيعُوا۟ ٱلرَّسُولَ ۖ فَإِن تَوَلَّوْا۟ فَإِنَّمَا عَلَيْهِ مَا حُمِّلَ وَعَلَيْكُم مَّا حُمِّلْتُمْ ۖ وَإِن تُطِيعُوهُ تَهْتَدُوا۟ ۚ وَمَا عَلَى ٱلرَّسُولِ إِلَّا ٱلْبَلَٰغُ ٱلْمُبِينُ ﴿٥٤﴾ تاسو (صلی الله علیه وعلی آله واصحابه وسلم) ووایئ: د الله اطاعت وکړئ او د رسول (صلی الله علیه وعلی آله واصحابه وسلم) اطاعت وکړئ؛ بیا که مو مخ واړاوه، نو بې‌شکه د رسول (صلی الله علیه وعلی آله واصحابه وسلم) پر غاړه همغه دي چې پر هغه لازم شوي، او ستاسو پر غاړه همغه دي چې پر تاسو لازم شوي؛ او که د هغه اطاعت وکړئ، نو هدایت به ومومئ؛ او د رسول (صلی الله علیه وعلی آله واصحابه وسلم) پر غاړه خو یوازې په ښکاره ډول (د حق خبرې) رسول دي.
p وَعَدَ ٱللَّهُ ٱلَّذِينَ ءَامَنُوا۟ مِنكُمْ وَعَمِلُوا۟ ٱلصَّٰلِحَٰتِ لَيَسْتَخْلِفَنَّهُمْ فِى ٱلْأَرْضِ كَمَا ٱسْتَخْلَفَ ٱلَّذِينَ مِن قَبْلِهِمْ وَلَيُمَكِّنَنَّ لَهُمْ دِينَهُمُ ٱلَّذِى ٱرْتَضَىٰ لَهُمْ وَلَيُبَدِّلَنَّهُم مِّنۢ بَعْدِ خَوْفِهِمْ أَمْنًۭا ۚ يَعْبُدُونَنِى لَا يُشْرِكُونَ بِى شَيْـًۭٔا ۚ وَمَن كَفَرَ بَعْدَ ذَٰلِكَ فَأُو۟لَٰٓئِكَ هُمُ ٱلْفَٰسِقُونَ ﴿٥٥﴾ اللہ نے تم میں سے اُن لوگوں سے وعدہ فرمایا ہے جو ایمان لائے اور جنھوں نے نیک اعمال کیے کہ وہ ضرور بالضرور انھیں زمین میں خلافت عطا فرمائے گا جس طرح اُس نے اُن لوگوں کو خلافت عطا فرمائی جو ان سے پہلے تھے اور اُن کے لیے اُس دین کو ضرور غالب کرے گا جو اُس نے اُن کے لیے پسند کیا اور اُن کے خوف (کی کیفیت) کو ضرور امن سے بدل دے گا (بس) وہ میری عبادت کریں گے میرے ساتھ کسی کو شریک نہ ٹھہرائیں گے اور جو اس کے بعد (بھی) ناشکری کریں تو ایسے ہی لوگ فاسق ہیں۔ || وَعَدَ ٱللَّهُ ٱلَّذِينَ ءَامَنُوا۟ مِنكُمْ وَعَمِلُوا۟ ٱلصَّٰلِحَٰتِ لَيَسْتَخْلِفَنَّهُمْ فِى ٱلْأَرْضِ كَمَا ٱسْتَخْلَفَ ٱلَّذِينَ مِن قَبْلِهِمْ وَلَيُمَكِّنَنَّ لَهُمْ دِينَهُمُ ٱلَّذِى ٱرْتَضَىٰ لَهُمْ وَلَيُبَدِّلَنَّهُم مِّنۢ بَعْدِ خَوْفِهِمْ أَمْنًۭا ۚ يَعْبُدُونَنِى لَا يُشْرِكُونَ بِى شَيْـًۭٔا ۚ وَمَن كَفَرَ بَعْدَ ذَٰلِكَ فَأُو۟لَٰٓئِكَ هُمُ ٱلْفَٰسِقُونَ ﴿٥٥﴾ Allah hath promised such of you as believe and do good work that He will surely make them to succeed (the present rulers) in the earth even as He caused those who were before them to succeed (others); and that He will surely establish for them their religion which He hath approved for them, and will give them in exchange safety after their fear. They serve Me. They ascribe no thing as partner unto Me. Those who disbelieve henceforth, they are the miscreants. (24:55) || وَعَدَ ٱللَّهُ ٱلَّذِينَ ءَامَنُوا۟ مِنكُمْ وَعَمِلُوا۟ ٱلصَّٰلِحَٰتِ لَيَسْتَخْلِفَنَّهُمْ فِى ٱلْأَرْضِ كَمَا ٱسْتَخْلَفَ ٱلَّذِينَ مِن قَبْلِهِمْ وَلَيُمَكِّنَنَّ لَهُمْ دِينَهُمُ ٱلَّذِى ٱرْتَضَىٰ لَهُمْ وَلَيُبَدِّلَنَّهُم مِّنۢ بَعْدِ خَوْفِهِمْ أَمْنًۭا ۚ يَعْبُدُونَنِى لَا يُشْرِكُونَ بِى شَيْـًۭٔا ۚ وَمَن كَفَرَ بَعْدَ ذَٰلِكَ فَأُو۟لَٰٓئِكَ هُمُ ٱلْفَٰسِقُونَ ﴿٥٥﴾ الله له تاسو څخه له هغو کسانو سره وعده کړې چې ایمان یې راوړی او نېک عملونه یې کړي، چې خامخا به یې په ځمکه کې خلافت ورکړي، لکه څنګه یې چې هغو کسانو ته خلافت ورکړ چې له دوی مخکې وو؛ او د دوی لپاره به هغه دین خامخا غالب کړي چې د دوی لپاره یې خوښ کړی، او د دوی د وېرې (حالت) به خامخا په امن بدل کړي؛ (نو) زما به عبادت کوي او له ما سره به هېڅوک شریک نه نیسي؛ او څوک چې له دې وروسته (هم) ناشکري وکړي، نو همدا خلک فاسقان دي.
p وَأَقِيمُوا۟ ٱلصَّلَوٰةَ وَءَاتُوا۟ ٱلزَّكَوٰةَ وَأَطِيعُوا۟ ٱلرَّسُولَ لَعَلَّكُمْ تُرْحَمُونَ ﴿٥٦﴾ اور نماز قائم کرو اور زکوٰۃ ادا کرو اور رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کی اطاعت کرو تاکہ تم پر رحم کیا جائے۔ || وَأَقِيمُوا۟ ٱلصَّلَوٰةَ وَءَاتُوا۟ ٱلزَّكَوٰةَ وَأَطِيعُوا۟ ٱلرَّسُولَ لَعَلَّكُمْ تُرْحَمُونَ ﴿٥٦﴾ Establish worship and pay the poor-due and obey the messenger, that haply ye may find mercy. (24:56) || وَأَقِيمُوا۟ ٱلصَّلَوٰةَ وَءَاتُوا۟ ٱلزَّكَوٰةَ وَأَطِيعُوا۟ ٱلرَّسُولَ لَعَلَّكُمْ تُرْحَمُونَ ﴿٥٦﴾ او لمونځ قائم کړئ، زکات ورکړئ او د رسول (صلی الله علیه وعلی آله واصحابه وسلم) اطاعت وکړئ تر څو پر تاسو رحم وشي.
p لَا تَحْسَبَنَّ ٱلَّذِينَ كَفَرُوا۟ مُعْجِزِينَ فِى ٱلْأَرْضِ ۚ وَمَأْوَىٰهُمُ ٱلنَّارُ ۖ وَلَبِئْسَ ٱلْمَصِيرُ ﴿٥٧﴾ ہرگز یہ گمان نہ کرنا کہ جن لوگوں نے کفر کیا وہ (اللہ کو) زمین میں عاجز کرنے والے ہیں اور اُن کا ٹھکانا (جہنم کی) آگ ہے اور یقیناً وہ بہت بُرا ٹھکانا ہے۔ || لَا تَحْسَبَنَّ ٱلَّذِينَ كَفَرُوا۟ مُعْجِزِينَ فِى ٱلْأَرْضِ ۚ وَمَأْوَىٰهُمُ ٱلنَّارُ ۖ وَلَبِئْسَ ٱلْمَصِيرُ ﴿٥٧﴾ Think not that the disbelievers can escape in the land. Fire will be their home - a hapless journey's end! (24:57) || لَا تَحْسَبَنَّ ٱلَّذِينَ كَفَرُوا۟ مُعْجِزِينَ فِى ٱلْأَرْضِ ۚ وَمَأْوَىٰهُمُ ٱلنَّارُ ۖ وَلَبِئْسَ ٱلْمَصِيرُ ﴿٥٧﴾ هېڅکله دا ګومان مه کوه چې کومو کسانو کفر کړی، هغوی (الله) په ځمکه کې عاجزوونکي دي؛ او د هغوی ځای د (دوزخ) اور دی، او یقیناً هغه ډېر بد ځای دی.
