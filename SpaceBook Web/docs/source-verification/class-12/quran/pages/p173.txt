p وَمَا لَكُمْ لَا تُؤْمِنُونَ بِٱللَّهِ ۙ وَٱلرَّسُولُ يَدْعُوكُمْ لِتُؤْمِنُوا۟ بِرَبِّكُمْ وَقَدْ أَخَذَ مِيثَٰقَكُمْ إِن كُنتُم مُّؤْمِنِينَ ﴿٨﴾ اور تمھیں کیا ہو گیا ہے! کہ تم اللہ پر ایمان نہیں لاتے جب کہ رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) تمھیں بلا رہے ہیں کہ اپنے رب پر ایمان لاؤ اور یقیناً وہ (اللہ) تم سے پختہ عہد لے چکا ہے اگر تم ماننے والے ہو۔ || وَمَا لَكُمْ لَا تُؤْمِنُونَ بِٱللَّهِ ۙ وَٱلرَّسُولُ يَدْعُوكُمْ لِتُؤْمِنُوا۟ بِرَبِّكُمْ وَقَدْ أَخَذَ مِيثَٰقَكُمْ إِن كُنتُم مُّؤْمِنِينَ ﴿٨﴾ What aileth you that ye believe not in Allah, when the messenger calleth you to believe in your Lord, and He hath already made a covenant with you, if ye are believers? (57:8) || وَمَا لَكُمْ لَا تُؤْمِنُونَ بِٱللَّهِ ۙ وَٱلرَّسُولُ يَدْعُوكُمْ لِتُؤْمِنُوا۟ بِرَبِّكُمْ وَقَدْ أَخَذَ مِيثَٰقَكُمْ إِن كُنتُم مُّؤْمِنِينَ ﴿٨﴾ او تاسو ته څه شوي! چې پر الله ایمان نه راوړئ حال دا چې رسول (صلی الله علیه وعلی آله واصحابه وسلم) تاسو رابولي چې پر خپل رب ایمان راوړئ، او یقیناً هغه (الله) له تاسو ټینګه ژمنه اخیستې، که منونکي یاست.
p هُوَ ٱلَّذِى يُنَزِّلُ عَلَىٰ عَبْدِهِۦٓ ءَايَٰتٍۭ بَيِّنَٰتٍۢ لِّيُخْرِجَكُم مِّنَ ٱلظُّلُمَٰتِ إِلَى ٱلنُّورِ ۚ وَإِنَّ ٱللَّهَ بِكُمْ لَرَءُوفٌۭ رَّحِيمٌۭ ﴿٩﴾ وہی ہے جو اپنے بندہ پر واضح آیتیں نازل فرماتا ہے تاکہ وہ تمھیں اندھیروں سے نکال کر روشنی کی طرف لے آئے اور بے شک اللہ تم پر یقیناً بہت شفقت فرمانے والا نہایت رحم فرمانے والا ہے۔ || هُوَ ٱلَّذِى يُنَزِّلُ عَلَىٰ عَبْدِهِۦٓ ءَايَٰتٍۭ بَيِّنَٰتٍۢ لِّيُخْرِجَكُم مِّنَ ٱلظُّلُمَٰتِ إِلَى ٱلنُّورِ ۚ وَإِنَّ ٱللَّهَ بِكُمْ لَرَءُوفٌۭ رَّحِيمٌۭ ﴿٩﴾ He it is Who sendeth down clear revelations unto His slave, that He may bring you forth from darkness unto light; and lo! for you, Allah is Full of Pity, Merciful. (57:9) || هُوَ ٱلَّذِى يُنَزِّلُ عَلَىٰ عَبْدِهِۦٓ ءَايَٰتٍۭ بَيِّنَٰتٍۢ لِّيُخْرِجَكُم مِّنَ ٱلظُّلُمَٰتِ إِلَى ٱلنُّورِ ۚ وَإِنَّ ٱللَّهَ بِكُمْ لَرَءُوفٌۭ رَّحِيمٌۭ ﴿٩﴾ همغه دی چې پر خپل بنده روښانه آیتونه نازلوي تر څو تاسو له تیارو راوباسي او رڼا ته مو راولي؛ او بې‌شکه الله پر تاسو یقیناً ډېر شفقت کوونکی، بې‌حده مهربان دی.
p وَمَا لَكُمْ أَلَّا تُنفِقُوا۟ فِى سَبِيلِ ٱللَّهِ وَلِلَّهِ مِيرَٰثُ ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ ۚ لَا يَسْتَوِى مِنكُم مَّنْ أَنفَقَ مِن قَبْلِ ٱلْفَتْحِ وَقَٰتَلَ ۚ أُو۟لَٰٓئِكَ أَعْظَمُ دَرَجَةًۭ مِّنَ ٱلَّذِينَ أَنفَقُوا۟ مِنۢ بَعْدُ وَقَٰتَلُوا۟ ۚ وَكُلًّۭا وَعَدَ ٱللَّهُ ٱلْحُسْنَىٰ ۚ وَٱللَّهُ بِمَا تَعْمَلُونَ خَبِيرٌۭ ﴿١٠﴾ اور تمھیں کیا ہو گیا ہے! کہ تم اللہ کی راہ میں خرچ نہیں کرتے جب کہ آسمانوں اور زمینوں کا وارث اللہ ہی ہے تم میں سے وہ جنھوں نے فتح (مکہ) سے پہلے خرچ کیا اور قتال کیا وہ (دوسروں کے) برابر نہیں ہو سکتے وہ ان لوگوں سے درجہ میں بہت بڑھے ہوئے ہیں جنھوں نے (فتح مکہ کے) بعد خرچ کیا اور قتال کیا اور اللہ نے سب سے بھلائی کا وعدہ فرما رکھا ہے اور جو کچھ تم کر رہے ہو اللہ اس سے خوب واقف ہے۔ || وَمَا لَكُمْ أَلَّا تُنفِقُوا۟ فِى سَبِيلِ ٱللَّهِ وَلِلَّهِ مِيرَٰثُ ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ ۚ لَا يَسْتَوِى مِنكُم مَّنْ أَنفَقَ مِن قَبْلِ ٱلْفَتْحِ وَقَٰتَلَ ۚ أُو۟لَٰٓئِكَ أَعْظَمُ دَرَجَةًۭ مِّنَ ٱلَّذِينَ أَنفَقُوا۟ مِنۢ بَعْدُ وَقَٰتَلُوا۟ ۚ وَكُلًّۭا وَعَدَ ٱللَّهُ ٱلْحُسْنَىٰ ۚ وَٱللَّهُ بِمَا تَعْمَلُونَ خَبِيرٌۭ ﴿١٠﴾ And what aileth you that ye spend not in the way of Allah when unto Allah belongeth the inheritance of the heavens and the earth? Those who spent and fought before the victory are not upon a level (with the rest of you). Such are greater in rank than those who spent and fought afterwards. Unto each hath Allah promised good. And Allah is Informed of what ye do. (57:10) || وَمَا لَكُمْ أَلَّا تُنفِقُوا۟ فِى سَبِيلِ ٱللَّهِ وَلِلَّهِ مِيرَٰثُ ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ ۚ لَا يَسْتَوِى مِنكُم مَّنْ أَنفَقَ مِن قَبْلِ ٱلْفَتْحِ وَقَٰتَلَ ۚ أُو۟لَٰٓئِكَ أَعْظَمُ دَرَجَةًۭ مِّنَ ٱلَّذِينَ أَنفَقُوا۟ مِنۢ بَعْدُ وَقَٰتَلُوا۟ ۚ وَكُلًّۭا وَعَدَ ٱللَّهُ ٱلْحُسْنَىٰ ۚ وَٱللَّهُ بِمَا تَعْمَلُونَ خَبِيرٌۭ ﴿١٠﴾ او تاسو ته څه شوي! چې د الله په لار کې نه لګوئ حال دا چې د آسمانونو او ځمکو وارث یوازې الله دی؛ له تاسو هغه کسان چې له (مکې) بریا مخکې یې لګښت وکړ او جګړه یې وکړه، د (نورو) سره برابر نه دي؛ هغوی په درجه کې له هغو کسانو ډېر لوړ دي چې له (مکې بریا) وروسته یې لګښت وکړ او جګړه یې وکړه؛ او الله له ټولو سره د ښېګڼې وعده کړې؛ او څه چې تاسو کوئ الله ترې ښه خبر دی.
p مَّن ذَا ٱلَّذِى يُقْرِضُ ٱللَّهَ قَرْضًا حَسَنًۭا فَيُضَٰعِفَهُۥ لَهُۥ وَلَهُۥٓ أَجْرٌۭ كَرِيمٌۭ ﴿١١﴾ کون ہے جو اللہ کو قرضِ حسنہ دے؟ تو (اللہ) اس کو اُس کے لیے کئی گنا بڑھاتا رہے اور اُس کے لیے عزت والا اجر ہے۔ || مَّن ذَا ٱلَّذِى يُقْرِضُ ٱللَّهَ قَرْضًا حَسَنًۭا فَيُضَٰعِفَهُۥ لَهُۥ وَلَهُۥٓ أَجْرٌۭ كَرِيمٌۭ ﴿١١﴾ Who is he that will lend unto Allah a goodly loan, that He may double it for him and his may be a rich reward? (57:11) || مَّن ذَا ٱلَّذِى يُقْرِضُ ٱللَّهَ قَرْضًا حَسَنًۭا فَيُضَٰعِفَهُۥ لَهُۥ وَلَهُۥٓ أَجْرٌۭ كَرِيمٌۭ ﴿١١﴾ څوک دی چې الله ته ښه پور ورکړي؟ نو (الله) به یې د هغه لپاره څو چنده زیات کړي، او د هغه لپاره د عزت اجر دی.
p يَوْمَ تَرَى ٱلْمُؤْمِنِينَ وَٱلْمُؤْمِنَٰتِ يَسْعَىٰ نُورُهُم بَيْنَ أَيْدِيهِمْ وَبِأَيْمَٰنِهِم بُشْرَىٰكُمُ ٱلْيَوْمَ جَنَّٰتٌۭ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَٰرُ خَٰلِدِينَ فِيهَا ۚ ذَٰلِكَ هُوَ ٱلْفَوْزُ ٱلْعَظِيمُ ﴿١٢﴾ جس دن آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) مومن مردوں اور مومن عورتوں کو دیکھیں گے کہ اُن کا نور اُن کے آگے اور ان کے دائیں طرف دوڑ رہا ہوگا (اور ان سے کہا جائے گا کہ) آج تمھیں ایسے باغات کی خوش خبری ہے جن کے نیچے نہریں بہتی ہیں (تم) ان میں ہمیشہ رہو گے یہی بہت بڑی کامیابی ہے۔ || يَوْمَ تَرَى ٱلْمُؤْمِنِينَ وَٱلْمُؤْمِنَٰتِ يَسْعَىٰ نُورُهُم بَيْنَ أَيْدِيهِمْ وَبِأَيْمَٰنِهِم بُشْرَىٰكُمُ ٱلْيَوْمَ جَنَّٰتٌۭ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَٰرُ خَٰلِدِينَ فِيهَا ۚ ذَٰلِكَ هُوَ ٱلْفَوْزُ ٱلْعَظِيمُ ﴿١٢﴾ On the day when thou (Muhammad) wilt see the believers, men and women, their light shining forth before them and on their right hands, (and wilt hear it said unto them): Glad news for you this day: Gardens underneath which rivers flow, wherein ye are immortal. That is the supreme triumph. (57:12) || يَوْمَ تَرَى ٱلْمُؤْمِنِينَ وَٱلْمُؤْمِنَٰتِ يَسْعَىٰ نُورُهُم بَيْنَ أَيْدِيهِمْ وَبِأَيْمَٰنِهِم بُشْرَىٰكُمُ ٱلْيَوْمَ جَنَّٰتٌۭ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَٰرُ خَٰلِدِينَ فِيهَا ۚ ذَٰلِكَ هُوَ ٱلْفَوْزُ ٱلْعَظِيمُ ﴿١٢﴾ په کومه ورځ چې تاسو (صلی الله علیه وعلی آله واصحابه وسلم) مؤمن نارینه او مؤمنې ښځې ووینئ چې د هغوی نور به د هغوی مخې ته او ښي لوري ته منډې وهي، (او ورته به وویل شي چې) نن تاسو ته د داسې باغونو زېری دی چې لاندې یې ویالې بهېږي، (تاسو به) پکې تل پاتې یاست؛ همدا ډېره لویه بریا ده.
p يَوْمَ يَقُولُ ٱلْمُنَٰفِقُونَ وَٱلْمُنَٰفِقَٰتُ لِلَّذِينَ ءَامَنُوا۟ ٱنظُرُونَا نَقْتَبِسْ مِن نُّورِكُمْ قِيلَ ٱرْجِعُوا۟ وَرَآءَكُمْ فَٱلْتَمِسُوا۟ نُورًۭا فَضُرِبَ بَيْنَهُم بِسُورٍۢ لَّهُۥ بَابٌۢ بَاطِنُهُۥ فِيهِ ٱلرَّحْمَةُ وَظَٰهِرُهُۥ مِن قِبَلِهِ ٱلْعَذَابُ ﴿١٣﴾ جس دن منافق مرد اور منافق عورتیں (پلِ صراط پر) ایمان والوں سے کہیں گے کہ (ذرا) ہمارا انتظار کرو تاکہ ہم تمھارے نور سے کچھ روشنی حاصل کریں (ان سے) کہا جائے گا اپنے پیچھے لوٹ جاؤ پھر (وہاں سے) نور تلاش کرو تو اُن کے درمیان ایک دیوار کھڑی کر دی جائے گی جس میں ایک دروازہ ہوگا اُس کے اندرونی حصہ میں رحمت ہوگی اور اُس کے بیرونی جانب عذاب ہوگا۔ || يَوْمَ يَقُولُ ٱلْمُنَٰفِقُونَ وَٱلْمُنَٰفِقَٰتُ لِلَّذِينَ ءَامَنُوا۟ ٱنظُرُونَا نَقْتَبِسْ مِن نُّورِكُمْ قِيلَ ٱرْجِعُوا۟ وَرَآءَكُمْ فَٱلْتَمِسُوا۟ نُورًۭا فَضُرِبَ بَيْنَهُم بِسُورٍۢ لَّهُۥ بَابٌۢ بَاطِنُهُۥ فِيهِ ٱلرَّحْمَةُ وَظَٰهِرُهُۥ مِن قِبَلِهِ ٱلْعَذَابُ ﴿١٣﴾ On the day when the hypocritical men and the hypocritical women will say unto those who believe: Look on us that we may borrow from your light! it will be said: Go back and seek for light! Then there will separate them a wall wherein is a gate, the inner side whereof containeth mercy, while the outer side thereof is toward the doom. (57:13) || يَوْمَ يَقُولُ ٱلْمُنَٰفِقُونَ وَٱلْمُنَٰفِقَٰتُ لِلَّذِينَ ءَامَنُوا۟ ٱنظُرُونَا نَقْتَبِسْ مِن نُّورِكُمْ قِيلَ ٱرْجِعُوا۟ وَرَآءَكُمْ فَٱلْتَمِسُوا۟ نُورًۭا فَضُرِبَ بَيْنَهُم بِسُورٍۢ لَّهُۥ بَابٌۢ بَاطِنُهُۥ فِيهِ ٱلرَّحْمَةُ وَظَٰهِرُهُۥ مِن قِبَلِهِ ٱلْعَذَابُ ﴿١٣﴾ په کومه ورځ چې منافق نارینه او منافقې ښځې به (پر پل صراط) مؤمنانو ته ووايي: (لږ) زموږ انتظار وکړئ تر څو ستاسو له نوره څه رڼا واخلو؛ (هغوی ته) به وویل شي: خپل شا ته بېرته لاړ شئ، بیا (له هغه ځایه) نور ولټوئ؛ نو د هغوی ترمنځ به یو دېوال ودرول شي چې یوه دروازه به پکې وي؛ د هغه دننه برخه کې به رحمت وي او د هغه بهر لوري ته به عذاب وي.
