p وَلَئِنْ أَذَقْنَٰهُ رَحْمَةًۭ مِّنَّا مِنۢ بَعْدِ ضَرَّآءَ مَسَّتْهُ لَيَقُولَنَّ هَٰذَا لِى وَمَآ أَظُنُّ ٱلسَّاعَةَ قَآئِمَةًۭ وَلَئِن رُّجِعْتُ إِلَىٰ رَبِّىٓ إِنَّ لِى عِندَهُۥ لَلْحُسْنَىٰ ۚ فَلَنُنَبِّئَنَّ ٱلَّذِينَ كَفَرُوا۟ بِمَا عَمِلُوا۟ وَلَنُذِيقَنَّهُم مِّنْ عَذَابٍ غَلِيظٍۢ ﴿٥٠﴾ اور اگر ہم اسے اپنی جانب سے رحمت سے لطف اندوز کریں اُس تکلیف کے بعد جو اُسے پہنچی تھی تو وہ ضرور کہنے لگتا ہے کہ یہ تو میرا حق تھا اور میں یہ خیال نہیں کرتا کہ قیامت قائم ہوگی اور اگر میں اپنے رب کے پاس لوٹایا بھی گیا (تو) یقیناً میرے لیے اس کے ہاں بھلائی ہی ہوگی تو ہم کافروں کو ضرور بتائیں گے جو وہ کیا کرتے تھے اور انھیں سخت عذاب کا مزہ ضرور چکھائیں گے۔ || وَلَئِنْ أَذَقْنَٰهُ رَحْمَةًۭ مِّنَّا مِنۢ بَعْدِ ضَرَّآءَ مَسَّتْهُ لَيَقُولَنَّ هَٰذَا لِى وَمَآ أَظُنُّ ٱلسَّاعَةَ قَآئِمَةًۭ وَلَئِن رُّجِعْتُ إِلَىٰ رَبِّىٓ إِنَّ لِى عِندَهُۥ لَلْحُسْنَىٰ ۚ فَلَنُنَبِّئَنَّ ٱلَّذِينَ كَفَرُوا۟ بِمَا عَمِلُوا۟ وَلَنُذِيقَنَّهُم مِّنْ عَذَابٍ غَلِيظٍۢ ﴿٥٠﴾ And verily, if We cause him to taste mercy after some hurt that hath touched him, he will say: This is my own; and I deem not that the Hour will ever rise, and if I am brought back to my Lord, I surely shall be better off with Him - But We verily shall tell those who disbelieve (all) that they did, and We verily shall make them taste hard punishment. (41:50) || وَلَئِنْ أَذَقْنَٰهُ رَحْمَةًۭ مِّنَّا مِنۢ بَعْدِ ضَرَّآءَ مَسَّتْهُ لَيَقُولَنَّ هَٰذَا لِى وَمَآ أَظُنُّ ٱلسَّاعَةَ قَآئِمَةًۭ وَلَئِن رُّجِعْتُ إِلَىٰ رَبِّىٓ إِنَّ لِى عِندَهُۥ لَلْحُسْنَىٰ ۚ فَلَنُنَبِّئَنَّ ٱلَّذِينَ كَفَرُوا۟ بِمَا عَمِلُوا۟ وَلَنُذِيقَنَّهُم مِّنْ عَذَابٍ غَلِيظٍۢ ﴿٥٠﴾ او که موږ هغه ته د هغه تکلیف وروسته چې ورته رسېدلی و، له خپل لوري رحمت وڅکوو، نو خامخا وايي: دا خو زما حق و، او زه دا ګومان نه کوم چې قیامت به قائم شي، او که زه خپل رب ته بېرته ورګرځول شم هم، (نو) یقیناً د هغه سره به زما لپاره یوازې ښېګڼه وي؛ نو موږ به کافرانو ته خامخا ووایو چې څه به یې کول، او خامخا به یې د سخت عذاب خوند وڅکوو.
p وَإِذَآ أَنْعَمْنَا عَلَى ٱلْإِنسَٰنِ أَعْرَضَ وَنَـَٔا بِجَانِبِهِۦ وَإِذَا مَسَّهُ ٱلشَّرُّ فَذُو دُعَآءٍ عَرِيضٍۢ ﴿٥١﴾ اور جب ہم انسان پر انعام فرماتے ہیں (تو) وہ منہ موڑ لیتا ہے اور پہلو پھیر لیتا ہے اور جب اسے کوئی تکلیف پہنچتی ہے تو وہ لمبی چوڑی دعائیں کرنے لگتا ہے۔ || وَإِذَآ أَنْعَمْنَا عَلَى ٱلْإِنسَٰنِ أَعْرَضَ وَنَـَٔا بِجَانِبِهِۦ وَإِذَا مَسَّهُ ٱلشَّرُّ فَذُو دُعَآءٍ عَرِيضٍۢ ﴿٥١﴾ When We show favour unto man, he withdraweth and turneth aside, but when ill toucheth him then he aboundeth in prayer. (41:51) || وَإِذَآ أَنْعَمْنَا عَلَى ٱلْإِنسَٰنِ أَعْرَضَ وَنَـَٔا بِجَانِبِهِۦ وَإِذَا مَسَّهُ ٱلشَّرُّ فَذُو دُعَآءٍ عَرِيضٍۢ ﴿٥١﴾ او کله چې موږ پر انسان انعام وکړو، (نو) مخ اړوي او اړخ ګرځوي؛ او کله چې ورته کوم تکلیف ورسېږي، نو اوږدې پلنې دعاوې پیلوي.
p قُلْ أَرَءَيْتُمْ إِن كَانَ مِنْ عِندِ ٱللَّهِ ثُمَّ كَفَرْتُم بِهِۦ مَنْ أَضَلُّ مِمَّنْ هُوَ فِى شِقَاقٍۭ بَعِيدٍۢ ﴿٥٢﴾ آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) فرما دیجیے بھلا دیکھو اگر یہ (قرآن) اللہ کی طرف سے ہے پھر تم نے اس کا انکار کر دیا تو اس سے زیادہ کون گم راہ ہوگا جو بہت دور کی مخالفت میں پڑا ہو۔ || قُلْ أَرَءَيْتُمْ إِن كَانَ مِنْ عِندِ ٱللَّهِ ثُمَّ كَفَرْتُم بِهِۦ مَنْ أَضَلُّ مِمَّنْ هُوَ فِى شِقَاقٍۭ بَعِيدٍۢ ﴿٥٢﴾ Bethink you: If it is from Allah and ye reject it - Who is further astray than one who is at open feud (with Allah)? (41:52) || قُلْ أَرَءَيْتُمْ إِن كَانَ مِنْ عِندِ ٱللَّهِ ثُمَّ كَفَرْتُم بِهِۦ مَنْ أَضَلُّ مِمَّنْ هُوَ فِى شِقَاقٍۭ بَعِيدٍۢ ﴿٥٢﴾ تاسو (صلی الله علیه وعلی آله واصحابه وسلم) ووایئ: ښه وګورئ، که دا (قرآن) د الله له لوري وي، بیا مو یې انکار وکړ، نو له هغه چا به څوک ډېر ګمراه وي چې په ډېره لرې مخالفت کې پروت وي؟
p سَنُرِيهِمْ ءَايَٰتِنَا فِى ٱلْءَافَاقِ وَفِىٓ أَنفُسِهِمْ حَتَّىٰ يَتَبَيَّنَ لَهُمْ أَنَّهُ ٱلْحَقُّ ۗ أَوَلَمْ يَكْفِ بِرَبِّكَ أَنَّهُۥ عَلَىٰ كُلِّ شَىْءٍۢ شَهِيدٌ ﴿٥٣﴾ عنقریب ہم انھیں اپنی نشانیاں دکھائیں گے آفاق (عالم) میں بھی اور (خود) ان کی جانوں میں بھی یہاں تک کہ ان پر واضح ہو جائے گا کہ بے شک وہ (قرآن) حق ہے کیا یہ بات کافی نہیں کہ بے شک آپ کا رب ہر شے پر گواہ ہے۔ || سَنُرِيهِمْ ءَايَٰتِنَا فِى ٱلْءَافَاقِ وَفِىٓ أَنفُسِهِمْ حَتَّىٰ يَتَبَيَّنَ لَهُمْ أَنَّهُ ٱلْحَقُّ ۗ أَوَلَمْ يَكْفِ بِرَبِّكَ أَنَّهُۥ عَلَىٰ كُلِّ شَىْءٍۢ شَهِيدٌ ﴿٥٣﴾ We shall show them Our portents on the horizons and within themselves until it will be manifest unto them that it is the Truth. (41:53) || سَنُرِيهِمْ ءَايَٰتِنَا فِى ٱلْءَافَاقِ وَفِىٓ أَنفُسِهِمْ حَتَّىٰ يَتَبَيَّنَ لَهُمْ أَنَّهُ ٱلْحَقُّ ۗ أَوَلَمْ يَكْفِ بِرَبِّكَ أَنَّهُۥ عَلَىٰ كُلِّ شَىْءٍۢ شَهِيدٌ ﴿٥٣﴾ ژر به موږ هغوی ته خپلې نښې وښیو په آفاقو (نړۍ) کې هم او (پخپله) د هغوی په ځانونو کې هم، تر دې چې پر هغوی به روښانه شي چې بې‌شکه هغه (قرآن) حق دی؛ ایا دا خبره بس نه ده چې بې‌شکه ستاسو رب پر هر شي شاهد دی؟
p أَلَآ إِنَّهُمْ فِى مِرْيَةٍۢ مِّن لِّقَآءِ رَبِّهِمْ ۗ أَلَآ إِنَّهُۥ بِكُلِّ شَىْءٍۢ مُّحِيطٌۢ ﴿٥٤﴾ آگاہ ہو جاؤ! یقیناً وہ لوگ اپنے رب کی ملاقات کے بارے میں شک میں ہیں آگاہ ہو جاؤ! بے شک وہ ہر چیز کا احاطہ فرمائے ہوئے ہے۔ || أَلَآ إِنَّهُمْ فِى مِرْيَةٍۢ مِّن لِّقَآءِ رَبِّهِمْ ۗ أَلَآ إِنَّهُۥ بِكُلِّ شَىْءٍۢ مُّحِيطٌۢ ﴿٥٤﴾ How! ## علم و عمل کی باتیں || Matters of knowledge and action || د علم او عمل خبرې
- انسان جرم کرتے ہوئے بھول جاتا ہے کہ اللہ تعالیٰ کے علاوہ اس کے اپنے اعضاء مثلاً کان، آنکھ اور کھال اس کے جرم کے چشم دید گواہ ہیں جو اس کے خلاف قیامت کے دن گواہی دیں گے۔ || While committing a crime man forgets that, besides Allah Almighty, his own limbs, for example ears, eyes and skin, are eyewitnesses of his crime, and they will testify against him on the Day of Resurrection. || انسان د جرم پر مهال هېروي چې له الله تعالی پرته د هغه خپل غړي لکه غوږ، سترګه او پوستکی د هغه د جرم سترګو لیدلي شاهدان دي چې د قیامت په ورځ به یې پر خلاف شاهدي ورکړي.
- ایمان پر استقامت اختیار کرنے والوں کو فرشتے یہ خوش خبری بھی دیتے ہیں کہ ہم دنیا و آخرت میں تمھارے مددگار و رفیق ہیں۔ سب سے اچھی بات اس شخص کی ہے جو لوگوں کو اللہ تعالیٰ کی طرف بلائے اور خود بھی نیک ہو اور اپنے آپ کو عام مسلمانوں میں شمار کرے۔ || The angels also give those who adopt steadfastness in faith this glad tidings: that we are your helpers and companions in this world and the Hereafter. The best speech is of that person who calls people towards Allah Almighty and is himself righteous too, and counts himself among the ordinary Muslims. || پر ایمان د استقامت غوره کوونکو ته پرښتې دا زېری هم ورکوي چې موږ په دنیا او آخرت کې ستاسو مرستندویان او ملګري یو. تر ټولو ښه خبره د هغه چا ده چې خلک الله تعالی ته راوبولي، پخپله هم نېک وي او ځان په عامو مسلمانانو کې وشمېري.
- قرآنِ حکیم اللہ تعالیٰ کی طرف سے برحق کلام ہے۔ انسان کائنات میں اور خود اپنی جان میں موجود اللہ تعالیٰ کی نشانیوں پر جتنا غور کرتا جائے گا اسی قدر قرآنِ حکیم کی عظمت اور حقانیت واضح ہوتی جائے گی۔ || The Qur'an-e-Hakim is true speech from Allah Almighty. The more a person reflects on the signs of Allah Almighty present in the universe and in his own self, the more the greatness and truth of the Qur'an-e-Hakim will become clear. || قرآن حکیم د الله تعالی له لوري برحقه کلام دی. انسان په کائنات او پخپل ځان کې د الله تعالی په موجودو نښو څومره چې فکر کوي، هومره به د قرآن حکیم عظمت او حقانیت روښانه کېږي.
