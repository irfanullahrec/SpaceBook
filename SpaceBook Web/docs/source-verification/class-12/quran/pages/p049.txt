p وَإِن مِّنْ أَهْلِ ٱلْكِتَٰبِ إِلَّا لَيُؤْمِنَنَّ بِهِۦ قَبْلَ مَوْتِهِۦ ۖ وَيَوْمَ ٱلْقِيَٰمَةِ يَكُونُ عَلَيْهِمْ شَهِيدًۭا ﴿١٥٩﴾ اور اہلِ کتاب میں سے کوئی ایسا نہیں جو ان کی موت سے پہلے ان پر ایمان نہ لائے اور قیامت کے دن وہ (مسیح علیہ السلام) ان پر گواہ ہوں گے۔ || وَإِن مِّنْ أَهْلِ ٱلْكِتَٰبِ إِلَّا لَيُؤْمِنَنَّ بِهِۦ قَبْلَ مَوْتِهِۦ ۖ وَيَوْمَ ٱلْقِيَٰمَةِ يَكُونُ عَلَيْهِمْ شَهِيدًۭا ﴿١٥٩﴾ There is not one of the People of the Scripture but will believe in him before his death, and on the Day of Resurrection he will be a witness against them - (4:159) || وَإِن مِّنْ أَهْلِ ٱلْكِتَٰبِ إِلَّا لَيُؤْمِنَنَّ بِهِۦ قَبْلَ مَوْتِهِۦ ۖ وَيَوْمَ ٱلْقِيَٰمَةِ يَكُونُ عَلَيْهِمْ شَهِيدًۭا ﴿١٥٩﴾ او په اهل کتابو کې هېڅوک داسې نشته چې د هغه له مړینې مخکې پرې ایمان رانه وړي، او د قیامت په ورځ به هغه (مسیح علیه السلام) پر هغوی شاهد وي.
p فَبِظُلْمٍۢ مِّنَ ٱلَّذِينَ هَادُوا۟ حَرَّمْنَا عَلَيْهِمْ طَيِّبَٰتٍ أُحِلَّتْ لَهُمْ وَبِصَدِّهِمْ عَن سَبِيلِ ٱللَّهِ كَثِيرًۭا ﴿١٦٠﴾ تو پھر یہودیوں کے ظلم کی وجہ سے ہم نے اُن پر وہ پاکیزہ چیزیں حرام کر دی تھیں جو اُن کے لیے حلال کی گئی تھیں اور ان کے کثرت سے اللہ کی راہ سے روکنے کی وجہ سے (بھی)۔ || فَبِظُلْمٍۢ مِّنَ ٱلَّذِينَ هَادُوا۟ حَرَّمْنَا عَلَيْهِمْ طَيِّبَٰتٍ أُحِلَّتْ لَهُمْ وَبِصَدِّهِمْ عَن سَبِيلِ ٱللَّهِ كَثِيرًۭا ﴿١٦٠﴾ Because of the wrongdoing of the Jews We forbade them good things which were (before) made lawful unto them, and because of their much hindering from Allah's way, (4:160) || فَبِظُلْمٍۢ مِّنَ ٱلَّذِينَ هَادُوا۟ حَرَّمْنَا عَلَيْهِمْ طَيِّبَٰتٍ أُحِلَّتْ لَهُمْ وَبِصَدِّهِمْ عَن سَبِيلِ ٱللَّهِ كَثِيرًۭا ﴿١٦٠﴾ نو بیا د یهودو د ظلم له امله موږ پر هغوی هغه پاک شیان حرام کړل چې ورته حلال شوي وو، او د الله له لارې د هغوی د ډېر منع کولو له امله (هم).
p وَأَخْذِهِمُ ٱلرِّبَوٰا۟ وَقَدْ نُهُوا۟ عَنْهُ وَأَكْلِهِمْ أَمْوَٰلَ ٱلنَّاسِ بِٱلْبَٰطِلِ ۚ وَأَعْتَدْنَا لِلْكَٰفِرِينَ مِنْهُمْ عَذَابًا أَلِيمًۭا ﴿١٦١﴾ اور اُن کے سود لینے کی وجہ سے (بھی) حالاں کہ انھیں اس سے منع کیا گیا تھا اور ان کے لوگوں کے مال ناحق کھانے کی وجہ سے ہے اور ہم نے اُن میں سے کافروں کے لیے دردناک عذاب تیار کر رکھا ہے۔ || وَأَخْذِهِمُ ٱلرِّبَوٰا۟ وَقَدْ نُهُوا۟ عَنْهُ وَأَكْلِهِمْ أَمْوَٰلَ ٱلنَّاسِ بِٱلْبَٰطِلِ ۚ وَأَعْتَدْنَا لِلْكَٰفِرِينَ مِنْهُمْ عَذَابًا أَلِيمًۭا ﴿١٦١﴾ And of their taking usury when they were forbidden it, and of their devouring people's wealth by false pretences, We have prepared for those of them who disbelieve a painful doom. (4:161) || وَأَخْذِهِمُ ٱلرِّبَوٰا۟ وَقَدْ نُهُوا۟ عَنْهُ وَأَكْلِهِمْ أَمْوَٰلَ ٱلنَّاسِ بِٱلْبَٰطِلِ ۚ وَأَعْتَدْنَا لِلْكَٰفِرِينَ مِنْهُمْ عَذَابًا أَلِيمًۭا ﴿١٦١﴾ او د هغوی د سود اخیستو له امله (هم)، حال دا چې ترې منع شوي وو، او د خلکو د مالونو د ناحقه خوړلو له امله؛ او موږ له هغوی څخه کافرانو ته دردناک عذاب چمتو کړی.
p لَّٰكِنِ ٱلرَّٰسِخُونَ فِى ٱلْعِلْمِ مِنْهُمْ وَٱلْمُؤْمِنُونَ يُؤْمِنُونَ بِمَآ أُنزِلَ إِلَيْكَ وَمَآ أُنزِلَ مِن قَبْلِكَ ۚ وَٱلْمُقِيمِينَ ٱلصَّلَوٰةَ ۚ وَٱلْمُؤْتُونَ ٱلزَّكَوٰةَ وَٱلْمُؤْمِنُونَ بِٱللَّهِ وَٱلْيَوْمِ ٱلْءَاخِرِ أُو۟لَٰٓئِكَ سَنُؤْتِيهِمْ أَجْرًا عَظِيمًا ﴿١٦٢﴾ لیکن اُن میں سے جو علم میں پختہ ہیں اور وہ مومن جو اس پر ایمان رکھتے ہیں جو آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کی طرف نازل فرمایا گیا اور اس پر (بھی) جو آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) سے پہلے نازل فرمایا گیا اور نماز قائم کرنے والے ہیں اور زکوٰۃ دینے والے ہیں اور جو اللہ اور آخرت کے دن پر ایمان لانے والے ہیں یہ وہ لوگ ہیں جن کو ہم عنقریب اجرِ عظیم دیں گے۔ || لَّٰكِنِ ٱلرَّٰسِخُونَ فِى ٱلْعِلْمِ مِنْهُمْ وَٱلْمُؤْمِنُونَ يُؤْمِنُونَ بِمَآ أُنزِلَ إِلَيْكَ وَمَآ أُنزِلَ مِن قَبْلِكَ ۚ وَٱلْمُقِيمِينَ ٱلصَّلَوٰةَ ۚ وَٱلْمُؤْتُونَ ٱلزَّكَوٰةَ وَٱلْمُؤْمِنُونَ بِٱللَّهِ وَٱلْيَوْمِ ٱلْءَاخِرِ أُو۟لَٰٓئِكَ سَنُؤْتِيهِمْ أَجْرًا عَظِيمًا ﴿١٦٢﴾ But those of them who are firm in knowledge and the believers believe in that which is revealed unto thee, and that which was revealed before thee, especially the diligent in prayer and those who pay the poor-due, the believers in Allah and the Last Day. Upon these We shall bestow immense reward. (4:162) || لَّٰكِنِ ٱلرَّٰسِخُونَ فِى ٱلْعِلْمِ مِنْهُمْ وَٱلْمُؤْمِنُونَ يُؤْمِنُونَ بِمَآ أُنزِلَ إِلَيْكَ وَمَآ أُنزِلَ مِن قَبْلِكَ ۚ وَٱلْمُقِيمِينَ ٱلصَّلَوٰةَ ۚ وَٱلْمُؤْتُونَ ٱلزَّكَوٰةَ وَٱلْمُؤْمِنُونَ بِٱللَّهِ وَٱلْيَوْمِ ٱلْءَاخِرِ أُو۟لَٰٓئِكَ سَنُؤْتِيهِمْ أَجْرًا عَظِيمًا ﴿١٦٢﴾ خو له هغوی څخه هغه کسان چې په علم کې پاخه دي، او هغه مؤمنان چې پر هغه څه ایمان لري چې تاسو (صلی الله علیه وعلی آله واصحابه وسلم) ته نازل شوي او پر هغه (هم) چې له تاسو (صلی الله علیه وعلی آله واصحابه وسلم) مخکې نازل شوي، او لمونځ قایموونکي او زکات ورکوونکي دي، او پر الله او د آخرت پر ورځ ایمان راوړونکي دي — دا هغه خلک دي چې ژر به یې لوی اجر ورکړو.
p ۞ إِنَّآ أَوْحَيْنَآ إِلَيْكَ كَمَآ أَوْحَيْنَآ إِلَىٰ نُوحٍۢ وَٱلنَّبِيِّۦنَ مِنۢ بَعْدِهِۦ ۚ وَأَوْحَيْنَآ إِلَىٰٓ إِبْرَٰهِيمَ وَإِسْمَٰعِيلَ وَإِسْحَٰقَ وَيَعْقُوبَ وَٱلْأَسْبَاطِ وَعِيسَىٰ وَأَيُّوبَ وَيُونُسَ وَهَٰرُونَ وَسُلَيْمَٰنَ ۚ وَءَاتَيْنَا دَاوُۥدَ زَبُورًۭا ﴿١٦٣﴾ بے شک ہم نے آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کی طرف وحی فرمائی جیسے ہم نے نوح (علیہ السلام) اور اُن کے بعد انبیاء کی طرف وحی فرمائی اور ہم نے ابراہیم اور اسماعیل اور اسحاق اور یعقوب (علیہم السلام) اور اُن کی اولاد کی طرف وحی فرمائی اور عیسیٰ اور ایوب اور یونس اور ہارون اور سلیمان (علیہم السلام) کی طرف (بھی وحی بھیجی) اور ہم نے داؤد (علیہ السلام) کو زبور عطا فرمائی۔ || ۞ إِنَّآ أَوْحَيْنَآ إِلَيْكَ كَمَآ أَوْحَيْنَآ إِلَىٰ نُوحٍۢ وَٱلنَّبِيِّۦنَ مِنۢ بَعْدِهِۦ ۚ وَأَوْحَيْنَآ إِلَىٰٓ إِبْرَٰهِيمَ وَإِسْمَٰعِيلَ وَإِسْحَٰقَ وَيَعْقُوبَ وَٱلْأَسْبَاطِ وَعِيسَىٰ وَأَيُّوبَ وَيُونُسَ وَهَٰرُونَ وَسُلَيْمَٰنَ ۚ وَءَاتَيْنَا دَاوُۥدَ زَبُورًۭا ﴿١٦٣﴾ Lo! We inspire thee as We inspired Noah and the prophets after him, as We inspired Abraham and Ishmael and Isaac and Jacob and the tribes, and Jesus and Job and Jonah and Aaron and Solomon, and as We imparted unto David the Psalms; (4:163) || ۞ إِنَّآ أَوْحَيْنَآ إِلَيْكَ كَمَآ أَوْحَيْنَآ إِلَىٰ نُوحٍۢ وَٱلنَّبِيِّۦنَ مِنۢ بَعْدِهِۦ ۚ وَأَوْحَيْنَآ إِلَىٰٓ إِبْرَٰهِيمَ وَإِسْمَٰعِيلَ وَإِسْحَٰقَ وَيَعْقُوبَ وَٱلْأَسْبَاطِ وَعِيسَىٰ وَأَيُّوبَ وَيُونُسَ وَهَٰرُونَ وَسُلَيْمَٰنَ ۚ وَءَاتَيْنَا دَاوُۥدَ زَبُورًۭا ﴿١٦٣﴾ بې‌شکه موږ تاسو (صلی الله علیه وعلی آله واصحابه وسلم) ته وحي وکړه لکه څنګه چې موږ نوح (علیه السلام) او له هغه وروسته انبیاوو ته وحي کړې وه، او موږ ابراهیم، اسماعیل، اسحاق او یعقوب (علیهم السلام) او د هغوی اولاد ته وحي وکړه، او عیسی، ایوب، یونس، هارون او سلیمان (علیهم السلام) ته (هم وحي ولېږله)، او موږ داود (علیه السلام) ته زبور ورکړ.
p وَرُسُلًۭا قَدْ قَصَصْنَٰهُمْ عَلَيْكَ مِن قَبْلُ وَرُسُلًۭا لَّمْ نَقْصُصْهُمْ عَلَيْكَ ۚ وَكَلَّمَ ٱللَّهُ مُوسَىٰ تَكْلِيمًۭا ﴿١٦٤﴾ اور ایسے بھی رسول ہیں یقیناً جن کے حالات (اس سے) پہلے ہم آپ سے بیان کر چکے ہیں اور وہ رسول بھی ہیں جن کے حالات ہم نے آپ سے بیان نہیں کیے اور اللہ نے موسیٰ (علیہ السلام) سے (خاص طور پر) خوب کلام فرمایا۔ || وَرُسُلًۭا قَدْ قَصَصْنَٰهُمْ عَلَيْكَ مِن قَبْلُ وَرُسُلًۭا لَّمْ نَقْصُصْهُمْ عَلَيْكَ ۚ وَكَلَّمَ ٱللَّهُ مُوسَىٰ تَكْلِيمًۭا ﴿١٦٤﴾ And messengers We have mentioned unto thee before and messengers We have not mentioned unto thee; and Allah spake directly unto Moses; (4:164) || وَرُسُلًۭا قَدْ قَصَصْنَٰهُمْ عَلَيْكَ مِن قَبْلُ وَرُسُلًۭا لَّمْ نَقْصُصْهُمْ عَلَيْكَ ۚ وَكَلَّمَ ٱللَّهُ مُوسَىٰ تَكْلِيمًۭا ﴿١٦٤﴾ او داسې رسولان هم دي چې یقیناً د هغوی حالات موږ (له دې) مخکې تاسو ته بیان کړي، او داسې رسولان هم دي چې د هغوی حالات موږ تاسو ته نه دي بیان کړي؛ او الله له موسی (علیه السلام) سره (په ځانګړي ډول) ښې خبرې وکړې.
