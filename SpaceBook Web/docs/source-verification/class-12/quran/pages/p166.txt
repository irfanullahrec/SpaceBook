p وَٱعْلَمُوٓا۟ أَنَّ فِيكُمْ رَسُولَ ٱللَّهِ ۚ لَوْ يُطِيعُكُمْ فِى كَثِيرٍۢ مِّنَ ٱلْأَمْرِ لَعَنِتُّمْ وَلَٰكِنَّ ٱللَّهَ حَبَّبَ إِلَيْكُمُ ٱلْإِيمَٰنَ وَزَيَّنَهُۥ فِى قُلُوبِكُمْ وَكَرَّهَ إِلَيْكُمُ ٱلْكُفْرَ وَٱلْفُسُوقَ وَٱلْعِصْيَانَ ۚ أُو۟لَٰٓئِكَ هُمُ ٱلرَّٰشِدُونَ ﴿٧﴾ اور جان لو! تمھارے درمیان اللہ کے رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) موجود ہیں اگر وہ بہت سے معاملات میں تمھارا کہنا مان لیں تو ضرور تم بڑی مشکل میں پڑ جاؤ گے لیکن اللہ نے ایمان کو تمھارے لیے محبوب بنا دیا اور اُسے تمھارے دلوں میں مزین فرما دیا اور تمھارے نزدیک کفر، فسق اور نافرمانی کو قابلِ نفرت بنا دیا یہی لوگ راہِ راست پر ہیں۔ || وَٱعْلَمُوٓا۟ أَنَّ فِيكُمْ رَسُولَ ٱللَّهِ ۚ لَوْ يُطِيعُكُمْ فِى كَثِيرٍۢ مِّنَ ٱلْأَمْرِ لَعَنِتُّمْ وَلَٰكِنَّ ٱللَّهَ حَبَّبَ إِلَيْكُمُ ٱلْإِيمَٰنَ وَزَيَّنَهُۥ فِى قُلُوبِكُمْ وَكَرَّهَ إِلَيْكُمُ ٱلْكُفْرَ وَٱلْفُسُوقَ وَٱلْعِصْيَانَ ۚ أُو۟لَٰٓئِكَ هُمُ ٱلرَّٰشِدُونَ ﴿٧﴾ And know that the messenger of Allah is among you. If he were to obey you in much of the government, ye would surely be in trouble; but Allah hath endeared the faith to you and hath beautified it in your hearts, and hath made disbelief and lewdness and rebellion hateful unto you. Such are they who are the rightly guided. (49:7) || وَٱعْلَمُوٓا۟ أَنَّ فِيكُمْ رَسُولَ ٱللَّهِ ۚ لَوْ يُطِيعُكُمْ فِى كَثِيرٍۢ مِّنَ ٱلْأَمْرِ لَعَنِتُّمْ وَلَٰكِنَّ ٱللَّهَ حَبَّبَ إِلَيْكُمُ ٱلْإِيمَٰنَ وَزَيَّنَهُۥ فِى قُلُوبِكُمْ وَكَرَّهَ إِلَيْكُمُ ٱلْكُفْرَ وَٱلْفُسُوقَ وَٱلْعِصْيَانَ ۚ أُو۟لَٰٓئِكَ هُمُ ٱلرَّٰشِدُونَ ﴿٧﴾ او پوه شئ! ستاسو په منځ کې د الله رسول (صلی الله علیه وعلی آله واصحابه وسلم) شته؛ که هغه په ډېرو چارو کې ستاسو خبره ومني، نو خامخا به په لویه ستونزه کې ولوېږئ؛ خو الله ایمان ستاسو لپاره محبوب کړ، او ستاسو په زړونو کې یې ښکلی کړ، او کفر، فسق او نافرماني یې ستاسو په نزد د کرکې وړ وګرځول؛ همدا خلک پر سمه لار دي.
p فَضْلًۭا مِّنَ ٱللَّهِ وَنِعْمَةًۭ ۚ وَٱللَّهُ عَلِيمٌ حَكِيمٌۭ ﴿٨﴾ (یہ) اللہ کی طرف سے فضل اور (اس کی) نعمت ہے اور اللہ خوب جاننے والا بڑی حکمت والا ہے۔ || فَضْلًۭا مِّنَ ٱللَّهِ وَنِعْمَةًۭ ۚ وَٱللَّهُ عَلِيمٌ حَكِيمٌۭ ﴿٨﴾ (It is) a bounty and a grace from Allah; and Allah is Knower, Wise. (49:8) || فَضْلًۭا مِّنَ ٱللَّهِ وَنِعْمَةًۭ ۚ وَٱللَّهُ عَلِيمٌ حَكِيمٌۭ ﴿٨﴾ (دا) د الله له لوري فضل او (د هغه) نعمت دی؛ او الله ښه پوه، ډېر حکمتناک دی.
p وَإِن طَآئِفَتَانِ مِنَ ٱلْمُؤْمِنِينَ ٱقْتَتَلُوا۟ فَأَصْلِحُوا۟ بَيْنَهُمَا ۖ فَإِنۢ بَغَتْ إِحْدَىٰهُمَا عَلَى ٱلْأُخْرَىٰ فَقَٰتِلُوا۟ ٱلَّتِى تَبْغِى حَتَّىٰ تَفِىٓءَ إِلَىٰٓ أَمْرِ ٱللَّهِ ۚ فَإِن فَآءَتْ فَأَصْلِحُوا۟ بَيْنَهُمَا بِٱلْعَدْلِ وَأَقْسِطُوٓا۟ ۖ إِنَّ ٱللَّهَ يُحِبُّ ٱلْمُقْسِطِينَ ﴿٩﴾ اور اگر مومنوں میں سے دو گروہ آپس میں لڑ پڑیں تو ان دونوں کے درمیان صلح کرا دو پھر اگر دونوں میں سے ایک گروہ دوسرے پر زیادتی کرے تو اُس سے لڑو جو زیادتی کرتا ہے یہاں تک کہ وہ اللہ کے حکم کی طرف پلٹ آئے پھر اگر وہ پلٹ آئے تو تم عدل کے ساتھ ان کے درمیان صلح کرا دو اور (ہر معاملے میں) انصاف کرو بے شک اللہ انصاف کرنے والوں کو پسند فرماتا ہے۔ || وَإِن طَآئِفَتَانِ مِنَ ٱلْمُؤْمِنِينَ ٱقْتَتَلُوا۟ فَأَصْلِحُوا۟ بَيْنَهُمَا ۖ فَإِنۢ بَغَتْ إِحْدَىٰهُمَا عَلَى ٱلْأُخْرَىٰ فَقَٰتِلُوا۟ ٱلَّتِى تَبْغِى حَتَّىٰ تَفِىٓءَ إِلَىٰٓ أَمْرِ ٱللَّهِ ۚ فَإِن فَآءَتْ فَأَصْلِحُوا۟ بَيْنَهُمَا بِٱلْعَدْلِ وَأَقْسِطُوٓا۟ ۖ إِنَّ ٱللَّهَ يُحِبُّ ٱلْمُقْسِطِينَ ﴿٩﴾ And if two parties of believers fall to fighting, then make peace between them. And if one party of them doeth wrong to the other, fight ye that which doeth wrong till it return unto the ordinance of Allah; then, if it return, make peace between them justly, and act equitably. Lo! Allah loveth the equitable. (49:9) || وَإِن طَآئِفَتَانِ مِنَ ٱلْمُؤْمِنِينَ ٱقْتَتَلُوا۟ فَأَصْلِحُوا۟ بَيْنَهُمَا ۖ فَإِنۢ بَغَتْ إِحْدَىٰهُمَا عَلَى ٱلْأُخْرَىٰ فَقَٰتِلُوا۟ ٱلَّتِى تَبْغِى حَتَّىٰ تَفِىٓءَ إِلَىٰٓ أَمْرِ ٱللَّهِ ۚ فَإِن فَآءَتْ فَأَصْلِحُوا۟ بَيْنَهُمَا بِٱلْعَدْلِ وَأَقْسِطُوٓا۟ ۖ إِنَّ ٱللَّهَ يُحِبُّ ٱلْمُقْسِطِينَ ﴿٩﴾ او که له مؤمنانو دوه ډلې سره وجنګېږي، نو د دواړو ترمنځ سوله وکړئ؛ بیا که له دواړو یوه ډله پر بلې زیاتی وکړي، نو له هغې سره وجنګېږئ چې زیاتی کوي تر دې چې د الله حکم ته راوګرځي؛ بیا که راوګرځېده، نو د دواړو ترمنځ په عدل سره سوله وکړئ، او (په هره چاره کې) انصاف وکړئ؛ بې‌شکه الله انصاف کوونکي خوښوي.
p إِنَّمَا ٱلْمُؤْمِنُونَ إِخْوَةٌۭ فَأَصْلِحُوا۟ بَيْنَ أَخَوَيْكُمْ ۚ وَٱتَّقُوا۟ ٱللَّهَ لَعَلَّكُمْ تُرْحَمُونَ ﴿١٠﴾ بے شک سب اہلِ ایمان (تو آپس میں) بھائی بھائی ہیں پس اپنے دو بھائیوں کے درمیان صلح کرا دیا کرو اور اللہ (کی نافرمانی) سے ڈرو تاکہ تم پر رحم کیا جائے۔ || إِنَّمَا ٱلْمُؤْمِنُونَ إِخْوَةٌۭ فَأَصْلِحُوا۟ بَيْنَ أَخَوَيْكُمْ ۚ وَٱتَّقُوا۟ ٱللَّهَ لَعَلَّكُمْ تُرْحَمُونَ ﴿١٠﴾ The believers are naught else than brothers. Therefore make peace between your brethren and observe your duty to Allah that haply ye may obtain mercy. (49:10) || إِنَّمَا ٱلْمُؤْمِنُونَ إِخْوَةٌۭ فَأَصْلِحُوا۟ بَيْنَ أَخَوَيْكُمْ ۚ وَٱتَّقُوا۟ ٱللَّهَ لَعَلَّكُمْ تُرْحَمُونَ ﴿١٠﴾ بې‌شکه ټول مؤمنان (په خپل منځ کې) وروڼه دي؛ نو د خپلو دوو وروڼو ترمنځ سوله کوئ، او له الله (د نافرمانۍ) ووېرېږئ تر څو پر تاسو رحم وشي.
p يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ لَا يَسْخَرْ قَوْمٌۭ مِّن قَوْمٍ عَسَىٰٓ أَن يَكُونُوا۟ خَيْرًۭا مِّنْهُمْ وَلَا نِسَآءٌۭ مِّن نِّسَآءٍ عَسَىٰٓ أَن يَكُنَّ خَيْرًۭا مِّنْهُنَّ ۖ وَلَا تَلْمِزُوٓا۟ أَنفُسَكُمْ وَلَا تَنَابَزُوا۟ بِٱلْأَلْقَٰبِ ۖ بِئْسَ ٱلِٱسْمُ ٱلْفُسُوقُ بَعْدَ ٱلْإِيمَٰنِ ۚ وَمَن لَّمْ يَتُبْ فَأُو۟لَٰٓئِكَ هُمُ ٱلظَّٰلِمُونَ ﴿١١﴾ اے ایمان والو! مردوں کا کوئی گروہ دوسرے مردوں کا مذاق نہ اڑائے ممکن ہے وہ لوگ اُن (مذاق اڑانے والوں) سے بہتر ہوں اور نہ عورتیں دوسری عورتوں کا (مذاق اڑائیں) ممکن ہے کہ وہ ان (مذاق اڑانے والیوں) سے بہتر ہوں اور آپس میں ایک دوسرے کو طعنہ نہ دیا کرو اور ایک دوسرے کو برے القاب کے ساتھ نہ پکارو ایمان کے بعد فسق بہت برا نام ہے اور جو لوگ (اس روش سے) توبہ نہ کریں تو وہی ظالم ہیں۔ || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ لَا يَسْخَرْ قَوْمٌۭ مِّن قَوْمٍ عَسَىٰٓ أَن يَكُونُوا۟ خَيْرًۭا مِّنْهُمْ وَلَا نِسَآءٌۭ مِّن نِّسَآءٍ عَسَىٰٓ أَن يَكُنَّ خَيْرًۭا مِّنْهُنَّ ۖ وَلَا تَلْمِزُوٓا۟ أَنفُسَكُمْ وَلَا تَنَابَزُوا۟ بِٱلْأَلْقَٰبِ ۖ بِئْسَ ٱلِٱسْمُ ٱلْفُسُوقُ بَعْدَ ٱلْإِيمَٰنِ ۚ وَمَن لَّمْ يَتُبْ فَأُو۟لَٰٓئِكَ هُمُ ٱلظَّٰلِمُونَ ﴿١١﴾ O ye who believe! Let not a folk deride a folk who may be better than they (are), not let women (deride) women who may be better than they are; neither defame one another, nor insult one another by nicknames. Bad is the name of lewdness after faith. And whoso turneth not in repentance, such are evil-doers. (49:11) || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ لَا يَسْخَرْ قَوْمٌۭ مِّن قَوْمٍ عَسَىٰٓ أَن يَكُونُوا۟ خَيْرًۭا مِّنْهُمْ وَلَا نِسَآءٌۭ مِّن نِّسَآءٍ عَسَىٰٓ أَن يَكُنَّ خَيْرًۭا مِّنْهُنَّ ۖ وَلَا تَلْمِزُوٓا۟ أَنفُسَكُمْ وَلَا تَنَابَزُوا۟ بِٱلْأَلْقَٰبِ ۖ بِئْسَ ٱلِٱسْمُ ٱلْفُسُوقُ بَعْدَ ٱلْإِيمَٰنِ ۚ وَمَن لَّمْ يَتُبْ فَأُو۟لَٰٓئِكَ هُمُ ٱلظَّٰلِمُونَ ﴿١١﴾ ای مؤمنانو! د نارینه وو یوه ډله دې پر نورو نارینه وو ملنډې نه وهي، کېدای شي هغوی له دې (ملنډې وهونکو) غوره وي؛ او نه دې ښځې پر نورو ښځو (ملنډې وهي)، کېدای شي هغوی له دې (ملنډې وهونکو) غوره وي؛ او په خپل منځ کې یو بل ته پېغور مه ورکوئ، او یو بل په بدو نومونو مه غږوئ؛ له ایمان وروسته فسق ډېر بد نوم دی؛ او هغه کسان چې (له دې چلند) توبه ونه باسي، نو همدوی ظالمان دي.
p يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ ٱجْتَنِبُوا۟ كَثِيرًۭا مِّنَ ٱلظَّنِّ إِنَّ بَعْضَ ٱلظَّنِّ إِثْمٌۭ ۖ وَلَا تَجَسَّسُوا۟ وَلَا يَغْتَب بَّعْضُكُم بَعْضًا ۚ أَيُحِبُّ أَحَدُكُمْ أَن يَأْكُلَ لَحْمَ أَخِيهِ مَيْتًۭا فَكَرِهْتُمُوهُ ۚ وَٱتَّقُوا۟ ٱللَّهَ ۚ إِنَّ ٱللَّهَ تَوَّابٌۭ رَّحِيمٌۭ ﴿١٢﴾ اے ایمان والو! بدگمانی سے بہت زیادہ بچو بے شک بعض گمان گناہ ہیں اور نہ (کسی کے متعلق) جاسوسی کرو اور نہ ایک دوسرے کی غیبت کرو کیا تم میں سے کوئی شخص پسند کرے گا کہ وہ اپنے مُردہ بھائی کا گوشت کھائے پس اس سے تو تم (انتہائی) نفرت کرتے ہو اور اللہ (کی نافرمانی) سے ڈرو بے شک اللہ بہت توبہ قبول فرمانے والا نہایت رحم فرمانے والا ہے۔ || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ ٱجْتَنِبُوا۟ كَثِيرًۭا مِّنَ ٱلظَّنِّ إِنَّ بَعْضَ ٱلظَّنِّ إِثْمٌۭ ۖ وَلَا تَجَسَّسُوا۟ وَلَا يَغْتَب بَّعْضُكُم بَعْضًا ۚ أَيُحِبُّ أَحَدُكُمْ أَن يَأْكُلَ لَحْمَ أَخِيهِ مَيْتًۭا فَكَرِهْتُمُوهُ ۚ وَٱتَّقُوا۟ ٱللَّهَ ۚ إِنَّ ٱللَّهَ تَوَّابٌۭ رَّحِيمٌۭ ﴿١٢﴾ O ye who believe! Shun much suspicion; for lo! some suspicion is a crime. And spy not, neither backbite one another. Would one of you love to eat the flesh of his dead brother? Ye abhor that (so abhor the other)! And keep your duty (to Allah). Lo! Allah is Relenting, Merciful. (49:12) || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ ٱجْتَنِبُوا۟ كَثِيرًۭا مِّنَ ٱلظَّنِّ إِنَّ بَعْضَ ٱلظَّنِّ إِثْمٌۭ ۖ وَلَا تَجَسَّسُوا۟ وَلَا يَغْتَب بَّعْضُكُم بَعْضًا ۚ أَيُحِبُّ أَحَدُكُمْ أَن يَأْكُلَ لَحْمَ أَخِيهِ مَيْتًۭا فَكَرِهْتُمُوهُ ۚ وَٱتَّقُوا۟ ٱللَّهَ ۚ إِنَّ ٱللَّهَ تَوَّابٌۭ رَّحِيمٌۭ ﴿١٢﴾ ای مؤمنانو! له بدګومانۍ ډېر ډډه وکړئ؛ بې‌شکه ځینې ګومانونه ګناه دي؛ او نه د (چا په اړه) جاسوسي وکړئ او نه د یو بل غیبت وکړئ؛ ایا له تاسو څوک خوښوي چې د خپل مړه ورور غوښه وخوري؟ نو له دې خو تاسو (ډېره) کرکه کوئ؛ او له الله (د نافرمانۍ) ووېرېږئ؛ بې‌شکه الله ډېر توبه منونکی، بې‌حده مهربان دی.
