p يَعِظُكُمُ ٱللَّهُ أَن تَعُودُوا۟ لِمِثْلِهِۦٓ أَبَدًا إِن كُنتُم مُّؤْمِنِينَ ﴿١٧﴾ اللہ تمھیں نصیحت فرماتا ہے کہ پھر کبھی بھی ایسی بات نہ کرنا اگر تم مومن ہو۔ || يَعِظُكُمُ ٱللَّهُ أَن تَعُودُوا۟ لِمِثْلِهِۦٓ أَبَدًا إِن كُنتُم مُّؤْمِنِينَ ﴿١٧﴾ Allah admonisheth you that ye repeat not the like thereof ever, if ye are (in truth) believers. (24:17) || يَعِظُكُمُ ٱللَّهُ أَن تَعُودُوا۟ لِمِثْلِهِۦٓ أَبَدًا إِن كُنتُم مُّؤْمِنِينَ ﴿١٧﴾ الله تاسو ته نصیحت کوي چې بیا هېڅکله داسې خبره ونه کړئ که مؤمنان یاست.
p وَيُبَيِّنُ ٱللَّهُ لَكُمُ ٱلْءَايَٰتِ ۚ وَٱللَّهُ عَلِيمٌ حَكِيمٌ ﴿١٨﴾ اور اللہ تمھارے لیے آیات کھول کھول کر بیان فرماتا ہے اور اللہ خوب جاننے والا بڑی حکمت والا ہے۔ || وَيُبَيِّنُ ٱللَّهُ لَكُمُ ٱلْءَايَٰتِ ۚ وَٱللَّهُ عَلِيمٌ حَكِيمٌ ﴿١٨﴾ And He expoundeth unto you the revelations. Allah is Knower, Wise. (24:18) || وَيُبَيِّنُ ٱللَّهُ لَكُمُ ٱلْءَايَٰتِ ۚ وَٱللَّهُ عَلِيمٌ حَكِيمٌ ﴿١٨﴾ او الله تاسو ته آیتونه ښه روښانه بیانوي، او الله ښه پوه، ډېر حکمتناک دی.
p إِنَّ ٱلَّذِينَ يُحِبُّونَ أَن تَشِيعَ ٱلْفَٰحِشَةُ فِى ٱلَّذِينَ ءَامَنُوا۟ لَهُمْ عَذَابٌ أَلِيمٌۭ فِى ٱلدُّنْيَا وَٱلْءَاخِرَةِ ۚ وَٱللَّهُ يَعْلَمُ وَأَنتُمْ لَا تَعْلَمُونَ ﴿١٩﴾ بے شک جو لوگ پسند کرتے ہیں کہ اہلِ ایمان میں بے حیائی پھیلے اُن کے لیے دنیا و آخرت میں دردناک عذاب ہے اور اللہ جانتا ہے اور تم نہیں جانتے۔ || إِنَّ ٱلَّذِينَ يُحِبُّونَ أَن تَشِيعَ ٱلْفَٰحِشَةُ فِى ٱلَّذِينَ ءَامَنُوا۟ لَهُمْ عَذَابٌ أَلِيمٌۭ فِى ٱلدُّنْيَا وَٱلْءَاخِرَةِ ۚ وَٱللَّهُ يَعْلَمُ وَأَنتُمْ لَا تَعْلَمُونَ ﴿١٩﴾ Lo! those who love that slander should be spread concerning those who believe, theirs will be a painful punishment in the world and the Hereafter. Allah knoweth. Ye know not. (24:19) || إِنَّ ٱلَّذِينَ يُحِبُّونَ أَن تَشِيعَ ٱلْفَٰحِشَةُ فِى ٱلَّذِينَ ءَامَنُوا۟ لَهُمْ عَذَابٌ أَلِيمٌۭ فِى ٱلدُّنْيَا وَٱلْءَاخِرَةِ ۚ وَٱللَّهُ يَعْلَمُ وَأَنتُمْ لَا تَعْلَمُونَ ﴿١٩﴾ بې‌شکه هغه کسان چې خوښوي چې په مؤمنانو کې بې‌حیایي خپره شي، د هغوی لپاره په دنیا او آخرت کې دردناک عذاب دی؛ او الله پوهېږي او تاسو نه پوهېږئ.
p وَلَوْلَا فَضْلُ ٱللَّهِ عَلَيْكُمْ وَرَحْمَتُهُۥ وَأَنَّ ٱللَّهَ رَءُوفٌۭ رَّحِيمٌۭ ﴿٢٠﴾ اور اگر تم پر اللہ کا فضل اور اُس کی رحمت نہ ہوتی (تو تم پر عذاب آ جاتا) اور بے شک اللہ بہت شفقت فرمانے والا نہایت رحم فرمانے والا ہے۔ || وَلَوْلَا فَضْلُ ٱللَّهِ عَلَيْكُمْ وَرَحْمَتُهُۥ وَأَنَّ ٱللَّهَ رَءُوفٌۭ رَّحِيمٌۭ ﴿٢٠﴾ Had it not been for the grace of Allah and His mercy unto you, and that Allah is Clement, Merciful, (ye had been undone). (24:20) || وَلَوْلَا فَضْلُ ٱللَّهِ عَلَيْكُمْ وَرَحْمَتُهُۥ وَأَنَّ ٱللَّهَ رَءُوفٌۭ رَّحِيمٌۭ ﴿٢٠﴾ او که پر تاسو د الله فضل او د هغه رحمت نه وای، (نو پر تاسو به عذاب راغلی و)؛ او بې‌شکه الله ډېر شفقت کوونکی، بې‌حده مهربان دی.
p ۞ يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ لَا تَتَّبِعُوا۟ خُطُوَٰتِ ٱلشَّيْطَٰنِ ۚ وَمَن يَتَّبِعْ خُطُوَٰتِ ٱلشَّيْطَٰنِ فَإِنَّهُۥ يَأْمُرُ بِٱلْفَحْشَآءِ وَٱلْمُنكَرِ ۚ وَلَوْلَا فَضْلُ ٱللَّهِ عَلَيْكُمْ وَرَحْمَتُهُۥ مَا زَكَىٰ مِنكُم مِّنْ أَحَدٍ أَبَدًۭا وَلَٰكِنَّ ٱللَّهَ يُزَكِّى مَن يَشَآءُ ۗ وَٱللَّهُ سَمِيعٌ عَلِيمٌۭ ﴿٢١﴾ اے ایمان والو! شیطان کے نقشِ قدم کی پیروی نہ کرو اور جو شخص شیطان کے نقشِ قدم کی پیروی کرتا ہے تو بے شک وہ (اسے) بے حیائی اور بُرائی کا حکم دیتا ہے اور اگر تم پر اللہ کا فضل اور اُس کی رحمت نہ ہوتی تو تم میں سے کوئی بھی کبھی پاک نہ ہوتا لیکن اللہ جسے چاہتا ہے پاک فرما دیتا ہے اور اللہ خوب سننے والا خوب جاننے والا ہے۔ || ۞ يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ لَا تَتَّبِعُوا۟ خُطُوَٰتِ ٱلشَّيْطَٰنِ ۚ وَمَن يَتَّبِعْ خُطُوَٰتِ ٱلشَّيْطَٰنِ فَإِنَّهُۥ يَأْمُرُ بِٱلْفَحْشَآءِ وَٱلْمُنكَرِ ۚ وَلَوْلَا فَضْلُ ٱللَّهِ عَلَيْكُمْ وَرَحْمَتُهُۥ مَا زَكَىٰ مِنكُم مِّنْ أَحَدٍ أَبَدًۭا وَلَٰكِنَّ ٱللَّهَ يُزَكِّى مَن يَشَآءُ ۗ وَٱللَّهُ سَمِيعٌ عَلِيمٌۭ ﴿٢١﴾ O ye who believe! Follow not the footsteps of the devil. Unto whomsoever followeth the footsteps of the devil, lo! he commandeth filthiness and wrong. Had it not been for the grace of Allah and His mercy unto you, not one of you would ever have grown pure. But Allah causeth whom He will to grow. And Allah is Hearer, Knower. (24:21) || ۞ يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ لَا تَتَّبِعُوا۟ خُطُوَٰتِ ٱلشَّيْطَٰنِ ۚ وَمَن يَتَّبِعْ خُطُوَٰتِ ٱلشَّيْطَٰنِ فَإِنَّهُۥ يَأْمُرُ بِٱلْفَحْشَآءِ وَٱلْمُنكَرِ ۚ وَلَوْلَا فَضْلُ ٱللَّهِ عَلَيْكُمْ وَرَحْمَتُهُۥ مَا زَكَىٰ مِنكُم مِّنْ أَحَدٍ أَبَدًۭا وَلَٰكِنَّ ٱللَّهَ يُزَكِّى مَن يَشَآءُ ۗ وَٱللَّهُ سَمِيعٌ عَلِيمٌۭ ﴿٢١﴾ ای مؤمنانو! د شیطان د پلونو پیروي مه کوئ؛ او څوک چې د شیطان د پلونو پیروي کوي، نو بې‌شکه هغه (هغه ته) د بې‌حیایۍ او بدۍ حکم کوي؛ او که پر تاسو د الله فضل او د هغه رحمت نه وای، نو له تاسو به هېڅوک هېڅکله پاک نه وای؛ خو الله چې څوک وغواړي پاکوي یې؛ او الله ښه اورېدونکی، ښه پوه دی.
p وَلَا يَأْتَلِ أُو۟لُوا۟ ٱلْفَضْلِ مِنكُمْ وَٱلسَّعَةِ أَن يُؤْتُوٓا۟ أُو۟لِى ٱلْقُرْبَىٰ وَٱلْمَسَٰكِينَ وَٱلْمُهَٰجِرِينَ فِى سَبِيلِ ٱللَّهِ ۖ وَلْيَعْفُوا۟ وَلْيَصْفَحُوٓا۟ ۗ أَلَا تُحِبُّونَ أَن يَغْفِرَ ٱللَّهُ لَكُمْ ۗ وَٱللَّهُ غَفُورٌۭ رَّحِيمٌ ﴿٢٢﴾ اور قسم نہ کھائیں جو تم میں سے (دینی) فضیلت اور (مالی) وسعت والے ہیں کہ وہ رشتہ داروں اور مسکینوں اور اللہ کی راہ میں ہجرت کرنے والوں کو (کچھ) نہ دیں گے اور انھیں چاہیے کہ معاف کر دیں اور درگزر کریں کیا تم پسند نہیں کرتے کہ اللہ تمھیں بخش دے اور اللہ بڑا ہی بخشنے والا نہایت رحم فرمانے والا ہے۔ || وَلَا يَأْتَلِ أُو۟لُوا۟ ٱلْفَضْلِ مِنكُمْ وَٱلسَّعَةِ أَن يُؤْتُوٓا۟ أُو۟لِى ٱلْقُرْبَىٰ وَٱلْمَسَٰكِينَ وَٱلْمُهَٰجِرِينَ فِى سَبِيلِ ٱللَّهِ ۖ وَلْيَعْفُوا۟ وَلْيَصْفَحُوٓا۟ ۗ أَلَا تُحِبُّونَ أَن يَغْفِرَ ٱللَّهُ لَكُمْ ۗ وَٱللَّهُ غَفُورٌۭ رَّحِيمٌ ﴿٢٢﴾ And let not those who possess dignity and ease among you swear not to give to the near of kin and to the needy, and to fugitives for the cause of Allah. Let them forgive and show indulgence. Yearn ye not that Allah may forgive you? Allah is Forgiving, Merciful. (24:22) || وَلَا يَأْتَلِ أُو۟لُوا۟ ٱلْفَضْلِ مِنكُمْ وَٱلسَّعَةِ أَن يُؤْتُوٓا۟ أُو۟لِى ٱلْقُرْبَىٰ وَٱلْمَسَٰكِينَ وَٱلْمُهَٰجِرِينَ فِى سَبِيلِ ٱللَّهِ ۖ وَلْيَعْفُوا۟ وَلْيَصْفَحُوٓا۟ ۗ أَلَا تُحِبُّونَ أَن يَغْفِرَ ٱللَّهُ لَكُمْ ۗ وَٱللَّهُ غَفُورٌۭ رَّحِيمٌ ﴿٢٢﴾ او له تاسو څخه د (دیني) فضیلت او (مالي) پراخۍ خاوندان دې قسم ونه خوري چې خپلوانو، مسکینانو او د الله په لار کې هجرت کوونکو ته به (څه) نه ورکوي؛ او باید وبښي او تېر شي؛ ایا تاسو نه خوښوئ چې الله تاسو وبښي؟ او الله ډېر بښونکی، بې‌حده مهربان دی.
p إِنَّ ٱلَّذِينَ يَرْمُونَ ٱلْمُحْصَنَٰتِ ٱلْغَٰفِلَٰتِ ٱلْمُؤْمِنَٰتِ لُعِنُوا۟ فِى ٱلدُّنْيَا وَٱلْءَاخِرَةِ وَلَهُمْ عَذَابٌ عَظِيمٌۭ ﴿٢٣﴾ بے شک وہ لوگ جو پاک دامن، بھولی بھالی، مومن عورتوں پر تہمت لگاتے ہیں اُن پر دنیا اور آخرت میں لعنت کی گئی ہے اور ان کے لیے بہت بڑا عذاب ہے۔ || إِنَّ ٱلَّذِينَ يَرْمُونَ ٱلْمُحْصَنَٰتِ ٱلْغَٰفِلَٰتِ ٱلْمُؤْمِنَٰتِ لُعِنُوا۟ فِى ٱلدُّنْيَا وَٱلْءَاخِرَةِ وَلَهُمْ عَذَابٌ عَظِيمٌۭ ﴿٢٣﴾ Lo! as for those who traduce virtuous, believing women (who are) careless, cursed are they in the world and the Hereafter. Theirs will be an awful doom (24:23) || إِنَّ ٱلَّذِينَ يَرْمُونَ ٱلْمُحْصَنَٰتِ ٱلْغَٰفِلَٰتِ ٱلْمُؤْمِنَٰتِ لُعِنُوا۟ فِى ٱلدُّنْيَا وَٱلْءَاخِرَةِ وَلَهُمْ عَذَابٌ عَظِيمٌۭ ﴿٢٣﴾ بې‌شکه هغه کسان چې پر پاکلمنو، ساده‌زړو، مؤمنو ښځو تور لګوي، پر هغوی په دنیا او آخرت کې لعنت شوی، او د هغوی لپاره ډېر لوی عذاب دی.
p يَوْمَ تَشْهَدُ عَلَيْهِمْ أَلْسِنَتُهُمْ وَأَيْدِيهِمْ وَأَرْجُلُهُم بِمَا كَانُوا۟ يَعْمَلُونَ ﴿٢٤﴾ (وہ اس دن کو یاد رکھیں) جس دن گواہی دیں گی اُن کے خلاف اُن کی زبانیں اور اُن کے ہاتھ اور ان کے پاؤں اُن (اعمال) پر جو وہ کیا کرتے تھے۔ || يَوْمَ تَشْهَدُ عَلَيْهِمْ أَلْسِنَتُهُمْ وَأَيْدِيهِمْ وَأَرْجُلُهُم بِمَا كَانُوا۟ يَعْمَلُونَ ﴿٢٤﴾ On the day when their tongues and their hands and their feet testify against them as to what they used to do, (24:24) || يَوْمَ تَشْهَدُ عَلَيْهِمْ أَلْسِنَتُهُمْ وَأَيْدِيهِمْ وَأَرْجُلُهُم بِمَا كَانُوا۟ يَعْمَلُونَ ﴿٢٤﴾ (هغه ورځ دې یاده ولري) چې په هغې کې به د هغوی پر خلاف د هغوی ژبې، لاسونه او پښې د هغو (عملونو) شاهدي ورکړي چې کول به یې.
p يَوْمَئِذٍۢ يُوَفِّيهِمُ ٱللَّهُ دِينَهُمُ ٱلْحَقَّ وَيَعْلَمُونَ أَنَّ ٱللَّهَ هُوَ ٱلْحَقُّ ٱلْمُبِينُ ﴿٢٥﴾ اس دن اللہ انھیں اُن (کے اعمال) کا حقیقی بدلہ پورا پورا عطا فرمائے گا اور وہ جان لیں گے کہ بے شک اللہ ہی حق ہے (اور حق کا) ظاہر فرمانے والا ہے۔ || يَوْمَئِذٍۢ يُوَفِّيهِمُ ٱللَّهُ دِينَهُمُ ٱلْحَقَّ وَيَعْلَمُونَ أَنَّ ٱللَّهَ هُوَ ٱلْحَقُّ ٱلْمُبِينُ ﴿٢٥﴾ On that day Allah will pay them their just due, and they will know that Allah, He is the Manifest Truth. (24:25) || يَوْمَئِذٍۢ يُوَفِّيهِمُ ٱللَّهُ دِينَهُمُ ٱلْحَقَّ وَيَعْلَمُونَ أَنَّ ٱللَّهَ هُوَ ٱلْحَقُّ ٱلْمُبِينُ ﴿٢٥﴾ په هغه ورځ به الله هغوی ته د هغوی (د عملونو) اصلي بدله پوره پوره ورکړي، او هغوی به پوه شي چې بې‌شکه یوازې الله حق دی (او د حق) ښکاره کوونکی دی.
