p رُّسُلًۭا مُّبَشِّرِينَ وَمُنذِرِينَ لِئَلَّا يَكُونَ لِلنَّاسِ عَلَى ٱللَّهِ حُجَّةٌۢ بَعْدَ ٱلرُّسُلِ ۚ وَكَانَ ٱللَّهُ عَزِيزًا حَكِيمًۭا ﴿١٦٥﴾ تمام رسول (جنت کی) بشارت دینے والے اور (جہنم کا) ڈر سنانے والے (بنا کر بھیجے گئے) تھے تاکہ لوگوں کے پاس اللہ کے ہاں کوئی حجت نہ رہے رسولوں کے آنے کے بعد اور اللہ بڑا غالب بڑی حکمت والا ہے۔ || رُّسُلًۭا مُّبَشِّرِينَ وَمُنذِرِينَ لِئَلَّا يَكُونَ لِلنَّاسِ عَلَى ٱللَّهِ حُجَّةٌۢ بَعْدَ ٱلرُّسُلِ ۚ وَكَانَ ٱللَّهُ عَزِيزًا حَكِيمًۭا ﴿١٦٥﴾ Messengers of good cheer and of warning, in order that mankind might have no argument against Allah after the messengers. Allah was ever Mighty, Wise. (4:165) || رُّسُلًۭا مُّبَشِّرِينَ وَمُنذِرِينَ لِئَلَّا يَكُونَ لِلنَّاسِ عَلَى ٱللَّهِ حُجَّةٌۢ بَعْدَ ٱلرُّسُلِ ۚ وَكَانَ ٱللَّهُ عَزِيزًا حَكِيمًۭا ﴿١٦٥﴾ ټول رسولان د (جنت) زېری ورکوونکي او د (دوزخ) وېره ورکوونکي (لېږل شوي) وو، تر څو د رسولانو له راتګ وروسته د خلکو لپاره د الله په وړاندې کوم دلیل پاتې نه شي؛ او الله ډېر غالب، ډېر حکمتناک دی.
p لَّٰكِنِ ٱللَّهُ يَشْهَدُ بِمَآ أَنزَلَ إِلَيْكَ ۖ أَنزَلَهُۥ بِعِلْمِهِۦ ۖ وَٱلْمَلَٰٓئِكَةُ يَشْهَدُونَ ۚ وَكَفَىٰ بِٱللَّهِ شَهِيدًا ﴿١٦٦﴾ (کافر مانیں یا نہ مانیں) لیکن اللہ گواہی دیتا ہے کہ جو اُس نے آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کی طرف نازل فرمایا وہ اپنے علم سے نازل فرمایا ہے اور فرشتے بھی گواہی دیتے ہیں اور اللہ ہی (بطور) گواہ کافی ہے۔ || لَّٰكِنِ ٱللَّهُ يَشْهَدُ بِمَآ أَنزَلَ إِلَيْكَ ۖ أَنزَلَهُۥ بِعِلْمِهِۦ ۖ وَٱلْمَلَٰٓئِكَةُ يَشْهَدُونَ ۚ وَكَفَىٰ بِٱللَّهِ شَهِيدًا ﴿١٦٦﴾ But Allah (Himself) testifieth concerning that which He hath revealeth unto thee; in His knowledge hath He revealed it; and the angels also testify. And Allah is sufficient Witness. (4:166) || لَّٰكِنِ ٱللَّهُ يَشْهَدُ بِمَآ أَنزَلَ إِلَيْكَ ۖ أَنزَلَهُۥ بِعِلْمِهِۦ ۖ وَٱلْمَلَٰٓئِكَةُ يَشْهَدُونَ ۚ وَكَفَىٰ بِٱللَّهِ شَهِيدًا ﴿١٦٦﴾ (کافران ومني که نه) خو الله شاهدي ورکوي چې هغه څه چې تاسو (صلی الله علیه وعلی آله واصحابه وسلم) ته یې نازل کړي، په خپل علم یې نازل کړي، او پرښتې هم شاهدي ورکوي؛ او الله د شاهد په توګه بس دی.
p إِنَّ ٱلَّذِينَ كَفَرُوا۟ وَصَدُّوا۟ عَن سَبِيلِ ٱللَّهِ قَدْ ضَلُّوا۟ ضَلَٰلًۢا بَعِيدًا ﴿١٦٧﴾ بے شک جن لوگوں نے کفر کیا اور اللہ کی راہ سے روکا یقیناً وہ دور کی گم راہی میں جا پڑے۔ || إِنَّ ٱلَّذِينَ كَفَرُوا۟ وَصَدُّوا۟ عَن سَبِيلِ ٱللَّهِ قَدْ ضَلُّوا۟ ضَلَٰلًۢا بَعِيدًا ﴿١٦٧﴾ Lo! those who disbelieve and hinder (others) from the way of Allah, they verily have wandered far astray. (4:167) || إِنَّ ٱلَّذِينَ كَفَرُوا۟ وَصَدُّوا۟ عَن سَبِيلِ ٱللَّهِ قَدْ ضَلُّوا۟ ضَلَٰلًۢا بَعِيدًا ﴿١٦٧﴾ بې‌شکه هغه کسان چې کفر یې وکړ او د الله له لارې یې منع کړه، یقیناً لرې ګمراهۍ ته ولوېدل.
p إِنَّ ٱلَّذِينَ كَفَرُوا۟ وَظَلَمُوا۟ لَمْ يَكُنِ ٱللَّهُ لِيَغْفِرَ لَهُمْ وَلَا لِيَهْدِيَهُمْ طَرِيقًا ﴿١٦٨﴾ بے شک جن لوگوں نے کفر کیا اور ظلم کیا اللہ انھیں ہرگز نہیں بخشے گا اور نہ انھیں کوئی راستہ دکھائے گا۔ || إِنَّ ٱلَّذِينَ كَفَرُوا۟ وَظَلَمُوا۟ لَمْ يَكُنِ ٱللَّهُ لِيَغْفِرَ لَهُمْ وَلَا لِيَهْدِيَهُمْ طَرِيقًا ﴿١٦٨﴾ Lo! those who disbelieve and deal in wrong, Allah will never forgive them, neither will He guide them unto a road, (4:168) || إِنَّ ٱلَّذِينَ كَفَرُوا۟ وَظَلَمُوا۟ لَمْ يَكُنِ ٱللَّهُ لِيَغْفِرَ لَهُمْ وَلَا لِيَهْدِيَهُمْ طَرِيقًا ﴿١٦٨﴾ بې‌شکه هغه کسان چې کفر یې وکړ او ظلم یې وکړ، الله به یې هېڅکله ونه بښي او نه به یې کومه لاره ورته وښیي.
p إِلَّا طَرِيقَ جَهَنَّمَ خَٰلِدِينَ فِيهَآ أَبَدًۭا ۚ وَكَانَ ذَٰلِكَ عَلَى ٱللَّهِ يَسِيرًۭا ﴿١٦٩﴾ سوائے جہنم کے راستہ کے وہ اس میں ہمیشہ ہمیشہ رہیں گے اور یہ اللہ پر بہت آسان ہے۔ || إِلَّا طَرِيقَ جَهَنَّمَ خَٰلِدِينَ فِيهَآ أَبَدًۭا ۚ وَكَانَ ذَٰلِكَ عَلَى ٱللَّهِ يَسِيرًۭا ﴿١٦٩﴾ Except the road of hell, wherein they will abide for ever. And that is ever easy for Allah. (4:169) || إِلَّا طَرِيقَ جَهَنَّمَ خَٰلِدِينَ فِيهَآ أَبَدًۭا ۚ وَكَانَ ذَٰلِكَ عَلَى ٱللَّهِ يَسِيرًۭا ﴿١٦٩﴾ پرته د دوزخ له لارې، چې پکې به تل تر تله وي؛ او دا پر الله ډېر اسانه دی.
p يَٰٓأَيُّهَا ٱلنَّاسُ قَدْ جَآءَكُمُ ٱلرَّسُولُ بِٱلْحَقِّ مِن رَّبِّكُمْ فَـَٔامِنُوا۟ خَيْرًۭا لَّكُمْ ۚ وَإِن تَكْفُرُوا۟ فَإِنَّ لِلَّهِ مَا فِى ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ ۚ وَكَانَ ٱللَّهُ عَلِيمًا حَكِيمًۭا ﴿١٧٠﴾ اے لوگو! تمھارے پاس تمھارے رب کی طرف سے رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) تشریف لا چکے ہیں حق کے ساتھ پس ایمان لے آؤ (یہی) تمھارے حق میں بہتر ہے اور اگر تم کفر کرو گے تو بے شک اللہ ہی کا ہے جو کچھ آسمانوں اور زمینوں میں ہے اور اللہ خوب جاننے والا بڑی حکمت والا ہے۔ || يَٰٓأَيُّهَا ٱلنَّاسُ قَدْ جَآءَكُمُ ٱلرَّسُولُ بِٱلْحَقِّ مِن رَّبِّكُمْ فَـَٔامِنُوا۟ خَيْرًۭا لَّكُمْ ۚ وَإِن تَكْفُرُوا۟ فَإِنَّ لِلَّهِ مَا فِى ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ ۚ وَكَانَ ٱللَّهُ عَلِيمًا حَكِيمًۭا ﴿١٧٠﴾ O mankind! The messenger hath come unto you with the Truth from your Lord. Therefor believe; (it is) better for you. But if ye disbelieve, still, lo! unto Allah belongeth whatsoever is in the heavens and the earth. Allah is ever Knower, Wise. (4:170) || يَٰٓأَيُّهَا ٱلنَّاسُ قَدْ جَآءَكُمُ ٱلرَّسُولُ بِٱلْحَقِّ مِن رَّبِّكُمْ فَـَٔامِنُوا۟ خَيْرًۭا لَّكُمْ ۚ وَإِن تَكْفُرُوا۟ فَإِنَّ لِلَّهِ مَا فِى ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ ۚ وَكَانَ ٱللَّهُ عَلِيمًا حَكِيمًۭا ﴿١٧٠﴾ ای خلکو! تاسو ته ستاسو د رب له لوري رسول (صلی الله علیه وعلی آله واصحابه وسلم) له حق سره راغلی؛ نو ایمان راوړئ، (همدا) ستاسو په ګټه غوره دی؛ او که کفر وکړئ، نو بې‌شکه د الله دي هغه څه چې په آسمانونو او ځمکه کې دي؛ او الله ښه پوه، ډېر حکمتناک دی.
p يَٰٓأَهْلَ ٱلْكِتَٰبِ لَا تَغْلُوا۟ فِى دِينِكُمْ وَلَا تَقُولُوا۟ عَلَى ٱللَّهِ إِلَّا ٱلْحَقَّ ۚ إِنَّمَا ٱلْمَسِيحُ عِيسَى ٱبْنُ مَرْيَمَ رَسُولُ ٱللَّهِ وَكَلِمَتُهُۥٓ أَلْقَىٰهَآ إِلَىٰ مَرْيَمَ وَرُوحٌۭ مِّنْهُ ۖ فَـَٔامِنُوا۟ بِٱللَّهِ وَرُسُلِهِۦ ۖ وَلَا تَقُولُوا۟ ثَلَٰثَةٌ ۚ ٱنتَهُوا۟ خَيْرًۭا لَّكُمْ ۚ إِنَّمَا ٱللَّهُ إِلَٰهٌۭ وَٰحِدٌۭ ۖ سُبْحَٰنَهُۥٓ أَن يَكُونَ لَهُۥ وَلَدٌۭ ۘ لَّهُۥ مَا فِى ٱلسَّمَٰوَٰتِ وَمَا فِى ٱلْأَرْضِ ۗ وَكَفَىٰ بِٱللَّهِ وَكِيلًۭا ﴿١٧١﴾ اے اہلِ کتاب! اپنے دین میں حد سے نہ بڑھو اور اللہ کے بارے میں سوائے حق بات کے کچھ مت کہو بے شک مسیح عیسیٰ ابن مریم (علیہ السلام) تو صرف اللہ کے رسول ہیں اور وہ اُس کا کلمہ ہیں جو اُس نے مریم (علیہا السلام) کی طرف پہنچایا اور وہ اُس (اللہ) کی طرف سے ایک روح ہیں پس اللہ پر اور اس کے رسولوں پر ایمان لاؤ اور مت کہو کہ تین (معبود) ہیں (اس سے) باز آ جاؤ (یہی) تمھارے لیے بہتر ہے بے شک اللہ ایک ہی معبود ہے وہ (اللہ) اِس سے پاک ہے کہ اُس کا کوئی بیٹا ہو اسی کا ہے جو کچھ آسمانوں میں ہے اور جو کچھ زمینوں میں ہے اور اللہ ہی (بطور) کارساز کافی ہے۔ || يَٰٓأَهْلَ ٱلْكِتَٰبِ لَا تَغْلُوا۟ فِى دِينِكُمْ وَلَا تَقُولُوا۟ عَلَى ٱللَّهِ إِلَّا ٱلْحَقَّ ۚ إِنَّمَا ٱلْمَسِيحُ عِيسَى ٱبْنُ مَرْيَمَ رَسُولُ ٱللَّهِ وَكَلِمَتُهُۥٓ أَلْقَىٰهَآ إِلَىٰ مَرْيَمَ وَرُوحٌۭ مِّنْهُ ۖ فَـَٔامِنُوا۟ بِٱللَّهِ وَرُسُلِهِۦ ۖ وَلَا تَقُولُوا۟ ثَلَٰثَةٌ ۚ ٱنتَهُوا۟ خَيْرًۭا لَّكُمْ ۚ إِنَّمَا ٱللَّهُ إِلَٰهٌۭ وَٰحِدٌۭ ۖ سُبْحَٰنَهُۥٓ أَن يَكُونَ لَهُۥ وَلَدٌۭ ۘ لَّهُۥ مَا فِى ٱلسَّمَٰوَٰتِ وَمَا فِى ٱلْأَرْضِ ۗ وَكَفَىٰ بِٱللَّهِ وَكِيلًۭا ﴿١٧١﴾ O People of the Scripture! Do not exaggerate in your religion nor utter aught concerning Allah save the truth. The Messiah, Jesus son of Mary, was only a messenger of Allah, and His word which He conveyed unto Mary, and a spirit from Him. So believe in Allah and His messengers, and say not "Three" - Cease! (it is) better for you! - Allah is only One Allah. Far is it removed from His Transcendent Majesty that He should have a son. His is all that is in the heavens and all that is in the earth. And Allah is sufficient as Defender. (4:171) || يَٰٓأَهْلَ ٱلْكِتَٰبِ لَا تَغْلُوا۟ فِى دِينِكُمْ وَلَا تَقُولُوا۟ عَلَى ٱللَّهِ إِلَّا ٱلْحَقَّ ۚ إِنَّمَا ٱلْمَسِيحُ عِيسَى ٱبْنُ مَرْيَمَ رَسُولُ ٱللَّهِ وَكَلِمَتُهُۥٓ أَلْقَىٰهَآ إِلَىٰ مَرْيَمَ وَرُوحٌۭ مِّنْهُ ۖ فَـَٔامِنُوا۟ بِٱللَّهِ وَرُسُلِهِۦ ۖ وَلَا تَقُولُوا۟ ثَلَٰثَةٌ ۚ ٱنتَهُوا۟ خَيْرًۭا لَّكُمْ ۚ إِنَّمَا ٱللَّهُ إِلَٰهٌۭ وَٰحِدٌۭ ۖ سُبْحَٰنَهُۥٓ أَن يَكُونَ لَهُۥ وَلَدٌۭ ۘ لَّهُۥ مَا فِى ٱلسَّمَٰوَٰتِ وَمَا فِى ٱلْأَرْضِ ۗ وَكَفَىٰ بِٱللَّهِ وَكِيلًۭا ﴿١٧١﴾ ای اهل کتابو! په خپل دین کې له حده مه اوړئ او د الله په اړه له حق خبرې پرته هېڅ مه وایئ؛ بې‌شکه مسیح عیسی ابن مریم (علیه السلام) یوازې د الله رسول دی او د هغه کلمه ده چې مریم (علیها السلام) ته یې ورسوله، او د هغه (الله) له لوري یو روح دی؛ نو پر الله او د هغه پر رسولانو ایمان راوړئ او مه وایئ چې درې (معبودان) دي؛ (له دې) لاس واخلئ، (همدا) ستاسو لپاره غوره دی؛ بې‌شکه الله یو معبود دی؛ هغه (الله) له دې پاک دی چې کوم زوی ولري؛ د هغه دي هغه څه چې په آسمانونو کې دي او هغه څه چې په ځمکه کې دي؛ او الله د کارساز په توګه بس دی.
