q 1 | يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓا۟ أَوْفُوا۟ بِٱلْعُقُودِ ۚ أُحِلَّتْ لَكُم بَهِيمَةُ ٱلْأَنْعَٰمِ إِلَّا مَا يُتْلَىٰ عَلَيْكُمْ غَيْرَ مُحِلِّى ٱلصَّيْدِ وَأَنتُمْ حُرُمٌ — آیت کریمہ میں اہلِ ایمان کو کیا تعلیمات دی گئیں ہیں؟ || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓا۟ أَوْفُوا۟ بِٱلْعُقُودِ ۚ أُحِلَّتْ لَكُم بَهِيمَةُ ٱلْأَنْعَٰمِ إِلَّا مَا يُتْلَىٰ عَلَيْكُمْ غَيْرَ مُحِلِّى ٱلصَّيْدِ وَأَنتُمْ حُرُمٌ — What teachings have been given to the believers in the verse? || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓا۟ أَوْفُوا۟ بِٱلْعُقُودِ ۚ أُحِلَّتْ لَكُم بَهِيمَةُ ٱلْأَنْعَٰمِ إِلَّا مَا يُتْلَىٰ عَلَيْكُمْ غَيْرَ مُحِلِّى ٱلصَّيْدِ وَأَنتُمْ حُرُمٌ — په آیت کې مؤمنانو ته کومې لارښوونې شوې؟
sol جواب ## اے ایمان والو! عہد و پیمان (عقود) پورے کرو؛ تمہارے لیے چوپائے (مویشی) حلال کیے گئے سوائے ان کے جن کا حکم تمہیں سنایا جائے گا، مگر احرام کی حالت میں شکار کو حلال نہ سمجھو؛ اللہ جو چاہتا ہے حکم دیتا ہے (آیت 1)۔
q 2 | ٱلْيَوْمَ أَكْمَلْتُ لَكُمْ دِينَكُمْ وَأَتْمَمْتُ عَلَيْكُمْ نِعْمَتِى وَرَضِيتُ لَكُمُ ٱلْإِسْلَٰمَ دِينًۭا — آیت کریمہ میں اللہ تعالیٰ کی طرف سے کیا اعلان فرمایا گیا ہے؟ || ٱلْيَوْمَ أَكْمَلْتُ لَكُمْ دِينَكُمْ وَأَتْمَمْتُ عَلَيْكُمْ نِعْمَتِى وَرَضِيتُ لَكُمُ ٱلْإِسْلَٰمَ دِينًۭا — What has been announced by Allah Almighty in the verse? || ٱلْيَوْمَ أَكْمَلْتُ لَكُمْ دِينَكُمْ وَأَتْمَمْتُ عَلَيْكُمْ نِعْمَتِى وَرَضِيتُ لَكُمُ ٱلْإِسْلَٰمَ دِينًۭا — په آیت کې د الله تعالی له لوري څه اعلان شوی؟
sol جواب ## اللہ نے اعلان فرمایا: آج میں نے تمہارے لیے تمہارا دین مکمل کر دیا، تم پر اپنی نعمت پوری کر دی اور تمہارے لیے اسلام کو دین کے طور پر پسند کر لیا — یہ آیت حجۃ الوداع کے موقع پر عرفہ کے دن نازل ہوئی (آیت 3)۔
q 3 | فَكُلُوا۟ مِمَّآ أَمْسَكْنَ عَلَيْكُمْ وَٱذْكُرُوا۟ ٱسْمَ ٱللَّهِ عَلَيْهِ ۖ وَٱتَّقُوا۟ ٱللَّهَ ۚ إِنَّ ٱللَّهَ سَرِيعُ ٱلْحِسَابِ — آیت کریمہ میں شکاری جانور کے حوالے سے کیا احکامات دیے گئے ہیں؟ || فَكُلُوا۟ مِمَّآ أَمْسَكْنَ عَلَيْكُمْ وَٱذْكُرُوا۟ ٱسْمَ ٱللَّهِ عَلَيْهِ ۖ وَٱتَّقُوا۟ ٱللَّهَ ۚ إِنَّ ٱللَّهَ سَرِيعُ ٱلْحِسَابِ — What commands have been given in the verse concerning hunting animals? || فَكُلُوا۟ مِمَّآ أَمْسَكْنَ عَلَيْكُمْ وَٱذْكُرُوا۟ ٱسْمَ ٱللَّهِ عَلَيْهِ ۖ وَٱتَّقُوا۟ ٱللَّهَ ۚ إِنَّ ٱللَّهَ سَرِيعُ ٱلْحِسَابِ — په آیت کې د ښکاري ځناور په اړه کوم حکمونه شوي؟
sol جواب ## سدھائے ہوئے شکاری جانور (کتے، باز) جو شکار تمہارے لیے روک رکھیں اسے کھا سکتے ہو؛ شکار پر چھوڑتے وقت اللہ کا نام لو؛ اور اللہ سے ڈرتے رہو، بے شک اللہ جلد حساب لینے والا ہے (آیت 4)۔
q 4 | يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓا۟ إِذَا قُمْتُمْ إِلَى ٱلصَّلَوٰةِ فَٱغْسِلُوا۟ وُجُوهَكُمْ وَأَيْدِيَكُمْ إِلَى ٱلْمَرَافِقِ وَٱمْسَحُوا۟ بِرُءُوسِكُمْ وَأَرْجُلَكُمْ إِلَى ٱلْكَعْبَيْنِ — آیت کریمہ میں وضو کے کیا فرائض بیان کیے گئے ہیں؟ || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓا۟ إِذَا قُمْتُمْ إِلَى ٱلصَّلَوٰةِ فَٱغْسِلُوا۟ وُجُوهَكُمْ وَأَيْدِيَكُمْ إِلَى ٱلْمَرَافِقِ وَٱمْسَحُوا۟ بِرُءُوسِكُمْ وَأَرْجُلَكُمْ إِلَى ٱلْكَعْبَيْنِ — What obligations of wudu have been stated in the verse? || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓا۟ إِذَا قُمْتُمْ إِلَى ٱلصَّلَوٰةِ فَٱغْسِلُوا۟ وُجُوهَكُمْ وَأَيْدِيَكُمْ إِلَى ٱلْمَرَافِقِ وَٱمْسَحُوا۟ بِرُءُوسِكُمْ وَأَرْجُلَكُمْ إِلَى ٱلْكَعْبَيْنِ — په آیت کې د اودس کوم فرضونه بیان شوي؟
sol جواب ## وضو کے فرائض: (1) چہرہ دھونا؛ (2) دونوں ہاتھ کہنیوں سمیت دھونا؛ (3) سر کا مسح کرنا؛ (4) دونوں پاؤں ٹخنوں سمیت دھونا (آیت 6)؛ جنابت ہو تو غسل اور پانی نہ ملے تو تیمم۔
q 5 | وَقَالَتِ ٱلْيَهُودُ وَٱلنَّصَٰرَىٰ نَحْنُ أَبْنَٰٓؤُا۟ ٱللَّهِ وَأَحِبَّٰٓؤُهُۥ — آیت کریمہ میں یہود و نصاریٰ کا کیا باطل عقیدہ بیان کیا گیا ہے؟ || وَقَالَتِ ٱلْيَهُودُ وَٱلنَّصَٰرَىٰ نَحْنُ أَبْنَٰٓؤُا۟ ٱللَّهِ وَأَحِبَّٰٓؤُهُۥ — What false belief of the Jews and Christians has been stated in the verse? || وَقَالَتِ ٱلْيَهُودُ وَٱلنَّصَٰرَىٰ نَحْنُ أَبْنَٰٓؤُا۟ ٱللَّهِ وَأَحِبَّٰٓؤُهُۥ — په آیت کې د یهودو او نصاراوو کومه باطله عقیده بیان شوې؟
sol جواب ## یہود و نصاریٰ کا جھوٹا عقیدہ کہ ہم اللہ کے بیٹے اور اس کے محبوب ہیں؛ جواب دیا گیا: پھر وہ تمہیں تمہارے گناہوں پر سزا کیوں دیتا ہے؟ بلکہ تم بھی اس کی مخلوق میں سے انسان ہی ہو (آیت 18)۔
q 6 | قَالُوا۟ يَٰمُوسَىٰٓ إِنَّا لَن نَّدْخُلَهَآ أَبَدًۭا مَّا دَامُوا۟ فِيهَا ۖ فَٱذْهَبْ أَنتَ وَرَبُّكَ فَقَٰتِلَآ إِنَّا هَٰهُنَا قَٰعِدُونَ — آیت کریمہ کے مطابق بنی اسرائیل نے سیدنا موسیٰ علیہ السلام کو جہاد کے مطالبہ کا کیا جواب دیا؟ || قَالُوا۟ يَٰمُوسَىٰٓ إِنَّا لَن نَّدْخُلَهَآ أَبَدًۭا مَّا دَامُوا۟ فِيهَا ۖ فَٱذْهَبْ أَنتَ وَرَبُّكَ فَقَٰتِلَآ إِنَّا هَٰهُنَا قَٰعِدُونَ — According to the verse, what answer did the Banu Isra'il give to Sayyiduna Musa's (peace be upon him) demand for jihad? || قَالُوا۟ يَٰمُوسَىٰٓ إِنَّا لَن نَّدْخُلَهَآ أَبَدًۭا مَّا دَامُوا۟ فِيهَا ۖ فَٱذْهَبْ أَنتَ وَرَبُّكَ فَقَٰتِلَآ إِنَّا هَٰهُنَا قَٰعِدُونَ — د آیت له مخې بني اسرائیلو د سیدنا موسی علیه السلام د جهاد غوښتنې ته څه ځواب ورکړ؟
sol جواب ## بنی اسرائیل نے موسیٰؑ سے کہا: اے موسیٰ! جب تک وہ (طاقتور قوم) وہاں ہیں ہم اس (ارضِ مقدس) میں ہرگز داخل نہ ہوں گے؛ تم اور تمہارا رب جا کر لڑو، ہم تو یہیں بیٹھے ہیں (آیت 24)۔
q 7 | قَالَ فَإِنَّهَا مُحَرَّمَةٌ عَلَيْهِمْ ۛ أَرْبَعِينَ سَنَةًۭ ۛ يَتِيهُونَ فِى ٱلْأَرْضِ — آیت کریمہ کے مطابق بنی اسرائیل کو کیا سزا دی گئی؟ || قَالَ فَإِنَّهَا مُحَرَّمَةٌ عَلَيْهِمْ ۛ أَرْبَعِينَ سَنَةًۭ ۛ يَتِيهُونَ فِى ٱلْأَرْضِ — According to the verse, what punishment was given to the Banu Isra'il? || قَالَ فَإِنَّهَا مُحَرَّمَةٌ عَلَيْهِمْ ۛ أَرْبَعِينَ سَنَةًۭ ۛ يَتِيهُونَ فِى ٱلْأَرْضِ — د آیت له مخې بني اسرائیلو ته څه سزا ورکړل شوه؟

ss ترجمہ: اے ایمان والو! اللہ کی اس نعمت کو یاد کرو جو تم پر ہوئی، جب ایک قوم نے ارادہ کیا کہ تم پر دست درازی کرے تو اللہ نے ان کے ہاتھ تم سے روک دیے؛ اور اللہ سے ڈرو، اور مومنوں کو اللہ ہی پر بھروسا کرنا چاہیے۔ (المائدہ 11)

qp 2 | (۱ب) سُوْرَةُ الْمَآئِدَةِ کا تعارف تحریر کریں۔ || (1b) Write the introduction of Surah al-Ma'idah. || (۱ب) د سورة المائده پېژندنه ولیکئ.
solution جواب
ss سورۃ المائدہ مدنی سورت ہے؛ اس میں 120 آیات اور 16 رکوع ہیں؛ ترتیب میں پانچویں سورت ہے؛ یہ آخری دور (صلح حدیبیہ کے بعد سے حجۃ الوداع تک) میں نازل ہوئی۔
ss نام: آیت 112–115 میں حواریوں کے لیے آسمان سے "مائدہ" (دستر خوان) اترنے کا ذکر ہے، اسی سے نام رکھا گیا۔
ss اہمیت: اسی میں «اَلْيَوْمَ اَكْمَلْتُ لَكُمْ دِيْنَكُمْ» نازل ہوئی؛ اس کا موضوع عہد و پیمان کی پابندی، حلال و حرام، اور اسلامی ریاست کے قوانین ہیں۔

qp 3 | (2a) وَإِذْ قَالَ مُوسَىٰ لِقَوْمِهِۦ يَٰقَوْمِ ٱذْكُرُوا۟ نِعْمَةَ ٱللَّهِ عَلَيْكُمْ إِذْ جَعَلَ فِيكُمْ أَنۢبِيَآءَ وَجَعَلَكُم مُّلُوكًۭا وَءَاتَىٰكُم مَّا لَمْ يُؤْتِ أَحَدًۭا مِّنَ ٱلْعَٰلَمِينَ — آیت کریمہ کا ترجمہ تحریر کریں۔ || (2a) وَإِذْ قَالَ مُوسَىٰ لِقَوْمِهِۦ يَٰقَوْمِ ٱذْكُرُوا۟ نِعْمَةَ ٱللَّهِ عَلَيْكُمْ إِذْ جَعَلَ فِيكُمْ أَنۢبِيَآءَ وَجَعَلَكُم مُّلُوكًۭا وَءَاتَىٰكُم مَّا لَمْ يُؤْتِ أَحَدًۭا مِّنَ ٱلْعَٰلَمِينَ — Write the translation of the verse. || (2a) وَإِذْ قَالَ مُوسَىٰ لِقَوْمِهِۦ يَٰقَوْمِ ٱذْكُرُوا۟ نِعْمَةَ ٱللَّهِ عَلَيْكُمْ إِذْ جَعَلَ فِيكُمْ أَنۢبِيَآءَ وَجَعَلَكُم مُّلُوكًۭا وَءَاتَىٰكُم مَّا لَمْ يُؤْتِ أَحَدًۭا مِّنَ ٱلْعَٰلَمِينَ — د آیت ژباړه ولیکئ.
solution جواب
ss ترجمہ: اور جب موسیٰ نے اپنی قوم سے کہا: اے میری قوم! اللہ کی اس نعمت کو یاد کرو جو تم پر ہوئی جب اس نے تم میں نبی پیدا کیے اور تمہیں بادشاہ (آزاد اور خود مختار) بنایا اور تمہیں وہ کچھ دیا جو دنیا والوں میں کسی کو نہ دیا تھا۔ (المائدہ 20)
ss تشریح: بنی اسرائیل کو فرعون کی غلامی سے نجات، انبیا کا سلسلہ، من و سلویٰ اور بادل کا سایہ جیسی نعمتیں ملیں، مگر انہوں نے ناشکری کی۔

qp 4 | (۲ب) سُوْرَةُ الْمَآئِدَةِ کے چند بنیادی مضامین تحریر کریں۔ || (2b) Write some basic themes of Surah al-Ma'idah. || (۲ب) د سورة المائده ځینې بنسټیز مضامین ولیکئ.
solution جواب
ss سورۃ المائدہ کے بنیادی مضامین: (1) عہد و پیمان پورے کرنا؛ (2) حلال و حرام جانور اور کھانے، شکار اور احرام کے احکام؛ (3) تکمیلِ دین کا اعلان؛ (4) وضو، غسل اور تیمم۔
ss (5) عدل و انصاف — دشمن کی دشمنی بھی ناانصافی پر آمادہ نہ کرے؛ نیکی اور تقویٰ میں تعاون؛ (6) قتل، ڈاکا (حرابہ) اور چوری کی سزائیں؛ ہابیل و قابیل کا واقعہ۔
ss (7) شراب، جوا، بت اور پانسوں کی حرمت؛ قسموں کا کفارہ؛ (8) یہود و نصاریٰ کی عہد شکنی، تحریفِ کتاب اور تثلیث کا رد؛ ان سے قلبی دوستی کی ممانعت؛ (9) حضرت عیسیٰؑ کے معجزات اور مائدہ کا واقعہ؛ قیامت میں حضرت عیسیٰؑ سے سوال۔
endsolution
