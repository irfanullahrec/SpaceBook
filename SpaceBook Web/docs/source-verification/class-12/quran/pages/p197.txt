p إِن يَثْقَفُوكُمْ يَكُونُوا۟ لَكُمْ أَعْدَآءًۭ وَيَبْسُطُوٓا۟ إِلَيْكُمْ أَيْدِيَهُمْ وَأَلْسِنَتَهُم بِٱلسُّوٓءِ وَوَدُّوا۟ لَوْ تَكْفُرُونَ ﴿٢﴾ اگر وہ تم پر قابو پا لیں تو تمھارے دشمن بن جائیں گے اور تم پر اپنے ہاتھ اور اپنی زبانیں برائی کے ساتھ دراز کریں گے اور ان کی خواہش یہ ہے کہ تم بھی کافر بن جاؤ۔ || إِن يَثْقَفُوكُمْ يَكُونُوا۟ لَكُمْ أَعْدَآءًۭ وَيَبْسُطُوٓا۟ إِلَيْكُمْ أَيْدِيَهُمْ وَأَلْسِنَتَهُم بِٱلسُّوٓءِ وَوَدُّوا۟ لَوْ تَكْفُرُونَ ﴿٢﴾ If they have the upper hand of you, they will be your foes, and will stretch out their hands and their tongues toward you with evil (intent), and they long for you to disbelieve. (60:2) || إِن يَثْقَفُوكُمْ يَكُونُوا۟ لَكُمْ أَعْدَآءًۭ وَيَبْسُطُوٓا۟ إِلَيْكُمْ أَيْدِيَهُمْ وَأَلْسِنَتَهُم بِٱلسُّوٓءِ وَوَدُّوا۟ لَوْ تَكْفُرُونَ ﴿٢﴾ که هغوی پر تاسو واکمن شي، نو ستاسو دښمنان به شي، او پر تاسو به خپل لاسونه او ژبې په بدۍ سره اوږدې کړي، او د هغوی غوښتنه دا ده چې تاسو هم کافران شئ.
p لَن تَنفَعَكُمْ أَرْحَامُكُمْ وَلَآ أَوْلَٰدُكُمْ ۚ يَوْمَ ٱلْقِيَٰمَةِ يَفْصِلُ بَيْنَكُمْ ۚ وَٱللَّهُ بِمَا تَعْمَلُونَ بَصِيرٌۭ ﴿٣﴾ قیامت کے دن تمھاری رشتہ داریاں اور تمھاری اولاد تمھیں ہرگز نفع نہ دیں گی وہ (اللہ) تمھارے درمیان فیصلہ فرما دے گا اور جو تم کر رہے ہو اللہ اُسے خوب دیکھنے والا ہے۔ || لَن تَنفَعَكُمْ أَرْحَامُكُمْ وَلَآ أَوْلَٰدُكُمْ ۚ يَوْمَ ٱلْقِيَٰمَةِ يَفْصِلُ بَيْنَكُمْ ۚ وَٱللَّهُ بِمَا تَعْمَلُونَ بَصِيرٌۭ ﴿٣﴾ Your ties of kindred and your children will avail you naught upon the Day of Resurrection. He will part you. Allah is Seer of what ye do. (60:3) || لَن تَنفَعَكُمْ أَرْحَامُكُمْ وَلَآ أَوْلَٰدُكُمْ ۚ يَوْمَ ٱلْقِيَٰمَةِ يَفْصِلُ بَيْنَكُمْ ۚ وَٱللَّهُ بِمَا تَعْمَلُونَ بَصِيرٌۭ ﴿٣﴾ د قیامت په ورځ به ستاسو خپلولۍ او ستاسو اولاد تاسو ته هېڅکله ګټه ونه رسوي؛ هغه (الله) به ستاسو ترمنځ پرېکړه وکړي؛ او څه چې تاسو کوئ، الله یې ښه لیدونکی دی.
p قَدْ كَانَتْ لَكُمْ أُسْوَةٌ حَسَنَةٌۭ فِىٓ إِبْرَٰهِيمَ وَٱلَّذِينَ مَعَهُۥٓ إِذْ قَالُوا۟ لِقَوْمِهِمْ إِنَّا بُرَءَٰٓؤُا۟ مِنكُمْ وَمِمَّا تَعْبُدُونَ مِن دُونِ ٱللَّهِ كَفَرْنَا بِكُمْ وَبَدَا بَيْنَنَا وَبَيْنَكُمُ ٱلْعَدَٰوَةُ وَٱلْبَغْضَآءُ أَبَدًا حَتَّىٰ تُؤْمِنُوا۟ بِٱللَّهِ وَحْدَهُۥٓ إِلَّا قَوْلَ إِبْرَٰهِيمَ لِأَبِيهِ لَأَسْتَغْفِرَنَّ لَكَ وَمَآ أَمْلِكُ لَكَ مِنَ ٱللَّهِ مِن شَىْءٍۢ ۖ رَّبَّنَا عَلَيْكَ تَوَكَّلْنَا وَإِلَيْكَ أَنَبْنَا وَإِلَيْكَ ٱلْمَصِيرُ ﴿٤﴾ یقیناً تمھارے لیے بہترین نمونہ ہے ابراہیم (علیہ السلام) اور ان لوگوں میں جو (ایمان اور اطاعت میں) ان کے ساتھ تھے جب انھوں نے اپنی قوم سے کہا بے شک ہم تم سے اور اُن سے بیزار ہیں جن کی تم اللہ کے سوا عبادت کرتے ہو ہم تمھارے (عقائد کے) منکر ہوئے ہمارے اور تمھارے درمیان عداوت اور بغض ہمیشہ کے لیے ظاہر ہو چکا ہے جب تک تم ایک اللہ پر ایمان نہ لاؤ مگر ابراہیم (علیہ السلام) کا اپنے باپ سے یہ کہنا (اس سے مستثنیٰ ہے) کہ میں تمھارے لیے ضرور بخشش طلب کروں گا اور میں اللہ کے مقابل تمھارے لیے کسی چیز کا مالک نہیں (ان کی دعا تھی) اے ہمارے رب! ہم نے تجھ ہی پر بھروسہ کیا اور تیری ہی طرف ہم نے رجوع کیا اور تیری ہی طرف پلٹنا ہے۔ || قَدْ كَانَتْ لَكُمْ أُسْوَةٌ حَسَنَةٌۭ فِىٓ إِبْرَٰهِيمَ وَٱلَّذِينَ مَعَهُۥٓ إِذْ قَالُوا۟ لِقَوْمِهِمْ إِنَّا بُرَءَٰٓؤُا۟ مِنكُمْ وَمِمَّا تَعْبُدُونَ مِن دُونِ ٱللَّهِ كَفَرْنَا بِكُمْ وَبَدَا بَيْنَنَا وَبَيْنَكُمُ ٱلْعَدَٰوَةُ وَٱلْبَغْضَآءُ أَبَدًا حَتَّىٰ تُؤْمِنُوا۟ بِٱللَّهِ وَحْدَهُۥٓ إِلَّا قَوْلَ إِبْرَٰهِيمَ لِأَبِيهِ لَأَسْتَغْفِرَنَّ لَكَ وَمَآ أَمْلِكُ لَكَ مِنَ ٱللَّهِ مِن شَىْءٍۢ ۖ رَّبَّنَا عَلَيْكَ تَوَكَّلْنَا وَإِلَيْكَ أَنَبْنَا وَإِلَيْكَ ٱلْمَصِيرُ ﴿٤﴾ There is a goodly pattern for you in Abraham and those with him, when they told their folk: Lo! we are guiltless of you and all that ye worship beside Allah. We have done with you. And there hath arisen between us and you hostility and hate for ever until ye believe in Allah only - save that which Abraham promised his father (when he said): I will ask forgiveness for thee, though I own nothing for thee from Allah - Our Lord! In Thee we put our trust, and unto Thee we turn repentant, and unto Thee is the journeying. (60:4) || قَدْ كَانَتْ لَكُمْ أُسْوَةٌ حَسَنَةٌۭ فِىٓ إِبْرَٰهِيمَ وَٱلَّذِينَ مَعَهُۥٓ إِذْ قَالُوا۟ لِقَوْمِهِمْ إِنَّا بُرَءَٰٓؤُا۟ مِنكُمْ وَمِمَّا تَعْبُدُونَ مِن دُونِ ٱللَّهِ كَفَرْنَا بِكُمْ وَبَدَا بَيْنَنَا وَبَيْنَكُمُ ٱلْعَدَٰوَةُ وَٱلْبَغْضَآءُ أَبَدًا حَتَّىٰ تُؤْمِنُوا۟ بِٱللَّهِ وَحْدَهُۥٓ إِلَّا قَوْلَ إِبْرَٰهِيمَ لِأَبِيهِ لَأَسْتَغْفِرَنَّ لَكَ وَمَآ أَمْلِكُ لَكَ مِنَ ٱللَّهِ مِن شَىْءٍۢ ۖ رَّبَّنَا عَلَيْكَ تَوَكَّلْنَا وَإِلَيْكَ أَنَبْنَا وَإِلَيْكَ ٱلْمَصِيرُ ﴿٤﴾ یقیناً ستاسو لپاره په ابراهیم (علیه السلام) او په هغو کسانو کې چې (په ایمان او اطاعت کې) ورسره وو، غوره بېلګه ده، کله چې هغوی خپل قوم ته وویل: بې‌شکه موږ له تاسو او له هغو څخه بېزاره یو چې تاسو یې له الله پرته عبادت کوئ؛ موږ ستاسو د (عقیدو) منکر شوو؛ زموږ او ستاسو ترمنځ دښمني او کینه د تل لپاره څرګنده شوه تر هغه چې پر یوه الله ایمان راوړئ؛ مګر د ابراهیم (علیه السلام) خپل پلار ته دا ویل (له دې مستثنی دي) چې زه به ستا لپاره خامخا بښنه وغواړم، او زه د الله په وړاندې ستا لپاره د هېڅ شي مالک نه یم؛ (د هغوی دعا وه:) ای زموږ ربه! موږ یوازې پر تا توکل وکړ، او یوازې تا ته مو رجوع وکړه، او یوازې تا ته ورګرځېدل دي.
p رَبَّنَا لَا تَجْعَلْنَا فِتْنَةًۭ لِّلَّذِينَ كَفَرُوا۟ وَٱغْفِرْ لَنَا رَبَّنَآ ۖ إِنَّكَ أَنتَ ٱلْعَزِيزُ ٱلْحَكِيمُ ﴿٥﴾ اے ہمارے رب! ہمیں کافروں کے لیے آزمائش نہ بنا اور اے ہمارے رب! ہمیں بخش دے بے شک تو بہت ہی غالب بڑی حکمت والا ہے۔ || رَبَّنَا لَا تَجْعَلْنَا فِتْنَةًۭ لِّلَّذِينَ كَفَرُوا۟ وَٱغْفِرْ لَنَا رَبَّنَآ ۖ إِنَّكَ أَنتَ ٱلْعَزِيزُ ٱلْحَكِيمُ ﴿٥﴾ Our Lord! Make us not a prey for those who disbelieve, and forgive us, our Lord! Lo! Thou, only Thou, are the Mighty, the Wise. (60:5) || رَبَّنَا لَا تَجْعَلْنَا فِتْنَةًۭ لِّلَّذِينَ كَفَرُوا۟ وَٱغْفِرْ لَنَا رَبَّنَآ ۖ إِنَّكَ أَنتَ ٱلْعَزِيزُ ٱلْحَكِيمُ ﴿٥﴾ ای زموږ ربه! موږ د کافرانو لپاره ازموینه مه ګرځوه، او ای زموږ ربه! موږ وبښه؛ بې‌شکه ته ډېر غالب، ډېر حکمتناک یې.
p لَقَدْ كَانَ لَكُمْ فِيهِمْ أُسْوَةٌ حَسَنَةٌۭ لِّمَن كَانَ يَرْجُوا۟ ٱللَّهَ وَٱلْيَوْمَ ٱلْءَاخِرَ ۚ وَمَن يَتَوَلَّ فَإِنَّ ٱللَّهَ هُوَ ٱلْغَنِىُّ ٱلْحَمِيدُ ﴿٦﴾ یقیناً تمھارے لیے اُن (کے طرزِ عمل) میں بہترین نمونہ ہے ہر اس شخص کے لیے جو اللہ اور آخرت کے دن کی (ملاقات کی) امید رکھتا ہو اور جو منہ موڑے تو (وہ یاد رکھے کہ) بے شک اللہ ہی بے نیاز بہت تعریفوں والا ہے۔ || لَقَدْ كَانَ لَكُمْ فِيهِمْ أُسْوَةٌ حَسَنَةٌۭ لِّمَن كَانَ يَرْجُوا۟ ٱللَّهَ وَٱلْيَوْمَ ٱلْءَاخِرَ ۚ وَمَن يَتَوَلَّ فَإِنَّ ٱللَّهَ هُوَ ٱلْغَنِىُّ ٱلْحَمِيدُ ﴿٦﴾ Verily ye have in them a goodly pattern for everyone who looketh to Allah and the Last Day. And whosoever may turn away, lo! still Allah, He is the Absolute, the Owner of Praise. (60:6) || لَقَدْ كَانَ لَكُمْ فِيهِمْ أُسْوَةٌ حَسَنَةٌۭ لِّمَن كَانَ يَرْجُوا۟ ٱللَّهَ وَٱلْيَوْمَ ٱلْءَاخِرَ ۚ وَمَن يَتَوَلَّ فَإِنَّ ٱللَّهَ هُوَ ٱلْغَنِىُّ ٱلْحَمِيدُ ﴿٦﴾ یقیناً ستاسو لپاره د هغوی (په چلند) کې غوره بېلګه ده د هر هغه چا لپاره چې د الله او د آخرت د ورځې (د ملاقات) هیله لري؛ او څوک چې مخ واړوي، نو (دې په یاد ولري چې) بې‌شکه یوازې الله بې‌نیازه، ډېر ستایل شوی دی.
