p ٱعْلَمُوٓا۟ أَنَّمَا ٱلْحَيَوٰةُ ٱلدُّنْيَا لَعِبٌۭ وَلَهْوٌۭ وَزِينَةٌۭ وَتَفَاخُرٌۢ بَيْنَكُمْ وَتَكَاثُرٌۭ فِى ٱلْأَمْوَٰلِ وَٱلْأَوْلَٰدِ ۖ كَمَثَلِ غَيْثٍ أَعْجَبَ ٱلْكُفَّارَ نَبَاتُهُۥ ثُمَّ يَهِيجُ فَتَرَىٰهُ مُصْفَرًّۭا ثُمَّ يَكُونُ حُطَٰمًۭا ۖ وَفِى ٱلْءَاخِرَةِ عَذَابٌۭ شَدِيدٌۭ وَمَغْفِرَةٌۭ مِّنَ ٱللَّهِ وَرِضْوَٰنٌۭ ۚ وَمَا ٱلْحَيَوٰةُ ٱلدُّنْيَآ إِلَّا مَتَٰعُ ٱلْغُرُورِ ﴿٢٠﴾ (خوب) جان لو کہ دنیا کی زندگی تو صرف کھیل اور تماشا اور زینت اور آپس میں فخر جتانا اور مال و اولاد میں ایک دوسرے پر کثرت حاصل کرنا ہے اس کی مثال اُس بارش کی طرح ہے جس سے اُگنے والی کھیتی نے کسان کو خوش کر دیا پھر وہ (پک کر) خشک ہو جاتی ہے پھر (اے دیکھنے والے) تو اُسے دیکھتا ہے کہ زرد پڑ گئی ہے پھر وہ چورا چورا ہو جاتی ہے اور آخرت میں (کفار کے لیے) سخت عذاب ہے اور (اہلِ ایمان کے لیے) اللہ کی طرف سے مغفرت اور رضامندی ہے اور دنیا کی زندگی تو صرف دھوکا کا سامان ہے۔ || ٱعْلَمُوٓا۟ أَنَّمَا ٱلْحَيَوٰةُ ٱلدُّنْيَا لَعِبٌۭ وَلَهْوٌۭ وَزِينَةٌۭ وَتَفَاخُرٌۢ بَيْنَكُمْ وَتَكَاثُرٌۭ فِى ٱلْأَمْوَٰلِ وَٱلْأَوْلَٰدِ ۖ كَمَثَلِ غَيْثٍ أَعْجَبَ ٱلْكُفَّارَ نَبَاتُهُۥ ثُمَّ يَهِيجُ فَتَرَىٰهُ مُصْفَرًّۭا ثُمَّ يَكُونُ حُطَٰمًۭا ۖ وَفِى ٱلْءَاخِرَةِ عَذَابٌۭ شَدِيدٌۭ وَمَغْفِرَةٌۭ مِّنَ ٱللَّهِ وَرِضْوَٰنٌۭ ۚ وَمَا ٱلْحَيَوٰةُ ٱلدُّنْيَآ إِلَّا مَتَٰعُ ٱلْغُرُورِ ﴿٢٠﴾ Know that the life of the world is only play, and idle talk, and pageantry, and boasting among you, and rivalry in respect of wealth and children; as the likeness of vegetation after rain, whereof the growth is pleasing to the husbandman, but afterward it drieth up and thou seest it turning yellow, then it becometh straw. And in the Hereafter there is grievous punishment, and (also) forgiveness from Allah and His good pleasure, whereas the life of the world is but matter of illusion. (57:20) || ٱعْلَمُوٓا۟ أَنَّمَا ٱلْحَيَوٰةُ ٱلدُّنْيَا لَعِبٌۭ وَلَهْوٌۭ وَزِينَةٌۭ وَتَفَاخُرٌۢ بَيْنَكُمْ وَتَكَاثُرٌۭ فِى ٱلْأَمْوَٰلِ وَٱلْأَوْلَٰدِ ۖ كَمَثَلِ غَيْثٍ أَعْجَبَ ٱلْكُفَّارَ نَبَاتُهُۥ ثُمَّ يَهِيجُ فَتَرَىٰهُ مُصْفَرًّۭا ثُمَّ يَكُونُ حُطَٰمًۭا ۖ وَفِى ٱلْءَاخِرَةِ عَذَابٌۭ شَدِيدٌۭ وَمَغْفِرَةٌۭ مِّنَ ٱللَّهِ وَرِضْوَٰنٌۭ ۚ وَمَا ٱلْحَيَوٰةُ ٱلدُّنْيَآ إِلَّا مَتَٰعُ ٱلْغُرُورِ ﴿٢٠﴾ (ښه) پوه شئ چې د دنیا ژوند خو یوازې لوبه، تماشه، سینګار، په خپل منځ کې وياړ، او په مال او اولاد کې پر یو بل د زیاتوالي غوښتل دي؛ د هغه مثال د هغه باران په څېر دی چې ترې زرغون شوي کښت کروندګر خوشاله کړ، بیا (پوخ شي او) وچ شي، بیا (ای لیدونکیه) ته یې وینې چې ژېړ شو، بیا ټوټې ټوټې شي؛ او په آخرت کې (د کافرانو لپاره) سخت عذاب دی، او (د مؤمنانو لپاره) د الله له لوري بښنه او رضا ده؛ او د دنیا ژوند خو یوازې د دوکې سامان دی.
p سَابِقُوٓا۟ إِلَىٰ مَغْفِرَةٍۢ مِّن رَّبِّكُمْ وَجَنَّةٍ عَرْضُهَا كَعَرْضِ ٱلسَّمَآءِ وَٱلْأَرْضِ أُعِدَّتْ لِلَّذِينَ ءَامَنُوا۟ بِٱللَّهِ وَرُسُلِهِۦ ۚ ذَٰلِكَ فَضْلُ ٱللَّهِ يُؤْتِيهِ مَن يَشَآءُ ۚ وَٱللَّهُ ذُو ٱلْفَضْلِ ٱلْعَظِيمِ ﴿٢١﴾ اپنے رب کی بخشش کی طرف ایک دوسرے پر سبقت لے جاؤ اور ایسی جنت (کے حصول) کی طرف جس کی چوڑائی آسمان اور زمین کی چوڑائی جیسی ہے وہ اُن کے لیے تیار کی گئی ہے جو اللہ اور اُس کے رسولوں پر ایمان لائے یہ اللہ کا فضل ہے وہ اِسے جسے چاہتا ہے عطا فرماتا ہے اور اللہ بہت بڑے فضل والا ہے۔ || سَابِقُوٓا۟ إِلَىٰ مَغْفِرَةٍۢ مِّن رَّبِّكُمْ وَجَنَّةٍ عَرْضُهَا كَعَرْضِ ٱلسَّمَآءِ وَٱلْأَرْضِ أُعِدَّتْ لِلَّذِينَ ءَامَنُوا۟ بِٱللَّهِ وَرُسُلِهِۦ ۚ ذَٰلِكَ فَضْلُ ٱللَّهِ يُؤْتِيهِ مَن يَشَآءُ ۚ وَٱللَّهُ ذُو ٱلْفَضْلِ ٱلْعَظِيمِ ﴿٢١﴾ Race one with another for forgiveness from your Lord and a Garden whereof the breadth is as the breadth of the heavens and the earth, which is in store for those who believe in Allah and His messengers. Such is the bounty of Allah, which He bestoweth upon whom He will, and Allah is of Infinite Bounty. (57:21) || سَابِقُوٓا۟ إِلَىٰ مَغْفِرَةٍۢ مِّن رَّبِّكُمْ وَجَنَّةٍ عَرْضُهَا كَعَرْضِ ٱلسَّمَآءِ وَٱلْأَرْضِ أُعِدَّتْ لِلَّذِينَ ءَامَنُوا۟ بِٱللَّهِ وَرُسُلِهِۦ ۚ ذَٰلِكَ فَضْلُ ٱللَّهِ يُؤْتِيهِ مَن يَشَآءُ ۚ وَٱللَّهُ ذُو ٱلْفَضْلِ ٱلْعَظِيمِ ﴿٢١﴾ د خپل رب د بښنې لوري ته او د داسې جنت (د ترلاسه کولو) لوري ته چې پلنوالی یې د آسمان او ځمکې د پلنوالي په څېر دی، پر یو بل مخکې شئ؛ هغه د هغو کسانو لپاره تیار شوی چې پر الله او د هغه پر رسولانو یې ایمان راوړی؛ دا د الله فضل دی، چا ته چې وغواړي ورکوي یې؛ او الله د ډېر لوی فضل خاوند دی.
p مَآ أَصَابَ مِن مُّصِيبَةٍۢ فِى ٱلْأَرْضِ وَلَا فِىٓ أَنفُسِكُمْ إِلَّا فِى كِتَٰبٍۢ مِّن قَبْلِ أَن نَّبْرَأَهَآ ۚ إِنَّ ذَٰلِكَ عَلَى ٱللَّهِ يَسِيرٌۭ ﴿٢٢﴾ نہیں آتی کوئی مصیبت زمین پر اور نہ تمھاری جانوں پر مگر وہ ایک کتاب (لوحِ محفوظ) میں (لکھی ہوئی) ہے اس سے پہلے کہ ہم اس کو پیدا کریں بے شک یہ اللہ پر بہت آسان ہے۔ || مَآ أَصَابَ مِن مُّصِيبَةٍۢ فِى ٱلْأَرْضِ وَلَا فِىٓ أَنفُسِكُمْ إِلَّا فِى كِتَٰبٍۢ مِّن قَبْلِ أَن نَّبْرَأَهَآ ۚ إِنَّ ذَٰلِكَ عَلَى ٱللَّهِ يَسِيرٌۭ ﴿٢٢﴾ Naught of disaster befalleth in the earth or in yourselves but it is in a Book before we bring it into being - Lo! that is easy for Allah - (57:22) || مَآ أَصَابَ مِن مُّصِيبَةٍۢ فِى ٱلْأَرْضِ وَلَا فِىٓ أَنفُسِكُمْ إِلَّا فِى كِتَٰبٍۢ مِّن قَبْلِ أَن نَّبْرَأَهَآ ۚ إِنَّ ذَٰلِكَ عَلَى ٱللَّهِ يَسِيرٌۭ ﴿٢٢﴾ هېڅ مصیبت نه پر ځمکه راځي او نه ستاسو پر ځانونو، مګر هغه په یوه کتاب (لوح محفوظ) کې (لیکل شوی) دی مخکې له دې چې موږ یې پیدا کړو؛ بې‌شکه دا پر الله ډېر اسان دی.
p لِّكَيْلَا تَأْسَوْا۟ عَلَىٰ مَا فَاتَكُمْ وَلَا تَفْرَحُوا۟ بِمَآ ءَاتَىٰكُمْ ۗ وَٱللَّهُ لَا يُحِبُّ كُلَّ مُخْتَالٍۢ فَخُورٍ ﴿٢٣﴾ (یہ اس لیے بتا دیا) تاکہ تم اس چیز پر غم نہ کرو جو تمھارے ہاتھ سے جاتی رہی اور اُس چیز پر نہ اتراؤ جو اُس نے تمھیں عطا کی ہے اور اللہ کسی اترانے والے فخر کرنے والے کو پسند نہیں فرماتا۔ || لِّكَيْلَا تَأْسَوْا۟ عَلَىٰ مَا فَاتَكُمْ وَلَا تَفْرَحُوا۟ بِمَآ ءَاتَىٰكُمْ ۗ وَٱللَّهُ لَا يُحِبُّ كُلَّ مُخْتَالٍۢ فَخُورٍ ﴿٢٣﴾ That ye grieve not for the sake of that which hath escaped you, nor yet exult because of that which hath been given. Allah loveth not all prideful boasters, (57:23) || لِّكَيْلَا تَأْسَوْا۟ عَلَىٰ مَا فَاتَكُمْ وَلَا تَفْرَحُوا۟ بِمَآ ءَاتَىٰكُمْ ۗ وَٱللَّهُ لَا يُحِبُّ كُلَّ مُخْتَالٍۢ فَخُورٍ ﴿٢٣﴾ (دا ځکه وویل شول) تر څو پر هغه شي غم ونه کړئ چې ستاسو له لاسه ووت، او پر هغه شي ونه ویاړئ چې هغه درکړی؛ او الله هېڅ کبرجن، وياړونکی نه خوښوي.
p ٱلَّذِينَ يَبْخَلُونَ وَيَأْمُرُونَ ٱلنَّاسَ بِٱلْبُخْلِ ۗ وَمَن يَتَوَلَّ فَإِنَّ ٱللَّهَ هُوَ ٱلْغَنِىُّ ٱلْحَمِيدُ ﴿٢٤﴾ جو بخل کرتے ہیں اور لوگوں کو بھی بخل کی تلقین کرتے ہیں اور جس نے (اللہ کے حکم سے) منہ پھیرا تو یقیناً اللہ (بھی) بے نیاز بہت تعریفوں والا ہے۔ || ٱلَّذِينَ يَبْخَلُونَ وَيَأْمُرُونَ ٱلنَّاسَ بِٱلْبُخْلِ ۗ وَمَن يَتَوَلَّ فَإِنَّ ٱللَّهَ هُوَ ٱلْغَنِىُّ ٱلْحَمِيدُ ﴿٢٤﴾ Who hoard and who enjoin upon the people avarice. And whosoever turneth away, still Allah is the Absolute, the Owner of Praise. (57:24) || ٱلَّذِينَ يَبْخَلُونَ وَيَأْمُرُونَ ٱلنَّاسَ بِٱلْبُخْلِ ۗ وَمَن يَتَوَلَّ فَإِنَّ ٱللَّهَ هُوَ ٱلْغَنِىُّ ٱلْحَمِيدُ ﴿٢٤﴾ هغه چې بخل کوي او خلکو ته هم د بخل لارښوونه کوي؛ او چا چې (د الله له حکم) مخ واړاوه، نو یقیناً الله (هم) بې‌نیازه، ډېر ستایل شوی دی.
p لَقَدْ أَرْسَلْنَا رُسُلَنَا بِٱلْبَيِّنَٰتِ وَأَنزَلْنَا مَعَهُمُ ٱلْكِتَٰبَ وَٱلْمِيزَانَ لِيَقُومَ ٱلنَّاسُ بِٱلْقِسْطِ ۖ وَأَنزَلْنَا ٱلْحَدِيدَ فِيهِ بَأْسٌۭ شَدِيدٌۭ وَمَنَٰفِعُ لِلنَّاسِ وَلِيَعْلَمَ ٱللَّهُ مَن يَنصُرُهُۥ وَرُسُلَهُۥ بِٱلْغَيْبِ ۚ إِنَّ ٱللَّهَ قَوِىٌّ عَزِيزٌۭ ﴿٢٥﴾ یقیناً ہم نے اپنے رسولوں کو واضح نشانیوں کے ساتھ بھیجا اور ہم نے اُن کے ساتھ کتاب بھی اتاری اور میزان (عدل) بھی تاکہ لوگ انصاف پر قائم رہیں اور ہم نے لوہا اتارا جس میں سخت (جنگی) طاقت بھی ہے اور لوگوں کے لیے دوسرے فائدے بھی اور تاکہ اللہ ظاہر فرما دے کہ کون اُس (کے دین) کی اور اُس کے رسولوں کی بن دیکھے مدد کرتا ہے بے شک اللہ بہت قوت والا بہت غالب ہے۔ || لَقَدْ أَرْسَلْنَا رُسُلَنَا بِٱلْبَيِّنَٰتِ وَأَنزَلْنَا مَعَهُمُ ٱلْكِتَٰبَ وَٱلْمِيزَانَ لِيَقُومَ ٱلنَّاسُ بِٱلْقِسْطِ ۖ وَأَنزَلْنَا ٱلْحَدِيدَ فِيهِ بَأْسٌۭ شَدِيدٌۭ وَمَنَٰفِعُ لِلنَّاسِ وَلِيَعْلَمَ ٱللَّهُ مَن يَنصُرُهُۥ وَرُسُلَهُۥ بِٱلْغَيْبِ ۚ إِنَّ ٱللَّهَ قَوِىٌّ عَزِيزٌۭ ﴿٢٥﴾ We verily sent Our messengers with clear proofs, and revealed with them the Scripture and the Balance, that mankind may observe right measure; and He revealed iron, wherein is mighty power and (many) uses for mankind, and that Allah may know him who helpeth Him and His messengers, though unseen. Lo! Allah is Strong, Almighty. (57:25) || لَقَدْ أَرْسَلْنَا رُسُلَنَا بِٱلْبَيِّنَٰتِ وَأَنزَلْنَا مَعَهُمُ ٱلْكِتَٰبَ وَٱلْمِيزَانَ لِيَقُومَ ٱلنَّاسُ بِٱلْقِسْطِ ۖ وَأَنزَلْنَا ٱلْحَدِيدَ فِيهِ بَأْسٌۭ شَدِيدٌۭ وَمَنَٰفِعُ لِلنَّاسِ وَلِيَعْلَمَ ٱللَّهُ مَن يَنصُرُهُۥ وَرُسُلَهُۥ بِٱلْغَيْبِ ۚ إِنَّ ٱللَّهَ قَوِىٌّ عَزِيزٌۭ ﴿٢٥﴾ یقیناً موږ خپل رسولان له روښانه نښو سره ولېږل، او له هغوی سره مو کتاب هم نازل کړ او میزان (عدل) هم، تر څو خلک پر انصاف ټینګ پاتې شي؛ او موږ اوسپنه نازله کړه چې په هغې کې سخت (جنګي) ځواک هم دی او د خلکو لپاره نورې ګټې هم؛ او تر څو الله ښکاره کړي چې څوک له لیدو پرته د هغه (د دین) او د هغه د رسولانو مرسته کوي؛ بې‌شکه الله ډېر ځواکمن، ډېر غالب دی.
