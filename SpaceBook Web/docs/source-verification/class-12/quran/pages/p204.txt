p يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ هَلْ أَدُلُّكُمْ عَلَىٰ تِجَٰرَةٍۢ تُنجِيكُم مِّنْ عَذَابٍ أَلِيمٍۢ ﴿١٠﴾ اے ایمان والو! کیا میں ایسی تجارت کی طرف تمھاری رہنمائی کروں جو تمھیں دردناک عذاب سے بچا لے۔ || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ هَلْ أَدُلُّكُمْ عَلَىٰ تِجَٰرَةٍۢ تُنجِيكُم مِّنْ عَذَابٍ أَلِيمٍۢ ﴿١٠﴾ O ye who believe! p تُؤْمِنُونَ بِٱللَّهِ وَرَسُولِهِۦ وَتُجَٰهِدُونَ فِى سَبِيلِ ٱللَّهِ بِأَمْوَٰلِكُمْ وَأَنفُسِكُمْ ۚ ذَٰلِكُمْ خَيْرٌۭ لَّكُمْ إِن كُنتُمْ تَعْلَمُونَ ﴿١١﴾ (وہ یہ ہے کہ) تم اللہ اور اُس کے رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) پر ایمان رکھو اور اللہ کی راہ میں جہاد کرو اپنے مالوں اور اپنی جانوں سے تمھارے لیے یہی بہتر ہے اگر تم جانتے ہو۔ || تُؤْمِنُونَ بِٱللَّهِ وَرَسُولِهِۦ وَتُجَٰهِدُونَ فِى سَبِيلِ ٱللَّهِ بِأَمْوَٰلِكُمْ وَأَنفُسِكُمْ ۚ ذَٰلِكُمْ خَيْرٌۭ لَّكُمْ إِن كُنتُمْ تَعْلَمُونَ ﴿١١﴾ Ye should believe in Allah and His messenger, and should strive for the cause of Allah with your wealth and your lives. That is better for you, if ye did but know. (61:11) || تُؤْمِنُونَ بِٱللَّهِ وَرَسُولِهِۦ وَتُجَٰهِدُونَ فِى سَبِيلِ ٱللَّهِ بِأَمْوَٰلِكُمْ وَأَنفُسِكُمْ ۚ ذَٰلِكُمْ خَيْرٌۭ لَّكُمْ إِن كُنتُمْ تَعْلَمُونَ ﴿١١﴾ (هغه دا ده چې) پر الله او د هغه پر رسول (صلی الله علیه وعلی آله واصحابه وسلم) ایمان ولرئ، او د الله په لار کې په خپلو مالونو او ځانونو جهاد وکړئ؛ ستاسو لپاره همدا غوره ده که پوهېږئ.
p يَغْفِرْ لَكُمْ ذُنُوبَكُمْ وَيُدْخِلْكُمْ جَنَّٰتٍۢ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَٰرُ وَمَسَٰكِنَ طَيِّبَةًۭ فِى جَنَّٰتِ عَدْنٍۢ ۚ ذَٰلِكَ ٱلْفَوْزُ ٱلْعَظِيمُ ﴿١٢﴾ وہ (اللہ) تمھارے گناہ بخش دے گا اور تمھیں (ایسے) باغات میں داخل فرمائے گا جن کے نیچے نہریں بہہ رہی ہوں گی اور نہایت عمدہ رہائش گاہوں میں (ٹھہرائے گا) جو ابدی جنتوں میں ہیں یہی بہت بڑی کامیابی ہے۔ || يَغْفِرْ لَكُمْ ذُنُوبَكُمْ وَيُدْخِلْكُمْ جَنَّٰتٍۢ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَٰرُ وَمَسَٰكِنَ طَيِّبَةًۭ فِى جَنَّٰتِ عَدْنٍۢ ۚ ذَٰلِكَ ٱلْفَوْزُ ٱلْعَظِيمُ ﴿١٢﴾ He will forgive you your sins and bring you into Gardens underneath which rivers flow, and pleasant dwellings in Gardens of Eden. That is the supreme triumph. (61:12) || يَغْفِرْ لَكُمْ ذُنُوبَكُمْ وَيُدْخِلْكُمْ جَنَّٰتٍۢ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَٰرُ وَمَسَٰكِنَ طَيِّبَةًۭ فِى جَنَّٰتِ عَدْنٍۢ ۚ ذَٰلِكَ ٱلْفَوْزُ ٱلْعَظِيمُ ﴿١٢﴾ هغه (الله) به ستاسو ګناهونه وبښي، او تاسو به (داسې) باغونو ته ننباسي چې لاندې به یې ویالې بهېږي، او په ډېرو غوره استوګنځایونو کې به (مو ځای کړي) چې په تلپاتې جنتونو کې دي؛ همدا ډېره لویه بریا ده.
p وَأُخْرَىٰ تُحِبُّونَهَا ۖ نَصْرٌۭ مِّنَ ٱللَّهِ وَفَتْحٌۭ قَرِيبٌۭ ۗ وَبَشِّرِ ٱلْمُؤْمِنِينَ ﴿١٣﴾ اور ایک دوسری (دنیوی نعمت بھی عطا فرمائے گا) جسے تم پسند کرتے ہو (وہ) اللہ کی طرف سے مدد اور جلد ملنے والی فتح ہے اور آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) مومنوں کو (اس بات کی) بشارت دے دیجیے۔ || وَأُخْرَىٰ تُحِبُّونَهَا ۖ نَصْرٌۭ مِّنَ ٱللَّهِ وَفَتْحٌۭ قَرِيبٌۭ ۗ وَبَشِّرِ ٱلْمُؤْمِنِينَ ﴿١٣﴾ And (He will give you) another (blessing) which ye love: help from Allah and present victory. Give good tidings (O Muhammad) to believers. (61:13) || وَأُخْرَىٰ تُحِبُّونَهَا ۖ نَصْرٌۭ مِّنَ ٱللَّهِ وَفَتْحٌۭ قَرِيبٌۭ ۗ وَبَشِّرِ ٱلْمُؤْمِنِينَ ﴿١٣﴾ او یو بل (دنیوي نعمت به هم درکړي) چې تاسو یې خوښوئ، (هغه) د الله له لوري مرسته او ژر راتلونکې بریا ده؛ او تاسو (صلی الله علیه وعلی آله واصحابه وسلم) مؤمنانو ته (د دې خبرې) زېری ورکړئ.
p يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ كُونُوٓا۟ أَنصَارَ ٱللَّهِ كَمَا قَالَ عِيسَى ٱبْنُ مَرْيَمَ لِلْحَوَارِيِّۦنَ مَنْ أَنصَارِىٓ إِلَى ٱللَّهِ ۖ قَالَ ٱلْحَوَارِيُّونَ نَحْنُ أَنصَارُ ٱللَّهِ ۖ فَـَٔامَنَت طَّآئِفَةٌۭ مِّنۢ بَنِىٓ إِسْرَٰٓءِيلَ وَكَفَرَت طَّآئِفَةٌۭ ۖ فَأَيَّدْنَا ٱلَّذِينَ ءَامَنُوا۟ عَلَىٰ عَدُوِّهِمْ فَأَصْبَحُوا۟ ظَٰهِرِينَ ﴿١٤﴾ اے ایمان والو! تم اللہ (کے دین) کے مددگار ہو جاؤ جس طرح عیسیٰ ابن مریم (علیہ السلام) نے حواریوں سے فرمایا تھا کون ہے جو اللہ کی راہ میں میرا مددگار بنے؟ حواریوں نے کہا ہم ہیں اللہ کے (دین کے) مددگار تو بنی اسرائیل کا ایک گروہ ایمان لے آیا اور ایک گروہ نے کفر اختیار کیا چنانچہ جو لوگ ایمان لائے تھے ہم نے ان کے دشمنوں کے مقابلہ میں ان کی مدد کی تو وہ غالب ہو گئے۔ || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ كُونُوٓا۟ أَنصَارَ ٱللَّهِ كَمَا قَالَ عِيسَى ٱبْنُ مَرْيَمَ لِلْحَوَارِيِّۦنَ مَنْ أَنصَارِىٓ إِلَى ٱللَّهِ ۖ قَالَ ٱلْحَوَارِيُّونَ نَحْنُ أَنصَارُ ٱللَّهِ ۖ فَـَٔامَنَت طَّآئِفَةٌۭ مِّنۢ بَنِىٓ إِسْرَٰٓءِيلَ وَكَفَرَت طَّآئِفَةٌۭ ۖ فَأَيَّدْنَا ٱلَّذِينَ ءَامَنُوا۟ عَلَىٰ عَدُوِّهِمْ فَأَصْبَحُوا۟ ظَٰهِرِينَ ﴿١٤﴾ O ye who believe! Be Allah's helpers, even as Jesus son of Mary said unto the disciples: Who are my helpers for Allah? They said: We are Allah's helpers. And a party of the Children of Israel believed, while a party disbelieved. Then We strengthened those who believed against their foe, and they became the uppermost. (61:14) || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ كُونُوٓا۟ أَنصَارَ ٱللَّهِ كَمَا قَالَ عِيسَى ٱبْنُ مَرْيَمَ لِلْحَوَارِيِّۦنَ مَنْ أَنصَارِىٓ إِلَى ٱللَّهِ ۖ قَالَ ٱلْحَوَارِيُّونَ نَحْنُ أَنصَارُ ٱللَّهِ ۖ فَـَٔامَنَت طَّآئِفَةٌۭ مِّنۢ بَنِىٓ إِسْرَٰٓءِيلَ وَكَفَرَت طَّآئِفَةٌۭ ۖ فَأَيَّدْنَا ٱلَّذِينَ ءَامَنُوا۟ عَلَىٰ عَدُوِّهِمْ فَأَصْبَحُوا۟ ظَٰهِرِينَ ﴿١٤﴾ ای مؤمنانو! د الله (د دین) مرستندویان شئ لکه څنګه چې عیسی ابن مریم (علیه السلام) حواریانو ته ویلي وو: څوک دی چې د الله په لار کې زما مرستندوی شي؟ حواریانو وویل: موږ د الله (د دین) مرستندویان یو؛ نو د بني اسرائیلو یوې ډلې ایمان راوړ او یوې ډلې کفر غوره کړ؛ نو چا چې ایمان راوړی و، موږ د هغوی د دښمنانو په مقابل کې د هغوی مرسته وکړه، نو هغوی غالب شول.
## علم و عمل کی باتیں || Matters of knowledge and action || د علم او عمل خبرې
- ایک سچے مسلمان کے قول اور عمل میں مطابقت ہونی چاہیے جو کچھ کہے اسی کے مطابق کر گزرے کیوں کہ قول و فعل کا تضاد انسان کو اللہ تعالیٰ کی ناراضگی میں مبتلا کرتا ہے۔ اندازہ کیا جا سکتا ہے کہ جو مسلمان قول و فعل کے تضاد میں مبتلا ہوتے ہیں وہ اللہ تعالیٰ کے کس قدر غیظ و غضب کا شکار ہو جاتے ہیں۔ || There should be harmony between the words and deeds of a true Muslim: whatever he says, he should carry out accordingly, because the contradiction between word and deed involves a person in the displeasure of Allah Almighty. One can estimate in what anger and wrath of Allah Almighty those Muslims become caught who are involved in contradiction of word and deed. || د یوه رښتیني مسلمان په خبره او عمل کې باید سمون وي؛ څه چې وايي، همغسې دې وکړي، ځکه د خبرې او عمل تضاد انسان د الله تعالی په ناخوښۍ کې اچوي. اټکل کېدای شي چې کوم مسلمانان د خبرې او عمل په تضاد کې اخته وي، د الله تعالی د څومره غوسې او غضب ښکار کېږي.
- سیدنا عیسیٰ علیہ السلام کی زبان سے نبی آخر الزمان صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم کی آمد کی بشارت آپ صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم کی عظمت و شان کا اظہار ہے۔ رسول اللہ صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم سب سے زیادہ حمد کرنے والے تھے، اس لیے آپ صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم کا اسمِ گرامی احمد صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم ہے۔ رسول اللہ صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم تمام مخلوقات اور سارے انبیاء کرام علیہم السلام سے بڑھ کر تعریف و توصیف کے لائق ہیں کیونکہ آپ صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم فضائل، مناقب اور اخلاق کے لحاظ سے سب سے اعلیٰ اور افضل ہیں۔ || The glad tidings of the coming of the Prophet of the end of time (peace and blessings of Allah be upon him) on the tongue of Sayyiduna 'Isa (peace be upon him) are an expression of the greatness and dignity of the Prophet (peace and blessings of Allah be upon him). The Messenger of Allah (peace and blessings of Allah be upon him) praised (Allah) more than anyone; for this reason the blessed name of the Prophet (peace and blessings of Allah be upon him) is Ahmad (peace and blessings of Allah be upon him). The Messenger of Allah (peace and blessings of Allah be upon him) is worthy of praise and description beyond all creation and all the noble Prophets (peace be upon them), because with respect to virtues, merits and character the Prophet (peace and blessings of Allah be upon him) is the highest and most excellent of all. || د سیدنا عیسی علیه السلام په ژبه د آخر الزمان نبي صلی الله علیه وعلی آله واصحابه وسلم د راتګ زېری د هغه صلی الله علیه وعلی آله واصحابه وسلم د عظمت او شان څرګندونه ده. رسول الله صلی الله علیه وعلی آله واصحابه وسلم تر ټولو ډېر حمد کوونکی و، له همدې امله د هغه صلی الله علیه وعلی آله واصحابه وسلم مبارک نوم احمد صلی الله علیه وعلی آله واصحابه وسلم دی. رسول الله صلی الله علیه وعلی آله واصحابه وسلم له ټولو مخلوقاتو او ټولو انبیاء کرامو علیهم السلام زیات د ستاینې وړ دی، ځکه چې هغه صلی الله علیه وعلی آله واصحابه وسلم د فضیلتونو، مناقبو او اخلاقو له مخې تر ټولو لوړ او غوره دی.
