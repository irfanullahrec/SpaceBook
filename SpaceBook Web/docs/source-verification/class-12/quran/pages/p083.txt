p فَإِنْ عُثِرَ عَلَىٰٓ أَنَّهُمَا ٱسْتَحَقَّآ إِثْمًۭا فَـَٔاخَرَانِ يَقُومَانِ مَقَامَهُمَا مِنَ ٱلَّذِينَ ٱسْتَحَقَّ عَلَيْهِمُ ٱلْأَوْلَيَٰنِ فَيُقْسِمَانِ بِٱللَّهِ لَشَهَٰدَتُنَآ أَحَقُّ مِن شَهَٰدَتِهِمَا وَمَا ٱعْتَدَيْنَآ إِنَّآ إِذًۭا لَّمِنَ ٱلظَّٰلِمِينَ ﴿١٠٧﴾ پھر اگر پتہ چلے کہ وہ دونوں (گواہ جھوٹ بول کر) گناہ کے مرتکب ہوئے ہیں تو اُن کی جگہ دو اور افراد کھڑے ہو جائیں اُن میں سے جن کا (جھوٹی گواہی کے ذریعہ) حق مارا گیا ہے جو (میت کے) قریب تر ہوں پھر وہ اللہ کی قسمیں کھائیں (کہ) یقیناً ہماری گواہی اُن دونوں کی گواہی سے زیادہ برحق ہے اور ہم نے (اس گواہی میں) کوئی زیادتی نہیں کی (اگر ہم ایسا کریں) تو اُس وقت یقیناً ہم ظالموں میں سے ہوں گے۔ || فَإِنْ عُثِرَ عَلَىٰٓ أَنَّهُمَا ٱسْتَحَقَّآ إِثْمًۭا فَـَٔاخَرَانِ يَقُومَانِ مَقَامَهُمَا مِنَ ٱلَّذِينَ ٱسْتَحَقَّ عَلَيْهِمُ ٱلْأَوْلَيَٰنِ فَيُقْسِمَانِ بِٱللَّهِ لَشَهَٰدَتُنَآ أَحَقُّ مِن شَهَٰدَتِهِمَا وَمَا ٱعْتَدَيْنَآ إِنَّآ إِذًۭا لَّمِنَ ٱلظَّٰلِمِينَ ﴿١٠٧﴾ But then, if it is afterwards ascertained that both of them merit (the suspicion of) sin, let two others take their place of those nearly concerned, and let them swear by Allah, (saying): Verily our testimony is truer than their testimony and we have not transgressed (the bounds of duty), for them indeed we should be of the evil-doers. (5:107) || فَإِنْ عُثِرَ عَلَىٰٓ أَنَّهُمَا ٱسْتَحَقَّآ إِثْمًۭا فَـَٔاخَرَانِ يَقُومَانِ مَقَامَهُمَا مِنَ ٱلَّذِينَ ٱسْتَحَقَّ عَلَيْهِمُ ٱلْأَوْلَيَٰنِ فَيُقْسِمَانِ بِٱللَّهِ لَشَهَٰدَتُنَآ أَحَقُّ مِن شَهَٰدَتِهِمَا وَمَا ٱعْتَدَيْنَآ إِنَّآ إِذًۭا لَّمِنَ ٱلظَّٰلِمِينَ ﴿١٠٧﴾ بیا که معلومه شي چې هغه دواړه (شاهدان په درواغو) د ګناه مرتکب شوي، نو د هغوی پر ځای دې بل دوه کسان ودرېږي له هغو څخه چې (د درواغجنې شاهدۍ له لارې) یې حق وهل شوی، چې (مړي ته) نږدې وي؛ بیا دې په الله قسم وخوري (چې) «یقیناً زموږ شاهدي د هغو دواړو له شاهدۍ ډېره رښتینې ده، او موږ (په دې شاهدۍ کې) هېڅ تېری نه دی کړی؛ (که داسې وکړو) نو په هغه وخت کې به یقیناً له ظالمانو یو.»
p ذَٰلِكَ أَدْنَىٰٓ أَن يَأْتُوا۟ بِٱلشَّهَٰدَةِ عَلَىٰ وَجْهِهَآ أَوْ يَخَافُوٓا۟ أَن تُرَدَّ أَيْمَٰنٌۢ بَعْدَ أَيْمَٰنِهِمْ ۗ وَٱتَّقُوا۟ ٱللَّهَ وَٱسْمَعُوا۟ ۗ وَٱللَّهُ لَا يَهْدِى ٱلْقَوْمَ ٱلْفَٰسِقِينَ ﴿١٠٨﴾ یہ اس کے زیادہ قریب ہے کہ لوگ (شروع ہی میں) اُس کی صحیح طور پر گواہی دیں یا پھر وہ (اس بات سے) ڈریں کہ (جھوٹی گواہی کی صورت میں) لوٹائی جائیں گی قسمیں (میت کے وارثوں کی طرف) ان کی قسموں کے بعد اور اللہ (کی نافرمانی) سے ڈرو اور (قبول کرنے کی نیت سے حکم) سنو اور اللہ نافرمان لوگوں کو ہدایت نہیں دیتا۔ || ذَٰلِكَ أَدْنَىٰٓ أَن يَأْتُوا۟ بِٱلشَّهَٰدَةِ عَلَىٰ وَجْهِهَآ أَوْ يَخَافُوٓا۟ أَن تُرَدَّ أَيْمَٰنٌۢ بَعْدَ أَيْمَٰنِهِمْ ۗ وَٱتَّقُوا۟ ٱللَّهَ وَٱسْمَعُوا۟ ۗ وَٱللَّهُ لَا يَهْدِى ٱلْقَوْمَ ٱلْفَٰسِقِينَ ﴿١٠٨﴾ Thus it is more likely that they will bear true witness or fear that after their oaths the oaths (of others) will be taken. So be mindful of your duty (to Allah) and hearken. Allah guideth not the froward folk. (5:108) || ذَٰلِكَ أَدْنَىٰٓ أَن يَأْتُوا۟ بِٱلشَّهَٰدَةِ عَلَىٰ وَجْهِهَآ أَوْ يَخَافُوٓا۟ أَن تُرَدَّ أَيْمَٰنٌۢ بَعْدَ أَيْمَٰنِهِمْ ۗ وَٱتَّقُوا۟ ٱللَّهَ وَٱسْمَعُوا۟ ۗ وَٱللَّهُ لَا يَهْدِى ٱلْقَوْمَ ٱلْفَٰسِقِينَ ﴿١٠٨﴾ دا دې ته ډېر نږدې دی چې خلک (له پیله) سمه شاهدي ورکړي، یا (له دې) ووېرېږي چې (د درواغجنې شاهدۍ په صورت کې) د هغوی له قسمونو وروسته به قسمونه (د مړي وارثانو ته) ورګرځول کېږي؛ او له الله (د نافرمانۍ) ووېرېږئ او (د منلو په نیت حکم) واورئ؛ او الله نافرمانو خلکو ته هدایت نه کوي.
p ۞ يَوْمَ يَجْمَعُ ٱللَّهُ ٱلرُّسُلَ فَيَقُولُ مَاذَآ أُجِبْتُمْ ۖ قَالُوا۟ لَا عِلْمَ لَنَآ ۖ إِنَّكَ أَنتَ عَلَّٰمُ ٱلْغُيُوبِ ﴿١٠٩﴾ (اس دن کا خوف کرو) جس دن اللہ تمام رسولوں کو جمع فرمائے گا پھر پوچھے گا (تمھاری اُمتوں کی طرف سے) تمھیں کیا جواب ملا؟ رسول عرض کریں گے کہ ہمیں کوئی علم نہیں بے شک تو ہی غیب کی ساری باتوں کو خوب جاننے والا ہے۔ || ۞ يَوْمَ يَجْمَعُ ٱللَّهُ ٱلرُّسُلَ فَيَقُولُ مَاذَآ أُجِبْتُمْ ۖ قَالُوا۟ لَا عِلْمَ لَنَآ ۖ إِنَّكَ أَنتَ عَلَّٰمُ ٱلْغُيُوبِ ﴿١٠٩﴾ In the day when Allah gathereth together the messengers, and saith: What was your response (from mankind)? they say: We have no knowledge. Lo! Thou, only Thou art the Knower of Things Hidden, (5:109) || ۞ يَوْمَ يَجْمَعُ ٱللَّهُ ٱلرُّسُلَ فَيَقُولُ مَاذَآ أُجِبْتُمْ ۖ قَالُوا۟ لَا عِلْمَ لَنَآ ۖ إِنَّكَ أَنتَ عَلَّٰمُ ٱلْغُيُوبِ ﴿١٠٩﴾ (له هغې ورځې ووېرېږئ) چې په هغې کې الله ټول رسولان راټولوي، بیا به پوښتنه وکړي: «(ستاسو له امتونو) څه ځواب درکړل شو؟» رسولان به عرض وکړي: «موږ ته هېڅ علم نشته؛ بې‌شکه یوازې ته د غیب په ټولو خبرو ښه پوه یې.»
p إِذْ قَالَ ٱللَّهُ يَٰعِيسَى ٱبْنَ مَرْيَمَ ٱذْكُرْ نِعْمَتِى عَلَيْكَ وَعَلَىٰ وَٰلِدَتِكَ إِذْ أَيَّدتُّكَ بِرُوحِ ٱلْقُدُسِ تُكَلِّمُ ٱلنَّاسَ فِى ٱلْمَهْدِ وَكَهْلًۭا ۖ وَإِذْ عَلَّمْتُكَ ٱلْكِتَٰبَ وَٱلْحِكْمَةَ وَٱلتَّوْرَىٰةَ وَٱلْإِنجِيلَ ۖ وَإِذْ تَخْلُقُ مِنَ ٱلطِّينِ كَهَيْـَٔةِ ٱلطَّيْرِ بِإِذْنِى فَتَنفُخُ فِيهَا فَتَكُونُ طَيْرًۢا بِإِذْنِى ۖ وَتُبْرِئُ ٱلْأَكْمَهَ وَٱلْأَبْرَصَ بِإِذْنِى ۖ وَإِذْ تُخْرِجُ ٱلْمَوْتَىٰ بِإِذْنِى ۖ وَإِذْ كَفَفْتُ بَنِىٓ إِسْرَٰٓءِيلَ عَنكَ إِذْ جِئْتَهُم بِٱلْبَيِّنَٰتِ فَقَالَ ٱلَّذِينَ كَفَرُوا۟ مِنْهُمْ إِنْ هَٰذَآ إِلَّا سِحْرٌۭ مُّبِينٌۭ ﴿١١٠﴾ (یہ واقعہ اس دن ہو گا) جب اللہ فرمائے گا اے عیسیٰ ابنِ مریم! تم اپنے اوپر اور اپنی والدہ پر میری نعمت یاد کرو جب میں نے رُوح القدس (جبریل علیہ السلام) کے ذریعہ تمھاری مدد کی تم گہوارے میں اور بڑی عمر میں بھی لوگوں سے بات کرتے تھے اور جب میں نے تمھیں کتاب اور حکمت اور تورات اور انجیل سکھائی اور جب تم میرے حکم سے گارے سے ایک پرندے کی سی صورت (کا پتلا) بناتے تھے اور پھر تم اس میں پھونک مارتے تھے تو وہ میرے حکم سے (اُڑتا ہوا) پرندہ بن جاتا تھا اور تم مادرزاد اندھے اور کوڑھ کے مریض کو میرے حکم سے اچھا کر دیتے تھے اور جب تم میرے حکم سے مُردوں کو (زندہ) نکال کھڑا کرتے تھے اور جب میں نے بنی اسرائیل کو تم سے روک دیا تھا جب تم اُن کے پاس واضح نشانیاں لے کر آئے تھے تو اُن میں سے جنھوں نے کفر کیا کہنے لگے کہ یہ تو کھلا جادو (ہی) ہے۔ || إِذْ قَالَ ٱللَّهُ يَٰعِيسَى ٱبْنَ مَرْيَمَ ٱذْكُرْ نِعْمَتِى عَلَيْكَ وَعَلَىٰ وَٰلِدَتِكَ إِذْ أَيَّدتُّكَ بِرُوحِ ٱلْقُدُسِ تُكَلِّمُ ٱلنَّاسَ فِى ٱلْمَهْدِ وَكَهْلًۭا ۖ وَإِذْ عَلَّمْتُكَ ٱلْكِتَٰبَ وَٱلْحِكْمَةَ وَٱلتَّوْرَىٰةَ وَٱلْإِنجِيلَ ۖ وَإِذْ تَخْلُقُ مِنَ ٱلطِّينِ كَهَيْـَٔةِ ٱلطَّيْرِ بِإِذْنِى فَتَنفُخُ فِيهَا فَتَكُونُ طَيْرًۢا بِإِذْنِى ۖ وَتُبْرِئُ ٱلْأَكْمَهَ وَٱلْأَبْرَصَ بِإِذْنِى ۖ وَإِذْ تُخْرِجُ ٱلْمَوْتَىٰ بِإِذْنِى ۖ وَإِذْ كَفَفْتُ بَنِىٓ إِسْرَٰٓءِيلَ عَنكَ إِذْ جِئْتَهُم بِٱلْبَيِّنَٰتِ فَقَالَ ٱلَّذِينَ كَفَرُوا۟ مِنْهُمْ إِنْ هَٰذَآ إِلَّا سِحْرٌۭ مُّبِينٌۭ ﴿١١٠﴾ When Allah saith: O Jesus, son of Mary! Remember My favour unto thee and unto thy mother; how I strengthened thee with the holy Spirit, so that thou spakest unto mankind in the cradle as in maturity; and how I taught thee the Scripture and Wisdom and the Torah and the Gospel; and how thou didst shape of clay as it were the likeness of a bird by My permission, and didst blow upon it and it was a bird by My permission, and thou didst heal him who was born blind and the leper by My permission; and how thou didst raise the dead by My permission; and how I restrained the Children of Israel from (harming) thee when thou camest unto them with clear proofs, and those of them who disbelieved exclaimed: This is naught else than mere magic; (5:110) || إِذْ قَالَ ٱللَّهُ يَٰعِيسَى ٱبْنَ مَرْيَمَ ٱذْكُرْ نِعْمَتِى عَلَيْكَ وَعَلَىٰ وَٰلِدَتِكَ إِذْ أَيَّدتُّكَ بِرُوحِ ٱلْقُدُسِ تُكَلِّمُ ٱلنَّاسَ فِى ٱلْمَهْدِ وَكَهْلًۭا ۖ وَإِذْ عَلَّمْتُكَ ٱلْكِتَٰبَ وَٱلْحِكْمَةَ وَٱلتَّوْرَىٰةَ وَٱلْإِنجِيلَ ۖ وَإِذْ تَخْلُقُ مِنَ ٱلطِّينِ كَهَيْـَٔةِ ٱلطَّيْرِ بِإِذْنِى فَتَنفُخُ فِيهَا فَتَكُونُ طَيْرًۢا بِإِذْنِى ۖ وَتُبْرِئُ ٱلْأَكْمَهَ وَٱلْأَبْرَصَ بِإِذْنِى ۖ وَإِذْ تُخْرِجُ ٱلْمَوْتَىٰ بِإِذْنِى ۖ وَإِذْ كَفَفْتُ بَنِىٓ إِسْرَٰٓءِيلَ عَنكَ إِذْ جِئْتَهُم بِٱلْبَيِّنَٰتِ فَقَالَ ٱلَّذِينَ كَفَرُوا۟ مِنْهُمْ إِنْ هَٰذَآ إِلَّا سِحْرٌۭ مُّبِينٌۭ ﴿١١٠﴾ (دا پېښه به په هغه ورځ وي) کله چې الله ووایي: «ای عیسی ابن مریم! پر خپل ځان او پر خپلې مور زما نعمت یاد کړه، کله چې ما د روح القدس (جبریل علیه السلام) په وسیله ستا مرسته وکړه؛ ته په زانګو کې او په لوی عمر کې هم له خلکو سره خبرې کولې؛ او کله چې ما تا ته کتاب، حکمت، تورات او انجیل ور زده کړل؛ او کله چې به تا زما په حکم له خټې د مرغه په بڼه (یو جسم) جوړاوه، بیا به دې پکې پوکی کاوه، نو زما په حکم به (الوتونکی) مرغه شو؛ او ته به زما په حکم مورزادی ړوند او د برګي ناروغ روغول؛ او کله چې به تا زما په حکم مړي (ژوندي) را ایستل؛ او کله چې ما بني اسرائیل له تا منع کړل کله چې ته ورته له ښکاره نښو سره راغلې، نو له هغوی هغو چې کفر یې کړی و، وویل: «دا خو (یوازې) ښکاره جادو دی.»»
