p وَٱلَّذِينَ كَفَرُوا۟ وَكَذَّبُوا۟ بِـَٔايَٰتِنَآ أُو۟لَٰٓئِكَ أَصْحَٰبُ ٱلنَّارِ خَٰلِدِينَ فِيهَا ۖ وَبِئْسَ ٱلْمَصِيرُ ﴿١٠﴾ اور جن لوگوں نے کفر کیا اور ہماری آیات کو جھٹلایا وہی جہنم والے ہیں وہ اس میں ہمیشہ رہیں گے اور وہ بدترین ٹھکانا ہے۔ || وَٱلَّذِينَ كَفَرُوا۟ وَكَذَّبُوا۟ بِـَٔايَٰتِنَآ أُو۟لَٰٓئِكَ أَصْحَٰبُ ٱلنَّارِ خَٰلِدِينَ فِيهَا ۖ وَبِئْسَ ٱلْمَصِيرُ ﴿١٠﴾ But those who disbelieve and deny Our revelations, such are owners of the Fire; they will abide therein - a hapless journey's end! (64:10) || وَٱلَّذِينَ كَفَرُوا۟ وَكَذَّبُوا۟ بِـَٔايَٰتِنَآ أُو۟لَٰٓئِكَ أَصْحَٰبُ ٱلنَّارِ خَٰلِدِينَ فِيهَا ۖ وَبِئْسَ ٱلْمَصِيرُ ﴿١٠﴾ او هغه کسان چې کفر یې وکړ او زموږ آیتونه یې دروغ وګڼل، همدوی دوزخیان دي، هغوی به پکې تل پاتې وي؛ او هغه تر ټولو بد ځای دی.
p مَآ أَصَابَ مِن مُّصِيبَةٍ إِلَّا بِإِذْنِ ٱللَّهِ ۗ وَمَن يُؤْمِنۢ بِٱللَّهِ يَهْدِ قَلْبَهُۥ ۚ وَٱللَّهُ بِكُلِّ شَىْءٍ عَلِيمٌۭ ﴿١١﴾ (کسی کو) کوئی مصیبت نہیں پہنچتی مگر اللہ کے اذن سے اور جو اللہ پر ایمان لاتا ہے وہ (اللہ) اُس کے دل کو ہدایت عطا فرماتا ہے اور اللہ ہر شے کو خوب جاننے والا ہے۔ || مَآ أَصَابَ مِن مُّصِيبَةٍ إِلَّا بِإِذْنِ ٱللَّهِ ۗ وَمَن يُؤْمِنۢ بِٱللَّهِ يَهْدِ قَلْبَهُۥ ۚ وَٱللَّهُ بِكُلِّ شَىْءٍ عَلِيمٌۭ ﴿١١﴾ No calamity befalleth save by Allah's leave. And whosoever believeth in Allah, He guideth his heart. And Allah is Knower of all things. (64:11) || مَآ أَصَابَ مِن مُّصِيبَةٍ إِلَّا بِإِذْنِ ٱللَّهِ ۗ وَمَن يُؤْمِنۢ بِٱللَّهِ يَهْدِ قَلْبَهُۥ ۚ وَٱللَّهُ بِكُلِّ شَىْءٍ عَلِيمٌۭ ﴿١١﴾ (چا ته) هېڅ مصیبت نه رسېږي مګر د الله په اجازه؛ او څوک چې پر الله ایمان راوړي، هغه (الله) د هغه زړه ته هدایت ورکوي؛ او الله په هر شي ښه پوه دی.
p وَأَطِيعُوا۟ ٱللَّهَ وَأَطِيعُوا۟ ٱلرَّسُولَ ۚ فَإِن تَوَلَّيْتُمْ فَإِنَّمَا عَلَىٰ رَسُولِنَا ٱلْبَلَٰغُ ٱلْمُبِينُ ﴿١٢﴾ اور تم اللہ کی اطاعت کرو اور رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کی اطاعت کرو پھر اگر تم نے منہ پھیرا تو بے شک ہمارے رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کے ذمہ تو واضح طور پر (پیغامِ حق) پہنچا دینا ہی ہے۔ || وَأَطِيعُوا۟ ٱللَّهَ وَأَطِيعُوا۟ ٱلرَّسُولَ ۚ فَإِن تَوَلَّيْتُمْ فَإِنَّمَا عَلَىٰ رَسُولِنَا ٱلْبَلَٰغُ ٱلْمُبِينُ ﴿١٢﴾ Obey Allah and obey His messenger; but if ye turn away, then the duty of Our messenger is only to convey (the message) plainly. (64:12) || وَأَطِيعُوا۟ ٱللَّهَ وَأَطِيعُوا۟ ٱلرَّسُولَ ۚ فَإِن تَوَلَّيْتُمْ فَإِنَّمَا عَلَىٰ رَسُولِنَا ٱلْبَلَٰغُ ٱلْمُبِينُ ﴿١٢﴾ او د الله اطاعت وکړئ او د رسول (صلی الله علیه وعلی آله واصحابه وسلم) اطاعت وکړئ؛ بیا که مو مخ واړاوه، نو بې‌شکه زموږ د رسول (صلی الله علیه وعلی آله واصحابه وسلم) پر غاړه خو یوازې په ښکاره ډول د (حق پیغام) رسول دي.
p ٱللَّهُ لَآ إِلَٰهَ إِلَّا هُوَ ۚ وَعَلَى ٱللَّهِ فَلْيَتَوَكَّلِ ٱلْمُؤْمِنُونَ ﴿١٣﴾ اللہ وہ ہے جس کے سوا کوئی عبادت کے لائق نہیں اور ایمان والوں کو اللہ ہی پر بھروسہ کرنا چاہیے۔ || ٱللَّهُ لَآ إِلَٰهَ إِلَّا هُوَ ۚ وَعَلَى ٱللَّهِ فَلْيَتَوَكَّلِ ٱلْمُؤْمِنُونَ ﴿١٣﴾ Allah! There is no Allah save Him. In Allah, therefore, let believers put their trust. (64:13) || ٱللَّهُ لَآ إِلَٰهَ إِلَّا هُوَ ۚ وَعَلَى ٱللَّهِ فَلْيَتَوَكَّلِ ٱلْمُؤْمِنُونَ ﴿١٣﴾ الله هغه دی چې له هغه پرته هېڅوک د عبادت وړ نه دی؛ او مؤمنان باید یوازې پر الله توکل وکړي.
p يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓا۟ إِنَّ مِنْ أَزْوَٰجِكُمْ وَأَوْلَٰدِكُمْ عَدُوًّۭا لَّكُمْ فَٱحْذَرُوهُمْ ۚ وَإِن تَعْفُوا۟ وَتَصْفَحُوا۟ وَتَغْفِرُوا۟ فَإِنَّ ٱللَّهَ غَفُورٌۭ رَّحِيمٌ ﴿١٤﴾ اے ایمان والو! بے شک تمھاری بیویوں اور تمھاری اولاد میں سے بعض تمھارے دشمن ہیں تو اُن سے ہوشیار رہو اور اگر تم معاف کر دو اور درگزر کرو اور بخش دو تو بے شک اللہ بڑا ہی بخشنے والا نہایت رحم فرمانے والا ہے۔ || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓا۟ إِنَّ مِنْ أَزْوَٰجِكُمْ وَأَوْلَٰدِكُمْ عَدُوًّۭا لَّكُمْ فَٱحْذَرُوهُمْ ۚ وَإِن تَعْفُوا۟ وَتَصْفَحُوا۟ وَتَغْفِرُوا۟ فَإِنَّ ٱللَّهَ غَفُورٌۭ رَّحِيمٌ ﴿١٤﴾ O ye who believe! Lo! among your wives and your children there are enemies for you, therefor beware of them. And if ye efface and overlook and forgive, then lo! Allah is Forgiving, Merciful. (64:14) || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓا۟ إِنَّ مِنْ أَزْوَٰجِكُمْ وَأَوْلَٰدِكُمْ عَدُوًّۭا لَّكُمْ فَٱحْذَرُوهُمْ ۚ وَإِن تَعْفُوا۟ وَتَصْفَحُوا۟ وَتَغْفِرُوا۟ فَإِنَّ ٱللَّهَ غَفُورٌۭ رَّحِيمٌ ﴿١٤﴾ ای مؤمنانو! بې‌شکه ستاسو له مېرمنو او اولادو ځینې ستاسو دښمنان دي، نو ترې هوښیار اوسئ؛ او که عفو وکړئ، تېر شئ او وبښئ، نو بې‌شکه الله ډېر بښونکی، بې‌حده مهربان دی.
p إِنَّمَآ أَمْوَٰلُكُمْ وَأَوْلَٰدُكُمْ فِتْنَةٌۭ ۚ وَٱللَّهُ عِندَهُۥٓ أَجْرٌ عَظِيمٌۭ ﴿١٥﴾ تمھارے مال اور تمھاری اولاد تو محض آزمائش ہیں اور اللہ ہی کے پاس بہت بڑا اجر ہے۔ || إِنَّمَآ أَمْوَٰلُكُمْ وَأَوْلَٰدُكُمْ فِتْنَةٌۭ ۚ وَٱللَّهُ عِندَهُۥٓ أَجْرٌ عَظِيمٌۭ ﴿١٥﴾ Your wealth and your children are only a temptation, whereas Allah! with Him is an immense reward. (64:15) || إِنَّمَآ أَمْوَٰلُكُمْ وَأَوْلَٰدُكُمْ فِتْنَةٌۭ ۚ وَٱللَّهُ عِندَهُۥٓ أَجْرٌ عَظِيمٌۭ ﴿١٥﴾ ستاسو مالونه او اولاد خو یوازې ازموینه ده؛ او یوازې له الله سره ډېر لوی اجر دی.
p فَٱتَّقُوا۟ ٱللَّهَ مَا ٱسْتَطَعْتُمْ وَٱسْمَعُوا۟ وَأَطِيعُوا۟ وَأَنفِقُوا۟ خَيْرًۭا لِّأَنفُسِكُمْ ۗ وَمَن يُوقَ شُحَّ نَفْسِهِۦ فَأُو۟لَٰٓئِكَ هُمُ ٱلْمُفْلِحُونَ ﴿١٦﴾ تو اللہ (کی نافرمانی) سے ڈرو جہاں تک تم سے ہو سکے اور (اس کے احکام) سنو اور اطاعت کرو اور (اللہ کی راہ میں) خرچ کرو یہ تمھارے (ہی) لیے بہتر ہے اور جو کوئی بچا لیا گیا اپنے نفس کی لالچ سے تو وہی لوگ کامیاب ہونے والے ہیں۔ || فَٱتَّقُوا۟ ٱللَّهَ مَا ٱسْتَطَعْتُمْ وَٱسْمَعُوا۟ وَأَطِيعُوا۟ وَأَنفِقُوا۟ خَيْرًۭا لِّأَنفُسِكُمْ ۗ وَمَن يُوقَ شُحَّ نَفْسِهِۦ فَأُو۟لَٰٓئِكَ هُمُ ٱلْمُفْلِحُونَ ﴿١٦﴾ So keep your duty to Allah as best ye can, and listen, and obey, and spend; that is better for your souls. And whoso is saved from his own greed, such are the successful. (64:16) || فَٱتَّقُوا۟ ٱللَّهَ مَا ٱسْتَطَعْتُمْ وَٱسْمَعُوا۟ وَأَطِيعُوا۟ وَأَنفِقُوا۟ خَيْرًۭا لِّأَنفُسِكُمْ ۗ وَمَن يُوقَ شُحَّ نَفْسِهِۦ فَأُو۟لَٰٓئِكَ هُمُ ٱلْمُفْلِحُونَ ﴿١٦﴾ نو له الله (د نافرمانۍ) ووېرېږئ تر هغه حده چې وس مو رسېږي، او (د هغه حکمونه) واورئ، اطاعت وکړئ، او (د الله په لار کې) ولګوئ؛ دا ستاسو (همدا) لپاره غوره ده؛ او څوک چې د خپل نفس له حرص وژغورل شي، نو همدوی بریالي دي.
p إِن تُقْرِضُوا۟ ٱللَّهَ قَرْضًا حَسَنًۭا يُضَٰعِفْهُ لَكُمْ وَيَغْفِرْ لَكُمْ ۚ وَٱللَّهُ شَكُورٌ حَلِيمٌ ﴿١٧﴾ اگر تم اللہ کو قرضِ حسن دو گے تو وہ اُسے تمھارے لیے کئی گنا بڑھا دے گا اور تمھیں بخش دے گا اور اللہ نہایت قدر دان بڑا حلم والا ہے۔ || إِن تُقْرِضُوا۟ ٱللَّهَ قَرْضًا حَسَنًۭا يُضَٰعِفْهُ لَكُمْ وَيَغْفِرْ لَكُمْ ۚ وَٱللَّهُ شَكُورٌ حَلِيمٌ ﴿١٧﴾ If ye lend unto Allah a goodly loan, He will double it for you and will forgive you, for Allah is Responsive, Clement, (64:17) || إِن تُقْرِضُوا۟ ٱللَّهَ قَرْضًا حَسَنًۭا يُضَٰعِفْهُ لَكُمْ وَيَغْفِرْ لَكُمْ ۚ وَٱللَّهُ شَكُورٌ حَلِيمٌ ﴿١٧﴾ که تاسو الله ته ښه پور ورکړئ، نو هغه به یې ستاسو لپاره څو چنده زیات کړي او تاسو به وبښي؛ او الله ډېر قدردان، ډېر زغمناک دی.
p عَٰلِمُ ٱلْغَيْبِ وَٱلشَّهَٰدَةِ ٱلْعَزِيزُ ٱلْحَكِيمُ ﴿١٨﴾ (اللہ ہر) پوشیدہ اور ظاہر کا جاننے والا بہت غالب بڑی حکمت والا ہے۔ || عَٰلِمُ ٱلْغَيْبِ وَٱلشَّهَٰدَةِ ٱلْعَزِيزُ ٱلْحَكِيمُ ﴿١٨﴾ Knower of the Invisible and the Visible, the Mighty, the Wise. (64:18) || عَٰلِمُ ٱلْغَيْبِ وَٱلشَّهَٰدَةِ ٱلْعَزِيزُ ٱلْحَكِيمُ ﴿١٨﴾ (الله په هر) پټ او ښکاره پوه، ډېر غالب، ډېر حکمتناک دی.
## علم و عمل کی باتیں || Matters of knowledge and action || د علم او عمل خبرې
- قیامت کا ایک نام يَوْمُ التَّغَابُنِ یعنی ہار جیت کا دن ہے۔ اصل ہار اور جیت تو آخرت ہی کی ہے جس دن نیک اہلِ ایمان ہمیشہ کے لیے جیت جائیں گے اور جنت کی نعمتوں سے لطف اندوز ہو رہے ہوں گے جب کہ کفار ہمیشہ کے لیے ہار جائیں گے اور عذابِ الٰہی میں مبتلا ہوں گے۔ || One name of the Day of Resurrection is "yawm al-taghabun", that is, the day of loss and gain. The real loss and gain is that of the Hereafter, on the day when the righteous people of faith will win forever and enjoy the blessings of Paradise, whereas the disbelievers will lose forever and be afflicted with the punishment of God. || د قیامت یو نوم «یوم التغابن» یعنې د ګټې او بایلنې ورځ ده. اصلي ګټه او بایلنه خو د آخرت ده، په هغه ورځ چې نېک مؤمنان به د تل لپاره وګټي او د جنت له نعمتونو به خوند اخلي، حال دا چې کافران به د تل لپاره وبایلي او په الهي عذاب کې به اخته وي.
