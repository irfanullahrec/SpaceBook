p لَّن يَسْتَنكِفَ ٱلْمَسِيحُ أَن يَكُونَ عَبْدًۭا لِّلَّهِ وَلَا ٱلْمَلَٰٓئِكَةُ ٱلْمُقَرَّبُونَ ۚ وَمَن يَسْتَنكِفْ عَنْ عِبَادَتِهِۦ وَيَسْتَكْبِرْ فَسَيَحْشُرُهُمْ إِلَيْهِ جَمِيعًۭا ﴿١٧٢﴾ ہرگز عار نہیں سمجھیں گے مسیح کہ وہ اللہ کے بندے ہوں اور نہ ہی مقرب فرشتے (اسے عار سمجھتے ہیں) اور جو اس (اللہ) کی عبادت کو عار سمجھتا ہے اور تکبر کرتا ہے تو عنقریب اللہ ان سب کو (ایک دن) اپنے پاس جمع فرمائے گا۔ || لَّن يَسْتَنكِفَ ٱلْمَسِيحُ أَن يَكُونَ عَبْدًۭا لِّلَّهِ وَلَا ٱلْمَلَٰٓئِكَةُ ٱلْمُقَرَّبُونَ ۚ وَمَن يَسْتَنكِفْ عَنْ عِبَادَتِهِۦ وَيَسْتَكْبِرْ فَسَيَحْشُرُهُمْ إِلَيْهِ جَمِيعًۭا ﴿١٧٢﴾ The Messiah will never scorn to be a slave unto Allah, nor will the favoured angels. Whoso scorneth His service and is proud, all such will He assemble unto Him; (4:172) || لَّن يَسْتَنكِفَ ٱلْمَسِيحُ أَن يَكُونَ عَبْدًۭا لِّلَّهِ وَلَا ٱلْمَلَٰٓئِكَةُ ٱلْمُقَرَّبُونَ ۚ وَمَن يَسْتَنكِفْ عَنْ عِبَادَتِهِۦ وَيَسْتَكْبِرْ فَسَيَحْشُرُهُمْ إِلَيْهِ جَمِيعًۭا ﴿١٧٢﴾ مسیح به هېڅکله دا شرم ونه ګڼي چې د الله بنده وي، او نه مقربې پرښتې (دا شرم ګڼي)؛ او څوک چې د هغه (الله) عبادت شرم ګڼي او کبر کوي، نو ژر به الله دوی ټول (په یوه ورځ) خپل لوري ته راټول کړي.
p فَأَمَّا ٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّٰلِحَٰتِ فَيُوَفِّيهِمْ أُجُورَهُمْ وَيَزِيدُهُم مِّن فَضْلِهِۦ ۖ وَأَمَّا ٱلَّذِينَ ٱسْتَنكَفُوا۟ وَٱسْتَكْبَرُوا۟ فَيُعَذِّبُهُمْ عَذَابًا أَلِيمًۭا وَلَا يَجِدُونَ لَهُم مِّن دُونِ ٱللَّهِ وَلِيًّۭا وَلَا نَصِيرًۭا ﴿١٧٣﴾ تو جو لوگ ایمان لائے اور جنھوں نے نیک اعمال کیے تو وہ (اللہ) انھیں ان کا اجر پورا پورا عطا فرمائے گا اور انھیں اپنے فضل سے اور مزید عطا فرمائے گا اور وہ لوگ جنھوں نے (اللہ کی بندگی سے) عار کیا اور تکبر کیا تو اللہ انھیں دردناک عذاب دے گا اور وہ اپنے لیے اللہ کے سوا نہ کوئی دوست پائیں گے اور نہ کوئی مدد گار۔ || فَأَمَّا ٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّٰلِحَٰتِ فَيُوَفِّيهِمْ أُجُورَهُمْ وَيَزِيدُهُم مِّن فَضْلِهِۦ ۖ وَأَمَّا ٱلَّذِينَ ٱسْتَنكَفُوا۟ وَٱسْتَكْبَرُوا۟ فَيُعَذِّبُهُمْ عَذَابًا أَلِيمًۭا وَلَا يَجِدُونَ لَهُم مِّن دُونِ ٱللَّهِ وَلِيًّۭا وَلَا نَصِيرًۭا ﴿١٧٣﴾ Then, as for those who believed and did good works, unto them will He pay their wages in full, adding unto them of His bounty; and as for those who were scornful and proud, them will He punish with a painful doom. And they will not find for them, against Allah, any protecting friend or helper. (4:173) || فَأَمَّا ٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّٰلِحَٰتِ فَيُوَفِّيهِمْ أُجُورَهُمْ وَيَزِيدُهُم مِّن فَضْلِهِۦ ۖ وَأَمَّا ٱلَّذِينَ ٱسْتَنكَفُوا۟ وَٱسْتَكْبَرُوا۟ فَيُعَذِّبُهُمْ عَذَابًا أَلِيمًۭا وَلَا يَجِدُونَ لَهُم مِّن دُونِ ٱللَّهِ وَلِيًّۭا وَلَا نَصِيرًۭا ﴿١٧٣﴾ نو هغه کسان چې ایمان یې راوړ او نېک عملونه یې وکړل، هغه (الله) به یې اجر پوره پوره ورکړي او له خپل فضله به یې نور هم ورکړي؛ او هغه کسان چې (د الله له بندګۍ) یې شرم وګاڼه او کبر یې وکړ، نو الله به یې دردناک عذاب ورکړي، او ځان ته به له الله پرته نه کوم دوست ومومي او نه کوم مرستندوی.
p يَٰٓأَيُّهَا ٱلنَّاسُ قَدْ جَآءَكُم بُرْهَٰنٌۭ مِّن رَّبِّكُمْ وَأَنزَلْنَآ إِلَيْكُمْ نُورًۭا مُّبِينًۭا ﴿١٧٤﴾ اے لوگو! یقیناً تمھارے پاس تمھارے رب کی طرف سے کھلی دلیل آ چکی ہے اور ہم نے تمھاری طرف واضح نور اتارا ہے۔ || يَٰٓأَيُّهَا ٱلنَّاسُ قَدْ جَآءَكُم بُرْهَٰنٌۭ مِّن رَّبِّكُمْ وَأَنزَلْنَآ إِلَيْكُمْ نُورًۭا مُّبِينًۭا ﴿١٧٤﴾ O mankind! Now hath a proof from your Lord come unto you, and We have sent down unto you a clear light; (4:174) || يَٰٓأَيُّهَا ٱلنَّاسُ قَدْ جَآءَكُم بُرْهَٰنٌۭ مِّن رَّبِّكُمْ وَأَنزَلْنَآ إِلَيْكُمْ نُورًۭا مُّبِينًۭا ﴿١٧٤﴾ ای خلکو! یقیناً تاسو ته ستاسو د رب له لوري ښکاره دلیل راغلی، او موږ تاسو ته ښکاره نور نازل کړی.
p فَأَمَّا ٱلَّذِينَ ءَامَنُوا۟ بِٱللَّهِ وَٱعْتَصَمُوا۟ بِهِۦ فَسَيُدْخِلُهُمْ فِى رَحْمَةٍۢ مِّنْهُ وَفَضْلٍۢ وَيَهْدِيهِمْ إِلَيْهِ صِرَٰطًۭا مُّسْتَقِيمًۭا ﴿١٧٥﴾ تو جو لوگ اللہ پر ایمان لائے اور اُس (کی رسی) کو مضبوطی سے تھام لیا تو عنقریب وہ انھیں اپنی رحمت اور فضل میں داخل فرمائے گا اور انھیں اپنی طرف آنے کے سیدھے راستے کی ہدایت عطا فرمائے گا۔ || فَأَمَّا ٱلَّذِينَ ءَامَنُوا۟ بِٱللَّهِ وَٱعْتَصَمُوا۟ بِهِۦ فَسَيُدْخِلُهُمْ فِى رَحْمَةٍۢ مِّنْهُ وَفَضْلٍۢ وَيَهْدِيهِمْ إِلَيْهِ صِرَٰطًۭا مُّسْتَقِيمًۭا ﴿١٧٥﴾ As for those who believe in Allah, and hold fast unto Him, them He will cause to enter into His mercy and grace, and will guide them unto Him by a straight road. (4:175) || فَأَمَّا ٱلَّذِينَ ءَامَنُوا۟ بِٱللَّهِ وَٱعْتَصَمُوا۟ بِهِۦ فَسَيُدْخِلُهُمْ فِى رَحْمَةٍۢ مِّنْهُ وَفَضْلٍۢ وَيَهْدِيهِمْ إِلَيْهِ صِرَٰطًۭا مُّسْتَقِيمًۭا ﴿١٧٥﴾ نو هغه کسان چې پر الله یې ایمان راوړ او د هغه (رسۍ) ته یې کلکې منګولې ولګولې، ژر به یې هغه په خپل رحمت او فضل کې ننباسي او خپل لوري ته د راتلو سمې لارې ته به یې هدایت وکړي.
p يَسْتَفْتُونَكَ قُلِ ٱللَّهُ يُفْتِيكُمْ فِى ٱلْكَلَٰلَةِ ۚ إِنِ ٱمْرُؤٌا۟ هَلَكَ لَيْسَ لَهُۥ وَلَدٌۭ وَلَهُۥٓ أُخْتٌۭ فَلَهَا نِصْفُ مَا تَرَكَ ۚ وَهُوَ يَرِثُهَآ إِن لَّمْ يَكُن لَّهَا وَلَدٌۭ ۚ فَإِن كَانَتَا ٱثْنَتَيْنِ فَلَهُمَا ٱلثُّلُثَانِ مِمَّا تَرَكَ ۚ وَإِن كَانُوٓا۟ إِخْوَةًۭ رِّجَالًۭا وَنِسَآءًۭ فَلِلذَّكَرِ مِثْلُ حَظِّ ٱلْأُنثَيَيْنِ ۗ يُبَيِّنُ ٱللَّهُ لَكُمْ أَن تَضِلُّوا۟ ۗ وَٱللَّهُ بِكُلِّ شَىْءٍ عَلِيمٌۢ ﴿١٧٦﴾ وہ آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) سے پوچھتے ہیں آپ فرما دیجیے اللہ تمھیں کلالہ (کی میراث) کے بارے میں حکم دیتا ہے اگر کوئی (ایسا) مرد وفات پا گیا جس کی کوئی اولاد نہ ہو (اور نہ باپ) اور اُس کی (سگی یا باپ کی طرف سے) ایک بہن ہو تو اس (بہن) کے لیے اُس (ترکہ) کا نصف ہے جو اس (بھائی) نے چھوڑا اور وہ (بھائی) اپنی بہن (کے کل ترکہ) کا وارث ہو گا اگر (بہن وفات پا گئی اور) اُس کی کوئی اولاد نہ ہو پھر اگر وہ دو بہنیں ہوں تو اُن دونوں کا اُس میں سے دو تہائی ہے جو اُس (بھائی) نے چھوڑ دیا اور اگر (مرنے والے کے) کئی بہن بھائی ہوں (یعنی) مرد اور عورتیں تو مرد کے لیے دو عورتوں کے حصہ کے برابر ہے اللہ تمھارے لیے (احکام) کھول کھول کر بیان فرماتا ہے کہ کہیں تم گم راہ نہ ہو جاؤ اور اللہ ہر شے کو خوب جاننے والا ہے۔ || يَسْتَفْتُونَكَ قُلِ ٱللَّهُ يُفْتِيكُمْ فِى ٱلْكَلَٰلَةِ ۚ إِنِ ٱمْرُؤٌا۟ هَلَكَ لَيْسَ لَهُۥ وَلَدٌۭ وَلَهُۥٓ أُخْتٌۭ فَلَهَا نِصْفُ مَا تَرَكَ ۚ وَهُوَ يَرِثُهَآ إِن لَّمْ يَكُن لَّهَا وَلَدٌۭ ۚ فَإِن كَانَتَا ٱثْنَتَيْنِ فَلَهُمَا ٱلثُّلُثَانِ مِمَّا تَرَكَ ۚ وَإِن كَانُوٓا۟ إِخْوَةًۭ رِّجَالًۭا وَنِسَآءًۭ فَلِلذَّكَرِ مِثْلُ حَظِّ ٱلْأُنثَيَيْنِ ۗ يُبَيِّنُ ٱللَّهُ لَكُمْ أَن تَضِلُّوا۟ ۗ وَٱللَّهُ بِكُلِّ شَىْءٍ عَلِيمٌۢ ﴿١٧٦﴾ They ask thee for a pronouncement. Say: Allah hath pronounced for you concerning distant kindred. If a man die childless and he have a sister, hers is half the heritage, and he would have inherited from her had she died childless. And if there be two sisters, then theirs are two-thirds of the heritage, and if they be brethren, men and women, unto the male is the equivalent of the share of two females. Allah expoundeth unto you, so that ye err not. Allah is Knower of all things. (4:176) || يَسْتَفْتُونَكَ قُلِ ٱللَّهُ يُفْتِيكُمْ فِى ٱلْكَلَٰلَةِ ۚ إِنِ ٱمْرُؤٌا۟ هَلَكَ لَيْسَ لَهُۥ وَلَدٌۭ وَلَهُۥٓ أُخْتٌۭ فَلَهَا نِصْفُ مَا تَرَكَ ۚ وَهُوَ يَرِثُهَآ إِن لَّمْ يَكُن لَّهَا وَلَدٌۭ ۚ فَإِن كَانَتَا ٱثْنَتَيْنِ فَلَهُمَا ٱلثُّلُثَانِ مِمَّا تَرَكَ ۚ وَإِن كَانُوٓا۟ إِخْوَةًۭ رِّجَالًۭا وَنِسَآءًۭ فَلِلذَّكَرِ مِثْلُ حَظِّ ٱلْأُنثَيَيْنِ ۗ يُبَيِّنُ ٱللَّهُ لَكُمْ أَن تَضِلُّوا۟ ۗ وَٱللَّهُ بِكُلِّ شَىْءٍ عَلِيمٌۢ ﴿١٧٦﴾ هغوی له تاسو (صلی الله علیه وعلی آله واصحابه وسلم) پوښتنه کوي؛ ووایئ: الله تاسو ته د کلالې (د میراث) په اړه حکم درکوي: که کوم (داسې) نارینه مړ شو چې هېڅ اولاد یې نه وي (او نه پلار)، او د هغه یوه (سکه یا پلارنۍ) خور وي، نو هغې (خور) ته د هغه (ترکې) نیمایي ده چې هغه (ورور) پرې ایښې؛ او هغه (ورور) به د خپلې خور (د ټولې ترکې) وارث وي که (خور مړه شي او) هېڅ اولاد یې نه وي؛ بیا که دوه خویندې وي، نو دواړو ته د هغه څه دوه پر درېیمه برخه ده چې هغه (ورور) پرې ایښې؛ او که (د مړي) څو خویندې او وروڼه وي (یعنې) نارینه او ښځې، نو نارینه ته د دوو ښځو د برخې په اندازه ده. الله تاسو ته (حکمونه) ښه روښانه بیانوي چې ګمراه نه شئ؛ او الله په هر شي ښه پوه دی.
