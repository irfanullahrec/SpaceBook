p وَيَوْمَ يُحْشَرُ أَعْدَآءُ ٱللَّهِ إِلَى ٱلنَّارِ فَهُمْ يُوزَعُونَ ﴿١٩﴾ اور جس دن اللہ کے دشمن آگ کی طرف جمع کیے جائیں گے پھر (نظم و ترتیب کے لیے) انھیں روکا جائے گا۔ || وَيَوْمَ يُحْشَرُ أَعْدَآءُ ٱللَّهِ إِلَى ٱلنَّارِ فَهُمْ يُوزَعُونَ ﴿١٩﴾ And (make mention of) the day when the enemies of Allah are gathered unto the Fire, they are driven on (41:19) || وَيَوْمَ يُحْشَرُ أَعْدَآءُ ٱللَّهِ إِلَى ٱلنَّارِ فَهُمْ يُوزَعُونَ ﴿١٩﴾ او په کومه ورځ چې د الله دښمنان د اور لوري ته راټول شي، بیا به (د نظم او ترتیب لپاره) ودرول شي.
p حَتَّىٰٓ إِذَا مَا جَآءُوهَا شَهِدَ عَلَيْهِمْ سَمْعُهُمْ وَأَبْصَٰرُهُمْ وَجُلُودُهُم بِمَا كَانُوا۟ يَعْمَلُونَ ﴿٢٠﴾ یہاں تک کہ جب وہ اس (جہنم) کے پاس پہنچ جائیں گے تو ان کے کان ان کی آنکھیں اور ان کی کھالیں ان کے خلاف گواہی دیں گی ان اعمال کی جو وہ کیا کرتے تھے۔ || حَتَّىٰٓ إِذَا مَا جَآءُوهَا شَهِدَ عَلَيْهِمْ سَمْعُهُمْ وَأَبْصَٰرُهُمْ وَجُلُودُهُم بِمَا كَانُوا۟ يَعْمَلُونَ ﴿٢٠﴾ Till, when they reach it, their ears and their eyes and their skins testify against them as to what they used to do. (41:20) || حَتَّىٰٓ إِذَا مَا جَآءُوهَا شَهِدَ عَلَيْهِمْ سَمْعُهُمْ وَأَبْصَٰرُهُمْ وَجُلُودُهُم بِمَا كَانُوا۟ يَعْمَلُونَ ﴿٢٠﴾ تر دې چې کله هغه (دوزخ) ته ورسېږي، نو د هغوی غوږونه، سترګې او پوستکي به یې د هغوی پر خلاف د هغو عملونو شاهدي ورکړي چې کول به یې.
p وَقَالُوا۟ لِجُلُودِهِمْ لِمَ شَهِدتُّمْ عَلَيْنَا ۖ قَالُوٓا۟ أَنطَقَنَا ٱللَّهُ ٱلَّذِىٓ أَنطَقَ كُلَّ شَىْءٍۢ وَهُوَ خَلَقَكُمْ أَوَّلَ مَرَّةٍۢ وَإِلَيْهِ تُرْجَعُونَ ﴿٢١﴾ اور وہ اپنی کھالوں سے کہیں گے تم نے ہمارے خلاف گواہی کیوں دی وہ کہیں گی ہمیں اُسی اللہ نے بولنا سکھایا ہے جس نے ہر شے کو بولنا سکھایا ہے اور اسی نے تمھیں پہلی بار پیدا فرمایا ہے اور اسی کی طرف تم سب لوٹائے جاؤ گے۔ || وَقَالُوا۟ لِجُلُودِهِمْ لِمَ شَهِدتُّمْ عَلَيْنَا ۖ قَالُوٓا۟ أَنطَقَنَا ٱللَّهُ ٱلَّذِىٓ أَنطَقَ كُلَّ شَىْءٍۢ وَهُوَ خَلَقَكُمْ أَوَّلَ مَرَّةٍۢ وَإِلَيْهِ تُرْجَعُونَ ﴿٢١﴾ And they say unto their skins: Why testify ye against us? They say: Allah hath given us speech Who giveth speech to all things, and Who created you at the first, and unto Whom ye are returned. (41:21) || وَقَالُوا۟ لِجُلُودِهِمْ لِمَ شَهِدتُّمْ عَلَيْنَا ۖ قَالُوٓا۟ أَنطَقَنَا ٱللَّهُ ٱلَّذِىٓ أَنطَقَ كُلَّ شَىْءٍۢ وَهُوَ خَلَقَكُمْ أَوَّلَ مَرَّةٍۢ وَإِلَيْهِ تُرْجَعُونَ ﴿٢١﴾ او هغوی به خپلو پوستکو ته ووايي: تاسو زموږ پر خلاف ولې شاهدي ورکړه؟ هغه به ووايي: موږ ته همغه الله د خبرو کولو توان راکړ چې هر شي ته یې د خبرو کولو توان ورکړی، او همغه تاسو لومړی ځل پیدا کړئ، او همغه ته به ټول بېرته ورګرځول کېږئ.
p وَمَا كُنتُمْ تَسْتَتِرُونَ أَن يَشْهَدَ عَلَيْكُمْ سَمْعُكُمْ وَلَآ أَبْصَٰرُكُمْ وَلَا جُلُودُكُمْ وَلَٰكِن ظَنَنتُمْ أَنَّ ٱللَّهَ لَا يَعْلَمُ كَثِيرًۭا مِّمَّا تَعْمَلُونَ ﴿٢٢﴾ اور تم (تو گناہ کرتے وقت) اس (خوف) سے بھی پردہ نہیں کرتے تھے کہ تمھارے خلاف گواہی دیں گے تمھارے کان اور تمھاری آنکھیں اور تمھاری کھالیں لیکن تم تو گمان کرتے تھے کہ اللہ (تمھارے) بہت سے (اعمال) جانتا ہی نہیں جو تم کرتے ہو۔ || وَمَا كُنتُمْ تَسْتَتِرُونَ أَن يَشْهَدَ عَلَيْكُمْ سَمْعُكُمْ وَلَآ أَبْصَٰرُكُمْ وَلَا جُلُودُكُمْ وَلَٰكِن ظَنَنتُمْ أَنَّ ٱللَّهَ لَا يَعْلَمُ كَثِيرًۭا مِّمَّا تَعْمَلُونَ ﴿٢٢﴾ Ye did not hide yourselves lest your ears and your eyes and your skins should testify against you, but ye deemed that Allah knew not much of what ye did. (41:22) || وَمَا كُنتُمْ تَسْتَتِرُونَ أَن يَشْهَدَ عَلَيْكُمْ سَمْعُكُمْ وَلَآ أَبْصَٰرُكُمْ وَلَا جُلُودُكُمْ وَلَٰكِن ظَنَنتُمْ أَنَّ ٱللَّهَ لَا يَعْلَمُ كَثِيرًۭا مِّمَّا تَعْمَلُونَ ﴿٢٢﴾ او تاسو (د ګناه پر مهال) له دې (وېرې) هم پرده نه کوله چې ستاسو غوږونه، سترګې او پوستکي به ستاسو پر خلاف شاهدي ورکړي، بلکې تاسو ګومان کاوه چې الله ستاسو ډېر هغه (عملونه) نه پېژني چې تاسو یې کوئ.
p وَذَٰلِكُمْ ظَنُّكُمُ ٱلَّذِى ظَنَنتُم بِرَبِّكُمْ أَرْدَىٰكُمْ فَأَصْبَحْتُم مِّنَ ٱلْخَٰسِرِينَ ﴿٢٣﴾ اور تمھارے اسی گمان نے جو تم اپنے رب کے بارے میں کیا کرتے تھے تمھیں ہلاک کر دیا لہٰذا تم نقصان اٹھانے والوں میں سے ہو گئے۔ || وَذَٰلِكُمْ ظَنُّكُمُ ٱلَّذِى ظَنَنتُم بِرَبِّكُمْ أَرْدَىٰكُمْ فَأَصْبَحْتُم مِّنَ ٱلْخَٰسِرِينَ ﴿٢٣﴾ That, your thought which ye did think about your Lord, hath ruined you; and ye find yourselves (this day) among the lost. (41:23) || وَذَٰلِكُمْ ظَنُّكُمُ ٱلَّذِى ظَنَنتُم بِرَبِّكُمْ أَرْدَىٰكُمْ فَأَصْبَحْتُم مِّنَ ٱلْخَٰسِرِينَ ﴿٢٣﴾ او ستاسو همدې ګومان چې د خپل رب په اړه به مو کاوه، تاسو هلاک کړئ؛ نو تاسو له زیانمنو څخه شوئ.
p فَإِن يَصْبِرُوا۟ فَٱلنَّارُ مَثْوًۭى لَّهُمْ ۖ وَإِن يَسْتَعْتِبُوا۟ فَمَا هُم مِّنَ ٱلْمُعْتَبِينَ ﴿٢٤﴾ پھر اگر وہ صبر کریں تو بھی ان کا ٹھکانا جہنم ہی ہے اور اگر وہ توبہ کرنا چاہیں تو انھیں توبہ کا موقع نہیں دیا جائے گا۔ || فَإِن يَصْبِرُوا۟ فَٱلنَّارُ مَثْوًۭى لَّهُمْ ۖ وَإِن يَسْتَعْتِبُوا۟ فَمَا هُم مِّنَ ٱلْمُعْتَبِينَ ﴿٢٤﴾ And though they are resigned, yet the Fire is still their home; and if they ask for favour, yet they are not of those unto whom favour can be shown. (41:24) || فَإِن يَصْبِرُوا۟ فَٱلنَّارُ مَثْوًۭى لَّهُمْ ۖ وَإِن يَسْتَعْتِبُوا۟ فَمَا هُم مِّنَ ٱلْمُعْتَبِينَ ﴿٢٤﴾ بیا که صبر وکړي، نو هم یې ځای دوزخ دی؛ او که توبه کول وغواړي، نو د توبې موقع به ورنه کړل شي.
p ۞ وَقَيَّضْنَا لَهُمْ قُرَنَآءَ فَزَيَّنُوا۟ لَهُم مَّا بَيْنَ أَيْدِيهِمْ وَمَا خَلْفَهُمْ وَحَقَّ عَلَيْهِمُ ٱلْقَوْلُ فِىٓ أُمَمٍۢ قَدْ خَلَتْ مِن قَبْلِهِم مِّنَ ٱلْجِنِّ وَٱلْإِنسِ ۖ إِنَّهُمْ كَانُوا۟ خَٰسِرِينَ ﴿٢٥﴾ اور ہم نے ان کے لیے کچھ ساتھی مقرر کر دیے ہیں تو انھوں نے ان کے لیے ان کے آگے پیچھے (کے برے اعمال) کو خوش نما کر دکھایا اور ان پر (عذاب کا) فیصلہ ثابت ہو گیا ان قوموں کی طرح جو ان سے پہلے گزر چکی تھیں جنات اور انسانوں میں سے بے شک وہ نقصان اٹھانے والے تھے۔ || ۞ وَقَيَّضْنَا لَهُمْ قُرَنَآءَ فَزَيَّنُوا۟ لَهُم مَّا بَيْنَ أَيْدِيهِمْ وَمَا خَلْفَهُمْ وَحَقَّ عَلَيْهِمُ ٱلْقَوْلُ فِىٓ أُمَمٍۢ قَدْ خَلَتْ مِن قَبْلِهِم مِّنَ ٱلْجِنِّ وَٱلْإِنسِ ۖ إِنَّهُمْ كَانُوا۟ خَٰسِرِينَ ﴿٢٥﴾ And We assigned them comrades (in the world), who made their present and their past fairseeming unto them. And the Word concerning nations of the jinn and humankind who passed away before them hath effect for them. Lo! they were ever losers. (41:25) || ۞ وَقَيَّضْنَا لَهُمْ قُرَنَآءَ فَزَيَّنُوا۟ لَهُم مَّا بَيْنَ أَيْدِيهِمْ وَمَا خَلْفَهُمْ وَحَقَّ عَلَيْهِمُ ٱلْقَوْلُ فِىٓ أُمَمٍۢ قَدْ خَلَتْ مِن قَبْلِهِم مِّنَ ٱلْجِنِّ وَٱلْإِنسِ ۖ إِنَّهُمْ كَانُوا۟ خَٰسِرِينَ ﴿٢٥﴾ او موږ د هغوی لپاره ځینې ملګري ټاکلي، نو هغوی د هغوی مخکې او وروسته (بد عملونه) ورته ښکلي ښکاره کړل، او پر هغوی د (عذاب) پرېکړه ثابته شوه لکه هغه قومونه چې له دوی مخکې له پېریانو او انسانانو تېر شوي وو؛ بې‌شکه هغوی زیانمن وو.
p وَقَالَ ٱلَّذِينَ كَفَرُوا۟ لَا تَسْمَعُوا۟ لِهَٰذَا ٱلْقُرْءَانِ وَٱلْغَوْا۟ فِيهِ لَعَلَّكُمْ تَغْلِبُونَ ﴿٢٦﴾ اور کافروں نے کہا اس قرآن کو مت سنو اور اس میں شور مچایا کرو (جب یہ سنایا جائے) تاکہ تم غالب رہو۔ || وَقَالَ ٱلَّذِينَ كَفَرُوا۟ لَا تَسْمَعُوا۟ لِهَٰذَا ٱلْقُرْءَانِ وَٱلْغَوْا۟ فِيهِ لَعَلَّكُمْ تَغْلِبُونَ ﴿٢٦﴾ Those who disbelieve say: Heed not this Qur'an, and drown the hearing of it; haply ye may conquer. (41:26) || وَقَالَ ٱلَّذِينَ كَفَرُوا۟ لَا تَسْمَعُوا۟ لِهَٰذَا ٱلْقُرْءَانِ وَٱلْغَوْا۟ فِيهِ لَعَلَّكُمْ تَغْلِبُونَ ﴿٢٦﴾ او کافرانو وویل: دا قرآن مه اورئ او په هغه کې شور ماشور جوړ کړئ (کله چې اورول کېږي) تر څو تاسو غالب پاتې شئ.
