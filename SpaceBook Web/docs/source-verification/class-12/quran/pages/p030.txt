p وَيَقُولُونَ طَاعَةٌۭ فَإِذَا بَرَزُوا۟ مِنْ عِندِكَ بَيَّتَ طَآئِفَةٌۭ مِّنْهُمْ غَيْرَ ٱلَّذِى تَقُولُ ۖ وَٱللَّهُ يَكْتُبُ مَا يُبَيِّتُونَ ۖ فَأَعْرِضْ عَنْهُمْ وَتَوَكَّلْ عَلَى ٱللَّهِ ۚ وَكَفَىٰ بِٱللَّهِ وَكِيلًا ﴿٨١﴾ اور وہ (منافق آپ کے سامنے تو) اطاعت کا کہتے ہیں پھر جب وہ آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کے پاس سے (اٹھ کر) باہر جاتے ہیں تو اُن میں سے ایک گروہ رات کے وقت مشورہ کرتا ہے اُس کے خلاف جو آپ کہتے ہیں اور جو مشورہ وہ رات کو کرتے ہیں اللہ اُسے لکھ رہا ہے تو آپ اُن کو نظر انداز کریں اور اللہ پر بھروسہ رکھیں اور اللہ ہی (بطور) کارساز کافی ہے۔ || وَيَقُولُونَ طَاعَةٌۭ فَإِذَا بَرَزُوا۟ مِنْ عِندِكَ بَيَّتَ طَآئِفَةٌۭ مِّنْهُمْ غَيْرَ ٱلَّذِى تَقُولُ ۖ وَٱللَّهُ يَكْتُبُ مَا يُبَيِّتُونَ ۖ فَأَعْرِضْ عَنْهُمْ وَتَوَكَّلْ عَلَى ٱللَّهِ ۚ وَكَفَىٰ بِٱللَّهِ وَكِيلًا ﴿٨١﴾ And they say: (It is) obedience; but when they have gone forth from thee a party of them spend the night in planning other than what thou sayest. Allah recordeth what they plan by night. So oppose them and put thy trust in Allah. Allah is sufficient as Trustee. (4:81) || وَيَقُولُونَ طَاعَةٌۭ فَإِذَا بَرَزُوا۟ مِنْ عِندِكَ بَيَّتَ طَآئِفَةٌۭ مِّنْهُمْ غَيْرَ ٱلَّذِى تَقُولُ ۖ وَٱللَّهُ يَكْتُبُ مَا يُبَيِّتُونَ ۖ فَأَعْرِضْ عَنْهُمْ وَتَوَكَّلْ عَلَى ٱللَّهِ ۚ وَكَفَىٰ بِٱللَّهِ وَكِيلًا ﴿٨١﴾ او هغوی (منافقان ستاسو مخې ته خو) د اطاعت خبره کوي، بیا چې ستاسو (صلی الله علیه وعلی آله واصحابه وسلم) له خوا (پاڅېږي او) ووځي، نو له هغوی یوه ډله د شپې د هغه څه پر خلاف مشوره کوي چې تاسو یې وایئ؛ او هغه مشوره چې دوی یې د شپې کوي، الله یې لیکي؛ نو تاسو ترې سترګې پټې کړئ او پر الله توکل وکړئ، او الله د کارساز په توګه بس دی.
p أَفَلَا يَتَدَبَّرُونَ ٱلْقُرْءَانَ ۚ وَلَوْ كَانَ مِنْ عِندِ غَيْرِ ٱللَّهِ لَوَجَدُوا۟ فِيهِ ٱخْتِلَٰفًۭا كَثِيرًۭا ﴿٨٢﴾ تو کیا وہ قرآن میں غور نہیں کرتے؟ اور اگر یہ اللہ کے سوا کسی اور کی طرف سے ہوتا تو یقیناً وہ اس میں بہت اختلاف پاتے۔ || أَفَلَا يَتَدَبَّرُونَ ٱلْقُرْءَانَ ۚ وَلَوْ كَانَ مِنْ عِندِ غَيْرِ ٱللَّهِ لَوَجَدُوا۟ فِيهِ ٱخْتِلَٰفًۭا كَثِيرًۭا ﴿٨٢﴾ Will they not then ponder on the Qur'an? If it had been from other than Allah they would have found therein much incongruity. (4:82) || أَفَلَا يَتَدَبَّرُونَ ٱلْقُرْءَانَ ۚ وَلَوْ كَانَ مِنْ عِندِ غَيْرِ ٱللَّهِ لَوَجَدُوا۟ فِيهِ ٱخْتِلَٰفًۭا كَثِيرًۭا ﴿٨٢﴾ نو ایا هغوی په قرآن کې غور نه کوي؟ او که دا له الله پرته د بل چا له لوري وای، نو یقیناً به یې پکې ډېر اختلاف موندلی وای.
p وَإِذَا جَآءَهُمْ أَمْرٌۭ مِّنَ ٱلْأَمْنِ أَوِ ٱلْخَوْفِ أَذَاعُوا۟ بِهِۦ ۖ وَلَوْ رَدُّوهُ إِلَى ٱلرَّسُولِ وَإِلَىٰٓ أُو۟لِى ٱلْأَمْرِ مِنْهُمْ لَعَلِمَهُ ٱلَّذِينَ يَسْتَنۢبِطُونَهُۥ مِنْهُمْ ۗ وَلَوْلَا فَضْلُ ٱللَّهِ عَلَيْكُمْ وَرَحْمَتُهُۥ لَٱتَّبَعْتُمُ ٱلشَّيْطَٰنَ إِلَّا قَلِيلًۭا ﴿٨٣﴾ اور جب اُن کے پاس امن یا خوف کی کوئی خبر آتی ہے تو اُسے مشہور کر دیتے ہیں اور اگر وہ اُسے رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کی طرف اور اپنے میں سے اصحابِ اختیار تک پہنچا دیتے تو یقیناً اُن میں سے تحقیق کرنے والے اُس (خبر) کی حقیقت کو جان لیتے اور اگر تم پر اللہ کا فضل اور اُس کی رحمت نہ ہوتی تو یقیناً چند ایک کے سوا تم (سب) شیطان کی پیروی کرنے لگ جاتے۔ || وَإِذَا جَآءَهُمْ أَمْرٌۭ مِّنَ ٱلْأَمْنِ أَوِ ٱلْخَوْفِ أَذَاعُوا۟ بِهِۦ ۖ وَلَوْ رَدُّوهُ إِلَى ٱلرَّسُولِ وَإِلَىٰٓ أُو۟لِى ٱلْأَمْرِ مِنْهُمْ لَعَلِمَهُ ٱلَّذِينَ يَسْتَنۢبِطُونَهُۥ مِنْهُمْ ۗ وَلَوْلَا فَضْلُ ٱللَّهِ عَلَيْكُمْ وَرَحْمَتُهُۥ لَٱتَّبَعْتُمُ ٱلشَّيْطَٰنَ إِلَّا قَلِيلًۭا ﴿٨٣﴾ And if any tidings, whether of safety or fear, come unto them, they noise it abroad, whereas if they had referred it to the messenger and to such of them as are in authority, those among them who are able to think out the matter would have known it. If it had not been for the grace of Allah upon you and His mercy ye would have followed Satan, save a few (of you). (4:83) || وَإِذَا جَآءَهُمْ أَمْرٌۭ مِّنَ ٱلْأَمْنِ أَوِ ٱلْخَوْفِ أَذَاعُوا۟ بِهِۦ ۖ وَلَوْ رَدُّوهُ إِلَى ٱلرَّسُولِ وَإِلَىٰٓ أُو۟لِى ٱلْأَمْرِ مِنْهُمْ لَعَلِمَهُ ٱلَّذِينَ يَسْتَنۢبِطُونَهُۥ مِنْهُمْ ۗ وَلَوْلَا فَضْلُ ٱللَّهِ عَلَيْكُمْ وَرَحْمَتُهُۥ لَٱتَّبَعْتُمُ ٱلشَّيْطَٰنَ إِلَّا قَلِيلًۭا ﴿٨٣﴾ او کله چې هغوی ته د امن یا وېرې کوم خبر راشي، نو خپور یې کړي؛ او که هغوی دا رسول (صلی الله علیه وعلی آله واصحابه وسلم) او له ځانونو د واک خاوندانو ته رسولی وای، نو یقیناً به له هغوی څخه څېړونکو د هغه (خبر) حقیقت پېژندلی وای؛ او که پر تاسو د الله فضل او د هغه رحمت نه وای، نو یقیناً به له څو تنو پرته تاسو (ټولو) د شیطان پیروي پیل کړې وای.
p فَقَٰتِلْ فِى سَبِيلِ ٱللَّهِ لَا تُكَلَّفُ إِلَّا نَفْسَكَ ۚ وَحَرِّضِ ٱلْمُؤْمِنِينَ ۖ عَسَى ٱللَّهُ أَن يَكُفَّ بَأْسَ ٱلَّذِينَ كَفَرُوا۟ ۚ وَٱللَّهُ أَشَدُّ بَأْسًۭا وَأَشَدُّ تَنكِيلًۭا ﴿٨٤﴾ تو آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) اللہ کی راہ میں قتال کیجیے آپ پر صرف اپنی ذات کی ذمہ داری ڈالی جاتی ہے اور مومنوں کو ترغیب دلایے قریب ہے کہ جنھوں نے کفر کیا ہے اللہ اُن کی جنگ کا زور روک دے اور اللہ پکڑ کرنے میں بہت سخت ہے اور سزا دینے میں (بھی) بہت سخت ہے۔ || فَقَٰتِلْ فِى سَبِيلِ ٱللَّهِ لَا تُكَلَّفُ إِلَّا نَفْسَكَ ۚ وَحَرِّضِ ٱلْمُؤْمِنِينَ ۖ عَسَى ٱللَّهُ أَن يَكُفَّ بَأْسَ ٱلَّذِينَ كَفَرُوا۟ ۚ وَٱللَّهُ أَشَدُّ بَأْسًۭا وَأَشَدُّ تَنكِيلًۭا ﴿٨٤﴾ So fight (O Muhammad) in the way of Allah Thou art not taxed (with the responsibility for anyone) except thyself - and urge on the believers. Peradventure Allah will restrain the might of those who disbelieve. Allah is stronger in might and stronger in inflicting punishment. (4:84) || فَقَٰتِلْ فِى سَبِيلِ ٱللَّهِ لَا تُكَلَّفُ إِلَّا نَفْسَكَ ۚ وَحَرِّضِ ٱلْمُؤْمِنِينَ ۖ عَسَى ٱللَّهُ أَن يَكُفَّ بَأْسَ ٱلَّذِينَ كَفَرُوا۟ ۚ وَٱللَّهُ أَشَدُّ بَأْسًۭا وَأَشَدُّ تَنكِيلًۭا ﴿٨٤﴾ نو تاسو (صلی الله علیه وعلی آله واصحابه وسلم) د الله په لار کې جګړه وکړئ؛ پر تاسو یوازې د خپل ځان مسؤلیت اچول کېږي؛ او مؤمنان وهڅوئ؛ نږدې ده چې الله د کافرانو د جګړې زور ودروي؛ او الله په نیونه کې ډېر سخت دی او په سزا ورکولو کې (هم) ډېر سخت دی.
p مَّن يَشْفَعْ شَفَٰعَةً حَسَنَةًۭ يَكُن لَّهُۥ نَصِيبٌۭ مِّنْهَا ۖ وَمَن يَشْفَعْ شَفَٰعَةًۭ سَيِّئَةًۭ يَكُن لَّهُۥ كِفْلٌۭ مِّنْهَا ۗ وَكَانَ ٱللَّهُ عَلَىٰ كُلِّ شَىْءٍۢ مُّقِيتًۭا ﴿٨٥﴾ جو اچھی سفارش کرے گا اُس میں سے اُس کا حصہ ہو گا اور جو بُری سفارش کرے گا اُس کے لیے اُس میں سے بوجھ ہو گا اور اللہ ہر چیز پر قدرت رکھنے والا ہے۔ || مَّن يَشْفَعْ شَفَٰعَةً حَسَنَةًۭ يَكُن لَّهُۥ نَصِيبٌۭ مِّنْهَا ۖ وَمَن يَشْفَعْ شَفَٰعَةًۭ سَيِّئَةًۭ يَكُن لَّهُۥ كِفْلٌۭ مِّنْهَا ۗ وَكَانَ ٱللَّهُ عَلَىٰ كُلِّ شَىْءٍۢ مُّقِيتًۭا ﴿٨٥﴾ Whoso interveneth in a good cause will have the reward thereof, and whoso interveneth in an evil cause will bear the consequence thereof. Allah overseeth all things. (4:85) || مَّن يَشْفَعْ شَفَٰعَةً حَسَنَةًۭ يَكُن لَّهُۥ نَصِيبٌۭ مِّنْهَا ۖ وَمَن يَشْفَعْ شَفَٰعَةًۭ سَيِّئَةًۭ يَكُن لَّهُۥ كِفْلٌۭ مِّنْهَا ۗ وَكَانَ ٱللَّهُ عَلَىٰ كُلِّ شَىْءٍۢ مُّقِيتًۭا ﴿٨٥﴾ څوک چې ښه سپارښتنه وکړي، په هغې کې به یې برخه وي، او څوک چې بده سپارښتنه وکړي، په هغې کې به یې بار وي؛ او الله پر هر شي قدرت لرونکی دی.
p وَإِذَا حُيِّيتُم بِتَحِيَّةٍۢ فَحَيُّوا۟ بِأَحْسَنَ مِنْهَآ أَوْ رُدُّوهَآ ۗ إِنَّ ٱللَّهَ كَانَ عَلَىٰ كُلِّ شَىْءٍ حَسِيبًا ﴿٨٦﴾ جب تمھیں سلام کیا جائے تو تم اُس سلام کا بہتر جواب دو یا اُسی کو لوٹا دو بے شک اللہ ہر چیز کا حساب لینے والا ہے۔ || وَإِذَا حُيِّيتُم بِتَحِيَّةٍۢ فَحَيُّوا۟ بِأَحْسَنَ مِنْهَآ أَوْ رُدُّوهَآ ۗ إِنَّ ٱللَّهَ كَانَ عَلَىٰ كُلِّ شَىْءٍ حَسِيبًا ﴿٨٦﴾ When ye are greeted with a greeting, greet ye with a better than it or return it. Lo! Allah taketh count of all things. (4:86) || وَإِذَا حُيِّيتُم بِتَحِيَّةٍۢ فَحَيُّوا۟ بِأَحْسَنَ مِنْهَآ أَوْ رُدُّوهَآ ۗ إِنَّ ٱللَّهَ كَانَ عَلَىٰ كُلِّ شَىْءٍ حَسِيبًا ﴿٨٦﴾ کله چې تاسو ته سلام وشي، نو د هغه سلام غوره ځواب ورکړئ یا هماغه بېرته ورکړئ؛ بې‌شکه الله د هر شي حساب اخیستونکی دی.
p ٱللَّهُ لَآ إِلَٰهَ إِلَّا هُوَ ۚ لَيَجْمَعَنَّكُمْ إِلَىٰ يَوْمِ ٱلْقِيَٰمَةِ لَا رَيْبَ فِيهِ ۗ وَمَنْ أَصْدَقُ مِنَ ٱللَّهِ حَدِيثًۭا ﴿٨٧﴾ اللہ (وہی ہے) جس کے سوا کوئی عبادت کے لائق نہیں وہ تمھیں قیامت کے دن ضرور جمع کرے گا جس (کے آنے) میں کوئی شک نہیں اور کون ہے جو اللہ سے زیادہ بات کا سچا ہو؟ || ٱللَّهُ لَآ إِلَٰهَ إِلَّا هُوَ ۚ لَيَجْمَعَنَّكُمْ إِلَىٰ يَوْمِ ٱلْقِيَٰمَةِ لَا رَيْبَ فِيهِ ۗ وَمَنْ أَصْدَقُ مِنَ ٱللَّهِ حَدِيثًۭا ﴿٨٧﴾ Allah! There is no Allah save Him. He gathereth you all unto a Day of Resurrection whereof there is no doubt. Who is more true in statement than Allah? (4:87) || ٱللَّهُ لَآ إِلَٰهَ إِلَّا هُوَ ۚ لَيَجْمَعَنَّكُمْ إِلَىٰ يَوْمِ ٱلْقِيَٰمَةِ لَا رَيْبَ فِيهِ ۗ وَمَنْ أَصْدَقُ مِنَ ٱللَّهِ حَدِيثًۭا ﴿٨٧﴾ الله (هماغه دی) چې له هغه پرته د عبادت وړ هېڅوک نشته؛ هغه به تاسو خامخا د قیامت په ورځ راټول کړي چې (په راتلو) کې یې هېڅ شک نشته؛ او څوک دی چې له الله په خبره کې رښتینی وي؟
