p أَعَدَّ ٱللَّهُ لَهُمْ عَذَابًۭا شَدِيدًۭا ۖ فَٱتَّقُوا۟ ٱللَّهَ يَٰٓأُو۟لِى ٱلْأَلْبَٰبِ ٱلَّذِينَ ءَامَنُوا۟ ۚ قَدْ أَنزَلَ ٱللَّهُ إِلَيْكُمْ ذِكْرًۭا ﴿١٠﴾ (آخرت میں) اللہ نے اُن کے لیے سخت عذاب تیار کر رکھا ہے تو اللہ (کی نافرمانی) سے ڈرو اے عقل مندو! جو ایمان لائے ہو یقیناً اللہ نے تمھاری طرف (ایک) ذکر (قرآنِ پاک) نازل فرمایا ہے۔ || أَعَدَّ ٱللَّهُ لَهُمْ عَذَابًۭا شَدِيدًۭا ۖ فَٱتَّقُوا۟ ٱللَّهَ يَٰٓأُو۟لِى ٱلْأَلْبَٰبِ ٱلَّذِينَ ءَامَنُوا۟ ۚ قَدْ أَنزَلَ ٱللَّهُ إِلَيْكُمْ ذِكْرًۭا ﴿١٠﴾ Allah hath prepared for them stern punishment; so keep your duty to Allah, O men of understanding! O ye who believe! Now Allah hath sent down unto you a reminder, (65:10) || أَعَدَّ ٱللَّهُ لَهُمْ عَذَابًۭا شَدِيدًۭا ۖ فَٱتَّقُوا۟ ٱللَّهَ يَٰٓأُو۟لِى ٱلْأَلْبَٰبِ ٱلَّذِينَ ءَامَنُوا۟ ۚ قَدْ أَنزَلَ ٱللَّهُ إِلَيْكُمْ ذِكْرًۭا ﴿١٠﴾ (په آخرت کې) الله د هغوی لپاره سخت عذاب تیار کړی؛ نو له الله (د نافرمانۍ) ووېرېږئ، ای عقلمندانو چې ایمان مو راوړی! یقیناً الله تاسو ته (یو) ذکر (پاک قرآن) نازل کړی.
p رَّسُولًۭا يَتْلُوا۟ عَلَيْكُمْ ءَايَٰتِ ٱللَّهِ مُبَيِّنَٰتٍۢ لِّيُخْرِجَ ٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّٰلِحَٰتِ مِنَ ٱلظُّلُمَٰتِ إِلَى ٱلنُّورِ ۚ وَمَن يُؤْمِنۢ بِٱللَّهِ وَيَعْمَلْ صَٰلِحًۭا يُدْخِلْهُ جَنَّٰتٍۢ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَٰرُ خَٰلِدِينَ فِيهَآ أَبَدًۭا ۖ قَدْ أَحْسَنَ ٱللَّهُ لَهُۥ رِزْقًا ﴿١١﴾ (اور) رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم کو بھیجا) جو تم پر اللہ کی واضح آیات تلاوت کرتے ہیں تاکہ اُن لوگوں کو جو ایمان لائے اور جنھوں نے نیک اعمال کیے اندھیروں سے نکال کر روشنی کی طرف لے آئیں اور جو شخص اللہ پر ایمان لائے اور نیک عمل کرے وہ اُسے اُن جنتوں میں داخل فرمائے گا جن کے نیچے نہریں بہتی ہیں (وہ) ان میں ہمیشہ رہیں گے یقیناً اللہ نے اُس کے لیے بہترین رزق طے فرما دیا ہے۔ || رَّسُولًۭا يَتْلُوا۟ عَلَيْكُمْ ءَايَٰتِ ٱللَّهِ مُبَيِّنَٰتٍۢ لِّيُخْرِجَ ٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّٰلِحَٰتِ مِنَ ٱلظُّلُمَٰتِ إِلَى ٱلنُّورِ ۚ وَمَن يُؤْمِنۢ بِٱللَّهِ وَيَعْمَلْ صَٰلِحًۭا يُدْخِلْهُ جَنَّٰتٍۢ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَٰرُ خَٰلِدِينَ فِيهَآ أَبَدًۭا ۖ قَدْ أَحْسَنَ ٱللَّهُ لَهُۥ رِزْقًا ﴿١١﴾ A messenger reciting unto you the revelations of Allah made plain, that He may bring forth those who believe and do good works from darkness unto light. And whosoever believeth in Allah and doeth right, He will bring him into Gardens underneath which rivers flow, therein to abide for ever. Allah hath made good provision for him. (65:11) || رَّسُولًۭا يَتْلُوا۟ عَلَيْكُمْ ءَايَٰتِ ٱللَّهِ مُبَيِّنَٰتٍۢ لِّيُخْرِجَ ٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّٰلِحَٰتِ مِنَ ٱلظُّلُمَٰتِ إِلَى ٱلنُّورِ ۚ وَمَن يُؤْمِنۢ بِٱللَّهِ وَيَعْمَلْ صَٰلِحًۭا يُدْخِلْهُ جَنَّٰتٍۢ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَٰرُ خَٰلِدِينَ فِيهَآ أَبَدًۭا ۖ قَدْ أَحْسَنَ ٱللَّهُ لَهُۥ رِزْقًا ﴿١١﴾ (او) رسول (صلی الله علیه وعلی آله واصحابه وسلم یې ولېږه) چې پر تاسو د الله روښانه آیتونه تلاوتوي، تر څو هغه کسان چې ایمان یې راوړی او نېک عملونه یې کړي، له تیارو راوباسي او رڼا ته یې راولي؛ او څوک چې پر الله ایمان راوړي او نېک عمل وکړي، هغه به یې په هغو جنتونو کې ننباسي چې لاندې یې ویالې بهېږي، (هغوی به) پکې تل پاتې وي؛ یقیناً الله د هغه لپاره غوره رزق ټاکلی دی.
p ٱللَّهُ ٱلَّذِى خَلَقَ سَبْعَ سَمَٰوَٰتٍۢ وَمِنَ ٱلْأَرْضِ مِثْلَهُنَّ يَتَنَزَّلُ ٱلْأَمْرُ بَيْنَهُنَّ لِتَعْلَمُوٓا۟ أَنَّ ٱللَّهَ عَلَىٰ كُلِّ شَىْءٍۢ قَدِيرٌۭ وَأَنَّ ٱللَّهَ قَدْ أَحَاطَ بِكُلِّ شَىْءٍ عِلْمًۢا ﴿١٢﴾ اللہ (ہی) ہے جس نے سات آسمان پیدا فرمائے اور ان ہی کی طرح اتنی ہی زمینیں بھی (بنائی ہیں) (اللہ کا) حکم اُن کے درمیان نازل ہوتا رہتا ہے تاکہ تم جان لو بے شک اللہ ہر چیز پر پوری طرح قادر ہے اور بے شک اللہ نے ہر چیز کا (اپنے) علم سے احاطہ فرما رکھا ہے۔ || ٱللَّهُ ٱلَّذِى خَلَقَ سَبْعَ سَمَٰوَٰتٍۢ وَمِنَ ٱلْأَرْضِ مِثْلَهُنَّ يَتَنَزَّلُ ٱلْأَمْرُ بَيْنَهُنَّ لِتَعْلَمُوٓا۟ أَنَّ ٱللَّهَ عَلَىٰ كُلِّ شَىْءٍۢ قَدِيرٌۭ وَأَنَّ ٱللَّهَ قَدْ أَحَاطَ بِكُلِّ شَىْءٍ عِلْمًۢا ﴿١٢﴾ Allah it is who hath created seven heavens, and of the earth the like thereof. The commandment cometh down among them slowly, that ye may know that Allah is Able to do all things, and that Allah surroundeth all things in knowledge. (65:12) || ٱللَّهُ ٱلَّذِى خَلَقَ سَبْعَ سَمَٰوَٰتٍۢ وَمِنَ ٱلْأَرْضِ مِثْلَهُنَّ يَتَنَزَّلُ ٱلْأَمْرُ بَيْنَهُنَّ لِتَعْلَمُوٓا۟ أَنَّ ٱللَّهَ عَلَىٰ كُلِّ شَىْءٍۢ قَدِيرٌۭ وَأَنَّ ٱللَّهَ قَدْ أَحَاطَ بِكُلِّ شَىْءٍ عِلْمًۢا ﴿١٢﴾ یوازې الله دی چې اووه آسمانونه یې پیدا کړل، او د هغو په څېر هومره ځمکې یې هم (جوړې کړې)؛ د (الله) حکم د هغو ترمنځ نازلېږي تر څو تاسو پوه شئ چې بې‌شکه الله پر هر شي پوره قادر دی، او بې‌شکه الله پر هر شي په (خپل) علم احاطه کړې ده.
## علم و عمل کی باتیں || Matters of knowledge and action || د علم او عمل خبرې
- جو شخص تقویٰ اختیار کرتا ہے تو اللہ تعالیٰ اسے پانچ برکات عطا فرماتا ہے: ۱۔ مشکلات سے نکلنے کی راہ عطا فرماتا ہے۔ ۲۔ وہاں سے رزق عطا فرماتا ہے جہاں سے اس کا گمان بھی نہیں ہوتا۔ ۳۔ اس کے لیے اس کے کام میں آسانی پیدا فرما دیتا ہے۔ ۴۔ اس کے گناہ معاف فرما دیتا ہے۔ ۵۔ اس کا اجر و ثواب بڑھا دیتا ہے۔ || Whoever adopts piety, Allah Almighty grants him five blessings: (1) He grants him a way out of difficulties; (2) He grants him provision from where he does not even imagine; (3) He creates ease for him in his affairs; (4) He forgives his sins; (5) He increases his reward and recompense. || څوک چې تقوا غوره کړي، الله تعالی ورته پنځه برکتونه ورکوي: ۱. له ستونزو د وتلو لار ورکوي؛ ۲. له داسې ځایه رزق ورکوي چې ګومان یې هم نه کوي؛ ۳. د هغه لپاره یې په کار کې اساني پیدا کوي؛ ۴. ګناهونه یې بښي؛ ۵. اجر او ثواب یې زیاتوي.
- ہمارا دین زندگی کے ہر مرحلہ کے لیے رہنمائی عطا فرماتا ہے۔ طلاق جیسے ناخوش گوار اور تلخ معاملہ کے بارے میں بھی خصوصی ہدایات عطا کی گئی ہیں اور معاملات کو خوش اُسلوبی سے پایۂ تکمیل تک پہنچانے کا حکم دیا گیا ہے۔ || Our religion grants guidance for every stage of life. Special instructions have been given even about a matter as unpleasant and bitter as divorce, and the command has been given to bring matters to completion in a graceful manner. || زموږ دین د ژوند هر پړاو ته لارښوونه کوي؛ حتی د طلاق په څېر د ناخوښې او ترخې چارې په اړه هم ځانګړې لارښوونې ورکړل شوې، او حکم شوی چې چارې په ښه توګه بشپړې کړای شي.
- عائلی قوانین اور ازدواجی زندگی کے بارے میں قرآنِ حکیم اور نبی کریم صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم کے عطا کردہ قواعد و ضوابط پر عمل پیرا ہو کر اُمتِ مسلمہ گمراہی کے ان اندھیروں سے بچ سکتی ہے جن اندھیروں میں وحیِ الٰہی کی تعلیم کو فراموش کرنے والی اقوام بھٹکتی پھر رہی ہیں۔ || By acting upon the rules and regulations granted by the Qur'an-e-Hakim and the Holy Prophet (peace and blessings of Allah be upon him) concerning family laws and married life, the Muslim Ummah can be saved from those darknesses of misguidance in which the nations that forgot the teaching of divine revelation are wandering. || د کورنیو قوانینو او ازدواجي ژوند په اړه د قرآن حکیم او نبي کریم صلی الله علیه وعلی آله واصحابه وسلم پر ورکړل شویو قواعدو او مقرراتو په عمل سره مسلمه امت د ګمراهۍ له هغو تیارو ژغورل کېدای شي چې د الهي وحې ښوونه هېروونکي قومونه پکې سرګردانه ګرځي.
