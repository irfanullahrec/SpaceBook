p يَٰٓأَهْلَ ٱلْكِتَٰبِ قَدْ جَآءَكُمْ رَسُولُنَا يُبَيِّنُ لَكُمْ عَلَىٰ فَتْرَةٍۢ مِّنَ ٱلرُّسُلِ أَن تَقُولُوا۟ مَا جَآءَنَا مِنۢ بَشِيرٍۢ وَلَا نَذِيرٍۢ ۖ فَقَدْ جَآءَكُم بَشِيرٌۭ وَنَذِيرٌۭ ۗ وَٱللَّهُ عَلَىٰ كُلِّ شَىْءٍۢ قَدِيرٌۭ ﴿١٩﴾ اے اہلِ کتاب! تمھارے پاس ہمارے (یہ) رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) آ چکے ہیں وہ تم پر (احکامِ شریعت) کھول کھول کر بیان فرماتے ہیں بعد اس کے کہ رسولوں کی آمد ایک مدت تک رُکی رہی تھی کہ کہیں تم یہ نہ کہہ سکو ہمارے پاس نہ خوش خبری دینے والا آیا اور نہ ہی ڈرانے والا تو یقیناً تمھارے پاس خوش خبری دینے والے اور ڈرانے والے (رسول کریم صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) آ چکے ہیں اور اللہ ہر چیز پر خوب قدرت رکھنے والا ہے۔ || يَٰٓأَهْلَ ٱلْكِتَٰبِ قَدْ جَآءَكُمْ رَسُولُنَا يُبَيِّنُ لَكُمْ عَلَىٰ فَتْرَةٍۢ مِّنَ ٱلرُّسُلِ أَن تَقُولُوا۟ مَا جَآءَنَا مِنۢ بَشِيرٍۢ وَلَا نَذِيرٍۢ ۖ فَقَدْ جَآءَكُم بَشِيرٌۭ وَنَذِيرٌۭ ۗ وَٱللَّهُ عَلَىٰ كُلِّ شَىْءٍۢ قَدِيرٌۭ ﴿١٩﴾ O People of the Scripture! Now hath Our messenger come unto you to make things plain unto you after an interval (of cessation) of the messengers, lest ye should say: There came not unto us a messenger of cheer nor any warner. Now hath a messenger of cheer and a warner come unto you. Allah is Able to do all things. (5:19) || يَٰٓأَهْلَ ٱلْكِتَٰبِ قَدْ جَآءَكُمْ رَسُولُنَا يُبَيِّنُ لَكُمْ عَلَىٰ فَتْرَةٍۢ مِّنَ ٱلرُّسُلِ أَن تَقُولُوا۟ مَا جَآءَنَا مِنۢ بَشِيرٍۢ وَلَا نَذِيرٍۢ ۖ فَقَدْ جَآءَكُم بَشِيرٌۭ وَنَذِيرٌۭ ۗ وَٱللَّهُ عَلَىٰ كُلِّ شَىْءٍۢ قَدِيرٌۭ ﴿١٩﴾ ای اهل کتابو! تاسو ته زموږ (دا) رسول (صلی الله علیه وعلی آله واصحابه وسلم) راغلی چې تاسو ته (د شریعت حکمونه) ښه روښانه بیانوي، وروسته له دې چې د رسولانو راتګ یوه موده بند پاتې و، تر څو داسې ونه وایئ چې موږ ته نه زېری ورکوونکی راغی او نه وېرونکی؛ نو یقیناً تاسو ته زېری ورکوونکی او وېرونکی (رسول کریم صلی الله علیه وعلی آله واصحابه وسلم) راغلی؛ او الله پر هر شي ښه قدرت لرونکی دی.
p وَإِذْ قَالَ مُوسَىٰ لِقَوْمِهِۦ يَٰقَوْمِ ٱذْكُرُوا۟ نِعْمَةَ ٱللَّهِ عَلَيْكُمْ إِذْ جَعَلَ فِيكُمْ أَنۢبِيَآءَ وَجَعَلَكُم مُّلُوكًۭا وَءَاتَىٰكُم مَّا لَمْ يُؤْتِ أَحَدًۭا مِّنَ ٱلْعَٰلَمِينَ ﴿٢٠﴾ اور جب موسیٰ (علیہ السلام) نے اپنی قوم سے فرمایا اے میری قوم! اپنے اوپر اللہ کی نعمت کو یاد کرو جب اُس نے تم میں انبیاء (علیہم السلام) پیدا فرمائے اور تمھیں بادشاہ بنایا اور اُس نے تمھیں وہ کچھ عطا فرمایا جو (تم سے پہلے) تمام دنیا جہان میں کسی کو نہیں دیا تھا۔ || وَإِذْ قَالَ مُوسَىٰ لِقَوْمِهِۦ يَٰقَوْمِ ٱذْكُرُوا۟ نِعْمَةَ ٱللَّهِ عَلَيْكُمْ إِذْ جَعَلَ فِيكُمْ أَنۢبِيَآءَ وَجَعَلَكُم مُّلُوكًۭا وَءَاتَىٰكُم مَّا لَمْ يُؤْتِ أَحَدًۭا مِّنَ ٱلْعَٰلَمِينَ ﴿٢٠﴾ And (remember) when Moses said unto his people: O my people! Remember Allah's favour unto you, how He placed among you prophets, and He made you kings, and gave you that (which) He gave not to any (other) of (His) creatures. (5:20) || وَإِذْ قَالَ مُوسَىٰ لِقَوْمِهِۦ يَٰقَوْمِ ٱذْكُرُوا۟ نِعْمَةَ ٱللَّهِ عَلَيْكُمْ إِذْ جَعَلَ فِيكُمْ أَنۢبِيَآءَ وَجَعَلَكُم مُّلُوكًۭا وَءَاتَىٰكُم مَّا لَمْ يُؤْتِ أَحَدًۭا مِّنَ ٱلْعَٰلَمِينَ ﴿٢٠﴾ او کله چې موسی (علیه السلام) خپل قوم ته وویل: «ای زما قومه! پر خپل ځان د الله نعمت یاد کړئ، کله چې یې په تاسو کې انبیاء (علیهم السلام) پیدا کړل، تاسو یې پاچاهان کړئ، او تاسو ته یې هغه څه درکړل چې (له تاسو مخکې) یې د نړۍ هېچا ته نه وو ورکړي.»
p يَٰقَوْمِ ٱدْخُلُوا۟ ٱلْأَرْضَ ٱلْمُقَدَّسَةَ ٱلَّتِى كَتَبَ ٱللَّهُ لَكُمْ وَلَا تَرْتَدُّوا۟ عَلَىٰٓ أَدْبَارِكُمْ فَتَنقَلِبُوا۟ خَٰسِرِينَ ﴿٢١﴾ اے میری قوم! اس مقدس سرزمین میں داخل ہو جاؤ جو اللہ نے تمھارے لیے لکھ دی ہے اور اپنی پیٹھ پھیرتے ہوئے پیچھے نہ ہٹنا ورنہ تم خسارہ اٹھانے والے ہو کر پلٹو گے۔ || يَٰقَوْمِ ٱدْخُلُوا۟ ٱلْأَرْضَ ٱلْمُقَدَّسَةَ ٱلَّتِى كَتَبَ ٱللَّهُ لَكُمْ وَلَا تَرْتَدُّوا۟ عَلَىٰٓ أَدْبَارِكُمْ فَتَنقَلِبُوا۟ خَٰسِرِينَ ﴿٢١﴾ O my people! Go into the holy land which Allah hath ordained for you. Turn not in flight, for surely ye turn back as losers: (5:21) || يَٰقَوْمِ ٱدْخُلُوا۟ ٱلْأَرْضَ ٱلْمُقَدَّسَةَ ٱلَّتِى كَتَبَ ٱللَّهُ لَكُمْ وَلَا تَرْتَدُّوا۟ عَلَىٰٓ أَدْبَارِكُمْ فَتَنقَلِبُوا۟ خَٰسِرِينَ ﴿٢١﴾ «ای زما قومه! هغې سپېڅلې ځمکې ته ننوځئ چې الله ستاسو لپاره لیکلې، او شا مه اړوئ او شاته مه کېږئ، ګنې زیانمن به بېرته وګرځئ.»
p قَالُوا۟ يَٰمُوسَىٰٓ إِنَّ فِيهَا قَوْمًۭا جَبَّارِينَ وَإِنَّا لَن نَّدْخُلَهَا حَتَّىٰ يَخْرُجُوا۟ مِنْهَا فَإِن يَخْرُجُوا۟ مِنْهَا فَإِنَّا دَٰخِلُونَ ﴿٢٢﴾ (بنی اسرائیل) نے کہا کہ اے موسیٰ (علیہ السلام)! بے شک اس بستی میں تو سخت جابر لوگ آباد ہیں اور ہم ہرگز اس میں داخل نہ ہوں گے یہاں تک کہ وہ اس سے نکل جائیں پھر اگر وہ اِس سے نکل جائیں تو بے شک ہم داخل ہوں گے۔ || قَالُوا۟ يَٰمُوسَىٰٓ إِنَّ فِيهَا قَوْمًۭا جَبَّارِينَ وَإِنَّا لَن نَّدْخُلَهَا حَتَّىٰ يَخْرُجُوا۟ مِنْهَا فَإِن يَخْرُجُوا۟ مِنْهَا فَإِنَّا دَٰخِلُونَ ﴿٢٢﴾ They said: O Moses! Lo! a giant people (dwell) therein and lo! we go not in till they go forth from thence. When they go forth from thence, then we will enter (not till then). (5:22) || قَالُوا۟ يَٰمُوسَىٰٓ إِنَّ فِيهَا قَوْمًۭا جَبَّارِينَ وَإِنَّا لَن نَّدْخُلَهَا حَتَّىٰ يَخْرُجُوا۟ مِنْهَا فَإِن يَخْرُجُوا۟ مِنْهَا فَإِنَّا دَٰخِلُونَ ﴿٢٢﴾ (بني اسرائیلو) وویل: «ای موسی (علیه السلام)! بې‌شکه په دې کلي کې سخت زورواکي خلک اوسېږي، او موږ به هېڅکله ورننه وځو تر څو چې هغوی ترې ووځي؛ بیا که هغوی ترې ووتل، نو بې‌شکه موږ به ورننوځو.»
p قَالَ رَجُلَانِ مِنَ ٱلَّذِينَ يَخَافُونَ أَنْعَمَ ٱللَّهُ عَلَيْهِمَا ٱدْخُلُوا۟ عَلَيْهِمُ ٱلْبَابَ فَإِذَا دَخَلْتُمُوهُ فَإِنَّكُمْ غَٰلِبُونَ ۚ وَعَلَى ٱللَّهِ فَتَوَكَّلُوٓا۟ إِن كُنتُم مُّؤْمِنِينَ ﴿٢٣﴾ (اللہ سے) ڈرنے والوں میں سے دو مردوں نے کہا جن دونوں پر اللہ نے انعام فرمایا تھا کہ تم ان پر (حملہ کرتے ہوئے شہر کے) دروازے میں داخل ہو جاؤ پس جب تم اس میں داخل ہو جاؤ گے تو بے شک تم ہی غالب رہو گے اور اللہ ہی پر بھروسہ رکھو اگر تم مومن ہو۔ || قَالَ رَجُلَانِ مِنَ ٱلَّذِينَ يَخَافُونَ أَنْعَمَ ٱللَّهُ عَلَيْهِمَا ٱدْخُلُوا۟ عَلَيْهِمُ ٱلْبَابَ فَإِذَا دَخَلْتُمُوهُ فَإِنَّكُمْ غَٰلِبُونَ ۚ وَعَلَى ٱللَّهِ فَتَوَكَّلُوٓا۟ إِن كُنتُم مُّؤْمِنِينَ ﴿٢٣﴾ Then out spake two of those who feared (their Lord, men) unto whom Allah had been gracious: Enter in upon them by the gate, for if ye enter by it, lo! ye will be victorious. So put your trust (in Allah) if ye are indeed believers. (5:23) || قَالَ رَجُلَانِ مِنَ ٱلَّذِينَ يَخَافُونَ أَنْعَمَ ٱللَّهُ عَلَيْهِمَا ٱدْخُلُوا۟ عَلَيْهِمُ ٱلْبَابَ فَإِذَا دَخَلْتُمُوهُ فَإِنَّكُمْ غَٰلِبُونَ ۚ وَعَلَى ٱللَّهِ فَتَوَكَّلُوٓا۟ إِن كُنتُم مُّؤْمِنِينَ ﴿٢٣﴾ له (الله) وېرېدونکو څخه دوو سړیو — چې الله پر دواړو انعام کړی و — وویل: «پر هغوی (په برید) د (ښار) دروازې ته ننوځئ؛ نو چې کله پکې ننوتلئ، بې‌شکه تاسو به غالب یاست؛ او یوازې پر الله توکل وکړئ که مؤمنان یاست.»
p قَالُوا۟ يَٰمُوسَىٰٓ إِنَّا لَن نَّدْخُلَهَآ أَبَدًۭا مَّا دَامُوا۟ فِيهَا ۖ فَٱذْهَبْ أَنتَ وَرَبُّكَ فَقَٰتِلَآ إِنَّا هَٰهُنَا قَٰعِدُونَ ﴿٢٤﴾ انھوں نے کہا اے موسیٰ (علیہ السلام)! بے شک ہم ہرگز اس (سرزمین) میں کبھی بھی داخل نہ ہوں گے جب تک وہ اس میں موجود ہیں پس تم اور تمھارا رب جاؤ پھر دونوں لڑو بے شک ہم تو یہیں بیٹھے ہیں۔ || قَالُوا۟ يَٰمُوسَىٰٓ إِنَّا لَن نَّدْخُلَهَآ أَبَدًۭا مَّا دَامُوا۟ فِيهَا ۖ فَٱذْهَبْ أَنتَ وَرَبُّكَ فَقَٰتِلَآ إِنَّا هَٰهُنَا قَٰعِدُونَ ﴿٢٤﴾ They said: O Moses! We will never enter (the land) while they are in it. So go thou and thy Lord and fight! We will sit here. (5:24) || قَالُوا۟ يَٰمُوسَىٰٓ إِنَّا لَن نَّدْخُلَهَآ أَبَدًۭا مَّا دَامُوا۟ فِيهَا ۖ فَٱذْهَبْ أَنتَ وَرَبُّكَ فَقَٰتِلَآ إِنَّا هَٰهُنَا قَٰعِدُونَ ﴿٢٤﴾ هغوی وویل: «ای موسی (علیه السلام)! بې‌شکه موږ به هېڅکله هم دې (ځمکې) ته ورننه وځو تر څو چې هغوی پکې وي؛ نو ته او ستا رب ورشئ او دواړه وجنګېږئ؛ بې‌شکه موږ دلته ناست یو.»
p قَالَ رَبِّ إِنِّى لَآ أَمْلِكُ إِلَّا نَفْسِى وَأَخِى ۖ فَٱفْرُقْ بَيْنَنَا وَبَيْنَ ٱلْقَوْمِ ٱلْفَٰسِقِينَ ﴿٢٥﴾ موسیٰ (علیہ السلام) نے عرض کیا اے میرے رب! بے شک میں اپنی ذات اور اپنے بھائی کے سوا (کسی اور پر) کچھ اختیار نہیں رکھتا پس ہمارے اور اس نافرمان قوم کے درمیان جدائی ڈال دے۔ || قَالَ رَبِّ إِنِّى لَآ أَمْلِكُ إِلَّا نَفْسِى وَأَخِى ۖ فَٱفْرُقْ بَيْنَنَا وَبَيْنَ ٱلْقَوْمِ ٱلْفَٰسِقِينَ ﴿٢٥﴾ He said: My Lord! I have control of none but myself and my brother, so distinguish between us and the wrong-doing folk. (5:25) || قَالَ رَبِّ إِنِّى لَآ أَمْلِكُ إِلَّا نَفْسِى وَأَخِى ۖ فَٱفْرُقْ بَيْنَنَا وَبَيْنَ ٱلْقَوْمِ ٱلْفَٰسِقِينَ ﴿٢٥﴾ موسی (علیه السلام) عرض وکړ: «ای زما ربه! بې‌شکه زه له خپل ځان او خپل ورور پرته (پر بل چا) هېڅ واک نه لرم؛ نو زموږ او د دې نافرمان قوم ترمنځ جلاوالی راوله.»
