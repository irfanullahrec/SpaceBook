p لَقَدْ أَخَذْنَا مِيثَٰقَ بَنِىٓ إِسْرَٰٓءِيلَ وَأَرْسَلْنَآ إِلَيْهِمْ رُسُلًۭا ۖ كُلَّمَا جَآءَهُمْ رَسُولٌۢ بِمَا لَا تَهْوَىٰٓ أَنفُسُهُمْ فَرِيقًۭا كَذَّبُوا۟ وَفَرِيقًۭا يَقْتُلُونَ ﴿٧٠﴾ یقیناً ہم نے بنی اسرائیل سے پختہ عہد لیا اور ہم نے ان کی طرف رسولوں کو بھیجا تو جب کبھی ان کے پاس کوئی رسول (ایسا حکم) لے کر آیا جسے ان کے نفس پسند نہ کرتے تھے تو انھوں نے بعض کو جھٹلایا اور بعض کو قتل (شہید) کیا۔ || لَقَدْ أَخَذْنَا مِيثَٰقَ بَنِىٓ إِسْرَٰٓءِيلَ وَأَرْسَلْنَآ إِلَيْهِمْ رُسُلًۭا ۖ كُلَّمَا جَآءَهُمْ رَسُولٌۢ بِمَا لَا تَهْوَىٰٓ أَنفُسُهُمْ فَرِيقًۭا كَذَّبُوا۟ وَفَرِيقًۭا يَقْتُلُونَ ﴿٧٠﴾ We made a covenant of old with the Children of Israel and We sent unto them messengers. As often as a messenger came unto them with that which their souls desired not (they became rebellious). Some (of them) they denied and some they slew. (5:70) || لَقَدْ أَخَذْنَا مِيثَٰقَ بَنِىٓ إِسْرَٰٓءِيلَ وَأَرْسَلْنَآ إِلَيْهِمْ رُسُلًۭا ۖ كُلَّمَا جَآءَهُمْ رَسُولٌۢ بِمَا لَا تَهْوَىٰٓ أَنفُسُهُمْ فَرِيقًۭا كَذَّبُوا۟ وَفَرِيقًۭا يَقْتُلُونَ ﴿٧٠﴾ یقیناً موږ له بني اسرائیلو ټینګه ژمنه واخیسته او هغوی ته مو رسولان ولېږل؛ نو هرکله چې هغوی ته کوم رسول (داسې حکم) راوړ چې د هغوی نفسونو نه خوښاوه، نو ځینې یې درواغجن وګڼل او ځینې یې ووژل (شهیدان کړل).
p وَحَسِبُوٓا۟ أَلَّا تَكُونَ فِتْنَةٌۭ فَعَمُوا۟ وَصَمُّوا۟ ثُمَّ تَابَ ٱللَّهُ عَلَيْهِمْ ثُمَّ عَمُوا۟ وَصَمُّوا۟ كَثِيرٌۭ مِّنْهُمْ ۚ وَٱللَّهُ بَصِيرٌۢ بِمَا يَعْمَلُونَ ﴿٧١﴾ اور وہ سمجھ بیٹھے کہ (ایسا کرنے سے) کوئی عذاب نہ ہو گا تو وہ اندھے اور بہرے ہو گئے پھر اللہ نے اُن کی توبہ قبول فرمائی پھر اُن میں سے بہت سے اندھے اور بہرے ہو گئے اور اللہ خوب دیکھنے والا ہے جو کچھ وہ کر رہے ہیں۔ || وَحَسِبُوٓا۟ أَلَّا تَكُونَ فِتْنَةٌۭ فَعَمُوا۟ وَصَمُّوا۟ ثُمَّ تَابَ ٱللَّهُ عَلَيْهِمْ ثُمَّ عَمُوا۟ وَصَمُّوا۟ كَثِيرٌۭ مِّنْهُمْ ۚ وَٱللَّهُ بَصِيرٌۢ بِمَا يَعْمَلُونَ ﴿٧١﴾ They thought no harm would come of it, so they were wilfully blind and deaf. And afterward Allah turned (in mercy) toward them. Now (even after that) are many of them wilfully blind and deaf. Allah is Seer of what they do. (5:71) || وَحَسِبُوٓا۟ أَلَّا تَكُونَ فِتْنَةٌۭ فَعَمُوا۟ وَصَمُّوا۟ ثُمَّ تَابَ ٱللَّهُ عَلَيْهِمْ ثُمَّ عَمُوا۟ وَصَمُّوا۟ كَثِيرٌۭ مِّنْهُمْ ۚ وَٱللَّهُ بَصِيرٌۢ بِمَا يَعْمَلُونَ ﴿٧١﴾ او هغوی ګومان وکړ چې (په داسې کولو) به هېڅ عذاب نه وي، نو ړانده او کاڼه شول؛ بیا الله یې توبه قبوله کړه؛ بیا له هغوی ډېر ړانده او کاڼه شول؛ او الله ښه لیدونکی دی چې هغوی څه کوي.
p لَقَدْ كَفَرَ ٱلَّذِينَ قَالُوٓا۟ إِنَّ ٱللَّهَ هُوَ ٱلْمَسِيحُ ٱبْنُ مَرْيَمَ ۖ وَقَالَ ٱلْمَسِيحُ يَٰبَنِىٓ إِسْرَٰٓءِيلَ ٱعْبُدُوا۟ ٱللَّهَ رَبِّى وَرَبَّكُمْ ۖ إِنَّهُۥ مَن يُشْرِكْ بِٱللَّهِ فَقَدْ حَرَّمَ ٱللَّهُ عَلَيْهِ ٱلْجَنَّةَ وَمَأْوَىٰهُ ٱلنَّارُ ۖ وَمَا لِلظَّٰلِمِينَ مِنْ أَنصَارٍۢ ﴿٧٢﴾ یقیناً اُن لوگوں نے کفر کیا جنھوں نے کہا بے شک اللہ مسیح ابنِ مریم ہی ہے حالاں کہ (خود) مسیح (علیہ السلام) نے فرمایا تھا اے بنی اسرائیل! اللہ کی عبادت کرو جو میرا (بھی) رب ہے اور تمھارا (بھی) رب ہے بے شک جو اللہ کے ساتھ کسی کو شریک کرتا ہے تو یقیناً اللہ نے اُس پر جنت حرام کر دی ہے اور اُس کا ٹھکانا (جہنم کی) آگ ہے اور ظالموں کا کوئی مدد گار نہیں۔ || لَقَدْ كَفَرَ ٱلَّذِينَ قَالُوٓا۟ إِنَّ ٱللَّهَ هُوَ ٱلْمَسِيحُ ٱبْنُ مَرْيَمَ ۖ وَقَالَ ٱلْمَسِيحُ يَٰبَنِىٓ إِسْرَٰٓءِيلَ ٱعْبُدُوا۟ ٱللَّهَ رَبِّى وَرَبَّكُمْ ۖ إِنَّهُۥ مَن يُشْرِكْ بِٱللَّهِ فَقَدْ حَرَّمَ ٱللَّهُ عَلَيْهِ ٱلْجَنَّةَ وَمَأْوَىٰهُ ٱلنَّارُ ۖ وَمَا لِلظَّٰلِمِينَ مِنْ أَنصَارٍۢ ﴿٧٢﴾ They surely disbelieve who say: Lo! Allah is the Messiah, son of Mary. The Messiah (himself) said: O Children of Israel, worship Allah, my Lord and your Lord. Lo! whoso ascribeth partners unto Allah, for him Allah hath forbidden paradise. His abode is the Fire. For evil-doers there will be no helpers. (5:72) || لَقَدْ كَفَرَ ٱلَّذِينَ قَالُوٓا۟ إِنَّ ٱللَّهَ هُوَ ٱلْمَسِيحُ ٱبْنُ مَرْيَمَ ۖ وَقَالَ ٱلْمَسِيحُ يَٰبَنِىٓ إِسْرَٰٓءِيلَ ٱعْبُدُوا۟ ٱللَّهَ رَبِّى وَرَبَّكُمْ ۖ إِنَّهُۥ مَن يُشْرِكْ بِٱللَّهِ فَقَدْ حَرَّمَ ٱللَّهُ عَلَيْهِ ٱلْجَنَّةَ وَمَأْوَىٰهُ ٱلنَّارُ ۖ وَمَا لِلظَّٰلِمِينَ مِنْ أَنصَارٍۢ ﴿٧٢﴾ یقیناً هغو کسانو کفر وکړ چې ویل یې: «بې‌شکه الله همدا مسیح ابن مریم دی»، حال دا چې (پخپله) مسیح (علیه السلام) ویلي وو: «ای بني اسرائیلو! د الله عبادت وکړئ چې زما (هم) رب دی او ستاسو (هم) رب دی؛ بې‌شکه څوک چې له الله سره څوک شریک کړي، نو یقیناً الله پر هغه جنت حرام کړی او ټیکانه یې (د دوزخ) اور دی، او د ظالمانو هېڅ مرستندوی نشته.»
p لَّقَدْ كَفَرَ ٱلَّذِينَ قَالُوٓا۟ إِنَّ ٱللَّهَ ثَالِثُ ثَلَٰثَةٍۢ ۘ وَمَا مِنْ إِلَٰهٍ إِلَّآ إِلَٰهٌۭ وَٰحِدٌۭ ۚ وَإِن لَّمْ يَنتَهُوا۟ عَمَّا يَقُولُونَ لَيَمَسَّنَّ ٱلَّذِينَ كَفَرُوا۟ مِنْهُمْ عَذَابٌ أَلِيمٌ ﴿٧٣﴾ یقیناً اُن لوگوں نے کفر کیا جنھوں نے کہا بے شک اللہ تین (معبودوں) میں سے تیسرا ہے حالاں کہ معبودِ واحد (اللہ) کے سوا کوئی معبود نہیں اور اگر وہ اُس بات سے باز نہ آئے جو وہ لوگ کہتے ہیں تو یقیناً اُن میں سے جنھوں نے کفر کیا اُن کو ضرور دردناک عذاب پہنچے گا۔ || لَّقَدْ كَفَرَ ٱلَّذِينَ قَالُوٓا۟ إِنَّ ٱللَّهَ ثَالِثُ ثَلَٰثَةٍۢ ۘ وَمَا مِنْ إِلَٰهٍ إِلَّآ إِلَٰهٌۭ وَٰحِدٌۭ ۚ وَإِن لَّمْ يَنتَهُوا۟ عَمَّا يَقُولُونَ لَيَمَسَّنَّ ٱلَّذِينَ كَفَرُوا۟ مِنْهُمْ عَذَابٌ أَلِيمٌ ﴿٧٣﴾ They surely disbelieve who say: Lo! Allah is the third of three; when there is no Allah save the One Allah. If they desist not from so saying a painful doom will fall on those of them who disbelieve. (5:73) || لَّقَدْ كَفَرَ ٱلَّذِينَ قَالُوٓا۟ إِنَّ ٱللَّهَ ثَالِثُ ثَلَٰثَةٍۢ ۘ وَمَا مِنْ إِلَٰهٍ إِلَّآ إِلَٰهٌۭ وَٰحِدٌۭ ۚ وَإِن لَّمْ يَنتَهُوا۟ عَمَّا يَقُولُونَ لَيَمَسَّنَّ ٱلَّذِينَ كَفَرُوا۟ مِنْهُمْ عَذَابٌ أَلِيمٌ ﴿٧٣﴾ یقیناً هغو کسانو کفر وکړ چې ویل یې: «بې‌شکه الله له درېیو (معبودانو) درېیم دی»، حال دا چې له یوه معبود (الله) پرته هېڅ معبود نشته؛ او که له هغې خبرې لاس وانه خلي چې وایي یې، نو یقیناً له هغوی هغو ته چې کفر یې کړی، خامخا دردناک عذاب رسېږي.
p أَفَلَا يَتُوبُونَ إِلَى ٱللَّهِ وَيَسْتَغْفِرُونَهُۥ ۚ وَٱللَّهُ غَفُورٌۭ رَّحِيمٌۭ ﴿٧٤﴾ آخر وہ اللہ کے حضور توبہ کیوں نہیں کرتے اور اُس سے مغفرت طلب کیوں نہیں کرتے؟ اور اللہ بڑا ہی بخشنے والا نہایت رحم فرمانے والا ہے۔ || أَفَلَا يَتُوبُونَ إِلَى ٱللَّهِ وَيَسْتَغْفِرُونَهُۥ ۚ وَٱللَّهُ غَفُورٌۭ رَّحِيمٌۭ ﴿٧٤﴾ Will they not rather turn unto Allah and seek forgiveness of Him? For Allah is Forgiving, Merciful. (5:74) || أَفَلَا يَتُوبُونَ إِلَى ٱللَّهِ وَيَسْتَغْفِرُونَهُۥ ۚ وَٱللَّهُ غَفُورٌۭ رَّحِيمٌۭ ﴿٧٤﴾ نو ولې د الله په وړاندې توبه نه کوي او ترې بښنه نه غواړي؟ او الله ډېر بښونکی، بې‌حده مهربان دی.
p مَّا ٱلْمَسِيحُ ٱبْنُ مَرْيَمَ إِلَّا رَسُولٌۭ قَدْ خَلَتْ مِن قَبْلِهِ ٱلرُّسُلُ وَأُمُّهُۥ صِدِّيقَةٌۭ ۖ كَانَا يَأْكُلَانِ ٱلطَّعَامَ ۗ ٱنظُرْ كَيْفَ نُبَيِّنُ لَهُمُ ٱلْءَايَٰتِ ثُمَّ ٱنظُرْ أَنَّىٰ يُؤْفَكُونَ ﴿٧٥﴾ مسیح ابنِ مریم (علیہ السلام) تو صرف (اللہ کے) رسول ہیں اُن سے پہلے (بھی) بہت سے رسول گزر چکے ہیں اور اُن کی والدہ صدیقہ تھیں دونوں کھانا کھایا کرتے تھے آپ دیکھیے ہم کیسے ان کے لیے آیات کھول کھول کر بیان کرتے ہیں پھر یہ کہاں بھٹکے جا رہے ہیں؟ || مَّا ٱلْمَسِيحُ ٱبْنُ مَرْيَمَ إِلَّا رَسُولٌۭ قَدْ خَلَتْ مِن قَبْلِهِ ٱلرُّسُلُ وَأُمُّهُۥ صِدِّيقَةٌۭ ۖ كَانَا يَأْكُلَانِ ٱلطَّعَامَ ۗ ٱنظُرْ كَيْفَ نُبَيِّنُ لَهُمُ ٱلْءَايَٰتِ ثُمَّ ٱنظُرْ أَنَّىٰ يُؤْفَكُونَ ﴿٧٥﴾ The Messiah, son of Mary, was no other than a messenger, messengers (the like of whom) had passed away before him. And his mother was a saintly woman. And they both used to eat (earthly) food. See how We make the revelations clear for them, and see how they are turned away! (5:75) || مَّا ٱلْمَسِيحُ ٱبْنُ مَرْيَمَ إِلَّا رَسُولٌۭ قَدْ خَلَتْ مِن قَبْلِهِ ٱلرُّسُلُ وَأُمُّهُۥ صِدِّيقَةٌۭ ۖ كَانَا يَأْكُلَانِ ٱلطَّعَامَ ۗ ٱنظُرْ كَيْفَ نُبَيِّنُ لَهُمُ ٱلْءَايَٰتِ ثُمَّ ٱنظُرْ أَنَّىٰ يُؤْفَكُونَ ﴿٧٥﴾ مسیح ابن مریم (علیه السلام) خو یوازې د (الله) رسول دی؛ له هغه مخکې (هم) ډېر رسولان تېر شوي، او مور یې صدیقه وه؛ دواړو به خواړه خوړل. وګورئ موږ څنګه ورته آیتونه ښه روښانه بیانوو، بیا دوی چېرې اړول کېږي؟
p قُلْ أَتَعْبُدُونَ مِن دُونِ ٱللَّهِ مَا لَا يَمْلِكُ لَكُمْ ضَرًّۭا وَلَا نَفْعًۭا ۚ وَٱللَّهُ هُوَ ٱلسَّمِيعُ ٱلْعَلِيمُ ﴿٧٦﴾ آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) فرما دیجیے کیا تم اللہ کے سوا اس کو پوجتے ہو جو تمھارے نقصان اور نفع کا کچھ اختیار نہیں رکھتا اور اللہ ہی خوب سننے والا خوب جاننے والا ہے۔ || قُلْ أَتَعْبُدُونَ مِن دُونِ ٱللَّهِ مَا لَا يَمْلِكُ لَكُمْ ضَرًّۭا وَلَا نَفْعًۭا ۚ وَٱللَّهُ هُوَ ٱلسَّمِيعُ ٱلْعَلِيمُ ﴿٧٦﴾ Say: Serve ye in place of Allah that which possesseth for you neither hurt nor use? Allah it is Who is the Hearer, the Knower. (5:76) || قُلْ أَتَعْبُدُونَ مِن دُونِ ٱللَّهِ مَا لَا يَمْلِكُ لَكُمْ ضَرًّۭا وَلَا نَفْعًۭا ۚ وَٱللَّهُ هُوَ ٱلسَّمِيعُ ٱلْعَلِيمُ ﴿٧٦﴾ تاسو (صلی الله علیه وعلی آله واصحابه وسلم) ووایئ: «ایا له الله پرته د هغه چا عبادت کوئ چې ستاسو د زیان او ګټې هېڅ واک نه لري؟» او یوازې الله ښه اورېدونکی، ښه پوه دی.
