p مَّثَلُ ٱلْجَنَّةِ ٱلَّتِى وُعِدَ ٱلْمُتَّقُونَ ۖ فِيهَآ أَنْهَٰرٌۭ مِّن مَّآءٍ غَيْرِ ءَاسِنٍۢ وَأَنْهَٰرٌۭ مِّن لَّبَنٍۢ لَّمْ يَتَغَيَّرْ طَعْمُهُۥ وَأَنْهَٰرٌۭ مِّنْ خَمْرٍۢ لَّذَّةٍۢ لِّلشَّٰرِبِينَ وَأَنْهَٰرٌۭ مِّنْ عَسَلٍۢ مُّصَفًّۭى ۖ وَلَهُمْ فِيهَا مِن كُلِّ ٱلثَّمَرَٰتِ وَمَغْفِرَةٌۭ مِّن رَّبِّهِمْ ۖ كَمَنْ هُوَ خَٰلِدٌۭ فِى ٱلنَّارِ وَسُقُوا۟ مَآءً حَمِيمًۭا فَقَطَّعَ أَمْعَآءَهُمْ ﴿١٥﴾ پرہیزگاروں سے جس جنت کا وعدہ کیا گیا ہے اس کی کیفیت یہ ہے کہ اس میں ایسے پانی کی نہریں ہیں جو (کبھی) بدبودار نہ ہوگا اور ایسے دودھ کی نہریں ہیں جس کا ذائقہ نہیں بدلے گا اور ایسی شراب کی نہریں ہیں جو پینے والوں کے لیے لذت بخش ہے اور صاف کیے ہوئے شہد کی نہریں (بھی) ہیں اور اُن کے لیے ان باغات میں ہر قسم کے پھل ہیں اور ان کے رب کی طرف سے مغفرت ہے کیا یہ (لوگ) اُس کی طرح ہو سکتے ہیں جو آگ میں ہمیشہ رہنے والا ہے اور جنھیں کھولتا ہوا پانی پلایا جائے گا تو وہ ان کی آنتیں کاٹ کر رکھ دے گا۔ || مَّثَلُ ٱلْجَنَّةِ ٱلَّتِى وُعِدَ ٱلْمُتَّقُونَ ۖ فِيهَآ أَنْهَٰرٌۭ مِّن مَّآءٍ غَيْرِ ءَاسِنٍۢ وَأَنْهَٰرٌۭ مِّن لَّبَنٍۢ لَّمْ يَتَغَيَّرْ طَعْمُهُۥ وَأَنْهَٰرٌۭ مِّنْ خَمْرٍۢ لَّذَّةٍۢ لِّلشَّٰرِبِينَ وَأَنْهَٰرٌۭ مِّنْ عَسَلٍۢ مُّصَفًّۭى ۖ وَلَهُمْ فِيهَا مِن كُلِّ ٱلثَّمَرَٰتِ وَمَغْفِرَةٌۭ مِّن رَّبِّهِمْ ۖ كَمَنْ هُوَ خَٰلِدٌۭ فِى ٱلنَّارِ وَسُقُوا۟ مَآءً حَمِيمًۭا فَقَطَّعَ أَمْعَآءَهُمْ ﴿١٥﴾ A similitude of the Garden which those who keep their duty (to Allah) are promised: Therein are rivers of water unpolluted, and rivers of milk whereof the flavour changeth not, and rivers of wine delicious to the drinkers, and rivers of clear-run honey; therein for them is every kind of fruit, with pardon from their Lord. (Are those who enjoy all this) like those who are immortal in the Fire and are given boiling water to drink so that it teareth their bowels? (47:15) || مَّثَلُ ٱلْجَنَّةِ ٱلَّتِى وُعِدَ ٱلْمُتَّقُونَ ۖ فِيهَآ أَنْهَٰرٌۭ مِّن مَّآءٍ غَيْرِ ءَاسِنٍۢ وَأَنْهَٰرٌۭ مِّن لَّبَنٍۢ لَّمْ يَتَغَيَّرْ طَعْمُهُۥ وَأَنْهَٰرٌۭ مِّنْ خَمْرٍۢ لَّذَّةٍۢ لِّلشَّٰرِبِينَ وَأَنْهَٰرٌۭ مِّنْ عَسَلٍۢ مُّصَفًّۭى ۖ وَلَهُمْ فِيهَا مِن كُلِّ ٱلثَّمَرَٰتِ وَمَغْفِرَةٌۭ مِّن رَّبِّهِمْ ۖ كَمَنْ هُوَ خَٰلِدٌۭ فِى ٱلنَّارِ وَسُقُوا۟ مَآءً حَمِيمًۭا فَقَطَّعَ أَمْعَآءَهُمْ ﴿١٥﴾ د هغه جنت حالت چې له پرهېزګارانو سره یې وعده شوې دا دی چې په هغه کې د داسې اوبو ویالې دي چې (هېڅکله) بدبویه نه شي، او د داسې شیدو ویالې دي چې خوند یې نه بدلېږي، او د داسې شرابو ویالې دي چې څښونکو ته خوندوره ده، او د صاف شوي شاتو ویالې (هم) دي، او د هغوی لپاره په دې باغونو کې هر ډول مېوې دي، او د هغوی د رب له لوري بښنه ده؛ ایا دا (خلک) د هغه چا په څېر کېدای شي چې تل په اور کې پاتې کېدونکی دی او ایشېدلې اوبه به ورته ورکړل شي چې کولمې به یې ټوټې ټوټې کړي؟
p وَمِنْهُم مَّن يَسْتَمِعُ إِلَيْكَ حَتَّىٰٓ إِذَا خَرَجُوا۟ مِنْ عِندِكَ قَالُوا۟ لِلَّذِينَ أُوتُوا۟ ٱلْعِلْمَ مَاذَا قَالَ ءَانِفًا ۚ أُو۟لَٰٓئِكَ ٱلَّذِينَ طَبَعَ ٱللَّهُ عَلَىٰ قُلُوبِهِمْ وَٱتَّبَعُوٓا۟ أَهْوَآءَهُمْ ﴿١٦﴾ اور ان میں سے کچھ (لوگ) آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کی طرف کان لگا کر سنتے ہیں یہاں تک کہ جب وہ آپ کے پاس سے نکل کر جاتے ہیں (تو) ان سے کہتے ہیں جنھیں علم عطا فرمایا گیا ہے کہ ابھی ان صاحب (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) نے کیا کہا تھا یہی لوگ ہیں کہ جن کے دلوں پر اللہ نے مہر لگا دی ہے اور انھوں نے اپنی خواہشات کی پیروی کی ہے۔ || وَمِنْهُم مَّن يَسْتَمِعُ إِلَيْكَ حَتَّىٰٓ إِذَا خَرَجُوا۟ مِنْ عِندِكَ قَالُوا۟ لِلَّذِينَ أُوتُوا۟ ٱلْعِلْمَ مَاذَا قَالَ ءَانِفًا ۚ أُو۟لَٰٓئِكَ ٱلَّذِينَ طَبَعَ ٱللَّهُ عَلَىٰ قُلُوبِهِمْ وَٱتَّبَعُوٓا۟ أَهْوَآءَهُمْ ﴿١٦﴾ Among them are some who give ear unto thee (Muhammad) till, when they go forth from thy presence they say unto those who have been given knowledge: What was that he said just now? Those are they whose hearts Allah hath sealed, and they follow their own lusts. (47:16) || وَمِنْهُم مَّن يَسْتَمِعُ إِلَيْكَ حَتَّىٰٓ إِذَا خَرَجُوا۟ مِنْ عِندِكَ قَالُوا۟ لِلَّذِينَ أُوتُوا۟ ٱلْعِلْمَ مَاذَا قَالَ ءَانِفًا ۚ أُو۟لَٰٓئِكَ ٱلَّذِينَ طَبَعَ ٱللَّهُ عَلَىٰ قُلُوبِهِمْ وَٱتَّبَعُوٓا۟ أَهْوَآءَهُمْ ﴿١٦﴾ او له هغوی ځینې (خلک) تاسو (صلی الله علیه وعلی آله واصحابه وسلم) ته غوږ نیسي، تر دې چې کله ستاسو له خوا ووځي، (نو) هغو کسانو ته چې علم ورکړل شوی وايي: دې صاحب (صلی الله علیه وعلی آله واصحابه وسلم) همدا اوس څه وویل؟ همدا هغه خلک دي چې الله یې پر زړونو مهر لګولی، او د خپلو غوښتنو پیروي یې کړې.
p وَٱلَّذِينَ ٱهْتَدَوْا۟ زَادَهُمْ هُدًۭى وَءَاتَىٰهُمْ تَقْوَىٰهُمْ ﴿١٧﴾ اور جنھوں نے ہدایت اختیار کی اللہ نے انھیں ہدایت میں بڑھا دیا اور انھیں ان کی پرہیزگاری عطا فرمائی۔ || وَٱلَّذِينَ ٱهْتَدَوْا۟ زَادَهُمْ هُدًۭى وَءَاتَىٰهُمْ تَقْوَىٰهُمْ ﴿١٧﴾ While as for those who walk aright, He addeth to their guidance, and giveth them their protection (against evil). (47:17) || وَٱلَّذِينَ ٱهْتَدَوْا۟ زَادَهُمْ هُدًۭى وَءَاتَىٰهُمْ تَقْوَىٰهُمْ ﴿١٧﴾ او چا چې هدایت غوره کړ، الله یې په هدایت کې زیات کړل او د هغوی پرهېزګاري یې ورکړه.
p فَهَلْ يَنظُرُونَ إِلَّا ٱلسَّاعَةَ أَن تَأْتِيَهُم بَغْتَةًۭ ۖ فَقَدْ جَآءَ أَشْرَاطُهَا ۚ فَأَنَّىٰ لَهُمْ إِذَا جَآءَتْهُمْ ذِكْرَىٰهُمْ ﴿١٨﴾ پھر کیا یہ صرف قیامت کا انتظار کر رہے ہیں کہ وہ ان پر اچانک آ جائے؟ تو یقیناً اُس کی نشانیاں تو آ چکی ہیں پھر انھیں ان کی نصیحت کہاں (مفید) ہوگی جب وہ (قیامت) آ پہنچے گی؟ || فَهَلْ يَنظُرُونَ إِلَّا ٱلسَّاعَةَ أَن تَأْتِيَهُم بَغْتَةًۭ ۖ فَقَدْ جَآءَ أَشْرَاطُهَا ۚ فَأَنَّىٰ لَهُمْ إِذَا جَآءَتْهُمْ ذِكْرَىٰهُمْ ﴿١٨﴾ Await they aught save the Hour, that it should come upon them unawares? And the beginnings thereof have already come. But how, when it hath come upon them, can they take their warning? (47:18) || فَهَلْ يَنظُرُونَ إِلَّا ٱلسَّاعَةَ أَن تَأْتِيَهُم بَغْتَةًۭ ۖ فَقَدْ جَآءَ أَشْرَاطُهَا ۚ فَأَنَّىٰ لَهُمْ إِذَا جَآءَتْهُمْ ذِكْرَىٰهُمْ ﴿١٨﴾ بیا ایا دوی یوازې د قیامت انتظار کوي چې ناڅاپه پرې راشي؟ نو یقیناً د هغې نښې خو راغلې دي؛ بیا به هغوی ته د هغوی نصیحت څه (ګټه) ورکړي کله چې هغه (قیامت) راورسېږي؟
p فَٱعْلَمْ أَنَّهُۥ لَآ إِلَٰهَ إِلَّا ٱللَّهُ وَٱسْتَغْفِرْ لِذَنۢبِكَ وَلِلْمُؤْمِنِينَ وَٱلْمُؤْمِنَٰتِ ۗ وَٱللَّهُ يَعْلَمُ مُتَقَلَّبَكُمْ وَمَثْوَىٰكُمْ ﴿١٩﴾ تو آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) جان لیجیے کہ اللہ کے سوا کوئی معبود نہیں اور اپنے (بظاہر) خلافِ اولیٰ کام پر استغفار کیجیے اور مومن مردوں اور مومن عورتوں کے لیے بھی (معافی طلب کیجیے) اور اللہ تمھاری آمد و رفت اور تمھارے ٹھکانے کو (خوب) جانتا ہے۔ || فَٱعْلَمْ أَنَّهُۥ لَآ إِلَٰهَ إِلَّا ٱللَّهُ وَٱسْتَغْفِرْ لِذَنۢبِكَ وَلِلْمُؤْمِنِينَ وَٱلْمُؤْمِنَٰتِ ۗ وَٱللَّهُ يَعْلَمُ مُتَقَلَّبَكُمْ وَمَثْوَىٰكُمْ ﴿١٩﴾ So know (O Muhammad) that there is no Allah save Allah, and ask forgiveness for thy sin and for believing men and believing women. Allah knoweth (both) your place of turmoil and your place of rest. (47:19) || فَٱعْلَمْ أَنَّهُۥ لَآ إِلَٰهَ إِلَّا ٱللَّهُ وَٱسْتَغْفِرْ لِذَنۢبِكَ وَلِلْمُؤْمِنِينَ وَٱلْمُؤْمِنَٰتِ ۗ وَٱللَّهُ يَعْلَمُ مُتَقَلَّبَكُمْ وَمَثْوَىٰكُمْ ﴿١٩﴾ نو تاسو (صلی الله علیه وعلی آله واصحابه وسلم) پوه شئ چې له الله پرته هېڅ معبود نشته، او په خپل (ظاهراً) خلاف اولی کار استغفار وکړئ، او د مؤمنو نارینه وو او مؤمنو ښځو لپاره هم (بښنه وغواړئ)؛ او الله ستاسو راتګ او تګ او ستاسو ځای (ښه) پېژني.
p وَيَقُولُ ٱلَّذِينَ ءَامَنُوا۟ لَوْلَا نُزِّلَتْ سُورَةٌۭ ۖ فَإِذَآ أُنزِلَتْ سُورَةٌۭ مُّحْكَمَةٌۭ وَذُكِرَ فِيهَا ٱلْقِتَالُ ۙ رَأَيْتَ ٱلَّذِينَ فِى قُلُوبِهِم مَّرَضٌۭ يَنظُرُونَ إِلَيْكَ نَظَرَ ٱلْمَغْشِىِّ عَلَيْهِ مِنَ ٱلْمَوْتِ ۖ فَأَوْلَىٰ لَهُمْ ﴿٢٠﴾ اور جو لوگ ایمان لائے وہ کہتے ہیں (قتال کے بارے میں) کوئی (نئی) سورت کیوں نازل نہیں کی گئی؟ پھر جب کوئی واضح سورت نازل کی گئی اور اس میں قتال کا ذکر کیا گیا تو آپ اُن کو دیکھیں گے جن کے دلوں میں (نفاق کی) بیماری ہے وہ آپ کی طرف (اس طرح) دیکھتے ہیں جیسے وہ شخص دیکھتا ہے جس پر موت کی غشی طاری ہو تو اُن کے لیے بربادی ہے۔ || وَيَقُولُ ٱلَّذِينَ ءَامَنُوا۟ لَوْلَا نُزِّلَتْ سُورَةٌۭ ۖ فَإِذَآ أُنزِلَتْ سُورَةٌۭ مُّحْكَمَةٌۭ وَذُكِرَ فِيهَا ٱلْقِتَالُ ۙ رَأَيْتَ ٱلَّذِينَ فِى قُلُوبِهِم مَّرَضٌۭ يَنظُرُونَ إِلَيْكَ نَظَرَ ٱلْمَغْشِىِّ عَلَيْهِ مِنَ ٱلْمَوْتِ ۖ فَأَوْلَىٰ لَهُمْ ﴿٢٠﴾ And those who believe say: If only a surah were revealed!
