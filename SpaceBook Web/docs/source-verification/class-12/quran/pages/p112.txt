p يَٰٓأَيُّهَا ٱلنَّبِىُّ قُل لِّأَزْوَٰجِكَ إِن كُنتُنَّ تُرِدْنَ ٱلْحَيَوٰةَ ٱلدُّنْيَا وَزِينَتَهَا فَتَعَالَيْنَ أُمَتِّعْكُنَّ وَأُسَرِّحْكُنَّ سَرَاحًۭا جَمِيلًۭا ﴿٢٨﴾ اے نبی (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم)! اپنی ازواج سے فرمائیے اگر تم دنیاوی زندگی اور اُس کی زینت چاہتی ہو تو آؤ میں تمھیں کچھ مال دیتا ہوں اور تمھیں خوش اسلوبی سے رخصت کر دیتا ہوں۔ || يَٰٓأَيُّهَا ٱلنَّبِىُّ قُل لِّأَزْوَٰجِكَ إِن كُنتُنَّ تُرِدْنَ ٱلْحَيَوٰةَ ٱلدُّنْيَا وَزِينَتَهَا فَتَعَالَيْنَ أُمَتِّعْكُنَّ وَأُسَرِّحْكُنَّ سَرَاحًۭا جَمِيلًۭا ﴿٢٨﴾ O Prophet! Say unto thy wives: If ye desire the world's life and its adornment, come! I will content you and will release you with a fair release. (33:28) || يَٰٓأَيُّهَا ٱلنَّبِىُّ قُل لِّأَزْوَٰجِكَ إِن كُنتُنَّ تُرِدْنَ ٱلْحَيَوٰةَ ٱلدُّنْيَا وَزِينَتَهَا فَتَعَالَيْنَ أُمَتِّعْكُنَّ وَأُسَرِّحْكُنَّ سَرَاحًۭا جَمِيلًۭا ﴿٢٨﴾ ای نبي (صلی الله علیه وعلی آله واصحابه وسلم)! خپلو مېرمنو ته ووایئ: که تاسو د دنیا ژوند او د هغه ښکلا غواړئ، نو راشئ چې څه مال درکړم او په ښه توګه مو رخصت کړم.
p وَإِن كُنتُنَّ تُرِدْنَ ٱللَّهَ وَرَسُولَهُۥ وَٱلدَّارَ ٱلْءَاخِرَةَ فَإِنَّ ٱللَّهَ أَعَدَّ لِلْمُحْسِنَٰتِ مِنكُنَّ أَجْرًا عَظِيمًۭا ﴿٢٩﴾ اور اگر تم چاہتی ہو اللہ اور اُس کے رسول کو اور آخرت کے گھر کو تو بے شک تم میں سے جو نیکی کرنے والیاں ہیں اللہ نے اُن کے لیے اجرِ عظیم تیار کر رکھا ہے۔ || وَإِن كُنتُنَّ تُرِدْنَ ٱللَّهَ وَرَسُولَهُۥ وَٱلدَّارَ ٱلْءَاخِرَةَ فَإِنَّ ٱللَّهَ أَعَدَّ لِلْمُحْسِنَٰتِ مِنكُنَّ أَجْرًا عَظِيمًۭا ﴿٢٩﴾ But if ye desire Allah and His messenger and the abode of the Hereafter, then lo! Allah hath prepared for the good among you an immense reward. (33:29) || وَإِن كُنتُنَّ تُرِدْنَ ٱللَّهَ وَرَسُولَهُۥ وَٱلدَّارَ ٱلْءَاخِرَةَ فَإِنَّ ٱللَّهَ أَعَدَّ لِلْمُحْسِنَٰتِ مِنكُنَّ أَجْرًا عَظِيمًۭا ﴿٢٩﴾ او که تاسو الله، د هغه رسول او د آخرت کور غواړئ، نو بې‌شکه له تاسو څخه چې کومې نېکي کوونکې دي، الله د هغوی لپاره لوی اجر تیار کړی دی.
p يَٰنِسَآءَ ٱلنَّبِىِّ مَن يَأْتِ مِنكُنَّ بِفَٰحِشَةٍۢ مُّبَيِّنَةٍۢ يُضَٰعَفْ لَهَا ٱلْعَذَابُ ضِعْفَيْنِ ۚ وَكَانَ ذَٰلِكَ عَلَى ٱللَّهِ يَسِيرًۭا ﴿٣٠﴾ اے نبی (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کی ازواج! اگر (بالفرض) تم میں سے کسی نے کھلی بے حیائی کا ارتکاب کیا اُس کے لیے عذاب کو بڑھا کر دوگنا کر دیا جائے گا اور یہ بات اللہ کے لیے (بالکل) آسان ہے۔ || يَٰنِسَآءَ ٱلنَّبِىِّ مَن يَأْتِ مِنكُنَّ بِفَٰحِشَةٍۢ مُّبَيِّنَةٍۢ يُضَٰعَفْ لَهَا ٱلْعَذَابُ ضِعْفَيْنِ ۚ وَكَانَ ذَٰلِكَ عَلَى ٱللَّهِ يَسِيرًۭا ﴿٣٠﴾ O ye wives of the Prophet! Whosoever of you committeth manifest lewdness, the punishment for her will be doubled, and that is easy for Allah. (33:30) || يَٰنِسَآءَ ٱلنَّبِىِّ مَن يَأْتِ مِنكُنَّ بِفَٰحِشَةٍۢ مُّبَيِّنَةٍۢ يُضَٰعَفْ لَهَا ٱلْعَذَابُ ضِعْفَيْنِ ۚ وَكَانَ ذَٰلِكَ عَلَى ٱللَّهِ يَسِيرًۭا ﴿٣٠﴾ ای د نبي (صلی الله علیه وعلی آله واصحابه وسلم) مېرمنو! که (فرضاً) له تاسو چا ښکاره بې‌حیایي وکړه، د هغې لپاره به عذاب زیات او دوه چنده کړای شي؛ او دا خبره پر الله (بېخي) اسانه ده.
p ۞ وَمَن يَقْنُتْ مِنكُنَّ لِلَّهِ وَرَسُولِهِۦ وَتَعْمَلْ صَٰلِحًۭا نُّؤْتِهَآ أَجْرَهَا مَرَّتَيْنِ وَأَعْتَدْنَا لَهَا رِزْقًۭا كَرِيمًۭا ﴿٣١﴾ اور تم میں سے جو اللہ اور اُس کے رسول کی فرماں برداری کرے اور نیک عمل کرے ہم اُسے اُس کا اجر (بھی) دوگنا کر دیں گے اور ہم نے اُس کے لیے باعزت رزق تیار کر رکھا ہے۔ || ۞ وَمَن يَقْنُتْ مِنكُنَّ لِلَّهِ وَرَسُولِهِۦ وَتَعْمَلْ صَٰلِحًۭا نُّؤْتِهَآ أَجْرَهَا مَرَّتَيْنِ وَأَعْتَدْنَا لَهَا رِزْقًۭا كَرِيمًۭا ﴿٣١﴾ And whosoever of you is submissive unto Allah and His messenger and doeth right, We shall give her her reward twice over, and We have prepared for her a rich provision. (33:31) || ۞ وَمَن يَقْنُتْ مِنكُنَّ لِلَّهِ وَرَسُولِهِۦ وَتَعْمَلْ صَٰلِحًۭا نُّؤْتِهَآ أَجْرَهَا مَرَّتَيْنِ وَأَعْتَدْنَا لَهَا رِزْقًۭا كَرِيمًۭا ﴿٣١﴾ او له تاسو چې څوک د الله او د هغه رسول فرمانبرداري وکړي او نېک عمل وکړي، موږ به یې اجر (هم) دوه چنده ورکړو، او موږ د هغې لپاره د عزت رزق تیار کړی دی.
p يَٰنِسَآءَ ٱلنَّبِىِّ لَسْتُنَّ كَأَحَدٍۢ مِّنَ ٱلنِّسَآءِ ۚ إِنِ ٱتَّقَيْتُنَّ فَلَا تَخْضَعْنَ بِٱلْقَوْلِ فَيَطْمَعَ ٱلَّذِى فِى قَلْبِهِۦ مَرَضٌۭ وَقُلْنَ قَوْلًۭا مَّعْرُوفًۭا ﴿٣٢﴾ اے نبی (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کی ازواج! تم (عام) عورتوں کی طرح نہیں ہو اگر تم پرہیزگاری اختیار کرو تو (نامحرم سے) بات کرنے میں نرم لہجہ نہ رکھو کہ کہیں وہ شخص جس کے دل میں (نفاق کی) بیماری ہے لالچ کر بیٹھے اور (دستور کے مطابق) بھلی بات کہو۔ || يَٰنِسَآءَ ٱلنَّبِىِّ لَسْتُنَّ كَأَحَدٍۢ مِّنَ ٱلنِّسَآءِ ۚ إِنِ ٱتَّقَيْتُنَّ فَلَا تَخْضَعْنَ بِٱلْقَوْلِ فَيَطْمَعَ ٱلَّذِى فِى قَلْبِهِۦ مَرَضٌۭ وَقُلْنَ قَوْلًۭا مَّعْرُوفًۭا ﴿٣٢﴾ O ye wives of the Prophet! Ye are not like any other women. If ye keep your duty (to Allah), then be not soft of speech, lest he in whose heart is a disease aspire (to you), but utter customary speech. (33:32) || يَٰنِسَآءَ ٱلنَّبِىِّ لَسْتُنَّ كَأَحَدٍۢ مِّنَ ٱلنِّسَآءِ ۚ إِنِ ٱتَّقَيْتُنَّ فَلَا تَخْضَعْنَ بِٱلْقَوْلِ فَيَطْمَعَ ٱلَّذِى فِى قَلْبِهِۦ مَرَضٌۭ وَقُلْنَ قَوْلًۭا مَّعْرُوفًۭا ﴿٣٢﴾ ای د نبي (صلی الله علیه وعلی آله واصحابه وسلم) مېرمنو! تاسو د (عامو) ښځو په څېر نه یاست؛ که پرهېزګاري غوره کړئ، نو (له نامحرم سره) په خبرو کې نرم لهجه مه کوئ، هسې نه هغه څوک چې په زړه کې یې د (نفاق) ناروغي ده طمع وکړي؛ او (د دستور سره سم) ښه خبره کوئ.
p وَقَرْنَ فِى بُيُوتِكُنَّ وَلَا تَبَرَّجْنَ تَبَرُّجَ ٱلْجَٰهِلِيَّةِ ٱلْأُولَىٰ ۖ وَأَقِمْنَ ٱلصَّلَوٰةَ وَءَاتِينَ ٱلزَّكَوٰةَ وَأَطِعْنَ ٱللَّهَ وَرَسُولَهُۥٓ ۚ إِنَّمَا يُرِيدُ ٱللَّهُ لِيُذْهِبَ عَنكُمُ ٱلرِّجْسَ أَهْلَ ٱلْبَيْتِ وَيُطَهِّرَكُمْ تَطْهِيرًۭا ﴿٣٣﴾ اور اپنے گھروں میں ٹھہری رہو اور سابقہ جاہلیت کی طرح زینت کی نمائش نہ کرو اور نماز قائم کرو اور زکوٰۃ ادا کرو اور اللہ اور اُس کے رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کی اطاعت کرو بے شک اللہ یہی چاہتا ہے کہ تم سے (ہر قسم کی) ناپاکی دور کر دے اے (نبی صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم کے) اہلِ بیت! اور تمھیں خوب اچھی طرح پاکیزگی عطا فرمائے۔ || وَقَرْنَ فِى بُيُوتِكُنَّ وَلَا تَبَرَّجْنَ تَبَرُّجَ ٱلْجَٰهِلِيَّةِ ٱلْأُولَىٰ ۖ وَأَقِمْنَ ٱلصَّلَوٰةَ وَءَاتِينَ ٱلزَّكَوٰةَ وَأَطِعْنَ ٱللَّهَ وَرَسُولَهُۥٓ ۚ إِنَّمَا يُرِيدُ ٱللَّهُ لِيُذْهِبَ عَنكُمُ ٱلرِّجْسَ أَهْلَ ٱلْبَيْتِ وَيُطَهِّرَكُمْ تَطْهِيرًۭا ﴿٣٣﴾ And stay in your houses. Bedizen not yourselves with the bedizenment of the Time of Ignorance. Be regular in prayer, and pay the poor-due, and obey Allah and His messenger. Allah's wish is but to remove uncleanness far from you, O Folk of the Household, and cleanse you with a thorough cleansing. (33:33) || وَقَرْنَ فِى بُيُوتِكُنَّ وَلَا تَبَرَّجْنَ تَبَرُّجَ ٱلْجَٰهِلِيَّةِ ٱلْأُولَىٰ ۖ وَأَقِمْنَ ٱلصَّلَوٰةَ وَءَاتِينَ ٱلزَّكَوٰةَ وَأَطِعْنَ ٱللَّهَ وَرَسُولَهُۥٓ ۚ إِنَّمَا يُرِيدُ ٱللَّهُ لِيُذْهِبَ عَنكُمُ ٱلرِّجْسَ أَهْلَ ٱلْبَيْتِ وَيُطَهِّرَكُمْ تَطْهِيرًۭا ﴿٣٣﴾ او په خپلو کورونو کې پاتې شئ، او د پخواني جاهلیت په څېر د ښکلا نندارې مه کوئ، او لمونځ قائم کړئ، زکات ورکړئ، او د الله او د هغه رسول (صلی الله علیه وعلی آله واصحابه وسلم) اطاعت وکړئ؛ بې‌شکه الله همدا غواړي چې له تاسو (هر ډول) ناپاکي لرې کړي، ای د (نبي صلی الله علیه وعلی آله واصحابه وسلم) اهل بیته! او تاسو ته ښه پوره پاکي درکړي.
p وَٱذْكُرْنَ مَا يُتْلَىٰ فِى بُيُوتِكُنَّ مِنْ ءَايَٰتِ ٱللَّهِ وَٱلْحِكْمَةِ ۚ إِنَّ ٱللَّهَ كَانَ لَطِيفًا خَبِيرًا ﴿٣٤﴾ اور یاد رکھو جو (بھی) تمھارے گھروں میں تلاوت کی جاتی ہیں اللہ کی آیات اور حکمت (کی باتیں) بے شک اللہ بہت باریک بین خوب باخبر ہے۔ || وَٱذْكُرْنَ مَا يُتْلَىٰ فِى بُيُوتِكُنَّ مِنْ ءَايَٰتِ ٱللَّهِ وَٱلْحِكْمَةِ ۚ إِنَّ ٱللَّهَ كَانَ لَطِيفًا خَبِيرًا ﴿٣٤﴾ And bear in mind that which is recited in your houses of the revelations of Allah and wisdom. Lo! Allah is Subtile, Aware. (33:34) || وَٱذْكُرْنَ مَا يُتْلَىٰ فِى بُيُوتِكُنَّ مِنْ ءَايَٰتِ ٱللَّهِ وَٱلْحِكْمَةِ ۚ إِنَّ ٱللَّهَ كَانَ لَطِيفًا خَبِيرًا ﴿٣٤﴾ او په یاد ولرئ هغه چې ستاسو په کورونو کې تلاوت کېږي، د الله آیتونه او د حکمت (خبرې)؛ بې‌شکه الله ډېر باریک‌بین، ښه خبر دی.
