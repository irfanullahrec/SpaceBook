p إِنَّمَا ٱلسَّبِيلُ عَلَى ٱلَّذِينَ يَظْلِمُونَ ٱلنَّاسَ وَيَبْغُونَ فِى ٱلْأَرْضِ بِغَيْرِ ٱلْحَقِّ ۚ أُو۟لَٰٓئِكَ لَهُمْ عَذَابٌ أَلِيمٌۭ ﴿٤٢﴾ الزام تو ان ہی پر ہے جو لوگوں پر ظلم کرتے ہیں اور زمین میں ناحق سرکشی کرتے ہیں یہی لوگ ہیں جن کے لیے دردناک عذاب ہے۔ || إِنَّمَا ٱلسَّبِيلُ عَلَى ٱلَّذِينَ يَظْلِمُونَ ٱلنَّاسَ وَيَبْغُونَ فِى ٱلْأَرْضِ بِغَيْرِ ٱلْحَقِّ ۚ أُو۟لَٰٓئِكَ لَهُمْ عَذَابٌ أَلِيمٌۭ ﴿٤٢﴾ The way (of blame) is only against those who oppress mankind, and wrongfully rebel in the earth. For such there is a painful doom. (42:42) || إِنَّمَا ٱلسَّبِيلُ عَلَى ٱلَّذِينَ يَظْلِمُونَ ٱلنَّاسَ وَيَبْغُونَ فِى ٱلْأَرْضِ بِغَيْرِ ٱلْحَقِّ ۚ أُو۟لَٰٓئِكَ لَهُمْ عَذَابٌ أَلِيمٌۭ ﴿٤٢﴾ ملامتیا خو یوازې پر هغو ده چې پر خلکو ظلم کوي او په ځمکه کې په ناحقه سرکشي کوي؛ همدا هغه خلک دي چې د هغوی لپاره دردناک عذاب دی.
p وَلَمَن صَبَرَ وَغَفَرَ إِنَّ ذَٰلِكَ لَمِنْ عَزْمِ ٱلْأُمُورِ ﴿٤٣﴾ اور البتہ جو شخص صبر کرے اور معاف کر دے تو بے شک یہ ضرور بڑی ہمت کے کام ہیں۔ || وَلَمَن صَبَرَ وَغَفَرَ إِنَّ ذَٰلِكَ لَمِنْ عَزْمِ ٱلْأُمُورِ ﴿٤٣﴾ And verily whoso is patient and forgiveth - lo! that, verily, is (of) the steadfast heart of things. (42:43) || وَلَمَن صَبَرَ وَغَفَرَ إِنَّ ذَٰلِكَ لَمِنْ عَزْمِ ٱلْأُمُورِ ﴿٤٣﴾ او البته څوک چې صبر وکړي او وبښي، نو بې‌شکه دا خامخا د لویې همت کارونه دي.
p وَمَن يُضْلِلِ ٱللَّهُ فَمَا لَهُۥ مِن وَلِىٍّۢ مِّنۢ بَعْدِهِۦ ۗ وَتَرَى ٱلظَّٰلِمِينَ لَمَّا رَأَوُا۟ ٱلْعَذَابَ يَقُولُونَ هَلْ إِلَىٰ مَرَدٍّۢ مِّن سَبِيلٍۢ ﴿٤٤﴾ اور جسے اللہ گم راہ کر دے تو اس کے بعد اس کا کوئی کارساز نہیں اور آپ ظالموں کو دیکھیں گے جب وہ عذاب دیکھ لیں گے (تو) وہ کہیں گے کیا واپسی کا بھی کوئی راستہ ہے؟ || وَمَن يُضْلِلِ ٱللَّهُ فَمَا لَهُۥ مِن وَلِىٍّۢ مِّنۢ بَعْدِهِۦ ۗ وَتَرَى ٱلظَّٰلِمِينَ لَمَّا رَأَوُا۟ ٱلْعَذَابَ يَقُولُونَ هَلْ إِلَىٰ مَرَدٍّۢ مِّن سَبِيلٍۢ ﴿٤٤﴾ He whom Allah sendeth astray, for him there is no protecting friend after Him. And thou (Muhammad) wilt see the evil-doers when they see the doom, (how) they say: Is there any way of return? (42:44) || وَمَن يُضْلِلِ ٱللَّهُ فَمَا لَهُۥ مِن وَلِىٍّۢ مِّنۢ بَعْدِهِۦ ۗ وَتَرَى ٱلظَّٰلِمِينَ لَمَّا رَأَوُا۟ ٱلْعَذَابَ يَقُولُونَ هَلْ إِلَىٰ مَرَدٍّۢ مِّن سَبِيلٍۢ ﴿٤٤﴾ او څوک چې الله ګمراه کړي، نو له هغه وروسته یې هېڅ کارساز نشته؛ او تاسو به ظالمان ووینئ چې کله عذاب وویني، (نو) وايي به: ایا د بېرته ورتګ کومه لاره شته؟
p وَتَرَىٰهُمْ يُعْرَضُونَ عَلَيْهَا خَٰشِعِينَ مِنَ ٱلذُّلِّ يَنظُرُونَ مِن طَرْفٍ خَفِىٍّۢ ۗ وَقَالَ ٱلَّذِينَ ءَامَنُوٓا۟ إِنَّ ٱلْخَٰسِرِينَ ٱلَّذِينَ خَسِرُوٓا۟ أَنفُسَهُمْ وَأَهْلِيهِمْ يَوْمَ ٱلْقِيَٰمَةِ ۗ أَلَآ إِنَّ ٱلظَّٰلِمِينَ فِى عَذَابٍۢ مُّقِيمٍۢ ﴿٤٥﴾ اور آپ انھیں دیکھیں گے کہ وہ اس (جہنم) پر پیش کیے جائیں گے ذلت سے جھکے ہوئے وہ چھپی نگاہ سے دیکھ رہے ہوں گے اور ایمان والے کہیں گے بے شک نقصان اٹھانے والے (وہی) ہیں جنھوں نے اپنے آپ کو اور اپنے گھر والوں کو قیامت کے دن نقصان میں ڈالا آگاہ ہو جاؤ بے شک ظالم ہمیشہ رہنے والے عذاب میں ہوں گے۔ || وَتَرَىٰهُمْ يُعْرَضُونَ عَلَيْهَا خَٰشِعِينَ مِنَ ٱلذُّلِّ يَنظُرُونَ مِن طَرْفٍ خَفِىٍّۢ ۗ وَقَالَ ٱلَّذِينَ ءَامَنُوٓا۟ إِنَّ ٱلْخَٰسِرِينَ ٱلَّذِينَ خَسِرُوٓا۟ أَنفُسَهُمْ وَأَهْلِيهِمْ يَوْمَ ٱلْقِيَٰمَةِ ۗ أَلَآ إِنَّ ٱلظَّٰلِمِينَ فِى عَذَابٍۢ مُّقِيمٍۢ ﴿٤٥﴾ And thou wilt see them exposed to (the Fire), made humble by disgrace, and looking with veiled eyes. And those who believe will say: Lo! the (eternal) losers are they who lose themselves and their housefolk on the Day of Resurrection. Lo! are not the wrong-doers in perpetual torment? (42:45) || وَتَرَىٰهُمْ يُعْرَضُونَ عَلَيْهَا خَٰشِعِينَ مِنَ ٱلذُّلِّ يَنظُرُونَ مِن طَرْفٍ خَفِىٍّۢ ۗ وَقَالَ ٱلَّذِينَ ءَامَنُوٓا۟ إِنَّ ٱلْخَٰسِرِينَ ٱلَّذِينَ خَسِرُوٓا۟ أَنفُسَهُمْ وَأَهْلِيهِمْ يَوْمَ ٱلْقِيَٰمَةِ ۗ أَلَآ إِنَّ ٱلظَّٰلِمِينَ فِى عَذَابٍۢ مُّقِيمٍۢ ﴿٤٥﴾ او تاسو به هغوی ووینئ چې پر هغه (دوزخ) به وړاندې کړای شي په ذلت ټیټ شوي، په پټه نظر به ګوري؛ او مؤمنان به ووايي: بې‌شکه زیانمن (همغه) دي چې د قیامت په ورځ یې ځانونه او کورنۍ په زیان کې واچولې؛ خبر اوسئ! بې‌شکه ظالمان به په تلپاتې عذاب کې وي.
p وَمَا كَانَ لَهُم مِّنْ أَوْلِيَآءَ يَنصُرُونَهُم مِّن دُونِ ٱللَّهِ ۗ وَمَن يُضْلِلِ ٱللَّهُ فَمَا لَهُۥ مِن سَبِيلٍ ﴿٤٦﴾ اور ان کے لیے اللہ کے سوا کوئی نہ ہوگا دوستوں میں سے جو ان کی مدد کر سکیں اور جسے اللہ گم راہ کر دے اس کے لیے ہدایت کا کوئی راستہ نہیں۔ || وَمَا كَانَ لَهُم مِّنْ أَوْلِيَآءَ يَنصُرُونَهُم مِّن دُونِ ٱللَّهِ ۗ وَمَن يُضْلِلِ ٱللَّهُ فَمَا لَهُۥ مِن سَبِيلٍ ﴿٤٦﴾ And they will have no protecting friends to help them instead of Allah. He whom Allah sendeth astray, for him there is no road. (42:46) || وَمَا كَانَ لَهُم مِّنْ أَوْلِيَآءَ يَنصُرُونَهُم مِّن دُونِ ٱللَّهِ ۗ وَمَن يُضْلِلِ ٱللَّهُ فَمَا لَهُۥ مِن سَبِيلٍ ﴿٤٦﴾ او د هغوی لپاره به له الله پرته له دوستانو هېڅوک نه وي چې مرسته یې وکړي؛ او څوک چې الله ګمراه کړي، د هغه لپاره د هدایت هېڅ لاره نشته.
p ٱسْتَجِيبُوا۟ لِرَبِّكُم مِّن قَبْلِ أَن يَأْتِىَ يَوْمٌۭ لَّا مَرَدَّ لَهُۥ مِنَ ٱللَّهِ ۚ مَا لَكُم مِّن مَّلْجَإٍۢ يَوْمَئِذٍۢ وَمَا لَكُم مِّن نَّكِيرٍۢ ﴿٤٧﴾ (اے لوگو!) تم اپنے رب کا حکم مان لو اس دن کے آنے سے پہلے جس کے ٹلنے کی اللہ کی طرف سے کوئی صورت نہیں اس دن تمھارے لیے نہ کوئی پناہ گاہ ہوگی اور نہ تمھارے لیے (گناہوں سے) انکار کی کوئی صورت ہوگی۔ || ٱسْتَجِيبُوا۟ لِرَبِّكُم مِّن قَبْلِ أَن يَأْتِىَ يَوْمٌۭ لَّا مَرَدَّ لَهُۥ مِنَ ٱللَّهِ ۚ مَا لَكُم مِّن مَّلْجَإٍۢ يَوْمَئِذٍۢ وَمَا لَكُم مِّن نَّكِيرٍۢ ﴿٤٧﴾ Answer the call of your Lord before there cometh unto you from Allah a Day which there is no averting. Ye have no refuge on that Day, nor have a ye any (power of) refusal. (42:47) || ٱسْتَجِيبُوا۟ لِرَبِّكُم مِّن قَبْلِ أَن يَأْتِىَ يَوْمٌۭ لَّا مَرَدَّ لَهُۥ مِنَ ٱللَّهِ ۚ مَا لَكُم مِّن مَّلْجَإٍۢ يَوْمَئِذٍۢ وَمَا لَكُم مِّن نَّكِيرٍۢ ﴿٤٧﴾ (ای خلکو!) د خپل رب حکم ومنئ مخکې له دې چې هغه ورځ راشي چې د الله له لوري یې د اوښتو هېڅ لاره نشته؛ په هغه ورځ به ستاسو لپاره نه کوم پناه ځای وي او نه به ستاسو لپاره د (ګناهونو) د انکار کومه لاره وي.
p فَإِنْ أَعْرَضُوا۟ فَمَآ أَرْسَلْنَٰكَ عَلَيْهِمْ حَفِيظًا ۖ إِنْ عَلَيْكَ إِلَّا ٱلْبَلَٰغُ ۗ وَإِنَّآ إِذَآ أَذَقْنَا ٱلْإِنسَٰنَ مِنَّا رَحْمَةًۭ فَرِحَ بِهَا ۖ وَإِن تُصِبْهُمْ سَيِّئَةٌۢ بِمَا قَدَّمَتْ أَيْدِيهِمْ فَإِنَّ ٱلْإِنسَٰنَ كَفُورٌۭ ﴿٤٨﴾ پھر اگر یہ منہ پھیر لیں تو ہم نے آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کو ان کا نگہبان بنا کر نہیں بھیجا آپ کے ذمہ تو صرف (پیغامِ حق کو) پہنچا دینا ہے اور جب ہم انسان کو اپنی رحمت کا مزا چکھاتے ہیں تو وہ اس سے خوش ہو جاتا ہے اور اگر انھیں کوئی برائی پہنچتی ہے ان (گناہوں) کی وجہ سے جو ان کے ہاتھوں نے آگے بھیجے (تو شکایت کرتے ہیں) تو بے شک انسان بڑا ناشکرا ہے۔ || فَإِنْ أَعْرَضُوا۟ فَمَآ أَرْسَلْنَٰكَ عَلَيْهِمْ حَفِيظًا ۖ إِنْ عَلَيْكَ إِلَّا ٱلْبَلَٰغُ ۗ وَإِنَّآ إِذَآ أَذَقْنَا ٱلْإِنسَٰنَ مِنَّا رَحْمَةًۭ فَرِحَ بِهَا ۖ وَإِن تُصِبْهُمْ سَيِّئَةٌۢ بِمَا قَدَّمَتْ أَيْدِيهِمْ فَإِنَّ ٱلْإِنسَٰنَ كَفُورٌۭ ﴿٤٨﴾ But if they are averse, We have not sent thee as a warder over them. Thine is only to convey (the message). And lo! when We cause man to taste of mercy from Us he exulteth therefor. And if some evil striketh them because of that which their own hands have sent before, then lo! man is an ingrate. (42:48) || فَإِنْ أَعْرَضُوا۟ فَمَآ أَرْسَلْنَٰكَ عَلَيْهِمْ حَفِيظًا ۖ إِنْ عَلَيْكَ إِلَّا ٱلْبَلَٰغُ ۗ وَإِنَّآ إِذَآ أَذَقْنَا ٱلْإِنسَٰنَ مِنَّا رَحْمَةًۭ فَرِحَ بِهَا ۖ وَإِن تُصِبْهُمْ سَيِّئَةٌۢ بِمَا قَدَّمَتْ أَيْدِيهِمْ فَإِنَّ ٱلْإِنسَٰنَ كَفُورٌۭ ﴿٤٨﴾ بیا که دوی مخ واړوي، نو موږ تاسو (صلی الله علیه وعلی آله واصحابه وسلم) د دوی څارونکی نه یاست لېږلي؛ ستاسو پر غاړه خو یوازې د (حق پیغام) رسول دي؛ او کله چې موږ انسان ته د خپل رحمت خوند وڅکوو، نو پرې خوشاله شي؛ او که هغوی ته کومه بدي ورسېږي د هغو (ګناهونو) له امله چې د هغوی لاسونو مخکې لېږلي، (نو شکایت کوي)؛ نو بې‌شکه انسان ډېر ناشکره دی.
p لِّلَّهِ مُلْكُ ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ ۚ يَخْلُقُ مَا يَشَآءُ ۚ يَهَبُ لِمَن يَشَآءُ إِنَٰثًۭا وَيَهَبُ لِمَن يَشَآءُ ٱلذُّكُورَ ﴿٤٩﴾ اللہ ہی کے لیے آسمانوں اور زمینوں کی بادشاہت ہے وہ جو چاہتا ہے پیدا فرماتا ہے جس کو چاہتا ہے بیٹیاں عطا فرماتا ہے اور جس کو چاہتا ہے بیٹے عطا فرماتا ہے۔ || لِّلَّهِ مُلْكُ ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ ۚ يَخْلُقُ مَا يَشَآءُ ۚ يَهَبُ لِمَن يَشَآءُ إِنَٰثًۭا وَيَهَبُ لِمَن يَشَآءُ ٱلذُّكُورَ ﴿٤٩﴾ Unto Allah belongeth the Sovereignty of the heavens and the earth. He createth what He will. He bestoweth female (offspring) upon whom He will, and bestoweth male (offspring) upon whom He will; (42:49) || لِّلَّهِ مُلْكُ ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ ۚ يَخْلُقُ مَا يَشَآءُ ۚ يَهَبُ لِمَن يَشَآءُ إِنَٰثًۭا وَيَهَبُ لِمَن يَشَآءُ ٱلذُّكُورَ ﴿٤٩﴾ یوازې د الله لپاره د آسمانونو او ځمکو پاچاهي ده؛ څه چې وغواړي پیدا کوي یې؛ چا ته چې وغواړي لوڼې ورکوي او چا ته چې وغواړي زامن ورکوي.
