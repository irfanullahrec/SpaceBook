p ثُمَّ ٱسْتَوَىٰٓ إِلَى ٱلسَّمَآءِ وَهِىَ دُخَانٌۭ فَقَالَ لَهَا وَلِلْأَرْضِ ٱئْتِيَا طَوْعًا أَوْ كَرْهًۭا قَالَتَآ أَتَيْنَا طَآئِعِينَ ﴿١١﴾ پھر وہ آسمان کی طرف متوجہ ہوا اور وہ اس وقت دھواں تھا پھر اس سے اور زمین سے کہا کہ تم دونوں (وجود میں) آؤ خوشی یا ناخوشی سے انھوں نے کہا کہ ہم فرماں بردار بن کر حاضر ہو گئے۔ || ثُمَّ ٱسْتَوَىٰٓ إِلَى ٱلسَّمَآءِ وَهِىَ دُخَانٌۭ فَقَالَ لَهَا وَلِلْأَرْضِ ٱئْتِيَا طَوْعًا أَوْ كَرْهًۭا قَالَتَآ أَتَيْنَا طَآئِعِينَ ﴿١١﴾ Then turned He to the heaven when it was smoke, and said unto it and unto the earth: Come both of you, willingly or loth. They said: We come, obedient. (41:11) || ثُمَّ ٱسْتَوَىٰٓ إِلَى ٱلسَّمَآءِ وَهِىَ دُخَانٌۭ فَقَالَ لَهَا وَلِلْأَرْضِ ٱئْتِيَا طَوْعًا أَوْ كَرْهًۭا قَالَتَآ أَتَيْنَا طَآئِعِينَ ﴿١١﴾ بیا هغه آسمان ته متوجه شو چې هغه مهال لوګی و، بیا یې هغه او ځمکې ته وویل: تاسو دواړه (وجود ته) راشئ په خوښۍ یا ناخوښۍ؛ هغوی وویل: موږ فرمانبرداره حاضر شو.
p فَقَضَىٰهُنَّ سَبْعَ سَمَٰوَاتٍۢ فِى يَوْمَيْنِ وَأَوْحَىٰ فِى كُلِّ سَمَآءٍ أَمْرَهَا ۚ وَزَيَّنَّا ٱلسَّمَآءَ ٱلدُّنْيَا بِمَصَٰبِيحَ وَحِفْظًۭا ۚ ذَٰلِكَ تَقْدِيرُ ٱلْعَزِيزِ ٱلْعَلِيمِ ﴿١٢﴾ پھر اس نے دو دن میں سات آسمان بنا دیے اور ہر آسمان میں اس کے کام کی وحی فرمائی اور ہم نے دنیاوی آسمان کو چراغوں (ستاروں) سے زینت دی اور اس کو محفوظ فرما دیا یہ بہت غالب خوب جاننے والے کا (مقرر کردہ) اندازہ ہے۔ || فَقَضَىٰهُنَّ سَبْعَ سَمَٰوَاتٍۢ فِى يَوْمَيْنِ وَأَوْحَىٰ فِى كُلِّ سَمَآءٍ أَمْرَهَا ۚ وَزَيَّنَّا ٱلسَّمَآءَ ٱلدُّنْيَا بِمَصَٰبِيحَ وَحِفْظًۭا ۚ ذَٰلِكَ تَقْدِيرُ ٱلْعَزِيزِ ٱلْعَلِيمِ ﴿١٢﴾ Then He ordained them seven heavens in two Days and inspired in each heaven its mandate; and We decked the nether heaven with lamps, and rendered it inviolable. That is the measuring of the Mighty, the Knower. (41:12) || فَقَضَىٰهُنَّ سَبْعَ سَمَٰوَاتٍۢ فِى يَوْمَيْنِ وَأَوْحَىٰ فِى كُلِّ سَمَآءٍ أَمْرَهَا ۚ وَزَيَّنَّا ٱلسَّمَآءَ ٱلدُّنْيَا بِمَصَٰبِيحَ وَحِفْظًۭا ۚ ذَٰلِكَ تَقْدِيرُ ٱلْعَزِيزِ ٱلْعَلِيمِ ﴿١٢﴾ بیا یې په دوو ورځو کې اووه آسمانونه جوړ کړل او په هر آسمان کې یې د هغه د کار وحې وکړه؛ او موږ نږدې (دنیا) آسمان په څراغونو (ستورو) ښکلی کړ او خوندي مو کړ؛ دا د ډېر غالب، ښه پوه (ټاکلې) اندازه ده.
p فَإِنْ أَعْرَضُوا۟ فَقُلْ أَنذَرْتُكُمْ صَٰعِقَةًۭ مِّثْلَ صَٰعِقَةِ عَادٍۢ وَثَمُودَ ﴿١٣﴾ پھر اگر وہ منہ موڑ لیں تو آپ فرما دیجیے کہ میں تمھیں کڑک (کے عذاب) سے ڈراتا ہوں جیسی کڑک قوم عاد اور قوم ثمود (پر آئی تھی)۔ || فَإِنْ أَعْرَضُوا۟ فَقُلْ أَنذَرْتُكُمْ صَٰعِقَةًۭ مِّثْلَ صَٰعِقَةِ عَادٍۢ وَثَمُودَ ﴿١٣﴾ But if they turn away, then say: I warn you of a thunderbolt like the thunderbolt (which fell of old upon the tribes) of A'ad and Thamud; (41:13) || فَإِنْ أَعْرَضُوا۟ فَقُلْ أَنذَرْتُكُمْ صَٰعِقَةًۭ مِّثْلَ صَٰعِقَةِ عَادٍۢ وَثَمُودَ ﴿١٣﴾ بیا که هغوی مخ واړوي، نو تاسو ووایئ: زه تاسو له تندر (د عذاب) وېروم لکه هغه تندر چې پر عاد قوم او ثمود قوم (راغلی و).
p إِذْ جَآءَتْهُمُ ٱلرُّسُلُ مِنۢ بَيْنِ أَيْدِيهِمْ وَمِنْ خَلْفِهِمْ أَلَّا تَعْبُدُوٓا۟ إِلَّا ٱللَّهَ ۖ قَالُوا۟ لَوْ شَآءَ رَبُّنَا لَأَنزَلَ مَلَٰٓئِكَةًۭ فَإِنَّا بِمَآ أُرْسِلْتُم بِهِۦ كَٰفِرُونَ ﴿١٤﴾ جب ان کے پاس رسول آئے ان کے آگے سے اور ان کے پیچھے سے (اس پیغام کے ساتھ) کہ اللہ کے سوا کسی کی عبادت نہ کرو تو لوگوں نے کہا اگر ہمارا رب چاہتا تو فرشتوں کو نازل فرما دیتا تو بے شک جو کچھ تم دے کر بھیجے گئے ہو ہم اس کا انکار کرنے والے ہیں۔ || إِذْ جَآءَتْهُمُ ٱلرُّسُلُ مِنۢ بَيْنِ أَيْدِيهِمْ وَمِنْ خَلْفِهِمْ أَلَّا تَعْبُدُوٓا۟ إِلَّا ٱللَّهَ ۖ قَالُوا۟ لَوْ شَآءَ رَبُّنَا لَأَنزَلَ مَلَٰٓئِكَةًۭ فَإِنَّا بِمَآ أُرْسِلْتُم بِهِۦ كَٰفِرُونَ ﴿١٤﴾ When their messengers came unto them from before them and behind them, saying: Worship none but Allah! they said: If our Lord had willed, He surely would have sent down angels (unto us), so lo! we are disbelievers in that wherewith ye have been sent. (41:14) || إِذْ جَآءَتْهُمُ ٱلرُّسُلُ مِنۢ بَيْنِ أَيْدِيهِمْ وَمِنْ خَلْفِهِمْ أَلَّا تَعْبُدُوٓا۟ إِلَّا ٱللَّهَ ۖ قَالُوا۟ لَوْ شَآءَ رَبُّنَا لَأَنزَلَ مَلَٰٓئِكَةًۭ فَإِنَّا بِمَآ أُرْسِلْتُم بِهِۦ كَٰفِرُونَ ﴿١٤﴾ کله چې هغوی ته رسولان د هغوی له مخې او له شا راغلل (په دې پیغام) چې له الله پرته د هېچا عبادت مه کوئ، نو خلکو وویل: که زموږ رب غوښتي وای، پرښتې به یې نازلې کړې وای؛ نو بې‌شکه تاسو چې په څه لېږل شوي یاست، موږ یې منکران یو.
p فَأَمَّا عَادٌۭ فَٱسْتَكْبَرُوا۟ فِى ٱلْأَرْضِ بِغَيْرِ ٱلْحَقِّ وَقَالُوا۟ مَنْ أَشَدُّ مِنَّا قُوَّةً ۖ أَوَلَمْ يَرَوْا۟ أَنَّ ٱللَّهَ ٱلَّذِى خَلَقَهُمْ هُوَ أَشَدُّ مِنْهُمْ قُوَّةًۭ ۖ وَكَانُوا۟ بِـَٔايَٰتِنَا يَجْحَدُونَ ﴿١٥﴾ پھر جو (قوم) عاد تھے تو انھوں نے زمین میں ناحق تکبر کیا اور انھوں نے کہا کون قوت میں ہم سے بڑھ کر ہے؟ کیا انھوں نے نہیں دیکھا بے شک اللہ جس نے انھیں پیدا فرمایا ہے وہ قوت میں ان سے کہیں زیادہ بڑھ کر ہے اور وہ ہماری آیات کا انکار کرتے رہے۔ || فَأَمَّا عَادٌۭ فَٱسْتَكْبَرُوا۟ فِى ٱلْأَرْضِ بِغَيْرِ ٱلْحَقِّ وَقَالُوا۟ مَنْ أَشَدُّ مِنَّا قُوَّةً ۖ أَوَلَمْ يَرَوْا۟ أَنَّ ٱللَّهَ ٱلَّذِى خَلَقَهُمْ هُوَ أَشَدُّ مِنْهُمْ قُوَّةًۭ ۖ وَكَانُوا۟ بِـَٔايَٰتِنَا يَجْحَدُونَ ﴿١٥﴾ As for A'ad, they were arrogant in the land without right, and they said: Who is mightier than us in power? Could they not see that Allah Who created them, He was mightier than them in power? And they denied Our revelations. (41:15) || فَأَمَّا عَادٌۭ فَٱسْتَكْبَرُوا۟ فِى ٱلْأَرْضِ بِغَيْرِ ٱلْحَقِّ وَقَالُوا۟ مَنْ أَشَدُّ مِنَّا قُوَّةً ۖ أَوَلَمْ يَرَوْا۟ أَنَّ ٱللَّهَ ٱلَّذِى خَلَقَهُمْ هُوَ أَشَدُّ مِنْهُمْ قُوَّةًۭ ۖ وَكَانُوا۟ بِـَٔايَٰتِنَا يَجْحَدُونَ ﴿١٥﴾ بیا چې کوم (قوم) عاد و، هغوی په ځمکه کې په ناحقه لویي وکړه او ویې ویل: څوک په ځواک کې له موږ زیات دی؟ ایا هغوی ونه لیدل چې بې‌شکه الله چې هغوی یې پیدا کړي، په ځواک کې له هغوی ډېر زیات دی؟ او هغوی زموږ د آیتونو انکار کاوه.
p فَأَرْسَلْنَا عَلَيْهِمْ رِيحًۭا صَرْصَرًۭا فِىٓ أَيَّامٍۢ نَّحِسَاتٍۢ لِّنُذِيقَهُمْ عَذَابَ ٱلْخِزْىِ فِى ٱلْحَيَوٰةِ ٱلدُّنْيَا ۖ وَلَعَذَابُ ٱلْءَاخِرَةِ أَخْزَىٰ ۖ وَهُمْ لَا يُنصَرُونَ ﴿١٦﴾ پھر ہم نے ان پر تیز آندھی والی ہوا بھیجی نحوست کے دنوں میں تاکہ ہم انھیں رسوائی کے عذاب کا مزہ چکھائیں دنیا کی زندگی میں اور آخرت کا عذاب زیادہ رسوا کرنے والا ہے اور ان کی کوئی مدد نہیں کی جائے گی۔ || فَأَرْسَلْنَا عَلَيْهِمْ رِيحًۭا صَرْصَرًۭا فِىٓ أَيَّامٍۢ نَّحِسَاتٍۢ لِّنُذِيقَهُمْ عَذَابَ ٱلْخِزْىِ فِى ٱلْحَيَوٰةِ ٱلدُّنْيَا ۖ وَلَعَذَابُ ٱلْءَاخِرَةِ أَخْزَىٰ ۖ وَهُمْ لَا يُنصَرُونَ ﴿١٦﴾ Therefor We let loose on them a raging wind in evil days, that We might make them taste the torment of disgrace in the life of the world. And verily the doom of the Hereafter will be more shameful, and they will not be helped. (41:16) || فَأَرْسَلْنَا عَلَيْهِمْ رِيحًۭا صَرْصَرًۭا فِىٓ أَيَّامٍۢ نَّحِسَاتٍۢ لِّنُذِيقَهُمْ عَذَابَ ٱلْخِزْىِ فِى ٱلْحَيَوٰةِ ٱلدُّنْيَا ۖ وَلَعَذَابُ ٱلْءَاخِرَةِ أَخْزَىٰ ۖ وَهُمْ لَا يُنصَرُونَ ﴿١٦﴾ بیا موږ پر هغوی په شومو ورځو کې تېز توپاني باد ولېږه تر څو هغوی ته د دنیا په ژوند کې د رسوایۍ د عذاب خوند وڅکوو؛ او د آخرت عذاب ډېر رسوا کوونکی دی، او له هغوی سره به هېڅ مرسته ونه شي.
p وَأَمَّا ثَمُودُ فَهَدَيْنَٰهُمْ فَٱسْتَحَبُّوا۟ ٱلْعَمَىٰ عَلَى ٱلْهُدَىٰ فَأَخَذَتْهُمْ صَٰعِقَةُ ٱلْعَذَابِ ٱلْهُونِ بِمَا كَانُوا۟ يَكْسِبُونَ ﴿١٧﴾ اور جو (قوم) ثمود تھے تو ہم نے انھیں (سیدھا) راستہ دکھایا مگر انھوں نے ہدایت کے مقابلہ میں اندھا رہنے کو پسند کیا تو انھیں ذلت کے عذاب کی کڑک نے پکڑ لیا ان (اعمال) کے بدلہ جو وہ کیا کرتے تھے۔ || وَأَمَّا ثَمُودُ فَهَدَيْنَٰهُمْ فَٱسْتَحَبُّوا۟ ٱلْعَمَىٰ عَلَى ٱلْهُدَىٰ فَأَخَذَتْهُمْ صَٰعِقَةُ ٱلْعَذَابِ ٱلْهُونِ بِمَا كَانُوا۟ يَكْسِبُونَ ﴿١٧﴾ And as for Thamud, We gave them guidance, but they preferred blindness to the guidance, so the bolt of the doom of humiliation overtook them because of what they used to earn. (41:17) || وَأَمَّا ثَمُودُ فَهَدَيْنَٰهُمْ فَٱسْتَحَبُّوا۟ ٱلْعَمَىٰ عَلَى ٱلْهُدَىٰ فَأَخَذَتْهُمْ صَٰعِقَةُ ٱلْعَذَابِ ٱلْهُونِ بِمَا كَانُوا۟ يَكْسِبُونَ ﴿١٧﴾ او چې کوم (قوم) ثمود و، نو موږ هغوی ته (سمه) لار وښوده، خو هغوی د هدایت په پرتله ړوندوالی غوره کړ، نو د ذلت د عذاب تندر ونیول د هغو (عملونو) په بدل کې چې کول یې.
p وَنَجَّيْنَا ٱلَّذِينَ ءَامَنُوا۟ وَكَانُوا۟ يَتَّقُونَ ﴿١٨﴾ اور ہم نے ان لوگوں کو نجات دی جو ایمان لائے اور ڈرتے رہتے تھے۔ || وَنَجَّيْنَا ٱلَّذِينَ ءَامَنُوا۟ وَكَانُوا۟ يَتَّقُونَ ﴿١٨﴾ And We delivered those who believed and used to keep their duty to Allah. (41:18) || وَنَجَّيْنَا ٱلَّذِينَ ءَامَنُوا۟ وَكَانُوا۟ يَتَّقُونَ ﴿١٨﴾ او موږ هغه کسان وژغورل چې ایمان یې راوړی و او وېرېدل به.
