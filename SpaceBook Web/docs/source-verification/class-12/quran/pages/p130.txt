p فَلَنُذِيقَنَّ ٱلَّذِينَ كَفَرُوا۟ عَذَابًۭا شَدِيدًۭا وَلَنَجْزِيَنَّهُمْ أَسْوَأَ ٱلَّذِى كَانُوا۟ يَعْمَلُونَ ﴿٢٧﴾ تو ہم کافروں کو سخت عذاب کا مزہ ضرور چکھائیں گے اور ضرور انھیں ان کے بدترین اعمال کا بدلہ دیں گے۔ || فَلَنُذِيقَنَّ ٱلَّذِينَ كَفَرُوا۟ عَذَابًۭا شَدِيدًۭا وَلَنَجْزِيَنَّهُمْ أَسْوَأَ ٱلَّذِى كَانُوا۟ يَعْمَلُونَ ﴿٢٧﴾ But verily We shall cause those who disbelieve to taste an awful doom, and verily We shall requite them the worst of what they used to do. (41:27) || فَلَنُذِيقَنَّ ٱلَّذِينَ كَفَرُوا۟ عَذَابًۭا شَدِيدًۭا وَلَنَجْزِيَنَّهُمْ أَسْوَأَ ٱلَّذِى كَانُوا۟ يَعْمَلُونَ ﴿٢٧﴾ نو موږ به کافرانو ته خامخا د سخت عذاب خوند وڅکوو او خامخا به یې د هغوی د تر ټولو بدو عملونو بدله ورکړو.
p ذَٰلِكَ جَزَآءُ أَعْدَآءِ ٱللَّهِ ٱلنَّارُ ۖ لَهُمْ فِيهَا دَارُ ٱلْخُلْدِ ۖ جَزَآءًۢ بِمَا كَانُوا۟ بِـَٔايَٰتِنَا يَجْحَدُونَ ﴿٢٨﴾ یہ (جہنم کی) آگ اللہ کے دشمنوں کا بدلہ ہے ان کے لیے اسی میں ہمیشہ رہنے کا گھر ہے یہ اس کا بدلہ ہے جو وہ ہماری آیتوں کا انکار کیا کرتے تھے۔ || ذَٰلِكَ جَزَآءُ أَعْدَآءِ ٱللَّهِ ٱلنَّارُ ۖ لَهُمْ فِيهَا دَارُ ٱلْخُلْدِ ۖ جَزَآءًۢ بِمَا كَانُوا۟ بِـَٔايَٰتِنَا يَجْحَدُونَ ﴿٢٨﴾ That is the reward of Allah's enemies: the Fire. Therein is their immortal home, payment forasmuch as they denied Our revelations. (41:28) || ذَٰلِكَ جَزَآءُ أَعْدَآءِ ٱللَّهِ ٱلنَّارُ ۖ لَهُمْ فِيهَا دَارُ ٱلْخُلْدِ ۖ جَزَآءًۢ بِمَا كَانُوا۟ بِـَٔايَٰتِنَا يَجْحَدُونَ ﴿٢٨﴾ دا د (دوزخ) اور د الله د دښمنانو بدله ده؛ د هغوی لپاره په هغه کې د تل پاتې کېدو کور دی؛ دا د هغه څه بدله ده چې زموږ د آیتونو انکار به یې کاوه.
p وَقَالَ ٱلَّذِينَ كَفَرُوا۟ رَبَّنَآ أَرِنَا ٱلَّذَيْنِ أَضَلَّانَا مِنَ ٱلْجِنِّ وَٱلْإِنسِ نَجْعَلْهُمَا تَحْتَ أَقْدَامِنَا لِيَكُونَا مِنَ ٱلْأَسْفَلِينَ ﴿٢٩﴾ اور کافر کہیں گے اے ہمارے رب! تو ہمیں وہ دونوں دکھا دے جنھوں نے جنات اور انسانوں میں سے ہمیں گمراہ کیا ہم انھیں اپنے پاؤں تلے (روند) ڈالیں تاکہ وہ خوب ذلیل ہوں۔ || وَقَالَ ٱلَّذِينَ كَفَرُوا۟ رَبَّنَآ أَرِنَا ٱلَّذَيْنِ أَضَلَّانَا مِنَ ٱلْجِنِّ وَٱلْإِنسِ نَجْعَلْهُمَا تَحْتَ أَقْدَامِنَا لِيَكُونَا مِنَ ٱلْأَسْفَلِينَ ﴿٢٩﴾ And those who disbelieve will say: Our Lord! Show us those who beguiled us of the jinn and humankind. We will place them underneath our feet that they may be among the nethermost. (41:29) || وَقَالَ ٱلَّذِينَ كَفَرُوا۟ رَبَّنَآ أَرِنَا ٱلَّذَيْنِ أَضَلَّانَا مِنَ ٱلْجِنِّ وَٱلْإِنسِ نَجْعَلْهُمَا تَحْتَ أَقْدَامِنَا لِيَكُونَا مِنَ ٱلْأَسْفَلِينَ ﴿٢٩﴾ او کافران به ووايي: ای زموږ ربه! موږ ته هغه دواړه وښیه چې له پېریانو او انسانانو یې موږ ګمراه کړي، چې د خپلو پښو لاندې یې (لتاړ) کړو تر څو ښه ذلیل شي.
p إِنَّ ٱلَّذِينَ قَالُوا۟ رَبُّنَا ٱللَّهُ ثُمَّ ٱسْتَقَٰمُوا۟ تَتَنَزَّلُ عَلَيْهِمُ ٱلْمَلَٰٓئِكَةُ أَلَّا تَخَافُوا۟ وَلَا تَحْزَنُوا۟ وَأَبْشِرُوا۟ بِٱلْجَنَّةِ ٱلَّتِى كُنتُمْ تُوعَدُونَ ﴿٣٠﴾ بے شک جن لوگوں نے کہا اللہ ہی ہمارا رب ہے پھر وہ (اس پر) قائم رہے ان پر فرشتے نازل ہوتے ہیں (یہ کہتے ہوئے) کہ نہ تم خوف کرو اور نہ غم کرو اور اس جنت کی بشارت پر خوش ہو جاؤ جس کا تم سے وعدہ کیا جاتا تھا۔ || إِنَّ ٱلَّذِينَ قَالُوا۟ رَبُّنَا ٱللَّهُ ثُمَّ ٱسْتَقَٰمُوا۟ تَتَنَزَّلُ عَلَيْهِمُ ٱلْمَلَٰٓئِكَةُ أَلَّا تَخَافُوا۟ وَلَا تَحْزَنُوا۟ وَأَبْشِرُوا۟ بِٱلْجَنَّةِ ٱلَّتِى كُنتُمْ تُوعَدُونَ ﴿٣٠﴾ Lo! those who say: Our Lord is Allah, and afterward are upright, the angels descend upon them, saying: Fear not nor grieve, but hear good tidings of the paradise which ye are promised. (41:30) || إِنَّ ٱلَّذِينَ قَالُوا۟ رَبُّنَا ٱللَّهُ ثُمَّ ٱسْتَقَٰمُوا۟ تَتَنَزَّلُ عَلَيْهِمُ ٱلْمَلَٰٓئِكَةُ أَلَّا تَخَافُوا۟ وَلَا تَحْزَنُوا۟ وَأَبْشِرُوا۟ بِٱلْجَنَّةِ ٱلَّتِى كُنتُمْ تُوعَدُونَ ﴿٣٠﴾ بې‌شکه هغه کسان چې وې ویل: یوازې الله زموږ رب دی، بیا (پر دې) ټینګ پاتې شول، پر هغوی پرښتې نازلېږي (دا وایي) چې مه وېرېږئ او مه خپه کېږئ، او د هغه جنت په زېري خوشاله شئ چې له تاسو سره یې وعده کېده.
p نَحْنُ أَوْلِيَآؤُكُمْ فِى ٱلْحَيَوٰةِ ٱلدُّنْيَا وَفِى ٱلْءَاخِرَةِ ۖ وَلَكُمْ فِيهَا مَا تَشْتَهِىٓ أَنفُسُكُمْ وَلَكُمْ فِيهَا مَا تَدَّعُونَ ﴿٣١﴾ ہم دنیا کی زندگی میں بھی تمھارے مددگار ہیں اور آخرت میں بھی اور تمھارے لیے اس میں وہ سب کچھ ہوگا جو تمھارے جی چاہیں گے اور اس میں تمھارے لیے ہر وہ چیز ہے جو تم مانگو گے۔ || نَحْنُ أَوْلِيَآؤُكُمْ فِى ٱلْحَيَوٰةِ ٱلدُّنْيَا وَفِى ٱلْءَاخِرَةِ ۖ وَلَكُمْ فِيهَا مَا تَشْتَهِىٓ أَنفُسُكُمْ وَلَكُمْ فِيهَا مَا تَدَّعُونَ ﴿٣١﴾ We are your protecting friends in the life of the world and in the Hereafter. There ye will have (all) that your souls desire, and there ye will have (all) for which ye pray. (41:31) || نَحْنُ أَوْلِيَآؤُكُمْ فِى ٱلْحَيَوٰةِ ٱلدُّنْيَا وَفِى ٱلْءَاخِرَةِ ۖ وَلَكُمْ فِيهَا مَا تَشْتَهِىٓ أَنفُسُكُمْ وَلَكُمْ فِيهَا مَا تَدَّعُونَ ﴿٣١﴾ موږ د دنیا په ژوند کې هم ستاسو مرستندویان یو او په آخرت کې هم؛ او ستاسو لپاره به په هغه کې هر هغه څه وي چې ستاسو زړونه یې وغواړي، او په هغه کې ستاسو لپاره هر هغه شی دی چې وې غواړئ.
p نُزُلًۭا مِّنْ غَفُورٍۢ رَّحِيمٍۢ ﴿٣٢﴾ (یہ) بہت بخشنے والے نہایت رحم فرمانے والے کی طرف سے مہمانی ہے۔ || نُزُلًۭا مِّنْ غَفُورٍۢ رَّحِيمٍۢ ﴿٣٢﴾ A gift of welcome from One Forgiving, Merciful. (41:32) || نُزُلًۭا مِّنْ غَفُورٍۢ رَّحِيمٍۢ ﴿٣٢﴾ (دا) د ډېر بښونکي، بې‌حده مهربان له لوري مېلمستیا ده.
p وَمَنْ أَحْسَنُ قَوْلًۭا مِّمَّن دَعَآ إِلَى ٱللَّهِ وَعَمِلَ صَٰلِحًۭا وَقَالَ إِنَّنِى مِنَ ٱلْمُسْلِمِينَ ﴿٣٣﴾ اور اس شخص سے بہتر بات کس کی ہوگی جو اللہ کی طرف بلائے اور نیک عمل کرے اور کہے کہ بے شک میں فرماں برداروں میں سے ہوں۔ || وَمَنْ أَحْسَنُ قَوْلًۭا مِّمَّن دَعَآ إِلَى ٱللَّهِ وَعَمِلَ صَٰلِحًۭا وَقَالَ إِنَّنِى مِنَ ٱلْمُسْلِمِينَ ﴿٣٣﴾ And who is better in speech than him who prayeth unto his Lord and doeth right, and saith: Lo! I am of those who are muslims (surrender unto Him). (41:33) || وَمَنْ أَحْسَنُ قَوْلًۭا مِّمَّن دَعَآ إِلَى ٱللَّهِ وَعَمِلَ صَٰلِحًۭا وَقَالَ إِنَّنِى مِنَ ٱلْمُسْلِمِينَ ﴿٣٣﴾ او د هغه چا تر خبرې به د چا خبره غوره وي چې الله ته بلنه ورکړي، نېک عمل وکړي او ووايي: بې‌شکه زه له فرمانبردارانو یم.
p وَلَا تَسْتَوِى ٱلْحَسَنَةُ وَلَا ٱلسَّيِّئَةُ ۚ ٱدْفَعْ بِٱلَّتِى هِىَ أَحْسَنُ فَإِذَا ٱلَّذِى بَيْنَكَ وَبَيْنَهُۥ عَدَٰوَةٌۭ كَأَنَّهُۥ وَلِىٌّ حَمِيمٌۭ ﴿٣٤﴾ اور بھلائی اور برائی برابر نہیں ہوتی تم (بدی کو) بہترین نیکی سے دفع کرو پس (تم دیکھو گے) وہی شخص جس کے اور تمھارے درمیان دشمنی ہے وہ گویا گرم جوش دوست بن جائے گا۔ || وَلَا تَسْتَوِى ٱلْحَسَنَةُ وَلَا ٱلسَّيِّئَةُ ۚ ٱدْفَعْ بِٱلَّتِى هِىَ أَحْسَنُ فَإِذَا ٱلَّذِى بَيْنَكَ وَبَيْنَهُۥ عَدَٰوَةٌۭ كَأَنَّهُۥ وَلِىٌّ حَمِيمٌۭ ﴿٣٤﴾ The good deed and the evil deed are not alike. Repel the evil deed with one which is better, then lo! he, between whom and thee there was enmity (will become) as though he was a bosom friend. (41:34) || وَلَا تَسْتَوِى ٱلْحَسَنَةُ وَلَا ٱلسَّيِّئَةُ ۚ ٱدْفَعْ بِٱلَّتِى هِىَ أَحْسَنُ فَإِذَا ٱلَّذِى بَيْنَكَ وَبَيْنَهُۥ عَدَٰوَةٌۭ كَأَنَّهُۥ وَلِىٌّ حَمِيمٌۭ ﴿٣٤﴾ او ښېګڼه او بدي برابر نه دي؛ تاسو (بدي) په غوره ښېګڼه لرې کړئ، نو (وبه وینئ چې) همغه کس چې ستاسو او د هغه ترمنځ دښمني ده، ګواکې یو ګرمجوشه دوست به شي.
p وَمَا يُلَقَّىٰهَآ إِلَّا ٱلَّذِينَ صَبَرُوا۟ وَمَا يُلَقَّىٰهَآ إِلَّا ذُو حَظٍّ عَظِيمٍۢ ﴿٣٥﴾ اور یہ (توفیق) صرف انھی لوگوں کو نصیب ہوتی ہے جو صبر کرتے ہیں اور یہ (توفیق) اسی کو ملتی ہے جو بڑے نصیب والا ہوتا ہے۔ || وَمَا يُلَقَّىٰهَآ إِلَّا ٱلَّذِينَ صَبَرُوا۟ وَمَا يُلَقَّىٰهَآ إِلَّا ذُو حَظٍّ عَظِيمٍۢ ﴿٣٥﴾ But none is granted it save those who are steadfast, and none is granted it save the owner of great happiness. (41:35) || وَمَا يُلَقَّىٰهَآ إِلَّا ٱلَّذِينَ صَبَرُوا۟ وَمَا يُلَقَّىٰهَآ إِلَّا ذُو حَظٍّ عَظِيمٍۢ ﴿٣٥﴾ او دا (توفیق) یوازې هغو خلکو ته په برخه کېږي چې صبر کوي، او دا (توفیق) یوازې هغه چا ته ورکول کېږي چې د لوی نصیب خاوند وي.
p وَإِمَّا يَنزَغَنَّكَ مِنَ ٱلشَّيْطَٰنِ نَزْغٌۭ فَٱسْتَعِذْ بِٱللَّهِ ۖ إِنَّهُۥ هُوَ ٱلسَّمِيعُ ٱلْعَلِيمُ ﴿٣٦﴾ اور اگر تمھیں شیطان کی طرف سے کوئی وسوسہ آئے تو (فوراً) اللہ کی پناہ مانگ لیا کرو بے شک وہ خوب سننے والا خوب جاننے والا ہے۔ || وَإِمَّا يَنزَغَنَّكَ مِنَ ٱلشَّيْطَٰنِ نَزْغٌۭ فَٱسْتَعِذْ بِٱللَّهِ ۖ إِنَّهُۥ هُوَ ٱلسَّمِيعُ ٱلْعَلِيمُ ﴿٣٦﴾ And if a whisper from the devil reach thee (O Muhammad) then seek refuge in Allah. Lo! He is the Hearer, the Knower. (41:36) || وَإِمَّا يَنزَغَنَّكَ مِنَ ٱلشَّيْطَٰنِ نَزْغٌۭ فَٱسْتَعِذْ بِٱللَّهِ ۖ إِنَّهُۥ هُوَ ٱلسَّمِيعُ ٱلْعَلِيمُ ﴿٣٦﴾ او که تاسو ته د شیطان له لوري کومه وسوسه راشي، نو (سمدستي) د الله پناه وغواړئ؛ بې‌شکه هغه ښه اورېدونکی، ښه پوه دی.
