p ۞ فَمَا لَكُمْ فِى ٱلْمُنَٰفِقِينَ فِئَتَيْنِ وَٱللَّهُ أَرْكَسَهُم بِمَا كَسَبُوٓا۟ ۚ أَتُرِيدُونَ أَن تَهْدُوا۟ مَنْ أَضَلَّ ٱللَّهُ ۖ وَمَن يُضْلِلِ ٱللَّهُ فَلَن تَجِدَ لَهُۥ سَبِيلًۭا ﴿٨٨﴾ پھر تمھیں کیا ہو گیا ہے کہ منافقین کے بارے میں دو گروہ ہو گئے ہو حالاں کہ اللہ نے انھیں (کفر کی طرف) الٹا پھیر دیا ہے اُس وجہ سے جو کچھ انھوں نے کمایا ہے کیا تم چاہتے ہو؟ کہ اُسے ہدایت دو جسے اللہ نے گم راہ کر دیا اور جسے اللہ گم راہ کر دے پس (اے مخاطب!) تو اُس کے لیے (ہدایت کا) کوئی راستہ ہرگز نہیں پائے گا۔ || ۞ فَمَا لَكُمْ فِى ٱلْمُنَٰفِقِينَ فِئَتَيْنِ وَٱللَّهُ أَرْكَسَهُم بِمَا كَسَبُوٓا۟ ۚ أَتُرِيدُونَ أَن تَهْدُوا۟ مَنْ أَضَلَّ ٱللَّهُ ۖ وَمَن يُضْلِلِ ٱللَّهُ فَلَن تَجِدَ لَهُۥ سَبِيلًۭا ﴿٨٨﴾ What aileth you that ye are become two parties regarding the hypocrites, when Allah cast them back (to disbelief) because of what they earned? Seek ye to guide him whom Allah hath sent astray? He whom Allah sendeth astray, for him thou (O MUhammad) canst not find a road. (4:88) || ۞ فَمَا لَكُمْ فِى ٱلْمُنَٰفِقِينَ فِئَتَيْنِ وَٱللَّهُ أَرْكَسَهُم بِمَا كَسَبُوٓا۟ ۚ أَتُرِيدُونَ أَن تَهْدُوا۟ مَنْ أَضَلَّ ٱللَّهُ ۖ وَمَن يُضْلِلِ ٱللَّهُ فَلَن تَجِدَ لَهُۥ سَبِيلًۭا ﴿٨٨﴾ نو تاسو ته څه شوي چې د منافقانو په اړه دوه ډلې شوي یاست؟ حال دا چې الله هغوی د هغه څه له امله چې ګټلي یې دي، (کفر ته) سرچپه اړولي؛ ایا تاسو غواړئ هغه چا ته هدایت وکړئ چې الله ګمراه کړی؟ او څوک چې الله ګمراه کړي، نو (ای مخاطبه!) هېڅکله به یې لپاره (د هدایت) کومه لاره ونه مومې.
p وَدُّوا۟ لَوْ تَكْفُرُونَ كَمَا كَفَرُوا۟ فَتَكُونُونَ سَوَآءًۭ ۖ فَلَا تَتَّخِذُوا۟ مِنْهُمْ أَوْلِيَآءَ حَتَّىٰ يُهَاجِرُوا۟ فِى سَبِيلِ ٱللَّهِ ۚ فَإِن تَوَلَّوْا۟ فَخُذُوهُمْ وَٱقْتُلُوهُمْ حَيْثُ وَجَدتُّمُوهُمْ ۖ وَلَا تَتَّخِذُوا۟ مِنْهُمْ وَلِيًّۭا وَلَا نَصِيرًا ﴿٨٩﴾ وہ (تو) چاہتے ہیں کہ تم بھی کفر کرو جیسے انھوں نے کفر کیا پھر تم سب (کفر میں) برابر ہو جاؤ تو اُن میں سے (کسی کو) دوست مت بناؤ جب تک وہ اللہ کی راہ میں ہجرت نہ کریں پھر اگر وہ (ہجرت سے) اعراض کریں تو انھیں پکڑو اور انھیں جہاں (بھی) پاؤ قتل کر دو اور اُن میں سے نہ کسی کو اپنا دوست بناؤ اور نہ مدد گار۔ || وَدُّوا۟ لَوْ تَكْفُرُونَ كَمَا كَفَرُوا۟ فَتَكُونُونَ سَوَآءًۭ ۖ فَلَا تَتَّخِذُوا۟ مِنْهُمْ أَوْلِيَآءَ حَتَّىٰ يُهَاجِرُوا۟ فِى سَبِيلِ ٱللَّهِ ۚ فَإِن تَوَلَّوْا۟ فَخُذُوهُمْ وَٱقْتُلُوهُمْ حَيْثُ وَجَدتُّمُوهُمْ ۖ وَلَا تَتَّخِذُوا۟ مِنْهُمْ وَلِيًّۭا وَلَا نَصِيرًا ﴿٨٩﴾ They long that ye should disbelieve even as they disbelieve, that ye may be upon a level (with them). So choose not friends from them till they forsake their homes in the way of Allah; if they turn back (to enmity) then take them and kill them wherever ye find them, and choose no friend nor helper from among them, (4:89) || وَدُّوا۟ لَوْ تَكْفُرُونَ كَمَا كَفَرُوا۟ فَتَكُونُونَ سَوَآءًۭ ۖ فَلَا تَتَّخِذُوا۟ مِنْهُمْ أَوْلِيَآءَ حَتَّىٰ يُهَاجِرُوا۟ فِى سَبِيلِ ٱللَّهِ ۚ فَإِن تَوَلَّوْا۟ فَخُذُوهُمْ وَٱقْتُلُوهُمْ حَيْثُ وَجَدتُّمُوهُمْ ۖ وَلَا تَتَّخِذُوا۟ مِنْهُمْ وَلِيًّۭا وَلَا نَصِيرًا ﴿٨٩﴾ هغوی (خو) غواړي چې تاسو هم کفر وکړئ لکه څنګه چې هغوی کفر کړی، بیا ټول (په کفر کې) برابر شئ؛ نو له هغوی (هېڅوک) دوست مه نیسئ تر څو چې د الله په لار کې هجرت ونه کړي؛ بیا که (له هجرته) مخ واړوي، نو ویې نیسئ او چېرې یې چې (هم) ومومئ، ویې وژنئ؛ او له هغوی هېڅوک خپل دوست او مرستندوی مه نیسئ.
p إِلَّا ٱلَّذِينَ يَصِلُونَ إِلَىٰ قَوْمٍۭ بَيْنَكُمْ وَبَيْنَهُم مِّيثَٰقٌ أَوْ جَآءُوكُمْ حَصِرَتْ صُدُورُهُمْ أَن يُقَٰتِلُوكُمْ أَوْ يُقَٰتِلُوا۟ قَوْمَهُمْ ۚ وَلَوْ شَآءَ ٱللَّهُ لَسَلَّطَهُمْ عَلَيْكُمْ فَلَقَٰتَلُوكُمْ ۚ فَإِنِ ٱعْتَزَلُوكُمْ فَلَمْ يُقَٰتِلُوكُمْ وَأَلْقَوْا۟ إِلَيْكُمُ ٱلسَّلَمَ فَمَا جَعَلَ ٱللَّهُ لَكُمْ عَلَيْهِمْ سَبِيلًۭا ﴿٩٠﴾ مگر اُن لوگوں کو (قتل نہ کرنا) جو ایسی قوم سے جا ملیں کہ تمھارے اور اُن کے درمیان معاہدہ ہے یا وہ تمھارے پاس آئیں اس حال میں کہ اُن کے سینے تنگ ہوں کہ وہ تم سے لڑیں یا اپنی قوم سے لڑیں اور اگر اللہ چاہتا تو یقیناً انھیں تم پر مسلط کر دیتا تو وہ ضرور تم سے لڑتے پس اگر وہ تم سے الگ رہیں اور تم سے نہ لڑیں اور تمھیں صلح کا پیغام دیں تو (پھر) اللہ نے تمھارے لیے اُن کے خلاف (اقدام کی) کوئی راہ نہیں رکھی۔ || إِلَّا ٱلَّذِينَ يَصِلُونَ إِلَىٰ قَوْمٍۭ بَيْنَكُمْ وَبَيْنَهُم مِّيثَٰقٌ أَوْ جَآءُوكُمْ حَصِرَتْ صُدُورُهُمْ أَن يُقَٰتِلُوكُمْ أَوْ يُقَٰتِلُوا۟ قَوْمَهُمْ ۚ وَلَوْ شَآءَ ٱللَّهُ لَسَلَّطَهُمْ عَلَيْكُمْ فَلَقَٰتَلُوكُمْ ۚ فَإِنِ ٱعْتَزَلُوكُمْ فَلَمْ يُقَٰتِلُوكُمْ وَأَلْقَوْا۟ إِلَيْكُمُ ٱلسَّلَمَ فَمَا جَعَلَ ٱللَّهُ لَكُمْ عَلَيْهِمْ سَبِيلًۭا ﴿٩٠﴾ Except those who seek refuge with a people between whom and you there is a covenant, or (those who) come unto you because their hearts forbid them to make war on you or make war on their own folk. Had Allah willed He could have given them power over you so that assuredly they would have fought you. So, if they hold aloof from you and wage not war against you and offer you peace, Allah alloweth you no way against them. (4:90) || إِلَّا ٱلَّذِينَ يَصِلُونَ إِلَىٰ قَوْمٍۭ بَيْنَكُمْ وَبَيْنَهُم مِّيثَٰقٌ أَوْ جَآءُوكُمْ حَصِرَتْ صُدُورُهُمْ أَن يُقَٰتِلُوكُمْ أَوْ يُقَٰتِلُوا۟ قَوْمَهُمْ ۚ وَلَوْ شَآءَ ٱللَّهُ لَسَلَّطَهُمْ عَلَيْكُمْ فَلَقَٰتَلُوكُمْ ۚ فَإِنِ ٱعْتَزَلُوكُمْ فَلَمْ يُقَٰتِلُوكُمْ وَأَلْقَوْا۟ إِلَيْكُمُ ٱلسَّلَمَ فَمَا جَعَلَ ٱللَّهُ لَكُمْ عَلَيْهِمْ سَبِيلًۭا ﴿٩٠﴾ مګر هغه کسان (مه وژنئ) چې له داسې قوم سره یوځای شي چې ستاسو او د هغوی ترمنځ تړون وي، یا تاسو ته په داسې حال کې راشي چې سینې یې له دې تنګې وي چې له تاسو سره وجنګېږي یا له خپل قوم سره وجنګېږي؛ او که الله غوښتي وای، یقیناً به یې پر تاسو مسلط کړي وای، نو خامخا به یې له تاسو سره جګړه کړې وای؛ نو که له تاسو جلا پاتې شي، له تاسو سره ونه جنګېږي او تاسو ته د سولې پیغام درکړي، (نو بیا) الله تاسو ته د هغوی پر خلاف (د اقدام) هېڅ لاره نه ده پرېښې.
p سَتَجِدُونَ ءَاخَرِينَ يُرِيدُونَ أَن يَأْمَنُوكُمْ وَيَأْمَنُوا۟ قَوْمَهُمْ كُلَّ مَا رُدُّوٓا۟ إِلَى ٱلْفِتْنَةِ أُرْكِسُوا۟ فِيهَا ۚ فَإِن لَّمْ يَعْتَزِلُوكُمْ وَيُلْقُوٓا۟ إِلَيْكُمُ ٱلسَّلَمَ وَيَكُفُّوٓا۟ أَيْدِيَهُمْ فَخُذُوهُمْ وَٱقْتُلُوهُمْ حَيْثُ ثَقِفْتُمُوهُمْ ۚ وَأُو۟لَٰٓئِكُمْ جَعَلْنَا لَكُمْ عَلَيْهِمْ سُلْطَٰنًۭا مُّبِينًۭا ﴿٩١﴾ عنقریب تم کچھ دوسرے لوگوں (منافقوں) کو پاؤ گے وہ چاہتے ہیں کہ تم سے (بھی) امن میں رہیں اور اپنی قوم سے بھی امن میں رہیں لیکن جب بھی کسی فتنہ انگیزی کی طرف پھیرے جاتے ہیں تو وہ اُس فتنے میں (اوندھے منہ) کود پڑتے ہیں تو اگر وہ تم سے الگ نہ رہیں اور تمھیں صلح کا پیغام نہ دیں اور اپنے ہاتھ نہ روکیں تو انھیں پکڑو اور انھیں جہاں (بھی) پاؤ قتل کر دو اور یہ وہ لوگ ہیں جن کے خلاف ہم نے تمھیں کھلا اختیار دیا ہے۔ || سَتَجِدُونَ ءَاخَرِينَ يُرِيدُونَ أَن يَأْمَنُوكُمْ وَيَأْمَنُوا۟ قَوْمَهُمْ كُلَّ مَا رُدُّوٓا۟ إِلَى ٱلْفِتْنَةِ أُرْكِسُوا۟ فِيهَا ۚ فَإِن لَّمْ يَعْتَزِلُوكُمْ وَيُلْقُوٓا۟ إِلَيْكُمُ ٱلسَّلَمَ وَيَكُفُّوٓا۟ أَيْدِيَهُمْ فَخُذُوهُمْ وَٱقْتُلُوهُمْ حَيْثُ ثَقِفْتُمُوهُمْ ۚ وَأُو۟لَٰٓئِكُمْ جَعَلْنَا لَكُمْ عَلَيْهِمْ سُلْطَٰنًۭا مُّبِينًۭا ﴿٩١﴾ Ye will find others who desire that they should have security from you, and security from their own folk. So often as they are returned to hostility they are plunged therein. If they keep not aloof from you nor offer you peace nor hold their hands, then take them and kill them wherever ye find them. Against such We have given you clear warrant. (4:91) || سَتَجِدُونَ ءَاخَرِينَ يُرِيدُونَ أَن يَأْمَنُوكُمْ وَيَأْمَنُوا۟ قَوْمَهُمْ كُلَّ مَا رُدُّوٓا۟ إِلَى ٱلْفِتْنَةِ أُرْكِسُوا۟ فِيهَا ۚ فَإِن لَّمْ يَعْتَزِلُوكُمْ وَيُلْقُوٓا۟ إِلَيْكُمُ ٱلسَّلَمَ وَيَكُفُّوٓا۟ أَيْدِيَهُمْ فَخُذُوهُمْ وَٱقْتُلُوهُمْ حَيْثُ ثَقِفْتُمُوهُمْ ۚ وَأُو۟لَٰٓئِكُمْ جَعَلْنَا لَكُمْ عَلَيْهِمْ سُلْطَٰنًۭا مُّبِينًۭا ﴿٩١﴾ ژر به تاسو ځینې نور خلک (منافقان) ومومئ چې غواړي له تاسو (هم) په امن کې وي او له خپل قوم سره هم په امن کې وي؛ خو هرکله چې کوم فتنې ته واړول شي، نو په هغې فتنې کې (پړمخې) ورټوپ کړي؛ نو که له تاسو جلا پاتې نه شي، تاسو ته د سولې پیغام در نه کړي او خپل لاسونه ونه ساتي، نو ویې نیسئ او چېرې یې چې (هم) ومومئ، ویې وژنئ؛ او دا هغه خلک دي چې د هغوی پر خلاف موږ تاسو ته ښکاره واک درکړی.
