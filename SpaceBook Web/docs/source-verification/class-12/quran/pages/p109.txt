p ٱلنَّبِىُّ أَوْلَىٰ بِٱلْمُؤْمِنِينَ مِنْ أَنفُسِهِمْ ۖ وَأَزْوَٰجُهُۥٓ أُمَّهَٰتُهُمْ ۗ وَأُو۟لُوا۟ ٱلْأَرْحَامِ بَعْضُهُمْ أَوْلَىٰ بِبَعْضٍۢ فِى كِتَٰبِ ٱللَّهِ مِنَ ٱلْمُؤْمِنِينَ وَٱلْمُهَٰجِرِينَ إِلَّآ أَن تَفْعَلُوٓا۟ إِلَىٰٓ أَوْلِيَآئِكُم مَّعْرُوفًۭا ۚ كَانَ ذَٰلِكَ فِى ٱلْكِتَٰبِ مَسْطُورًۭا ﴿٦﴾ نبی (کریم صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) اہلِ ایمان کی جانوں پر خود ان سے زیادہ حق رکھتے ہیں اور آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کی ازواج (مطہرات رضی اللہ عنہن) اُن کی مائیں ہیں اور (قریبی) رشتہ دار اللہ کی کتاب میں (میراث کے لحاظ سے) ایک دوسرے کے زیادہ حق دار ہیں دوسرے مومنین اور مہاجرین کی نسبت مگر یہ کہ تم اپنے دوستوں پر (وصیت کر کے) احسان کرنا چاہو یہ حکم کتاب (الٰہی) میں لکھا ہوا ہے۔ || ٱلنَّبِىُّ أَوْلَىٰ بِٱلْمُؤْمِنِينَ مِنْ أَنفُسِهِمْ ۖ وَأَزْوَٰجُهُۥٓ أُمَّهَٰتُهُمْ ۗ وَأُو۟لُوا۟ ٱلْأَرْحَامِ بَعْضُهُمْ أَوْلَىٰ بِبَعْضٍۢ فِى كِتَٰبِ ٱللَّهِ مِنَ ٱلْمُؤْمِنِينَ وَٱلْمُهَٰجِرِينَ إِلَّآ أَن تَفْعَلُوٓا۟ إِلَىٰٓ أَوْلِيَآئِكُم مَّعْرُوفًۭا ۚ كَانَ ذَٰلِكَ فِى ٱلْكِتَٰبِ مَسْطُورًۭا ﴿٦﴾ The Prophet is closer to the believers than their selves, and his wives are (as) their mothers. And the owners of kinship are closer one to another in the ordinance of Allah than (other) believers and the fugitives (who fled from Mecca), except that ye should do kindness to your friends. This is written in the Book (of nature). (33:6) || ٱلنَّبِىُّ أَوْلَىٰ بِٱلْمُؤْمِنِينَ مِنْ أَنفُسِهِمْ ۖ وَأَزْوَٰجُهُۥٓ أُمَّهَٰتُهُمْ ۗ وَأُو۟لُوا۟ ٱلْأَرْحَامِ بَعْضُهُمْ أَوْلَىٰ بِبَعْضٍۢ فِى كِتَٰبِ ٱللَّهِ مِنَ ٱلْمُؤْمِنِينَ وَٱلْمُهَٰجِرِينَ إِلَّآ أَن تَفْعَلُوٓا۟ إِلَىٰٓ أَوْلِيَآئِكُم مَّعْرُوفًۭا ۚ كَانَ ذَٰلِكَ فِى ٱلْكِتَٰبِ مَسْطُورًۭا ﴿٦﴾ نبي (کریم صلی الله علیه وعلی آله واصحابه وسلم) د مؤمنانو پر ځانونو له هغوی پخپله ډېر حق لري، او د هغه (صلی الله علیه وعلی آله واصحابه وسلم) مېرمنې (پاکې مېرمنې رضي الله عنهن) د هغوی میندې دي؛ او (نږدې) خپلوان د الله په کتاب کې (د میراث له مخې) د نورو مؤمنانو او مهاجرینو په پرتله یو د بل ډېر حقدار دي، مګر دا چې تاسو پر خپلو دوستانو (په وصیت سره) احسان وکړئ؛ دا حکم په (الهي) کتاب کې لیکل شوی دی.
p وَإِذْ أَخَذْنَا مِنَ ٱلنَّبِيِّۦنَ مِيثَٰقَهُمْ وَمِنكَ وَمِن نُّوحٍۢ وَإِبْرَٰهِيمَ وَمُوسَىٰ وَعِيسَى ٱبْنِ مَرْيَمَ ۖ وَأَخَذْنَا مِنْهُم مِّيثَٰقًا غَلِيظًۭا ﴿٧﴾ اور جب ہم نے نبیوں سے پختہ عہد لیا اور (خصوصاً) آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) سے اور نوح اور ابراہیم اور موسیٰ اور عیسیٰ ابن مریم (علیہم السلام) سے اور ہم نے اُن سے پختہ عہد لیا۔ || وَإِذْ أَخَذْنَا مِنَ ٱلنَّبِيِّۦنَ مِيثَٰقَهُمْ وَمِنكَ وَمِن نُّوحٍۢ وَإِبْرَٰهِيمَ وَمُوسَىٰ وَعِيسَى ٱبْنِ مَرْيَمَ ۖ وَأَخَذْنَا مِنْهُم مِّيثَٰقًا غَلِيظًۭا ﴿٧﴾ And when We exacted a covenant from the prophets, and from thee (O Muhammad) and from Noah and Abraham and Moses and Jesus son of Mary. We took from them a solemn covenant; (33:7) || وَإِذْ أَخَذْنَا مِنَ ٱلنَّبِيِّۦنَ مِيثَٰقَهُمْ وَمِنكَ وَمِن نُّوحٍۢ وَإِبْرَٰهِيمَ وَمُوسَىٰ وَعِيسَى ٱبْنِ مَرْيَمَ ۖ وَأَخَذْنَا مِنْهُم مِّيثَٰقًا غَلِيظًۭا ﴿٧﴾ او کله چې موږ له نبیانو ټینګه ژمنه واخیسته، او (په ځانګړي ډول) له تاسو (صلی الله علیه وعلی آله واصحابه وسلم) او له نوح، ابراهیم، موسی او عیسی ابن مریم (علیهم السلام) څخه، او له هغوی مو ټینګه ژمنه واخیسته.
p لِّيَسْـَٔلَ ٱلصَّٰدِقِينَ عَن صِدْقِهِمْ ۚ وَأَعَدَّ لِلْكَٰفِرِينَ عَذَابًا أَلِيمًۭا ﴿٨﴾ تاکہ اللہ سچوں سے اُن کے سچ کے بارے میں پوچھے اور اُس نے کافروں کے لیے دردناک عذاب تیار کر رکھا ہے۔ || لِّيَسْـَٔلَ ٱلصَّٰدِقِينَ عَن صِدْقِهِمْ ۚ وَأَعَدَّ لِلْكَٰفِرِينَ عَذَابًا أَلِيمًۭا ﴿٨﴾ That He may ask the loyal of their loyalty. And He hath prepared a painful doom for the unfaithful. (33:8) || لِّيَسْـَٔلَ ٱلصَّٰدِقِينَ عَن صِدْقِهِمْ ۚ وَأَعَدَّ لِلْكَٰفِرِينَ عَذَابًا أَلِيمًۭا ﴿٨﴾ تر څو الله له رښتینو د هغوی د رښتیا په اړه وپوښتي؛ او هغه د کافرانو لپاره دردناک عذاب تیار کړی دی.
p يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ ٱذْكُرُوا۟ نِعْمَةَ ٱللَّهِ عَلَيْكُمْ إِذْ جَآءَتْكُمْ جُنُودٌۭ فَأَرْسَلْنَا عَلَيْهِمْ رِيحًۭا وَجُنُودًۭا لَّمْ تَرَوْهَا ۚ وَكَانَ ٱللَّهُ بِمَا تَعْمَلُونَ بَصِيرًا ﴿٩﴾ اے ایمان والو! اپنے اوپر اللہ کا احسان یاد کرو جب تم پر (کفار کے) لشکر چڑھ آئے پھر ہم نے (تمھاری مدد کے لیے) ان پر (تیز) آندھی بھیجی اور (فرشتوں کے) لشکر (بھیجے) جنھیں تم نے نہیں دیکھا اور جو تم کر رہے تھے اللہ اُسے خوب دیکھنے والا تھا۔ || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ ٱذْكُرُوا۟ نِعْمَةَ ٱللَّهِ عَلَيْكُمْ إِذْ جَآءَتْكُمْ جُنُودٌۭ فَأَرْسَلْنَا عَلَيْهِمْ رِيحًۭا وَجُنُودًۭا لَّمْ تَرَوْهَا ۚ وَكَانَ ٱللَّهُ بِمَا تَعْمَلُونَ بَصِيرًا ﴿٩﴾ O ye who believe! Remember Allah's favour unto you when there came against you hosts, and We sent against them a great wind and hosts ye could not see. And Allah is ever Seer of what ye do. (33:9) || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ ٱذْكُرُوا۟ نِعْمَةَ ٱللَّهِ عَلَيْكُمْ إِذْ جَآءَتْكُمْ جُنُودٌۭ فَأَرْسَلْنَا عَلَيْهِمْ رِيحًۭا وَجُنُودًۭا لَّمْ تَرَوْهَا ۚ وَكَانَ ٱللَّهُ بِمَا تَعْمَلُونَ بَصِيرًا ﴿٩﴾ ای مؤمنانو! پر خپل ځان د الله احسان یاد کړئ کله چې پر تاسو د (کافرانو) لښکرې راغلې، بیا موږ (ستاسو د مرستې لپاره) پر هغوی (تېز) توپان ولېږه او د (پرښتو) لښکرې (مو ولېږلې) چې تاسو نه لیدلې؛ او څه چې تاسو کول، الله یې ښه لیدونکی و.
p إِذْ جَآءُوكُم مِّن فَوْقِكُمْ وَمِنْ أَسْفَلَ مِنكُمْ وَإِذْ زَاغَتِ ٱلْأَبْصَٰرُ وَبَلَغَتِ ٱلْقُلُوبُ ٱلْحَنَاجِرَ وَتَظُنُّونَ بِٱللَّهِ ٱلظُّنُونَا۠ ﴿١٠﴾ جب وہ (کافر) تمھارے اوپر اور تمھارے نیچے سے تم پر چڑھ آئے اور جب (دہشت سے) آنکھیں پتھرا گئیں اور دل (کلیجے) منہ کو آ گئے اور تم اللہ کے متعلق (خوف اور امید میں) طرح طرح کے گمان کرنے لگے۔ || إِذْ جَآءُوكُم مِّن فَوْقِكُمْ وَمِنْ أَسْفَلَ مِنكُمْ وَإِذْ زَاغَتِ ٱلْأَبْصَٰرُ وَبَلَغَتِ ٱلْقُلُوبُ ٱلْحَنَاجِرَ وَتَظُنُّونَ بِٱللَّهِ ٱلظُّنُونَا۠ ﴿١٠﴾ When they came upon you from above you and from below you, and when eyes grew wild and hearts reached to the throats, and ye were imagining vain thoughts concerning Allah. (33:10) || إِذْ جَآءُوكُم مِّن فَوْقِكُمْ وَمِنْ أَسْفَلَ مِنكُمْ وَإِذْ زَاغَتِ ٱلْأَبْصَٰرُ وَبَلَغَتِ ٱلْقُلُوبُ ٱلْحَنَاجِرَ وَتَظُنُّونَ بِٱللَّهِ ٱلظُّنُونَا۠ ﴿١٠﴾ کله چې هغوی (کافران) له پاسه او له لاندې پر تاسو راغلل، او کله چې (له وېرې) سترګې ټکانې شوې او زړونه (ځیګرونه) ستوني ته راورسېدل، او تاسو د الله په اړه (د وېرې او هیلې په منځ کې) ډول ډول ګومانونه کول.
p هُنَالِكَ ٱبْتُلِىَ ٱلْمُؤْمِنُونَ وَزُلْزِلُوا۟ زِلْزَالًۭا شَدِيدًۭا ﴿١١﴾ اس موقع پر مومن خوب آزمائے گئے اور وہ پوری شدت سے ہلا دیے گئے۔ || هُنَالِكَ ٱبْتُلِىَ ٱلْمُؤْمِنُونَ وَزُلْزِلُوا۟ زِلْزَالًۭا شَدِيدًۭا ﴿١١﴾ There were the believers sorely tried, and shaken with a mighty shock. (33:11) || هُنَالِكَ ٱبْتُلِىَ ٱلْمُؤْمِنُونَ وَزُلْزِلُوا۟ زِلْزَالًۭا شَدِيدًۭا ﴿١١﴾ په همدې ځای کې مؤمنان ښه وازمویل شول او په پوره شدت وښورول شول.
p وَإِذْ يَقُولُ ٱلْمُنَٰفِقُونَ وَٱلَّذِينَ فِى قُلُوبِهِم مَّرَضٌۭ مَّا وَعَدَنَا ٱللَّهُ وَرَسُولُهُۥٓ إِلَّا غُرُورًۭا ﴿١٢﴾ اور جب منافقین نے کہا اور اُن لوگوں نے جن کے دلوں میں (شک کی) بیماری تھی کہ اللہ اور اُس کے رسول نے ہم سے صرف دھوکے اور فریب کے لیے (فتح کا) وعدہ کیا تھا۔ || وَإِذْ يَقُولُ ٱلْمُنَٰفِقُونَ وَٱلَّذِينَ فِى قُلُوبِهِم مَّرَضٌۭ مَّا وَعَدَنَا ٱللَّهُ وَرَسُولُهُۥٓ إِلَّا غُرُورًۭا ﴿١٢﴾ And when the hypocrites, and those in whose hearts is a disease, were saying: Allah and His messenger promised us naught but delusion. (33:12) || وَإِذْ يَقُولُ ٱلْمُنَٰفِقُونَ وَٱلَّذِينَ فِى قُلُوبِهِم مَّرَضٌۭ مَّا وَعَدَنَا ٱللَّهُ وَرَسُولُهُۥٓ إِلَّا غُرُورًۭا ﴿١٢﴾ او کله چې منافقانو او هغو کسانو چې په زړونو کې یې د (شک) ناروغي وه وویل: الله او د هغه رسول له موږ سره د (بریا) وعده یوازې د دوکې او فرېب لپاره کړې وه.
p وَإِذْ قَالَت طَّآئِفَةٌۭ مِّنْهُمْ يَٰٓأَهْلَ يَثْرِبَ لَا مُقَامَ لَكُمْ فَٱرْجِعُوا۟ ۚ وَيَسْتَـْٔذِنُ فَرِيقٌۭ مِّنْهُمُ ٱلنَّبِىَّ يَقُولُونَ إِنَّ بُيُوتَنَا عَوْرَةٌۭ وَمَا هِىَ بِعَوْرَةٍ ۖ إِن يُرِيدُونَ إِلَّا فِرَارًۭا ﴿١٣﴾ اور جب ان ہی میں سے ایک گروہ نے کہا اے یثرب والو! تمھارے لیے یہاں ٹھہرنے کا کوئی موقع نہیں (جان عزیز ہے) تو واپس (اپنے گھروں کو) چلے جاؤ اور ان میں سے ایک گروہ نبی (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) سے (واپس جانے کی) اجازت مانگ رہا تھا یہ کہتے ہوئے کہ بے شک ہمارے گھر غیر محفوظ ہیں حالانکہ وہ غیر محفوظ نہیں تھے وہ تو صرف بھاگ جانا چاہتے تھے۔ || وَإِذْ قَالَت طَّآئِفَةٌۭ مِّنْهُمْ يَٰٓأَهْلَ يَثْرِبَ لَا مُقَامَ لَكُمْ فَٱرْجِعُوا۟ ۚ وَيَسْتَـْٔذِنُ فَرِيقٌۭ مِّنْهُمُ ٱلنَّبِىَّ يَقُولُونَ إِنَّ بُيُوتَنَا عَوْرَةٌۭ وَمَا هِىَ بِعَوْرَةٍ ۖ إِن يُرِيدُونَ إِلَّا فِرَارًۭا ﴿١٣﴾ And when a party of them said: O folk of Yathrib! There is no stand (possible) for you, therefor turn back. And certain of them (even) sought permission of the Prophet, saying: Our homes lie open (to the enemy). And they lay not open. (33:13) || وَإِذْ قَالَت طَّآئِفَةٌۭ مِّنْهُمْ يَٰٓأَهْلَ يَثْرِبَ لَا مُقَامَ لَكُمْ فَٱرْجِعُوا۟ ۚ وَيَسْتَـْٔذِنُ فَرِيقٌۭ مِّنْهُمُ ٱلنَّبِىَّ يَقُولُونَ إِنَّ بُيُوتَنَا عَوْرَةٌۭ وَمَا هِىَ بِعَوْرَةٍ ۖ إِن يُرِيدُونَ إِلَّا فِرَارًۭا ﴿١٣﴾ او کله چې له همدوی یوې ډلې وویل: ای د یثرب خلکو! دلته ستاسو د پاتې کېدو هېڅ ځای نشته (ساه ګرانه ده)، نو بېرته (خپلو کورونو ته) لاړ شئ؛ او له هغوی یوې ډلې له نبي (صلی الله علیه وعلی آله واصحابه وسلم) څخه د (بېرته تګ) اجازه غوښته او ویل یې: بې‌شکه زموږ کورونه خوندي نه دي، حال دا چې هغه ناخوندي نه وو؛ هغوی یوازې تېښته غوښته.
