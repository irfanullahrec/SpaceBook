p وَإِذَا سَمِعُوا۟ مَآ أُنزِلَ إِلَى ٱلرَّسُولِ تَرَىٰٓ أَعْيُنَهُمْ تَفِيضُ مِنَ ٱلدَّمْعِ مِمَّا عَرَفُوا۟ مِنَ ٱلْحَقِّ ۖ يَقُولُونَ رَبَّنَآ ءَامَنَّا فَٱكْتُبْنَا مَعَ ٱلشَّٰهِدِينَ ﴿٨٣﴾ اور جب وہ اُس (قرآن) کو سنتے ہیں جو رسول کی طرف نازل کیا گیا تو آپ دیکھتے ہیں اُن کی آنکھیں آنسوؤں سے بہہ رہی ہوتی ہیں اس لیے کہ وہ حق کو پہچان چکے ہیں وہ عرض کرتے ہیں اے ہمارے رب! ہم ایمان لے آئے پس ہمیں (حق کے) گواہوں کے ساتھ لکھ لے۔ || وَإِذَا سَمِعُوا۟ مَآ أُنزِلَ إِلَى ٱلرَّسُولِ تَرَىٰٓ أَعْيُنَهُمْ تَفِيضُ مِنَ ٱلدَّمْعِ مِمَّا عَرَفُوا۟ مِنَ ٱلْحَقِّ ۖ يَقُولُونَ رَبَّنَآ ءَامَنَّا فَٱكْتُبْنَا مَعَ ٱلشَّٰهِدِينَ ﴿٨٣﴾ When they listen to that which hath been revealed unto the messengers, thou seest their eyes overflow with tears because of their recognition of the Truth. They say: Our Lord, we believe. Inscribe us as among the witnesses. (5:83) || وَإِذَا سَمِعُوا۟ مَآ أُنزِلَ إِلَى ٱلرَّسُولِ تَرَىٰٓ أَعْيُنَهُمْ تَفِيضُ مِنَ ٱلدَّمْعِ مِمَّا عَرَفُوا۟ مِنَ ٱلْحَقِّ ۖ يَقُولُونَ رَبَّنَآ ءَامَنَّا فَٱكْتُبْنَا مَعَ ٱلشَّٰهِدِينَ ﴿٨٣﴾ او کله چې هغه (قرآن) واوري چې رسول ته نازل شوی، نو تاسو وینئ چې سترګې یې له اوښکو بهېږي، ځکه چې حق یې پېژندلی؛ عرض کوي: «ای زموږ ربه! موږ ایمان راوړ، نو موږ د (حق) له شاهدانو سره ولیکه.»
p وَمَا لَنَا لَا نُؤْمِنُ بِٱللَّهِ وَمَا جَآءَنَا مِنَ ٱلْحَقِّ وَنَطْمَعُ أَن يُدْخِلَنَا رَبُّنَا مَعَ ٱلْقَوْمِ ٱلصَّٰلِحِينَ ﴿٨٤﴾ اور ہمارے لیے کیا (وجہ) ہے کہ ہم اللہ پر ایمان نہ لائیں اور اس حق پر جو ہمارے پاس آ چکا ہے اور ہم اُمید رکھتے ہیں کہ ہمارا رب ہمیں نیک لوگوں کے ساتھ داخل فرما دے۔ || وَمَا لَنَا لَا نُؤْمِنُ بِٱللَّهِ وَمَا جَآءَنَا مِنَ ٱلْحَقِّ وَنَطْمَعُ أَن يُدْخِلَنَا رَبُّنَا مَعَ ٱلْقَوْمِ ٱلصَّٰلِحِينَ ﴿٨٤﴾ How should we not believe in Allah and that which hath come unto us of the Truth. And (how should we not) hope that our Lord will bring us in along with righteous folk? (5:84) || وَمَا لَنَا لَا نُؤْمِنُ بِٱللَّهِ وَمَا جَآءَنَا مِنَ ٱلْحَقِّ وَنَطْمَعُ أَن يُدْخِلَنَا رَبُّنَا مَعَ ٱلْقَوْمِ ٱلصَّٰلِحِينَ ﴿٨٤﴾ «او موږ ته څه (لامل) دی چې پر الله او پر هغه حق ایمان رانه وړو چې موږ ته راغلی، او هیله لرو چې زموږ رب موږ له نېکو خلکو سره ننباسي؟»
p فَأَثَٰبَهُمُ ٱللَّهُ بِمَا قَالُوا۟ جَنَّٰتٍۢ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَٰرُ خَٰلِدِينَ فِيهَا ۚ وَذَٰلِكَ جَزَآءُ ٱلْمُحْسِنِينَ ﴿٨٥﴾ تو اللہ نے انھیں اس قول کے بدلے میں عطا فرمائے ایسے باغات جن کے نیچے نہریں بہتی ہیں وہ ان میں ہمیشہ رہنے والے ہیں اور یہی نیکی کرنے والوں کا بدلہ ہے۔ || فَأَثَٰبَهُمُ ٱللَّهُ بِمَا قَالُوا۟ جَنَّٰتٍۢ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَٰرُ خَٰلِدِينَ فِيهَا ۚ وَذَٰلِكَ جَزَآءُ ٱلْمُحْسِنِينَ ﴿٨٥﴾ Allah hath rewarded them for that their saying - Gardens underneath which rivers flow, wherein they will abide for ever. That is the reward of the good. (5:85) || فَأَثَٰبَهُمُ ٱللَّهُ بِمَا قَالُوا۟ جَنَّٰتٍۢ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَٰرُ خَٰلِدِينَ فِيهَا ۚ وَذَٰلِكَ جَزَآءُ ٱلْمُحْسِنِينَ ﴿٨٥﴾ نو الله هغوی ته د دې خبرې په بدل کې داسې باغونه ورکړل چې لاندې یې ویالې بهېږي، په هغو کې به تل وي؛ او همدا د نېکوکارانو بدله ده.
p وَٱلَّذِينَ كَفَرُوا۟ وَكَذَّبُوا۟ بِـَٔايَٰتِنَآ أُو۟لَٰٓئِكَ أَصْحَٰبُ ٱلْجَحِيمِ ﴿٨٦﴾ اور جنھوں نے کفر کیا اور ہماری آیتوں کو جھٹلایا (تو) یہی لوگ جہنم والے ہیں۔ || وَٱلَّذِينَ كَفَرُوا۟ وَكَذَّبُوا۟ بِـَٔايَٰتِنَآ أُو۟لَٰٓئِكَ أَصْحَٰبُ ٱلْجَحِيمِ ﴿٨٦﴾ But those who disbelieve and deny Our revelations, they are owners of hell-fire. (5:86) || وَٱلَّذِينَ كَفَرُوا۟ وَكَذَّبُوا۟ بِـَٔايَٰتِنَآ أُو۟لَٰٓئِكَ أَصْحَٰبُ ٱلْجَحِيمِ ﴿٨٦﴾ او هغه کسان چې کفر یې وکړ او زموږ آیتونه یې درواغ وګڼل، (نو) همدا خلک دوزخیان دي.
p يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ لَا تُحَرِّمُوا۟ طَيِّبَٰتِ مَآ أَحَلَّ ٱللَّهُ لَكُمْ وَلَا تَعْتَدُوٓا۟ ۚ إِنَّ ٱللَّهَ لَا يُحِبُّ ٱلْمُعْتَدِينَ ﴿٨٧﴾ اے ایمان والو! پاکیزہ چیزیں حرام نہ ٹھہراؤ جنھیں اللہ نے تمھارے لیے حلال فرمایا ہے اور حد سے نہ بڑھو بے شک اللہ حد سے بڑھنے والوں کو پسند نہیں کرتا۔ || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ لَا تُحَرِّمُوا۟ طَيِّبَٰتِ مَآ أَحَلَّ ٱللَّهُ لَكُمْ وَلَا تَعْتَدُوٓا۟ ۚ إِنَّ ٱللَّهَ لَا يُحِبُّ ٱلْمُعْتَدِينَ ﴿٨٧﴾ O ye who believe! Forbid not the good things which Allah hath made lawful for you, and transgress not, Lo! Allah loveth not transgressors. (5:87) || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ لَا تُحَرِّمُوا۟ طَيِّبَٰتِ مَآ أَحَلَّ ٱللَّهُ لَكُمْ وَلَا تَعْتَدُوٓا۟ ۚ إِنَّ ٱللَّهَ لَا يُحِبُّ ٱلْمُعْتَدِينَ ﴿٨٧﴾ ای مؤمنانو! هغه پاک شیان مه حراموئ چې الله تاسو ته حلال کړي، او له حده مه اوړئ؛ بې‌شکه الله له حده اوښتونکي نه خوښوي.
p وَكُلُوا۟ مِمَّا رَزَقَكُمُ ٱللَّهُ حَلَٰلًۭا طَيِّبًۭا ۚ وَٱتَّقُوا۟ ٱللَّهَ ٱلَّذِىٓ أَنتُم بِهِۦ مُؤْمِنُونَ ﴿٨٨﴾ اور اُس میں سے کھاؤ جو رزق اللہ نے تمھیں دیا ہے حلال پاکیزہ اور اللہ (کی نافرمانی) سے ڈرو جس پر تم ایمان رکھنے والے ہو۔ || وَكُلُوا۟ مِمَّا رَزَقَكُمُ ٱللَّهُ حَلَٰلًۭا طَيِّبًۭا ۚ وَٱتَّقُوا۟ ٱللَّهَ ٱلَّذِىٓ أَنتُم بِهِۦ مُؤْمِنُونَ ﴿٨٨﴾ Eat of that which Allah hath bestowed on you as food lawful and good, and keep your duty to Allah in Whom ye are believers. (5:88) || وَكُلُوا۟ مِمَّا رَزَقَكُمُ ٱللَّهُ حَلَٰلًۭا طَيِّبًۭا ۚ وَٱتَّقُوا۟ ٱللَّهَ ٱلَّذِىٓ أَنتُم بِهِۦ مُؤْمِنُونَ ﴿٨٨﴾ او له هغه رزقه وخورئ چې الله درکړی — حلال او پاک — او له هغه الله (د نافرمانۍ) ووېرېږئ چې پرې ایمان لرئ.
p لَا يُؤَاخِذُكُمُ ٱللَّهُ بِٱللَّغْوِ فِىٓ أَيْمَٰنِكُمْ وَلَٰكِن يُؤَاخِذُكُم بِمَا عَقَّدتُّمُ ٱلْأَيْمَٰنَ ۖ فَكَفَّٰرَتُهُۥٓ إِطْعَامُ عَشَرَةِ مَسَٰكِينَ مِنْ أَوْسَطِ مَا تُطْعِمُونَ أَهْلِيكُمْ أَوْ كِسْوَتُهُمْ أَوْ تَحْرِيرُ رَقَبَةٍۢ ۖ فَمَن لَّمْ يَجِدْ فَصِيَامُ ثَلَٰثَةِ أَيَّامٍۢ ۚ ذَٰلِكَ كَفَّٰرَةُ أَيْمَٰنِكُمْ إِذَا حَلَفْتُمْ ۚ وَٱحْفَظُوٓا۟ أَيْمَٰنَكُمْ ۚ كَذَٰلِكَ يُبَيِّنُ ٱللَّهُ لَكُمْ ءَايَٰتِهِۦ لَعَلَّكُمْ تَشْكُرُونَ ﴿٨٩﴾ اللہ تمھاری لایعنی (بے مقصد) قسموں پر مواخذہ نہیں کرے گا لیکن وہ تمھارا ان (قسموں) پر مواخذہ کرے گا جو تم نے پختہ ارادے سے کھائی ہیں تو اُس کا کفارہ دس مسکینوں کو کھانا کھلانا ہے اُس اوسط درجہ کا جو تم اپنے گھر والوں کو کھلاتے ہو یا انھیں لباس پہنانا یا ایک غلام کو آزاد کرنا ہے پھر جو یہ نہ پائے تو تین دن کے روزے (رکھے) یہ تمھاری قسموں کا کفارہ ہے جب تم قسم کھاؤ (اور اسے توڑ دو) اور اپنی قسموں کی حفاظت کرو اسی طرح اللہ اپنی آیات تمھارے لیے کھول کھول کر بیان فرماتا ہے تاکہ تم شکر ادا کرو۔ || لَا يُؤَاخِذُكُمُ ٱللَّهُ بِٱللَّغْوِ فِىٓ أَيْمَٰنِكُمْ وَلَٰكِن يُؤَاخِذُكُم بِمَا عَقَّدتُّمُ ٱلْأَيْمَٰنَ ۖ فَكَفَّٰرَتُهُۥٓ إِطْعَامُ عَشَرَةِ مَسَٰكِينَ مِنْ أَوْسَطِ مَا تُطْعِمُونَ أَهْلِيكُمْ أَوْ كِسْوَتُهُمْ أَوْ تَحْرِيرُ رَقَبَةٍۢ ۖ فَمَن لَّمْ يَجِدْ فَصِيَامُ ثَلَٰثَةِ أَيَّامٍۢ ۚ ذَٰلِكَ كَفَّٰرَةُ أَيْمَٰنِكُمْ إِذَا حَلَفْتُمْ ۚ وَٱحْفَظُوٓا۟ أَيْمَٰنَكُمْ ۚ كَذَٰلِكَ يُبَيِّنُ ٱللَّهُ لَكُمْ ءَايَٰتِهِۦ لَعَلَّكُمْ تَشْكُرُونَ ﴿٨٩﴾ Allah will not take you to task for that which is unintentional in your oaths, but He will take you to task for the oaths which ye swear in earnest. The expiation thereof is the feeding of ten of the needy with the average of that wherewith ye feed your own folk, or the clothing of them, or the liberation of a slave, and for him who findeth not (the wherewithal to do so) then a three days' fast. This is the expiation of your oaths when ye have sworn; and keep your oaths. Thus Allah expoundeth unto you His revelations in order that ye may give thanks. (5:89) || لَا يُؤَاخِذُكُمُ ٱللَّهُ بِٱللَّغْوِ فِىٓ أَيْمَٰنِكُمْ وَلَٰكِن يُؤَاخِذُكُم بِمَا عَقَّدتُّمُ ٱلْأَيْمَٰنَ ۖ فَكَفَّٰرَتُهُۥٓ إِطْعَامُ عَشَرَةِ مَسَٰكِينَ مِنْ أَوْسَطِ مَا تُطْعِمُونَ أَهْلِيكُمْ أَوْ كِسْوَتُهُمْ أَوْ تَحْرِيرُ رَقَبَةٍۢ ۖ فَمَن لَّمْ يَجِدْ فَصِيَامُ ثَلَٰثَةِ أَيَّامٍۢ ۚ ذَٰلِكَ كَفَّٰرَةُ أَيْمَٰنِكُمْ إِذَا حَلَفْتُمْ ۚ وَٱحْفَظُوٓا۟ أَيْمَٰنَكُمْ ۚ كَذَٰلِكَ يُبَيِّنُ ٱللَّهُ لَكُمْ ءَايَٰتِهِۦ لَعَلَّكُمْ تَشْكُرُونَ ﴿٨٩﴾ الله به ستاسو پر بې‌ګټې (بې‌موخې) قسمونو نیونه ونه کړي، خو پر هغو (قسمونو) به مو نیونه وکړي چې په ټینګه اراده مو خوړلي؛ نو د هغې کفاره د لسو مسکینانو خوړول دي له هغې منځنۍ درجې چې خپلې کورنۍ ته یې ورکوئ، یا یې جامې ورواغوستل، یا د یوه غلام آزادول؛ بیا څوک چې دا ونه مومي، نو درې ورځې روژې (ونیسي)؛ دا ستاسو د قسمونو کفاره ده کله چې قسم وخورئ (او مات یې کړئ)؛ او د خپلو قسمونو ساتنه وکړئ؛ همداسې الله خپل آیتونه تاسو ته ښه روښانه بیانوي تر څو شکر ادا کړئ.
