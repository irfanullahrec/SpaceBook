p وَمِنْ ءَايَٰتِهِ ٱلَّيْلُ وَٱلنَّهَارُ وَٱلشَّمْسُ وَٱلْقَمَرُ ۚ لَا تَسْجُدُوا۟ لِلشَّمْسِ وَلَا لِلْقَمَرِ وَٱسْجُدُوا۟ لِلَّهِ ٱلَّذِى خَلَقَهُنَّ إِن كُنتُمْ إِيَّاهُ تَعْبُدُونَ ﴿٣٧﴾ اور اسی کی نشانیوں میں سے ہیں رات اور دن اور سورج اور چاند نہ تم سورج کو سجدہ کرو اور نہ چاند کو اور اللہ ہی کو سجدہ کرو جس نے ان سب کو پیدا فرمایا ہے اگر تم صرف اسی کی عبادت کرتے ہو۔ || وَمِنْ ءَايَٰتِهِ ٱلَّيْلُ وَٱلنَّهَارُ وَٱلشَّمْسُ وَٱلْقَمَرُ ۚ لَا تَسْجُدُوا۟ لِلشَّمْسِ وَلَا لِلْقَمَرِ وَٱسْجُدُوا۟ لِلَّهِ ٱلَّذِى خَلَقَهُنَّ إِن كُنتُمْ إِيَّاهُ تَعْبُدُونَ ﴿٣٧﴾ And of His portents are the night and the day and the sun and the moon. Do not prostrate to the sun or the moon; but prostrate to Allah Who created them, if it is in truth Him Whom ye worship. (41:37) || وَمِنْ ءَايَٰتِهِ ٱلَّيْلُ وَٱلنَّهَارُ وَٱلشَّمْسُ وَٱلْقَمَرُ ۚ لَا تَسْجُدُوا۟ لِلشَّمْسِ وَلَا لِلْقَمَرِ وَٱسْجُدُوا۟ لِلَّهِ ٱلَّذِى خَلَقَهُنَّ إِن كُنتُمْ إِيَّاهُ تَعْبُدُونَ ﴿٣٧﴾ او د هغه له نښو څخه شپه او ورځ، لمر او سپوږمۍ دي؛ نه لمر ته سجده وکړئ او نه سپوږمۍ ته، او یوازې الله ته سجده وکړئ چې دا ټول یې پیدا کړي، که تاسو یوازې د هغه عبادت کوئ.
p فَإِنِ ٱسْتَكْبَرُوا۟ فَٱلَّذِينَ عِندَ رَبِّكَ يُسَبِّحُونَ لَهُۥ بِٱلَّيْلِ وَٱلنَّهَارِ وَهُمْ لَا يَسْـَٔمُونَ ۩ ﴿٣٨﴾ پھر اگر وہ تکبر کریں تو جو (فرشتے) آپ کے رب کے پاس ہیں وہ رات اور دن اس کی تسبیح کرتے ہیں اور وہ تھکتے نہیں ہیں۔ || فَإِنِ ٱسْتَكْبَرُوا۟ فَٱلَّذِينَ عِندَ رَبِّكَ يُسَبِّحُونَ لَهُۥ بِٱلَّيْلِ وَٱلنَّهَارِ وَهُمْ لَا يَسْـَٔمُونَ ۩ ﴿٣٨﴾ But if they are too proud - still those who are with thy Lord glorify Him night and day, and tire not. (41:38) || فَإِنِ ٱسْتَكْبَرُوا۟ فَٱلَّذِينَ عِندَ رَبِّكَ يُسَبِّحُونَ لَهُۥ بِٱلَّيْلِ وَٱلنَّهَارِ وَهُمْ لَا يَسْـَٔمُونَ ۩ ﴿٣٨﴾ بیا که هغوی لویي وکړي، نو هغه (پرښتې) چې ستاسو د رب سره دي، شپه او ورځ د هغه تسبیح وايي او نه ستړي کېږي.
p وَمِنْ ءَايَٰتِهِۦٓ أَنَّكَ تَرَى ٱلْأَرْضَ خَٰشِعَةًۭ فَإِذَآ أَنزَلْنَا عَلَيْهَا ٱلْمَآءَ ٱهْتَزَّتْ وَرَبَتْ ۚ إِنَّ ٱلَّذِىٓ أَحْيَاهَا لَمُحْىِ ٱلْمَوْتَىٰٓ ۚ إِنَّهُۥ عَلَىٰ كُلِّ شَىْءٍۢ قَدِيرٌ ﴿٣٩﴾ اور اس کی نشانیوں میں سے (یہ بھی) ہے کہ تم زمین کو دیکھتے ہو کہ وہ دبی (خشک) پڑی ہوتی ہے پھر جب ہم اس پر پانی برساتے ہیں تو وہ ابھر آتی ہے اور پھولتی ہے یقیناً وہ (اللہ) جس نے اس کو زندہ کیا ہے یقیناً وہی مردوں کو زندہ فرمانے والا ہے بے شک وہ ہر شے پر بڑی قدرت رکھنے والا ہے۔ || وَمِنْ ءَايَٰتِهِۦٓ أَنَّكَ تَرَى ٱلْأَرْضَ خَٰشِعَةًۭ فَإِذَآ أَنزَلْنَا عَلَيْهَا ٱلْمَآءَ ٱهْتَزَّتْ وَرَبَتْ ۚ إِنَّ ٱلَّذِىٓ أَحْيَاهَا لَمُحْىِ ٱلْمَوْتَىٰٓ ۚ إِنَّهُۥ عَلَىٰ كُلِّ شَىْءٍۢ قَدِيرٌ ﴿٣٩﴾ And of His portents (is this): that thou seest the earth lowly, but when We send down water thereon it thrilleth and groweth. Lo! He Who quickeneth it is verily the Quickener of the Dead. Lo! He is Able to do all things. (41:39) || وَمِنْ ءَايَٰتِهِۦٓ أَنَّكَ تَرَى ٱلْأَرْضَ خَٰشِعَةًۭ فَإِذَآ أَنزَلْنَا عَلَيْهَا ٱلْمَآءَ ٱهْتَزَّتْ وَرَبَتْ ۚ إِنَّ ٱلَّذِىٓ أَحْيَاهَا لَمُحْىِ ٱلْمَوْتَىٰٓ ۚ إِنَّهُۥ عَلَىٰ كُلِّ شَىْءٍۢ قَدِيرٌ ﴿٣٩﴾ او د هغه له نښو څخه (دا هم) ده چې ته ځمکه وینې چې ځپلې (وچه) پرته ده، بیا چې موږ پرې اوبه ووروو، نو راپورته کېږي او پړسېږي؛ یقیناً هغه (الله) چې دا یې ژوندۍ کړه، یقیناً همغه د مړو ژوندي کوونکی دی؛ بې‌شکه هغه پر هر شي لوی قدرت لرونکی دی.
p إِنَّ ٱلَّذِينَ يُلْحِدُونَ فِىٓ ءَايَٰتِنَا لَا يَخْفَوْنَ عَلَيْنَآ ۗ أَفَمَن يُلْقَىٰ فِى ٱلنَّارِ خَيْرٌ أَم مَّن يَأْتِىٓ ءَامِنًۭا يَوْمَ ٱلْقِيَٰمَةِ ۚ ٱعْمَلُوا۟ مَا شِئْتُمْ ۖ إِنَّهُۥ بِمَا تَعْمَلُونَ بَصِيرٌ ﴿٤٠﴾ بے شک جو لوگ ہماری آیتوں میں ٹیڑھا راستہ اختیار کرتے ہیں وہ ہم سے چھپے ہوئے نہیں ہیں بھلا جو آگ میں ڈالا جائے (وہ) بہتر ہے یا وہ جو قیامت کے دن امن کی حالت میں آئے گا؟ تم جو چاہو کرو بے شک وہ خوب دیکھ رہا ہے جو تم کر رہے ہو۔ || إِنَّ ٱلَّذِينَ يُلْحِدُونَ فِىٓ ءَايَٰتِنَا لَا يَخْفَوْنَ عَلَيْنَآ ۗ أَفَمَن يُلْقَىٰ فِى ٱلنَّارِ خَيْرٌ أَم مَّن يَأْتِىٓ ءَامِنًۭا يَوْمَ ٱلْقِيَٰمَةِ ۚ ٱعْمَلُوا۟ مَا شِئْتُمْ ۖ إِنَّهُۥ بِمَا تَعْمَلُونَ بَصِيرٌ ﴿٤٠﴾ Lo! those who distort Our revelations are not hid from Us. Is he who is hurled into the Fire better, or he who cometh secure on the Day of Resurrection? Do what ye will. Lo! He is Seer of what ye do. (41:40) || إِنَّ ٱلَّذِينَ يُلْحِدُونَ فِىٓ ءَايَٰتِنَا لَا يَخْفَوْنَ عَلَيْنَآ ۗ أَفَمَن يُلْقَىٰ فِى ٱلنَّارِ خَيْرٌ أَم مَّن يَأْتِىٓ ءَامِنًۭا يَوْمَ ٱلْقِيَٰمَةِ ۚ ٱعْمَلُوا۟ مَا شِئْتُمْ ۖ إِنَّهُۥ بِمَا تَعْمَلُونَ بَصِيرٌ ﴿٤٠﴾ بې‌شکه هغه کسان چې زموږ په آیتونو کې کږه لار غوره کوي، له موږ پټ نه دي؛ ایا هغه چې په اور کې واچول شي غوره دی، که هغه چې د قیامت په ورځ په امن کې راشي؟ څه چې غواړئ وې کړئ؛ بې‌شکه هغه ښه ویني چې تاسو څه کوئ.
p إِنَّ ٱلَّذِينَ كَفَرُوا۟ بِٱلذِّكْرِ لَمَّا جَآءَهُمْ ۖ وَإِنَّهُۥ لَكِتَٰبٌ عَزِيزٌۭ ﴿٤١﴾ بے شک جن لوگوں نے (کتاب) نصیحت کا انکار کیا جب کہ وہ ان کے پاس آ چکی تھی (وہ عذاب سے نہ بچیں گے) اور یقیناً یہ ایک زبردست کتاب ہے۔ || إِنَّ ٱلَّذِينَ كَفَرُوا۟ بِٱلذِّكْرِ لَمَّا جَآءَهُمْ ۖ وَإِنَّهُۥ لَكِتَٰبٌ عَزِيزٌۭ ﴿٤١﴾ Lo! those who disbelieve in the Reminder when it cometh unto them (are guilty), for lo! it is an unassailable Scripture. (41:41) || إِنَّ ٱلَّذِينَ كَفَرُوا۟ بِٱلذِّكْرِ لَمَّا جَآءَهُمْ ۖ وَإِنَّهُۥ لَكِتَٰبٌ عَزِيزٌۭ ﴿٤١﴾ بې‌شکه هغه کسان چې د نصیحت (کتاب) انکار یې وکړ کله چې هغوی ته راغی، (هغوی به له عذابه ونه ژغورل شي)؛ او یقیناً دا یو زورور کتاب دی.
p لَّا يَأْتِيهِ ٱلْبَٰطِلُ مِنۢ بَيْنِ يَدَيْهِ وَلَا مِنْ خَلْفِهِۦ ۖ تَنزِيلٌۭ مِّنْ حَكِيمٍ حَمِيدٍۢ ﴿٤٢﴾ باطل اس (قرآن) کے پاس نہ اس کے آگے سے آ سکتا ہے اور نہ ہی اس کے پیچھے سے (یہ) بڑی حکمت والے بہت تعریفوں والے (اللہ) کا نازل کیا ہوا ہے۔ || لَّا يَأْتِيهِ ٱلْبَٰطِلُ مِنۢ بَيْنِ يَدَيْهِ وَلَا مِنْ خَلْفِهِۦ ۖ تَنزِيلٌۭ مِّنْ حَكِيمٍ حَمِيدٍۢ ﴿٤٢﴾ Falsehood cannot come at it from before it or from behind it. (It is) a revelation from the Wise, the Owner of Praise. (41:42) || لَّا يَأْتِيهِ ٱلْبَٰطِلُ مِنۢ بَيْنِ يَدَيْهِ وَلَا مِنْ خَلْفِهِۦ ۖ تَنزِيلٌۭ مِّنْ حَكِيمٍ حَمِيدٍۢ ﴿٤٢﴾ باطل دې (قرآن) ته نه د هغه له مخې راتلای شي او نه یې له شا؛ (دا) د ډېر حکمتناک، ډېر ستایل شوي (الله) نازل کړی دی.
p مَّا يُقَالُ لَكَ إِلَّا مَا قَدْ قِيلَ لِلرُّسُلِ مِن قَبْلِكَ ۚ إِنَّ رَبَّكَ لَذُو مَغْفِرَةٍۢ وَذُو عِقَابٍ أَلِيمٍۢ ﴿٤٣﴾ آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) سے وہی باتیں کہی جاتی ہیں جو آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) سے پہلے رسولوں سے کہی جا چکی ہیں یقیناً آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کا رب مغفرت والا (بھی) ہے اور دردناک سزا دینے والا (بھی) ہے۔ || مَّا يُقَالُ لَكَ إِلَّا مَا قَدْ قِيلَ لِلرُّسُلِ مِن قَبْلِكَ ۚ إِنَّ رَبَّكَ لَذُو مَغْفِرَةٍۢ وَذُو عِقَابٍ أَلِيمٍۢ ﴿٤٣﴾ Naught is said unto thee (Muhammad) save what was said unto the messengers before thee. Lo! thy Lord is owner of forgiveness, and owner (also) of dire punishment. (41:43) || مَّا يُقَالُ لَكَ إِلَّا مَا قَدْ قِيلَ لِلرُّسُلِ مِن قَبْلِكَ ۚ إِنَّ رَبَّكَ لَذُو مَغْفِرَةٍۢ وَذُو عِقَابٍ أَلِيمٍۢ ﴿٤٣﴾ تاسو (صلی الله علیه وعلی آله واصحابه وسلم) ته همغه خبرې ویل کېږي چې له تاسو (صلی الله علیه وعلی آله واصحابه وسلم) مخکې رسولانو ته ویل شوې دي؛ یقیناً ستاسو (صلی الله علیه وعلی آله واصحابه وسلم) رب د بښنې خاوند (هم) دی او دردناکه سزا ورکوونکی (هم) دی.
