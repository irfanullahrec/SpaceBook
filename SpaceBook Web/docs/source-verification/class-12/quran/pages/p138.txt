p كَذَٰلِكَ يُوحِىٓ إِلَيْكَ وَإِلَى ٱلَّذِينَ مِن قَبْلِكَ ٱللَّهُ ٱلْعَزِيزُ ٱلْحَكِيمُ ﴿٣﴾ اسی طرح آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کی طرف وحی فرماتا رہا ہے اور ان (رسولوں) کی طرف جو آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) سے پہلے تھے وہ اللہ جو بہت غالب بڑی حکمت والا ہے۔ || كَذَٰلِكَ يُوحِىٓ إِلَيْكَ وَإِلَى ٱلَّذِينَ مِن قَبْلِكَ ٱللَّهُ ٱلْعَزِيزُ ٱلْحَكِيمُ ﴿٣﴾ Thus Allah the Mighty, the Knower inspireth thee (Muhammad) as (He inspired) those before thee. (42:3) || كَذَٰلِكَ يُوحِىٓ إِلَيْكَ وَإِلَى ٱلَّذِينَ مِن قَبْلِكَ ٱللَّهُ ٱلْعَزِيزُ ٱلْحَكِيمُ ﴿٣﴾ همداسې تاسو (صلی الله علیه وعلی آله واصحابه وسلم) ته او هغو (رسولانو) ته چې له تاسو (صلی الله علیه وعلی آله واصحابه وسلم) مخکې وو، وحې کوي هغه الله چې ډېر غالب، ډېر حکمتناک دی.
p لَهُۥ مَا فِى ٱلسَّمَٰوَٰتِ وَمَا فِى ٱلْأَرْضِ ۖ وَهُوَ ٱلْعَلِىُّ ٱلْعَظِيمُ ﴿٤﴾ اسی کا ہے جو کچھ آسمانوں اور زمینوں میں ہے اور وہی سب سے بلند عظمت والا ہے۔ || لَهُۥ مَا فِى ٱلسَّمَٰوَٰتِ وَمَا فِى ٱلْأَرْضِ ۖ وَهُوَ ٱلْعَلِىُّ ٱلْعَظِيمُ ﴿٤﴾ Unto Him belongeth all that is in the heavens and all that is in the earth, and He is the Sublime, the Tremendous. (42:4) || لَهُۥ مَا فِى ٱلسَّمَٰوَٰتِ وَمَا فِى ٱلْأَرْضِ ۖ وَهُوَ ٱلْعَلِىُّ ٱلْعَظِيمُ ﴿٤﴾ څه چې په آسمانونو او ځمکو کې دي، د هماغه دي، او همغه تر ټولو لوړ، د عظمت خاوند دی.
p تَكَادُ ٱلسَّمَٰوَٰتُ يَتَفَطَّرْنَ مِن فَوْقِهِنَّ ۚ وَٱلْمَلَٰٓئِكَةُ يُسَبِّحُونَ بِحَمْدِ رَبِّهِمْ وَيَسْتَغْفِرُونَ لِمَن فِى ٱلْأَرْضِ ۗ أَلَآ إِنَّ ٱللَّهَ هُوَ ٱلْغَفُورُ ٱلرَّحِيمُ ﴿٥﴾ قریب ہے کہ آسمان اپنے اوپر سے پھٹ پڑیں اور فرشتے اپنے رب کی حمد کے ساتھ تسبیح کرتے رہتے ہیں اور زمین والوں کے لیے مغفرت مانگتے رہتے ہیں آگاہ ہو جاؤ بے شک اللہ ہی بہت بخشنے والا نہایت رحم فرمانے والا ہے۔ || تَكَادُ ٱلسَّمَٰوَٰتُ يَتَفَطَّرْنَ مِن فَوْقِهِنَّ ۚ وَٱلْمَلَٰٓئِكَةُ يُسَبِّحُونَ بِحَمْدِ رَبِّهِمْ وَيَسْتَغْفِرُونَ لِمَن فِى ٱلْأَرْضِ ۗ أَلَآ إِنَّ ٱللَّهَ هُوَ ٱلْغَفُورُ ٱلرَّحِيمُ ﴿٥﴾ Almost might the heavens above be rent asunder while the angels hymn the praise of their Lord and ask forgiveness for those on the earth. Lo! Allah, He is the Forgiver, the Merciful. (42:5) || تَكَادُ ٱلسَّمَٰوَٰتُ يَتَفَطَّرْنَ مِن فَوْقِهِنَّ ۚ وَٱلْمَلَٰٓئِكَةُ يُسَبِّحُونَ بِحَمْدِ رَبِّهِمْ وَيَسْتَغْفِرُونَ لِمَن فِى ٱلْأَرْضِ ۗ أَلَآ إِنَّ ٱللَّهَ هُوَ ٱلْغَفُورُ ٱلرَّحِيمُ ﴿٥﴾ نږدې ده چې آسمانونه له پاسه وچوي؛ او پرښتې د خپل رب له ستاینې سره تسبیح وايي او د ځمکې د اوسېدونکو لپاره بښنه غواړي؛ خبر اوسئ! بې‌شکه یوازې الله ډېر بښونکی، بې‌حده مهربان دی.
p وَٱلَّذِينَ ٱتَّخَذُوا۟ مِن دُونِهِۦٓ أَوْلِيَآءَ ٱللَّهُ حَفِيظٌ عَلَيْهِمْ وَمَآ أَنتَ عَلَيْهِم بِوَكِيلٍۢ ﴿٦﴾ اور جن لوگوں نے اس کے سوا دوسرے کارساز بنا رکھے ہیں اللہ ان پر نگہبان ہے اور آپ ان کے ذمہ دار نہیں ہیں۔ || وَٱلَّذِينَ ٱتَّخَذُوا۟ مِن دُونِهِۦٓ أَوْلِيَآءَ ٱللَّهُ حَفِيظٌ عَلَيْهِمْ وَمَآ أَنتَ عَلَيْهِم بِوَكِيلٍۢ ﴿٦﴾ And as for those who choose protecting friends beside Him, Allah is Warden over them, and thou art in no wise a guardian over them. (42:6) || وَٱلَّذِينَ ٱتَّخَذُوا۟ مِن دُونِهِۦٓ أَوْلِيَآءَ ٱللَّهُ حَفِيظٌ عَلَيْهِمْ وَمَآ أَنتَ عَلَيْهِم بِوَكِيلٍۢ ﴿٦﴾ او هغه کسان چې له هغه پرته یې نور کارسازان نیولي، الله پرې څارونکی دی، او تاسو د هغوی مسؤل نه یاست.
p وَكَذَٰلِكَ أَوْحَيْنَآ إِلَيْكَ قُرْءَانًا عَرَبِيًّۭا لِّتُنذِرَ أُمَّ ٱلْقُرَىٰ وَمَنْ حَوْلَهَا وَتُنذِرَ يَوْمَ ٱلْجَمْعِ لَا رَيْبَ فِيهِ ۚ فَرِيقٌۭ فِى ٱلْجَنَّةِ وَفَرِيقٌۭ فِى ٱلسَّعِيرِ ﴿٧﴾ اور اسی طرح ہم نے آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کی طرف وحی فرمایا عربی (زبان میں) قرآن تاکہ آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) مکہ والوں اور اس کے آس پاس والوں کو ڈرائیں اور جمع ہونے کے دن سے ڈرائیں جس (کے آنے) میں کوئی شک نہیں (اس دن) ایک گروہ جنت میں ہوگا اور ایک گروہ بھڑکتی آگ میں۔ || وَكَذَٰلِكَ أَوْحَيْنَآ إِلَيْكَ قُرْءَانًا عَرَبِيًّۭا لِّتُنذِرَ أُمَّ ٱلْقُرَىٰ وَمَنْ حَوْلَهَا وَتُنذِرَ يَوْمَ ٱلْجَمْعِ لَا رَيْبَ فِيهِ ۚ فَرِيقٌۭ فِى ٱلْجَنَّةِ وَفَرِيقٌۭ فِى ٱلسَّعِيرِ ﴿٧﴾ And thus We have inspired in thee a Lecture in Arabic, that thou mayst warn the mother-town and those around it, and mayst warn of a day of assembling whereof there is no doubt. A host will be in the Garden, and a host of them in the Flame. (42:7) || وَكَذَٰلِكَ أَوْحَيْنَآ إِلَيْكَ قُرْءَانًا عَرَبِيًّۭا لِّتُنذِرَ أُمَّ ٱلْقُرَىٰ وَمَنْ حَوْلَهَا وَتُنذِرَ يَوْمَ ٱلْجَمْعِ لَا رَيْبَ فِيهِ ۚ فَرِيقٌۭ فِى ٱلْجَنَّةِ وَفَرِيقٌۭ فِى ٱلسَّعِيرِ ﴿٧﴾ او همداسې موږ تاسو (صلی الله علیه وعلی آله واصحابه وسلم) ته په عربي (ژبه) قرآن وحې کړ تر څو تاسو (صلی الله علیه وعلی آله واصحابه وسلم) د مکې خلک او د هغې د شاوخوا خلک ووېروئ، او د راټولېدو له هغې ورځې یې ووېروئ چې (په راتګ) کې یې هېڅ شک نشته؛ (په هغه ورځ) یوه ډله به په جنت کې وي او یوه ډله به په بل اور کې.
p وَلَوْ شَآءَ ٱللَّهُ لَجَعَلَهُمْ أُمَّةًۭ وَٰحِدَةًۭ وَلَٰكِن يُدْخِلُ مَن يَشَآءُ فِى رَحْمَتِهِۦ ۚ وَٱلظَّٰلِمُونَ مَا لَهُم مِّن وَلِىٍّۢ وَلَا نَصِيرٍ ﴿٨﴾ اور اگر اللہ چاہتا تو یقیناً انھیں ایک ہی امت بنا دیتا لیکن وہ جسے چاہتا ہے اپنی رحمت میں داخل فرماتا ہے اور ظالموں کے لیے نہ کوئی دوست ہے اور نہ کوئی مددگار۔ || وَلَوْ شَآءَ ٱللَّهُ لَجَعَلَهُمْ أُمَّةًۭ وَٰحِدَةًۭ وَلَٰكِن يُدْخِلُ مَن يَشَآءُ فِى رَحْمَتِهِۦ ۚ وَٱلظَّٰلِمُونَ مَا لَهُم مِّن وَلِىٍّۢ وَلَا نَصِيرٍ ﴿٨﴾ Had Allah willed, He could have made them one community, but Allah bringeth whom He will into His mercy. And the wrong-doers have no friend nor helper. (42:8) || وَلَوْ شَآءَ ٱللَّهُ لَجَعَلَهُمْ أُمَّةًۭ وَٰحِدَةًۭ وَلَٰكِن يُدْخِلُ مَن يَشَآءُ فِى رَحْمَتِهِۦ ۚ وَٱلظَّٰلِمُونَ مَا لَهُم مِّن وَلِىٍّۢ وَلَا نَصِيرٍ ﴿٨﴾ او که الله غوښتي وای، یقیناً به یې هغوی یو امت ګرځولي وای؛ خو هغه چې څوک وغواړي په خپل رحمت کې یې ننباسي؛ او د ظالمانو لپاره نه کوم دوست شته او نه کوم مرستندوی.
p أَمِ ٱتَّخَذُوا۟ مِن دُونِهِۦٓ أَوْلِيَآءَ ۖ فَٱللَّهُ هُوَ ٱلْوَلِىُّ وَهُوَ يُحْىِ ٱلْمَوْتَىٰ وَهُوَ عَلَىٰ كُلِّ شَىْءٍۢ قَدِيرٌۭ ﴿٩﴾ کیا انھوں نے اس کے سوا دوسروں کو مددگار بنا رکھا ہے تو اللہ ہی مددگار ہے اور وہی مُردوں کو زندہ فرمائے گا اور وہ ہر چیز پر بڑی قدرت رکھنے والا ہے۔ || أَمِ ٱتَّخَذُوا۟ مِن دُونِهِۦٓ أَوْلِيَآءَ ۖ فَٱللَّهُ هُوَ ٱلْوَلِىُّ وَهُوَ يُحْىِ ٱلْمَوْتَىٰ وَهُوَ عَلَىٰ كُلِّ شَىْءٍۢ قَدِيرٌۭ ﴿٩﴾ Or have they chosen protecting friends besides Him? But Allah, He (alone) is the Protecting Friend. He quickeneth the dead, and He is Able to do all things. (42:9) || أَمِ ٱتَّخَذُوا۟ مِن دُونِهِۦٓ أَوْلِيَآءَ ۖ فَٱللَّهُ هُوَ ٱلْوَلِىُّ وَهُوَ يُحْىِ ٱلْمَوْتَىٰ وَهُوَ عَلَىٰ كُلِّ شَىْءٍۢ قَدِيرٌۭ ﴿٩﴾ ایا هغوی له هغه پرته نور مرستندویان نیولي؟ نو یوازې الله مرستندوی دی، او همغه به مړي ژوندي کړي، او هغه پر هر شي لوی قدرت لرونکی دی.
p وَمَا ٱخْتَلَفْتُمْ فِيهِ مِن شَىْءٍۢ فَحُكْمُهُۥٓ إِلَى ٱللَّهِ ۚ ذَٰلِكُمُ ٱللَّهُ رَبِّى عَلَيْهِ تَوَكَّلْتُ وَإِلَيْهِ أُنِيبُ ﴿١٠﴾ اور تم جس بات میں بھی اختلاف کرتے ہو اس کا فیصلہ اللہ ہی کی طرف (سے) ہوگا یہی اللہ میرا رب ہے اسی پر میں نے بھروسہ کیا اور اسی کی طرف میں رجوع کرتا ہوں۔ || وَمَا ٱخْتَلَفْتُمْ فِيهِ مِن شَىْءٍۢ فَحُكْمُهُۥٓ إِلَى ٱللَّهِ ۚ ذَٰلِكُمُ ٱللَّهُ رَبِّى عَلَيْهِ تَوَكَّلْتُ وَإِلَيْهِ أُنِيبُ ﴿١٠﴾ And in whatsoever ye differ, the verdict therein belongeth to Allah. Such is my Lord, in Whom I put my trust, and unto Whom I turn. (42:10) || وَمَا ٱخْتَلَفْتُمْ فِيهِ مِن شَىْءٍۢ فَحُكْمُهُۥٓ إِلَى ٱللَّهِ ۚ ذَٰلِكُمُ ٱللَّهُ رَبِّى عَلَيْهِ تَوَكَّلْتُ وَإِلَيْهِ أُنِيبُ ﴿١٠﴾ او تاسو په هره خبره کې چې اختلاف کوئ، د هغې پرېکړه به یوازې د الله (له لوري) وي؛ همدا الله زما رب دی، پر هماغه مې توکل کړی او هماغه ته رجوع کوم.
p فَاطِرُ ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ ۚ جَعَلَ لَكُم مِّنْ أَنفُسِكُمْ أَزْوَٰجًۭا وَمِنَ ٱلْأَنْعَٰمِ أَزْوَٰجًۭا ۖ يَذْرَؤُكُمْ فِيهِ ۚ لَيْسَ كَمِثْلِهِۦ شَىْءٌۭ ۖ وَهُوَ ٱلسَّمِيعُ ٱلْبَصِيرُ ﴿١١﴾ (اللہ) آسمانوں اور زمینوں کا پیدا فرمانے والا ہے اس نے تمھارے لیے تم ہی میں سے جوڑے بنائے اور چوپایوں میں سے جوڑے بنائے (اس طریقے سے) وہ تمھیں اس (دنیا) میں پھیلاتا ہے اس (اللہ) جیسی کوئی شے نہیں اور وہ خوب سننے والا خوب دیکھنے والا ہے۔ || فَاطِرُ ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ ۚ جَعَلَ لَكُم مِّنْ أَنفُسِكُمْ أَزْوَٰجًۭا وَمِنَ ٱلْأَنْعَٰمِ أَزْوَٰجًۭا ۖ يَذْرَؤُكُمْ فِيهِ ۚ لَيْسَ كَمِثْلِهِۦ شَىْءٌۭ ۖ وَهُوَ ٱلسَّمِيعُ ٱلْبَصِيرُ ﴿١١﴾ The Creator of the heavens and the earth. He hath made for you pairs of yourselves, and of the cattle also pairs, whereby He multiplieth you. Naught is as His likeness; and He is the Hearer, the Seer. (42:11) || فَاطِرُ ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ ۚ جَعَلَ لَكُم مِّنْ أَنفُسِكُمْ أَزْوَٰجًۭا وَمِنَ ٱلْأَنْعَٰمِ أَزْوَٰجًۭا ۖ يَذْرَؤُكُمْ فِيهِ ۚ لَيْسَ كَمِثْلِهِۦ شَىْءٌۭ ۖ وَهُوَ ٱلسَّمِيعُ ٱلْبَصِيرُ ﴿١١﴾ (الله) د آسمانونو او ځمکو پیدا کوونکی دی؛ هغه ستاسو لپاره له تاسو څخه جوړې جوړې کړې او له څارویو څخه یې جوړې جوړې کړې؛ (په دې طریقه) هغه تاسو په دې (نړۍ) کې خپروي؛ د هغه (الله) په څېر هېڅ شی نشته، او هغه ښه اورېدونکی، ښه لیدونکی دی.
