p يَٰٓأَيُّهَا ٱلنَّاسُ إِنَّا خَلَقْنَٰكُم مِّن ذَكَرٍۢ وَأُنثَىٰ وَجَعَلْنَٰكُمْ شُعُوبًۭا وَقَبَآئِلَ لِتَعَارَفُوٓا۟ ۚ إِنَّ أَكْرَمَكُمْ عِندَ ٱللَّهِ أَتْقَىٰكُمْ ۚ إِنَّ ٱللَّهَ عَلِيمٌ خَبِيرٌۭ ﴿١٣﴾ اے لوگو! بے شک ہم نے تمھیں ایک مرد اور ایک عورت سے پیدا فرمایا اور ہم نے تمھیں (مختلف) قومیں اور قبیلے بنایا تاکہ تم ایک دوسرے کو پہچان سکو بے شک اللہ کے نزدیک تم میں زیادہ عزت والا وہ ہے جو تم میں سے زیادہ پرہیزگار ہو بے شک اللہ خوب جاننے والا بڑا باخبر ہے۔ || يَٰٓأَيُّهَا ٱلنَّاسُ إِنَّا خَلَقْنَٰكُم مِّن ذَكَرٍۢ وَأُنثَىٰ وَجَعَلْنَٰكُمْ شُعُوبًۭا وَقَبَآئِلَ لِتَعَارَفُوٓا۟ ۚ إِنَّ أَكْرَمَكُمْ عِندَ ٱللَّهِ أَتْقَىٰكُمْ ۚ إِنَّ ٱللَّهَ عَلِيمٌ خَبِيرٌۭ ﴿١٣﴾ O mankind! Lo! We have created you male and female, and have made you nations and tribes that ye may know one another. Lo! the noblest of you, in the sight of Allah, is the best in conduct. Lo! Allah is Knower, Aware. (49:13) || يَٰٓأَيُّهَا ٱلنَّاسُ إِنَّا خَلَقْنَٰكُم مِّن ذَكَرٍۢ وَأُنثَىٰ وَجَعَلْنَٰكُمْ شُعُوبًۭا وَقَبَآئِلَ لِتَعَارَفُوٓا۟ ۚ إِنَّ أَكْرَمَكُمْ عِندَ ٱللَّهِ أَتْقَىٰكُمْ ۚ إِنَّ ٱللَّهَ عَلِيمٌ خَبِيرٌۭ ﴿١٣﴾ ای خلکو! بې‌شکه موږ تاسو له یوه نارینه او یوې ښځې پیدا کړئ، او تاسو مو (بېلابېل) قومونه او قبیلې کړئ تر څو یو بل وپېژنئ؛ بې‌شکه د الله په نزد ستاسو تر ټولو ډېر عزتمن هغه دی چې له تاسو ډېر پرهېزګار وي؛ بې‌شکه الله ښه پوه، ډېر خبر دی.
p ۞ قَالَتِ ٱلْأَعْرَابُ ءَامَنَّا ۖ قُل لَّمْ تُؤْمِنُوا۟ وَلَٰكِن قُولُوٓا۟ أَسْلَمْنَا وَلَمَّا يَدْخُلِ ٱلْإِيمَٰنُ فِى قُلُوبِكُمْ ۖ وَإِن تُطِيعُوا۟ ٱللَّهَ وَرَسُولَهُۥ لَا يَلِتْكُم مِّنْ أَعْمَٰلِكُمْ شَيْـًٔا ۚ إِنَّ ٱللَّهَ غَفُورٌۭ رَّحِيمٌ ﴿١٤﴾ دیہاتیوں نے کہا ہم ایمان لے آئے آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) فرما دیجیے تم ایمان تو نہیں لائے لیکن یہ کہو ہم اسلام لے آئے اور ابھی تک ایمان تمھارے دلوں میں داخل ہی نہیں ہوا اور اگر تم نے اللہ اور اُس کے رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کی اطاعت کی (تو) وہ تمھارے اعمال (کے ثواب میں) سے کچھ بھی کمی نہیں کرے گا بے شک اللہ بڑا ہی بخشنے والا نہایت رحم فرمانے والا ہے۔ || ۞ قَالَتِ ٱلْأَعْرَابُ ءَامَنَّا ۖ قُل لَّمْ تُؤْمِنُوا۟ وَلَٰكِن قُولُوٓا۟ أَسْلَمْنَا وَلَمَّا يَدْخُلِ ٱلْإِيمَٰنُ فِى قُلُوبِكُمْ ۖ وَإِن تُطِيعُوا۟ ٱللَّهَ وَرَسُولَهُۥ لَا يَلِتْكُم مِّنْ أَعْمَٰلِكُمْ شَيْـًٔا ۚ إِنَّ ٱللَّهَ غَفُورٌۭ رَّحِيمٌ ﴿١٤﴾ The wandering Arabs say: We believe. Say (unto them, O Muhammad): Ye believe not, but rather say "We submit," for the faith hath not yet entered into your hearts. Yet, if ye obey Allah and His messenger, He will not withhold from you aught of (the reward of) your deeds. Lo! Allah is Forgiving, Merciful. (49:14) || ۞ قَالَتِ ٱلْأَعْرَابُ ءَامَنَّا ۖ قُل لَّمْ تُؤْمِنُوا۟ وَلَٰكِن قُولُوٓا۟ أَسْلَمْنَا وَلَمَّا يَدْخُلِ ٱلْإِيمَٰنُ فِى قُلُوبِكُمْ ۖ وَإِن تُطِيعُوا۟ ٱللَّهَ وَرَسُولَهُۥ لَا يَلِتْكُم مِّنْ أَعْمَٰلِكُمْ شَيْـًٔا ۚ إِنَّ ٱللَّهَ غَفُورٌۭ رَّحِيمٌ ﴿١٤﴾ کلیوالو وویل: موږ ایمان راوړ؛ تاسو (صلی الله علیه وعلی آله واصحابه وسلم) ووایئ: تاسو خو ایمان نه دی راوړی، خو داسې ووایئ چې موږ اسلام راوړ؛ او لا تر اوسه ایمان ستاسو زړونو ته نه دی ننوتی؛ او که تاسو د الله او د هغه رسول (صلی الله علیه وعلی آله واصحابه وسلم) اطاعت وکړ، (نو) هغه به ستاسو د عملونو (له ثوابه) هېڅ کم نه کړي؛ بې‌شکه الله ډېر بښونکی، بې‌حده مهربان دی.
p إِنَّمَا ٱلْمُؤْمِنُونَ ٱلَّذِينَ ءَامَنُوا۟ بِٱللَّهِ وَرَسُولِهِۦ ثُمَّ لَمْ يَرْتَابُوا۟ وَجَٰهَدُوا۟ بِأَمْوَٰلِهِمْ وَأَنفُسِهِمْ فِى سَبِيلِ ٱللَّهِ ۚ أُو۟لَٰٓئِكَ هُمُ ٱلصَّٰدِقُونَ ﴿١٥﴾ بے شک ایمان والے تو صرف وہ لوگ ہیں جو اللہ اور اُس کے رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) پر ایمان لائے پھر کسی شک میں نہ پڑے اور انھوں نے اللہ کی راہ میں اپنے مالوں اور اپنی جانوں سے جہاد کیا یہی لوگ سچے ہیں۔ || إِنَّمَا ٱلْمُؤْمِنُونَ ٱلَّذِينَ ءَامَنُوا۟ بِٱللَّهِ وَرَسُولِهِۦ ثُمَّ لَمْ يَرْتَابُوا۟ وَجَٰهَدُوا۟ بِأَمْوَٰلِهِمْ وَأَنفُسِهِمْ فِى سَبِيلِ ٱللَّهِ ۚ أُو۟لَٰٓئِكَ هُمُ ٱلصَّٰدِقُونَ ﴿١٥﴾ The (true) believers are those only who believe in Allah and His messenger and afterward doubt not, but strive with their wealth and their lives for the cause of Allah. Such are the sincere. (49:15) || إِنَّمَا ٱلْمُؤْمِنُونَ ٱلَّذِينَ ءَامَنُوا۟ بِٱللَّهِ وَرَسُولِهِۦ ثُمَّ لَمْ يَرْتَابُوا۟ وَجَٰهَدُوا۟ بِأَمْوَٰلِهِمْ وَأَنفُسِهِمْ فِى سَبِيلِ ٱللَّهِ ۚ أُو۟لَٰٓئِكَ هُمُ ٱلصَّٰدِقُونَ ﴿١٥﴾ بې‌شکه مؤمنان خو یوازې هغه کسان دي چې پر الله او د هغه پر رسول (صلی الله علیه وعلی آله واصحابه وسلم) یې ایمان راوړ، بیا په هېڅ شک کې ونه لوېدل، او د الله په لار کې یې په خپلو مالونو او ځانونو جهاد وکړ؛ همدا خلک رښتیني دي.
p قُلْ أَتُعَلِّمُونَ ٱللَّهَ بِدِينِكُمْ وَٱللَّهُ يَعْلَمُ مَا فِى ٱلسَّمَٰوَٰتِ وَمَا فِى ٱلْأَرْضِ ۚ وَٱللَّهُ بِكُلِّ شَىْءٍ عَلِيمٌۭ ﴿١٦﴾ آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) فرما دیجیے کیا تم اللہ کو اپنی دین داری جتلاتے ہو حالاں کہ اللہ جانتا ہے جو کچھ آسمانوں میں ہے اور جو کچھ زمینوں میں ہے اور اللہ ہر چیز کو خوب جاننے والا ہے۔ || قُلْ أَتُعَلِّمُونَ ٱللَّهَ بِدِينِكُمْ وَٱللَّهُ يَعْلَمُ مَا فِى ٱلسَّمَٰوَٰتِ وَمَا فِى ٱلْأَرْضِ ۚ وَٱللَّهُ بِكُلِّ شَىْءٍ عَلِيمٌۭ ﴿١٦﴾ Say (unto them, O Muhammad): Would ye teach Allah your religion, when Allah knoweth all that is in the heavens and all that is in the earth, and Allah is Aware of all things? (49:16) || قُلْ أَتُعَلِّمُونَ ٱللَّهَ بِدِينِكُمْ وَٱللَّهُ يَعْلَمُ مَا فِى ٱلسَّمَٰوَٰتِ وَمَا فِى ٱلْأَرْضِ ۚ وَٱللَّهُ بِكُلِّ شَىْءٍ عَلِيمٌۭ ﴿١٦﴾ تاسو (صلی الله علیه وعلی آله واصحابه وسلم) ووایئ: ایا تاسو الله ته خپله دینداري ښکاره کوئ؟ حال دا چې الله پوهېږي څه چې په آسمانونو کې دي او څه چې په ځمکو کې دي؛ او الله په هر شي ښه پوه دی.
p يَمُنُّونَ عَلَيْكَ أَنْ أَسْلَمُوا۟ ۖ قُل لَّا تَمُنُّوا۟ عَلَىَّ إِسْلَٰمَكُم ۖ بَلِ ٱللَّهُ يَمُنُّ عَلَيْكُمْ أَنْ هَدَىٰكُمْ لِلْإِيمَٰنِ إِن كُنتُمْ صَٰدِقِينَ ﴿١٧﴾ وہ آپ پر احسان جتلاتے ہیں کہ وہ اسلام لے آئے آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) فرما دیجیے تم اپنے اسلام کا مجھ پر احسان نہ جتلاؤ بلکہ اللہ تم پر احسان فرماتا ہے کہ اُس نے تمھیں ایمان کی طرف ہدایت فرمائی اگر تم سچے ہو۔ || يَمُنُّونَ عَلَيْكَ أَنْ أَسْلَمُوا۟ ۖ قُل لَّا تَمُنُّوا۟ عَلَىَّ إِسْلَٰمَكُم ۖ بَلِ ٱللَّهُ يَمُنُّ عَلَيْكُمْ أَنْ هَدَىٰكُمْ لِلْإِيمَٰنِ إِن كُنتُمْ صَٰدِقِينَ ﴿١٧﴾ They make it a favour unto thee (Muhammad) that they have surrendered (unto Him). Say: Deem not your Surrender a favour unto me; but Allah doth confer a favour on you, inasmuch as He hath led you to the Faith, if ye are earnest. (49:17) || يَمُنُّونَ عَلَيْكَ أَنْ أَسْلَمُوا۟ ۖ قُل لَّا تَمُنُّوا۟ عَلَىَّ إِسْلَٰمَكُم ۖ بَلِ ٱللَّهُ يَمُنُّ عَلَيْكُمْ أَنْ هَدَىٰكُمْ لِلْإِيمَٰنِ إِن كُنتُمْ صَٰدِقِينَ ﴿١٧﴾ هغوی پر تاسو احسان بادوي چې اسلام یې راوړی؛ تاسو (صلی الله علیه وعلی آله واصحابه وسلم) ووایئ: پر ما د خپل اسلام احسان مه بادوئ، بلکې الله پر تاسو احسان کوي چې تاسو یې ایمان ته لارښوونه وکړه، که رښتیني یاست.
p إِنَّ ٱللَّهَ يَعْلَمُ غَيْبَ ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ ۚ وَٱللَّهُ بَصِيرٌۢ بِمَا تَعْمَلُونَ ﴿١٨﴾ بے شک اللہ آسمانوں اور زمینوں کی (سب) پوشیدہ باتیں جانتا ہے اور اللہ خوب دیکھنے والا ہے جو کچھ تم کر رہے ہو۔ || إِنَّ ٱللَّهَ يَعْلَمُ غَيْبَ ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ ۚ وَٱللَّهُ بَصِيرٌۢ بِمَا تَعْمَلُونَ ﴿١٨﴾ Lo! Allah knoweth the Unseen of the heavens and the earth. And Allah is Seer of what ye do. (49:18) || إِنَّ ٱللَّهَ يَعْلَمُ غَيْبَ ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ ۚ وَٱللَّهُ بَصِيرٌۢ بِمَا تَعْمَلُونَ ﴿١٨﴾ بې‌شکه الله د آسمانونو او ځمکو (ټولې) پټې خبرې پېژني؛ او الله ښه لیدونکی دی په هغه څه چې تاسو یې کوئ.
