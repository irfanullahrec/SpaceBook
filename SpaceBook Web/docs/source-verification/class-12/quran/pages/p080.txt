p إِنَّمَا يُرِيدُ ٱلشَّيْطَٰنُ أَن يُوقِعَ بَيْنَكُمُ ٱلْعَدَٰوَةَ وَٱلْبَغْضَآءَ فِى ٱلْخَمْرِ وَٱلْمَيْسِرِ وَيَصُدَّكُمْ عَن ذِكْرِ ٱللَّهِ وَعَنِ ٱلصَّلَوٰةِ ۖ فَهَلْ أَنتُم مُّنتَهُونَ ﴿٩١﴾ بے شک شیطان یہی چاہتا ہے کہ تمھارے درمیان شراب اور جوئے کے ذریعہ عداوت اور بغض ڈال دے اور تمھیں اللہ کے ذکر اور نماز سے روک دے تو کیا تم (ان ناپاک چیزوں سے) باز آنے والے ہو؟ || إِنَّمَا يُرِيدُ ٱلشَّيْطَٰنُ أَن يُوقِعَ بَيْنَكُمُ ٱلْعَدَٰوَةَ وَٱلْبَغْضَآءَ فِى ٱلْخَمْرِ وَٱلْمَيْسِرِ وَيَصُدَّكُمْ عَن ذِكْرِ ٱللَّهِ وَعَنِ ٱلصَّلَوٰةِ ۖ فَهَلْ أَنتُم مُّنتَهُونَ ﴿٩١﴾ Satan seeketh only to cast among you enmity and hatred by means of strong drink and games of chance, and to turn you from remembrance of Allah and from (His) worship. Will ye then have done? (5:91) || إِنَّمَا يُرِيدُ ٱلشَّيْطَٰنُ أَن يُوقِعَ بَيْنَكُمُ ٱلْعَدَٰوَةَ وَٱلْبَغْضَآءَ فِى ٱلْخَمْرِ وَٱلْمَيْسِرِ وَيَصُدَّكُمْ عَن ذِكْرِ ٱللَّهِ وَعَنِ ٱلصَّلَوٰةِ ۖ فَهَلْ أَنتُم مُّنتَهُونَ ﴿٩١﴾ بې‌شکه شیطان یوازې دا غواړي چې د شرابو او قمار له لارې ستاسو ترمنځ دښمني او کینه واچوي او تاسو د الله له ذکر او لمانځه منع کړي؛ نو ایا تاسو (له دې ناپاکو شیانو) لاس اخیستونکي یاست؟
p وَأَطِيعُوا۟ ٱللَّهَ وَأَطِيعُوا۟ ٱلرَّسُولَ وَٱحْذَرُوا۟ ۚ فَإِن تَوَلَّيْتُمْ فَٱعْلَمُوٓا۟ أَنَّمَا عَلَىٰ رَسُولِنَا ٱلْبَلَٰغُ ٱلْمُبِينُ ﴿٩٢﴾ اور اللہ کی اطاعت کرو اور رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کی اطاعت کرو اور (نافرمانی سے) بچتے رہو پس اگر تم منہ پھیرے رہو تو جان لو کہ بے شک ہمارے رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کے ذمہ تو صرف واضح طور پر (ہمارے احکامات) پہنچا دینا ہے۔ || وَأَطِيعُوا۟ ٱللَّهَ وَأَطِيعُوا۟ ٱلرَّسُولَ وَٱحْذَرُوا۟ ۚ فَإِن تَوَلَّيْتُمْ فَٱعْلَمُوٓا۟ أَنَّمَا عَلَىٰ رَسُولِنَا ٱلْبَلَٰغُ ٱلْمُبِينُ ﴿٩٢﴾ Obey Allah and obey the messenger, and beware! But if ye turn away, then know that the duty of Our messenger is only plain conveyance (of the message). (5:92) || وَأَطِيعُوا۟ ٱللَّهَ وَأَطِيعُوا۟ ٱلرَّسُولَ وَٱحْذَرُوا۟ ۚ فَإِن تَوَلَّيْتُمْ فَٱعْلَمُوٓا۟ أَنَّمَا عَلَىٰ رَسُولِنَا ٱلْبَلَٰغُ ٱلْمُبِينُ ﴿٩٢﴾ او د الله اطاعت وکړئ، د رسول (صلی الله علیه وعلی آله واصحابه وسلم) اطاعت وکړئ او (له نافرمانۍ) ځان وساتئ؛ نو که مخ مو واړاوه، پوه شئ چې بې‌شکه زموږ پر رسول (صلی الله علیه وعلی آله واصحابه وسلم) خو یوازې په ښکاره ډول (زموږ د حکمونو) رسول دي.
p لَيْسَ عَلَى ٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّٰلِحَٰتِ جُنَاحٌۭ فِيمَا طَعِمُوٓا۟ إِذَا مَا ٱتَّقَوا۟ وَّءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّٰلِحَٰتِ ثُمَّ ٱتَّقَوا۟ وَّءَامَنُوا۟ ثُمَّ ٱتَّقَوا۟ وَّأَحْسَنُوا۟ ۗ وَٱللَّهُ يُحِبُّ ٱلْمُحْسِنِينَ ﴿٩٣﴾ ایمان والوں اور نیک عمل کرنے والوں پر کوئی گناہ نہیں اس میں جو (حرمت کا حکم نازل ہونے سے پہلے) انھوں نے کھایا (پیا) جب کہ وہ اللہ سے ڈرتے رہے اور ایمان لائے اور انھوں نے نیک کام کیے پھر ڈرتے رہے اور (آنے والے حکم پر) ایمان لائے پھر ڈرتے رہے اور انھوں نے نیک کام کیے اور اللہ نیک کام کرنے والوں کو پسند فرماتا ہے۔ || لَيْسَ عَلَى ٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّٰلِحَٰتِ جُنَاحٌۭ فِيمَا طَعِمُوٓا۟ إِذَا مَا ٱتَّقَوا۟ وَّءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّٰلِحَٰتِ ثُمَّ ٱتَّقَوا۟ وَّءَامَنُوا۟ ثُمَّ ٱتَّقَوا۟ وَّأَحْسَنُوا۟ ۗ وَٱللَّهُ يُحِبُّ ٱلْمُحْسِنِينَ ﴿٩٣﴾ There shall be no sin (imputed) unto those who believe and do good works for what they may have eaten (in the past). So be mindful of your duty (to Allah), and believe, and do good works; and again: be mindful of your duty, and believe; and once again: be mindful of your duty, and do right. Allah loveth the good. (5:93) || لَيْسَ عَلَى ٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّٰلِحَٰتِ جُنَاحٌۭ فِيمَا طَعِمُوٓا۟ إِذَا مَا ٱتَّقَوا۟ وَّءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّٰلِحَٰتِ ثُمَّ ٱتَّقَوا۟ وَّءَامَنُوا۟ ثُمَّ ٱتَّقَوا۟ وَّأَحْسَنُوا۟ ۗ وَٱللَّهُ يُحِبُّ ٱلْمُحْسِنِينَ ﴿٩٣﴾ پر مؤمنانو او نېک عمل کوونکو په هغه څه کې هېڅ ګناه نشته چې (د حرمت د حکم له نازلېدو مخکې) یې خوړلي (څښلي)، کله چې له الله وېرېدل، ایمان یې راوړ او نېک کارونه یې وکړل، بیا وېرېدل او (پر راتلونکي حکم) یې ایمان راوړ، بیا وېرېدل او نېک کارونه یې وکړل؛ او الله نېکوکاران خوښوي.
p يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ لَيَبْلُوَنَّكُمُ ٱللَّهُ بِشَىْءٍۢ مِّنَ ٱلصَّيْدِ تَنَالُهُۥٓ أَيْدِيكُمْ وَرِمَاحُكُمْ لِيَعْلَمَ ٱللَّهُ مَن يَخَافُهُۥ بِٱلْغَيْبِ ۚ فَمَنِ ٱعْتَدَىٰ بَعْدَ ذَٰلِكَ فَلَهُۥ عَذَابٌ أَلِيمٌۭ ﴿٩٤﴾ اے ایمان والو! اللہ ضرور تمھیں آزمائے گا کسی قدر (اس) شکار سے جس تک تمھارے ہاتھ اور تمھارے نیزے پہنچ سکتے ہیں تاکہ اللہ ظاہر کر دے کہ کون اُس سے بغیر دیکھے ڈرتا ہے پس جس کسی نے اس کے بعد بھی زیادتی کی تو اس کے لیے دردناک عذاب ہے۔ || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ لَيَبْلُوَنَّكُمُ ٱللَّهُ بِشَىْءٍۢ مِّنَ ٱلصَّيْدِ تَنَالُهُۥٓ أَيْدِيكُمْ وَرِمَاحُكُمْ لِيَعْلَمَ ٱللَّهُ مَن يَخَافُهُۥ بِٱلْغَيْبِ ۚ فَمَنِ ٱعْتَدَىٰ بَعْدَ ذَٰلِكَ فَلَهُۥ عَذَابٌ أَلِيمٌۭ ﴿٩٤﴾ O ye who believe! Allah will surely try you somewhat (in the matter) of the game which ye take with your hands and your spears, that Allah may know him who feareth Him in secret. Whoso transgresseth after this, for him there is a painful doom. (5:94) || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ لَيَبْلُوَنَّكُمُ ٱللَّهُ بِشَىْءٍۢ مِّنَ ٱلصَّيْدِ تَنَالُهُۥٓ أَيْدِيكُمْ وَرِمَاحُكُمْ لِيَعْلَمَ ٱللَّهُ مَن يَخَافُهُۥ بِٱلْغَيْبِ ۚ فَمَنِ ٱعْتَدَىٰ بَعْدَ ذَٰلِكَ فَلَهُۥ عَذَابٌ أَلِيمٌۭ ﴿٩٤﴾ ای مؤمنانو! الله به خامخا تاسو په یو څه (هغه) ښکار وازمويي چې ستاسو لاسونه او نېزې ورته رسېږي، تر څو الله ښکاره کړي چې څوک ترې په غیب کې وېرېږي؛ نو چا چې له دې وروسته هم تېری وکړ، د هغه لپاره دردناک عذاب دی.
p يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ لَا تَقْتُلُوا۟ ٱلصَّيْدَ وَأَنتُمْ حُرُمٌۭ ۚ وَمَن قَتَلَهُۥ مِنكُم مُّتَعَمِّدًۭا فَجَزَآءٌۭ مِّثْلُ مَا قَتَلَ مِنَ ٱلنَّعَمِ يَحْكُمُ بِهِۦ ذَوَا عَدْلٍۢ مِّنكُمْ هَدْيًۢا بَٰلِغَ ٱلْكَعْبَةِ أَوْ كَفَّٰرَةٌۭ طَعَامُ مَسَٰكِينَ أَوْ عَدْلُ ذَٰلِكَ صِيَامًۭا لِّيَذُوقَ وَبَالَ أَمْرِهِۦ ۗ عَفَا ٱللَّهُ عَمَّا سَلَفَ ۚ وَمَنْ عَادَ فَيَنتَقِمُ ٱللَّهُ مِنْهُ ۗ وَٱللَّهُ عَزِيزٌۭ ذُو ٱنتِقَامٍ ﴿٩٥﴾ اے ایمان والو! احرام کی حالت میں تم شکار نہ مارو اور جس نے تم میں سے جان بوجھ کر شکار مارا تو (اس) کا بدلہ ہے مویشیوں میں سے اسی کی مثل جو اس نے مارا تم میں سے دو عادل آدمی اُس کا فیصلہ کریں گے یہ قربانی ہو گی جسے کعبہ تک پہنچایا ہو گا یا اُس کا کفارہ مساکین کو کھانا کھلانا ہو گا یا اُس کا بدل روزہ رکھنا ہو گا تاکہ وہ اپنے کیے (کی سزا) کا مزہ چکھے اللہ نے وہ معاف فرما دیا جو پہلے ہو چکا اور جو دوبارہ ایسا کرے تو اللہ اُس سے انتقام لے گا اور اللہ بہت غالب، انتقام لینے والا ہے۔ || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ لَا تَقْتُلُوا۟ ٱلصَّيْدَ وَأَنتُمْ حُرُمٌۭ ۚ وَمَن قَتَلَهُۥ مِنكُم مُّتَعَمِّدًۭا فَجَزَآءٌۭ مِّثْلُ مَا قَتَلَ مِنَ ٱلنَّعَمِ يَحْكُمُ بِهِۦ ذَوَا عَدْلٍۢ مِّنكُمْ هَدْيًۢا بَٰلِغَ ٱلْكَعْبَةِ أَوْ كَفَّٰرَةٌۭ طَعَامُ مَسَٰكِينَ أَوْ عَدْلُ ذَٰلِكَ صِيَامًۭا لِّيَذُوقَ وَبَالَ أَمْرِهِۦ ۗ عَفَا ٱللَّهُ عَمَّا سَلَفَ ۚ وَمَنْ عَادَ فَيَنتَقِمُ ٱللَّهُ مِنْهُ ۗ وَٱللَّهُ عَزِيزٌۭ ذُو ٱنتِقَامٍ ﴿٩٥﴾ O ye who believe! Kill no wild game while ye are on the pilgrimage. Whoso of you killeth it of set purpose he shall pay its forfeit in the equivalent of that which he hath killed, of domestic animals, the judge to be two men among you known for justice, (the forfeit) to be brought as an offering to the Ka'bah; or, for expiation, he shall feed poor persons, or the equivalent thereof in fasting, that he may taste the evil consequences of his deed. Allah forgiveth whatever (of this kind) may have happened in the past, but whoso relapseth, Allah will take retribution from him. Allah is Mighty, Able to Requite (the wrong). (5:95) || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ لَا تَقْتُلُوا۟ ٱلصَّيْدَ وَأَنتُمْ حُرُمٌۭ ۚ وَمَن قَتَلَهُۥ مِنكُم مُّتَعَمِّدًۭا فَجَزَآءٌۭ مِّثْلُ مَا قَتَلَ مِنَ ٱلنَّعَمِ يَحْكُمُ بِهِۦ ذَوَا عَدْلٍۢ مِّنكُمْ هَدْيًۢا بَٰلِغَ ٱلْكَعْبَةِ أَوْ كَفَّٰرَةٌۭ طَعَامُ مَسَٰكِينَ أَوْ عَدْلُ ذَٰلِكَ صِيَامًۭا لِّيَذُوقَ وَبَالَ أَمْرِهِۦ ۗ عَفَا ٱللَّهُ عَمَّا سَلَفَ ۚ وَمَنْ عَادَ فَيَنتَقِمُ ٱللَّهُ مِنْهُ ۗ وَٱللَّهُ عَزِيزٌۭ ذُو ٱنتِقَامٍ ﴿٩٥﴾ ای مؤمنانو! د احرام په حال کې ښکار مه وژنئ؛ او له تاسو څخه چا چې په قصد ښکار ووژه، نو (د هغه) بدله له څارویو د هغه په څېر ده چې وژلی یې دی، چې له تاسو دوه عادل سړي به یې پرېکړه کوي، دا به قرباني وي چې کعبې ته ورسول شي، یا یې کفاره مسکینانو ته خواړه ورکول وي، یا یې په بدل کې روژې نیول وي، تر څو د خپل کار (د سزا) خوند وڅکي؛ الله هغه بښلي چې مخکې تېر شوي، او څوک چې بیا داسې وکړي، نو الله به ترې انتقام واخلي؛ او الله ډېر غالب، انتقام اخیستونکی دی.
