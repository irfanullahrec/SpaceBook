p وَٱلَّذِينَ يُظَٰهِرُونَ مِن نِّسَآئِهِمْ ثُمَّ يَعُودُونَ لِمَا قَالُوا۟ فَتَحْرِيرُ رَقَبَةٍۢ مِّن قَبْلِ أَن يَتَمَآسَّا ۚ ذَٰلِكُمْ تُوعَظُونَ بِهِۦ ۚ وَٱللَّهُ بِمَا تَعْمَلُونَ خَبِيرٌۭ ﴿٣﴾ اور جو لوگ اپنی بیویوں سے ظہار کر بیٹھیں پھر جو انھوں نے کہا ہے اُس سے رجوع کرنا چاہیں تو (ان کے ذمہ) ایک غلام (یا کنیز) آزاد کرنا ہے اس سے پہلے کہ وہ ایک دوسرے کو ہاتھ لگائیں یہ وہ (کفارہ) ہے جس کی تمھیں نصیحت کی جاتی ہے اور جو کچھ تم کر رہے ہو اللہ اس سے خوب خبردار ہے۔ || وَٱلَّذِينَ يُظَٰهِرُونَ مِن نِّسَآئِهِمْ ثُمَّ يَعُودُونَ لِمَا قَالُوا۟ فَتَحْرِيرُ رَقَبَةٍۢ مِّن قَبْلِ أَن يَتَمَآسَّا ۚ ذَٰلِكُمْ تُوعَظُونَ بِهِۦ ۚ وَٱللَّهُ بِمَا تَعْمَلُونَ خَبِيرٌۭ ﴿٣﴾ Those who put away their wives (by saying they are as their mothers) and afterward would go back on that which they have said, (the penalty) in that case (is) the freeing of a slave before they touch one another. Unto this ye are exhorted; and Allah is Informed of what ye do. (58:3) || وَٱلَّذِينَ يُظَٰهِرُونَ مِن نِّسَآئِهِمْ ثُمَّ يَعُودُونَ لِمَا قَالُوا۟ فَتَحْرِيرُ رَقَبَةٍۢ مِّن قَبْلِ أَن يَتَمَآسَّا ۚ ذَٰلِكُمْ تُوعَظُونَ بِهِۦ ۚ وَٱللَّهُ بِمَا تَعْمَلُونَ خَبِيرٌۭ ﴿٣﴾ او هغه کسان چې له خپلو مېرمنو ظهار وکړي، بیا له هغه څه چې ویلي یې دي رجوع کول وغواړي، نو (د هغوی پر غاړه) د یوه غلام (یا وینځې) آزادول دي مخکې له دې چې یو بل ته لاس ور وړي؛ دا هغه (کفاره) ده چې تاسو ته یې نصیحت کېږي؛ او څه چې تاسو کوئ، الله ترې ښه خبر دی.
p فَمَن لَّمْ يَجِدْ فَصِيَامُ شَهْرَيْنِ مُتَتَابِعَيْنِ مِن قَبْلِ أَن يَتَمَآسَّا ۖ فَمَن لَّمْ يَسْتَطِعْ فَإِطْعَامُ سِتِّينَ مِسْكِينًۭا ۚ ذَٰلِكَ لِتُؤْمِنُوا۟ بِٱللَّهِ وَرَسُولِهِۦ ۚ وَتِلْكَ حُدُودُ ٱللَّهِ ۗ وَلِلْكَٰفِرِينَ عَذَابٌ أَلِيمٌ ﴿٤﴾ پھر جو (غلام) نہ پائے تو (اس پر) دو مہینے لگاتار روزے (رکھنا لازم) ہیں اس سے پہلے کہ وہ ایک دوسرے کو ہاتھ لگائیں پھر جو اس کی (بھی) استطاعت نہ رکھے تو (اس پر) ساٹھ مسکینوں کو کھانا کھلانا (لازم) ہے یہ اس لیے تاکہ تم اللہ پر اور اُس کے رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) پر ایمان رکھو اور یہ اللہ کی (مقرر کردہ) حدود ہیں اور کافروں کے لیے دردناک عذاب ہے۔ || فَمَن لَّمْ يَجِدْ فَصِيَامُ شَهْرَيْنِ مُتَتَابِعَيْنِ مِن قَبْلِ أَن يَتَمَآسَّا ۖ فَمَن لَّمْ يَسْتَطِعْ فَإِطْعَامُ سِتِّينَ مِسْكِينًۭا ۚ ذَٰلِكَ لِتُؤْمِنُوا۟ بِٱللَّهِ وَرَسُولِهِۦ ۚ وَتِلْكَ حُدُودُ ٱللَّهِ ۗ وَلِلْكَٰفِرِينَ عَذَابٌ أَلِيمٌ ﴿٤﴾ And he who findeth not (the wherewithal), let him fast for two successive months before they touch one another; and for him who is unable to do so (the penance is) the feeding of sixty needy ones. This, that ye may put trust in Allah and His messenger. Such are the limits (imposed by Allah); and for disbelievers is a painful doom. (58:4) || فَمَن لَّمْ يَجِدْ فَصِيَامُ شَهْرَيْنِ مُتَتَابِعَيْنِ مِن قَبْلِ أَن يَتَمَآسَّا ۖ فَمَن لَّمْ يَسْتَطِعْ فَإِطْعَامُ سِتِّينَ مِسْكِينًۭا ۚ ذَٰلِكَ لِتُؤْمِنُوا۟ بِٱللَّهِ وَرَسُولِهِۦ ۚ وَتِلْكَ حُدُودُ ٱللَّهِ ۗ وَلِلْكَٰفِرِينَ عَذَابٌ أَلِيمٌ ﴿٤﴾ بیا څوک چې (غلام) ونه مومي، نو (پر هغه) دوه میاشتې پرله‌پسې روژې (لازمې) دي مخکې له دې چې یو بل ته لاس ور وړي؛ بیا څوک چې د دې (هم) وس ونه لري، نو (پر هغه) شپېتو مسکینانو ته خواړه ورکول (لازم) دي؛ دا ځکه تر څو پر الله او د هغه پر رسول (صلی الله علیه وعلی آله واصحابه وسلم) ایمان ولرئ؛ او دا د الله (ټاکلې) پولې دي، او د کافرانو لپاره دردناک عذاب دی.
p إِنَّ ٱلَّذِينَ يُحَآدُّونَ ٱللَّهَ وَرَسُولَهُۥ كُبِتُوا۟ كَمَا كُبِتَ ٱلَّذِينَ مِن قَبْلِهِمْ ۚ وَقَدْ أَنزَلْنَآ ءَايَٰتٍۭ بَيِّنَٰتٍۢ ۚ وَلِلْكَٰفِرِينَ عَذَابٌۭ مُّهِينٌۭ ﴿٥﴾ بے شک وہ لوگ جو اللہ اور اُس کے رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کی مخالفت کرتے ہیں وہ (اسی طرح) ذلیل کیے جائیں گے جس طرح ان سے پہلے لوگ ذلیل کیے جا چکے ہیں اور یقیناً ہم نے واضح آیات نازل فرما دی ہیں اور کافروں کے لیے ذلت والا عذاب ہے۔ || إِنَّ ٱلَّذِينَ يُحَآدُّونَ ٱللَّهَ وَرَسُولَهُۥ كُبِتُوا۟ كَمَا كُبِتَ ٱلَّذِينَ مِن قَبْلِهِمْ ۚ وَقَدْ أَنزَلْنَآ ءَايَٰتٍۭ بَيِّنَٰتٍۢ ۚ وَلِلْكَٰفِرِينَ عَذَابٌۭ مُّهِينٌۭ ﴿٥﴾ Lo! those who oppose Allah and His messenger will be abased even as those before them were abased; and We have sent down clear tokens, and for disbelievers is a shameful doom (58:5) || إِنَّ ٱلَّذِينَ يُحَآدُّونَ ٱللَّهَ وَرَسُولَهُۥ كُبِتُوا۟ كَمَا كُبِتَ ٱلَّذِينَ مِن قَبْلِهِمْ ۚ وَقَدْ أَنزَلْنَآ ءَايَٰتٍۭ بَيِّنَٰتٍۢ ۚ وَلِلْكَٰفِرِينَ عَذَابٌۭ مُّهِينٌۭ ﴿٥﴾ بې‌شکه هغه کسان چې د الله او د هغه رسول (صلی الله علیه وعلی آله واصحابه وسلم) مخالفت کوي، (همداسې) به ذلیل کړای شي لکه څنګه چې له دوی مخکې خلک ذلیل کړای شوي؛ او یقیناً موږ روښانه آیتونه نازل کړي، او د کافرانو لپاره د ذلت عذاب دی.
p يَوْمَ يَبْعَثُهُمُ ٱللَّهُ جَمِيعًۭا فَيُنَبِّئُهُم بِمَا عَمِلُوٓا۟ ۚ أَحْصَىٰهُ ٱللَّهُ وَنَسُوهُ ۚ وَٱللَّهُ عَلَىٰ كُلِّ شَىْءٍۢ شَهِيدٌ ﴿٦﴾ جس دن اللہ اُن سب کو (دوبارہ زندہ کر کے) اٹھائے گا پھر وہ انھیں آگاہ فرمائے گا جو انھوں نے کیا اللہ نے وہ (سب) (محفوظ کر کے) رکھا ہے اور وہ اسے بھول گئے اور اللہ ہر چیز پر گواہ ہے۔ || يَوْمَ يَبْعَثُهُمُ ٱللَّهُ جَمِيعًۭا فَيُنَبِّئُهُم بِمَا عَمِلُوٓا۟ ۚ أَحْصَىٰهُ ٱللَّهُ وَنَسُوهُ ۚ وَٱللَّهُ عَلَىٰ كُلِّ شَىْءٍۢ شَهِيدٌ ﴿٦﴾ On the day when Allah will raise them all together and inform them of what they did. Allah hath kept account of it while they forgot it. And Allah is Witness over all things. (58:6) || يَوْمَ يَبْعَثُهُمُ ٱللَّهُ جَمِيعًۭا فَيُنَبِّئُهُم بِمَا عَمِلُوٓا۟ ۚ أَحْصَىٰهُ ٱللَّهُ وَنَسُوهُ ۚ وَٱللَّهُ عَلَىٰ كُلِّ شَىْءٍۢ شَهِيدٌ ﴿٦﴾ په کومه ورځ چې الله دوی ټول (بیا ژوندي کړي او) راپورته کړي، بیا یې خبروي په هغه څه چې دوی کړي؛ الله هغه (ټول) (خوندي) ساتلي او دوی هېر کړي؛ او الله پر هر شي شاهد دی.
p أَلَمْ تَرَ أَنَّ ٱللَّهَ يَعْلَمُ مَا فِى ٱلسَّمَٰوَٰتِ وَمَا فِى ٱلْأَرْضِ ۖ مَا يَكُونُ مِن نَّجْوَىٰ ثَلَٰثَةٍ إِلَّا هُوَ رَابِعُهُمْ وَلَا خَمْسَةٍ إِلَّا هُوَ سَادِسُهُمْ وَلَآ أَدْنَىٰ مِن ذَٰلِكَ وَلَآ أَكْثَرَ إِلَّا هُوَ مَعَهُمْ أَيْنَ مَا كَانُوا۟ ۖ ثُمَّ يُنَبِّئُهُم بِمَا عَمِلُوا۟ يَوْمَ ٱلْقِيَٰمَةِ ۚ إِنَّ ٱللَّهَ بِكُلِّ شَىْءٍ عَلِيمٌ ﴿٧﴾ (اے مخاطب!) کیا تو نے نہیں دیکھا کہ اللہ سب کچھ جانتا ہے جو کچھ آسمانوں میں ہے اور جو کچھ زمینوں میں ہے؟ کہیں بھی تین (آدمیوں) کی کوئی سرگوشی نہیں ہوتی مگر وہ (اللہ) ان کا چوتھا ہوتا ہے اور نہ ہی پانچ کی (سرگوشی ہوتی ہے) مگر وہ ان کا چھٹا ہوتا ہے اور نہ اس سے کم ہوتے ہیں اور نہ زیادہ مگر وہ ان کے ساتھ ہوتا ہے جہاں کہیں بھی وہ ہوتے ہیں پھر وہ قیامت کے دن انھیں آگاہ فرما دے گا جو کچھ انھوں نے کیا بے شک اللہ ہر شے کو خوب جاننے والا ہے۔ || أَلَمْ تَرَ أَنَّ ٱللَّهَ يَعْلَمُ مَا فِى ٱلسَّمَٰوَٰتِ وَمَا فِى ٱلْأَرْضِ ۖ مَا يَكُونُ مِن نَّجْوَىٰ ثَلَٰثَةٍ إِلَّا هُوَ رَابِعُهُمْ وَلَا خَمْسَةٍ إِلَّا هُوَ سَادِسُهُمْ وَلَآ أَدْنَىٰ مِن ذَٰلِكَ وَلَآ أَكْثَرَ إِلَّا هُوَ مَعَهُمْ أَيْنَ مَا كَانُوا۟ ۖ ثُمَّ يُنَبِّئُهُم بِمَا عَمِلُوا۟ يَوْمَ ٱلْقِيَٰمَةِ ۚ إِنَّ ٱللَّهَ بِكُلِّ شَىْءٍ عَلِيمٌ ﴿٧﴾ Hast thou not seen that Allah knoweth all that is in the heavens and all that is in the earth? There is no secret conference of three but He is their fourth, nor of five but He is their sixth, nor of less than that or more but He is with them wheresoever they may be; and afterward, on the Day of Resurrection, He will inform them of what they did. Lo! Allah is Knower of all things. (58:7) || أَلَمْ تَرَ أَنَّ ٱللَّهَ يَعْلَمُ مَا فِى ٱلسَّمَٰوَٰتِ وَمَا فِى ٱلْأَرْضِ ۖ مَا يَكُونُ مِن نَّجْوَىٰ ثَلَٰثَةٍ إِلَّا هُوَ رَابِعُهُمْ وَلَا خَمْسَةٍ إِلَّا هُوَ سَادِسُهُمْ وَلَآ أَدْنَىٰ مِن ذَٰلِكَ وَلَآ أَكْثَرَ إِلَّا هُوَ مَعَهُمْ أَيْنَ مَا كَانُوا۟ ۖ ثُمَّ يُنَبِّئُهُم بِمَا عَمِلُوا۟ يَوْمَ ٱلْقِيَٰمَةِ ۚ إِنَّ ٱللَّهَ بِكُلِّ شَىْءٍ عَلِيمٌ ﴿٧﴾ (ای مخاطبه!) ایا تا نه دي لیدلي چې الله په هر هغه څه پوهېږي چې په آسمانونو کې دي او په هر هغه څه چې په ځمکو کې دي؟ هېڅ ځای د درېیو (کسانو) پټه مشوره نه کېږي مګر هغه (الله) د هغوی څلورم وي، او نه د پنځو (پټه مشوره کېږي) مګر هغه د هغوی شپږم وي، او نه له دې لږ وي او نه زیات، مګر هغه له هغوی سره وي هر چېرې چې وي؛ بیا به یې د قیامت په ورځ خبر کړي په هغه څه چې کړي یې دي؛ بې‌شکه الله په هر شي ښه پوه دی.
