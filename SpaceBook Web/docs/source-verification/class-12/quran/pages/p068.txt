p ۞ يَٰٓأَيُّهَا ٱلرَّسُولُ لَا يَحْزُنكَ ٱلَّذِينَ يُسَٰرِعُونَ فِى ٱلْكُفْرِ مِنَ ٱلَّذِينَ قَالُوٓا۟ ءَامَنَّا بِأَفْوَٰهِهِمْ وَلَمْ تُؤْمِن قُلُوبُهُمْ ۛ وَمِنَ ٱلَّذِينَ هَادُوا۟ ۛ سَمَّٰعُونَ لِلْكَذِبِ سَمَّٰعُونَ لِقَوْمٍ ءَاخَرِينَ لَمْ يَأْتُوكَ ۖ يُحَرِّفُونَ ٱلْكَلِمَ مِنۢ بَعْدِ مَوَاضِعِهِۦ ۖ يَقُولُونَ إِنْ أُوتِيتُمْ هَٰذَا فَخُذُوهُ وَإِن لَّمْ تُؤْتَوْهُ فَٱحْذَرُوا۟ ۚ وَمَن يُرِدِ ٱللَّهُ فِتْنَتَهُۥ فَلَن تَمْلِكَ لَهُۥ مِنَ ٱللَّهِ شَيْـًٔا ۚ أُو۟لَٰٓئِكَ ٱلَّذِينَ لَمْ يُرِدِ ٱللَّهُ أَن يُطَهِّرَ قُلُوبَهُمْ ۚ لَهُمْ فِى ٱلدُّنْيَا خِزْىٌۭ ۖ وَلَهُمْ فِى ٱلْءَاخِرَةِ عَذَابٌ عَظِيمٌۭ ﴿٤١﴾ اے رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم)! آپ کو وہ لوگ غمگین نہ کریں جو کفر میں تیزی دکھاتے ہیں اُن میں سے جنھوں نے اپنے منہ سے کہا ہم ایمان لائے حالاں کہ ان کے دل ایمان نہیں لائے اور اُن لوگوں میں سے جو یہودی ہوئے، جھوٹی باتیں بنانے کے لیے وہ (آپ کی باتوں کو) خوب سنتے ہیں (حقیقت میں) وہ دوسروں کے لیے (جاسوسی کی خاطر) بہت سننے والے ہیں جو (ابھی تک) آپ کے پاس نہیں آئے جو کلام کو اُس کے مقام سے بدل دیتے ہیں وہ کہتے ہیں اگر تمھیں یہ (من پسند) حکم دیا جائے تو اسے قبول کرنا اور اگر تمھیں یہ (حکم) نہ دیا جائے تو بچ کر رہنا اور جسے اللہ گم راہ کرنا چاہے تو (اے مخاطب!) تو اس کو اللہ سے (بچانے کا) ہرگز کوئی اختیار نہیں رکھتا یہ وہ لوگ ہیں کہ (ان کی نافرمانی کی وجہ سے) اللہ نے ان کے دلوں کو پاک کرنے کا ارادہ نہیں کیا اُن کے لیے دنیا میں رسوائی ہے اور آخرت میں اُن کے لیے بڑا عذاب ہے۔ || ۞ يَٰٓأَيُّهَا ٱلرَّسُولُ لَا يَحْزُنكَ ٱلَّذِينَ يُسَٰرِعُونَ فِى ٱلْكُفْرِ مِنَ ٱلَّذِينَ قَالُوٓا۟ ءَامَنَّا بِأَفْوَٰهِهِمْ وَلَمْ تُؤْمِن قُلُوبُهُمْ ۛ وَمِنَ ٱلَّذِينَ هَادُوا۟ ۛ سَمَّٰعُونَ لِلْكَذِبِ سَمَّٰعُونَ لِقَوْمٍ ءَاخَرِينَ لَمْ يَأْتُوكَ ۖ يُحَرِّفُونَ ٱلْكَلِمَ مِنۢ بَعْدِ مَوَاضِعِهِۦ ۖ يَقُولُونَ إِنْ أُوتِيتُمْ هَٰذَا فَخُذُوهُ وَإِن لَّمْ تُؤْتَوْهُ فَٱحْذَرُوا۟ ۚ وَمَن يُرِدِ ٱللَّهُ فِتْنَتَهُۥ فَلَن تَمْلِكَ لَهُۥ مِنَ ٱللَّهِ شَيْـًٔا ۚ أُو۟لَٰٓئِكَ ٱلَّذِينَ لَمْ يُرِدِ ٱللَّهُ أَن يُطَهِّرَ قُلُوبَهُمْ ۚ لَهُمْ فِى ٱلدُّنْيَا خِزْىٌۭ ۖ وَلَهُمْ فِى ٱلْءَاخِرَةِ عَذَابٌ عَظِيمٌۭ ﴿٤١﴾ O Messenger! Let not them grieve thee who vie one with another in the race to disbelief, of such as say with their mouths: "We believe," but their hearts believe not, and of the Jews: listeners for the sake of falsehood, listeners on behalf of other folk who come not unto thee, changing words from their context and saying: If this be given unto you, receive it, but if this be not given unto you, then beware! He whom Allah doometh unto sin, thou (by thine efforts) wilt avail him naught against Allah. Those are they for whom the Will of Allah is that He cleanse not their hearts. Theirs in the world will be ignominy, and in the Hereafter an awful doom; (5:41) || ۞ يَٰٓأَيُّهَا ٱلرَّسُولُ لَا يَحْزُنكَ ٱلَّذِينَ يُسَٰرِعُونَ فِى ٱلْكُفْرِ مِنَ ٱلَّذِينَ قَالُوٓا۟ ءَامَنَّا بِأَفْوَٰهِهِمْ وَلَمْ تُؤْمِن قُلُوبُهُمْ ۛ وَمِنَ ٱلَّذِينَ هَادُوا۟ ۛ سَمَّٰعُونَ لِلْكَذِبِ سَمَّٰعُونَ لِقَوْمٍ ءَاخَرِينَ لَمْ يَأْتُوكَ ۖ يُحَرِّفُونَ ٱلْكَلِمَ مِنۢ بَعْدِ مَوَاضِعِهِۦ ۖ يَقُولُونَ إِنْ أُوتِيتُمْ هَٰذَا فَخُذُوهُ وَإِن لَّمْ تُؤْتَوْهُ فَٱحْذَرُوا۟ ۚ وَمَن يُرِدِ ٱللَّهُ فِتْنَتَهُۥ فَلَن تَمْلِكَ لَهُۥ مِنَ ٱللَّهِ شَيْـًٔا ۚ أُو۟لَٰٓئِكَ ٱلَّذِينَ لَمْ يُرِدِ ٱللَّهُ أَن يُطَهِّرَ قُلُوبَهُمْ ۚ لَهُمْ فِى ٱلدُّنْيَا خِزْىٌۭ ۖ وَلَهُمْ فِى ٱلْءَاخِرَةِ عَذَابٌ عَظِيمٌۭ ﴿٤١﴾ ای رسوله (صلی الله علیه وعلی آله واصحابه وسلم)! هغه کسان دې تاسو غمجن نه کړي چې په کفر کې بېړه کوي: له هغو څخه چې په خپلو خولو یې وویل «موږ ایمان راوړ» حال دا چې زړونو یې ایمان نه دی راوړی، او له هغو خلکو چې یهودان شول — هغوی د درواغو جوړولو لپاره (ستاسو خبرې) ښې اوري؛ (په حقیقت کې) هغوی د نورو لپاره (د جاسوسۍ په خاطر) ډېر اورېدونکي دي چې (تر اوسه) تاسو ته نه دي راغلي؛ هغوی کلام له خپل ځایه اړوي، وایي: «که تاسو ته دا (خوښ) حکم درکړل شو، ومنئ یې، او که دا (حکم) درنه کړل شو، ځان ترې وساتئ.» او څوک چې الله وغواړي ګمراه یې کړي، نو (ای مخاطبه!) ته یې له الله (د ژغورلو) هېڅ واک نه لرې. دا هغه خلک دي چې (د هغوی د نافرمانۍ له امله) الله د هغوی د زړونو د پاکولو اراده نه ده کړې؛ د هغوی لپاره په دنیا کې رسوایي او په آخرت کې لوی عذاب دی.
p سَمَّٰعُونَ لِلْكَذِبِ أَكَّٰلُونَ لِلسُّحْتِ ۚ فَإِن جَآءُوكَ فَٱحْكُم بَيْنَهُمْ أَوْ أَعْرِضْ عَنْهُمْ ۖ وَإِن تُعْرِضْ عَنْهُمْ فَلَن يَضُرُّوكَ شَيْـًۭٔا ۖ وَإِنْ حَكَمْتَ فَٱحْكُم بَيْنَهُم بِٱلْقِسْطِ ۚ إِنَّ ٱللَّهَ يُحِبُّ ٱلْمُقْسِطِينَ ﴿٤٢﴾ وہ (آپ کی باتوں کو) جھوٹی باتیں بنانے کے لیے خوب سنتے ہیں بہت زیادہ حرام کھانے والے ہیں تو اگر وہ آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کے پاس (کوئی مقدمہ لے کر) آئیں تو آپ (کو اختیار ہے کہ) چاہیں تو ان کے درمیان فیصلہ فرما دیں یا ان سے منہ پھیر لیں اور اگر آپ ان سے منہ پھیر لیں تو یہ ہرگز آپ کو کچھ بھی نقصان نہ پہنچا سکیں گے اور اگر آپ فیصلہ فرمائیں تو ان کے درمیان انصاف کے ساتھ فیصلہ فرما دیجیے بے شک اللہ انصاف کرنے والوں کو پسند فرماتا ہے۔ || سَمَّٰعُونَ لِلْكَذِبِ أَكَّٰلُونَ لِلسُّحْتِ ۚ فَإِن جَآءُوكَ فَٱحْكُم بَيْنَهُمْ أَوْ أَعْرِضْ عَنْهُمْ ۖ وَإِن تُعْرِضْ عَنْهُمْ فَلَن يَضُرُّوكَ شَيْـًۭٔا ۖ وَإِنْ حَكَمْتَ فَٱحْكُم بَيْنَهُم بِٱلْقِسْطِ ۚ إِنَّ ٱللَّهَ يُحِبُّ ٱلْمُقْسِطِينَ ﴿٤٢﴾ Listeners for the sake of falsehood! Greedy for illicit gain! If then they have recourse unto thee (Muhammad) judge between them or disclaim jurisdiction. If thou disclaimest jurisdiction, then they cannot harm thee at all. But if thou judgest, judge between them with equity. Lo! Allah loveth the equitable. (5:42) || سَمَّٰعُونَ لِلْكَذِبِ أَكَّٰلُونَ لِلسُّحْتِ ۚ فَإِن جَآءُوكَ فَٱحْكُم بَيْنَهُمْ أَوْ أَعْرِضْ عَنْهُمْ ۖ وَإِن تُعْرِضْ عَنْهُمْ فَلَن يَضُرُّوكَ شَيْـًۭٔا ۖ وَإِنْ حَكَمْتَ فَٱحْكُم بَيْنَهُم بِٱلْقِسْطِ ۚ إِنَّ ٱللَّهَ يُحِبُّ ٱلْمُقْسِطِينَ ﴿٤٢﴾ هغوی (ستاسو خبرې) د درواغو جوړولو لپاره ښې اوري، ډېر حرام خوړونکي دي؛ نو که تاسو (صلی الله علیه وعلی آله واصحابه وسلم) ته (کومه قضیه) راوړي، نو تاسو (واک لرئ چې) که وغواړئ د هغوی ترمنځ پرېکړه وکړئ یا ترې مخ واړوئ؛ او که ترې مخ واړوئ، نو هېڅکله به تاسو ته هېڅ زیان ونه رسوي؛ او که پرېکړه کوئ، نو د هغوی ترمنځ په انصاف پرېکړه وکړئ؛ بې‌شکه الله انصاف کوونکي خوښوي.
p وَكَيْفَ يُحَكِّمُونَكَ وَعِندَهُمُ ٱلتَّوْرَىٰةُ فِيهَا حُكْمُ ٱللَّهِ ثُمَّ يَتَوَلَّوْنَ مِنۢ بَعْدِ ذَٰلِكَ ۚ وَمَآ أُو۟لَٰٓئِكَ بِٱلْمُؤْمِنِينَ ﴿٤٣﴾ اور یہ کیسے آپ سے فیصلہ کرائیں گے حالاں کہ ان کے پاس تورات ہے جس میں اللہ کا حکم (موجود) ہے پھر یہ اُس کے بعد (بھی) منہ موڑتے ہیں اور یہ لوگ ایمان لانے والے نہیں ہیں۔ || وَكَيْفَ يُحَكِّمُونَكَ وَعِندَهُمُ ٱلتَّوْرَىٰةُ فِيهَا حُكْمُ ٱللَّهِ ثُمَّ يَتَوَلَّوْنَ مِنۢ بَعْدِ ذَٰلِكَ ۚ وَمَآ أُو۟لَٰٓئِكَ بِٱلْمُؤْمِنِينَ ﴿٤٣﴾ How come they unto thee for judgment when they have the Torah, wherein Allah hath delivered judgment (for them)? Yet even after that they turn away. Such (folk) are not believers. (5:43) || وَكَيْفَ يُحَكِّمُونَكَ وَعِندَهُمُ ٱلتَّوْرَىٰةُ فِيهَا حُكْمُ ٱللَّهِ ثُمَّ يَتَوَلَّوْنَ مِنۢ بَعْدِ ذَٰلِكَ ۚ وَمَآ أُو۟لَٰٓئِكَ بِٱلْمُؤْمِنِينَ ﴿٤٣﴾ او دا به څنګه له تاسو پرېکړه وغواړي، حال دا چې له هغوی سره تورات دی چې پکې د الله حکم (موجود) دی؟ بیا له هغه وروسته (هم) مخ اړوي؛ او دا خلک ایمان راوړونکي نه دي.
