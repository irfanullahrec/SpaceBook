p هُوَ ٱلَّذِىٓ أَنزَلَ ٱلسَّكِينَةَ فِى قُلُوبِ ٱلْمُؤْمِنِينَ لِيَزْدَادُوٓا۟ إِيمَٰنًۭا مَّعَ إِيمَٰنِهِمْ ۗ وَلِلَّهِ جُنُودُ ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ ۚ وَكَانَ ٱللَّهُ عَلِيمًا حَكِيمًۭا ﴿٤﴾ وہی تو ہے جس نے مومنوں کے دلوں میں سکون (و اطمینان) نازل فرمایا تاکہ وہ اپنے ایمان کے ساتھ ایمان میں زیادہ ہو جائیں اور آسمانوں اور زمینوں کے لشکر اللہ کے لیے ہیں اور اللہ خوب جاننے والا بڑی حکمت والا ہے۔ || هُوَ ٱلَّذِىٓ أَنزَلَ ٱلسَّكِينَةَ فِى قُلُوبِ ٱلْمُؤْمِنِينَ لِيَزْدَادُوٓا۟ إِيمَٰنًۭا مَّعَ إِيمَٰنِهِمْ ۗ وَلِلَّهِ جُنُودُ ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ ۚ وَكَانَ ٱللَّهُ عَلِيمًا حَكِيمًۭا ﴿٤﴾ He it is Who sent down peace of reassurance into the hearts of the believers that they might add faith unto their faith. Allah's are the hosts of the heavens and the earth, and Allah is ever Knower, Wise - (48:4) || هُوَ ٱلَّذِىٓ أَنزَلَ ٱلسَّكِينَةَ فِى قُلُوبِ ٱلْمُؤْمِنِينَ لِيَزْدَادُوٓا۟ إِيمَٰنًۭا مَّعَ إِيمَٰنِهِمْ ۗ وَلِلَّهِ جُنُودُ ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ ۚ وَكَانَ ٱللَّهُ عَلِيمًا حَكِيمًۭا ﴿٤﴾ همغه دی چې د مؤمنانو په زړونو کې یې سکون (او ډاډ) نازل کړ تر څو له خپل ایمان سره په ایمان کې زیات شي؛ او د آسمانونو او ځمکو لښکرې د الله دي؛ او الله ښه پوه، ډېر حکمتناک دی.
p لِّيُدْخِلَ ٱلْمُؤْمِنِينَ وَٱلْمُؤْمِنَٰتِ جَنَّٰتٍۢ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَٰرُ خَٰلِدِينَ فِيهَا وَيُكَفِّرَ عَنْهُمْ سَيِّـَٔاتِهِمْ ۚ وَكَانَ ذَٰلِكَ عِندَ ٱللَّهِ فَوْزًا عَظِيمًۭا ﴿٥﴾ تاکہ وہ مومن مردوں اور مومن عورتوں کو ایسی جنتوں میں داخل فرمائے جن کے نیچے نہریں بہتی ہیں وہ ان میں ہمیشہ رہنے والے ہیں اور وہ (اللہ) ان سے ان کی برائیاں دور فرما دے اور یہ اللہ کے نزدیک بہت بڑی کامیابی ہے۔ || لِّيُدْخِلَ ٱلْمُؤْمِنِينَ وَٱلْمُؤْمِنَٰتِ جَنَّٰتٍۢ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَٰرُ خَٰلِدِينَ فِيهَا وَيُكَفِّرَ عَنْهُمْ سَيِّـَٔاتِهِمْ ۚ وَكَانَ ذَٰلِكَ عِندَ ٱللَّهِ فَوْزًا عَظِيمًۭا ﴿٥﴾ That He may bring the believing men and the believing women into Gardens underneath which rivers flow, wherein they will abide, and may remit from them their evil deeds - That, in the sight of Allah, is the supreme triumph - (48:5) || لِّيُدْخِلَ ٱلْمُؤْمِنِينَ وَٱلْمُؤْمِنَٰتِ جَنَّٰتٍۢ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَٰرُ خَٰلِدِينَ فِيهَا وَيُكَفِّرَ عَنْهُمْ سَيِّـَٔاتِهِمْ ۚ وَكَانَ ذَٰلِكَ عِندَ ٱللَّهِ فَوْزًا عَظِيمًۭا ﴿٥﴾ تر څو مؤمن نارینه او مؤمنې ښځې داسې جنتونو ته ننباسي چې لاندې یې ویالې بهېږي، هغوی به پکې تل پاتې وي، او هغه (الله) له هغوی بدۍ لرې کړي؛ او دا د الله په وړاندې ډېره لویه بریا ده.
p وَيُعَذِّبَ ٱلْمُنَٰفِقِينَ وَٱلْمُنَٰفِقَٰتِ وَٱلْمُشْرِكِينَ وَٱلْمُشْرِكَٰتِ ٱلظَّآنِّينَ بِٱللَّهِ ظَنَّ ٱلسَّوْءِ ۚ عَلَيْهِمْ دَآئِرَةُ ٱلسَّوْءِ ۖ وَغَضِبَ ٱللَّهُ عَلَيْهِمْ وَلَعَنَهُمْ وَأَعَدَّ لَهُمْ جَهَنَّمَ ۖ وَسَآءَتْ مَصِيرًۭا ﴿٦﴾ اور (تاکہ) اللہ عذاب دے منافق مردوں اور منافق عورتوں کو اور مشرک مردوں اور مشرک عورتوں کو جو اللہ کے بارے میں برا گمان کرنے والے ہیں (حقیقت میں) انھی پر بری گردش ہے اور اللہ اُن پر غضب ناک ہوا اور اُس نے ان پر لعنت کر دی اور اس نے ان کے لیے جہنم تیار کر رکھی ہے اور وہ بہت برا ٹھکانا ہے۔ || وَيُعَذِّبَ ٱلْمُنَٰفِقِينَ وَٱلْمُنَٰفِقَٰتِ وَٱلْمُشْرِكِينَ وَٱلْمُشْرِكَٰتِ ٱلظَّآنِّينَ بِٱللَّهِ ظَنَّ ٱلسَّوْءِ ۚ عَلَيْهِمْ دَآئِرَةُ ٱلسَّوْءِ ۖ وَغَضِبَ ٱللَّهُ عَلَيْهِمْ وَلَعَنَهُمْ وَأَعَدَّ لَهُمْ جَهَنَّمَ ۖ وَسَآءَتْ مَصِيرًۭا ﴿٦﴾ And may punish the hypocritical men and the hypocritical women, and the idolatrous men and the idolatrous women, who think an evil thought concerning Allah. For them is the evil turn of fortune, and Allah is wroth against them and hath cursed them, and hath made ready for them hell, a hapless journey's end. (48:6) || وَيُعَذِّبَ ٱلْمُنَٰفِقِينَ وَٱلْمُنَٰفِقَٰتِ وَٱلْمُشْرِكِينَ وَٱلْمُشْرِكَٰتِ ٱلظَّآنِّينَ بِٱللَّهِ ظَنَّ ٱلسَّوْءِ ۚ عَلَيْهِمْ دَآئِرَةُ ٱلسَّوْءِ ۖ وَغَضِبَ ٱللَّهُ عَلَيْهِمْ وَلَعَنَهُمْ وَأَعَدَّ لَهُمْ جَهَنَّمَ ۖ وَسَآءَتْ مَصِيرًۭا ﴿٦﴾ او (تر څو) الله منافقو نارینه وو او منافقو ښځو او مشرکو نارینه وو او مشرکو ښځو ته عذاب ورکړي چې د الله په اړه بد ګومان کوي؛ (په حقیقت کې) بده ګرځیدنه پر همدوی ده؛ او الله پرې غضبناک شو، لعنت یې پرې وکړ، او د هغوی لپاره یې دوزخ تیار کړی؛ او هغه ډېر بد ځای دی.
p وَلِلَّهِ جُنُودُ ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ ۚ وَكَانَ ٱللَّهُ عَزِيزًا حَكِيمًا ﴿٧﴾ اور آسمانوں اور زمینوں کے لشکر اللہ کے لیے ہیں اور اللہ بہت غالب بڑی حکمت والا ہے۔ || وَلِلَّهِ جُنُودُ ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ ۚ وَكَانَ ٱللَّهُ عَزِيزًا حَكِيمًا ﴿٧﴾ Allah's are the hosts of the heavens and the earth, and Allah is ever Mighty, Wise. (48:7) || وَلِلَّهِ جُنُودُ ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ ۚ وَكَانَ ٱللَّهُ عَزِيزًا حَكِيمًا ﴿٧﴾ او د آسمانونو او ځمکو لښکرې د الله دي؛ او الله ډېر غالب، ډېر حکمتناک دی.
p إِنَّآ أَرْسَلْنَٰكَ شَٰهِدًۭا وَمُبَشِّرًۭا وَنَذِيرًۭا ﴿٨﴾ بے شک ہم نے آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کو گواہی دینے اور خوش خبری سنانے اور ڈرانے والا (رسول) بنا کر بھیجا ہے۔ || إِنَّآ أَرْسَلْنَٰكَ شَٰهِدًۭا وَمُبَشِّرًۭا وَنَذِيرًۭا ﴿٨﴾ Lo! We have sent thee (O Muhammad) as a witness and a bearer of good tidings and a warner, (48:8) || إِنَّآ أَرْسَلْنَٰكَ شَٰهِدًۭا وَمُبَشِّرًۭا وَنَذِيرًۭا ﴿٨﴾ بې‌شکه موږ تاسو (صلی الله علیه وعلی آله واصحابه وسلم) شاهدي ورکوونکی، زېری ورکوونکی او وېروونکی (رسول) لېږلی یاست.
p لِّتُؤْمِنُوا۟ بِٱللَّهِ وَرَسُولِهِۦ وَتُعَزِّرُوهُ وَتُوَقِّرُوهُ وَتُسَبِّحُوهُ بُكْرَةًۭ وَأَصِيلًا ﴿٩﴾ تاکہ (اے لوگو!) تم اللہ اور اُس کے رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) پر ایمان لاؤ اور ان (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کی مدد کرو اور ان (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کی تعظیم کرو اور صبح و شام اس (اللہ) کی تسبیح کرو۔ || لِّتُؤْمِنُوا۟ بِٱللَّهِ وَرَسُولِهِۦ وَتُعَزِّرُوهُ وَتُوَقِّرُوهُ وَتُسَبِّحُوهُ بُكْرَةًۭ وَأَصِيلًا ﴿٩﴾ That ye (mankind) may believe in Allah and His messenger, and may honour Him, and may revere Him, and may glorify Him at early dawn and at the close of day. (48:9) || لِّتُؤْمِنُوا۟ بِٱللَّهِ وَرَسُولِهِۦ وَتُعَزِّرُوهُ وَتُوَقِّرُوهُ وَتُسَبِّحُوهُ بُكْرَةًۭ وَأَصِيلًا ﴿٩﴾ تر څو (ای خلکو!) تاسو پر الله او د هغه پر رسول (صلی الله علیه وعلی آله واصحابه وسلم) ایمان راوړئ، د هغه (صلی الله علیه وعلی آله واصحابه وسلم) مرسته وکړئ، د هغه (صلی الله علیه وعلی آله واصحابه وسلم) درناوی وکړئ، او سهار او ماښام د هغه (الله) تسبیح ووایئ.
p إِنَّ ٱلَّذِينَ يُبَايِعُونَكَ إِنَّمَا يُبَايِعُونَ ٱللَّهَ يَدُ ٱللَّهِ فَوْقَ أَيْدِيهِمْ ۚ فَمَن نَّكَثَ فَإِنَّمَا يَنكُثُ عَلَىٰ نَفْسِهِۦ ۖ وَمَنْ أَوْفَىٰ بِمَا عَٰهَدَ عَلَيْهُ ٱللَّهَ فَسَيُؤْتِيهِ أَجْرًا عَظِيمًۭا ﴿١٠﴾ بے شک جو لوگ آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) سے بیعت کرتے ہیں وہ یقیناً اللہ ہی سے بیعت کرتے ہیں اُن کے ہاتھوں پر اللہ کا ہاتھ ہے پھر جو کوئی (اس) عہد کو توڑے گا تو اس کا وبال اسی پر ہوگا اور جس نے وہ بات پوری کی، جس پر اُس نے اللہ سے عہد کیا تھا تو (اللہ) عنقریب اُسے بہت بڑا اجر عطا فرمائے گا۔ || إِنَّ ٱلَّذِينَ يُبَايِعُونَكَ إِنَّمَا يُبَايِعُونَ ٱللَّهَ يَدُ ٱللَّهِ فَوْقَ أَيْدِيهِمْ ۚ فَمَن نَّكَثَ فَإِنَّمَا يَنكُثُ عَلَىٰ نَفْسِهِۦ ۖ وَمَنْ أَوْفَىٰ بِمَا عَٰهَدَ عَلَيْهُ ٱللَّهَ فَسَيُؤْتِيهِ أَجْرًا عَظِيمًۭا ﴿١٠﴾ Lo! those who swear allegiance unto thee (Muhammad), swear allegiance only unto Allah. The Hand of Allah is above their hands. So whosoever breaketh his oath, breaketh it only to his soul's hurt; while whosoever keepeth his covenant with Allah, on him will He bestow immense reward. (48:10) || إِنَّ ٱلَّذِينَ يُبَايِعُونَكَ إِنَّمَا يُبَايِعُونَ ٱللَّهَ يَدُ ٱللَّهِ فَوْقَ أَيْدِيهِمْ ۚ فَمَن نَّكَثَ فَإِنَّمَا يَنكُثُ عَلَىٰ نَفْسِهِۦ ۖ وَمَنْ أَوْفَىٰ بِمَا عَٰهَدَ عَلَيْهُ ٱللَّهَ فَسَيُؤْتِيهِ أَجْرًا عَظِيمًۭا ﴿١٠﴾ بې‌شکه هغه کسان چې له تاسو (صلی الله علیه وعلی آله واصحابه وسلم) سره بیعت کوي، هغوی یقیناً له الله سره بیعت کوي؛ د هغوی پر لاسونو د الله لاس دی؛ بیا څوک چې (دا) ژمنه ماته کړي، نو د هغې وبال پر همغه دی؛ او چا چې هغه خبره پوره کړه چې له الله سره یې پرې ژمنه کړې وه، نو (الله) به ژر هغه ته ډېر لوی اجر ورکړي.
