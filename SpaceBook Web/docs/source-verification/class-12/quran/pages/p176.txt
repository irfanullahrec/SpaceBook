p وَلَقَدْ أَرْسَلْنَا نُوحًۭا وَإِبْرَٰهِيمَ وَجَعَلْنَا فِى ذُرِّيَّتِهِمَا ٱلنُّبُوَّةَ وَٱلْكِتَٰبَ ۖ فَمِنْهُم مُّهْتَدٍۢ ۖ وَكَثِيرٌۭ مِّنْهُمْ فَٰسِقُونَ ﴿٢٦﴾ اور یقیناً ہم نے نوح (علیہ السلام) اور ابراہیم (علیہ السلام) کو (رسول بنا کر) بھیجا اور ہم نے ان دونوں کی اولاد میں نبوت اور کتاب کا سلسلہ جاری فرمایا پھر ان میں سے کچھ تو ہدایت یافتہ ہوئے اور ان میں سے اکثر فاسق رہے۔ || وَلَقَدْ أَرْسَلْنَا نُوحًۭا وَإِبْرَٰهِيمَ وَجَعَلْنَا فِى ذُرِّيَّتِهِمَا ٱلنُّبُوَّةَ وَٱلْكِتَٰبَ ۖ فَمِنْهُم مُّهْتَدٍۢ ۖ وَكَثِيرٌۭ مِّنْهُمْ فَٰسِقُونَ ﴿٢٦﴾ And We verily sent Noah and Abraham and placed the prophethood and the scripture among their seed, and among them there is he who goeth right, but many of them are evil-livers. (57:26) || وَلَقَدْ أَرْسَلْنَا نُوحًۭا وَإِبْرَٰهِيمَ وَجَعَلْنَا فِى ذُرِّيَّتِهِمَا ٱلنُّبُوَّةَ وَٱلْكِتَٰبَ ۖ فَمِنْهُم مُّهْتَدٍۢ ۖ وَكَثِيرٌۭ مِّنْهُمْ فَٰسِقُونَ ﴿٢٦﴾ او یقیناً موږ نوح (علیه السلام) او ابراهیم (علیه السلام) (رسولان کړل او) ولېږل، او د دواړو په اولاد کې مو د نبوت او کتاب لړۍ جاري کړه؛ بیا له هغوی ځینې هدایت موندونکي شول او ډېری یې فاسقان پاتې شول.
p ثُمَّ قَفَّيْنَا عَلَىٰٓ ءَاثَٰرِهِم بِرُسُلِنَا وَقَفَّيْنَا بِعِيسَى ٱبْنِ مَرْيَمَ وَءَاتَيْنَٰهُ ٱلْإِنجِيلَ وَجَعَلْنَا فِى قُلُوبِ ٱلَّذِينَ ٱتَّبَعُوهُ رَأْفَةًۭ وَرَحْمَةًۭ وَرَهْبَانِيَّةً ٱبْتَدَعُوهَا مَا كَتَبْنَٰهَا عَلَيْهِمْ إِلَّا ٱبْتِغَآءَ رِضْوَٰنِ ٱللَّهِ فَمَا رَعَوْهَا حَقَّ رِعَايَتِهَا ۖ فَـَٔاتَيْنَا ٱلَّذِينَ ءَامَنُوا۟ مِنْهُمْ أَجْرَهُمْ ۖ وَكَثِيرٌۭ مِّنْهُمْ فَٰسِقُونَ ﴿٢٧﴾ پھر ہم نے ان کے بعد ان ہی کے نقوشِ قدم پر اپنے رسولوں کو بھیجا اور ان کے بعد عیسیٰ ابن مریم (علیہ السلام) کو بھیجا اور ہم نے انھیں انجیل عطا فرمائی اور جنھوں نے ان کی پیروی کی ہم نے ان کے دلوں میں نرمی اور رحمت پیدا کر دی اور رہبانیت تو انھوں نے خود ایجاد کر لی تھی ہم نے اسے ان پر فرض نہیں کیا تھا مگر اللہ کی رضا جوئی (کے لیے انھوں نے خود سے یہ ایجاد کی) پھر وہ اسے نباہ نہ سکے جیسے اُس کے نباہنے کا حق تھا تو جو ان میں سے ایمان لائے ہم نے ان لوگوں کو ان کا اجر دیا اور ان میں سے اکثر فاسق رہے۔ || ثُمَّ قَفَّيْنَا عَلَىٰٓ ءَاثَٰرِهِم بِرُسُلِنَا وَقَفَّيْنَا بِعِيسَى ٱبْنِ مَرْيَمَ وَءَاتَيْنَٰهُ ٱلْإِنجِيلَ وَجَعَلْنَا فِى قُلُوبِ ٱلَّذِينَ ٱتَّبَعُوهُ رَأْفَةًۭ وَرَحْمَةًۭ وَرَهْبَانِيَّةً ٱبْتَدَعُوهَا مَا كَتَبْنَٰهَا عَلَيْهِمْ إِلَّا ٱبْتِغَآءَ رِضْوَٰنِ ٱللَّهِ فَمَا رَعَوْهَا حَقَّ رِعَايَتِهَا ۖ فَـَٔاتَيْنَا ٱلَّذِينَ ءَامَنُوا۟ مِنْهُمْ أَجْرَهُمْ ۖ وَكَثِيرٌۭ مِّنْهُمْ فَٰسِقُونَ ﴿٢٧﴾ Then We caused Our messengers to follow in their footsteps; and We caused Jesus, son of Mary, to follow, and gave him the Gospel, and placed compassion and mercy in the hearts of those who followed him. But monasticism they invented - We ordained it not for them - only seeking Allah's pleasure, and they observed it not with right observance. So We give those of them who believe their reward, but many of them are evil-livers. (57:27) || ثُمَّ قَفَّيْنَا عَلَىٰٓ ءَاثَٰرِهِم بِرُسُلِنَا وَقَفَّيْنَا بِعِيسَى ٱبْنِ مَرْيَمَ وَءَاتَيْنَٰهُ ٱلْإِنجِيلَ وَجَعَلْنَا فِى قُلُوبِ ٱلَّذِينَ ٱتَّبَعُوهُ رَأْفَةًۭ وَرَحْمَةًۭ وَرَهْبَانِيَّةً ٱبْتَدَعُوهَا مَا كَتَبْنَٰهَا عَلَيْهِمْ إِلَّا ٱبْتِغَآءَ رِضْوَٰنِ ٱللَّهِ فَمَا رَعَوْهَا حَقَّ رِعَايَتِهَا ۖ فَـَٔاتَيْنَا ٱلَّذِينَ ءَامَنُوا۟ مِنْهُمْ أَجْرَهُمْ ۖ وَكَثِيرٌۭ مِّنْهُمْ فَٰسِقُونَ ﴿٢٧﴾ بیا موږ له هغوی وروسته د هماغوی په پلونو خپل رسولان ولېږل، او له هغوی وروسته مو عیسی ابن مریم (علیه السلام) ولېږه او انجیل مو ورکړ؛ او چا چې د هغه پیروي وکړه، د هغوی په زړونو کې مو نرمي او رحمت پیدا کړ؛ او رهبانیت خو هغوی پخپله ایجاد کړی و، موږ پرې فرض کړی نه و، مګر د الله د رضا د غوښتلو (لپاره یې پخپله ایجاد کړ)، بیا یې هغه داسې پالنه ونه شوه کړی لکه څنګه چې یې د پالنې حق و؛ نو له هغوی چې څوک ایمان لرونکي وو، هغوی ته مو د هغوی اجر ورکړ، او له هغوی ډېری فاسقان پاتې شول.
p يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ ٱتَّقُوا۟ ٱللَّهَ وَءَامِنُوا۟ بِرَسُولِهِۦ يُؤْتِكُمْ كِفْلَيْنِ مِن رَّحْمَتِهِۦ وَيَجْعَل لَّكُمْ نُورًۭا تَمْشُونَ بِهِۦ وَيَغْفِرْ لَكُمْ ۚ وَٱللَّهُ غَفُورٌۭ رَّحِيمٌۭ ﴿٢٨﴾ اے ایمان والو! اللہ (کی نافرمانی) سے ڈرو اور اُس کے رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) پر ایمان لاؤ تو وہ تمھیں اپنی رحمت میں سے (ثواب کے) دو حصے عطا فرمائے گا اور بنا دے گا تمھارے لیے ایسا نور جس میں تم چلو گے اور تمھیں بخش دے گا اور اللہ بڑا ہی بخشنے والا نہایت رحم فرمانے والا ہے۔ || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ ٱتَّقُوا۟ ٱللَّهَ وَءَامِنُوا۟ بِرَسُولِهِۦ يُؤْتِكُمْ كِفْلَيْنِ مِن رَّحْمَتِهِۦ وَيَجْعَل لَّكُمْ نُورًۭا تَمْشُونَ بِهِۦ وَيَغْفِرْ لَكُمْ ۚ وَٱللَّهُ غَفُورٌۭ رَّحِيمٌۭ ﴿٢٨﴾ O ye who believe! Be mindful of your duty to Allah and put faith in His messenger. He will give you twofold of His mercy and will appoint for you a light wherein ye shall walk, and will forgive you. Allah is Forgiving, Merciful; (57:28) || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ ٱتَّقُوا۟ ٱللَّهَ وَءَامِنُوا۟ بِرَسُولِهِۦ يُؤْتِكُمْ كِفْلَيْنِ مِن رَّحْمَتِهِۦ وَيَجْعَل لَّكُمْ نُورًۭا تَمْشُونَ بِهِۦ وَيَغْفِرْ لَكُمْ ۚ وَٱللَّهُ غَفُورٌۭ رَّحِيمٌۭ ﴿٢٨﴾ ای مؤمنانو! له الله (د نافرمانۍ) ووېرېږئ او د هغه پر رسول (صلی الله علیه وعلی آله واصحابه وسلم) ایمان راوړئ، نو هغه به تاسو ته له خپل رحمت څخه (د ثواب) دوه برخې درکړي، او ستاسو لپاره به داسې نور جوړ کړي چې په هغه کې به ګرځئ، او تاسو به وبښي؛ او الله ډېر بښونکی، بې‌حده مهربان دی.
p لِّئَلَّا يَعْلَمَ أَهْلُ ٱلْكِتَٰبِ أَلَّا يَقْدِرُونَ عَلَىٰ شَىْءٍۢ مِّن فَضْلِ ٱللَّهِ ۙ وَأَنَّ ٱلْفَضْلَ بِيَدِ ٱللَّهِ يُؤْتِيهِ مَن يَشَآءُ ۚ وَٱللَّهُ ذُو ٱلْفَضْلِ ٱلْعَظِيمِ ﴿٢٩﴾ تاکہ اہلِ کتاب (جو آپ صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم پر ایمان نہیں لائے) جان لیں کہ انھیں اللہ کے فضل میں سے کسی چیز پر کوئی اختیار نہیں اور بے شک فضل تو اللہ ہی کے ہاتھ میں ہے وہ اسے جسے چاہتا ہے عطا فرماتا ہے اور اللہ بہت بڑے فضل والا ہے۔ || لِّئَلَّا يَعْلَمَ أَهْلُ ٱلْكِتَٰبِ أَلَّا يَقْدِرُونَ عَلَىٰ شَىْءٍۢ مِّن فَضْلِ ٱللَّهِ ۙ وَأَنَّ ٱلْفَضْلَ بِيَدِ ٱللَّهِ يُؤْتِيهِ مَن يَشَآءُ ۚ وَٱللَّهُ ذُو ٱلْفَضْلِ ٱلْعَظِيمِ ﴿٢٩﴾ That the People of the Scripture may know that they control naught of the bounty of Allah, but that the bounty is in Allah's hand to give to whom He will. And Allah is of Infinite Bounty. (57:29) || لِّئَلَّا يَعْلَمَ أَهْلُ ٱلْكِتَٰبِ أَلَّا يَقْدِرُونَ عَلَىٰ شَىْءٍۢ مِّن فَضْلِ ٱللَّهِ ۙ وَأَنَّ ٱلْفَضْلَ بِيَدِ ٱللَّهِ يُؤْتِيهِ مَن يَشَآءُ ۚ وَٱللَّهُ ذُو ٱلْفَضْلِ ٱلْعَظِيمِ ﴿٢٩﴾ تر څو اهل کتاب (چې پر هغه صلی الله علیه وعلی آله واصحابه وسلم یې ایمان نه دی راوړی) پوه شي چې د الله له فضل څخه پر هېڅ شي اختیار نه لري، او بې‌شکه فضل خو یوازې د الله په لاس کې دی، چا ته چې وغواړي ورکوي یې؛ او الله د ډېر لوی فضل خاوند دی.
## علم و عمل کی باتیں || Matters of knowledge and action || د علم او عمل خبرې
- قرآنِ حکیم کی واضح آیات ہمیں کفر، شرک، الحاد، مادہ پرستی، ظلم اور تمام قسم کی گمراہیوں کے اندھیروں سے نکال کر توحید و رسالت کی روشنی میں لاتی ہیں اور اللہ تعالیٰ کی رحمت، ہدایت اور ایمان کے حصول کا بھی ذریعہ بنتی ہیں۔ || The clear verses of the Qur'an-e-Hakim bring us out of the darkness of disbelief, polytheism, atheism, materialism, oppression and every kind of misguidance into the light of Tawhid and Prophethood, and they are also a means of attaining the mercy of Allah Almighty, guidance and faith. || د قرآن حکیم روښانه آیتونه موږ د کفر، شرک، الحاد، مادیت، ظلم او هر ډول ګمراهیو له تیارو راباسي او د توحید او رسالت رڼا ته مو راولي، او د الله تعالی د رحمت، هدایت او ایمان د ترلاسه کولو وسیله هم کېږي.
