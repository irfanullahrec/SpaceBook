p أَلَمْ تَرَ إِلَى ٱلَّذِينَ قِيلَ لَهُمْ كُفُّوٓا۟ أَيْدِيَكُمْ وَأَقِيمُوا۟ ٱلصَّلَوٰةَ وَءَاتُوا۟ ٱلزَّكَوٰةَ فَلَمَّا كُتِبَ عَلَيْهِمُ ٱلْقِتَالُ إِذَا فَرِيقٌۭ مِّنْهُمْ يَخْشَوْنَ ٱلنَّاسَ كَخَشْيَةِ ٱللَّهِ أَوْ أَشَدَّ خَشْيَةًۭ ۚ وَقَالُوا۟ رَبَّنَا لِمَ كَتَبْتَ عَلَيْنَا ٱلْقِتَالَ لَوْلَآ أَخَّرْتَنَآ إِلَىٰٓ أَجَلٍۢ قَرِيبٍۢ ۗ قُلْ مَتَٰعُ ٱلدُّنْيَا قَلِيلٌۭ وَٱلْءَاخِرَةُ خَيْرٌۭ لِّمَنِ ٱتَّقَىٰ وَلَا تُظْلَمُونَ فَتِيلًا ﴿٧٧﴾ کیا آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) نے ان لوگوں (کے حال) کو نہیں دیکھا جن سے کہا گیا تھا کہ اپنے ہاتھ روکے رکھو (قتال میں جلد بازی نہ کرو) اور نماز قائم کرو اور زکوٰۃ دو پھر جب اُن پر قتال کرنا لکھ دیا گیا اُس وقت اُن میں سے ایک گروہ لوگوں سے ڈرنے لگا جیسے اللہ سے ڈرا جاتا ہے یا اس سے بھی بڑھ کر ڈرنے (لگا) اور انھوں نے کہا اے ہمارے رب! تو نے ہم پر قتال کرنا کیوں فرض کر دیا ہمیں ایک قریبی مدت تک مہلت کیوں نہ دی آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) فرما دیجیے دنیا کا سامان تو بہت ہی تھوڑا ہے اور آخرت اُس کے لیے بہت بہتر ہے جس نے پرہیزگاری اختیار کی اور تم پر ذرہ برابر بھی ظلم نہیں کیا جائے گا۔ || أَلَمْ تَرَ إِلَى ٱلَّذِينَ قِيلَ لَهُمْ كُفُّوٓا۟ أَيْدِيَكُمْ وَأَقِيمُوا۟ ٱلصَّلَوٰةَ وَءَاتُوا۟ ٱلزَّكَوٰةَ فَلَمَّا كُتِبَ عَلَيْهِمُ ٱلْقِتَالُ إِذَا فَرِيقٌۭ مِّنْهُمْ يَخْشَوْنَ ٱلنَّاسَ كَخَشْيَةِ ٱللَّهِ أَوْ أَشَدَّ خَشْيَةًۭ ۚ وَقَالُوا۟ رَبَّنَا لِمَ كَتَبْتَ عَلَيْنَا ٱلْقِتَالَ لَوْلَآ أَخَّرْتَنَآ إِلَىٰٓ أَجَلٍۢ قَرِيبٍۢ ۗ قُلْ مَتَٰعُ ٱلدُّنْيَا قَلِيلٌۭ وَٱلْءَاخِرَةُ خَيْرٌۭ لِّمَنِ ٱتَّقَىٰ وَلَا تُظْلَمُونَ فَتِيلًا ﴿٧٧﴾ Hast thou not seen those unto whom it was said: Withhold your hands, establish worship and pay the poordue, but when fighting was prescribed for them behold! a party of them fear mankind even as their fear of Allah or with greater fear, and say: Our Lord! Why hast Thou ordained fighting for us? If only Thou wouldst give us respite yet a while! Say (unto them, O Muhammad): The comfort of this world is scant; the Hereafter will be better for him who wardeth off (evil); and ye will not be wronged the down upon a date-stone. (4:77) || أَلَمْ تَرَ إِلَى ٱلَّذِينَ قِيلَ لَهُمْ كُفُّوٓا۟ أَيْدِيَكُمْ وَأَقِيمُوا۟ ٱلصَّلَوٰةَ وَءَاتُوا۟ ٱلزَّكَوٰةَ فَلَمَّا كُتِبَ عَلَيْهِمُ ٱلْقِتَالُ إِذَا فَرِيقٌۭ مِّنْهُمْ يَخْشَوْنَ ٱلنَّاسَ كَخَشْيَةِ ٱللَّهِ أَوْ أَشَدَّ خَشْيَةًۭ ۚ وَقَالُوا۟ رَبَّنَا لِمَ كَتَبْتَ عَلَيْنَا ٱلْقِتَالَ لَوْلَآ أَخَّرْتَنَآ إِلَىٰٓ أَجَلٍۢ قَرِيبٍۢ ۗ قُلْ مَتَٰعُ ٱلدُّنْيَا قَلِيلٌۭ وَٱلْءَاخِرَةُ خَيْرٌۭ لِّمَنِ ٱتَّقَىٰ وَلَا تُظْلَمُونَ فَتِيلًا ﴿٧٧﴾ ایا تاسو (صلی الله علیه وعلی آله واصحابه وسلم) د هغو خلکو (حال) نه دی لیدلی چې ورته ویل شوي وو: خپل لاسونه وساتئ (په جګړه کې بېړه مه کوئ)، لمونځ قایم کړئ او زکات ورکړئ؟ بیا کله چې پرې جګړه فرض کړای شوه، نو په هغه وخت کې له هغوی یوه ډله له خلکو داسې ووېرېده لکه له الله چې وېره کېږي یا له هغه هم زیاته (وېره پیل کړه)، او ویې ویل: «ای زموږ ربه! پر موږ دې جګړه ولې فرض کړه؟ موږ ته دې تر یوې نږدې مودې مهلت ولې رانه کړ؟» تاسو (صلی الله علیه وعلی آله واصحابه وسلم) ووایئ: د دنیا سامان ډېر لږ دی، او آخرت د هغه چا لپاره ډېر غوره دی چې پرهېزګاري یې غوره کړې، او پر تاسو به د ذرې په اندازه هم ظلم ونه شي.
p أَيْنَمَا تَكُونُوا۟ يُدْرِككُّمُ ٱلْمَوْتُ وَلَوْ كُنتُمْ فِى بُرُوجٍۢ مُّشَيَّدَةٍۢ ۗ وَإِن تُصِبْهُمْ حَسَنَةٌۭ يَقُولُوا۟ هَٰذِهِۦ مِنْ عِندِ ٱللَّهِ ۖ وَإِن تُصِبْهُمْ سَيِّئَةٌۭ يَقُولُوا۟ هَٰذِهِۦ مِنْ عِندِكَ ۚ قُلْ كُلٌّۭ مِّنْ عِندِ ٱللَّهِ ۖ فَمَالِ هَٰٓؤُلَآءِ ٱلْقَوْمِ لَا يَكَادُونَ يَفْقَهُونَ حَدِيثًۭا ﴿٧٨﴾ تم جہاں کہیں بھی ہو موت تمھیں آ پکڑے گی اگرچہ تم مضبوط قلعوں میں ہو اور اگر انھیں کوئی سکھ پہنچے تو وہ کہتے ہیں یہ اللہ کے پاس سے ہے اور اگر انھیں کوئی دکھ پہنچے تو وہ کہتے ہیں یہ آپ کی وجہ سے ہے آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) فرما دیجیے یہ سب اللہ کے پاس سے ہے تو پھر اس قوم کو کیا ہوا ہے کہ کوئی بات سمجھنے کے قریب بھی نہیں آتے۔ || أَيْنَمَا تَكُونُوا۟ يُدْرِككُّمُ ٱلْمَوْتُ وَلَوْ كُنتُمْ فِى بُرُوجٍۢ مُّشَيَّدَةٍۢ ۗ وَإِن تُصِبْهُمْ حَسَنَةٌۭ يَقُولُوا۟ هَٰذِهِۦ مِنْ عِندِ ٱللَّهِ ۖ وَإِن تُصِبْهُمْ سَيِّئَةٌۭ يَقُولُوا۟ هَٰذِهِۦ مِنْ عِندِكَ ۚ قُلْ كُلٌّۭ مِّنْ عِندِ ٱللَّهِ ۖ فَمَالِ هَٰٓؤُلَآءِ ٱلْقَوْمِ لَا يَكَادُونَ يَفْقَهُونَ حَدِيثًۭا ﴿٧٨﴾ Wheresoever ye may be, death will overtake you, even though ye were in lofty towers. Yet if a happy thing befalleth them they say: This is from Allah; and if an evil thing befalleth them they say: This is of thy doing (O Muhammad). Say (unto them): All is from Allah. What is amiss with these people that they come not nigh to understand a happening? (4:78) || أَيْنَمَا تَكُونُوا۟ يُدْرِككُّمُ ٱلْمَوْتُ وَلَوْ كُنتُمْ فِى بُرُوجٍۢ مُّشَيَّدَةٍۢ ۗ وَإِن تُصِبْهُمْ حَسَنَةٌۭ يَقُولُوا۟ هَٰذِهِۦ مِنْ عِندِ ٱللَّهِ ۖ وَإِن تُصِبْهُمْ سَيِّئَةٌۭ يَقُولُوا۟ هَٰذِهِۦ مِنْ عِندِكَ ۚ قُلْ كُلٌّۭ مِّنْ عِندِ ٱللَّهِ ۖ فَمَالِ هَٰٓؤُلَآءِ ٱلْقَوْمِ لَا يَكَادُونَ يَفْقَهُونَ حَدِيثًۭا ﴿٧٨﴾ تاسو چې هر چېرې یاست، مرګ به مو ونیسي که څه هم په ټینګو کلاګانو کې یاست؛ او که هغوی ته کومه ښېګڼه ورسېږي، وایي: «دا د الله له لوري ده»، او که کوم تکلیف ورسېږي، وایي: «دا ستاسو له امله دی». تاسو (صلی الله علیه وعلی آله واصحابه وسلم) ووایئ: «دا ټول د الله له لوري دي». نو دې قوم ته څه شوي چې د کومې خبرې پوهېدو ته نږدې هم نه ورځي؟
p مَّآ أَصَابَكَ مِنْ حَسَنَةٍۢ فَمِنَ ٱللَّهِ ۖ وَمَآ أَصَابَكَ مِن سَيِّئَةٍۢ فَمِن نَّفْسِكَ ۚ وَأَرْسَلْنَٰكَ لِلنَّاسِ رَسُولًۭا ۚ وَكَفَىٰ بِٱللَّهِ شَهِيدًۭا ﴿٧٩﴾ (اے مخاطب!) جو سکھ تجھے پہنچے تو وہ اللہ کی طرف سے ہے اور جو دکھ تجھے پہنچے تو وہ تیری طرف سے ہے اور ہم نے آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کو لوگوں کے لیے رسول بنا کر بھیجا ہے اور اللہ (آپ صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم کی رسالت پر بطور) گواہ کافی ہے۔ || مَّآ أَصَابَكَ مِنْ حَسَنَةٍۢ فَمِنَ ٱللَّهِ ۖ وَمَآ أَصَابَكَ مِن سَيِّئَةٍۢ فَمِن نَّفْسِكَ ۚ وَأَرْسَلْنَٰكَ لِلنَّاسِ رَسُولًۭا ۚ وَكَفَىٰ بِٱللَّهِ شَهِيدًۭا ﴿٧٩﴾ Whatever of good befalleth thee (O man) it is from Allah, and whatever of ill befalleth thee it is from thyself. We have sent thee (Muhammad) as a messenger unto mankind and Allah is sufficient as Witness. (4:79) || مَّآ أَصَابَكَ مِنْ حَسَنَةٍۢ فَمِنَ ٱللَّهِ ۖ وَمَآ أَصَابَكَ مِن سَيِّئَةٍۢ فَمِن نَّفْسِكَ ۚ وَأَرْسَلْنَٰكَ لِلنَّاسِ رَسُولًۭا ۚ وَكَفَىٰ بِٱللَّهِ شَهِيدًۭا ﴿٧٩﴾ (ای مخاطبه!) کومه ښېګڼه چې تا ته ورسېږي، هغه د الله له لوري ده، او کوم تکلیف چې تا ته ورسېږي، هغه ستا له لوري دی؛ او موږ تاسو (صلی الله علیه وعلی آله واصحابه وسلم) خلکو ته رسول لېږلی یاست، او الله (ستاسو پر رسالت د) شاهد (په توګه) بس دی.
p مَّن يُطِعِ ٱلرَّسُولَ فَقَدْ أَطَاعَ ٱللَّهَ ۖ وَمَن تَوَلَّىٰ فَمَآ أَرْسَلْنَٰكَ عَلَيْهِمْ حَفِيظًۭا ﴿٨٠﴾ جس نے رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کی اطاعت کی تو یقیناً اُسی نے اللہ کی اطاعت کی اور جس نے (اطاعت سے) منہ پھیرا تو ہم نے آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کو اُن پر نگہبان بنا کر نہیں بھیجا۔ || مَّن يُطِعِ ٱلرَّسُولَ فَقَدْ أَطَاعَ ٱللَّهَ ۖ وَمَن تَوَلَّىٰ فَمَآ أَرْسَلْنَٰكَ عَلَيْهِمْ حَفِيظًۭا ﴿٨٠﴾ Whoso obeyeth the messenger hath obeyed Allah, and whoso turneth away: We have not sent thee as a warder over them. (4:80) || مَّن يُطِعِ ٱلرَّسُولَ فَقَدْ أَطَاعَ ٱللَّهَ ۖ وَمَن تَوَلَّىٰ فَمَآ أَرْسَلْنَٰكَ عَلَيْهِمْ حَفِيظًۭا ﴿٨٠﴾ چا چې د رسول (صلی الله علیه وعلی آله واصحابه وسلم) اطاعت وکړ، یقیناً هغه د الله اطاعت وکړ؛ او چا چې (له اطاعته) مخ واړاوه، نو موږ تاسو (صلی الله علیه وعلی آله واصحابه وسلم) پر هغوی ساتونکی نه یاست لېږلي.
