p وَٱلَّٰٓـِٔى يَئِسْنَ مِنَ ٱلْمَحِيضِ مِن نِّسَآئِكُمْ إِنِ ٱرْتَبْتُمْ فَعِدَّتُهُنَّ ثَلَٰثَةُ أَشْهُرٍۢ وَٱلَّٰٓـِٔى لَمْ يَحِضْنَ ۚ وَأُو۟لَٰتُ ٱلْأَحْمَالِ أَجَلُهُنَّ أَن يَضَعْنَ حَمْلَهُنَّ ۚ وَمَن يَتَّقِ ٱللَّهَ يَجْعَل لَّهُۥ مِنْ أَمْرِهِۦ يُسْرًۭا ﴿٤﴾ اور تمھاری عورتوں میں سے جو حیض سے ناامید ہو چکی ہوں اگر تمھیں (ان کی عدت کے بارے میں) شک ہو تو ان کی عدت تین مہینے ہے اور ان (عورتوں) کی (عدت) بھی (تین مہینے ہے) جنھیں (ابھی) حیض نہیں آیا اور حاملہ عورتوں کی عدت ان کے وضعِ حمل (بچہ جننے) تک ہے اور جو اللہ (کی نافرمانی) سے ڈرتا رہے وہ اُس کے لیے اُس کے کام میں آسانی فرما دیتا ہے۔ || وَٱلَّٰٓـِٔى يَئِسْنَ مِنَ ٱلْمَحِيضِ مِن نِّسَآئِكُمْ إِنِ ٱرْتَبْتُمْ فَعِدَّتُهُنَّ ثَلَٰثَةُ أَشْهُرٍۢ وَٱلَّٰٓـِٔى لَمْ يَحِضْنَ ۚ وَأُو۟لَٰتُ ٱلْأَحْمَالِ أَجَلُهُنَّ أَن يَضَعْنَ حَمْلَهُنَّ ۚ وَمَن يَتَّقِ ٱللَّهَ يَجْعَل لَّهُۥ مِنْ أَمْرِهِۦ يُسْرًۭا ﴿٤﴾ And for such of your women as despair of menstruation, if ye doubt, their period (of waiting) shall be three months, along with those who have it not. And for those with child, their period shall be till they bring forth their burden. And whosoever keepeth his duty to Allah, He maketh his course easy for him. (65:4) || وَٱلَّٰٓـِٔى يَئِسْنَ مِنَ ٱلْمَحِيضِ مِن نِّسَآئِكُمْ إِنِ ٱرْتَبْتُمْ فَعِدَّتُهُنَّ ثَلَٰثَةُ أَشْهُرٍۢ وَٱلَّٰٓـِٔى لَمْ يَحِضْنَ ۚ وَأُو۟لَٰتُ ٱلْأَحْمَالِ أَجَلُهُنَّ أَن يَضَعْنَ حَمْلَهُنَّ ۚ وَمَن يَتَّقِ ٱللَّهَ يَجْعَل لَّهُۥ مِنْ أَمْرِهِۦ يُسْرًۭا ﴿٤﴾ او ستاسو له ښځو هغه چې له حیضه ناهیلې شوې وي، که تاسو ته (د هغوی د عدت په اړه) شک وي، نو د هغوی عدت درې میاشتې دی، او د هغو (ښځو) (عدت) هم (درې میاشتې دی) چې (لا) حیض نه دی ورته راغلی؛ او د امیندوارو ښځو عدت د هغوی تر وضع حمل (ماشوم زېږولو) پورې دی؛ او څوک چې له الله (د نافرمانۍ) وېرېږي، هغه د هغه لپاره په کار کې اساني پیدا کوي.
p ذَٰلِكَ أَمْرُ ٱللَّهِ أَنزَلَهُۥٓ إِلَيْكُمْ ۚ وَمَن يَتَّقِ ٱللَّهَ يُكَفِّرْ عَنْهُ سَيِّـَٔاتِهِۦ وَيُعْظِمْ لَهُۥٓ أَجْرًا ﴿٥﴾ یہ اللہ کا حکم ہے جو اُس نے تمھاری طرف نازل فرمایا ہے اور جو اللہ (کی نافرمانی) سے ڈرتا ہے اللہ اُس کے گناہ دور کر دے گا اور اس کے اجر کو اُس کے لیے بڑا کر دے گا۔ || ذَٰلِكَ أَمْرُ ٱللَّهِ أَنزَلَهُۥٓ إِلَيْكُمْ ۚ وَمَن يَتَّقِ ٱللَّهَ يُكَفِّرْ عَنْهُ سَيِّـَٔاتِهِۦ وَيُعْظِمْ لَهُۥٓ أَجْرًا ﴿٥﴾ That is the commandment of Allah which He revealeth unto you. And whoso keepeth his duty to Allah, He will remit from him his evil deeds and magnify reward for him. (65:5) || ذَٰلِكَ أَمْرُ ٱللَّهِ أَنزَلَهُۥٓ إِلَيْكُمْ ۚ وَمَن يَتَّقِ ٱللَّهَ يُكَفِّرْ عَنْهُ سَيِّـَٔاتِهِۦ وَيُعْظِمْ لَهُۥٓ أَجْرًا ﴿٥﴾ دا د الله حکم دی چې تاسو ته یې نازل کړی؛ او څوک چې له الله (د نافرمانۍ) وېرېږي، الله به یې ګناهونه لرې کړي او اجر به یې د هغه لپاره لوی کړي.
p أَسْكِنُوهُنَّ مِنْ حَيْثُ سَكَنتُم مِّن وُجْدِكُمْ وَلَا تُضَآرُّوهُنَّ لِتُضَيِّقُوا۟ عَلَيْهِنَّ ۚ وَإِن كُنَّ أُو۟لَٰتِ حَمْلٍۢ فَأَنفِقُوا۟ عَلَيْهِنَّ حَتَّىٰ يَضَعْنَ حَمْلَهُنَّ ۚ فَإِنْ أَرْضَعْنَ لَكُمْ فَـَٔاتُوهُنَّ أُجُورَهُنَّ ۖ وَأْتَمِرُوا۟ بَيْنَكُم بِمَعْرُوفٍۢ ۖ وَإِن تَعَاسَرْتُمْ فَسَتُرْضِعُ لَهُۥٓ أُخْرَىٰ ﴿٦﴾ (دورانِ عدت) ان (مطلقہ) عورتوں کو اپنی حیثیت کے مطابق اُس جگہ رہائش مہیا کرو جہاں تم رہتے ہو اور انھیں تنگ کرنے کے لیے انھیں تکلیف مت پہنچاؤ اور اگر وہ حاملہ ہوں تو انھیں خرچ دیتے رہو یہاں تک کہ وہ اپنا (بچہ جن کر) وضعِ حمل کر لیں پھر اگر وہ تمھاری خاطر (بچے کو) دودھ پلائیں تو انھیں اُن کی اجرت ادا کرو اور (اجرت مقرر کرنے کے لیے) عمدہ طریقہ سے باہم مشورہ کر لیا کرو اور اگر تم باہم دشواری محسوس کرو تو اُس (ماں کی جگہ) اُسے کوئی دوسری عورت دودھ پلائے گی۔ || أَسْكِنُوهُنَّ مِنْ حَيْثُ سَكَنتُم مِّن وُجْدِكُمْ وَلَا تُضَآرُّوهُنَّ لِتُضَيِّقُوا۟ عَلَيْهِنَّ ۚ وَإِن كُنَّ أُو۟لَٰتِ حَمْلٍۢ فَأَنفِقُوا۟ عَلَيْهِنَّ حَتَّىٰ يَضَعْنَ حَمْلَهُنَّ ۚ فَإِنْ أَرْضَعْنَ لَكُمْ فَـَٔاتُوهُنَّ أُجُورَهُنَّ ۖ وَأْتَمِرُوا۟ بَيْنَكُم بِمَعْرُوفٍۢ ۖ وَإِن تَعَاسَرْتُمْ فَسَتُرْضِعُ لَهُۥٓ أُخْرَىٰ ﴿٦﴾ Lodge them where ye dwell, according to your wealth, and harass them not so as to straiten life for them. And if they are with child, then spend for them till they bring forth their burden. Then, if they give suck for you, give them their due payment and consult together in kindness; but if ye make difficulties for one another, then let some other woman give suck for him (the father of the child). (65:6) || أَسْكِنُوهُنَّ مِنْ حَيْثُ سَكَنتُم مِّن وُجْدِكُمْ وَلَا تُضَآرُّوهُنَّ لِتُضَيِّقُوا۟ عَلَيْهِنَّ ۚ وَإِن كُنَّ أُو۟لَٰتِ حَمْلٍۢ فَأَنفِقُوا۟ عَلَيْهِنَّ حَتَّىٰ يَضَعْنَ حَمْلَهُنَّ ۚ فَإِنْ أَرْضَعْنَ لَكُمْ فَـَٔاتُوهُنَّ أُجُورَهُنَّ ۖ وَأْتَمِرُوا۟ بَيْنَكُم بِمَعْرُوفٍۢ ۖ وَإِن تَعَاسَرْتُمْ فَسَتُرْضِعُ لَهُۥٓ أُخْرَىٰ ﴿٦﴾ (د عدت پر مهال) دې (طلاقې شوې) ښځو ته د خپل وس سره سم په هغه ځای کې استوګنه برابره کړئ چې تاسو پکې اوسېږئ، او د تنګولو لپاره یې مه ځوروئ؛ او که امیندوارې وي، نو لګښت ورکوئ تر دې چې خپل (ماشوم وزېږوي او) وضع حمل وکړي؛ بیا که ستاسو لپاره (ماشوم ته) شیدې ورکړي، نو د هغوی مزدوري ورکړئ، او (د مزدورۍ ټاکلو لپاره) په ښه توګه سره مشوره وکړئ؛ او که په خپل منځ کې ستونزه احساس کړئ، نو د هغه (د مور پر ځای) به بله ښځه شیدې ورکړي.
p لِيُنفِقْ ذُو سَعَةٍۢ مِّن سَعَتِهِۦ ۖ وَمَن قُدِرَ عَلَيْهِ رِزْقُهُۥ فَلْيُنفِقْ مِمَّآ ءَاتَىٰهُ ٱللَّهُ ۚ لَا يُكَلِّفُ ٱللَّهُ نَفْسًا إِلَّا مَآ ءَاتَىٰهَا ۚ سَيَجْعَلُ ٱللَّهُ بَعْدَ عُسْرٍۢ يُسْرًۭا ﴿٧﴾ وسعت والے کو چاہیے کہ وہ اپنی وسعت کے مطابق خرچ دے اور جس شخص پر اس کا رزق تنگ کر دیا گیا ہو تو اسے چاہیے کہ جتنا اللہ نے اسے عطا فرمایا ہے اس میں سے خرچ دے اللہ کسی بھی شخص پر ذمہ داری نہیں ڈالتا مگر اسی قدر جتنا اس نے اُسے عطا کیا ہے عنقریب اللہ تنگ دستی کے بعد آسانی پیدا فرما دے گا۔ || لِيُنفِقْ ذُو سَعَةٍۢ مِّن سَعَتِهِۦ ۖ وَمَن قُدِرَ عَلَيْهِ رِزْقُهُۥ فَلْيُنفِقْ مِمَّآ ءَاتَىٰهُ ٱللَّهُ ۚ لَا يُكَلِّفُ ٱللَّهُ نَفْسًا إِلَّا مَآ ءَاتَىٰهَا ۚ سَيَجْعَلُ ٱللَّهُ بَعْدَ عُسْرٍۢ يُسْرًۭا ﴿٧﴾ Let him who hath abundance spend of his abundance, and he whose provision is measured, let him spend of that which Allah hath given him. Allah asketh naught of any soul save that which He hath given it. Allah will vouchsafe, after hardship, ease. (65:7) || لِيُنفِقْ ذُو سَعَةٍۢ مِّن سَعَتِهِۦ ۖ وَمَن قُدِرَ عَلَيْهِ رِزْقُهُۥ فَلْيُنفِقْ مِمَّآ ءَاتَىٰهُ ٱللَّهُ ۚ لَا يُكَلِّفُ ٱللَّهُ نَفْسًا إِلَّا مَآ ءَاتَىٰهَا ۚ سَيَجْعَلُ ٱللَّهُ بَعْدَ عُسْرٍۢ يُسْرًۭا ﴿٧﴾ پراخي لرونکی دې د خپلې پراخۍ سره سم لګښت ورکړي؛ او هغه څوک چې رزق یې پرې تنګ شوی وي، باید له هغه څه چې الله ورته ورکړي، لګښت ورکړي؛ الله پر هېڅ کس مسؤلیت نه اچوي مګر هومره چې ورته یې ورکړي؛ ژر به الله له تنګسې وروسته اساني پیدا کړي.
p وَكَأَيِّن مِّن قَرْيَةٍ عَتَتْ عَنْ أَمْرِ رَبِّهَا وَرُسُلِهِۦ فَحَاسَبْنَٰهَا حِسَابًۭا شَدِيدًۭا وَعَذَّبْنَٰهَا عَذَابًۭا نُّكْرًۭا ﴿٨﴾ اور کتنی ہی بستیاں ایسی تھیں جن کے رہنے والوں نے اپنے رب اور اُس کے رسولوں کے حکم سے سرکشی کی تو ہم نے اُن سے سخت حساب لیا اور انھیں (ایسے) سخت عذاب میں مبتلا کیا (جو نہ دیکھا نہ سنا گیا تھا)۔ || وَكَأَيِّن مِّن قَرْيَةٍ عَتَتْ عَنْ أَمْرِ رَبِّهَا وَرُسُلِهِۦ فَحَاسَبْنَٰهَا حِسَابًۭا شَدِيدًۭا وَعَذَّبْنَٰهَا عَذَابًۭا نُّكْرًۭا ﴿٨﴾ And how many a community revolted against the ordinance of its Lord and His messengers, and We called it to a stern account and punished it with dire punishment, (65:8) || وَكَأَيِّن مِّن قَرْيَةٍ عَتَتْ عَنْ أَمْرِ رَبِّهَا وَرُسُلِهِۦ فَحَاسَبْنَٰهَا حِسَابًۭا شَدِيدًۭا وَعَذَّبْنَٰهَا عَذَابًۭا نُّكْرًۭا ﴿٨﴾ او څومره ډېر کلي وو چې اوسېدونکو یې د خپل رب او د هغه د رسولانو له حکمه سرکشي وکړه، نو موږ ترې سخت حساب واخیست او هغوی مو په (داسې) سخت عذاب اخته کړل (چې نه لیدل شوی و نه اورېدل شوی).
p فَذَاقَتْ وَبَالَ أَمْرِهَا وَكَانَ عَٰقِبَةُ أَمْرِهَا خُسْرًا ﴿٩﴾ تو انھوں نے اپنے کیے کا وبال چکھا اور اُن کے کام کا انجام خسارہ ہی ہوا۔ || فَذَاقَتْ وَبَالَ أَمْرِهَا وَكَانَ عَٰقِبَةُ أَمْرِهَا خُسْرًا ﴿٩﴾ So that it tasted the ill-effects of its conduct, and the consequence of its conduct was loss. (65:9) || فَذَاقَتْ وَبَالَ أَمْرِهَا وَكَانَ عَٰقِبَةُ أَمْرِهَا خُسْرًا ﴿٩﴾ نو هغوی د خپلو کړنو وبال وڅکه، او د هغوی د کار پایله یوازې زیان شو.
