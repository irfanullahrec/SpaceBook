p إِنَّ ٱلْمُنَٰفِقِينَ فِى ٱلدَّرْكِ ٱلْأَسْفَلِ مِنَ ٱلنَّارِ وَلَن تَجِدَ لَهُمْ نَصِيرًا ﴿١٤٥﴾ بے شک منافقین جہنم کے سب سے نچلے طبقہ میں ہوں گے اور تم اُن کے لیے ہرگز کوئی مدد گار نہ پاؤ گے۔ || إِنَّ ٱلْمُنَٰفِقِينَ فِى ٱلدَّرْكِ ٱلْأَسْفَلِ مِنَ ٱلنَّارِ وَلَن تَجِدَ لَهُمْ نَصِيرًا ﴿١٤٥﴾ Lo! the hypocrites (will be) in the lowest deep of the Fire, and thou wilt find no helper for them; (4:145) || إِنَّ ٱلْمُنَٰفِقِينَ فِى ٱلدَّرْكِ ٱلْأَسْفَلِ مِنَ ٱلنَّارِ وَلَن تَجِدَ لَهُمْ نَصِيرًا ﴿١٤٥﴾ بې‌شکه منافقان به د دوزخ په تر ټولو ښکته طبقه کې وي، او هېڅکله به ورته کوم مرستندوی ونه مومئ.
p إِلَّا ٱلَّذِينَ تَابُوا۟ وَأَصْلَحُوا۟ وَٱعْتَصَمُوا۟ بِٱللَّهِ وَأَخْلَصُوا۟ دِينَهُمْ لِلَّهِ فَأُو۟لَٰٓئِكَ مَعَ ٱلْمُؤْمِنِينَ ۖ وَسَوْفَ يُؤْتِ ٱللَّهُ ٱلْمُؤْمِنِينَ أَجْرًا عَظِيمًۭا ﴿١٤٦﴾ سوائے اُن لوگوں کے جنھوں نے توبہ کی اور (اپنی) اصلاح کر لی اور اللہ (کے سہارے) کو مضبوطی سے تھام لیا اور اپنے دین کو اللہ کے لیے خالص کر لیا تو ایسے لوگ مومنوں کے ساتھ ہیں اور عنقریب اللہ مومنوں کو اجرِ عظیم عطا فرمائے گا۔ || إِلَّا ٱلَّذِينَ تَابُوا۟ وَأَصْلَحُوا۟ وَٱعْتَصَمُوا۟ بِٱللَّهِ وَأَخْلَصُوا۟ دِينَهُمْ لِلَّهِ فَأُو۟لَٰٓئِكَ مَعَ ٱلْمُؤْمِنِينَ ۖ وَسَوْفَ يُؤْتِ ٱللَّهُ ٱلْمُؤْمِنِينَ أَجْرًا عَظِيمًۭا ﴿١٤٦﴾ Save those who repent and amend and hold fast to Allah and make their religion pure for Allah (only). Those are with the believers. And Allah will bestow on the believers an immense reward. (4:146) || إِلَّا ٱلَّذِينَ تَابُوا۟ وَأَصْلَحُوا۟ وَٱعْتَصَمُوا۟ بِٱللَّهِ وَأَخْلَصُوا۟ دِينَهُمْ لِلَّهِ فَأُو۟لَٰٓئِكَ مَعَ ٱلْمُؤْمِنِينَ ۖ وَسَوْفَ يُؤْتِ ٱللَّهُ ٱلْمُؤْمِنِينَ أَجْرًا عَظِيمًۭا ﴿١٤٦﴾ پرته له هغو کسانو چې توبه یې وکړه، (ځانونه یې) سم کړل، د الله (پر ملاتړ) یې په کلکه منګولې ولګولې او خپل دین یې الله ته خالص کړ؛ نو داسې خلک له مؤمنانو سره دي، او ژر به الله مؤمنانو ته لوی اجر ورکړي.
p مَّا يَفْعَلُ ٱللَّهُ بِعَذَابِكُمْ إِن شَكَرْتُمْ وَءَامَنتُمْ ۚ وَكَانَ ٱللَّهُ شَاكِرًا عَلِيمًۭا ﴿١٤٧﴾ اللہ تمھیں عذاب دے کر کیا کرے گا اگر تم شکر ادا کرو اور ایمان لے آؤ اور اللہ بڑا قدردان خوب جاننے والا ہے۔ || مَّا يَفْعَلُ ٱللَّهُ بِعَذَابِكُمْ إِن شَكَرْتُمْ وَءَامَنتُمْ ۚ وَكَانَ ٱللَّهُ شَاكِرًا عَلِيمًۭا ﴿١٤٧﴾ What concern hath Allah for your punishment if ye are thankful (for His mercies) and believe (in Him)? Allah was ever Responsive, Aware. (4:147) || مَّا يَفْعَلُ ٱللَّهُ بِعَذَابِكُمْ إِن شَكَرْتُمْ وَءَامَنتُمْ ۚ وَكَانَ ٱللَّهُ شَاكِرًا عَلِيمًۭا ﴿١٤٧﴾ الله به تاسو په عذاب څه وکړي که شکر ادا کړئ او ایمان راوړئ؟ او الله ډېر قدردان، ښه پوه دی.
p ۞ لَّا يُحِبُّ ٱللَّهُ ٱلْجَهْرَ بِٱلسُّوٓءِ مِنَ ٱلْقَوْلِ إِلَّا مَن ظُلِمَ ۚ وَكَانَ ٱللَّهُ سَمِيعًا عَلِيمًا ﴿١٤٨﴾ اللہ اس بات کو پسند نہیں کرتا کہ (کسی کی) کوئی بُری بات علانیہ کہی جائے سوائے اُس کے کہ جس پر ظلم کیا گیا اور اللہ خوب سننے والا خوب جاننے والا ہے۔ || ۞ لَّا يُحِبُّ ٱللَّهُ ٱلْجَهْرَ بِٱلسُّوٓءِ مِنَ ٱلْقَوْلِ إِلَّا مَن ظُلِمَ ۚ وَكَانَ ٱللَّهُ سَمِيعًا عَلِيمًا ﴿١٤٨﴾ Allah loveth not the utterance of harsh speech save by one who hath been wronged. Allah is ever Hearer, Knower. (4:148) || ۞ لَّا يُحِبُّ ٱللَّهُ ٱلْجَهْرَ بِٱلسُّوٓءِ مِنَ ٱلْقَوْلِ إِلَّا مَن ظُلِمَ ۚ وَكَانَ ٱللَّهُ سَمِيعًا عَلِيمًا ﴿١٤٨﴾ الله دا نه خوښوي چې د (چا) کومه بده خبره په ښکاره وویل شي، پرته له هغه چا چې ظلم پرې شوی؛ او الله ښه اورېدونکی، ښه پوه دی.
p إِن تُبْدُوا۟ خَيْرًا أَوْ تُخْفُوهُ أَوْ تَعْفُوا۟ عَن سُوٓءٍۢ فَإِنَّ ٱللَّهَ كَانَ عَفُوًّۭا قَدِيرًا ﴿١٤٩﴾ اگر تم کوئی بھلائی علانیہ کرو یا اُسے پوشیدہ رکھو یا (کسی کی) برائی کو معاف کر دو تو بے شک اللہ بہت معاف فرمانے والا خوب قدرت رکھنے والا ہے۔ || إِن تُبْدُوا۟ خَيْرًا أَوْ تُخْفُوهُ أَوْ تَعْفُوا۟ عَن سُوٓءٍۢ فَإِنَّ ٱللَّهَ كَانَ عَفُوًّۭا قَدِيرًا ﴿١٤٩﴾ If ye do good openly or keep it secret, or forgive evil, lo! Allah is ever Forgiving, Powerful. (4:149) || إِن تُبْدُوا۟ خَيْرًا أَوْ تُخْفُوهُ أَوْ تَعْفُوا۟ عَن سُوٓءٍۢ فَإِنَّ ٱللَّهَ كَانَ عَفُوًّۭا قَدِيرًا ﴿١٤٩﴾ که کومه ښېګڼه په ښکاره وکړئ یا یې پټه وساتئ یا د (چا) بدي وبښئ، نو بې‌شکه الله ډېر معاف کوونکی، ښه قدرت لرونکی دی.
p إِنَّ ٱلَّذِينَ يَكْفُرُونَ بِٱللَّهِ وَرُسُلِهِۦ وَيُرِيدُونَ أَن يُفَرِّقُوا۟ بَيْنَ ٱللَّهِ وَرُسُلِهِۦ وَيَقُولُونَ نُؤْمِنُ بِبَعْضٍۢ وَنَكْفُرُ بِبَعْضٍۢ وَيُرِيدُونَ أَن يَتَّخِذُوا۟ بَيْنَ ذَٰلِكَ سَبِيلًا ﴿١٥٠﴾ بے شک جو لوگ اللہ اور اُس کے رسولوں کا انکار کرتے ہیں اور چاہتے ہیں کہ اللہ اور اُس کے رسولوں کے درمیان فرق کریں اور وہ کہتے ہیں کہ ہم بعض (رسولوں) پر ایمان لائیں گے اور بعض کا انکار کریں گے اور چاہتے ہیں کہ اس طرح (ایمان و کفر) کے درمیان کوئی راہ نکال لیں۔ || إِنَّ ٱلَّذِينَ يَكْفُرُونَ بِٱللَّهِ وَرُسُلِهِۦ وَيُرِيدُونَ أَن يُفَرِّقُوا۟ بَيْنَ ٱللَّهِ وَرُسُلِهِۦ وَيَقُولُونَ نُؤْمِنُ بِبَعْضٍۢ وَنَكْفُرُ بِبَعْضٍۢ وَيُرِيدُونَ أَن يَتَّخِذُوا۟ بَيْنَ ذَٰلِكَ سَبِيلًا ﴿١٥٠﴾ Lo! those who disbelieve in Allah and His messengers, and seek to make distinction between Allah and His messengers, and say: We believe in some and disbelieve in others, and seek to choose a way in between; (4:150) || إِنَّ ٱلَّذِينَ يَكْفُرُونَ بِٱللَّهِ وَرُسُلِهِۦ وَيُرِيدُونَ أَن يُفَرِّقُوا۟ بَيْنَ ٱللَّهِ وَرُسُلِهِۦ وَيَقُولُونَ نُؤْمِنُ بِبَعْضٍۢ وَنَكْفُرُ بِبَعْضٍۢ وَيُرِيدُونَ أَن يَتَّخِذُوا۟ بَيْنَ ذَٰلِكَ سَبِيلًا ﴿١٥٠﴾ بې‌شکه هغه کسان چې له الله او د هغه له رسولانو انکار کوي، غواړي چې د الله او د هغه د رسولانو ترمنځ توپیر وکړي، او وایي: «پر ځینو (رسولانو) به ایمان راوړو او له ځینو به انکار کوو»، او غواړي چې په دې ډول د (ایمان او کفر) ترمنځ کومه لاره وباسي.
p أُو۟لَٰٓئِكَ هُمُ ٱلْكَٰفِرُونَ حَقًّۭا ۚ وَأَعْتَدْنَا لِلْكَٰفِرِينَ عَذَابًۭا مُّهِينًۭا ﴿١٥١﴾ یہی لوگ پکے کافر ہیں اور ہم نے کافروں کے لیے ذلت آمیز عذاب تیار کر رکھا ہے۔ || أُو۟لَٰٓئِكَ هُمُ ٱلْكَٰفِرُونَ حَقًّۭا ۚ وَأَعْتَدْنَا لِلْكَٰفِرِينَ عَذَابًۭا مُّهِينًۭا ﴿١٥١﴾ Such are disbelievers in truth; and for disbelievers We prepare a shameful doom. (4:151) || أُو۟لَٰٓئِكَ هُمُ ٱلْكَٰفِرُونَ حَقًّۭا ۚ وَأَعْتَدْنَا لِلْكَٰفِرِينَ عَذَابًۭا مُّهِينًۭا ﴿١٥١﴾ همدا خلک پاخه کافران دي، او موږ کافرانو ته سپکوونکی عذاب چمتو کړی.
p وَٱلَّذِينَ ءَامَنُوا۟ بِٱللَّهِ وَرُسُلِهِۦ وَلَمْ يُفَرِّقُوا۟ بَيْنَ أَحَدٍۢ مِّنْهُمْ أُو۟لَٰٓئِكَ سَوْفَ يُؤْتِيهِمْ أُجُورَهُمْ ۗ وَكَانَ ٱللَّهُ غَفُورًۭا رَّحِيمًۭا ﴿١٥٢﴾ اور جو لوگ اللہ اور اُس کے رسولوں پر ایمان لائے اور ان میں سے کسی ایک کے درمیان بھی فرق نہیں کیا یہ وہ لوگ ہیں جنھیں اللہ ان کے اجر جلد عطا فرمائے گا اور اللہ بڑا ہی بخشنے والا نہایت رحم فرمانے والا ہے۔ || وَٱلَّذِينَ ءَامَنُوا۟ بِٱللَّهِ وَرُسُلِهِۦ وَلَمْ يُفَرِّقُوا۟ بَيْنَ أَحَدٍۢ مِّنْهُمْ أُو۟لَٰٓئِكَ سَوْفَ يُؤْتِيهِمْ أُجُورَهُمْ ۗ وَكَانَ ٱللَّهُ غَفُورًۭا رَّحِيمًۭا ﴿١٥٢﴾ But those who believe in Allah and His messengers and make no distinction between any of them, unto them Allah will give their wages; and Allah was ever Forgiving, Merciful. (4:152) || وَٱلَّذِينَ ءَامَنُوا۟ بِٱللَّهِ وَرُسُلِهِۦ وَلَمْ يُفَرِّقُوا۟ بَيْنَ أَحَدٍۢ مِّنْهُمْ أُو۟لَٰٓئِكَ سَوْفَ يُؤْتِيهِمْ أُجُورَهُمْ ۗ وَكَانَ ٱللَّهُ غَفُورًۭا رَّحِيمًۭا ﴿١٥٢﴾ او هغه کسان چې پر الله او د هغه پر رسولانو یې ایمان راوړ او د هغوی د یوه ترمنځ یې هم توپیر ونه کړ، دا هغه خلک دي چې الله به یې ژر اجرونه ورکړي؛ او الله ډېر بښونکی، بې‌حده مهربان دی.
