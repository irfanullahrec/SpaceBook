p ٱلَّذِينَ يَتَّخِذُونَ ٱلْكَٰفِرِينَ أَوْلِيَآءَ مِن دُونِ ٱلْمُؤْمِنِينَ ۚ أَيَبْتَغُونَ عِندَهُمُ ٱلْعِزَّةَ فَإِنَّ ٱلْعِزَّةَ لِلَّهِ جَمِيعًۭا ﴿١٣٩﴾ (یہ منافقین) وہ ہیں جو مومنوں کو چھوڑ کر کافروں کو دوست بناتے ہیں کیا وہ اُن کے پاس عزت تلاش کرتے ہیں؟ تو بے شک عزت تو سب کی سب اللہ کے لیے ہے۔ || ٱلَّذِينَ يَتَّخِذُونَ ٱلْكَٰفِرِينَ أَوْلِيَآءَ مِن دُونِ ٱلْمُؤْمِنِينَ ۚ أَيَبْتَغُونَ عِندَهُمُ ٱلْعِزَّةَ فَإِنَّ ٱلْعِزَّةَ لِلَّهِ جَمِيعًۭا ﴿١٣٩﴾ Those who chose disbelievers for their friends instead of believers! Do they look for power at their hands? Lo! all power appertaineth to Allah. (4:139) || ٱلَّذِينَ يَتَّخِذُونَ ٱلْكَٰفِرِينَ أَوْلِيَآءَ مِن دُونِ ٱلْمُؤْمِنِينَ ۚ أَيَبْتَغُونَ عِندَهُمُ ٱلْعِزَّةَ فَإِنَّ ٱلْعِزَّةَ لِلَّهِ جَمِيعًۭا ﴿١٣٩﴾ (دا منافقان) هغه دي چې مؤمنان پرېږدي او کافران دوستان نیسي؛ ایا له هغوی سره عزت لټوي؟ نو بې‌شکه ټول عزت د الله دی.
p وَقَدْ نَزَّلَ عَلَيْكُمْ فِى ٱلْكِتَٰبِ أَنْ إِذَا سَمِعْتُمْ ءَايَٰتِ ٱللَّهِ يُكْفَرُ بِهَا وَيُسْتَهْزَأُ بِهَا فَلَا تَقْعُدُوا۟ مَعَهُمْ حَتَّىٰ يَخُوضُوا۟ فِى حَدِيثٍ غَيْرِهِۦٓ ۚ إِنَّكُمْ إِذًۭا مِّثْلُهُمْ ۗ إِنَّ ٱللَّهَ جَامِعُ ٱلْمُنَٰفِقِينَ وَٱلْكَٰفِرِينَ فِى جَهَنَّمَ جَمِيعًا ﴿١٤٠﴾ اور یقیناً (اللہ نے) تم پر کتاب میں (یہ حکم) نازل فرمایا کہ جب تم سنو کہ اللہ کی آیات کا انکار کیا جا رہا ہے اور اُن کا مذاق اڑایا جا رہا ہے تو اُن (مذاق اڑانے والوں) کے ساتھ مت بیٹھو یہاں تک کہ وہ کسی دوسری بات میں مشغول ہو جائیں ورنہ تم بھی اُن ہی جیسے ہو جاؤ گے بے شک اللہ منافقوں اور کافروں کو جہنم میں ایک ساتھ جمع فرمانے والا ہے۔ || وَقَدْ نَزَّلَ عَلَيْكُمْ فِى ٱلْكِتَٰبِ أَنْ إِذَا سَمِعْتُمْ ءَايَٰتِ ٱللَّهِ يُكْفَرُ بِهَا وَيُسْتَهْزَأُ بِهَا فَلَا تَقْعُدُوا۟ مَعَهُمْ حَتَّىٰ يَخُوضُوا۟ فِى حَدِيثٍ غَيْرِهِۦٓ ۚ إِنَّكُمْ إِذًۭا مِّثْلُهُمْ ۗ إِنَّ ٱللَّهَ جَامِعُ ٱلْمُنَٰفِقِينَ وَٱلْكَٰفِرِينَ فِى جَهَنَّمَ جَمِيعًا ﴿١٤٠﴾ He hath already revealed unto you in the Scripture that, when ye hear the revelations of Allah rejected and derided, (ye) sit not with them (who disbelieve and mock) until they engage in some other conversation. Lo! in that case (if ye stayed) ye would be like unto them. Lo! Allah will gather hypocrites and disbelievers, all together, into hell; (4:140) || وَقَدْ نَزَّلَ عَلَيْكُمْ فِى ٱلْكِتَٰبِ أَنْ إِذَا سَمِعْتُمْ ءَايَٰتِ ٱللَّهِ يُكْفَرُ بِهَا وَيُسْتَهْزَأُ بِهَا فَلَا تَقْعُدُوا۟ مَعَهُمْ حَتَّىٰ يَخُوضُوا۟ فِى حَدِيثٍ غَيْرِهِۦٓ ۚ إِنَّكُمْ إِذًۭا مِّثْلُهُمْ ۗ إِنَّ ٱللَّهَ جَامِعُ ٱلْمُنَٰفِقِينَ وَٱلْكَٰفِرِينَ فِى جَهَنَّمَ جَمِيعًا ﴿١٤٠﴾ او یقیناً (الله) پر تاسو په کتاب کې (دا حکم) نازل کړی چې کله واورئ چې د الله له آیتونو انکار کېږي او پرې ملنډې وهل کېږي، نو له هغو (ملنډې وهونکو) سره مه کېنئ تر دې چې په بله خبره بوخت شي، ګنې تاسو به هم د هغوی په څېر شئ؛ بې‌شکه الله منافقان او کافران په دوزخ کې یوځای راټولوونکی دی.
p ٱلَّذِينَ يَتَرَبَّصُونَ بِكُمْ فَإِن كَانَ لَكُمْ فَتْحٌۭ مِّنَ ٱللَّهِ قَالُوٓا۟ أَلَمْ نَكُن مَّعَكُمْ وَإِن كَانَ لِلْكَٰفِرِينَ نَصِيبٌۭ قَالُوٓا۟ أَلَمْ نَسْتَحْوِذْ عَلَيْكُمْ وَنَمْنَعْكُم مِّنَ ٱلْمُؤْمِنِينَ ۚ فَٱللَّهُ يَحْكُمُ بَيْنَكُمْ يَوْمَ ٱلْقِيَٰمَةِ ۗ وَلَن يَجْعَلَ ٱللَّهُ لِلْكَٰفِرِينَ عَلَى ٱلْمُؤْمِنِينَ سَبِيلًا ﴿١٤١﴾ وہ (منافقین) جو تمھارے (انجام کے) بارے میں انتظار کر رہے ہیں پس اگر تمھارے حق میں اللہ کی طرف سے فتح ہو تو کہتے ہیں کہ کیا ہم تمھارے ساتھ نہیں تھے؟ اور اگر کافروں کے لیے (فتح میں سے) کوئی حصہ ہو تو کہتے ہیں کیا ہم نے تمھیں گھیر نہ لیا تھا اور (کیا اس کے باوجود) ہم نے تمھیں مومنوں سے بچایا نہیں تھا؟ تو (اے منافقو!) اللہ تمھارے درمیان قیامت کے دن فیصلہ فرما دے گا اور اللہ کافروں کو مومنوں پر (غلبہ کی) کوئی راہ ہرگز نہیں دے گا۔ || ٱلَّذِينَ يَتَرَبَّصُونَ بِكُمْ فَإِن كَانَ لَكُمْ فَتْحٌۭ مِّنَ ٱللَّهِ قَالُوٓا۟ أَلَمْ نَكُن مَّعَكُمْ وَإِن كَانَ لِلْكَٰفِرِينَ نَصِيبٌۭ قَالُوٓا۟ أَلَمْ نَسْتَحْوِذْ عَلَيْكُمْ وَنَمْنَعْكُم مِّنَ ٱلْمُؤْمِنِينَ ۚ فَٱللَّهُ يَحْكُمُ بَيْنَكُمْ يَوْمَ ٱلْقِيَٰمَةِ ۗ وَلَن يَجْعَلَ ٱللَّهُ لِلْكَٰفِرِينَ عَلَى ٱلْمُؤْمِنِينَ سَبِيلًا ﴿١٤١﴾ Those who wait upon occasion in regard to you and, if a victory cometh unto you from Allah, say: Are we not with you? and if the disbelievers meet with a success say: Had we not the mastery of you, and did we not protect you from the believers? - Allah will judge between you at the Day of Resurrection, and Allah will not give the disbelievers any way (of success) against the believers. (4:141) || ٱلَّذِينَ يَتَرَبَّصُونَ بِكُمْ فَإِن كَانَ لَكُمْ فَتْحٌۭ مِّنَ ٱللَّهِ قَالُوٓا۟ أَلَمْ نَكُن مَّعَكُمْ وَإِن كَانَ لِلْكَٰفِرِينَ نَصِيبٌۭ قَالُوٓا۟ أَلَمْ نَسْتَحْوِذْ عَلَيْكُمْ وَنَمْنَعْكُم مِّنَ ٱلْمُؤْمِنِينَ ۚ فَٱللَّهُ يَحْكُمُ بَيْنَكُمْ يَوْمَ ٱلْقِيَٰمَةِ ۗ وَلَن يَجْعَلَ ٱللَّهُ لِلْكَٰفِرِينَ عَلَى ٱلْمُؤْمِنِينَ سَبِيلًا ﴿١٤١﴾ هغه (منافقان) چې ستاسو د (پایلې) په اړه انتظار کوي: نو که ستاسو په ګټه د الله له لوري فتحه وشي، وایي: «ایا موږ له تاسو سره نه وو؟» او که د کافرانو لپاره (له فتحې) کومه برخه وي، وایي: «ایا موږ تاسو نه وئ را ګېر کړي او (سره له دې) ایا موږ تاسو له مؤمنانو نه وئ ژغورلي؟» نو (ای منافقانو!) الله به د قیامت په ورځ ستاسو ترمنځ پرېکړه وکړي، او الله به کافرانو ته پر مؤمنانو د (غلبې) هېڅ لاره ورنه کړي.
p إِنَّ ٱلْمُنَٰفِقِينَ يُخَٰدِعُونَ ٱللَّهَ وَهُوَ خَٰدِعُهُمْ وَإِذَا قَامُوٓا۟ إِلَى ٱلصَّلَوٰةِ قَامُوا۟ كُسَالَىٰ يُرَآءُونَ ٱلنَّاسَ وَلَا يَذْكُرُونَ ٱللَّهَ إِلَّا قَلِيلًۭا ﴿١٤٢﴾ بے شک منافقین اللہ کو دھوکا دینا چاہتے ہیں اور وہ (اللہ) انھیں (ان کے) دھوکے کی سزا دینے والا ہے اور جب وہ نماز کے لیے کھڑے ہوتے ہیں تو سستی کے ساتھ کھڑے ہوتے ہیں لوگوں کو دکھاوا کرتے ہوئے اور اللہ کو کم ہی یاد کرتے ہیں۔ || إِنَّ ٱلْمُنَٰفِقِينَ يُخَٰدِعُونَ ٱللَّهَ وَهُوَ خَٰدِعُهُمْ وَإِذَا قَامُوٓا۟ إِلَى ٱلصَّلَوٰةِ قَامُوا۟ كُسَالَىٰ يُرَآءُونَ ٱلنَّاسَ وَلَا يَذْكُرُونَ ٱللَّهَ إِلَّا قَلِيلًۭا ﴿١٤٢﴾ Lo! the hypocrites seek to beguile Allah, but it is He Who beguileth them. When they stand up to worship they perform it languidly and to be seen of men, and are mindful of Allah but little; (4:142) || إِنَّ ٱلْمُنَٰفِقِينَ يُخَٰدِعُونَ ٱللَّهَ وَهُوَ خَٰدِعُهُمْ وَإِذَا قَامُوٓا۟ إِلَى ٱلصَّلَوٰةِ قَامُوا۟ كُسَالَىٰ يُرَآءُونَ ٱلنَّاسَ وَلَا يَذْكُرُونَ ٱللَّهَ إِلَّا قَلِيلًۭا ﴿١٤٢﴾ بې‌شکه منافقان غواړي الله وغولوي، او هغه (الله) د هغوی د دوکې سزا ورکوونکی دی؛ او کله چې لمانځه ته ودرېږي، په سستۍ ودرېږي، خلکو ته ځان ښیي، او الله لږ یادوي.
p مُّذَبْذَبِينَ بَيْنَ ذَٰلِكَ لَآ إِلَىٰ هَٰٓؤُلَآءِ وَلَآ إِلَىٰ هَٰٓؤُلَآءِ ۚ وَمَن يُضْلِلِ ٱللَّهُ فَلَن تَجِدَ لَهُۥ سَبِيلًۭا ﴿١٤٣﴾ وہ اس (کفر اور ایمان) کے درمیان تذبذب میں ہیں نہ ان (مسلمانوں) کی طرف ہیں اور نہ ان (کافروں) کی طرف اور جسے اللہ گم راہ کر دے پھر آپ اُس کے لیے (ہدایت کا) کوئی راستہ نہیں پائیں گے۔ || مُّذَبْذَبِينَ بَيْنَ ذَٰلِكَ لَآ إِلَىٰ هَٰٓؤُلَآءِ وَلَآ إِلَىٰ هَٰٓؤُلَآءِ ۚ وَمَن يُضْلِلِ ٱللَّهُ فَلَن تَجِدَ لَهُۥ سَبِيلًۭا ﴿١٤٣﴾ Swaying between this (and that), (belonging) neither to these nor to those. He whom Allah causeth to go astray, thou (O Muhammad) wilt not find a way for him: (4:143) || مُّذَبْذَبِينَ بَيْنَ ذَٰلِكَ لَآ إِلَىٰ هَٰٓؤُلَآءِ وَلَآ إِلَىٰ هَٰٓؤُلَآءِ ۚ وَمَن يُضْلِلِ ٱللَّهُ فَلَن تَجِدَ لَهُۥ سَبِيلًۭا ﴿١٤٣﴾ هغوی د دې (کفر او ایمان) ترمنځ په تردد کې دي، نه د دې (مسلمانانو) لوري ته دي او نه د هغو (کافرانو) لوري ته؛ او څوک چې الله ګمراه کړي، بیا به تاسو ورته (د هدایت) کومه لاره ونه مومئ.
p يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ لَا تَتَّخِذُوا۟ ٱلْكَٰفِرِينَ أَوْلِيَآءَ مِن دُونِ ٱلْمُؤْمِنِينَ ۚ أَتُرِيدُونَ أَن تَجْعَلُوا۟ لِلَّهِ عَلَيْكُمْ سُلْطَٰنًۭا مُّبِينًا ﴿١٤٤﴾ اے ایمان والو! مومنوں کو چھوڑ کر کافروں کو دوست نہ بناؤ کیا تم چاہتے ہو کہ اللہ کے لیے اپنے خلاف ایک واضح دلیل بنا لو۔ || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ لَا تَتَّخِذُوا۟ ٱلْكَٰفِرِينَ أَوْلِيَآءَ مِن دُونِ ٱلْمُؤْمِنِينَ ۚ أَتُرِيدُونَ أَن تَجْعَلُوا۟ لِلَّهِ عَلَيْكُمْ سُلْطَٰنًۭا مُّبِينًا ﴿١٤٤﴾ O ye who believe! Choose not disbelievers for (your) friends in place of believers. Would ye give Allah a clear warrant against you? (4:144) || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ لَا تَتَّخِذُوا۟ ٱلْكَٰفِرِينَ أَوْلِيَآءَ مِن دُونِ ٱلْمُؤْمِنِينَ ۚ أَتُرِيدُونَ أَن تَجْعَلُوا۟ لِلَّهِ عَلَيْكُمْ سُلْطَٰنًۭا مُّبِينًا ﴿١٤٤﴾ ای مؤمنانو! مؤمنان پرېږدئ او کافران دوستان مه نیسئ؛ ایا غواړئ چې د الله لپاره د ځان پر خلاف یو ښکاره دلیل جوړ کړئ؟
