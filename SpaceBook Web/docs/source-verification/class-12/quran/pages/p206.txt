p وَءَاخَرِينَ مِنْهُمْ لَمَّا يَلْحَقُوا۟ بِهِمْ ۚ وَهُوَ ٱلْعَزِيزُ ٱلْحَكِيمُ ﴿٣﴾ اور (ان) دوسرے لوگوں کے لیے بھی (مبعوث ہوئے) جو ابھی اُن سے نہیں ملے اور وہ بہت غالب بڑی حکمت والا ہے۔ || وَءَاخَرِينَ مِنْهُمْ لَمَّا يَلْحَقُوا۟ بِهِمْ ۚ وَهُوَ ٱلْعَزِيزُ ٱلْحَكِيمُ ﴿٣﴾ Along with others of them who have not yet joined them. He is the Mighty, the Wise. (62:3) || وَءَاخَرِينَ مِنْهُمْ لَمَّا يَلْحَقُوا۟ بِهِمْ ۚ وَهُوَ ٱلْعَزِيزُ ٱلْحَكِيمُ ﴿٣﴾ او د (هغو) نورو خلکو لپاره هم (لېږل شوی) چې لا له دوی سره نه دي یوځای شوي؛ او هغه ډېر غالب، ډېر حکمتناک دی.
p ذَٰلِكَ فَضْلُ ٱللَّهِ يُؤْتِيهِ مَن يَشَآءُ ۚ وَٱللَّهُ ذُو ٱلْفَضْلِ ٱلْعَظِيمِ ﴿٤﴾ یہ اللہ کا فضل ہے وہ اسے عطا فرماتا ہے جسے چاہتا ہے اور اللہ بہت بڑے فضل والا ہے۔ || ذَٰلِكَ فَضْلُ ٱللَّهِ يُؤْتِيهِ مَن يَشَآءُ ۚ وَٱللَّهُ ذُو ٱلْفَضْلِ ٱلْعَظِيمِ ﴿٤﴾ That is the bounty of Allah; which He giveth unto whom He will. Allah is of Infinite Bounty. (62:4) || ذَٰلِكَ فَضْلُ ٱللَّهِ يُؤْتِيهِ مَن يَشَآءُ ۚ وَٱللَّهُ ذُو ٱلْفَضْلِ ٱلْعَظِيمِ ﴿٤﴾ دا د الله فضل دی، چا ته چې وغواړي ورکوي یې؛ او الله د ډېر لوی فضل خاوند دی.
p مَثَلُ ٱلَّذِينَ حُمِّلُوا۟ ٱلتَّوْرَىٰةَ ثُمَّ لَمْ يَحْمِلُوهَا كَمَثَلِ ٱلْحِمَارِ يَحْمِلُ أَسْفَارًۢا ۚ بِئْسَ مَثَلُ ٱلْقَوْمِ ٱلَّذِينَ كَذَّبُوا۟ بِـَٔايَٰتِ ٱللَّهِ ۚ وَٱللَّهُ لَا يَهْدِى ٱلْقَوْمَ ٱلظَّٰلِمِينَ ﴿٥﴾ ان (لوگوں) کی مثال جن پر تورات کی ذمہ داری ڈالی گئی تھی پھر انھوں نے اس کو نہیں نبھایا اس گدھے کی سی ہے جو پیٹھ پر بہت سی کتابیں لادے ہوئے ہو بہت بری ہے ان لوگوں کی مثال جنھوں نے اللہ کی آیات کو جھٹلایا اور اللہ ظالم قوم کو ہدایت (کی توفیق) نہیں دیتا۔ || مَثَلُ ٱلَّذِينَ حُمِّلُوا۟ ٱلتَّوْرَىٰةَ ثُمَّ لَمْ يَحْمِلُوهَا كَمَثَلِ ٱلْحِمَارِ يَحْمِلُ أَسْفَارًۢا ۚ بِئْسَ مَثَلُ ٱلْقَوْمِ ٱلَّذِينَ كَذَّبُوا۟ بِـَٔايَٰتِ ٱللَّهِ ۚ وَٱللَّهُ لَا يَهْدِى ٱلْقَوْمَ ٱلظَّٰلِمِينَ ﴿٥﴾ The likeness of those who are entrusted with the Law of Moses, yet apply it not, is as the likeness of the ass carrying books. Wretched is the likeness of folk who deny the revelations of Allah. And Allah guideth not wrongdoing folk. (62:5) || مَثَلُ ٱلَّذِينَ حُمِّلُوا۟ ٱلتَّوْرَىٰةَ ثُمَّ لَمْ يَحْمِلُوهَا كَمَثَلِ ٱلْحِمَارِ يَحْمِلُ أَسْفَارًۢا ۚ بِئْسَ مَثَلُ ٱلْقَوْمِ ٱلَّذِينَ كَذَّبُوا۟ بِـَٔايَٰتِ ٱللَّهِ ۚ وَٱللَّهُ لَا يَهْدِى ٱلْقَوْمَ ٱلظَّٰلِمِينَ ﴿٥﴾ د هغو (خلکو) مثال چې د تورات مسؤلیت پرې اچول شوی و، بیا یې ونه پالو، د هغه خره په څېر دی چې پر شا ډېر کتابونه بار وي؛ د هغو خلکو مثال ډېر بد دی چې د الله آیتونه یې دروغ وګڼل؛ او الله ظالم قوم ته د هدایت (توفیق) نه ورکوي.
p قُلْ يَٰٓأَيُّهَا ٱلَّذِينَ هَادُوٓا۟ إِن زَعَمْتُمْ أَنَّكُمْ أَوْلِيَآءُ لِلَّهِ مِن دُونِ ٱلنَّاسِ فَتَمَنَّوُا۟ ٱلْمَوْتَ إِن كُنتُمْ صَٰدِقِينَ ﴿٦﴾ آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) فرما دیجیے اے وہ لوگو! جو یہودی بن گئے ہو اگر تمھارا یہ دعویٰ ہے کہ سارے لوگوں کو چھوڑ کر بس تم ہی اللہ کے دوست ہو تو (ذرا) موت کی تمنا تو کرو اگر تم سچے ہو۔ || قُلْ يَٰٓأَيُّهَا ٱلَّذِينَ هَادُوٓا۟ إِن زَعَمْتُمْ أَنَّكُمْ أَوْلِيَآءُ لِلَّهِ مِن دُونِ ٱلنَّاسِ فَتَمَنَّوُا۟ ٱلْمَوْتَ إِن كُنتُمْ صَٰدِقِينَ ﴿٦﴾ Say (O Muhammad): O ye who are Jews! If ye claim that ye are favoured of Allah apart from (all) mankind, then long for death if ye are truthful. (62:6) || قُلْ يَٰٓأَيُّهَا ٱلَّذِينَ هَادُوٓا۟ إِن زَعَمْتُمْ أَنَّكُمْ أَوْلِيَآءُ لِلَّهِ مِن دُونِ ٱلنَّاسِ فَتَمَنَّوُا۟ ٱلْمَوْتَ إِن كُنتُمْ صَٰدِقِينَ ﴿٦﴾ تاسو (صلی الله علیه وعلی آله واصحابه وسلم) ووایئ: ای هغو خلکو چې یهودیان شوي یاست! که ستاسو دا دعوه وي چې له ټولو خلکو پرته یوازې تاسو د الله دوستان یاست، نو (لږ) د مرګ هیله وکړئ که رښتیني یاست.
p وَلَا يَتَمَنَّوْنَهُۥٓ أَبَدًۢا بِمَا قَدَّمَتْ أَيْدِيهِمْ ۚ وَٱللَّهُ عَلِيمٌۢ بِٱلظَّٰلِمِينَ ﴿٧﴾ اور وہ کبھی بھی اس کی تمنا نہیں کریں گے ان (گناہوں) کی وجہ سے جو وہ اپنے ہاتھوں آگے بھیج چکے ہیں اور اللہ ظالموں کو خوب جاننے والا ہے۔ || وَلَا يَتَمَنَّوْنَهُۥٓ أَبَدًۢا بِمَا قَدَّمَتْ أَيْدِيهِمْ ۚ وَٱللَّهُ عَلِيمٌۢ بِٱلظَّٰلِمِينَ ﴿٧﴾ But they will never long for it because of all that their own hands have sent before, and Allah is Aware of evil-doers. (62:7) || وَلَا يَتَمَنَّوْنَهُۥٓ أَبَدًۢا بِمَا قَدَّمَتْ أَيْدِيهِمْ ۚ وَٱللَّهُ عَلِيمٌۢ بِٱلظَّٰلِمِينَ ﴿٧﴾ او هغوی به هېڅکله د هغه هیله ونه کړي د هغو (ګناهونو) له امله چې په خپلو لاسونو یې مخکې لېږلي؛ او الله په ظالمانو ښه پوه دی.
p قُلْ إِنَّ ٱلْمَوْتَ ٱلَّذِى تَفِرُّونَ مِنْهُ فَإِنَّهُۥ مُلَٰقِيكُمْ ۖ ثُمَّ تُرَدُّونَ إِلَىٰ عَٰلِمِ ٱلْغَيْبِ وَٱلشَّهَٰدَةِ فَيُنَبِّئُكُم بِمَا كُنتُمْ تَعْمَلُونَ ﴿٨﴾ آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) فرما دیجیے بے شک وہ موت جس سے تم بھاگتے ہو تو وہ ضرور تم سے آ ملنے والی ہے پھر تم (ہر) پوشیدہ اور ظاہر کے جاننے والے (اللہ) کی طرف لوٹائے جاؤ گے پھر وہ تمھیں بتا دے گا جو کچھ تم کیا کرتے تھے۔ || قُلْ إِنَّ ٱلْمَوْتَ ٱلَّذِى تَفِرُّونَ مِنْهُ فَإِنَّهُۥ مُلَٰقِيكُمْ ۖ ثُمَّ تُرَدُّونَ إِلَىٰ عَٰلِمِ ٱلْغَيْبِ وَٱلشَّهَٰدَةِ فَيُنَبِّئُكُم بِمَا كُنتُمْ تَعْمَلُونَ ﴿٨﴾ Say (unto them, O Muhammad): Lo! the death from which ye shrink will surely meet you, and afterward ye will be returned unto the Knower of the Invisible and the Visible, and He will tell you what ye used to do. (62:8) || قُلْ إِنَّ ٱلْمَوْتَ ٱلَّذِى تَفِرُّونَ مِنْهُ فَإِنَّهُۥ مُلَٰقِيكُمْ ۖ ثُمَّ تُرَدُّونَ إِلَىٰ عَٰلِمِ ٱلْغَيْبِ وَٱلشَّهَٰدَةِ فَيُنَبِّئُكُم بِمَا كُنتُمْ تَعْمَلُونَ ﴿٨﴾ تاسو (صلی الله علیه وعلی آله واصحابه وسلم) ووایئ: بې‌شکه هغه مرګ چې تاسو ترې تښتئ، هغه به خامخا له تاسو سره مخ شي؛ بیا به د (هر) پټ او ښکاره پوهېدونکي (الله) لوري ته ورګرځول شئ، بیا به تاسو ته ووايي چې څه به مو کول.
p يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓا۟ إِذَا نُودِىَ لِلصَّلَوٰةِ مِن يَوْمِ ٱلْجُمُعَةِ فَٱسْعَوْا۟ إِلَىٰ ذِكْرِ ٱللَّهِ وَذَرُوا۟ ٱلْبَيْعَ ۚ ذَٰلِكُمْ خَيْرٌۭ لَّكُمْ إِن كُنتُمْ تَعْلَمُونَ ﴿٩﴾ اے ایمان والو! جب جمعہ کے دن نماز (جمعہ) کے لیے اذان دی جائے تو اللہ کے ذکر کی طرف (جلدی) چل پڑو اور خرید و فروخت چھوڑ دو یہ تمھارے لیے بہتر ہے اگر تم سمجھو۔ || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓا۟ إِذَا نُودِىَ لِلصَّلَوٰةِ مِن يَوْمِ ٱلْجُمُعَةِ فَٱسْعَوْا۟ إِلَىٰ ذِكْرِ ٱللَّهِ وَذَرُوا۟ ٱلْبَيْعَ ۚ ذَٰلِكُمْ خَيْرٌۭ لَّكُمْ إِن كُنتُمْ تَعْلَمُونَ ﴿٩﴾ O ye who believe! When the call is heard for the prayer of the day of congregation, haste unto remembrance of Allah and leave your trading. That is better for you if ye did but know. (62:9) || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓا۟ إِذَا نُودِىَ لِلصَّلَوٰةِ مِن يَوْمِ ٱلْجُمُعَةِ فَٱسْعَوْا۟ إِلَىٰ ذِكْرِ ٱللَّهِ وَذَرُوا۟ ٱلْبَيْعَ ۚ ذَٰلِكُمْ خَيْرٌۭ لَّكُمْ إِن كُنتُمْ تَعْلَمُونَ ﴿٩﴾ ای مؤمنانو! کله چې د جمعې په ورځ د (جمعې) لمانځه لپاره اذان وشي، نو د الله ذکر ته (ژر) روان شئ او پېر و پلور پرېږدئ؛ دا ستاسو لپاره غوره ده که پوه شئ.
p فَإِذَا قُضِيَتِ ٱلصَّلَوٰةُ فَٱنتَشِرُوا۟ فِى ٱلْأَرْضِ وَٱبْتَغُوا۟ مِن فَضْلِ ٱللَّهِ وَٱذْكُرُوا۟ ٱللَّهَ كَثِيرًۭا لَّعَلَّكُمْ تُفْلِحُونَ ﴿١٠﴾ پھر جب نماز ادا ہو چکے تو (اپنے کاموں کی طرف) زمین میں پھیل جاؤ اور اللہ کا فضل تلاش کرو اور کثرت سے اللہ کو یاد کرو تاکہ تم فلاح پاؤ۔ || فَإِذَا قُضِيَتِ ٱلصَّلَوٰةُ فَٱنتَشِرُوا۟ فِى ٱلْأَرْضِ وَٱبْتَغُوا۟ مِن فَضْلِ ٱللَّهِ وَٱذْكُرُوا۟ ٱللَّهَ كَثِيرًۭا لَّعَلَّكُمْ تُفْلِحُونَ ﴿١٠﴾ And when the prayer is ended, then disperse in the land and seek of Allah's bounty, and remember Allah much, that ye may be successful. (62:10) || فَإِذَا قُضِيَتِ ٱلصَّلَوٰةُ فَٱنتَشِرُوا۟ فِى ٱلْأَرْضِ وَٱبْتَغُوا۟ مِن فَضْلِ ٱللَّهِ وَٱذْكُرُوا۟ ٱللَّهَ كَثِيرًۭا لَّعَلَّكُمْ تُفْلِحُونَ ﴿١٠﴾ بیا کله چې لمونځ ادا شي، نو (خپلو کارونو ته) په ځمکه کې خپاره شئ او د الله فضل ولټوئ، او الله ډېر یاد کړئ تر څو بریالي شئ.
p وَإِذَا رَأَوْا۟ تِجَٰرَةً أَوْ لَهْوًا ٱنفَضُّوٓا۟ إِلَيْهَا وَتَرَكُوكَ قَآئِمًۭا ۚ قُلْ مَا عِندَ ٱللَّهِ خَيْرٌۭ مِّنَ ٱللَّهْوِ وَمِنَ ٱلتِّجَٰرَةِ ۚ وَٱللَّهُ خَيْرُ ٱلرَّٰزِقِينَ ﴿١١﴾ اور جب (کچھ) لوگوں نے کوئی تجارت یا کوئی کھیل (تماشا) دیکھا تو اُس کی طرف تیزی سے چل دیے اور آپ کو (خطبہ میں) کھڑا ہوا چھوڑ گئے آپ فرما دیجیے کہ جو کچھ اللہ کے پاس ہے وہ کھیل (تماشے) اور تجارت سے کہیں بہتر ہے اور اللہ سب سے بہتر رزق عطا فرمانے والا ہے۔ || وَإِذَا رَأَوْا۟ تِجَٰرَةً أَوْ لَهْوًا ٱنفَضُّوٓا۟ إِلَيْهَا وَتَرَكُوكَ قَآئِمًۭا ۚ قُلْ مَا عِندَ ٱللَّهِ خَيْرٌۭ مِّنَ ٱللَّهْوِ وَمِنَ ٱلتِّجَٰرَةِ ۚ وَٱللَّهُ خَيْرُ ٱلرَّٰزِقِينَ ﴿١١﴾ But when they spy some merchandise or pastime they break away to it and leave thee standing. Say: That which Allah hath is better than pastime and than merchandise, and Allah is the Best of providers. (62:11) || وَإِذَا رَأَوْا۟ تِجَٰرَةً أَوْ لَهْوًا ٱنفَضُّوٓا۟ إِلَيْهَا وَتَرَكُوكَ قَآئِمًۭا ۚ قُلْ مَا عِندَ ٱللَّهِ خَيْرٌۭ مِّنَ ٱللَّهْوِ وَمِنَ ٱلتِّجَٰرَةِ ۚ وَٱللَّهُ خَيْرُ ٱلرَّٰزِقِينَ ﴿١١﴾ او کله چې (ځینو) خلکو کومه سوداګري یا کومه لوبه (تماشه) ولیده، نو په چټکۍ ورپسې لاړل او تاسو یې (په خطبه کې) ولاړ پرېښودئ؛ تاسو ووایئ: څه چې له الله سره دي، هغه له لوبې (تماشې) او سوداګرۍ ډېر غوره دي؛ او الله تر ټولو غوره رزق ورکوونکی دی.
