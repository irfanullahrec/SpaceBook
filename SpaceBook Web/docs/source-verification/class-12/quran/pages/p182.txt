p أَلَمْ تَرَ إِلَى ٱلَّذِينَ نُهُوا۟ عَنِ ٱلنَّجْوَىٰ ثُمَّ يَعُودُونَ لِمَا نُهُوا۟ عَنْهُ وَيَتَنَٰجَوْنَ بِٱلْإِثْمِ وَٱلْعُدْوَٰنِ وَمَعْصِيَتِ ٱلرَّسُولِ وَإِذَا جَآءُوكَ حَيَّوْكَ بِمَا لَمْ يُحَيِّكَ بِهِ ٱللَّهُ وَيَقُولُونَ فِىٓ أَنفُسِهِمْ لَوْلَا يُعَذِّبُنَا ٱللَّهُ بِمَا نَقُولُ ۚ حَسْبُهُمْ جَهَنَّمُ يَصْلَوْنَهَا ۖ فَبِئْسَ ٱلْمَصِيرُ ﴿٨﴾ کیا آپ نے ان لوگوں کو نہیں دیکھا جنھیں (بُری) سرگوشی سے منع کیا گیا تھا؟ پھر وہ وہی کر رہے ہیں جس سے انھیں منع کیا گیا تھا اور وہ آپس میں گناہ اور زیادتی اور رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کی نافرمانی کے لیے سرگوشی کرتے ہیں اور جب وہ آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کے پاس آتے ہیں تو آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کو ایسے (طریقہ اور) لفظوں سے سلام کہتے ہیں جن سے اللہ نے آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کو سلام نہیں کہا اور اپنے دلوں میں کہتے ہیں کہ اللہ ہمیں اس پر عذاب کیوں نہیں دیتا جو ہم کہتے ہیں اُن کے لیے جہنم ہی کافی ہے وہ اُس میں جھلسائے جائیں گے تو وہ بہت برا ٹھکانا ہے۔ || أَلَمْ تَرَ إِلَى ٱلَّذِينَ نُهُوا۟ عَنِ ٱلنَّجْوَىٰ ثُمَّ يَعُودُونَ لِمَا نُهُوا۟ عَنْهُ وَيَتَنَٰجَوْنَ بِٱلْإِثْمِ وَٱلْعُدْوَٰنِ وَمَعْصِيَتِ ٱلرَّسُولِ وَإِذَا جَآءُوكَ حَيَّوْكَ بِمَا لَمْ يُحَيِّكَ بِهِ ٱللَّهُ وَيَقُولُونَ فِىٓ أَنفُسِهِمْ لَوْلَا يُعَذِّبُنَا ٱللَّهُ بِمَا نَقُولُ ۚ حَسْبُهُمْ جَهَنَّمُ يَصْلَوْنَهَا ۖ فَبِئْسَ ٱلْمَصِيرُ ﴿٨﴾ Hast thou not observed those who were forbidden conspiracy and afterward returned to that which they had been forbidden, and (now) conspire together for crime and wrongdoing and disobedience toward the messenger? And when they come unto thee they greet thee with a greeting wherewith Allah greeteth thee not, and say within themselves: Why should Allah punish us for what we say? Hell will suffice them; they will feel the heat thereof - a hapless journey's end! (58:8) || أَلَمْ تَرَ إِلَى ٱلَّذِينَ نُهُوا۟ عَنِ ٱلنَّجْوَىٰ ثُمَّ يَعُودُونَ لِمَا نُهُوا۟ عَنْهُ وَيَتَنَٰجَوْنَ بِٱلْإِثْمِ وَٱلْعُدْوَٰنِ وَمَعْصِيَتِ ٱلرَّسُولِ وَإِذَا جَآءُوكَ حَيَّوْكَ بِمَا لَمْ يُحَيِّكَ بِهِ ٱللَّهُ وَيَقُولُونَ فِىٓ أَنفُسِهِمْ لَوْلَا يُعَذِّبُنَا ٱللَّهُ بِمَا نَقُولُ ۚ حَسْبُهُمْ جَهَنَّمُ يَصْلَوْنَهَا ۖ فَبِئْسَ ٱلْمَصِيرُ ﴿٨﴾ ایا تاسو هغه کسان نه دي لیدلي چې له (بدې) پټې مشورې منع شوي وو؟ بیا هم هغه څه کوي چې ترې منع شوي وو، او په خپل منځ کې د ګناه، زیاتي او د رسول (صلی الله علیه وعلی آله واصحابه وسلم) د نافرمانۍ لپاره پټې مشورې کوي؛ او کله چې تاسو (صلی الله علیه وعلی آله واصحابه وسلم) ته راشي، نو تاسو (صلی الله علیه وعلی آله واصحابه وسلم) ته په داسې (طریقه او) لفظونو سلام کوي چې الله پرې تاسو ته سلام نه دی کړی، او په خپلو زړونو کې وايي: الله موږ ته په هغه څه چې وایو ولې عذاب نه راکوي؟ د هغوی لپاره دوزخ بس دی، په هغه کې به وسوځول شي؛ نو هغه ډېر بد ځای دی.
p يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓا۟ إِذَا تَنَٰجَيْتُمْ فَلَا تَتَنَٰجَوْا۟ بِٱلْإِثْمِ وَٱلْعُدْوَٰنِ وَمَعْصِيَتِ ٱلرَّسُولِ وَتَنَٰجَوْا۟ بِٱلْبِرِّ وَٱلتَّقْوَىٰ ۖ وَٱتَّقُوا۟ ٱللَّهَ ٱلَّذِىٓ إِلَيْهِ تُحْشَرُونَ ﴿٩﴾ اے ایمان والو! جب تم آپس میں سرگوشی کرو تو گناہ اور زیادتی اور رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کی نافرمانی کے لیے سرگوشی نہ کرو بلکہ نیکی اور پرہیزگاری کے لیے سرگوشی کرو اور اُس اللہ (کی نافرمانی) سے ڈرو جس کی طرف تمھیں جمع کیا جائے گا۔ || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓا۟ إِذَا تَنَٰجَيْتُمْ فَلَا تَتَنَٰجَوْا۟ بِٱلْإِثْمِ وَٱلْعُدْوَٰنِ وَمَعْصِيَتِ ٱلرَّسُولِ وَتَنَٰجَوْا۟ بِٱلْبِرِّ وَٱلتَّقْوَىٰ ۖ وَٱتَّقُوا۟ ٱللَّهَ ٱلَّذِىٓ إِلَيْهِ تُحْشَرُونَ ﴿٩﴾ O ye who believe! When ye conspire together, conspire not together for crime and wrongdoing and disobedience toward the messenger, but conspire together for righteousness and piety, and keep your duty toward Allah, unto whom ye will be gathered. (58:9) || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓا۟ إِذَا تَنَٰجَيْتُمْ فَلَا تَتَنَٰجَوْا۟ بِٱلْإِثْمِ وَٱلْعُدْوَٰنِ وَمَعْصِيَتِ ٱلرَّسُولِ وَتَنَٰجَوْا۟ بِٱلْبِرِّ وَٱلتَّقْوَىٰ ۖ وَٱتَّقُوا۟ ٱللَّهَ ٱلَّذِىٓ إِلَيْهِ تُحْشَرُونَ ﴿٩﴾ ای مؤمنانو! کله چې په خپل منځ کې پټې مشورې کوئ، نو د ګناه، زیاتي او د رسول (صلی الله علیه وعلی آله واصحابه وسلم) د نافرمانۍ لپاره پټې مشورې مه کوئ، بلکې د نېکۍ او پرهېزګارۍ لپاره پټې مشورې وکړئ؛ او له هغه الله (د نافرمانۍ) ووېرېږئ چې د هغه لوري ته به راټول کړای شئ.
p إِنَّمَا ٱلنَّجْوَىٰ مِنَ ٱلشَّيْطَٰنِ لِيَحْزُنَ ٱلَّذِينَ ءَامَنُوا۟ وَلَيْسَ بِضَآرِّهِمْ شَيْـًٔا إِلَّا بِإِذْنِ ٱللَّهِ ۚ وَعَلَى ٱللَّهِ فَلْيَتَوَكَّلِ ٱلْمُؤْمِنُونَ ﴿١٠﴾ (بُری) سرگوشی تو شیطان ہی کی طرف سے ہے تاکہ وہ ایمان والوں کو غمگین کرے اور وہ اللہ کے اذن کے بغیر انھیں کچھ بھی نقصان نہیں پہنچا سکتا اور ایمان والوں کو اللہ ہی پر بھروسہ کرنا چاہیے۔ || إِنَّمَا ٱلنَّجْوَىٰ مِنَ ٱلشَّيْطَٰنِ لِيَحْزُنَ ٱلَّذِينَ ءَامَنُوا۟ وَلَيْسَ بِضَآرِّهِمْ شَيْـًٔا إِلَّا بِإِذْنِ ٱللَّهِ ۚ وَعَلَى ٱللَّهِ فَلْيَتَوَكَّلِ ٱلْمُؤْمِنُونَ ﴿١٠﴾ Lo! Conspiracy is only of the devil, that he may vex those who believe; but he can harm them not at all unless by Allah's leave. In Allah let believers put their trust. (58:10) || إِنَّمَا ٱلنَّجْوَىٰ مِنَ ٱلشَّيْطَٰنِ لِيَحْزُنَ ٱلَّذِينَ ءَامَنُوا۟ وَلَيْسَ بِضَآرِّهِمْ شَيْـًٔا إِلَّا بِإِذْنِ ٱللَّهِ ۚ وَعَلَى ٱللَّهِ فَلْيَتَوَكَّلِ ٱلْمُؤْمِنُونَ ﴿١٠﴾ (بده) پټه مشوره خو یوازې د شیطان له لوري ده تر څو مؤمنان خپه کړي؛ او هغه د الله له اجازې پرته هغوی ته هېڅ زیان نه شي رسولی؛ او مؤمنان باید یوازې پر الله توکل وکړي.
p يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓا۟ إِذَا قِيلَ لَكُمْ تَفَسَّحُوا۟ فِى ٱلْمَجَٰلِسِ فَٱفْسَحُوا۟ يَفْسَحِ ٱللَّهُ لَكُمْ ۖ وَإِذَا قِيلَ ٱنشُزُوا۟ فَٱنشُزُوا۟ يَرْفَعِ ٱللَّهُ ٱلَّذِينَ ءَامَنُوا۟ مِنكُمْ وَٱلَّذِينَ أُوتُوا۟ ٱلْعِلْمَ دَرَجَٰتٍۢ ۚ وَٱللَّهُ بِمَا تَعْمَلُونَ خَبِيرٌۭ ﴿١١﴾ اے ایمان والو! جب تم سے کہا جائے کہ مجلسوں میں کشادگی پیدا کرو تو کشادگی پیدا کر دیا کرو اللہ تمھیں کشادگی عطا فرمائے گا اور جب کہا جائے اُٹھ جاؤ تو اُٹھ جایا کرو اللہ ان لوگوں کے درجات بلند فرما دے گا جو تم میں سے (کامل) ایمان لائے اور جنھیں علم دیا گیا اور اللہ تمھارے تمام اعمال سے خوب خبردار ہے۔ || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓا۟ إِذَا قِيلَ لَكُمْ تَفَسَّحُوا۟ فِى ٱلْمَجَٰلِسِ فَٱفْسَحُوا۟ يَفْسَحِ ٱللَّهُ لَكُمْ ۖ وَإِذَا قِيلَ ٱنشُزُوا۟ فَٱنشُزُوا۟ يَرْفَعِ ٱللَّهُ ٱلَّذِينَ ءَامَنُوا۟ مِنكُمْ وَٱلَّذِينَ أُوتُوا۟ ٱلْعِلْمَ دَرَجَٰتٍۢ ۚ وَٱللَّهُ بِمَا تَعْمَلُونَ خَبِيرٌۭ ﴿١١﴾ O ye who believe! when it is said unto you, Make room! in assemblies, then make room; Allah will make way for you (hereafter). And when it is said, Come up higher! go up higher; Allah will exalt those who believe among you, and those who have knowledge, to high ranks. Allah is Informed of what ye do. (58:11) || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓا۟ إِذَا قِيلَ لَكُمْ تَفَسَّحُوا۟ فِى ٱلْمَجَٰلِسِ فَٱفْسَحُوا۟ يَفْسَحِ ٱللَّهُ لَكُمْ ۖ وَإِذَا قِيلَ ٱنشُزُوا۟ فَٱنشُزُوا۟ يَرْفَعِ ٱللَّهُ ٱلَّذِينَ ءَامَنُوا۟ مِنكُمْ وَٱلَّذِينَ أُوتُوا۟ ٱلْعِلْمَ دَرَجَٰتٍۢ ۚ وَٱللَّهُ بِمَا تَعْمَلُونَ خَبِيرٌۭ ﴿١١﴾ ای مؤمنانو! کله چې درته وویل شي چې په مجلسونو کې ځای پراخ کړئ، نو ځای پراخ کړئ، الله به تاسو ته پراخي درکړي؛ او کله چې وویل شي پاڅېږئ، نو پاڅېږئ؛ الله به د هغو کسانو درجې لوړې کړي چې له تاسو یې (بشپړ) ایمان راوړی او هغه کسان چې علم ورکړل شوی؛ او الله ستاسو په ټولو عملونو ښه خبر دی.
p يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓا۟ إِذَا نَٰجَيْتُمُ ٱلرَّسُولَ فَقَدِّمُوا۟ بَيْنَ يَدَىْ نَجْوَىٰكُمْ صَدَقَةًۭ ۚ ذَٰلِكَ خَيْرٌۭ لَّكُمْ وَأَطْهَرُ ۚ فَإِن لَّمْ تَجِدُوا۟ فَإِنَّ ٱللَّهَ غَفُورٌۭ رَّحِيمٌ ﴿١٢﴾ اے ایمان والو! جب تم رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) سے سرگوشی کرو تو اپنی سرگوشی سے پہلے صدقہ پیش کرو یہ تمھارے لیے زیادہ اچھا اور بہت پاکیزہ ہے پھر اگر تم (صدقہ کے لیے کچھ) نہ پاؤ تو بے شک اللہ بڑا ہی بخشنے والا نہایت رحم فرمانے والا ہے۔ || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓا۟ إِذَا نَٰجَيْتُمُ ٱلرَّسُولَ فَقَدِّمُوا۟ بَيْنَ يَدَىْ نَجْوَىٰكُمْ صَدَقَةًۭ ۚ ذَٰلِكَ خَيْرٌۭ لَّكُمْ وَأَطْهَرُ ۚ فَإِن لَّمْ تَجِدُوا۟ فَإِنَّ ٱللَّهَ غَفُورٌۭ رَّحِيمٌ ﴿١٢﴾ O ye who believe! When ye hold conference with the messenger, offer an alms before your conference. That is better and purer for you. But if ye cannot find (the wherewithal) then lo! Allah is Forgiving, Merciful. (58:12) || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓا۟ إِذَا نَٰجَيْتُمُ ٱلرَّسُولَ فَقَدِّمُوا۟ بَيْنَ يَدَىْ نَجْوَىٰكُمْ صَدَقَةًۭ ۚ ذَٰلِكَ خَيْرٌۭ لَّكُمْ وَأَطْهَرُ ۚ فَإِن لَّمْ تَجِدُوا۟ فَإِنَّ ٱللَّهَ غَفُورٌۭ رَّحِيمٌ ﴿١٢﴾ ای مؤمنانو! کله چې له رسول (صلی الله علیه وعلی آله واصحابه وسلم) سره پټې خبرې کوئ، نو له خپلو پټو خبرو مخکې صدقه وړاندې کړئ؛ دا ستاسو لپاره ډېره غوره او ډېره پاکه ده؛ بیا که (د صدقې لپاره څه) ونه مومئ، نو بې‌شکه الله ډېر بښونکی، بې‌حده مهربان دی.
