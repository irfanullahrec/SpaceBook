p يَحْسَبُونَ ٱلْأَحْزَابَ لَمْ يَذْهَبُوا۟ ۖ وَإِن يَأْتِ ٱلْأَحْزَابُ يَوَدُّوا۟ لَوْ أَنَّهُم بَادُونَ فِى ٱلْأَعْرَابِ يَسْـَٔلُونَ عَنْ أَنۢبَآئِكُمْ ۖ وَلَوْ كَانُوا۟ فِيكُم مَّا قَٰتَلُوٓا۟ إِلَّا قَلِيلًۭا ﴿٢٠﴾ یہ (منافق دشمن کے بھاگ جانے کے بعد بھی) سمجھ رہے ہیں کہ (کافروں کے) لشکر ابھی نہیں گئے اور اگر وہ لشکر (دوبارہ) آ جائیں تو یہ چاہیں گے کہ کاش وہ دیہاتوں میں جا کر رہیں (اور وہیں بیٹھے) تمھاری خبریں پوچھتے رہیں اور اگر وہ تمھارے درمیان موجود ہوتے تو وہ قتال نہ کرتے مگر برائے نام۔ || يَحْسَبُونَ ٱلْأَحْزَابَ لَمْ يَذْهَبُوا۟ ۖ وَإِن يَأْتِ ٱلْأَحْزَابُ يَوَدُّوا۟ لَوْ أَنَّهُم بَادُونَ فِى ٱلْأَعْرَابِ يَسْـَٔلُونَ عَنْ أَنۢبَآئِكُمْ ۖ وَلَوْ كَانُوا۟ فِيكُم مَّا قَٰتَلُوٓا۟ إِلَّا قَلِيلًۭا ﴿٢٠﴾ They hold that the clans have not retired (for good); and if the clans should advance (again), they would fain be in the desert with the wandering Arabs, asking for the news of you; and if they were among you, they would not give battle, save a little. (33:20) || يَحْسَبُونَ ٱلْأَحْزَابَ لَمْ يَذْهَبُوا۟ ۖ وَإِن يَأْتِ ٱلْأَحْزَابُ يَوَدُّوا۟ لَوْ أَنَّهُم بَادُونَ فِى ٱلْأَعْرَابِ يَسْـَٔلُونَ عَنْ أَنۢبَآئِكُمْ ۖ وَلَوْ كَانُوا۟ فِيكُم مَّا قَٰتَلُوٓا۟ إِلَّا قَلِيلًۭا ﴿٢٠﴾ دا (منافقان د دښمن له تېښتې وروسته هم) ګڼي چې د (کافرانو) لښکرې لا نه دي تللې؛ او که هغه لښکرې (بیا) راشي، نو دوی به هیله وکړي چې کاش په کلیو کې اوسېدلي وای (او هلته ناست) ستاسو خبرونه یې پوښتلي وای؛ او که ستاسو په منځ کې وای هم، نو جګړه به یې نه کوله مګر په نامه.
p لَّقَدْ كَانَ لَكُمْ فِى رَسُولِ ٱللَّهِ أُسْوَةٌ حَسَنَةٌۭ لِّمَن كَانَ يَرْجُوا۟ ٱللَّهَ وَٱلْيَوْمَ ٱلْءَاخِرَ وَذَكَرَ ٱللَّهَ كَثِيرًۭا ﴿٢١﴾ یقیناً تمھارے لیے اللہ کے رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم کی ذاتِ مبارکہ) میں بہترین نمونہ ہے اُس کے لیے جو اللہ (سے ملاقات) اور یوم آخرت کی امید رکھتا ہو اور کثرت سے اللہ کا ذکر کرتا ہو۔ || لَّقَدْ كَانَ لَكُمْ فِى رَسُولِ ٱللَّهِ أُسْوَةٌ حَسَنَةٌۭ لِّمَن كَانَ يَرْجُوا۟ ٱللَّهَ وَٱلْيَوْمَ ٱلْءَاخِرَ وَذَكَرَ ٱللَّهَ كَثِيرًۭا ﴿٢١﴾ Verily in the messenger of Allah ye have a good example for him who looketh unto Allah and the Last Day, and remembereth Allah much. (33:21) || لَّقَدْ كَانَ لَكُمْ فِى رَسُولِ ٱللَّهِ أُسْوَةٌ حَسَنَةٌۭ لِّمَن كَانَ يَرْجُوا۟ ٱللَّهَ وَٱلْيَوْمَ ٱلْءَاخِرَ وَذَكَرَ ٱللَّهَ كَثِيرًۭا ﴿٢١﴾ یقیناً ستاسو لپاره د الله په رسول (صلی الله علیه وعلی آله واصحابه وسلم د مبارکې ذات) کې غوره بېلګه ده، د هغه چا لپاره چې له الله (سره د لیدنې) او د آخرت د ورځې هیله لري او الله ډېر یادوي.
p وَلَمَّا رَءَا ٱلْمُؤْمِنُونَ ٱلْأَحْزَابَ قَالُوا۟ هَٰذَا مَا وَعَدَنَا ٱللَّهُ وَرَسُولُهُۥ وَصَدَقَ ٱللَّهُ وَرَسُولُهُۥ ۚ وَمَا زَادَهُمْ إِلَّآ إِيمَٰنًۭا وَتَسْلِيمًۭا ﴿٢٢﴾ اور جب مومنوں نے (کفار کے) لشکر دیکھے (تو) کہنے لگے یہ وہی ہے جس کا اللہ اور اُس کے رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) نے ہم سے وعدہ فرمایا تھا اور اللہ اور اُس کے رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) نے سچ فرمایا اور اس (صورتِ حال) سے اُن کے ایمان اور اطاعت گزاری میں اضافہ ہی ہوا۔ || وَلَمَّا رَءَا ٱلْمُؤْمِنُونَ ٱلْأَحْزَابَ قَالُوا۟ هَٰذَا مَا وَعَدَنَا ٱللَّهُ وَرَسُولُهُۥ وَصَدَقَ ٱللَّهُ وَرَسُولُهُۥ ۚ وَمَا زَادَهُمْ إِلَّآ إِيمَٰنًۭا وَتَسْلِيمًۭا ﴿٢٢﴾ And when the true believers saw the clans, they said: This is that which Allah and His messenger promised us. Allah and His messenger are true. It did but confirm them in their faith and resignation. (33:22) || وَلَمَّا رَءَا ٱلْمُؤْمِنُونَ ٱلْأَحْزَابَ قَالُوا۟ هَٰذَا مَا وَعَدَنَا ٱللَّهُ وَرَسُولُهُۥ وَصَدَقَ ٱللَّهُ وَرَسُولُهُۥ ۚ وَمَا زَادَهُمْ إِلَّآ إِيمَٰنًۭا وَتَسْلِيمًۭا ﴿٢٢﴾ او کله چې مؤمنانو د (کافرانو) لښکرې ولیدې، (نو) ویې ویل: دا همغه څه دي چې الله او د هغه رسول (صلی الله علیه وعلی آله واصحابه وسلم) له موږ سره وعده کړې وه، او الله او د هغه رسول (صلی الله علیه وعلی آله واصحابه وسلم) رښتیا ویلي وو؛ او له دې (حالت) د هغوی په ایمان او اطاعت کې یوازې زیاتوالی راغی.
p مِّنَ ٱلْمُؤْمِنِينَ رِجَالٌۭ صَدَقُوا۟ مَا عَٰهَدُوا۟ ٱللَّهَ عَلَيْهِ ۖ فَمِنْهُم مَّن قَضَىٰ نَحْبَهُۥ وَمِنْهُم مَّن يَنتَظِرُ ۖ وَمَا بَدَّلُوا۟ تَبْدِيلًۭا ﴿٢٣﴾ مومنوں میں سے کچھ (جواں) مرد ایسے ہیں جنھوں نے وہ بات سچ کر دکھائی جس پر انھوں نے اللہ سے عہد کیا تھا تو ان میں سے کوئی (شہید ہو کر) اپنی نذر پوری کر چکا ہے اور ان میں سے کوئی (اس کا) انتظار کر رہا ہے اور انھوں نے (اپنے عہد کو) ذرا بھی نہیں بدلا۔ || مِّنَ ٱلْمُؤْمِنِينَ رِجَالٌۭ صَدَقُوا۟ مَا عَٰهَدُوا۟ ٱللَّهَ عَلَيْهِ ۖ فَمِنْهُم مَّن قَضَىٰ نَحْبَهُۥ وَمِنْهُم مَّن يَنتَظِرُ ۖ وَمَا بَدَّلُوا۟ تَبْدِيلًۭا ﴿٢٣﴾ Of the believers are men who are true to that which they covenanted with Allah. Some of them have paid their vow by death (in battle), and some of them still are waiting; and they have not altered in the least; (33:23) || مِّنَ ٱلْمُؤْمِنِينَ رِجَالٌۭ صَدَقُوا۟ مَا عَٰهَدُوا۟ ٱللَّهَ عَلَيْهِ ۖ فَمِنْهُم مَّن قَضَىٰ نَحْبَهُۥ وَمِنْهُم مَّن يَنتَظِرُ ۖ وَمَا بَدَّلُوا۟ تَبْدِيلًۭا ﴿٢٣﴾ له مؤمنانو ځینې (زړور) سړي داسې دي چې هغه خبره یې رښتیا کړه چې له الله سره یې پرې ژمنه کړې وه؛ نو له هغوی ځینو (په شهادت) خپله نذر پوره کړې، او ځینې یې (د هغې) انتظار کوي؛ او هغوی (خپله ژمنه) هېڅ نه ده بدله کړې.
p لِّيَجْزِىَ ٱللَّهُ ٱلصَّٰدِقِينَ بِصِدْقِهِمْ وَيُعَذِّبَ ٱلْمُنَٰفِقِينَ إِن شَآءَ أَوْ يَتُوبَ عَلَيْهِمْ ۚ إِنَّ ٱللَّهَ كَانَ غَفُورًۭا رَّحِيمًۭا ﴿٢٤﴾ تاکہ اللہ سچوں کو اُن کی سچائی کا بدلہ دے اور اگر چاہے تو منافقوں کو عذاب دے یا اُن کی توبہ قبول فرما لے بے شک اللہ بڑا ہی بخشنے والا نہایت رحم فرمانے والا ہے۔ || لِّيَجْزِىَ ٱللَّهُ ٱلصَّٰدِقِينَ بِصِدْقِهِمْ وَيُعَذِّبَ ٱلْمُنَٰفِقِينَ إِن شَآءَ أَوْ يَتُوبَ عَلَيْهِمْ ۚ إِنَّ ٱللَّهَ كَانَ غَفُورًۭا رَّحِيمًۭا ﴿٢٤﴾ That Allah may reward the true men for their truth, and punish the hypocrites if He will, or relent toward them (if He will). Lo! Allah is Forgiving, Merciful. (33:24) || لِّيَجْزِىَ ٱللَّهُ ٱلصَّٰدِقِينَ بِصِدْقِهِمْ وَيُعَذِّبَ ٱلْمُنَٰفِقِينَ إِن شَآءَ أَوْ يَتُوبَ عَلَيْهِمْ ۚ إِنَّ ٱللَّهَ كَانَ غَفُورًۭا رَّحِيمًۭا ﴿٢٤﴾ تر څو الله رښتینو ته د هغوی د رښتینولۍ بدله ورکړي، او که وغواړي منافقانو ته عذاب ورکړي یا یې توبه ومني؛ بې‌شکه الله ډېر بښونکی، بې‌حده مهربان دی.
p وَرَدَّ ٱللَّهُ ٱلَّذِينَ كَفَرُوا۟ بِغَيْظِهِمْ لَمْ يَنَالُوا۟ خَيْرًۭا ۚ وَكَفَى ٱللَّهُ ٱلْمُؤْمِنِينَ ٱلْقِتَالَ ۚ وَكَانَ ٱللَّهُ قَوِيًّا عَزِيزًۭا ﴿٢٥﴾ اور اللہ نے کفر کرنے والوں کو اس حال میں پھیر دیا کہ وہ غصے میں (جل رہے) تھے وہ کوئی فائدہ حاصل نہ کر سکے اور قتال میں اللہ (خود) مومنوں کے لیے کافی ہو گیا اور اللہ بہت قوت والا بہت غالب ہے۔ || وَرَدَّ ٱللَّهُ ٱلَّذِينَ كَفَرُوا۟ بِغَيْظِهِمْ لَمْ يَنَالُوا۟ خَيْرًۭا ۚ وَكَفَى ٱللَّهُ ٱلْمُؤْمِنِينَ ٱلْقِتَالَ ۚ وَكَانَ ٱللَّهُ قَوِيًّا عَزِيزًۭا ﴿٢٥﴾ And Allah repulsed the disbelievers in their wrath; they gained no good. Allah averted their attack from the believers. Allah is ever Strong, Mighty. (33:25) || وَرَدَّ ٱللَّهُ ٱلَّذِينَ كَفَرُوا۟ بِغَيْظِهِمْ لَمْ يَنَالُوا۟ خَيْرًۭا ۚ وَكَفَى ٱللَّهُ ٱلْمُؤْمِنِينَ ٱلْقِتَالَ ۚ وَكَانَ ٱللَّهُ قَوِيًّا عَزِيزًۭا ﴿٢٥﴾ او الله کافران په داسې حال بېرته واړول چې په غوسه کې (سوځېدل)؛ هېڅ ګټه یې ترلاسه نه کړه؛ او په جګړه کې الله (پخپله) د مؤمنانو لپاره بس شو؛ او الله ډېر ځواکمن، ډېر غالب دی.
p وَأَنزَلَ ٱلَّذِينَ ظَٰهَرُوهُم مِّنْ أَهْلِ ٱلْكِتَٰبِ مِن صَيَاصِيهِمْ وَقَذَفَ فِى قُلُوبِهِمُ ٱلرُّعْبَ فَرِيقًۭا تَقْتُلُونَ وَتَأْسِرُونَ فَرِيقًۭا ﴿٢٦﴾ اور اہلِ کتاب میں سے جنھوں نے (دشمنوں کی) مدد کی تھی اللہ نے انھیں اُن کے قلعوں سے اتار دیا اور اُن کے دلوں میں (مسلمانوں کا ایسا) رعب ڈال دیا کہ ایک گروہ کو تم قتل کرنے لگے اور ایک گروہ کو تم نے قیدی بنا لیا۔ || وَأَنزَلَ ٱلَّذِينَ ظَٰهَرُوهُم مِّنْ أَهْلِ ٱلْكِتَٰبِ مِن صَيَاصِيهِمْ وَقَذَفَ فِى قُلُوبِهِمُ ٱلرُّعْبَ فَرِيقًۭا تَقْتُلُونَ وَتَأْسِرُونَ فَرِيقًۭا ﴿٢٦﴾ And He brought those of the People of the Scripture who supported them down from their strongholds, and cast panic into their hearts. Some ye slew, and ye made captive some. (33:26) || وَأَنزَلَ ٱلَّذِينَ ظَٰهَرُوهُم مِّنْ أَهْلِ ٱلْكِتَٰبِ مِن صَيَاصِيهِمْ وَقَذَفَ فِى قُلُوبِهِمُ ٱلرُّعْبَ فَرِيقًۭا تَقْتُلُونَ وَتَأْسِرُونَ فَرِيقًۭا ﴿٢٦﴾ او له اهل کتابو چا چې د (دښمنانو) مرسته کړې وه، الله هغوی له خپلو کلاګانو راښکته کړل او په زړونو کې یې (د مسلمانانو داسې) رعب واچاوه چې یوه ډله مو وژله او بله ډله مو بندي کړه.
p وَأَوْرَثَكُمْ أَرْضَهُمْ وَدِيَٰرَهُمْ وَأَمْوَٰلَهُمْ وَأَرْضًۭا لَّمْ تَطَـُٔوهَا ۚ وَكَانَ ٱللَّهُ عَلَىٰ كُلِّ شَىْءٍۢ قَدِيرًۭا ﴿٢٧﴾ اور (اللہ نے) تمھیں وارث بنایا اُن کی زمین کا اور اُن کے گھروں کا اور اُن کے مالوں کا اور اس زمین کا بھی جس پر تم نے قدم نہیں رکھا تھا اور اللہ ہر چیز پر خوب قدرت رکھنے والا ہے۔ || وَأَوْرَثَكُمْ أَرْضَهُمْ وَدِيَٰرَهُمْ وَأَمْوَٰلَهُمْ وَأَرْضًۭا لَّمْ تَطَـُٔوهَا ۚ وَكَانَ ٱللَّهُ عَلَىٰ كُلِّ شَىْءٍۢ قَدِيرًۭا ﴿٢٧﴾ And He caused you to inherit their land and their houses and their wealth, and land ye have not trodden. Allah is ever Able to do all things.
