p يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ تُوبُوٓا۟ إِلَى ٱللَّهِ تَوْبَةًۭ نَّصُوحًا عَسَىٰ رَبُّكُمْ أَن يُكَفِّرَ عَنكُمْ سَيِّـَٔاتِكُمْ وَيُدْخِلَكُمْ جَنَّٰتٍۢ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَٰرُ يَوْمَ لَا يُخْزِى ٱللَّهُ ٱلنَّبِىَّ وَٱلَّذِينَ ءَامَنُوا۟ مَعَهُۥ ۖ نُورُهُمْ يَسْعَىٰ بَيْنَ أَيْدِيهِمْ وَبِأَيْمَٰنِهِمْ يَقُولُونَ رَبَّنَآ أَتْمِمْ لَنَا نُورَنَا وَٱغْفِرْ لَنَآ ۖ إِنَّكَ عَلَىٰ كُلِّ شَىْءٍۢ قَدِيرٌۭ ﴿٨﴾ اے ایمان والو! اللہ کے حضور خالص توبہ کرو کچھ بعید نہیں کہ تمھارا رب تم سے تمھارے گناہ دور کر دے اور تمھیں ایسے باغات میں داخل کرے جن کے نیچے نہریں بہتی ہیں اس دن رسوا نہیں کرے گا اللہ نبی (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کو اور ان لوگوں کو جو ان کے ساتھ ایمان لائے ان کا نور ان کے آگے اور ان کے دائیں طرف دوڑ رہا ہوگا وہ عرض کرتے ہوں گے کہ اے ہمارے رب! ہمارے لیے ہمارا نور مکمل فرما دے اور ہماری مغفرت فرما دے بے شک تو ہر چیز پر بڑی قدرت رکھنے والا ہے۔ || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ تُوبُوٓا۟ إِلَى ٱللَّهِ تَوْبَةًۭ نَّصُوحًا عَسَىٰ رَبُّكُمْ أَن يُكَفِّرَ عَنكُمْ سَيِّـَٔاتِكُمْ وَيُدْخِلَكُمْ جَنَّٰتٍۢ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَٰرُ يَوْمَ لَا يُخْزِى ٱللَّهُ ٱلنَّبِىَّ وَٱلَّذِينَ ءَامَنُوا۟ مَعَهُۥ ۖ نُورُهُمْ يَسْعَىٰ بَيْنَ أَيْدِيهِمْ وَبِأَيْمَٰنِهِمْ يَقُولُونَ رَبَّنَآ أَتْمِمْ لَنَا نُورَنَا وَٱغْفِرْ لَنَآ ۖ إِنَّكَ عَلَىٰ كُلِّ شَىْءٍۢ قَدِيرٌۭ ﴿٨﴾ O ye who believe! Turn unto Allah in sincere repentance! It may be that your Lord will remit from you your evil deeds and bring you into Gardens underneath which rivers flow, on the day when Allah will not abase the Prophet and those who believe with him. Their light will run before them and on their right hands; they will say: Our Lord! Perfect our light for us, and forgive us! Lo! Thou art Able to do all things. (66:8) || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ تُوبُوٓا۟ إِلَى ٱللَّهِ تَوْبَةًۭ نَّصُوحًا عَسَىٰ رَبُّكُمْ أَن يُكَفِّرَ عَنكُمْ سَيِّـَٔاتِكُمْ وَيُدْخِلَكُمْ جَنَّٰتٍۢ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَٰرُ يَوْمَ لَا يُخْزِى ٱللَّهُ ٱلنَّبِىَّ وَٱلَّذِينَ ءَامَنُوا۟ مَعَهُۥ ۖ نُورُهُمْ يَسْعَىٰ بَيْنَ أَيْدِيهِمْ وَبِأَيْمَٰنِهِمْ يَقُولُونَ رَبَّنَآ أَتْمِمْ لَنَا نُورَنَا وَٱغْفِرْ لَنَآ ۖ إِنَّكَ عَلَىٰ كُلِّ شَىْءٍۢ قَدِيرٌۭ ﴿٨﴾ ای مؤمنانو! د الله حضور ته خالصه توبه وکړئ؛ لرې نه ده چې ستاسو رب له تاسو ستاسو ګناهونه لرې کړي او تاسو داسې باغونو ته ننباسي چې لاندې یې ویالې بهېږي؛ په هغه ورځ به الله نبي (صلی الله علیه وعلی آله واصحابه وسلم) او هغه کسان چې ورسره یې ایمان راوړی رسوا نه کړي؛ د هغوی نور به د هغوی مخې ته او ښي لوري ته منډې وهي، هغوی به عرض کوي: ای زموږ ربه! زموږ لپاره زموږ نور بشپړ کړه او موږ وبښه؛ بې‌شکه ته پر هر شي لوی قدرت لرونکی یې.
p يَٰٓأَيُّهَا ٱلنَّبِىُّ جَٰهِدِ ٱلْكُفَّارَ وَٱلْمُنَٰفِقِينَ وَٱغْلُظْ عَلَيْهِمْ ۚ وَمَأْوَىٰهُمْ جَهَنَّمُ ۖ وَبِئْسَ ٱلْمَصِيرُ ﴿٩﴾ اے نبی (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم)! آپ کافروں اور منافقوں سے جہاد کیجیے اور ان پر سختی کیجیے اور اُن کا ٹھکانا جہنم ہے اور وہ بہت برا ٹھکانا ہے۔ || يَٰٓأَيُّهَا ٱلنَّبِىُّ جَٰهِدِ ٱلْكُفَّارَ وَٱلْمُنَٰفِقِينَ وَٱغْلُظْ عَلَيْهِمْ ۚ وَمَأْوَىٰهُمْ جَهَنَّمُ ۖ وَبِئْسَ ٱلْمَصِيرُ ﴿٩﴾ O Prophet! Strive against the disbelievers and the hypocrites, and be stern with them. Hell will be their home, a hapless journey's end. (66:9) || يَٰٓأَيُّهَا ٱلنَّبِىُّ جَٰهِدِ ٱلْكُفَّارَ وَٱلْمُنَٰفِقِينَ وَٱغْلُظْ عَلَيْهِمْ ۚ وَمَأْوَىٰهُمْ جَهَنَّمُ ۖ وَبِئْسَ ٱلْمَصِيرُ ﴿٩﴾ ای نبي (صلی الله علیه وعلی آله واصحابه وسلم)! له کافرانو او منافقانو سره جهاد وکړئ او پر هغوی سختي وکړئ؛ او د هغوی ځای دوزخ دی، او هغه ډېر بد ځای دی.
p ضَرَبَ ٱللَّهُ مَثَلًۭا لِّلَّذِينَ كَفَرُوا۟ ٱمْرَأَتَ نُوحٍۢ وَٱمْرَأَتَ لُوطٍۢ ۖ كَانَتَا تَحْتَ عَبْدَيْنِ مِنْ عِبَادِنَا صَٰلِحَيْنِ فَخَانَتَاهُمَا فَلَمْ يُغْنِيَا عَنْهُمَا مِنَ ٱللَّهِ شَيْـًۭٔا وَقِيلَ ٱدْخُلَا ٱلنَّارَ مَعَ ٱلدَّٰخِلِينَ ﴿١٠﴾ اللہ نے کافروں کے لیے نوح (علیہ السلام) کی بیوی اور لوط (علیہ السلام) کی بیوی کی مثال بیان فرمائی ہے وہ دونوں ہمارے بندوں میں سے دو صالح بندوں کے نکاح میں تھیں پھر دونوں نے ان (شوہروں) سے (دین کے معاملے میں) خیانت کی تو وہ دونوں اللہ کے مقابلے میں ان دونوں (عورتوں) کے کچھ کام نہ آئے اور انھیں حکم دے دیا گیا کہ تم دونوں (عورتیں جہنم کی) آگ میں داخل ہو جاؤ داخل ہونے والوں کے ساتھ۔ || ضَرَبَ ٱللَّهُ مَثَلًۭا لِّلَّذِينَ كَفَرُوا۟ ٱمْرَأَتَ نُوحٍۢ وَٱمْرَأَتَ لُوطٍۢ ۖ كَانَتَا تَحْتَ عَبْدَيْنِ مِنْ عِبَادِنَا صَٰلِحَيْنِ فَخَانَتَاهُمَا فَلَمْ يُغْنِيَا عَنْهُمَا مِنَ ٱللَّهِ شَيْـًۭٔا وَقِيلَ ٱدْخُلَا ٱلنَّارَ مَعَ ٱلدَّٰخِلِينَ ﴿١٠﴾ Allah citeth an example for those who disbelieve: the wife of Noah and the wife of Lot, who were under two of Our righteous slaves yet betrayed them so that they (the husbands) availed them naught against Allah and it was said (unto them): Enter the Fire along with those who enter. (66:10) || ضَرَبَ ٱللَّهُ مَثَلًۭا لِّلَّذِينَ كَفَرُوا۟ ٱمْرَأَتَ نُوحٍۢ وَٱمْرَأَتَ لُوطٍۢ ۖ كَانَتَا تَحْتَ عَبْدَيْنِ مِنْ عِبَادِنَا صَٰلِحَيْنِ فَخَانَتَاهُمَا فَلَمْ يُغْنِيَا عَنْهُمَا مِنَ ٱللَّهِ شَيْـًۭٔا وَقِيلَ ٱدْخُلَا ٱلنَّارَ مَعَ ٱلدَّٰخِلِينَ ﴿١٠﴾ الله د کافرانو لپاره د نوح (علیه السلام) د مېرمنې او د لوط (علیه السلام) د مېرمنې مثال بیان کړی؛ هغه دواړه زموږ له بندګانو د دوو صالحو بندګانو په نکاح کې وې، بیا دواړو له هغو (مېړونو) سره (د دین په چاره کې) خیانت وکړ، نو هغوی دواړه د الله په مقابل کې د دې دواړو (ښځو) هېڅ په کار رانغلل، او هغوی ته حکم وشو چې تاسو دواړه (ښځې) له ننوتونکو سره د (دوزخ) اور ته ننوځئ.
p وَضَرَبَ ٱللَّهُ مَثَلًۭا لِّلَّذِينَ ءَامَنُوا۟ ٱمْرَأَتَ فِرْعَوْنَ إِذْ قَالَتْ رَبِّ ٱبْنِ لِى عِندَكَ بَيْتًۭا فِى ٱلْجَنَّةِ وَنَجِّنِى مِن فِرْعَوْنَ وَعَمَلِهِۦ وَنَجِّنِى مِنَ ٱلْقَوْمِ ٱلظَّٰلِمِينَ ﴿١١﴾ اور اللہ نے ایمان والوں کے لیے فرعون کی بیوی (حضرت آسیہ علیہا السلام) کی مثال بیان فرمائی ہے جب انھوں نے عرض کیا اے میرے رب! میرے لیے اپنے پاس جنت میں ایک گھر بنا دے اور مجھے فرعون اور اُس کے (کافرانہ) عمل سے نجات عطا فرما اور مجھے ظالم قوم سے نجات عطا فرما۔ || وَضَرَبَ ٱللَّهُ مَثَلًۭا لِّلَّذِينَ ءَامَنُوا۟ ٱمْرَأَتَ فِرْعَوْنَ إِذْ قَالَتْ رَبِّ ٱبْنِ لِى عِندَكَ بَيْتًۭا فِى ٱلْجَنَّةِ وَنَجِّنِى مِن فِرْعَوْنَ وَعَمَلِهِۦ وَنَجِّنِى مِنَ ٱلْقَوْمِ ٱلظَّٰلِمِينَ ﴿١١﴾ And Allah citeth an example for those who believe: the wife of Pharaoh when she said: My Lord! Build for me a home with thee in the Garden, and deliver me from Pharaoh and his work, and deliver me from evil-doing folk; (66:11) || وَضَرَبَ ٱللَّهُ مَثَلًۭا لِّلَّذِينَ ءَامَنُوا۟ ٱمْرَأَتَ فِرْعَوْنَ إِذْ قَالَتْ رَبِّ ٱبْنِ لِى عِندَكَ بَيْتًۭا فِى ٱلْجَنَّةِ وَنَجِّنِى مِن فِرْعَوْنَ وَعَمَلِهِۦ وَنَجِّنِى مِنَ ٱلْقَوْمِ ٱلظَّٰلِمِينَ ﴿١١﴾ او الله د مؤمنانو لپاره د فرعون د مېرمنې (حضرت آسیې علیها السلام) مثال بیان کړی، کله چې هغې عرض وکړ: ای زما ربه! زما لپاره له خپل ځان سره په جنت کې یو کور جوړ کړه، او ما له فرعون او د هغه له (کافرانه) عمله نجات راکړه، او ما له ظالم قوم نجات راکړه.
p وَمَرْيَمَ ٱبْنَتَ عِمْرَٰنَ ٱلَّتِىٓ أَحْصَنَتْ فَرْجَهَا فَنَفَخْنَا فِيهِ مِن رُّوحِنَا وَصَدَّقَتْ بِكَلِمَٰتِ رَبِّهَا وَكُتُبِهِۦ وَكَانَتْ مِنَ ٱلْقَٰنِتِينَ ﴿١٢﴾ اور عمران کی بیٹی مریم (علیہا السلام) کی (مثال بیان فرمائی ہے) جنھوں نے اپنی عصمت کی حفاظت کی تو ہم نے ان میں اپنی روح میں سے پھونک دیا اور انھوں نے اپنے رب کی باتوں اور اُس کی کتابوں کی تصدیق کی اور وہ اطاعت گزاروں میں سے تھیں۔ || وَمَرْيَمَ ٱبْنَتَ عِمْرَٰنَ ٱلَّتِىٓ أَحْصَنَتْ فَرْجَهَا فَنَفَخْنَا فِيهِ مِن رُّوحِنَا وَصَدَّقَتْ بِكَلِمَٰتِ رَبِّهَا وَكُتُبِهِۦ وَكَانَتْ مِنَ ٱلْقَٰنِتِينَ ﴿١٢﴾ And Mary, daughter of 'Imran, whose body was chaste, therefor We breathed therein something of Our Spirit. And she put faith in the words of her Lord and His scriptures, and was of the obedient. (66:12) || وَمَرْيَمَ ٱبْنَتَ عِمْرَٰنَ ٱلَّتِىٓ أَحْصَنَتْ فَرْجَهَا فَنَفَخْنَا فِيهِ مِن رُّوحِنَا وَصَدَّقَتْ بِكَلِمَٰتِ رَبِّهَا وَكُتُبِهِۦ وَكَانَتْ مِنَ ٱلْقَٰنِتِينَ ﴿١٢﴾ او د عمران د لور مریم (علیها السلام) (مثال یې بیان کړی) چې خپله عصمت یې وساته، نو موږ په هغې کې له خپل روح څخه پو وکړ؛ او هغې د خپل رب د خبرو او د هغه د کتابونو تصدیق وکړ، او هغه له اطاعت کوونکو څخه وه.
