p وَلَا تُؤْتُوا۟ ٱلسُّفَهَآءَ أَمْوَٰلَكُمُ ٱلَّتِى جَعَلَ ٱللَّهُ لَكُمْ قِيَٰمًۭا وَٱرْزُقُوهُمْ فِيهَا وَٱكْسُوهُمْ وَقُولُوا۟ لَهُمْ قَوْلًۭا مَّعْرُوفًۭا ﴿٥﴾ اور ناسمجھ (یتیموں) کو اپنے وہ مال حوالے نہ کرو جن کو اللہ نے تمھارے لیے (زندگی گزارنے کا) سہارا بنایا ہے اور انھیں اُس (مال) میں سے کھلاؤ اور انھیں پہناؤ اور اُن سے اچھی بات کہو۔ || وَلَا تُؤْتُوا۟ ٱلسُّفَهَآءَ أَمْوَٰلَكُمُ ٱلَّتِى جَعَلَ ٱللَّهُ لَكُمْ قِيَٰمًۭا وَٱرْزُقُوهُمْ فِيهَا وَٱكْسُوهُمْ وَقُولُوا۟ لَهُمْ قَوْلًۭا مَّعْرُوفًۭا ﴿٥﴾ Give not unto the foolish (what is in) your (keeping of their) wealth, which Allah hath given you to maintain; but feed and clothe them from it, and speak kindly unto them. (4:5) || وَلَا تُؤْتُوا۟ ٱلسُّفَهَآءَ أَمْوَٰلَكُمُ ٱلَّتِى جَعَلَ ٱللَّهُ لَكُمْ قِيَٰمًۭا وَٱرْزُقُوهُمْ فِيهَا وَٱكْسُوهُمْ وَقُولُوا۟ لَهُمْ قَوْلًۭا مَّعْرُوفًۭا ﴿٥﴾ او ناپوهو (یتیمانو) ته خپل هغه مالونه مه سپارئ چې الله ستاسو لپاره د (ژوند تېرولو) سهارا ګرځولي، او له هغه (مال) څخه یې خوراک ورکړئ، جامې یې ورواغوندئ او ورته ښه خبره وکړئ.
p وَٱبْتَلُوا۟ ٱلْيَتَٰمَىٰ حَتَّىٰٓ إِذَا بَلَغُوا۟ ٱلنِّكَاحَ فَإِنْ ءَانَسْتُم مِّنْهُمْ رُشْدًۭا فَٱدْفَعُوٓا۟ إِلَيْهِمْ أَمْوَٰلَهُمْ ۖ وَلَا تَأْكُلُوهَآ إِسْرَافًۭا وَبِدَارًا أَن يَكْبَرُوا۟ ۚ وَمَن كَانَ غَنِيًّۭا فَلْيَسْتَعْفِفْ ۖ وَمَن كَانَ فَقِيرًۭا فَلْيَأْكُلْ بِٱلْمَعْرُوفِ ۚ فَإِذَا دَفَعْتُمْ إِلَيْهِمْ أَمْوَٰلَهُمْ فَأَشْهِدُوا۟ عَلَيْهِمْ ۚ وَكَفَىٰ بِٱللَّهِ حَسِيبًۭا ﴿٦﴾ اور یتیموں کو آزماتے رہو یہاں تک کہ جب وہ نکاح (کی عمر) کو پہنچ جائیں تو اگر تم اُن میں سمجھ بوجھ پاؤ تو انھیں اُن کے مال لوٹا دو اور ان کے مال فضول خرچی کرتے ہوئے اور جلدی جلدی نہ کھاؤ کہ وہ بڑے ہو جائیں گے اور (یتیم کا) جو (ولی) مال دار ہے تو اُسے چاہیے کہ بچتا رہے اور جو (ولی) محتاج ہے تو اُسے چاہیے کہ وہ معروف طریقہ کے مطابق کھائے پھر جب تم اُن کے مال اُن کی طرف لوٹاؤ تو اُن پر گواہ بنا لیا کرو اور حساب لینے کے لیے اللہ کافی ہے۔ || وَٱبْتَلُوا۟ ٱلْيَتَٰمَىٰ حَتَّىٰٓ إِذَا بَلَغُوا۟ ٱلنِّكَاحَ فَإِنْ ءَانَسْتُم مِّنْهُمْ رُشْدًۭا فَٱدْفَعُوٓا۟ إِلَيْهِمْ أَمْوَٰلَهُمْ ۖ وَلَا تَأْكُلُوهَآ إِسْرَافًۭا وَبِدَارًا أَن يَكْبَرُوا۟ ۚ وَمَن كَانَ غَنِيًّۭا فَلْيَسْتَعْفِفْ ۖ وَمَن كَانَ فَقِيرًۭا فَلْيَأْكُلْ بِٱلْمَعْرُوفِ ۚ فَإِذَا دَفَعْتُمْ إِلَيْهِمْ أَمْوَٰلَهُمْ فَأَشْهِدُوا۟ عَلَيْهِمْ ۚ وَكَفَىٰ بِٱللَّهِ حَسِيبًۭا ﴿٦﴾ Prove orphans till they reach the marriageable age; then, if ye find them of sound judgment, deliver over unto them their fortune; and devour it not by squandering and in haste lest they should grow up Whoso (of the guardians) is rich, let him abstain generously (from taking of the property of orphans); and whoso is poor let him take thereof in reason (for his guardianship). And when ye deliver up their fortune unto orphans, have (the transaction) witnessed in their presence. Allah sufficeth as a Reckoner. (4:6) || وَٱبْتَلُوا۟ ٱلْيَتَٰمَىٰ حَتَّىٰٓ إِذَا بَلَغُوا۟ ٱلنِّكَاحَ فَإِنْ ءَانَسْتُم مِّنْهُمْ رُشْدًۭا فَٱدْفَعُوٓا۟ إِلَيْهِمْ أَمْوَٰلَهُمْ ۖ وَلَا تَأْكُلُوهَآ إِسْرَافًۭا وَبِدَارًا أَن يَكْبَرُوا۟ ۚ وَمَن كَانَ غَنِيًّۭا فَلْيَسْتَعْفِفْ ۖ وَمَن كَانَ فَقِيرًۭا فَلْيَأْكُلْ بِٱلْمَعْرُوفِ ۚ فَإِذَا دَفَعْتُمْ إِلَيْهِمْ أَمْوَٰلَهُمْ فَأَشْهِدُوا۟ عَلَيْهِمْ ۚ وَكَفَىٰ بِٱللَّهِ حَسِيبًۭا ﴿٦﴾ او یتیمان ازمویئ تر دې چې د نکاح (عمر) ته ورسېږي؛ نو که په هغوی کې پوهه او سمه لاره ومومئ، مالونه یې ورته وسپارئ؛ او د هغوی مالونه په اسراف او په بیړه مه خورئ چې ګواکې لوی به شي؛ او د (یتیم) کوم (سرپرست) چې شتمن وي، باید ځان وساتي؛ او کوم (سرپرست) چې اړ وي، باید په معروف (دودیز) ډول وخوري؛ بیا چې کله یې مالونه ورته وسپارئ، پر هغوی شاهدان ونیسئ؛ او د حساب اخیستلو لپاره الله بس دی.
p لِّلرِّجَالِ نَصِيبٌۭ مِّمَّا تَرَكَ ٱلْوَٰلِدَانِ وَٱلْأَقْرَبُونَ وَلِلنِّسَآءِ نَصِيبٌۭ مِّمَّا تَرَكَ ٱلْوَٰلِدَانِ وَٱلْأَقْرَبُونَ مِمَّا قَلَّ مِنْهُ أَوْ كَثُرَ ۚ نَصِيبًۭا مَّفْرُوضًۭا ﴿٧﴾ مردوں کے لیے اُس (ترکہ) میں سے حصہ ہے جو ماں باپ اور قریبی رشتہ دار چھوڑ گئے اور عورتوں کے لیے بھی اُس میں سے حصہ ہے جو ماں باپ اور قریبی رشتہ دار چھوڑ گئے (خواہ) وہ (ترکہ) تھوڑا ہو یا زیادہ (اللہ کی طرف سے) مقرر کردہ حصہ ہے۔ || لِّلرِّجَالِ نَصِيبٌۭ مِّمَّا تَرَكَ ٱلْوَٰلِدَانِ وَٱلْأَقْرَبُونَ وَلِلنِّسَآءِ نَصِيبٌۭ مِّمَّا تَرَكَ ٱلْوَٰلِدَانِ وَٱلْأَقْرَبُونَ مِمَّا قَلَّ مِنْهُ أَوْ كَثُرَ ۚ نَصِيبًۭا مَّفْرُوضًۭا ﴿٧﴾ Unto the men (of a family) belongeth a share of that which parents and near kindred leave, and unto the women a share of that which parents and near kindred leave, whether it be little or much - a legal share. (4:7) || لِّلرِّجَالِ نَصِيبٌۭ مِّمَّا تَرَكَ ٱلْوَٰلِدَانِ وَٱلْأَقْرَبُونَ وَلِلنِّسَآءِ نَصِيبٌۭ مِّمَّا تَرَكَ ٱلْوَٰلِدَانِ وَٱلْأَقْرَبُونَ مِمَّا قَلَّ مِنْهُ أَوْ كَثُرَ ۚ نَصِيبًۭا مَّفْرُوضًۭا ﴿٧﴾ نارینه وو لپاره په هغه (ترکه) کې برخه ده چې مور او پلار او نږدې خپلوانو پرې ایښې، او ښځو لپاره هم په هغه کې برخه ده چې مور او پلار او نږدې خپلوانو پرې ایښې — (که) هغه (ترکه) لږه وي که ډېره — (د الله له خوا) ټاکلې برخه ده.
p وَإِذَا حَضَرَ ٱلْقِسْمَةَ أُو۟لُوا۟ ٱلْقُرْبَىٰ وَٱلْيَتَٰمَىٰ وَٱلْمَسَٰكِينُ فَٱرْزُقُوهُم مِّنْهُ وَقُولُوا۟ لَهُمْ قَوْلًۭا مَّعْرُوفًۭا ﴿٨﴾ اور جب (وراثت کی) تقسیم کے وقت حاضر ہوں (غیر وارث) قرابت دار اور یتیم اور محتاج تو انھیں بھی اس میں سے کچھ دے دو اور اُن سے اچھی بات کہو۔ || وَإِذَا حَضَرَ ٱلْقِسْمَةَ أُو۟لُوا۟ ٱلْقُرْبَىٰ وَٱلْيَتَٰمَىٰ وَٱلْمَسَٰكِينُ فَٱرْزُقُوهُم مِّنْهُ وَقُولُوا۟ لَهُمْ قَوْلًۭا مَّعْرُوفًۭا ﴿٨﴾ And when kinsfolk and orphans and the needy are present at the division (of the heritage), bestow on them therefrom and speak kindly unto them. (4:8) || وَإِذَا حَضَرَ ٱلْقِسْمَةَ أُو۟لُوا۟ ٱلْقُرْبَىٰ وَٱلْيَتَٰمَىٰ وَٱلْمَسَٰكِينُ فَٱرْزُقُوهُم مِّنْهُ وَقُولُوا۟ لَهُمْ قَوْلًۭا مَّعْرُوفًۭا ﴿٨﴾ او کله چې د (میراث) د وېش پر مهال (ناوارث) خپلوان، یتیمان او اړمن حاضر وي، نو هغوی ته هم له هغه څخه څه ورکړئ او ورته ښه خبره وکړئ.
p وَلْيَخْشَ ٱلَّذِينَ لَوْ تَرَكُوا۟ مِنْ خَلْفِهِمْ ذُرِّيَّةًۭ ضِعَٰفًا خَافُوا۟ عَلَيْهِمْ فَلْيَتَّقُوا۟ ٱللَّهَ وَلْيَقُولُوا۟ قَوْلًۭا سَدِيدًا ﴿٩﴾ اور چاہیے کہ لوگ (یتیموں کے معاملے میں) ڈریں کہ اگر وہ اپنے پیچھے کمزور اولاد چھوڑ جاتے تو وہ اُن کے بارے میں (کتنے) فکر مند ہوتے تو انھیں چاہیے کہ اللہ (کی نافرمانی) سے ڈریں اور چاہیے کہ سیدھی بات کہیں۔ || وَلْيَخْشَ ٱلَّذِينَ لَوْ تَرَكُوا۟ مِنْ خَلْفِهِمْ ذُرِّيَّةًۭ ضِعَٰفًا خَافُوا۟ عَلَيْهِمْ فَلْيَتَّقُوا۟ ٱللَّهَ وَلْيَقُولُوا۟ قَوْلًۭا سَدِيدًا ﴿٩﴾ And let those fear (in their behaviour toward orphans) who if they left behind them weak offspring would be afraid for them. So let them mind their duty to Allah, and speak justly. (4:9) || وَلْيَخْشَ ٱلَّذِينَ لَوْ تَرَكُوا۟ مِنْ خَلْفِهِمْ ذُرِّيَّةًۭ ضِعَٰفًا خَافُوا۟ عَلَيْهِمْ فَلْيَتَّقُوا۟ ٱللَّهَ وَلْيَقُولُوا۟ قَوْلًۭا سَدِيدًا ﴿٩﴾ او خلک دې (د یتیمانو په اړه) ووېرېږي، چې که هغوی پخپله شاته کمزوري اولادونه پرېښي وای، نو د هغوی په اړه به (څومره) اندېښمن وای؛ نو باید له الله (د نافرمانۍ) ووېرېږي او سمه خبره وکړي.
p إِنَّ ٱلَّذِينَ يَأْكُلُونَ أَمْوَٰلَ ٱلْيَتَٰمَىٰ ظُلْمًا إِنَّمَا يَأْكُلُونَ فِى بُطُونِهِمْ نَارًۭا ۖ وَسَيَصْلَوْنَ سَعِيرًۭا ﴿١٠﴾ بے شک وہ لوگ جو یتیموں کا مال ناحق کھاتے ہیں بے شک وہ اپنے پیٹوں میں آگ (ہی) بھرتے ہیں اور وہ عنقریب بھڑکتی ہوئی آگ میں جھلسیں گے۔ || إِنَّ ٱلَّذِينَ يَأْكُلُونَ أَمْوَٰلَ ٱلْيَتَٰمَىٰ ظُلْمًا إِنَّمَا يَأْكُلُونَ فِى بُطُونِهِمْ نَارًۭا ۖ وَسَيَصْلَوْنَ سَعِيرًۭا ﴿١٠﴾ Lo! Those who devour the wealth of orphans wrongfully, they do but swallow fire into their bellies, and they will be exposed to burning flame. (4:10) || إِنَّ ٱلَّذِينَ يَأْكُلُونَ أَمْوَٰلَ ٱلْيَتَٰمَىٰ ظُلْمًا إِنَّمَا يَأْكُلُونَ فِى بُطُونِهِمْ نَارًۭا ۖ وَسَيَصْلَوْنَ سَعِيرًۭا ﴿١٠﴾ بې‌شکه هغه کسان چې د یتیمانو مالونه په ناحقه خوري، بې‌شکه هغوی په خپلو ګېډو کې (یوازې) اور ډکوي، او ژر به په بل شوي اور کې وسوځي.
