p لَأَنتُمْ أَشَدُّ رَهْبَةًۭ فِى صُدُورِهِم مِّنَ ٱللَّهِ ۚ ذَٰلِكَ بِأَنَّهُمْ قَوْمٌۭ لَّا يَفْقَهُونَ ﴿١٣﴾ (بے شک مسلمانو!) تمھارا ڈر اُن یہود کے دلوں میں اللہ سے (بھی) زیادہ ہے یہ اس لیے کہ بے شک وہ ایسے لوگ ہیں جو سمجھتے نہیں۔ || لَأَنتُمْ أَشَدُّ رَهْبَةًۭ فِى صُدُورِهِم مِّنَ ٱللَّهِ ۚ ذَٰلِكَ بِأَنَّهُمْ قَوْمٌۭ لَّا يَفْقَهُونَ ﴿١٣﴾ Ye are more awful as a fear in their bosoms than Allah. That is because they are a folk who understand not. (59:13) || لَأَنتُمْ أَشَدُّ رَهْبَةًۭ فِى صُدُورِهِم مِّنَ ٱللَّهِ ۚ ذَٰلِكَ بِأَنَّهُمْ قَوْمٌۭ لَّا يَفْقَهُونَ ﴿١٣﴾ (بې‌شکه ای مسلمانانو!) ستاسو وېره د هغو یهودو په زړونو کې له الله (هم) زیاته ده؛ دا ځکه چې بې‌شکه هغوی داسې خلک دي چې نه پوهېږي.
p لَا يُقَٰتِلُونَكُمْ جَمِيعًا إِلَّا فِى قُرًۭى مُّحَصَّنَةٍ أَوْ مِن وَرَآءِ جُدُرٍۭ ۚ بَأْسُهُم بَيْنَهُمْ شَدِيدٌۭ ۚ تَحْسَبُهُمْ جَمِيعًۭا وَقُلُوبُهُمْ شَتَّىٰ ۚ ذَٰلِكَ بِأَنَّهُمْ قَوْمٌۭ لَّا يَعْقِلُونَ ﴿١٤﴾ (مسلمانو!) یہ سب لوگ اکٹھے ہو کر (بھی) تم سے جنگ نہیں کر سکیں گے مگر قلعہ بند بستیوں میں یا دیواروں کی آڑ میں (چھپ کر) اُن کی لڑائی اُن کے آپس میں (ہی) بہت سخت ہے (اے مخاطب) تو انھیں اکٹھا سمجھتا ہے حالاں کہ اُن کے دل (ایک دوسرے سے) جدا جدا ہیں یہ اس وجہ سے (ہے) کہ بے شک وہ ایسے لوگ ہیں جو سمجھتے نہیں۔ || لَا يُقَٰتِلُونَكُمْ جَمِيعًا إِلَّا فِى قُرًۭى مُّحَصَّنَةٍ أَوْ مِن وَرَآءِ جُدُرٍۭ ۚ بَأْسُهُم بَيْنَهُمْ شَدِيدٌۭ ۚ تَحْسَبُهُمْ جَمِيعًۭا وَقُلُوبُهُمْ شَتَّىٰ ۚ ذَٰلِكَ بِأَنَّهُمْ قَوْمٌۭ لَّا يَعْقِلُونَ ﴿١٤﴾ They will not fight against you in a body save in fortified villages or from behind walls. Their adversity among themselves is very great. Ye think of them as a whole whereas their hearts are divers. That is because they are a folk who have no sense. (59:14) || لَا يُقَٰتِلُونَكُمْ جَمِيعًا إِلَّا فِى قُرًۭى مُّحَصَّنَةٍ أَوْ مِن وَرَآءِ جُدُرٍۭ ۚ بَأْسُهُم بَيْنَهُمْ شَدِيدٌۭ ۚ تَحْسَبُهُمْ جَمِيعًۭا وَقُلُوبُهُمْ شَتَّىٰ ۚ ذَٰلِكَ بِأَنَّهُمْ قَوْمٌۭ لَّا يَعْقِلُونَ ﴿١٤﴾ (ای مسلمانانو!) دا ټول خلک یوځای هم له تاسو سره جګړه نه شي کولی مګر په کلابندو کلیو کې یا د دېوالونو تر شا (پټ شوي)؛ د هغوی جګړه په خپل منځ کې (همدا) ډېره سخته ده؛ (ای مخاطبه!) ته یې یوځای ګڼې حال دا چې زړونه یې (له یو بله) جلا جلا دي؛ دا ځکه (دي) چې بې‌شکه هغوی داسې خلک دي چې عقل نه چلوي.
p كَمَثَلِ ٱلَّذِينَ مِن قَبْلِهِمْ قَرِيبًۭا ۖ ذَاقُوا۟ وَبَالَ أَمْرِهِمْ وَلَهُمْ عَذَابٌ أَلِيمٌۭ ﴿١٥﴾ ان کا حال تو ان ہی لوگوں جیسا ہے جو ان سے کچھ ہی مدت پہلے اپنے کیے کا وبال چکھ چکے ہیں اور اُن کے لیے دردناک عذاب ہے۔ || كَمَثَلِ ٱلَّذِينَ مِن قَبْلِهِمْ قَرِيبًۭا ۖ ذَاقُوا۟ وَبَالَ أَمْرِهِمْ وَلَهُمْ عَذَابٌ أَلِيمٌۭ ﴿١٥﴾ On the likeness of those (who suffered) a short time before them, they taste the ill-effects of their own conduct, and theirs is painful punishment. (59:15) || كَمَثَلِ ٱلَّذِينَ مِن قَبْلِهِمْ قَرِيبًۭا ۖ ذَاقُوا۟ وَبَالَ أَمْرِهِمْ وَلَهُمْ عَذَابٌ أَلِيمٌۭ ﴿١٥﴾ د هغوی حال د هماغو خلکو په څېر دی چې له دوی لږه موده مخکې یې د خپلو کړنو وبال وڅکه؛ او د هغوی لپاره دردناک عذاب دی.
p كَمَثَلِ ٱلشَّيْطَٰنِ إِذْ قَالَ لِلْإِنسَٰنِ ٱكْفُرْ فَلَمَّا كَفَرَ قَالَ إِنِّى بَرِىٓءٌۭ مِّنكَ إِنِّىٓ أَخَافُ ٱللَّهَ رَبَّ ٱلْعَٰلَمِينَ ﴿١٦﴾ ان کی مثال شیطان کی سی ہے جب اس نے انسان سے کہا کہ کافر ہو جا پھر جب وہ کافر ہو جاتا ہے تو کہتا ہے کہ بے شک میں تجھ سے بیزار ہوں بے شک میں تو اللہ سے ڈرتا ہوں جو تمام جہانوں کا رب ہے۔ || كَمَثَلِ ٱلشَّيْطَٰنِ إِذْ قَالَ لِلْإِنسَٰنِ ٱكْفُرْ فَلَمَّا كَفَرَ قَالَ إِنِّى بَرِىٓءٌۭ مِّنكَ إِنِّىٓ أَخَافُ ٱللَّهَ رَبَّ ٱلْعَٰلَمِينَ ﴿١٦﴾ (And the hypocrites are) on the likeness of the devil when he telleth man to disbelieve, then, when he disbelieveth saith: Lo! I am quit of thee. Lo! I fear Allah, the Lord of the Worlds. (59:16) || كَمَثَلِ ٱلشَّيْطَٰنِ إِذْ قَالَ لِلْإِنسَٰنِ ٱكْفُرْ فَلَمَّا كَفَرَ قَالَ إِنِّى بَرِىٓءٌۭ مِّنكَ إِنِّىٓ أَخَافُ ٱللَّهَ رَبَّ ٱلْعَٰلَمِينَ ﴿١٦﴾ د هغوی مثال د شیطان په څېر دی کله چې یې انسان ته وویل: کافر شه؛ بیا چې کله کافر شي، نو وايي: بې‌شکه زه له تا بېزاره یم؛ بې‌شکه زه له الله وېرېږم چې د ټولو نړیو رب دی.
p فَكَانَ عَٰقِبَتَهُمَآ أَنَّهُمَا فِى ٱلنَّارِ خَٰلِدَيْنِ فِيهَا ۚ وَذَٰلِكَ جَزَٰٓؤُا۟ ٱلظَّٰلِمِينَ ﴿١٧﴾ پھر اُن دونوں کا انجام یہ ہوا کہ وہ دونوں جہنمی ہو گئے جہاں وہ ہمیشہ رہیں گے اور ظالموں کی یہی سزا ہے۔ || فَكَانَ عَٰقِبَتَهُمَآ أَنَّهُمَا فِى ٱلنَّارِ خَٰلِدَيْنِ فِيهَا ۚ وَذَٰلِكَ جَزَٰٓؤُا۟ ٱلظَّٰلِمِينَ ﴿١٧﴾ And the consequence for both will be that they are in the Fire, therein abiding. Such is the reward of evil-doers. (59:17) || فَكَانَ عَٰقِبَتَهُمَآ أَنَّهُمَا فِى ٱلنَّارِ خَٰلِدَيْنِ فِيهَا ۚ وَذَٰلِكَ جَزَٰٓؤُا۟ ٱلظَّٰلِمِينَ ﴿١٧﴾ بیا د دواړو پایله دا شوه چې دواړه دوزخیان شول چې هلته به تل پاتې وي؛ او د ظالمانو همدا سزا ده.
p يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ ٱتَّقُوا۟ ٱللَّهَ وَلْتَنظُرْ نَفْسٌۭ مَّا قَدَّمَتْ لِغَدٍۢ ۖ وَٱتَّقُوا۟ ٱللَّهَ ۚ إِنَّ ٱللَّهَ خَبِيرٌۢ بِمَا تَعْمَلُونَ ﴿١٨﴾ اے ایمان والو! اللہ (کی نافرمانی) سے ڈرو اور چاہیے کہ ہر شخص دیکھتا رہے کہ اُس نے کل (قیامت کے دن) کے لیے کیا آگے بھیجا ہے اور اللہ (کی نافرمانی) سے ڈرو بے شک اللہ اُس سے خوب باخبر ہے جو تم کرتے ہو۔ || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ ٱتَّقُوا۟ ٱللَّهَ وَلْتَنظُرْ نَفْسٌۭ مَّا قَدَّمَتْ لِغَدٍۢ ۖ وَٱتَّقُوا۟ ٱللَّهَ ۚ إِنَّ ٱللَّهَ خَبِيرٌۢ بِمَا تَعْمَلُونَ ﴿١٨﴾ O ye who believe! Observe your duty to Allah. And let every soul look to that which it sendeth on before for the morrow. And observe your duty to Allah. Lo! Allah is Informed of what ye do. (59:18) || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ ٱتَّقُوا۟ ٱللَّهَ وَلْتَنظُرْ نَفْسٌۭ مَّا قَدَّمَتْ لِغَدٍۢ ۖ وَٱتَّقُوا۟ ٱللَّهَ ۚ إِنَّ ٱللَّهَ خَبِيرٌۢ بِمَا تَعْمَلُونَ ﴿١٨﴾ ای مؤمنانو! له الله (د نافرمانۍ) ووېرېږئ، او باید هر کس وګوري چې د سبا (د قیامت د ورځې) لپاره یې څه مخکې لېږلي؛ او له الله (د نافرمانۍ) ووېرېږئ؛ بې‌شکه الله په هغه څه ښه خبر دی چې تاسو یې کوئ.
p وَلَا تَكُونُوا۟ كَٱلَّذِينَ نَسُوا۟ ٱللَّهَ فَأَنسَىٰهُمْ أَنفُسَهُمْ ۚ أُو۟لَٰٓئِكَ هُمُ ٱلْفَٰسِقُونَ ﴿١٩﴾ اور اُن لوگوں کی طرح نہ ہو جاؤ جو اللہ کو بھول گئے تو اللہ نے انھیں خود اپنے آپ سے غافل کر دیا یہی لوگ نافرمان ہیں۔ || وَلَا تَكُونُوا۟ كَٱلَّذِينَ نَسُوا۟ ٱللَّهَ فَأَنسَىٰهُمْ أَنفُسَهُمْ ۚ أُو۟لَٰٓئِكَ هُمُ ٱلْفَٰسِقُونَ ﴿١٩﴾ And be not ye as those who forgot Allah, therefor He caused them to forget their souls. Such are the evil-doers. (59:19) || وَلَا تَكُونُوا۟ كَٱلَّذِينَ نَسُوا۟ ٱللَّهَ فَأَنسَىٰهُمْ أَنفُسَهُمْ ۚ أُو۟لَٰٓئِكَ هُمُ ٱلْفَٰسِقُونَ ﴿١٩﴾ او د هغو خلکو په څېر مه کېږئ چې الله یې هېر کړ، نو الله هغوی له خپل ځان غافل کړل؛ همدا خلک نافرمانان دي.
p لَا يَسْتَوِىٓ أَصْحَٰبُ ٱلنَّارِ وَأَصْحَٰبُ ٱلْجَنَّةِ ۚ أَصْحَٰبُ ٱلْجَنَّةِ هُمُ ٱلْفَآئِزُونَ ﴿٢٠﴾ جہنم والے اور جنت والے برابر نہیں ہو سکتے جنت والے ہی کامیاب ہیں۔ || لَا يَسْتَوِىٓ أَصْحَٰبُ ٱلنَّارِ وَأَصْحَٰبُ ٱلْجَنَّةِ ۚ أَصْحَٰبُ ٱلْجَنَّةِ هُمُ ٱلْفَآئِزُونَ ﴿٢٠﴾ Not equal are the owners of the Fire and the owners of the Garden. The owners of the Garden, they are the victorious. (59:20) || لَا يَسْتَوِىٓ أَصْحَٰبُ ٱلنَّارِ وَأَصْحَٰبُ ٱلْجَنَّةِ ۚ أَصْحَٰبُ ٱلْجَنَّةِ هُمُ ٱلْفَآئِزُونَ ﴿٢٠﴾ دوزخیان او جنتیان برابر نه شي کېدای؛ یوازې جنتیان بریالي دي.
