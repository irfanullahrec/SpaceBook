p يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ لِيَسْتَـْٔذِنكُمُ ٱلَّذِينَ مَلَكَتْ أَيْمَٰنُكُمْ وَٱلَّذِينَ لَمْ يَبْلُغُوا۟ ٱلْحُلُمَ مِنكُمْ ثَلَٰثَ مَرَّٰتٍۢ ۚ مِّن قَبْلِ صَلَوٰةِ ٱلْفَجْرِ وَحِينَ تَضَعُونَ ثِيَابَكُم مِّنَ ٱلظَّهِيرَةِ وَمِنۢ بَعْدِ صَلَوٰةِ ٱلْعِشَآءِ ۚ ثَلَٰثُ عَوْرَٰتٍۢ لَّكُمْ ۚ لَيْسَ عَلَيْكُمْ وَلَا عَلَيْهِمْ جُنَاحٌۢ بَعْدَهُنَّ ۚ طَوَّٰفُونَ عَلَيْكُم بَعْضُكُمْ عَلَىٰ بَعْضٍۢ ۚ كَذَٰلِكَ يُبَيِّنُ ٱللَّهُ لَكُمُ ٱلْءَايَٰتِ ۗ وَٱللَّهُ عَلِيمٌ حَكِيمٌۭ ﴿٥٨﴾ اے ایمان والو! چاہیے کہ تم سے اجازت طلب کیا کریں جو تمھاری ملکیت (غلام اور کنیزیں) ہیں اور وہ (بچے) جو تم میں سے ابھی بلوغت کو نہیں پہنچے تین اوقات میں فجر کی نماز سے پہلے اور دوپہر کے وقت جب تم اپنے کپڑے اتارے رکھتے ہو اور عشاء کی نماز کے بعد (یہ) تین (اوقات) تمھارے پردے کے ہیں ان اوقات کے علاوہ نہ تم پر کوئی حرج ہے اور نہ اُن پر (بقیہ اوقات میں وہ) تمھارے ہاں (کثرت کے ساتھ) ایک دوسرے کے پاس آتے جاتے رہتے ہیں اسی طرح اللہ تمھارے لیے آیتوں کو واضح فرماتا ہے اور اللہ خوب جاننے والا بڑی حکمت والا ہے۔ || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ لِيَسْتَـْٔذِنكُمُ ٱلَّذِينَ مَلَكَتْ أَيْمَٰنُكُمْ وَٱلَّذِينَ لَمْ يَبْلُغُوا۟ ٱلْحُلُمَ مِنكُمْ ثَلَٰثَ مَرَّٰتٍۢ ۚ مِّن قَبْلِ صَلَوٰةِ ٱلْفَجْرِ وَحِينَ تَضَعُونَ ثِيَابَكُم مِّنَ ٱلظَّهِيرَةِ وَمِنۢ بَعْدِ صَلَوٰةِ ٱلْعِشَآءِ ۚ ثَلَٰثُ عَوْرَٰتٍۢ لَّكُمْ ۚ لَيْسَ عَلَيْكُمْ وَلَا عَلَيْهِمْ جُنَاحٌۢ بَعْدَهُنَّ ۚ طَوَّٰفُونَ عَلَيْكُم بَعْضُكُمْ عَلَىٰ بَعْضٍۢ ۚ كَذَٰلِكَ يُبَيِّنُ ٱللَّهُ لَكُمُ ٱلْءَايَٰتِ ۗ وَٱللَّهُ عَلِيمٌ حَكِيمٌۭ ﴿٥٨﴾ O ye who believe! Let your slaves, and those of you who have not come to puberty, ask leave of you at three times (before they come into your presence): Before the prayer of dawn, and when ye lay aside your raiment for the heat of noon, and after the prayer of night. Three times of privacy for you. It is no sin for them or for you at other times, when some of you go round attendant upon others (if they come into your presence without leave). Thus Allah maketh clear the revelations for you. Allah is Knower, Wise. (24:58) || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ لِيَسْتَـْٔذِنكُمُ ٱلَّذِينَ مَلَكَتْ أَيْمَٰنُكُمْ وَٱلَّذِينَ لَمْ يَبْلُغُوا۟ ٱلْحُلُمَ مِنكُمْ ثَلَٰثَ مَرَّٰتٍۢ ۚ مِّن قَبْلِ صَلَوٰةِ ٱلْفَجْرِ وَحِينَ تَضَعُونَ ثِيَابَكُم مِّنَ ٱلظَّهِيرَةِ وَمِنۢ بَعْدِ صَلَوٰةِ ٱلْعِشَآءِ ۚ ثَلَٰثُ عَوْرَٰتٍۢ لَّكُمْ ۚ لَيْسَ عَلَيْكُمْ وَلَا عَلَيْهِمْ جُنَاحٌۢ بَعْدَهُنَّ ۚ طَوَّٰفُونَ عَلَيْكُم بَعْضُكُمْ عَلَىٰ بَعْضٍۢ ۚ كَذَٰلِكَ يُبَيِّنُ ٱللَّهُ لَكُمُ ٱلْءَايَٰتِ ۗ وَٱللَّهُ عَلِيمٌ حَكِيمٌۭ ﴿٥٨﴾ ای مؤمنانو! باید له تاسو اجازه وغواړي هغه چې ستاسو ملکیت (غلامان او وینځې) دي، او هغه (ماشومان) چې له تاسو لا بلوغ ته نه دي رسېدلي، په درېیو وختونو کې: د سهار له لمانځه مخکې، او د غرمې پر مهال کله چې خپلې جامې ایستلې ساتئ، او د ماخستن له لمانځه وروسته؛ (دا) درې (وختونه) ستاسو د پردې دي؛ له دې وختونو پرته نه پر تاسو څه حرج شته او نه پر هغوی؛ (په نورو وختونو کې هغوی) ستاسو کره (ډېر) یو د بل خوا ته ځي راځي؛ همداسې الله تاسو ته آیتونه روښانه کوي؛ او الله ښه پوه، ډېر حکمتناک دی.
p وَإِذَا بَلَغَ ٱلْأَطْفَٰلُ مِنكُمُ ٱلْحُلُمَ فَلْيَسْتَـْٔذِنُوا۟ كَمَا ٱسْتَـْٔذَنَ ٱلَّذِينَ مِن قَبْلِهِمْ ۚ كَذَٰلِكَ يُبَيِّنُ ٱللَّهُ لَكُمْ ءَايَٰتِهِۦ ۗ وَٱللَّهُ عَلِيمٌ حَكِيمٌۭ ﴿٥٩﴾ اور جب تمھارے بچے بلوغ کو پہنچ جائیں تو وہ اجازت طلب کیا کریں جیسا کہ ان سے پہلے (بالغ افراد) اجازت طلب کرتے ہیں اس طرح اللہ تمھارے لیے اپنی آیات (خوب) واضح فرماتا ہے اور اللہ خوب جاننے والا بڑی حکمت والا ہے۔ || وَإِذَا بَلَغَ ٱلْأَطْفَٰلُ مِنكُمُ ٱلْحُلُمَ فَلْيَسْتَـْٔذِنُوا۟ كَمَا ٱسْتَـْٔذَنَ ٱلَّذِينَ مِن قَبْلِهِمْ ۚ كَذَٰلِكَ يُبَيِّنُ ٱللَّهُ لَكُمْ ءَايَٰتِهِۦ ۗ وَٱللَّهُ عَلِيمٌ حَكِيمٌۭ ﴿٥٩﴾ And when the children among you come to puberty then let them ask leave even as those before them used to ask it. Thus Allah maketh clear His revelations for you. Allah is Knower, Wise. (24:59) || وَإِذَا بَلَغَ ٱلْأَطْفَٰلُ مِنكُمُ ٱلْحُلُمَ فَلْيَسْتَـْٔذِنُوا۟ كَمَا ٱسْتَـْٔذَنَ ٱلَّذِينَ مِن قَبْلِهِمْ ۚ كَذَٰلِكَ يُبَيِّنُ ٱللَّهُ لَكُمْ ءَايَٰتِهِۦ ۗ وَٱللَّهُ عَلِيمٌ حَكِيمٌۭ ﴿٥٩﴾ او کله چې ستاسو ماشومان بلوغ ته ورسېږي، نو اجازه دې غواړي لکه څنګه چې له دوی مخکې (بالغ کسان) اجازه غواړي؛ همداسې الله تاسو ته خپل آیتونه (ښه) روښانه کوي؛ او الله ښه پوه، ډېر حکمتناک دی.
p وَٱلْقَوَٰعِدُ مِنَ ٱلنِّسَآءِ ٱلَّٰتِى لَا يَرْجُونَ نِكَاحًۭا فَلَيْسَ عَلَيْهِنَّ جُنَاحٌ أَن يَضَعْنَ ثِيَابَهُنَّ غَيْرَ مُتَبَرِّجَٰتٍۭ بِزِينَةٍۢ ۖ وَأَن يَسْتَعْفِفْنَ خَيْرٌۭ لَّهُنَّ ۗ وَٱللَّهُ سَمِيعٌ عَلِيمٌۭ ﴿٦٠﴾ اور وہ خانہ نشین بوڑھی عورتیں جو نکاح کی آرزو نہیں رکھتیں اُن پر کوئی گناہ نہیں ہے کہ وہ اپنے (زائد) کپڑے اتار دیں بشرطیکہ وہ (بھی) زینت کو ظاہر کرنے والی نہ بنیں اور اگر وہ (اس سے بھی) احتیاط رکھیں تو اُن کے لیے بہتر ہے اور اللہ خوب سننے والا خوب جاننے والا ہے۔ || وَٱلْقَوَٰعِدُ مِنَ ٱلنِّسَآءِ ٱلَّٰتِى لَا يَرْجُونَ نِكَاحًۭا فَلَيْسَ عَلَيْهِنَّ جُنَاحٌ أَن يَضَعْنَ ثِيَابَهُنَّ غَيْرَ مُتَبَرِّجَٰتٍۭ بِزِينَةٍۢ ۖ وَأَن يَسْتَعْفِفْنَ خَيْرٌۭ لَّهُنَّ ۗ وَٱللَّهُ سَمِيعٌ عَلِيمٌۭ ﴿٦٠﴾ As for women past child-bearing, who have no hope of marriage, it is no sin for them if they discard their (outer) clothing in such a way as not to show adornment. But to refrain is better for them. Allah is Hearer, Knower. (24:60) || وَٱلْقَوَٰعِدُ مِنَ ٱلنِّسَآءِ ٱلَّٰتِى لَا يَرْجُونَ نِكَاحًۭا فَلَيْسَ عَلَيْهِنَّ جُنَاحٌ أَن يَضَعْنَ ثِيَابَهُنَّ غَيْرَ مُتَبَرِّجَٰتٍۭ بِزِينَةٍۢ ۖ وَأَن يَسْتَعْفِفْنَ خَيْرٌۭ لَّهُنَّ ۗ وَٱللَّهُ سَمِيعٌ عَلِيمٌۭ ﴿٦٠﴾ او هغه کورناستې زړې ښځې چې د نکاح هیله نه لري، پر هغوی څه ګناه نشته چې خپلې (اضافي) جامې لرې کړي، په دې شرط چې (هم) ښکلا ښکاره کوونکې نه شي؛ او که (له دې هم) احتیاط وکړي، نو د هغوی لپاره غوره ده؛ او الله ښه اورېدونکی، ښه پوه دی.

يُبَيِّنُ ٱللَّهُ لَكُمُ ٱلْءَايَٰتِ لَعَلَّكُمْ تَعْقِلُونَ ﴿٦١﴾ No blame is there upon the blind nor any blame upon the lame nor any blame upon the sick nor on yourselves if ye eat from your houses, or the houses of your fathers, or the houses of your mothers, or the houses of your brothers, or the houses of your sisters, or the houses of your fathers' brothers, or the houses of your fathers' sisters, or the houses of your mothers' brothers, or the houses of your mothers' sisters, or (from that) whereof ye hold the keys, or (from the house) of a friend. No sin shall it be for you whether ye eat together or apart. But when ye enter houses, salute one another with a greeting from Allah, blessed and sweet. Thus Allah maketh clear His revelations for you, that haply ye may understand. (24:61) || لَّيْسَ عَلَى ٱلْأَعْمَىٰ حَرَجٌۭ وَلَا عَلَى ٱلْأَعْرَجِ حَرَجٌۭ وَلَا عَلَى ٱلْمَرِيضِ حَرَجٌۭ وَلَا عَلَىٰٓ أَنفُسِكُمْ أَن تَأْكُلُوا۟ مِنۢ بُيُوتِكُمْ أَوْ بُيُوتِ ءَابَآئِكُمْ أَوْ بُيُوتِ أُمَّهَٰتِكُمْ أَوْ بُيُوتِ إِخْوَٰنِكُمْ أَوْ بُيُوتِ أَخَوَٰتِكُمْ أَوْ بُيُوتِ أَعْمَٰمِكُمْ أَوْ بُيُوتِ عَمَّٰتِكُمْ أَوْ بُيُوتِ أَخْوَٰلِكُمْ أَوْ بُيُوتِ خَٰلَٰتِكُمْ أَوْ مَا مَلَكْتُم مَّفَاتِحَهُۥٓ أَوْ صَدِيقِكُمْ ۚ لَيْسَ عَلَيْكُمْ جُنَاحٌ أَن تَأْكُلُوا۟ جَمِيعًا أَوْ أَشْتَاتًۭا ۚ فَإِذَا دَخَلْتُم بُيُوتًۭا فَسَلِّمُوا۟ عَلَىٰٓ أَنفُسِكُمْ تَحِيَّةًۭ مِّنْ عِندِ ٱللَّهِ مُبَٰرَكَةًۭ طَيِّبَةًۭ ۚ كَذَٰلِكَ يُبَيِّنُ ٱللَّهُ لَكُمُ ٱلْءَايَٰتِ لَعَلَّكُمْ تَعْقِلُونَ ﴿٦١﴾ پر ړانده څه حرج نشته، نه پر ګوډ څه حرج شته، نه پر ناروغ څه حرج شته، او نه پر تاسو پخپله چې له خپلو کورونو وخورئ، یا د خپلو پلرونو له کورونو، یا د خپلو میندو له کورونو، یا د خپلو وروڼو له کورونو، یا د خپلو خویندو له کورونو، یا د خپلو تره ګانو له کورونو، یا د خپلو عمه ګانو له کورونو، یا د خپلو ماما ګانو له کورونو، یا د خپلو تروریو له کورونو، یا له هغو کورونو چې د کیلیو اختیار یې لرئ، یا د خپل ملګري له کوره؛ پر تاسو څه حرج نشته چې ټول یوځای وخورئ یا بېل بېل؛ او کله چې (له دې) کوم کور ته ننوځئ، نو خپلو خلکو ته سلام وایئ؛ دا (د لیدنې) ښه دعا ده چې د الله له لوري برکتناکه (او) پاکه ده؛ همداسې الله تاسو ته آیتونه روښانه کوي تر څو پوه شئ.
