p ءَأَشْفَقْتُمْ أَن تُقَدِّمُوا۟ بَيْنَ يَدَىْ نَجْوَىٰكُمْ صَدَقَٰتٍۢ ۚ فَإِذْ لَمْ تَفْعَلُوا۟ وَتَابَ ٱللَّهُ عَلَيْكُمْ فَأَقِيمُوا۟ ٱلصَّلَوٰةَ وَءَاتُوا۟ ٱلزَّكَوٰةَ وَأَطِيعُوا۟ ٱللَّهَ وَرَسُولَهُۥ ۚ وَٱللَّهُ خَبِيرٌۢ بِمَا تَعْمَلُونَ ﴿١٣﴾ کیا تم (اس حکم سے) گھبرا گئے کہ اپنی سرگوشی سے پہلے کچھ صدقات پیش کرو؟ پس جب تم نے ایسا نہیں کیا اور اللہ نے تمھیں معاف فرما دیا تو نماز قائم کرو اور زکوٰۃ ادا کرو اور اللہ اور اُس کے رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کی اطاعت کرو اور اللہ تمھارے تمام اعمال سے اچھی طرح خبردار ہے۔ || ءَأَشْفَقْتُمْ أَن تُقَدِّمُوا۟ بَيْنَ يَدَىْ نَجْوَىٰكُمْ صَدَقَٰتٍۢ ۚ فَإِذْ لَمْ تَفْعَلُوا۟ وَتَابَ ٱللَّهُ عَلَيْكُمْ فَأَقِيمُوا۟ ٱلصَّلَوٰةَ وَءَاتُوا۟ ٱلزَّكَوٰةَ وَأَطِيعُوا۟ ٱللَّهَ وَرَسُولَهُۥ ۚ وَٱللَّهُ خَبِيرٌۢ بِمَا تَعْمَلُونَ ﴿١٣﴾ Fear ye to offer alms before your conference? Then, when ye do it not and Allah hath forgiven you, establish worship and pay the poor-due and obey Allah and His messenger. And Allah is Aware of what ye do. (58:13) || ءَأَشْفَقْتُمْ أَن تُقَدِّمُوا۟ بَيْنَ يَدَىْ نَجْوَىٰكُمْ صَدَقَٰتٍۢ ۚ فَإِذْ لَمْ تَفْعَلُوا۟ وَتَابَ ٱللَّهُ عَلَيْكُمْ فَأَقِيمُوا۟ ٱلصَّلَوٰةَ وَءَاتُوا۟ ٱلزَّكَوٰةَ وَأَطِيعُوا۟ ٱللَّهَ وَرَسُولَهُۥ ۚ وَٱللَّهُ خَبِيرٌۢ بِمَا تَعْمَلُونَ ﴿١٣﴾ ایا تاسو (له دې حکم) وېرېدلئ چې له خپلو پټو خبرو مخکې څه صدقې وړاندې کړئ؟ نو کله چې مو داسې ونه کړل او الله تاسو وبښلئ، نو لمونځ قائم کړئ، زکات ورکړئ، او د الله او د هغه رسول (صلی الله علیه وعلی آله واصحابه وسلم) اطاعت وکړئ؛ او الله ستاسو په ټولو عملونو ښه خبر دی.
p ۞ أَلَمْ تَرَ إِلَى ٱلَّذِينَ تَوَلَّوْا۟ قَوْمًا غَضِبَ ٱللَّهُ عَلَيْهِم مَّا هُم مِّنكُمْ وَلَا مِنْهُمْ وَيَحْلِفُونَ عَلَى ٱلْكَذِبِ وَهُمْ يَعْلَمُونَ ﴿١٤﴾ کیا آپ نے اُن لوگوں کو نہیں دیکھا جو ایسی قوم سے دوستی کرتے ہیں جن پر اللہ غضب ناک ہوا؟ نہ وہ تم میں سے ہیں اور نہ ہی اُن میں سے ہیں اور وہ جانتے بوجھتے جھوٹی (باتوں) پر قسمیں کھاتے ہیں۔ || ۞ أَلَمْ تَرَ إِلَى ٱلَّذِينَ تَوَلَّوْا۟ قَوْمًا غَضِبَ ٱللَّهُ عَلَيْهِم مَّا هُم مِّنكُمْ وَلَا مِنْهُمْ وَيَحْلِفُونَ عَلَى ٱلْكَذِبِ وَهُمْ يَعْلَمُونَ ﴿١٤﴾ Hast thou not seen those who take for friends a folk with whom Allah is wroth? They are neither of you nor of them, and they swear a false oath knowingly. (58:14) || ۞ أَلَمْ تَرَ إِلَى ٱلَّذِينَ تَوَلَّوْا۟ قَوْمًا غَضِبَ ٱللَّهُ عَلَيْهِم مَّا هُم مِّنكُمْ وَلَا مِنْهُمْ وَيَحْلِفُونَ عَلَى ٱلْكَذِبِ وَهُمْ يَعْلَمُونَ ﴿١٤﴾ ایا تاسو هغه کسان نه دي لیدلي چې له داسې قوم سره دوستي کوي چې الله پرې غضبناک شوی؟ نه هغوی له تاسو دي او نه له هغوی، او په پوهه پر دروغو (خبرو) قسمونه خوري.
p أَعَدَّ ٱللَّهُ لَهُمْ عَذَابًۭا شَدِيدًا ۖ إِنَّهُمْ سَآءَ مَا كَانُوا۟ يَعْمَلُونَ ﴿١٥﴾ اللہ نے اُن کے لیے سخت عذاب تیار کر رکھا ہے بے شک بہت برے کام ہیں جو وہ کر رہے ہیں۔ || أَعَدَّ ٱللَّهُ لَهُمْ عَذَابًۭا شَدِيدًا ۖ إِنَّهُمْ سَآءَ مَا كَانُوا۟ يَعْمَلُونَ ﴿١٥﴾ Allah hath prepared for them a dreadful doom. Evil indeed is that which they are wont to do. (58:15) || أَعَدَّ ٱللَّهُ لَهُمْ عَذَابًۭا شَدِيدًا ۖ إِنَّهُمْ سَآءَ مَا كَانُوا۟ يَعْمَلُونَ ﴿١٥﴾ الله د هغوی لپاره سخت عذاب تیار کړی؛ بې‌شکه ډېر بد کارونه دي چې هغوی یې کوي.
p ٱتَّخَذُوٓا۟ أَيْمَٰنَهُمْ جُنَّةًۭ فَصَدُّوا۟ عَن سَبِيلِ ٱللَّهِ فَلَهُمْ عَذَابٌۭ مُّهِينٌۭ ﴿١٦﴾ انھوں نے اپنی (جھوٹی) قسموں کو ڈھال بنا لیا ہے پھر (دوسروں کو بھی) اللہ کی راہ سے روکتے ہیں تو اُن کے لیے نہایت ذلت والا عذاب ہے۔ || ٱتَّخَذُوٓا۟ أَيْمَٰنَهُمْ جُنَّةًۭ فَصَدُّوا۟ عَن سَبِيلِ ٱللَّهِ فَلَهُمْ عَذَابٌۭ مُّهِينٌۭ ﴿١٦﴾ They make a shelter of their oaths and turn (men) from the way of Allah; so theirs will be a shameful doom. (58:16) || ٱتَّخَذُوٓا۟ أَيْمَٰنَهُمْ جُنَّةًۭ فَصَدُّوا۟ عَن سَبِيلِ ٱللَّهِ فَلَهُمْ عَذَابٌۭ مُّهِينٌۭ ﴿١٦﴾ هغوی خپل (دروغجن) قسمونه ډال ګرځولي، بیا (نور هم) د الله له لارې منع کوي؛ نو د هغوی لپاره ډېر ذلیلوونکی عذاب دی.
p لَّن تُغْنِىَ عَنْهُمْ أَمْوَٰلُهُمْ وَلَآ أَوْلَٰدُهُم مِّنَ ٱللَّهِ شَيْـًٔا ۚ أُو۟لَٰٓئِكَ أَصْحَٰبُ ٱلنَّارِ ۖ هُمْ فِيهَا خَٰلِدُونَ ﴿١٧﴾ اُن کے مال اور اُن کی اولاد اللہ کے (عذاب کے) مقابلے میں ہرگز کچھ بھی کام نہ آئیں گے وہی اہلِ دوزخ ہیں وہ اُس میں ہمیشہ رہنے والے ہیں۔ || لَّن تُغْنِىَ عَنْهُمْ أَمْوَٰلُهُمْ وَلَآ أَوْلَٰدُهُم مِّنَ ٱللَّهِ شَيْـًٔا ۚ أُو۟لَٰٓئِكَ أَصْحَٰبُ ٱلنَّارِ ۖ هُمْ فِيهَا خَٰلِدُونَ ﴿١٧﴾ Their wealth and their children will avail them naught against Allah. Such are rightful owners of the Fire; they will abide therein. (58:17) || لَّن تُغْنِىَ عَنْهُمْ أَمْوَٰلُهُمْ وَلَآ أَوْلَٰدُهُم مِّنَ ٱللَّهِ شَيْـًٔا ۚ أُو۟لَٰٓئِكَ أَصْحَٰبُ ٱلنَّارِ ۖ هُمْ فِيهَا خَٰلِدُونَ ﴿١٧﴾ د هغوی مالونه او اولاد به د الله (د عذاب) په وړاندې هېڅکله هېڅ په کار رانه شي؛ همدوی دوزخیان دي، هغوی به پکې تل پاتې وي.
p يَوْمَ يَبْعَثُهُمُ ٱللَّهُ جَمِيعًۭا فَيَحْلِفُونَ لَهُۥ كَمَا يَحْلِفُونَ لَكُمْ ۖ وَيَحْسَبُونَ أَنَّهُمْ عَلَىٰ شَىْءٍ ۚ أَلَآ إِنَّهُمْ هُمُ ٱلْكَٰذِبُونَ ﴿١٨﴾ جس دن اللہ اُن سب کو (زندہ کر کے) اٹھائے گا تو وہ اُس کے سامنے (بھی) ایسی قسمیں کھائیں گے جیسے وہ تمھارے سامنے قسمیں کھاتے ہیں اور وہ گمان کریں گے کہ وہ کسی (اچھی) شے (دلیل) پر ہیں آگاہ ہو جاؤ! یقیناً وہ جھوٹے ہیں۔ || يَوْمَ يَبْعَثُهُمُ ٱللَّهُ جَمِيعًۭا فَيَحْلِفُونَ لَهُۥ كَمَا يَحْلِفُونَ لَكُمْ ۖ وَيَحْسَبُونَ أَنَّهُمْ عَلَىٰ شَىْءٍ ۚ أَلَآ إِنَّهُمْ هُمُ ٱلْكَٰذِبُونَ ﴿١٨﴾ On the day when Allah will raise them all together, then will they swear unto Him as they (now) swear unto you, and they will fancy that they have some standing. Lo! is it not they who are the liars? (58:18) || يَوْمَ يَبْعَثُهُمُ ٱللَّهُ جَمِيعًۭا فَيَحْلِفُونَ لَهُۥ كَمَا يَحْلِفُونَ لَكُمْ ۖ وَيَحْسَبُونَ أَنَّهُمْ عَلَىٰ شَىْءٍ ۚ أَلَآ إِنَّهُمْ هُمُ ٱلْكَٰذِبُونَ ﴿١٨﴾ په کومه ورځ چې الله دوی ټول (ژوندي کړي او) راپورته کړي، نو د هغه په وړاندې به (هم) داسې قسمونه وخوري لکه څنګه چې ستاسو په وړاندې قسمونه خوري، او ګومان به کوي چې پر کوم (ښه) شي (دلیل) دي؛ خبر اوسئ! یقیناً هغوی دروغجن دي.
p ٱسْتَحْوَذَ عَلَيْهِمُ ٱلشَّيْطَٰنُ فَأَنسَىٰهُمْ ذِكْرَ ٱللَّهِ ۚ أُو۟لَٰٓئِكَ حِزْبُ ٱلشَّيْطَٰنِ ۚ أَلَآ إِنَّ حِزْبَ ٱلشَّيْطَٰنِ هُمُ ٱلْخَٰسِرُونَ ﴿١٩﴾ شیطان نے ان پر غلبہ پا لیا ہے تو اُس نے انھیں اللہ کا ذکر بھلا دیا یہی لوگ شیطان کا گروہ ہیں آگاہ ہو جاؤ! بے شک شیطان کا گروہ ہی خسارہ اٹھانے والا ہے۔ || ٱسْتَحْوَذَ عَلَيْهِمُ ٱلشَّيْطَٰنُ فَأَنسَىٰهُمْ ذِكْرَ ٱللَّهِ ۚ أُو۟لَٰٓئِكَ حِزْبُ ٱلشَّيْطَٰنِ ۚ أَلَآ إِنَّ حِزْبَ ٱلشَّيْطَٰنِ هُمُ ٱلْخَٰسِرُونَ ﴿١٩﴾ The devil hath engrossed them and so hath caused them to forget remembrance of Allah. They are the devil's party. Lo! is it not the devil's party who will be the losers? (58:19) || ٱسْتَحْوَذَ عَلَيْهِمُ ٱلشَّيْطَٰنُ فَأَنسَىٰهُمْ ذِكْرَ ٱللَّهِ ۚ أُو۟لَٰٓئِكَ حِزْبُ ٱلشَّيْطَٰنِ ۚ أَلَآ إِنَّ حِزْبَ ٱلشَّيْطَٰنِ هُمُ ٱلْخَٰسِرُونَ ﴿١٩﴾ شیطان پرې غلبه موندلې، نو د الله ذکر یې ترې هېر کړ؛ همدا خلک د شیطان ډله ده؛ خبر اوسئ! بې‌شکه د شیطان ډله همغه زیانمنه ده.
p إِنَّ ٱلَّذِينَ يُحَآدُّونَ ٱللَّهَ وَرَسُولَهُۥٓ أُو۟لَٰٓئِكَ فِى ٱلْأَذَلِّينَ ﴿٢٠﴾ بے شک جو لوگ اللہ اور اُس کے رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کی مخالفت کرتے ہیں وہی ذلیل ترین لوگوں میں سے ہیں۔ || إِنَّ ٱلَّذِينَ يُحَآدُّونَ ٱللَّهَ وَرَسُولَهُۥٓ أُو۟لَٰٓئِكَ فِى ٱلْأَذَلِّينَ ﴿٢٠﴾ Lo! those who oppose Allah and His messenger, they will be among the lowest. (58:20) || إِنَّ ٱلَّذِينَ يُحَآدُّونَ ٱللَّهَ وَرَسُولَهُۥٓ أُو۟لَٰٓئِكَ فِى ٱلْأَذَلِّينَ ﴿٢٠﴾ بې‌شکه هغه کسان چې د الله او د هغه رسول (صلی الله علیه وعلی آله واصحابه وسلم) مخالفت کوي، همدوی تر ټولو ذلیلو خلکو کې دي.
p كَتَبَ ٱللَّهُ لَأَغْلِبَنَّ أَنَا۠ وَرُسُلِىٓ ۚ إِنَّ ٱللَّهَ قَوِىٌّ عَزِيزٌۭ ﴿٢١﴾ اللہ نے لکھ دیا ہے کہ بے شک میں اور میرے رسول ضرور غالب آئیں گے بے شک اللہ بڑا ہی قوت والا بہت غالب ہے۔ || كَتَبَ ٱللَّهُ لَأَغْلِبَنَّ أَنَا۠ وَرُسُلِىٓ ۚ إِنَّ ٱللَّهَ قَوِىٌّ عَزِيزٌۭ ﴿٢١﴾ Allah hath decreed: Lo! I verily shall conquer, I and My messengers. Lo! Allah is Strong, Almighty. (58:21) || كَتَبَ ٱللَّهُ لَأَغْلِبَنَّ أَنَا۠ وَرُسُلِىٓ ۚ إِنَّ ٱللَّهَ قَوِىٌّ عَزِيزٌۭ ﴿٢١﴾ الله لیکلي چې بې‌شکه زه او زما رسولان به خامخا غالب شو؛ بې‌شکه الله ډېر ځواکمن، ډېر غالب دی.
