p أُو۟لَٰٓئِكَ ٱلَّذِينَ لَعَنَهُمُ ٱللَّهُ ۖ وَمَن يَلْعَنِ ٱللَّهُ فَلَن تَجِدَ لَهُۥ نَصِيرًا ﴿٥٢﴾ (یہ) وہ لوگ ہیں جن پر اللہ نے لعنت کر دی اور جس پر اللہ لعنت کرے پس (اے مخاطب!) تو ہرگز اُس کا کوئی مدد گار نہ پائے گا۔ || أُو۟لَٰٓئِكَ ٱلَّذِينَ لَعَنَهُمُ ٱللَّهُ ۖ وَمَن يَلْعَنِ ٱللَّهُ فَلَن تَجِدَ لَهُۥ نَصِيرًا ﴿٥٢﴾ Those are they whom Allah hath cursed, and he whom Allah hath cursed, thou (O Muhammad) wilt find for him no helper. (4:52) || أُو۟لَٰٓئِكَ ٱلَّذِينَ لَعَنَهُمُ ٱللَّهُ ۖ وَمَن يَلْعَنِ ٱللَّهُ فَلَن تَجِدَ لَهُۥ نَصِيرًا ﴿٥٢﴾ (دا) هغه خلک دي چې الله پرې لعنت کړی؛ او پر چا چې الله لعنت وکړي، نو (ای مخاطبه!) هېڅکله به یې کوم مرستندوی ونه مومې.
p أَمْ لَهُمْ نَصِيبٌۭ مِّنَ ٱلْمُلْكِ فَإِذًۭا لَّا يُؤْتُونَ ٱلنَّاسَ نَقِيرًا ﴿٥٣﴾ کیا اُن کا (کائنات کی) بادشاہی میں سے کوئی حصہ ہے؟ اگر ایسا ہوتا تو یہ لوگوں کو تل کے برابر بھی (کوئی چیز) نہ دیں۔ || أَمْ لَهُمْ نَصِيبٌۭ مِّنَ ٱلْمُلْكِ فَإِذًۭا لَّا يُؤْتُونَ ٱلنَّاسَ نَقِيرًا ﴿٥٣﴾ Or have they even a share in the Sovereignty? Then in that case, they would not give mankind even the speck on a date-stone. (4:53) || أَمْ لَهُمْ نَصِيبٌۭ مِّنَ ٱلْمُلْكِ فَإِذًۭا لَّا يُؤْتُونَ ٱلنَّاسَ نَقِيرًا ﴿٥٣﴾ ایا د هغوی د (کائنات) په پاچاهۍ کې کومه برخه شته؟ که داسې وای، نو دوی به خلکو ته د کونجي د دانې د غوټې په اندازه (هېڅ شی) هم نه ورکاوه.
p أَمْ يَحْسُدُونَ ٱلنَّاسَ عَلَىٰ مَآ ءَاتَىٰهُمُ ٱللَّهُ مِن فَضْلِهِۦ ۖ فَقَدْ ءَاتَيْنَآ ءَالَ إِبْرَٰهِيمَ ٱلْكِتَٰبَ وَٱلْحِكْمَةَ وَءَاتَيْنَٰهُم مُّلْكًا عَظِيمًۭا ﴿٥٤﴾ کیا وہ لوگوں سے حسد کرتے ہیں اس پر جو اللہ نے انھیں اپنے فضل سے عطا فرمایا ہے (اگر ایسا ہے) تو ہم نے ہی (تو) آلِ ابراہیم (علیہ السلام) کو کتاب اور حکمت عطا فرمائی اور ہم نے انھیں بہت بڑی بادشاہی عطا فرمائی۔ || أَمْ يَحْسُدُونَ ٱلنَّاسَ عَلَىٰ مَآ ءَاتَىٰهُمُ ٱللَّهُ مِن فَضْلِهِۦ ۖ فَقَدْ ءَاتَيْنَآ ءَالَ إِبْرَٰهِيمَ ٱلْكِتَٰبَ وَٱلْحِكْمَةَ وَءَاتَيْنَٰهُم مُّلْكًا عَظِيمًۭا ﴿٥٤﴾ Or are they jealous of mankind because of that which Allah of His bounty hath bestowed upon them? For We bestowed upon the house of Abraham (of old) the Scripture and wisdom, and We bestowed on them a mighty kingdom. (4:54) || أَمْ يَحْسُدُونَ ٱلنَّاسَ عَلَىٰ مَآ ءَاتَىٰهُمُ ٱللَّهُ مِن فَضْلِهِۦ ۖ فَقَدْ ءَاتَيْنَآ ءَالَ إِبْرَٰهِيمَ ٱلْكِتَٰبَ وَٱلْحِكْمَةَ وَءَاتَيْنَٰهُم مُّلْكًا عَظِيمًۭا ﴿٥٤﴾ ایا هغوی له خلکو سره د هغه څه له امله حسد کوي چې الله ورته له خپل فضله ورکړي؟ (که داسې ده) نو موږ خو د ابراهیم (علیه السلام) کورنۍ ته کتاب او حکمت ورکړی و، او موږ ورته ډېره لویه پاچاهي ورکړې وه.
p فَمِنْهُم مَّنْ ءَامَنَ بِهِۦ وَمِنْهُم مَّن صَدَّ عَنْهُ ۚ وَكَفَىٰ بِجَهَنَّمَ سَعِيرًا ﴿٥٥﴾ پھر اُن میں سے کوئی تو اس (کتاب) پر ایمان لایا اور اُن میں سے کسی نے اس سے رخ پھیر لیا اور (ان کے لیے) جہنم کی بھڑکتی ہوئی آگ کافی ہے۔ || فَمِنْهُم مَّنْ ءَامَنَ بِهِۦ وَمِنْهُم مَّن صَدَّ عَنْهُ ۚ وَكَفَىٰ بِجَهَنَّمَ سَعِيرًا ﴿٥٥﴾ And of them were (some) who believed therein and of them were (some) who turned away from it. Hell is sufficient for (their) burning. (4:55) || فَمِنْهُم مَّنْ ءَامَنَ بِهِۦ وَمِنْهُم مَّن صَدَّ عَنْهُ ۚ وَكَفَىٰ بِجَهَنَّمَ سَعِيرًا ﴿٥٥﴾ بیا له هغوی ځینو پر هغه (کتاب) ایمان راوړ او ځینو ترې مخ واړاوه؛ او (د هغوی لپاره) د دوزخ بل شوی اور بس دی.
p إِنَّ ٱلَّذِينَ كَفَرُوا۟ بِـَٔايَٰتِنَا سَوْفَ نُصْلِيهِمْ نَارًۭا كُلَّمَا نَضِجَتْ جُلُودُهُم بَدَّلْنَٰهُمْ جُلُودًا غَيْرَهَا لِيَذُوقُوا۟ ٱلْعَذَابَ ۗ إِنَّ ٱللَّهَ كَانَ عَزِيزًا حَكِيمًۭا ﴿٥٦﴾ بے شک جن لوگوں نے ہماری آیات کا انکار کیا ہم عنقریب انھیں آگ میں جھلسا دیں گے جب بھی اُن کی کھالیں جل کر پک جائیں گی (تو) ہم اُن کی کھالوں کو دوسری (کھالوں) سے بدل دیں گے تاکہ وہ (مسلسل) عذاب (کا مزہ) چکھتے رہیں بے شک اللہ بہت غالب بڑی حکمت والا ہے۔ || إِنَّ ٱلَّذِينَ كَفَرُوا۟ بِـَٔايَٰتِنَا سَوْفَ نُصْلِيهِمْ نَارًۭا كُلَّمَا نَضِجَتْ جُلُودُهُم بَدَّلْنَٰهُمْ جُلُودًا غَيْرَهَا لِيَذُوقُوا۟ ٱلْعَذَابَ ۗ إِنَّ ٱللَّهَ كَانَ عَزِيزًا حَكِيمًۭا ﴿٥٦﴾ Lo! Those who disbelieve Our revelations, We shall expose them to the Fire. As often as their skins are consumed We shall exchange them for fresh skins that they may taste the torment. Lo! Allah is ever Mighty, Wise. (4:56) || إِنَّ ٱلَّذِينَ كَفَرُوا۟ بِـَٔايَٰتِنَا سَوْفَ نُصْلِيهِمْ نَارًۭا كُلَّمَا نَضِجَتْ جُلُودُهُم بَدَّلْنَٰهُمْ جُلُودًا غَيْرَهَا لِيَذُوقُوا۟ ٱلْعَذَابَ ۗ إِنَّ ٱللَّهَ كَانَ عَزِيزًا حَكِيمًۭا ﴿٥٦﴾ بې‌شکه هغه کسان چې زموږ له آیتونو یې انکار کړی، ژر به یې په اور کې وسوځوو؛ هرکله چې یې څرمنې وسوځي او پخې شي، (نو) موږ به یې څرمنې په نورو (څرمنو) بدلې کړو تر څو (پرله‌پسې) د عذاب (خوند) وڅکي؛ بې‌شکه الله ډېر غالب، ډېر حکمتناک دی.
p وَٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّٰلِحَٰتِ سَنُدْخِلُهُمْ جَنَّٰتٍۢ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَٰرُ خَٰلِدِينَ فِيهَآ أَبَدًۭا ۖ لَّهُمْ فِيهَآ أَزْوَٰجٌۭ مُّطَهَّرَةٌۭ ۖ وَنُدْخِلُهُمْ ظِلًّۭا ظَلِيلًا ﴿٥٧﴾ اور جو لوگ ایمان لائے اور انھوں نے نیک عمل کیے ہم عنقریب انھیں ایسے باغات میں داخل فرمائیں گے جن کے نیچے سے نہریں بہتی ہیں وہ اُن میں ہمیشہ ہمیشہ رہیں گے اُن کے لیے وہاں پاکیزہ بیویاں ہوں گی اور ہم انھیں گھنے سائے میں داخل فرمائیں گے۔ || وَٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّٰلِحَٰتِ سَنُدْخِلُهُمْ جَنَّٰتٍۢ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَٰرُ خَٰلِدِينَ فِيهَآ أَبَدًۭا ۖ لَّهُمْ فِيهَآ أَزْوَٰجٌۭ مُّطَهَّرَةٌۭ ۖ وَنُدْخِلُهُمْ ظِلًّۭا ظَلِيلًا ﴿٥٧﴾ And as for those who believe and do good works, We shall make them enter Gardens underneath which rivers flow - to dwell therein for ever; there for them are pure companions - and We shall make them enter plenteous shade. (4:57) || وَٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّٰلِحَٰتِ سَنُدْخِلُهُمْ جَنَّٰتٍۢ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَٰرُ خَٰلِدِينَ فِيهَآ أَبَدًۭا ۖ لَّهُمْ فِيهَآ أَزْوَٰجٌۭ مُّطَهَّرَةٌۭ ۖ وَنُدْخِلُهُمْ ظِلًّۭا ظَلِيلًا ﴿٥٧﴾ او هغه کسان چې ایمان یې راوړ او نېک عملونه یې وکړل، ژر به یې داسې باغونو ته ننباسو چې لاندې یې ویالې بهېږي؛ هغوی به پکې تل تر تله وي؛ هلته به یې پاکې مېرمنې وي، او موږ به یې په ګڼ سیوري کې ننباسو.
p ۞ إِنَّ ٱللَّهَ يَأْمُرُكُمْ أَن تُؤَدُّوا۟ ٱلْأَمَٰنَٰتِ إِلَىٰٓ أَهْلِهَا وَإِذَا حَكَمْتُم بَيْنَ ٱلنَّاسِ أَن تَحْكُمُوا۟ بِٱلْعَدْلِ ۚ إِنَّ ٱللَّهَ نِعِمَّا يَعِظُكُم بِهِۦٓ ۗ إِنَّ ٱللَّهَ كَانَ سَمِيعًۢا بَصِيرًۭا ﴿٥٨﴾ بے شک اللہ تمھیں حکم دیتا ہے کہ امانتیں اُن کے حق داروں کو سپرد کر دو اور جب بھی لوگوں کے درمیان فیصلہ کرو (تو) عدل کے ساتھ فیصلہ کرو بے شک اللہ تمھیں اِس بارے میں کیا ہی خوب نصیحت فرماتا ہے بے شک اللہ خوب سننے والا، خوب دیکھنے والا ہے۔ || ۞ إِنَّ ٱللَّهَ يَأْمُرُكُمْ أَن تُؤَدُّوا۟ ٱلْأَمَٰنَٰتِ إِلَىٰٓ أَهْلِهَا وَإِذَا حَكَمْتُم بَيْنَ ٱلنَّاسِ أَن تَحْكُمُوا۟ بِٱلْعَدْلِ ۚ إِنَّ ٱللَّهَ نِعِمَّا يَعِظُكُم بِهِۦٓ ۗ إِنَّ ٱللَّهَ كَانَ سَمِيعًۢا بَصِيرًۭا ﴿٥٨﴾ Lo! Allah commandeth you that ye restore deposits to their owners, and, if ye judge between mankind, that ye judge justly. Lo! comely is this which Allah admonisheth you. Lo! Allah is ever Hearer, Seer. (4:58) || ۞ إِنَّ ٱللَّهَ يَأْمُرُكُمْ أَن تُؤَدُّوا۟ ٱلْأَمَٰنَٰتِ إِلَىٰٓ أَهْلِهَا وَإِذَا حَكَمْتُم بَيْنَ ٱلنَّاسِ أَن تَحْكُمُوا۟ بِٱلْعَدْلِ ۚ إِنَّ ٱللَّهَ نِعِمَّا يَعِظُكُم بِهِۦٓ ۗ إِنَّ ٱللَّهَ كَانَ سَمِيعًۢا بَصِيرًۭا ﴿٥٨﴾ بې‌شکه الله تاسو ته حکم کوي چې امانتونه خپلو حقدارانو ته وسپارئ، او کله چې د خلکو ترمنځ پرېکړه کوئ، (نو) په عدل پرېکړه وکړئ؛ بې‌شکه الله تاسو ته په دې اړه څومره ښه نصیحت کوي! بې‌شکه الله ښه اورېدونکی، ښه لیدونکی دی.
p يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓا۟ أَطِيعُوا۟ ٱللَّهَ وَأَطِيعُوا۟ ٱلرَّسُولَ وَأُو۟لِى ٱلْأَمْرِ مِنكُمْ ۖ فَإِن تَنَٰزَعْتُمْ فِى شَىْءٍۢ فَرُدُّوهُ إِلَى ٱللَّهِ وَٱلرَّسُولِ إِن كُنتُمْ تُؤْمِنُونَ بِٱللَّهِ وَٱلْيَوْمِ ٱلْءَاخِرِ ۚ ذَٰلِكَ خَيْرٌۭ وَأَحْسَنُ تَأْوِيلًا ﴿٥٩﴾ اے ایمان والو! اللہ کی اطاعت کرو اور (اُس کے) رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کی اطاعت کرو اور اُن کی جو تم میں سے صاحبِ اختیار ہیں پھر اگر تمھارے درمیان کسی معاملے میں نزاع ہو جائے تو اُسے اللہ اور رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کی طرف لوٹا دو اگر تم اللہ اور آخرت کے دن پر ایمان رکھتے ہو یہ (طریقہ) بہتر ہے اور (اس کا) انجام بھی بہت اچھا ہے۔ || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓا۟ أَطِيعُوا۟ ٱللَّهَ وَأَطِيعُوا۟ ٱلرَّسُولَ وَأُو۟لِى ٱلْأَمْرِ مِنكُمْ ۖ فَإِن تَنَٰزَعْتُمْ فِى شَىْءٍۢ فَرُدُّوهُ إِلَى ٱللَّهِ وَٱلرَّسُولِ إِن كُنتُمْ تُؤْمِنُونَ بِٱللَّهِ وَٱلْيَوْمِ ٱلْءَاخِرِ ۚ ذَٰلِكَ خَيْرٌۭ وَأَحْسَنُ تَأْوِيلًا ﴿٥٩﴾ O ye who believe! Obey Allah, and obey the messenger and those of you who are in authority; and if ye have a dispute concerning any matter, refer it to Allah and the messenger if ye are (in truth) believers in Allah and the Last Day. That is better and more seemly in the end. (4:59) || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوٓا۟ أَطِيعُوا۟ ٱللَّهَ وَأَطِيعُوا۟ ٱلرَّسُولَ وَأُو۟لِى ٱلْأَمْرِ مِنكُمْ ۖ فَإِن تَنَٰزَعْتُمْ فِى شَىْءٍۢ فَرُدُّوهُ إِلَى ٱللَّهِ وَٱلرَّسُولِ إِن كُنتُمْ تُؤْمِنُونَ بِٱللَّهِ وَٱلْيَوْمِ ٱلْءَاخِرِ ۚ ذَٰلِكَ خَيْرٌۭ وَأَحْسَنُ تَأْوِيلًا ﴿٥٩﴾ ای مؤمنانو! د الله اطاعت وکړئ او د (هغه د) رسول (صلی الله علیه وعلی آله واصحابه وسلم) اطاعت وکړئ او د هغو چې له تاسو څخه د واک خاوندان دي؛ بیا که ستاسو ترمنځ په کومه چاره کې شخړه پېښه شي، نو هغه الله او رسول (صلی الله علیه وعلی آله واصحابه وسلم) ته ور وګرځوئ، که پر الله او د آخرت پر ورځ ایمان لرئ؛ دا (طریقه) غوره ده او (د هغې) پایله هم ډېره ښه ده.
