p وَٱلَّذِينَ كَفَرُوٓا۟ أَعْمَٰلُهُمْ كَسَرَابٍۭ بِقِيعَةٍۢ يَحْسَبُهُ ٱلظَّمْـَٔانُ مَآءً حَتَّىٰٓ إِذَا جَآءَهُۥ لَمْ يَجِدْهُ شَيْـًۭٔا وَوَجَدَ ٱللَّهَ عِندَهُۥ فَوَفَّىٰهُ حِسَابَهُۥ ۗ وَٱللَّهُ سَرِيعُ ٱلْحِسَابِ ﴿٣٩﴾ اور وہ لوگ جنھوں نے کفر کیا ان کے اعمال کسی چٹیل میدان میں سراب کی طرح ہیں پیاسا اُسے پانی سمجھتا ہے یہاں تک کہ جب وہ اس کے پاس (اسے پینے کے لیے) آتا ہے تو اسے کچھ بھی نہیں پاتا اور وہ اللہ کو اپنے پاس پاتا ہے تو وہ (اللہ) اُسے اُس کا پورا حساب چکا دیتا ہے اور اللہ بہت جلد حساب لینے والا ہے۔ || وَٱلَّذِينَ كَفَرُوٓا۟ أَعْمَٰلُهُمْ كَسَرَابٍۭ بِقِيعَةٍۢ يَحْسَبُهُ ٱلظَّمْـَٔانُ مَآءً حَتَّىٰٓ إِذَا جَآءَهُۥ لَمْ يَجِدْهُ شَيْـًۭٔا وَوَجَدَ ٱللَّهَ عِندَهُۥ فَوَفَّىٰهُ حِسَابَهُۥ ۗ وَٱللَّهُ سَرِيعُ ٱلْحِسَابِ ﴿٣٩﴾ As for those who disbelieve, their deeds are as a mirage in a desert. The thirsty one supposeth it to be water till he cometh unto it and findeth it naught, and findeth, in the place thereof, Allah Who payeth him his due; and Allah is swift at reckoning. (24:39) || وَٱلَّذِينَ كَفَرُوٓا۟ أَعْمَٰلُهُمْ كَسَرَابٍۭ بِقِيعَةٍۢ يَحْسَبُهُ ٱلظَّمْـَٔانُ مَآءً حَتَّىٰٓ إِذَا جَآءَهُۥ لَمْ يَجِدْهُ شَيْـًۭٔا وَوَجَدَ ٱللَّهَ عِندَهُۥ فَوَفَّىٰهُ حِسَابَهُۥ ۗ وَٱللَّهُ سَرِيعُ ٱلْحِسَابِ ﴿٣٩﴾ او هغه کسان چې کفر یې کړی، د هغوی عملونه په یوه سپېره ډاګ کې د سراب په څېر دي؛ تږی یې اوبه ګڼي، تر دې چې کله ورته (د څښلو لپاره) راشي، نو هېڅ یې نه مومي؛ او الله له ځان سره مومي، نو (الله) یې پوره حساب ورسره صفا کوي؛ او الله ډېر ژر حساب اخیستونکی دی.
p أَوْ كَظُلُمَٰتٍۢ فِى بَحْرٍۢ لُّجِّىٍّۢ يَغْشَىٰهُ مَوْجٌۭ مِّن فَوْقِهِۦ مَوْجٌۭ مِّن فَوْقِهِۦ سَحَابٌۭ ۚ ظُلُمَٰتٌۢ بَعْضُهَا فَوْقَ بَعْضٍ إِذَآ أَخْرَجَ يَدَهُۥ لَمْ يَكَدْ يَرَىٰهَا ۗ وَمَن لَّمْ يَجْعَلِ ٱللَّهُ لَهُۥ نُورًۭا فَمَا لَهُۥ مِن نُّورٍ ﴿٤٠﴾ یا (کفار کے اعمال کی مثال ایسی ہے) جیسے گہرے سمندر میں اندھیرے ہوں جس پر ایک موج چھائی ہوئی ہو اُس موج پر ایک اور موج (ہو) (اور) اُس کے اوپر بادل (تہ در تہ) اندھیرے (ہی اندھیرے) ہیں اگر کوئی اپنا ہاتھ نکالے تو اسے بھی نہ دیکھ پائے اور جس کو اللہ نور عطا نہ کرے تو اُس کے لیے کوئی نور نہیں۔ || أَوْ كَظُلُمَٰتٍۢ فِى بَحْرٍۢ لُّجِّىٍّۢ يَغْشَىٰهُ مَوْجٌۭ مِّن فَوْقِهِۦ مَوْجٌۭ مِّن فَوْقِهِۦ سَحَابٌۭ ۚ ظُلُمَٰتٌۢ بَعْضُهَا فَوْقَ بَعْضٍ إِذَآ أَخْرَجَ يَدَهُۥ لَمْ يَكَدْ يَرَىٰهَا ۗ وَمَن لَّمْ يَجْعَلِ ٱللَّهُ لَهُۥ نُورًۭا فَمَا لَهُۥ مِن نُّورٍ ﴿٤٠﴾ Or as darkness on a vast, abysmal sea. There covereth him a wave, above which is a wave, above which is a cloud. Layer upon layer of darkness. When he holdeth out his hand he scarce can see it. And he for whom Allah hath not appointed light, for him there is no light. (24:40) || أَوْ كَظُلُمَٰتٍۢ فِى بَحْرٍۢ لُّجِّىٍّۢ يَغْشَىٰهُ مَوْجٌۭ مِّن فَوْقِهِۦ مَوْجٌۭ مِّن فَوْقِهِۦ سَحَابٌۭ ۚ ظُلُمَٰتٌۢ بَعْضُهَا فَوْقَ بَعْضٍ إِذَآ أَخْرَجَ يَدَهُۥ لَمْ يَكَدْ يَرَىٰهَا ۗ وَمَن لَّمْ يَجْعَلِ ٱللَّهُ لَهُۥ نُورًۭا فَمَا لَهُۥ مِن نُّورٍ ﴿٤٠﴾ یا (د کافرانو د عملونو مثال داسې دی) لکه په ژور سمندر کې تیارې چې پر هغه یوه څپه خوره وي، پر هغې څپې بله څپه (وي)، (او) د هغې پورته ورېځ؛ (پر یو بل پرتې) تیارې (همدا تیارې) دي؛ که څوک خپل لاس راوباسي، هغه هم ونه شي لیدلی؛ او چا ته چې الله نور ورنه کړي، نو د هغه لپاره هېڅ نور نشته.
p أَلَمْ تَرَ أَنَّ ٱللَّهَ يُسَبِّحُ لَهُۥ مَن فِى ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ وَٱلطَّيْرُ صَٰٓفَّٰتٍۢ ۖ كُلٌّۭ قَدْ عَلِمَ صَلَاتَهُۥ وَتَسْبِيحَهُۥ ۗ وَٱللَّهُ عَلِيمٌۢ بِمَا يَفْعَلُونَ ﴿٤١﴾ (اے مخاطب!) کیا تو نے نہیں دیکھا کہ بے شک اللہ ہی ہے جس کی تسبیح بیان کرتی ہے ہر وہ شے جو آسمانوں اور زمینوں میں ہے اور پر پھیلائے پرندے ہر ایک نے اپنی نماز اور اپنی تسبیح یقیناً جان رکھی ہے اور اللہ اُسے خوب جانتا ہے جو وہ کر رہے ہیں۔ || أَلَمْ تَرَ أَنَّ ٱللَّهَ يُسَبِّحُ لَهُۥ مَن فِى ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ وَٱلطَّيْرُ صَٰٓفَّٰتٍۢ ۖ كُلٌّۭ قَدْ عَلِمَ صَلَاتَهُۥ وَتَسْبِيحَهُۥ ۗ وَٱللَّهُ عَلِيمٌۢ بِمَا يَفْعَلُونَ ﴿٤١﴾ Hast thou not seen that Allah, He it is Whom all who are in the heavens and the earth praise, and the birds in their flight? Of each He knoweth verily the worship and the praise; and Allah is Aware of what they do. (24:41) || أَلَمْ تَرَ أَنَّ ٱللَّهَ يُسَبِّحُ لَهُۥ مَن فِى ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ وَٱلطَّيْرُ صَٰٓفَّٰتٍۢ ۖ كُلٌّۭ قَدْ عَلِمَ صَلَاتَهُۥ وَتَسْبِيحَهُۥ ۗ وَٱللَّهُ عَلِيمٌۢ بِمَا يَفْعَلُونَ ﴿٤١﴾ (ای مخاطبه!) ایا تا نه دي لیدلي چې بې‌شکه یوازې الله دی چې د هغه تسبیح هر هغه شی بیانوي چې په آسمانونو او ځمکو کې دی، او وزرونه غوړولي مرغان؟ هر یو خپل لمونځ او خپله تسبیح یقیناً پېژندلې ده؛ او الله په هغه څه ښه پوه دی چې هغوی یې کوي.
p وَلِلَّهِ مُلْكُ ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ ۖ وَإِلَى ٱللَّهِ ٱلْمَصِيرُ ﴿٤٢﴾ اور اللہ ہی کے لیے آسمانوں اور زمینوں کی بادشاہت ہے اور اللہ ہی کی طرف پلٹ کر جانا ہے۔ || وَلِلَّهِ مُلْكُ ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ ۖ وَإِلَى ٱللَّهِ ٱلْمَصِيرُ ﴿٤٢﴾ And unto Allah belongeth the Sovereignty of the heavens and the earth, and unto Allah is the journeying. (24:42) || وَلِلَّهِ مُلْكُ ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ ۖ وَإِلَى ٱللَّهِ ٱلْمَصِيرُ ﴿٤٢﴾ او یوازې د الله لپاره د آسمانونو او ځمکو پاچاهي ده، او یوازې الله ته بېرته ورتلل دي.
p أَلَمْ تَرَ أَنَّ ٱللَّهَ يُزْجِى سَحَابًۭا ثُمَّ يُؤَلِّفُ بَيْنَهُۥ ثُمَّ يَجْعَلُهُۥ رُكَامًۭا فَتَرَى ٱلْوَدْقَ يَخْرُجُ مِنْ خِلَٰلِهِۦ وَيُنَزِّلُ مِنَ ٱلسَّمَآءِ مِن جِبَالٍۢ فِيهَا مِنۢ بَرَدٍۢ فَيُصِيبُ بِهِۦ مَن يَشَآءُ وَيَصْرِفُهُۥ عَن مَّن يَشَآءُ ۖ يَكَادُ سَنَا بَرْقِهِۦ يَذْهَبُ بِٱلْأَبْصَٰرِ ﴿٤٣﴾ کیا تو نے نہیں دیکھا کہ اللہ بادلوں کو چلاتا ہے پھر انھیں آپس میں ملا دیتا ہے پھر اُس کو تہ بہ تہ بنا دیتا ہے پھر (اے مخاطب!) تو دیکھتا ہے کہ بارش کے قطرے اُس کے درمیان سے نکلتے ہیں اور وہ آسمان سے پہاڑوں (جیسے بادلوں) سے اولے برساتا ہے پھر جس پر چاہتا ہے اسے برسا دیتا ہے اور جس سے چاہتا ہے اسے ہٹا دیتا ہے قریب ہے کہ اُس (بادل) کی بجلی کی چمک ان کی آنکھوں کو ختم کر دے۔ || أَلَمْ تَرَ أَنَّ ٱللَّهَ يُزْجِى سَحَابًۭا ثُمَّ يُؤَلِّفُ بَيْنَهُۥ ثُمَّ يَجْعَلُهُۥ رُكَامًۭا فَتَرَى ٱلْوَدْقَ يَخْرُجُ مِنْ خِلَٰلِهِۦ وَيُنَزِّلُ مِنَ ٱلسَّمَآءِ مِن جِبَالٍۢ فِيهَا مِنۢ بَرَدٍۢ فَيُصِيبُ بِهِۦ مَن يَشَآءُ وَيَصْرِفُهُۥ عَن مَّن يَشَآءُ ۖ يَكَادُ سَنَا بَرْقِهِۦ يَذْهَبُ بِٱلْأَبْصَٰرِ ﴿٤٣﴾ Hast thou not seen how Allah wafteth the clouds, then gathereth them, then maketh them layers, and thou seest the rain come forth from between them; He sendeth down from the heaven mountains wherein is hail, and smiteth therewith whom He will, and averteth it from whom He will. The flashing of His lightning all but snatcheth away the sight. (24:43) || أَلَمْ تَرَ أَنَّ ٱللَّهَ يُزْجِى سَحَابًۭا ثُمَّ يُؤَلِّفُ بَيْنَهُۥ ثُمَّ يَجْعَلُهُۥ رُكَامًۭا فَتَرَى ٱلْوَدْقَ يَخْرُجُ مِنْ خِلَٰلِهِۦ وَيُنَزِّلُ مِنَ ٱلسَّمَآءِ مِن جِبَالٍۢ فِيهَا مِنۢ بَرَدٍۢ فَيُصِيبُ بِهِۦ مَن يَشَآءُ وَيَصْرِفُهُۥ عَن مَّن يَشَآءُ ۖ يَكَادُ سَنَا بَرْقِهِۦ يَذْهَبُ بِٱلْأَبْصَٰرِ ﴿٤٣﴾ ایا تا نه دي لیدلي چې الله ورېځې چلوي، بیا یې سره یوځای کوي، بیا یې پر یو بل پرتې کوي، بیا (ای مخاطبه!) ګورې چې د باران څاڅکي یې له منځه راوځي؛ او هغه له آسمانه د (ورېځو) غرونو څخه ږلۍ وروي، بیا یې پر چا چې وغواړي وروي او له چا چې وغواړي اړوي یې؛ نږدې ده چې د هغې (ورېځې) د برېښنا ځلا یې سترګې له منځه یوسي.
p يُقَلِّبُ ٱللَّهُ ٱلَّيْلَ وَٱلنَّهَارَ ۚ إِنَّ فِى ذَٰلِكَ لَعِبْرَةًۭ لِّأُو۟لِى ٱلْأَبْصَٰرِ ﴿٤٤﴾ اللہ رات اور دن کو بدلتا رہتا ہے یقیناً اس میں بصیرت رکھنے والوں کے لیے (سامانِ) عبرت ہے۔ || يُقَلِّبُ ٱللَّهُ ٱلَّيْلَ وَٱلنَّهَارَ ۚ إِنَّ فِى ذَٰلِكَ لَعِبْرَةًۭ لِّأُو۟لِى ٱلْأَبْصَٰرِ ﴿٤٤﴾ Allah causeth the revolution of the day and the night. Lo! herein is indeed a lesson for those who see. (24:44) || يُقَلِّبُ ٱللَّهُ ٱلَّيْلَ وَٱلنَّهَارَ ۚ إِنَّ فِى ذَٰلِكَ لَعِبْرَةًۭ لِّأُو۟لِى ٱلْأَبْصَٰرِ ﴿٤٤﴾ الله شپه او ورځ اړوي راړوي؛ یقیناً په دې کې د بصیرت خاوندانو لپاره عبرت دی.
