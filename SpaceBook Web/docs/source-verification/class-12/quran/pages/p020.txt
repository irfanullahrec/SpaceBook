p وَٱلَّذِينَ يُنفِقُونَ أَمْوَٰلَهُمْ رِئَآءَ ٱلنَّاسِ وَلَا يُؤْمِنُونَ بِٱللَّهِ وَلَا بِٱلْيَوْمِ ٱلْءَاخِرِ ۗ وَمَن يَكُنِ ٱلشَّيْطَٰنُ لَهُۥ قَرِينًۭا فَسَآءَ قَرِينًۭا ﴿٣٨﴾ اور وہ لوگ جو اپنے مال لوگوں کو دکھانے کے لیے خرچ کرتے ہیں اور وہ نہ اللہ پر ایمان لاتے ہیں اور نہ آخرت کے دن پر اور شیطان جس کا ساتھی ہو جائے تو وہ بہت بُرا ساتھی ہے۔ || وَٱلَّذِينَ يُنفِقُونَ أَمْوَٰلَهُمْ رِئَآءَ ٱلنَّاسِ وَلَا يُؤْمِنُونَ بِٱللَّهِ وَلَا بِٱلْيَوْمِ ٱلْءَاخِرِ ۗ وَمَن يَكُنِ ٱلشَّيْطَٰنُ لَهُۥ قَرِينًۭا فَسَآءَ قَرِينًۭا ﴿٣٨﴾ And (also) those who spend their wealth in order to be seen of men, and believe not in Allah nor the Last Day. Whoso taketh Satan for a comrade, a bad comrade hath he. (4:38) || وَٱلَّذِينَ يُنفِقُونَ أَمْوَٰلَهُمْ رِئَآءَ ٱلنَّاسِ وَلَا يُؤْمِنُونَ بِٱللَّهِ وَلَا بِٱلْيَوْمِ ٱلْءَاخِرِ ۗ وَمَن يَكُنِ ٱلشَّيْطَٰنُ لَهُۥ قَرِينًۭا فَسَآءَ قَرِينًۭا ﴿٣٨﴾ او هغه کسان چې خپل مالونه خلکو ته د ښودلو لپاره لګوي، او نه پر الله ایمان راوړي او نه د آخرت پر ورځ؛ او شیطان چې د چا ملګری شي، نو هغه ډېر بد ملګری دی.
p وَمَاذَا عَلَيْهِمْ لَوْ ءَامَنُوا۟ بِٱللَّهِ وَٱلْيَوْمِ ٱلْءَاخِرِ وَأَنفَقُوا۟ مِمَّا رَزَقَهُمُ ٱللَّهُ ۚ وَكَانَ ٱللَّهُ بِهِمْ عَلِيمًا ﴿٣٩﴾ اور اُن کا کیا نقصان ہوتا اگر وہ اللہ اور آخرت کے دن پر ایمان لے آتے اور اللہ نے انھیں جو رزق دیا ہے اُس میں سے خرچ کرتے اور اللہ اُن کو خوب جاننے والا ہے۔ || وَمَاذَا عَلَيْهِمْ لَوْ ءَامَنُوا۟ بِٱللَّهِ وَٱلْيَوْمِ ٱلْءَاخِرِ وَأَنفَقُوا۟ مِمَّا رَزَقَهُمُ ٱللَّهُ ۚ وَكَانَ ٱللَّهُ بِهِمْ عَلِيمًا ﴿٣٩﴾ What have they (to fear) if they believe in Allah and the Last Day and spend (aright) of that which Allah hath bestowed upon them, when Allah is ever Aware of them (and all they do)? (4:39) || وَمَاذَا عَلَيْهِمْ لَوْ ءَامَنُوا۟ بِٱللَّهِ وَٱلْيَوْمِ ٱلْءَاخِرِ وَأَنفَقُوا۟ مِمَّا رَزَقَهُمُ ٱللَّهُ ۚ وَكَانَ ٱللَّهُ بِهِمْ عَلِيمًا ﴿٣٩﴾ او د هغوی به څه زیان و که پر الله او د آخرت پر ورځ یې ایمان راوړی وای او له هغه رزق یې لګښت کړی وای چې الله ورکړی؛ او الله هغوی ښه پېژني.
p إِنَّ ٱللَّهَ لَا يَظْلِمُ مِثْقَالَ ذَرَّةٍۢ ۖ وَإِن تَكُ حَسَنَةًۭ يُضَٰعِفْهَا وَيُؤْتِ مِن لَّدُنْهُ أَجْرًا عَظِيمًۭا ﴿٤٠﴾ بے شک اللہ ایک ذرہ برابر بھی ظلم نہیں کرتا اور اگر کوئی نیکی ہو تو اُسے دُگنا کر دیتا ہے اور اپنے پاس سے اجرِ عظیم عطا فرماتا ہے۔ || إِنَّ ٱللَّهَ لَا يَظْلِمُ مِثْقَالَ ذَرَّةٍۢ ۖ وَإِن تَكُ حَسَنَةًۭ يُضَٰعِفْهَا وَيُؤْتِ مِن لَّدُنْهُ أَجْرًا عَظِيمًۭا ﴿٤٠﴾ Lo! Allah wrongeth not even of the weight of an ant; and if there is a good deed, He will double it and will give (the doer) from His presence an immense reward. (4:40) || إِنَّ ٱللَّهَ لَا يَظْلِمُ مِثْقَالَ ذَرَّةٍۢ ۖ وَإِن تَكُ حَسَنَةًۭ يُضَٰعِفْهَا وَيُؤْتِ مِن لَّدُنْهُ أَجْرًا عَظِيمًۭا ﴿٤٠﴾ بې‌شکه الله د یوې ذرې په اندازه هم ظلم نه کوي؛ او که کومه نېکي وي، دوه چنده یې کوي او له خپل لوري لوی اجر ورکوي.
p فَكَيْفَ إِذَا جِئْنَا مِن كُلِّ أُمَّةٍۭ بِشَهِيدٍۢ وَجِئْنَا بِكَ عَلَىٰ هَٰٓؤُلَآءِ شَهِيدًۭا ﴿٤١﴾ تو (اس دن) کیا حال ہو گا جب ہم ہر اُمت سے ایک گواہ لائیں گے اور ہم آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کو ان سب پر گواہ بنا کر لائیں گے۔ || فَكَيْفَ إِذَا جِئْنَا مِن كُلِّ أُمَّةٍۭ بِشَهِيدٍۢ وَجِئْنَا بِكَ عَلَىٰ هَٰٓؤُلَآءِ شَهِيدًۭا ﴿٤١﴾ But how (will it be with them) when We bring of every people a witness, and We bring thee (O Muhammad) a witness against these? (4:41) || فَكَيْفَ إِذَا جِئْنَا مِن كُلِّ أُمَّةٍۭ بِشَهِيدٍۢ وَجِئْنَا بِكَ عَلَىٰ هَٰٓؤُلَآءِ شَهِيدًۭا ﴿٤١﴾ نو (په هغه ورځ) به څه حال وي کله چې موږ له هر امت څخه یو شاهد راولو او تاسو (صلی الله علیه وعلی آله واصحابه وسلم) پر دې ټولو شاهد راولو؟
p يَوْمَئِذٍۢ يَوَدُّ ٱلَّذِينَ كَفَرُوا۟ وَعَصَوُا۟ ٱلرَّسُولَ لَوْ تُسَوَّىٰ بِهِمُ ٱلْأَرْضُ وَلَا يَكْتُمُونَ ٱللَّهَ حَدِيثًۭا ﴿٤٢﴾ اس دن وہ آرزو کریں گے جنھوں نے کفر کیا اور رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کی نافرمانی کی کہ کاش (انھیں مٹی میں دبا کر) زمین اُن پر برابر کر دی جائے اور وہ اللہ سے کوئی بات چھپا نہ سکیں گے۔ || يَوْمَئِذٍۢ يَوَدُّ ٱلَّذِينَ كَفَرُوا۟ وَعَصَوُا۟ ٱلرَّسُولَ لَوْ تُسَوَّىٰ بِهِمُ ٱلْأَرْضُ وَلَا يَكْتُمُونَ ٱللَّهَ حَدِيثًۭا ﴿٤٢﴾ On that day those who disbelieved and disobeyed the messenger will wish that they were level with the ground, and they can hide no fact from Allah. (4:42) || يَوْمَئِذٍۢ يَوَدُّ ٱلَّذِينَ كَفَرُوا۟ وَعَصَوُا۟ ٱلرَّسُولَ لَوْ تُسَوَّىٰ بِهِمُ ٱلْأَرْضُ وَلَا يَكْتُمُونَ ٱللَّهَ حَدِيثًۭا ﴿٤٢﴾ په هغه ورځ به هغه کسان چې کفر یې کړی او د رسول (صلی الله علیه وعلی آله واصحابه وسلم) نافرماني یې کړې، ارمان وکړي چې کاش (په خاوره کې ښخ شي او) ځمکه پرې برابره شي؛ او هغوی به له الله څخه هېڅ خبره پټه نه کړای شي.
p يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ لَا تَقْرَبُوا۟ ٱلصَّلَوٰةَ وَأَنتُمْ سُكَٰرَىٰ حَتَّىٰ تَعْلَمُوا۟ مَا تَقُولُونَ وَلَا جُنُبًا إِلَّا عَابِرِى سَبِيلٍ حَتَّىٰ تَغْتَسِلُوا۟ ۚ وَإِن كُنتُم مَّرْضَىٰٓ أَوْ عَلَىٰ سَفَرٍ أَوْ جَآءَ أَحَدٌۭ مِّنكُم مِّنَ ٱلْغَآئِطِ أَوْ لَٰمَسْتُمُ ٱلنِّسَآءَ فَلَمْ تَجِدُوا۟ مَآءًۭ فَتَيَمَّمُوا۟ صَعِيدًۭا طَيِّبًۭا فَٱمْسَحُوا۟ بِوُجُوهِكُمْ وَأَيْدِيكُمْ ۗ إِنَّ ٱللَّهَ كَانَ عَفُوًّا غَفُورًا ﴿٤٣﴾ اے ایمان والو! نماز کے قریب نہ جاؤ جب کہ تم نشے کی حالت میں ہو یہاں تک کہ جو تم کہہ رہے ہو اُسے جان لو اور نہ جنابت (ناپاکی) کی حالت میں (نماز کے قریب جاؤ) سوائے اِس کے کہ تم راستہ عبور کرنے والے ہو یہاں تک کہ تم غسل کر لو اور اگر تم بیمار ہو یا سفر میں ہو یا تم میں سے کوئی قضائے حاجت سے آئے یا تم نے بیویوں سے صحبت کی ہو پھر تم پانی نہ پاؤ تو پاک مٹی سے تیمم کر لیا کرو پس اپنے چہروں اور ہاتھوں کا مسح کر لیا کرو بے شک اللہ بہت معاف فرمانے والا بڑا ہی بخشنے والا ہے۔ || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ لَا تَقْرَبُوا۟ ٱلصَّلَوٰةَ وَأَنتُمْ سُكَٰرَىٰ حَتَّىٰ تَعْلَمُوا۟ مَا تَقُولُونَ وَلَا جُنُبًا إِلَّا عَابِرِى سَبِيلٍ حَتَّىٰ تَغْتَسِلُوا۟ ۚ وَإِن كُنتُم مَّرْضَىٰٓ أَوْ عَلَىٰ سَفَرٍ أَوْ جَآءَ أَحَدٌۭ مِّنكُم مِّنَ ٱلْغَآئِطِ أَوْ لَٰمَسْتُمُ ٱلنِّسَآءَ فَلَمْ تَجِدُوا۟ مَآءًۭ فَتَيَمَّمُوا۟ صَعِيدًۭا طَيِّبًۭا فَٱمْسَحُوا۟ بِوُجُوهِكُمْ وَأَيْدِيكُمْ ۗ إِنَّ ٱللَّهَ كَانَ عَفُوًّا غَفُورًا ﴿٤٣﴾ O ye who believe! Draw not near unto prayer when ye are drunken, till ye know that which ye utter, nor when ye are polluted, save when journeying upon the road, till ye have bathed. And if ye be ill, or on a journey, or one of you cometh from the closet, or ye have touched women, and ye find not water, then go to high clean soil and rub your faces and your hands (therewith). Lo! Allah is Benign, Forgiving. (4:43) || يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ لَا تَقْرَبُوا۟ ٱلصَّلَوٰةَ وَأَنتُمْ سُكَٰرَىٰ حَتَّىٰ تَعْلَمُوا۟ مَا تَقُولُونَ وَلَا جُنُبًا إِلَّا عَابِرِى سَبِيلٍ حَتَّىٰ تَغْتَسِلُوا۟ ۚ وَإِن كُنتُم مَّرْضَىٰٓ أَوْ عَلَىٰ سَفَرٍ أَوْ جَآءَ أَحَدٌۭ مِّنكُم مِّنَ ٱلْغَآئِطِ أَوْ لَٰمَسْتُمُ ٱلنِّسَآءَ فَلَمْ تَجِدُوا۟ مَآءًۭ فَتَيَمَّمُوا۟ صَعِيدًۭا طَيِّبًۭا فَٱمْسَحُوا۟ بِوُجُوهِكُمْ وَأَيْدِيكُمْ ۗ إِنَّ ٱللَّهَ كَانَ عَفُوًّا غَفُورًا ﴿٤٣﴾ ای مؤمنانو! لمانځه ته مه نږدې کېږئ په داسې حال کې چې نشه یاست تر دې چې پوه شئ څه وایئ، او نه د جنابت (ناپاکۍ) په حال کې (لمانځه ته نږدې کېږئ)، مګر دا چې د لارې تېرېدونکي یاست، تر دې چې غسل وکړئ؛ او که ناروغ یاست یا په سفر کې یاست یا له تاسو څوک له قضای حاجت راشي یا مو له مېرمنو سره صحبت کړی وي، بیا اوبه ونه مومئ، نو په پاکه خاوره تیمم وکړئ: خپلو مخونو او لاسونو ته مسح ورکړئ؛ بې‌شکه الله ډېر معاف کوونکی، ډېر بښونکی دی.
p أَلَمْ تَرَ إِلَى ٱلَّذِينَ أُوتُوا۟ نَصِيبًۭا مِّنَ ٱلْكِتَٰبِ يَشْتَرُونَ ٱلضَّلَٰلَةَ وَيُرِيدُونَ أَن تَضِلُّوا۟ ٱلسَّبِيلَ ﴿٤٤﴾ کیا آپ نے اُن لوگوں (کے حال) کو نہیں دیکھا جنھیں کتاب (تورات) کا ایک حصہ دیا گیا تھا وہ گم راہی کا سودا کرتے ہیں اور وہ چاہتے ہیں کہ تم سیدھے راستے سے بھٹک جاؤ۔ || أَلَمْ تَرَ إِلَى ٱلَّذِينَ أُوتُوا۟ نَصِيبًۭا مِّنَ ٱلْكِتَٰبِ يَشْتَرُونَ ٱلضَّلَٰلَةَ وَيُرِيدُونَ أَن تَضِلُّوا۟ ٱلسَّبِيلَ ﴿٤٤﴾ Seest thou not those unto whom a portion of the Scripture hath been given, how they purchase error, and seek to make you (Muslims) err from the right way? (4:44) || أَلَمْ تَرَ إِلَى ٱلَّذِينَ أُوتُوا۟ نَصِيبًۭا مِّنَ ٱلْكِتَٰبِ يَشْتَرُونَ ٱلضَّلَٰلَةَ وَيُرِيدُونَ أَن تَضِلُّوا۟ ٱلسَّبِيلَ ﴿٤٤﴾ ایا تاسو د هغو خلکو (حال) نه دی لیدلی چې د کتاب (تورات) یوه برخه ورکړل شوې وه؟ هغوی د ګمراهۍ سودا کوي او غواړي چې تاسو له سمې لارې واوړئ.
p وَٱللَّهُ أَعْلَمُ بِأَعْدَآئِكُمْ ۚ وَكَفَىٰ بِٱللَّهِ وَلِيًّۭا وَكَفَىٰ بِٱللَّهِ نَصِيرًۭا ﴿٤٥﴾ اور اللہ تمھارے دشمنوں کو خوب جانتا ہے اور اللہ ہی کافی ہے (بطور) کارساز اور اللہ ہی کافی ہے (بطور) مدد گار۔ || وَٱللَّهُ أَعْلَمُ بِأَعْدَآئِكُمْ ۚ وَكَفَىٰ بِٱللَّهِ وَلِيًّۭا وَكَفَىٰ بِٱللَّهِ نَصِيرًۭا ﴿٤٥﴾ Allah knoweth best (who are) your enemies. Allah is sufficient as a Guardian, and Allah is sufficient as a Supporter. (4:45) || وَٱللَّهُ أَعْلَمُ بِأَعْدَآئِكُمْ ۚ وَكَفَىٰ بِٱللَّهِ وَلِيًّۭا وَكَفَىٰ بِٱللَّهِ نَصِيرًۭا ﴿٤٥﴾ او الله ستاسو دښمنان ښه پېژني؛ او الله د کارساز په توګه بس دی، او الله د مرستندوی په توګه بس دی.
