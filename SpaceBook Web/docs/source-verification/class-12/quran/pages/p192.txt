p لَوْ أَنزَلْنَا هَٰذَا ٱلْقُرْءَانَ عَلَىٰ جَبَلٍۢ لَّرَأَيْتَهُۥ خَٰشِعًۭا مُّتَصَدِّعًۭا مِّنْ خَشْيَةِ ٱللَّهِ ۚ وَتِلْكَ ٱلْأَمْثَٰلُ نَضْرِبُهَا لِلنَّاسِ لَعَلَّهُمْ يَتَفَكَّرُونَ ﴿٢١﴾ اگر ہم اس قرآن کو کسی پہاڑ پر نازل کرتے تو (اے مخاطب!) یقیناً تو اسے اللہ کے خوف سے جھک کر پاش پاش ہوتے ہوئے دیکھتا اور یہ مثالیں ہم لوگوں کے لیے بیان کرتے ہیں تاکہ وہ غور و فکر کریں۔ || لَوْ أَنزَلْنَا هَٰذَا ٱلْقُرْءَانَ عَلَىٰ جَبَلٍۢ لَّرَأَيْتَهُۥ خَٰشِعًۭا مُّتَصَدِّعًۭا مِّنْ خَشْيَةِ ٱللَّهِ ۚ وَتِلْكَ ٱلْأَمْثَٰلُ نَضْرِبُهَا لِلنَّاسِ لَعَلَّهُمْ يَتَفَكَّرُونَ ﴿٢١﴾ If We had caused this Qur'an to descend upon a mountain, thou (O Muhammad) verily hadst seen it humbled, rent asunder by the fear of Allah. (59:21) || لَوْ أَنزَلْنَا هَٰذَا ٱلْقُرْءَانَ عَلَىٰ جَبَلٍۢ لَّرَأَيْتَهُۥ خَٰشِعًۭا مُّتَصَدِّعًۭا مِّنْ خَشْيَةِ ٱللَّهِ ۚ وَتِلْكَ ٱلْأَمْثَٰلُ نَضْرِبُهَا لِلنَّاسِ لَعَلَّهُمْ يَتَفَكَّرُونَ ﴿٢١﴾ که موږ دا قرآن پر کوم غره نازل کړی وای، نو (ای مخاطبه!) یقیناً به دې لیدلی وای چې د الله له وېرې ټیټ شوی او ټوټې ټوټې کېږي؛ او دا مثالونه موږ خلکو ته بیانوو تر څو فکر وکړي.
p هُوَ ٱللَّهُ ٱلَّذِى لَآ إِلَٰهَ إِلَّا هُوَ ۖ عَٰلِمُ ٱلْغَيْبِ وَٱلشَّهَٰدَةِ ۖ هُوَ ٱلرَّحْمَٰنُ ٱلرَّحِيمُ ﴿٢٢﴾ وہی اللہ ہے جس کے سوا کوئی معبود نہیں (وہ) غیب اور ظاہر کو جاننے والا ہے وہی بڑا مہربان نہایت رحم فرمانے والا ہے۔ || هُوَ ٱللَّهُ ٱلَّذِى لَآ إِلَٰهَ إِلَّا هُوَ ۖ عَٰلِمُ ٱلْغَيْبِ وَٱلشَّهَٰدَةِ ۖ هُوَ ٱلرَّحْمَٰنُ ٱلرَّحِيمُ ﴿٢٢﴾ He is Allah, than Whom there is no other Allah, the Knower of the Invisible and the Visible. He is the Beneficent, Merciful. (59:22) || هُوَ ٱللَّهُ ٱلَّذِى لَآ إِلَٰهَ إِلَّا هُوَ ۖ عَٰلِمُ ٱلْغَيْبِ وَٱلشَّهَٰدَةِ ۖ هُوَ ٱلرَّحْمَٰنُ ٱلرَّحِيمُ ﴿٢٢﴾ همغه الله دی چې له هغه پرته هېڅ معبود نشته؛ (هغه) په پټ او ښکاره پوه دی؛ همغه ډېر مهربان، بې‌حده رحم کوونکی دی.
p هُوَ ٱللَّهُ ٱلَّذِى لَآ إِلَٰهَ إِلَّا هُوَ ٱلْمَلِكُ ٱلْقُدُّوسُ ٱلسَّلَٰمُ ٱلْمُؤْمِنُ ٱلْمُهَيْمِنُ ٱلْعَزِيزُ ٱلْجَبَّارُ ٱلْمُتَكَبِّرُ ۚ سُبْحَٰنَ ٱللَّهِ عَمَّا يُشْرِكُونَ ﴿٢٣﴾ وہی اللہ ہے جس کے سوا کوئی معبود نہیں (وہ) بادشاہ ہے (ہر عیب سے) پاک ہے (ہر نقص سے) سالم امن دینے والا نگہبان بہت غالب زبردست طاقت والا بڑائی والا ہے اللہ اُس سے پاک ہے جو وہ (اُس کے ساتھ) شریک ٹھہراتے ہیں۔ || هُوَ ٱللَّهُ ٱلَّذِى لَآ إِلَٰهَ إِلَّا هُوَ ٱلْمَلِكُ ٱلْقُدُّوسُ ٱلسَّلَٰمُ ٱلْمُؤْمِنُ ٱلْمُهَيْمِنُ ٱلْعَزِيزُ ٱلْجَبَّارُ ٱلْمُتَكَبِّرُ ۚ سُبْحَٰنَ ٱللَّهِ عَمَّا يُشْرِكُونَ ﴿٢٣﴾ He is Allah, than Whom there is no other Allah, the Sovereign Lord, the Holy One, Peace, the Keeper of Faith, the Guardian, the Majestic, the Compeller, the Superb. Glorified be Allah from all that they ascribe as partner (unto Him). (59:23) || هُوَ ٱللَّهُ ٱلَّذِى لَآ إِلَٰهَ إِلَّا هُوَ ٱلْمَلِكُ ٱلْقُدُّوسُ ٱلسَّلَٰمُ ٱلْمُؤْمِنُ ٱلْمُهَيْمِنُ ٱلْعَزِيزُ ٱلْجَبَّارُ ٱلْمُتَكَبِّرُ ۚ سُبْحَٰنَ ٱللَّهِ عَمَّا يُشْرِكُونَ ﴿٢٣﴾ همغه الله دی چې له هغه پرته هېڅ معبود نشته؛ (هغه) پاچا دی، (له هر عیب) پاک، (له هر نقص) روغ، امن ورکوونکی، څارونکی، ډېر غالب، زورور ځواکمن، د لوییو خاوند؛ الله له هغه څه پاک دی چې هغوی (ورسره) شریکوي.
p هُوَ ٱللَّهُ ٱلْخَٰلِقُ ٱلْبَارِئُ ٱلْمُصَوِّرُ ۖ لَهُ ٱلْأَسْمَآءُ ٱلْحُسْنَىٰ ۚ يُسَبِّحُ لَهُۥ مَا فِى ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ ۖ وَهُوَ ٱلْعَزِيزُ ٱلْحَكِيمُ ﴿٢٤﴾ وہی اللہ ہے پیدا فرمانے والا (ٹھیک ٹھیک) ایجاد فرمانے والا صورت عطا فرمانے والا اُسی کے لیے تمام اچھے نام ہیں جو کچھ آسمانوں اور زمینوں میں ہے (وہ) سب اُس کی تسبیح کرتے ہیں اور وہی بہت غالب بڑی حکمت والا ہے۔ || هُوَ ٱللَّهُ ٱلْخَٰلِقُ ٱلْبَارِئُ ٱلْمُصَوِّرُ ۖ لَهُ ٱلْأَسْمَآءُ ٱلْحُسْنَىٰ ۚ يُسَبِّحُ لَهُۥ مَا فِى ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ ۖ وَهُوَ ٱلْعَزِيزُ ٱلْحَكِيمُ ﴿٢٤﴾ He is Allah, the Creator, the Shaper out of naught, the Fashioner. His are the most beautiful names. All that is in the heavens and the earth glorifieth Him, and He is the Mighty, the Wise. (59:24) || هُوَ ٱللَّهُ ٱلْخَٰلِقُ ٱلْبَارِئُ ٱلْمُصَوِّرُ ۖ لَهُ ٱلْأَسْمَآءُ ٱلْحُسْنَىٰ ۚ يُسَبِّحُ لَهُۥ مَا فِى ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ ۖ وَهُوَ ٱلْعَزِيزُ ٱلْحَكِيمُ ﴿٢٤﴾ همغه الله دی، پیدا کوونکی، (سم سم) جوړوونکی، بڼه ورکوونکی؛ ټول ښه نومونه د هماغه دي؛ څه چې په آسمانونو او ځمکو کې دي، (هغه) ټول د هغه تسبیح وايي؛ او همغه ډېر غالب، ډېر حکمتناک دی.
## علم و عمل کی باتیں || Matters of knowledge and action || د علم او عمل خبرې
- مالِ فے اُس مال کو کہتے ہیں جو دشمن سے بغیر جہاد و قتال کے حاصل ہو، خواہ اس طرح کہ وہ اپنا مال چھوڑ کر بھاگ جائیں یا رضامندی سے بصورت جزیہ و خراج وغیرہ ان سے حاصل ہو۔ مالِ غنیمت کی طرح یہ مجاہدین میں تقسیم نہیں کیا جاتا بلکہ اس مالِ فے میں تصرف کا حق اللہ تعالیٰ اور اس کے رسول صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم کا ہے۔ یہ مال آپ صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم کے قرابت داروں، یتیموں، مساکین، مسافروں، فقراء، مہاجرین، انصار اور ان کے بعد آنے والے مسلمانوں کے لیے بھی ہے۔ || Mal-e-fay' is that wealth which is obtained from the enemy without jihad and fighting, whether in the way that they leave their wealth and flee, or that it is obtained from them willingly in the form of jizyah, kharaj, etc. Like the spoils of war, this wealth is not distributed among the mujahidin; rather, the right of disposal over this mal-e-fay' belongs to Allah Almighty and His Messenger (peace and blessings of Allah be upon him). This wealth is also for the relatives of the Prophet (peace and blessings of Allah be upon him), orphans, the poor, travellers, the needy, the Muhajirin, the Ansar and the Muslims who come after them. || مال فیء هغه مال ته ویل کېږي چې له دښمن څخه له جهاد او قتال پرته ترلاسه شي، که په دې توګه چې خپل مال پرېږدي او وتښتي، یا په رضا د جزیې او خراج او داسې نورو په بڼه ترې ترلاسه شي. دا مال د غنیمت د مال په څېر په مجاهدینو نه وېشل کېږي، بلکې په دې مال فیء کې د تصرف حق د الله تعالی او د هغه د رسول صلی الله علیه وعلی آله واصحابه وسلم دی. دا مال د هغه صلی الله علیه وعلی آله واصحابه وسلم د خپلوانو، یتیمانو، مسکینانو، مسافرو، فقیرانو، مهاجرینو، انصارو او له هغوی وروسته د راتلونکو مسلمانانو لپاره هم دی.
- اسلامی نظامِ معیشت کا ایک اصول یہ ہے کہ دولت صرف مال داروں کے درمیان ہی گردش نہ کرتی رہے، بلکہ مالِ فے، مالِ غنیمت، زکوٰۃ و صدقات وغیرہ کی صورت میں مال و دولت فقراء، غریبوں، مسکینوں، یتیموں اور ضرورت مندوں تک بھی پہنچے تاکہ ان کی ضروریاتِ زندگی بھی پوری ہو سکیں۔ || One principle of the Islamic economic system is that wealth should not keep circulating only among the wealthy; rather, wealth in the form of mal-e-fay', spoils of war, zakat and charity, etc., should also reach the poor, the indigent, the orphans and the needy so that their necessities of life may also be fulfilled. || د اسلامي اقتصادي نظام یو اصل دا دی چې شتمني یوازې د شتمنو ترمنځ ونه ګرځي، بلکې د مال فیء، غنیمت، زکات او صدقو او داسې نورو په بڼه مال او شتمني فقیرانو، غریبانو، مسکینانو، یتیمانو او اړمنو ته هم ورسېږي تر څو د هغوی د ژوند اړتیاوې هم پوره شي.
- قرآنِ حکیم کی طرح سنت و حدیثِ رسول صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم بھی دین میں حجت ہے۔ جیسا کہ اللہ تعالیٰ نے حکم فرمایا ہے "اور جو کچھ رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) تمھیں عطا فرمائیں تو اُسے لے لو اور جس سے تمھیں منع فرمائیں تو (اُس سے) رُک جاؤ۔" اس آیت کی بنا پر رسول اللہ صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم کا ہر حکم قرآنِ حکیم ہی کا حکم ہے اور اس پر عمل کرنا ضروری قرار دیا گیا ہے۔ || Like the Qur'an-e-Hakim, the Sunnah and hadith of the Messenger (peace and blessings of Allah be upon him) are also proof in religion, as Allah Almighty has commanded: "And whatever the Messenger (peace and blessings of Allah be upon him) gives you, take it, and whatever he forbids you, refrain (from it)." On the basis of this verse every command of the Messenger of Allah (peace and blessings of Allah be upon him) is a command of the Qur'an-e-Hakim itself, and acting upon it has been declared necessary. || د قرآن حکیم په څېر د رسول صلی الله علیه وعلی آله واصحابه وسلم سنت او حدیث هم په دین کې حجت دی، لکه څنګه چې الله تعالی حکم کړی: «او رسول (صلی الله علیه وعلی آله واصحابه وسلم) چې څه درکړي، هغه واخلئ، او له څه چې منع کړي، (ترې) ودرېږئ.» د دې آیت پر بنسټ د رسول الله صلی الله علیه وعلی آله واصحابه وسلم هر حکم په حقیقت کې د قرآن حکیم حکم دی، او پرې عمل کول اړین ګرځول شوي.
- جو لوگ اللہ تعالیٰ کو بھلا دیتے ہیں، اللہ تعالیٰ بھی انھیں اپنے آپ سے غافل کر دیتا ہے۔ یہ لوگ اپنے آپ کو محض حیوان کی طرح انسانی جسم کا ایک ڈھانچہ ہی تصور کرتے ہیں اور روحانی تقاضوں سے اکثر غافل رہتے ہیں۔ || Those people who forget Allah Almighty, Allah Almighty too makes them heedless of themselves. These people consider themselves to be merely a frame of the human body like an animal, and remain mostly heedless of spiritual demands. || هغه کسان چې الله تعالی هېروي، الله تعالی هم هغوی له خپل ځان غافلوي؛ دا خلک ځانونه یوازې د څاروي په څېر د انساني بدن یو چوکاټ ګڼي او ډېری وخت له روحاني غوښتنو غافل پاتې کېږي.
- قرآنِ کریم کی تاثیر تو ایسی ہے کہ اگر اللہ تعالیٰ اسے کسی پہاڑ پر نازل فرماتا تو پہاڑ اللہ تعالیٰ کے خوف سے جھک جاتا اور ریزہ ریزہ ہو جاتا۔ کفار کے دل تو اس سے بھی زیادہ سخت ہو گئے کہ وہ قرآنِ کریم پر ایمان نہ لائے۔ غور طلب بات ہے کہ ہمارے دلوں پر قرآنِ کریم کا کتنا اثر ہوتا ہے؟ || The effect of the Qur'an Karim is such that if Allah Almighty had sent it down upon a mountain, the mountain would bow in fear of Allah Almighty and be shattered to pieces. The hearts of the disbelievers had become even harder than that, so that they did not believe in the Qur'an Karim. It is a matter for reflection how much effect the Qur'an Karim has on our hearts. || د قرآن کریم اغېز خو داسې دی چې که الله تعالی دا پر کوم غره نازل کړی وای، نو غر به د الله تعالی له وېرې ټیټ شوی او ذره ذره شوی وای. د کافرانو زړونه له دې هم سخت شول چې پر قرآن کریم یې ایمان رانه وړ. د فکر وړ خبره ده چې زموږ پر زړونو د قرآن کریم څومره اغېز کېږي؟
