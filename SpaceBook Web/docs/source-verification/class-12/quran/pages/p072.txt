p قُلْ يَٰٓأَهْلَ ٱلْكِتَٰبِ هَلْ تَنقِمُونَ مِنَّآ إِلَّآ أَنْ ءَامَنَّا بِٱللَّهِ وَمَآ أُنزِلَ إِلَيْنَا وَمَآ أُنزِلَ مِن قَبْلُ وَأَنَّ أَكْثَرَكُمْ فَٰسِقُونَ ﴿٥٩﴾ آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) فرما دیجیے کہ اے اہلِ کتاب! تمھیں ہماری کون سی بات بُری لگی سوائے اس کے کہ ہم اللہ پر ایمان لائے اور اس (کلام) پر جو ہماری طرف نازل کیا گیا اور اس پر (بھی) جو اس سے پہلے نازل کیا گیا اور بے شک تم میں سے اکثر نافرمان ہیں۔ || قُلْ يَٰٓأَهْلَ ٱلْكِتَٰبِ هَلْ تَنقِمُونَ مِنَّآ إِلَّآ أَنْ ءَامَنَّا بِٱللَّهِ وَمَآ أُنزِلَ إِلَيْنَا وَمَآ أُنزِلَ مِن قَبْلُ وَأَنَّ أَكْثَرَكُمْ فَٰسِقُونَ ﴿٥٩﴾ Say: O People of the Scripture! Do ye blame us for aught else than that we believe in Allah and that which is revealed unto us and that which was revealed aforetime, and because most of you are evil-livers? (5:59) || قُلْ يَٰٓأَهْلَ ٱلْكِتَٰبِ هَلْ تَنقِمُونَ مِنَّآ إِلَّآ أَنْ ءَامَنَّا بِٱللَّهِ وَمَآ أُنزِلَ إِلَيْنَا وَمَآ أُنزِلَ مِن قَبْلُ وَأَنَّ أَكْثَرَكُمْ فَٰسِقُونَ ﴿٥٩﴾ تاسو (صلی الله علیه وعلی آله واصحابه وسلم) ووایئ: «ای اهل کتابو! زموږ کومه خبره درته بده لګېدلې، پرته له دې چې موږ پر الله ایمان راوړی او پر هغه (کلام) چې موږ ته نازل شوی او پر هغه (هم) چې له دې مخکې نازل شوی؟ او بې‌شکه ستاسو ډېری نافرمانه دي.»
p قُلْ هَلْ أُنَبِّئُكُم بِشَرٍّۢ مِّن ذَٰلِكَ مَثُوبَةً عِندَ ٱللَّهِ ۚ مَن لَّعَنَهُ ٱللَّهُ وَغَضِبَ عَلَيْهِ وَجَعَلَ مِنْهُمُ ٱلْقِرَدَةَ وَٱلْخَنَازِيرَ وَعَبَدَ ٱلطَّٰغُوتَ ۚ أُو۟لَٰٓئِكَ شَرٌّۭ مَّكَانًۭا وَأَضَلُّ عَن سَوَآءِ ٱلسَّبِيلِ ﴿٦٠﴾ آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) فرما دیجیے کیا میں تمھیں بتاؤں اللہ کے ہاں سزا کے اعتبار سے اس سے زیادہ بُرا کون ہے؟ (بُرے) وہ لوگ ہیں جن پر اللہ نے لعنت کی اور وہ ان پر غضب ناک ہوا ہے اور اُن میں سے (بعض کو) بندر اور (بعض کو) خنزیر بنا دیا اور جنھوں نے طاغوت کی پوجا کی وہی لوگ ٹھکانے کے لحاظ سے بدترین ہیں اور سیدھی راہ سے بہت ہی بھٹکے ہوئے ہیں۔ || قُلْ هَلْ أُنَبِّئُكُم بِشَرٍّۢ مِّن ذَٰلِكَ مَثُوبَةً عِندَ ٱللَّهِ ۚ مَن لَّعَنَهُ ٱللَّهُ وَغَضِبَ عَلَيْهِ وَجَعَلَ مِنْهُمُ ٱلْقِرَدَةَ وَٱلْخَنَازِيرَ وَعَبَدَ ٱلطَّٰغُوتَ ۚ أُو۟لَٰٓئِكَ شَرٌّۭ مَّكَانًۭا وَأَضَلُّ عَن سَوَآءِ ٱلسَّبِيلِ ﴿٦٠﴾ Shall I tell thee of a worse (case) than theirs for retribution with Allah? (Worse is the case of him) whom Allah hath cursed, him on whom His wrath hath fallen and of whose sort Allah hath turned some to apes and swine, and who serveth idols. Such are in worse plight and further astray from the plain road. (5:60) || قُلْ هَلْ أُنَبِّئُكُم بِشَرٍّۢ مِّن ذَٰلِكَ مَثُوبَةً عِندَ ٱللَّهِ ۚ مَن لَّعَنَهُ ٱللَّهُ وَغَضِبَ عَلَيْهِ وَجَعَلَ مِنْهُمُ ٱلْقِرَدَةَ وَٱلْخَنَازِيرَ وَعَبَدَ ٱلطَّٰغُوتَ ۚ أُو۟لَٰٓئِكَ شَرٌّۭ مَّكَانًۭا وَأَضَلُّ عَن سَوَآءِ ٱلسَّبِيلِ ﴿٦٠﴾ تاسو (صلی الله علیه وعلی آله واصحابه وسلم) ووایئ: «ایا تاسو ته ووایم چې د الله په نزد د سزا له پلوه له دې بد څوک دی؟ (بد) هغه خلک دي چې الله پرې لعنت کړی او پرې غضبناک شوی، او له هغوی (ځینې) یې بیزوګان او (ځینې) یې خنزیران کړل، او هغوی چې د طاغوت عبادت یې وکړ — همدوی د ټیکانې له پلوه بدترین دي او له سمې لارې ډېر اوښتي دي.»
p وَإِذَا جَآءُوكُمْ قَالُوٓا۟ ءَامَنَّا وَقَد دَّخَلُوا۟ بِٱلْكُفْرِ وَهُمْ قَدْ خَرَجُوا۟ بِهِۦ ۚ وَٱللَّهُ أَعْلَمُ بِمَا كَانُوا۟ يَكْتُمُونَ ﴿٦١﴾ اور (اے مسلمانو!) جب وہ (منافقین) تمھارے پاس آتے ہیں تو کہتے ہیں کہ ہم ایمان لے آئے حالاں کہ یقیناً وہ (تمھارے پاس) کفر ہی کے ساتھ داخل ہوئے اور وہ یقیناً اسی (کفر) کے ساتھ نکلے اور اللہ خوب جانتا ہے جو وہ چھپاتے ہیں۔ || وَإِذَا جَآءُوكُمْ قَالُوٓا۟ ءَامَنَّا وَقَد دَّخَلُوا۟ بِٱلْكُفْرِ وَهُمْ قَدْ خَرَجُوا۟ بِهِۦ ۚ وَٱللَّهُ أَعْلَمُ بِمَا كَانُوا۟ يَكْتُمُونَ ﴿٦١﴾ When they come unto you (Muslims), they say: We believe; but they came in unbelief and they went out in the same; and Allah knoweth best what they were hiding. (5:61) || وَإِذَا جَآءُوكُمْ قَالُوٓا۟ ءَامَنَّا وَقَد دَّخَلُوا۟ بِٱلْكُفْرِ وَهُمْ قَدْ خَرَجُوا۟ بِهِۦ ۚ وَٱللَّهُ أَعْلَمُ بِمَا كَانُوا۟ يَكْتُمُونَ ﴿٦١﴾ او (ای مسلمانانو!) کله چې هغوی (منافقان) تاسو ته راشي، وایي: «موږ ایمان راوړ»، حال دا چې یقیناً هغوی (تاسو ته) له کفر سره ننوتل او یقیناً له همدې (کفر) سره ووتل؛ او الله ښه پوهېږي چې هغوی څه پټوي.
p وَتَرَىٰ كَثِيرًۭا مِّنْهُمْ يُسَٰرِعُونَ فِى ٱلْإِثْمِ وَٱلْعُدْوَٰنِ وَأَكْلِهِمُ ٱلسُّحْتَ ۚ لَبِئْسَ مَا كَانُوا۟ يَعْمَلُونَ ﴿٦٢﴾ اور آپ اُن میں سے بہت سے لوگوں کو دیکھتے ہیں کہ وہ گناہ اور زیادتی اور اپنی حرام خوری میں تیزی دکھاتے ہیں یقیناً بہت بُرا ہے جو کچھ وہ کر رہے ہیں۔ || وَتَرَىٰ كَثِيرًۭا مِّنْهُمْ يُسَٰرِعُونَ فِى ٱلْإِثْمِ وَٱلْعُدْوَٰنِ وَأَكْلِهِمُ ٱلسُّحْتَ ۚ لَبِئْسَ مَا كَانُوا۟ يَعْمَلُونَ ﴿٦٢﴾ And thou seest many of them vying one with another in sin and transgression and their devouring of illicit gain. Verily evil is what they do. (5:62) || وَتَرَىٰ كَثِيرًۭا مِّنْهُمْ يُسَٰرِعُونَ فِى ٱلْإِثْمِ وَٱلْعُدْوَٰنِ وَأَكْلِهِمُ ٱلسُّحْتَ ۚ لَبِئْسَ مَا كَانُوا۟ يَعْمَلُونَ ﴿٦٢﴾ او تاسو له هغوی ډېر خلک وینئ چې په ګناه، تېري او خپل حرام خوړلو کې بېړه کوي؛ یقیناً ډېر بد دي هغه څه چې هغوی یې کوي.
p لَوْلَا يَنْهَىٰهُمُ ٱلرَّبَّٰنِيُّونَ وَٱلْأَحْبَارُ عَن قَوْلِهِمُ ٱلْإِثْمَ وَأَكْلِهِمُ ٱلسُّحْتَ ۚ لَبِئْسَ مَا كَانُوا۟ يَصْنَعُونَ ﴿٦٣﴾ انھیں اللہ والے اور (ان کے) علماء کیوں منع نہیں کرتے اُن کی گناہ کی بات اور اُن کی حرام خوری سے یقیناً بہت بُرا ہے جو کچھ وہ کر رہے ہیں۔ || لَوْلَا يَنْهَىٰهُمُ ٱلرَّبَّٰنِيُّونَ وَٱلْأَحْبَارُ عَن قَوْلِهِمُ ٱلْإِثْمَ وَأَكْلِهِمُ ٱلسُّحْتَ ۚ لَبِئْسَ مَا كَانُوا۟ يَصْنَعُونَ ﴿٦٣﴾ Why do not the rabbis and the priests forbid their evil-speaking and their devouring of illicit gain? Verily evil is their handiwork. (5:63) || لَوْلَا يَنْهَىٰهُمُ ٱلرَّبَّٰنِيُّونَ وَٱلْأَحْبَارُ عَن قَوْلِهِمُ ٱلْإِثْمَ وَأَكْلِهِمُ ٱلسُّحْتَ ۚ لَبِئْسَ مَا كَانُوا۟ يَصْنَعُونَ ﴿٦٣﴾ هغوی ته الله والا او (د هغوی) علماء ولې د ګناه له خبرې او له حرام خوړلو منع نه کوي؟ یقیناً ډېر بد دي هغه څه چې هغوی یې کوي.
p وَقَالَتِ ٱلْيَهُودُ يَدُ ٱللَّهِ مَغْلُولَةٌ ۚ غُلَّتْ أَيْدِيهِمْ وَلُعِنُوا۟ بِمَا قَالُوا۟ ۘ بَلْ يَدَاهُ مَبْسُوطَتَانِ يُنفِقُ كَيْفَ يَشَآءُ ۚ وَلَيَزِيدَنَّ كَثِيرًۭا مِّنْهُم مَّآ أُنزِلَ إِلَيْكَ مِن رَّبِّكَ طُغْيَٰنًۭا وَكُفْرًۭا ۚ وَأَلْقَيْنَا بَيْنَهُمُ ٱلْعَدَٰوَةَ وَٱلْبَغْضَآءَ إِلَىٰ يَوْمِ ٱلْقِيَٰمَةِ ۚ كُلَّمَآ أَوْقَدُوا۟ نَارًۭا لِّلْحَرْبِ أَطْفَأَهَا ٱللَّهُ ۚ وَيَسْعَوْنَ فِى ٱلْأَرْضِ فَسَادًۭا ۚ وَٱللَّهُ لَا يُحِبُّ ٱلْمُفْسِدِينَ ﴿٦٤﴾ اور یہود نے کہا کہ اللہ کا ہاتھ بندھا ہوا ہے خود ان کے (اپنے) ہاتھ بندھ جائیں اور جو کچھ انھوں نے (گستاخی کرتے ہوئے) کہا اس کی وجہ سے ان پر لعنت ہو بلکہ اس (اللہ) کے دونوں ہاتھ کشادہ ہیں وہ جیسے چاہتا ہے عطا فرماتا ہے اور جو (کلام) آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) پر آپ کے رب کی طرف سے نازل کیا گیا ہے (وہ) یقیناً ان میں سے اکثر لوگوں کی سرکشی اور کفر میں ضرور مزید اضافہ کرے گا اور ہم نے اُن کے درمیان روزِ قیامت تک عداوت اور بغض ڈال دیا ہے جب کبھی وہ جنگ کی آگ بھڑکاتے ہیں اللہ اُسے بجھا دیتا ہے اور یہ لوگ زمین میں فساد پھیلانے کی کوشش کرتے ہیں اور اللہ فساد کرنے والوں کو پسند نہیں کرتا۔ || وَقَالَتِ ٱلْيَهُودُ يَدُ ٱللَّهِ مَغْلُولَةٌ ۚ غُلَّتْ أَيْدِيهِمْ وَلُعِنُوا۟ بِمَا قَالُوا۟ ۘ بَلْ يَدَاهُ مَبْسُوطَتَانِ يُنفِقُ كَيْفَ يَشَآءُ ۚ وَلَيَزِيدَنَّ كَثِيرًۭا مِّنْهُم مَّآ أُنزِلَ إِلَيْكَ مِن رَّبِّكَ طُغْيَٰنًۭا وَكُفْرًۭا ۚ وَأَلْقَيْنَا بَيْنَهُمُ ٱلْعَدَٰوَةَ وَٱلْبَغْضَآءَ إِلَىٰ يَوْمِ ٱلْقِيَٰمَةِ ۚ كُلَّمَآ أَوْقَدُوا۟ نَارًۭا لِّلْحَرْبِ أَطْفَأَهَا ٱللَّهُ ۚ وَيَسْعَوْنَ فِى ٱلْأَرْضِ فَسَادًۭا ۚ وَٱللَّهُ لَا يُحِبُّ ٱلْمُفْسِدِينَ ﴿٦٤﴾ The Jews say: Allah's hand is fettered. Their hands are fettered and they are accursed for saying so. Nay, but both His hands are spread out wide in bounty. He bestoweth as He will. That which hath been revealed unto thee from thy Lord is certain to increase the contumacy and disbelief of many of them, and We have cast among them enmity and hatred till the Day of Resurrection. As often as they light a fire for war, Allah extinguisheth it. Their effort is for corruption in the land, and Allah loveth not corrupters. (5:64) || وَقَالَتِ ٱلْيَهُودُ يَدُ ٱللَّهِ مَغْلُولَةٌ ۚ غُلَّتْ أَيْدِيهِمْ وَلُعِنُوا۟ بِمَا قَالُوا۟ ۘ بَلْ يَدَاهُ مَبْسُوطَتَانِ يُنفِقُ كَيْفَ يَشَآءُ ۚ وَلَيَزِيدَنَّ كَثِيرًۭا مِّنْهُم مَّآ أُنزِلَ إِلَيْكَ مِن رَّبِّكَ طُغْيَٰنًۭا وَكُفْرًۭا ۚ وَأَلْقَيْنَا بَيْنَهُمُ ٱلْعَدَٰوَةَ وَٱلْبَغْضَآءَ إِلَىٰ يَوْمِ ٱلْقِيَٰمَةِ ۚ كُلَّمَآ أَوْقَدُوا۟ نَارًۭا لِّلْحَرْبِ أَطْفَأَهَا ٱللَّهُ ۚ وَيَسْعَوْنَ فِى ٱلْأَرْضِ فَسَادًۭا ۚ وَٱللَّهُ لَا يُحِبُّ ٱلْمُفْسِدِينَ ﴿٦٤﴾ او یهودو وویل: «د الله لاس تړلی دی.» پخپله د هغوی لاسونه دې وتړل شي او د هغه څه له امله چې (په بې‌ادبۍ) یې وویل، پر هغوی دې لعنت وي؛ بلکې د هغه (الله) دواړه لاسونه پراخ دي، څنګه چې وغواړي ورکوي؛ او هغه (کلام) چې تاسو (صلی الله علیه وعلی آله واصحابه وسلم) ته ستاسو د رب له لوري نازل شوی، (هغه به) یقیناً د هغوی د ډېرو خلکو په سرکښۍ او کفر کې خامخا نور زیاتوالی راولي؛ او موږ د هغوی ترمنځ د قیامت تر ورځې دښمني او کینه اچولې؛ هرکله چې د جګړې اور بل کړي، الله یې مړ کړي؛ او دا خلک په ځمکه کې د فساد خپرولو هڅه کوي، او الله فساد کوونکي نه خوښوي.
