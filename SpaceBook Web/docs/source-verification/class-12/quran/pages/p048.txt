p يَسْـَٔلُكَ أَهْلُ ٱلْكِتَٰبِ أَن تُنَزِّلَ عَلَيْهِمْ كِتَٰبًۭا مِّنَ ٱلسَّمَآءِ ۚ فَقَدْ سَأَلُوا۟ مُوسَىٰٓ أَكْبَرَ مِن ذَٰلِكَ فَقَالُوٓا۟ أَرِنَا ٱللَّهَ جَهْرَةًۭ فَأَخَذَتْهُمُ ٱلصَّٰعِقَةُ بِظُلْمِهِمْ ۚ ثُمَّ ٱتَّخَذُوا۟ ٱلْعِجْلَ مِنۢ بَعْدِ مَا جَآءَتْهُمُ ٱلْبَيِّنَٰتُ فَعَفَوْنَا عَن ذَٰلِكَ ۚ وَءَاتَيْنَا مُوسَىٰ سُلْطَٰنًۭا مُّبِينًۭا ﴿١٥٣﴾ آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) سے اہلِ کتاب مطالبہ کرتے ہیں کہ آپ اُن پر آسمان سے ایک کتاب اتار لائیں پس وہ تو موسیٰ (علیہ السلام) سے اس سے زیادہ بڑا مطالبہ کر چکے ہیں تو ان لوگوں نے (موسیٰ علیہ السلام سے) کہا کہ ہمیں اللہ دکھا دو علانیہ طور پر تو اُن کے ظلم کی وجہ سے انھیں کڑک نے آ پکڑا پھر انھوں نے اس کے بعد بھی کہ ان کے پاس واضح دلائل آ چکے تھے بچھڑے کو (معبود) بنایا تو ہم نے اس (سنگین جرم) کو معاف کر دیا اور ہم نے موسیٰ (علیہ السلام) کو واضح غلبہ عطا فرمایا۔ || يَسْـَٔلُكَ أَهْلُ ٱلْكِتَٰبِ أَن تُنَزِّلَ عَلَيْهِمْ كِتَٰبًۭا مِّنَ ٱلسَّمَآءِ ۚ فَقَدْ سَأَلُوا۟ مُوسَىٰٓ أَكْبَرَ مِن ذَٰلِكَ فَقَالُوٓا۟ أَرِنَا ٱللَّهَ جَهْرَةًۭ فَأَخَذَتْهُمُ ٱلصَّٰعِقَةُ بِظُلْمِهِمْ ۚ ثُمَّ ٱتَّخَذُوا۟ ٱلْعِجْلَ مِنۢ بَعْدِ مَا جَآءَتْهُمُ ٱلْبَيِّنَٰتُ فَعَفَوْنَا عَن ذَٰلِكَ ۚ وَءَاتَيْنَا مُوسَىٰ سُلْطَٰنًۭا مُّبِينًۭا ﴿١٥٣﴾ The people of the Scripture ask of thee that thou shouldst cause an (actual) Book to descend upon them from heaven. They asked a greater thing of Moses aforetime, for they said: Show us Allah plainly. The storm of lightning seized them for their wickedness. Then (even) after that) they chose the calf (for worship) after clear proofs (of Allah's Sovereignty) had come unto them. And We forgave them that! And We bestowed on Moses evident authority. (4:153) || يَسْـَٔلُكَ أَهْلُ ٱلْكِتَٰبِ أَن تُنَزِّلَ عَلَيْهِمْ كِتَٰبًۭا مِّنَ ٱلسَّمَآءِ ۚ فَقَدْ سَأَلُوا۟ مُوسَىٰٓ أَكْبَرَ مِن ذَٰلِكَ فَقَالُوٓا۟ أَرِنَا ٱللَّهَ جَهْرَةًۭ فَأَخَذَتْهُمُ ٱلصَّٰعِقَةُ بِظُلْمِهِمْ ۚ ثُمَّ ٱتَّخَذُوا۟ ٱلْعِجْلَ مِنۢ بَعْدِ مَا جَآءَتْهُمُ ٱلْبَيِّنَٰتُ فَعَفَوْنَا عَن ذَٰلِكَ ۚ وَءَاتَيْنَا مُوسَىٰ سُلْطَٰنًۭا مُّبِينًۭا ﴿١٥٣﴾ اهل کتاب له تاسو (صلی الله علیه وعلی آله واصحابه وسلم) غوښتنه کوي چې پر هغوی له اسمانه یو کتاب راکوز کړئ؛ نو هغوی خو له موسی (علیه السلام) له دې هم لویه غوښتنه کړې وه: هغو خلکو (موسی علیه السلام ته) وویل: «موږ ته الله په ښکاره وښیه»، نو د هغوی د ظلم له امله تندر ونیول؛ بیا هغوی سره له دې چې ښکاره دلیلونه ورته راغلي وو، سخی (معبود) ونیو؛ نو موږ له دې (لوی جرم) تېر شو، او موږ موسی (علیه السلام) ته ښکاره غلبه ورکړه.
p وَرَفَعْنَا فَوْقَهُمُ ٱلطُّورَ بِمِيثَٰقِهِمْ وَقُلْنَا لَهُمُ ٱدْخُلُوا۟ ٱلْبَابَ سُجَّدًۭا وَقُلْنَا لَهُمْ لَا تَعْدُوا۟ فِى ٱلسَّبْتِ وَأَخَذْنَا مِنْهُم مِّيثَٰقًا غَلِيظًۭا ﴿١٥٤﴾ اور ہم نے اُن سے عہد لینے کے لیے اُن پر طور (پہاڑ) بلند کیا اور ہم نے اُن سے کہا کہ (شہر کے) دروازے سے سروں کو جھکاتے ہوئے داخل ہونا اور ہم نے اُن سے کہا تھا کہ ہفتہ کے دن کے بارے میں حد سے نہ بڑھنا اور ہم نے ان سے پختہ عہد لیا۔ || وَرَفَعْنَا فَوْقَهُمُ ٱلطُّورَ بِمِيثَٰقِهِمْ وَقُلْنَا لَهُمُ ٱدْخُلُوا۟ ٱلْبَابَ سُجَّدًۭا وَقُلْنَا لَهُمْ لَا تَعْدُوا۟ فِى ٱلسَّبْتِ وَأَخَذْنَا مِنْهُم مِّيثَٰقًا غَلِيظًۭا ﴿١٥٤﴾ And We caused the Mount to tower above them at (the taking of) their covenant: and We bade them: Enter the gate, prostrate! and We bode them: Transgress not the Sabbath! and We took from them a firm covenant. (4:154) || وَرَفَعْنَا فَوْقَهُمُ ٱلطُّورَ بِمِيثَٰقِهِمْ وَقُلْنَا لَهُمُ ٱدْخُلُوا۟ ٱلْبَابَ سُجَّدًۭا وَقُلْنَا لَهُمْ لَا تَعْدُوا۟ فِى ٱلسَّبْتِ وَأَخَذْنَا مِنْهُم مِّيثَٰقًا غَلِيظًۭا ﴿١٥٤﴾ او موږ د هغوی څخه د ژمنې اخیستو لپاره پر هغوی طور (غر) پورته کړ، او موږ ورته وویل چې د (ښار) له دروازې سرونه ټیټ ننوځئ، او موږ ورته ویلي وو چې د شنبې د ورځې په اړه له حده مه اوړئ، او موږ ترې ټینګه ژمنه واخیسته.
p فَبِمَا نَقْضِهِم مِّيثَٰقَهُمْ وَكُفْرِهِم بِـَٔايَٰتِ ٱللَّهِ وَقَتْلِهِمُ ٱلْأَنۢبِيَآءَ بِغَيْرِ حَقٍّۢ وَقَوْلِهِمْ قُلُوبُنَا غُلْفٌۢ ۚ بَلْ طَبَعَ ٱللَّهُ عَلَيْهَا بِكُفْرِهِمْ فَلَا يُؤْمِنُونَ إِلَّا قَلِيلًۭا ﴿١٥٥﴾ تو (ان پر لعنت کی گئی) اُن کے اپنے عہد کو توڑنے پر اور آیاتِ الٰہی سے انکار پر اور انبیاء کو ان کے ناحق قتل (شہید) کر دینے پر اور اُن کی اس بات پر کہ ہمارے دل غلاف میں (بند) ہیں (ایسا نہیں ہے) بلکہ اللہ نے ان کے کفر کی وجہ سے اُن پر مہر لگا دی پس چند لوگوں کے سوا وہ ایمان نہیں لائیں گے۔ || فَبِمَا نَقْضِهِم مِّيثَٰقَهُمْ وَكُفْرِهِم بِـَٔايَٰتِ ٱللَّهِ وَقَتْلِهِمُ ٱلْأَنۢبِيَآءَ بِغَيْرِ حَقٍّۢ وَقَوْلِهِمْ قُلُوبُنَا غُلْفٌۢ ۚ بَلْ طَبَعَ ٱللَّهُ عَلَيْهَا بِكُفْرِهِمْ فَلَا يُؤْمِنُونَ إِلَّا قَلِيلًۭا ﴿١٥٥﴾ Then because of their breaking of their covenant, and their disbelieving in the revelations of Allah, and their slaying of the prophets wrongfully, and their saying: Our hearts are hardened - Nay, but Allah set a seal upon them for their disbelief, so that they believe not save a few - (4:155) || فَبِمَا نَقْضِهِم مِّيثَٰقَهُمْ وَكُفْرِهِم بِـَٔايَٰتِ ٱللَّهِ وَقَتْلِهِمُ ٱلْأَنۢبِيَآءَ بِغَيْرِ حَقٍّۢ وَقَوْلِهِمْ قُلُوبُنَا غُلْفٌۢ ۚ بَلْ طَبَعَ ٱللَّهُ عَلَيْهَا بِكُفْرِهِمْ فَلَا يُؤْمِنُونَ إِلَّا قَلِيلًۭا ﴿١٥٥﴾ نو (پر هغوی لعنت وشو) د خپلې ژمنې د ماتولو، د الهي آیتونو د انکار، د انبیاوو د ناحقه وژلو (شهیدولو)، او د دې خبرې له امله چې «زموږ زړونه په پوښ کې (تړلي) دي» — (داسې نه ده) بلکې الله د هغوی د کفر له امله پرې مهر لګولی؛ نو له څو کسانو پرته به ایمان رانه وړي.
p وَبِكُفْرِهِمْ وَقَوْلِهِمْ عَلَىٰ مَرْيَمَ بُهْتَٰنًا عَظِيمًۭا ﴿١٥٦﴾ اور (ان پر لعنت کی گئی) اُن کے کفر اور اُن کے مریم (علیہا السلام) پر بہت بڑے بہتان کی بات کہنے کی وجہ سے۔ || وَبِكُفْرِهِمْ وَقَوْلِهِمْ عَلَىٰ مَرْيَمَ بُهْتَٰنًا عَظِيمًۭا ﴿١٥٦﴾ And because of their disbelief and of their speaking against Mary a tremendous calumny; (4:156) || وَبِكُفْرِهِمْ وَقَوْلِهِمْ عَلَىٰ مَرْيَمَ بُهْتَٰنًا عَظِيمًۭا ﴿١٥٦﴾ او (پر هغوی لعنت وشو) د هغوی د کفر او پر مریم (علیها السلام) د ډېر لوی بهتان ویلو له امله.
p وَقَوْلِهِمْ إِنَّا قَتَلْنَا ٱلْمَسِيحَ عِيسَى ٱبْنَ مَرْيَمَ رَسُولَ ٱللَّهِ وَمَا قَتَلُوهُ وَمَا صَلَبُوهُ وَلَٰكِن شُبِّهَ لَهُمْ ۚ وَإِنَّ ٱلَّذِينَ ٱخْتَلَفُوا۟ فِيهِ لَفِى شَكٍّۢ مِّنْهُ ۚ مَا لَهُم بِهِۦ مِنْ عِلْمٍ إِلَّا ٱتِّبَاعَ ٱلظَّنِّ ۚ وَمَا قَتَلُوهُ يَقِينًۢا ﴿١٥٧﴾ اور (ان پر لعنت کی گئی) اُن کے اس قول کی وجہ سے کہ بے شک ہم نے مسیح عیسیٰ ابن مریم (علیہ السلام) (یعنی) اللہ کے رسول کو قتل کیا جب کہ انھوں نے قتل نہیں کیا ان کو اور نہ انھیں سولی چڑھایا مگر (ہوا یہ کہ) ان کے لیے (کسی کو عیسیٰ علیہ السلام کا) مشابہ بنا دیا گیا اور بے شک جن لوگوں نے اس بارے میں اختلاف کیا یقیناً وہ اس معاملے میں شک میں مبتلا ہیں انھیں اس کا کچھ علم نہیں سوائے گمان کی پیروی کرنے کے اور انھوں نے یقیناً ان کو قتل نہیں کیا۔ || وَقَوْلِهِمْ إِنَّا قَتَلْنَا ٱلْمَسِيحَ عِيسَى ٱبْنَ مَرْيَمَ رَسُولَ ٱللَّهِ وَمَا قَتَلُوهُ وَمَا صَلَبُوهُ وَلَٰكِن شُبِّهَ لَهُمْ ۚ وَإِنَّ ٱلَّذِينَ ٱخْتَلَفُوا۟ فِيهِ لَفِى شَكٍّۢ مِّنْهُ ۚ مَا لَهُم بِهِۦ مِنْ عِلْمٍ إِلَّا ٱتِّبَاعَ ٱلظَّنِّ ۚ وَمَا قَتَلُوهُ يَقِينًۢا ﴿١٥٧﴾ And because of their saying: We slew the Messiah, Jesus son of Mary, Allah's messenger - they slew him not nor crucified him, but it appeared so unto them; and lo! those who disagree concerning it are in doubt thereof; they have no knowledge thereof save pursuit of a conjecture; they slew him not for certain. (4:157) || وَقَوْلِهِمْ إِنَّا قَتَلْنَا ٱلْمَسِيحَ عِيسَى ٱبْنَ مَرْيَمَ رَسُولَ ٱللَّهِ وَمَا قَتَلُوهُ وَمَا صَلَبُوهُ وَلَٰكِن شُبِّهَ لَهُمْ ۚ وَإِنَّ ٱلَّذِينَ ٱخْتَلَفُوا۟ فِيهِ لَفِى شَكٍّۢ مِّنْهُ ۚ مَا لَهُم بِهِۦ مِنْ عِلْمٍ إِلَّا ٱتِّبَاعَ ٱلظَّنِّ ۚ وَمَا قَتَلُوهُ يَقِينًۢا ﴿١٥٧﴾ او (پر هغوی لعنت وشو) د دې ویلو له امله چې «بې‌شکه موږ مسیح عیسی ابن مریم (علیه السلام)، (یعنې) د الله رسول ووژه» — حال دا چې هغوی نه دی وژلی او نه یې پر دار ځړولی، بلکې (داسې وشول چې) د هغوی لپاره (یو کس د عیسی علیه السلام) ورته کړای شو؛ او بې‌شکه هغه کسان چې په دې اړه یې اختلاف وکړ، یقیناً په دې چاره کې په شک کې دي؛ هغوی ته له ګومان له پیروۍ پرته پرې هېڅ علم نشته؛ او یقیناً یې نه دی وژلی.
p بَل رَّفَعَهُ ٱللَّهُ إِلَيْهِ ۚ وَكَانَ ٱللَّهُ عَزِيزًا حَكِيمًۭا ﴿١٥٨﴾ بلکہ اللہ نے انھیں اپنی طرف اٹھا لیا اور اللہ بڑا غالب بڑی حکمت والا ہے۔ || بَل رَّفَعَهُ ٱللَّهُ إِلَيْهِ ۚ وَكَانَ ٱللَّهُ عَزِيزًا حَكِيمًۭا ﴿١٥٨﴾ But Allah took him up unto Himself. Allah was ever Mighty, Wise. (4:158) || بَل رَّفَعَهُ ٱللَّهُ إِلَيْهِ ۚ وَكَانَ ٱللَّهُ عَزِيزًا حَكِيمًۭا ﴿١٥٨﴾ بلکې الله هغه خپل لوري ته پورته کړ، او الله ډېر غالب، ډېر حکمتناک دی.
