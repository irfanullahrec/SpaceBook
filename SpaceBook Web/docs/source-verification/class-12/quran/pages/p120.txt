p لَّا جُنَاحَ عَلَيْهِنَّ فِىٓ ءَابَآئِهِنَّ وَلَآ أَبْنَآئِهِنَّ وَلَآ إِخْوَٰنِهِنَّ وَلَآ أَبْنَآءِ إِخْوَٰنِهِنَّ وَلَآ أَبْنَآءِ أَخَوَٰتِهِنَّ وَلَا نِسَآئِهِنَّ وَلَا مَا مَلَكَتْ أَيْمَٰنُهُنَّ ۗ وَٱتَّقِينَ ٱللَّهَ ۚ إِنَّ ٱللَّهَ كَانَ عَلَىٰ كُلِّ شَىْءٍۢ شَهِيدًا ﴿٥٥﴾ ان (ازواجِ مطہرات) پر (پردہ نہ کرنے میں) کوئی حرج نہیں اپنے باپوں سے اور نہ اپنے بیٹوں سے اور نہ اپنے بھائیوں سے اور نہ اپنے بھتیجوں سے اور نہ اپنے بھانجوں سے اور نہ اپنی جیسی عورتوں سے اور نہ اپنی کنیزوں سے اور (اے ازواجِ نبی صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم!) اللہ (کی نافرمانی) سے ڈرو بے شک اللہ ہر چیز پر گواہ ہے۔ || لَّا جُنَاحَ عَلَيْهِنَّ فِىٓ ءَابَآئِهِنَّ وَلَآ أَبْنَآئِهِنَّ وَلَآ إِخْوَٰنِهِنَّ وَلَآ أَبْنَآءِ إِخْوَٰنِهِنَّ وَلَآ أَبْنَآءِ أَخَوَٰتِهِنَّ وَلَا نِسَآئِهِنَّ وَلَا مَا مَلَكَتْ أَيْمَٰنُهُنَّ ۗ وَٱتَّقِينَ ٱللَّهَ ۚ إِنَّ ٱللَّهَ كَانَ عَلَىٰ كُلِّ شَىْءٍۢ شَهِيدًا ﴿٥٥﴾ It is no sin for them (thy wives) to converse freely) with their fathers, or their sons, or their brothers, or their brothers' sons, or the sons of their sisters or of their own women, or their slaves. O women! Keep your duty to Allah. Lo! Allah is ever Witness over all things. (33:55) || لَّا جُنَاحَ عَلَيْهِنَّ فِىٓ ءَابَآئِهِنَّ وَلَآ أَبْنَآئِهِنَّ وَلَآ إِخْوَٰنِهِنَّ وَلَآ أَبْنَآءِ إِخْوَٰنِهِنَّ وَلَآ أَبْنَآءِ أَخَوَٰتِهِنَّ وَلَا نِسَآئِهِنَّ وَلَا مَا مَلَكَتْ أَيْمَٰنُهُنَّ ۗ وَٱتَّقِينَ ٱللَّهَ ۚ إِنَّ ٱللَّهَ كَانَ عَلَىٰ كُلِّ شَىْءٍۢ شَهِيدًا ﴿٥٥﴾ پر هغوی (پاکو مېرمنو) (د پردې نه کولو کې) څه حرج نشته له خپلو پلرونو، نه له خپلو زامنو، نه له خپلو وروڼو، نه د خپلو وروڼو له زامنو، نه د خپلو خویندو له زامنو، نه له خپلو په څېر ښځو، او نه له خپلو وینځو؛ او (ای د نبي صلی الله علیه وعلی آله واصحابه وسلم مېرمنو!) له الله (د نافرمانۍ) ووېرېږئ؛ بې‌شکه الله پر هر شي شاهد دی.
p إِنَّ ٱللَّهَ وَمَلَٰٓئِكَتَهُۥ يُصَلُّونَ عَلَى ٱلنَّبِىِّ ۚ يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ صَلُّوا۟ عَلَيْهِ وَسَلِّمُوا۟ تَسْلِيمًا ﴿٥٦﴾ بے شک اللہ اور اُس کے فرشتے نبی (کریم صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) پر درود بھیجتے ہیں اے ایمان والو! تم بھی اُن پر درود بھیجو اور خوب سلام بھیجو۔ || إِنَّ ٱللَّهَ وَمَلَٰٓئِكَتَهُۥ يُصَلُّونَ عَلَى ٱلنَّبِىِّ ۚ يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ صَلُّوا۟ عَلَيْهِ وَسَلِّمُوا۟ تَسْلِيمًا ﴿٥٦﴾ Lo! Allah and His angels shower blessings on the Prophet. O ye who believe! Ask blessings on him and salute him with a worthy salutation. (33:56) || إِنَّ ٱللَّهَ وَمَلَٰٓئِكَتَهُۥ يُصَلُّونَ عَلَى ٱلنَّبِىِّ ۚ يَٰٓأَيُّهَا ٱلَّذِينَ ءَامَنُوا۟ صَلُّوا۟ عَلَيْهِ وَسَلِّمُوا۟ تَسْلِيمًا ﴿٥٦﴾ بې‌شکه الله او د هغه پرښتې پر نبي (کریم صلی الله علیه وعلی آله واصحابه وسلم) درود لېږي؛ ای مؤمنانو! تاسو هم پر هغه درود ولېږئ او ښه سلام ولېږئ.
p إِنَّ ٱلَّذِينَ يُؤْذُونَ ٱللَّهَ وَرَسُولَهُۥ لَعَنَهُمُ ٱللَّهُ فِى ٱلدُّنْيَا وَٱلْءَاخِرَةِ وَأَعَدَّ لَهُمْ عَذَابًۭا مُّهِينًۭا ﴿٥٧﴾ بے شک جو لوگ اللہ اور اُس کے رسول (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کو اذیت پہنچاتے ہیں اللہ نے اُن پر دنیا اور آخرت میں لعنت کی ہے اور اُس نے اُن کے لیے ذلت والا عذاب تیار کر رکھا ہے۔ || إِنَّ ٱلَّذِينَ يُؤْذُونَ ٱللَّهَ وَرَسُولَهُۥ لَعَنَهُمُ ٱللَّهُ فِى ٱلدُّنْيَا وَٱلْءَاخِرَةِ وَأَعَدَّ لَهُمْ عَذَابًۭا مُّهِينًۭا ﴿٥٧﴾ Lo! those who malign Allah and His messenger, Allah hath cursed them in the world and the Hereafter, and hath prepared for them the doom of the disdained. (33:57) || إِنَّ ٱلَّذِينَ يُؤْذُونَ ٱللَّهَ وَرَسُولَهُۥ لَعَنَهُمُ ٱللَّهُ فِى ٱلدُّنْيَا وَٱلْءَاخِرَةِ وَأَعَدَّ لَهُمْ عَذَابًۭا مُّهِينًۭا ﴿٥٧﴾ بې‌شکه هغه کسان چې الله او د هغه رسول (صلی الله علیه وعلی آله واصحابه وسلم) ته اذیت رسوي، الله پر هغوی په دنیا او آخرت کې لعنت کړی، او د هغوی لپاره یې د ذلت عذاب تیار کړی دی.
p وَٱلَّذِينَ يُؤْذُونَ ٱلْمُؤْمِنِينَ وَٱلْمُؤْمِنَٰتِ بِغَيْرِ مَا ٱكْتَسَبُوا۟ فَقَدِ ٱحْتَمَلُوا۟ بُهْتَٰنًۭا وَإِثْمًۭا مُّبِينًۭا ﴿٥٨﴾ اور جو لوگ مومن مردوں اور مومن عورتوں کو اذیت پہنچاتے ہیں بغیر اس کے کہ انھوں نے کچھ (قصور) کیا ہو تو یقیناً انھوں نے بہتان اور واضح گناہ کا بوجھ اٹھا لیا۔ || وَٱلَّذِينَ يُؤْذُونَ ٱلْمُؤْمِنِينَ وَٱلْمُؤْمِنَٰتِ بِغَيْرِ مَا ٱكْتَسَبُوا۟ فَقَدِ ٱحْتَمَلُوا۟ بُهْتَٰنًۭا وَإِثْمًۭا مُّبِينًۭا ﴿٥٨﴾ And those who malign believing men and believing women undeservedly, they bear the guilt of slander and manifest sin. (33:58) || وَٱلَّذِينَ يُؤْذُونَ ٱلْمُؤْمِنِينَ وَٱلْمُؤْمِنَٰتِ بِغَيْرِ مَا ٱكْتَسَبُوا۟ فَقَدِ ٱحْتَمَلُوا۟ بُهْتَٰنًۭا وَإِثْمًۭا مُّبِينًۭا ﴿٥٨﴾ او هغه کسان چې مؤمنو نارینه وو او مؤمنو ښځو ته اذیت رسوي پرته له دې چې هغوی څه (قصور) کړی وي، نو یقیناً هغوی د بهتان او ښکاره ګناه بار پر غاړه واخیست.
p يَٰٓأَيُّهَا ٱلنَّبِىُّ قُل لِّأَزْوَٰجِكَ وَبَنَاتِكَ وَنِسَآءِ ٱلْمُؤْمِنِينَ يُدْنِينَ عَلَيْهِنَّ مِن جَلَٰبِيبِهِنَّ ۚ ذَٰلِكَ أَدْنَىٰٓ أَن يُعْرَفْنَ فَلَا يُؤْذَيْنَ ۗ وَكَانَ ٱللَّهُ غَفُورًۭا رَّحِيمًۭا ﴿٥٩﴾ اے نبی (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم)! آپ فرما دیں اپنی ازواج (مطہرات) سے اور اپنی صاحبزادیوں سے اور مومنوں کی عورتوں سے کہ وہ اپنی چادروں کا ایک حصہ اپنے اوپر (سے) لٹکا لیا کریں یہ (پردہ) اس کے زیادہ قریب ہے کہ وہ پہچان لی جائیں (کہ وہ شریف عورتیں ہیں) تو انھیں ستایا نہ جائے اور اللہ بہت بخشنے والا نہایت رحم فرمانے والا ہے۔ || يَٰٓأَيُّهَا ٱلنَّبِىُّ قُل لِّأَزْوَٰجِكَ وَبَنَاتِكَ وَنِسَآءِ ٱلْمُؤْمِنِينَ يُدْنِينَ عَلَيْهِنَّ مِن جَلَٰبِيبِهِنَّ ۚ ذَٰلِكَ أَدْنَىٰٓ أَن يُعْرَفْنَ فَلَا يُؤْذَيْنَ ۗ وَكَانَ ٱللَّهُ غَفُورًۭا رَّحِيمًۭا ﴿٥٩﴾ O Prophet! Tell thy wives and thy daughters and the women of the believers to draw their cloaks close round them (when they go abroad). That will be better, so that they may be recognised and not annoyed. Allah is ever Forgiving, Merciful. (33:59) || يَٰٓأَيُّهَا ٱلنَّبِىُّ قُل لِّأَزْوَٰجِكَ وَبَنَاتِكَ وَنِسَآءِ ٱلْمُؤْمِنِينَ يُدْنِينَ عَلَيْهِنَّ مِن جَلَٰبِيبِهِنَّ ۚ ذَٰلِكَ أَدْنَىٰٓ أَن يُعْرَفْنَ فَلَا يُؤْذَيْنَ ۗ وَكَانَ ٱللَّهُ غَفُورًۭا رَّحِيمًۭا ﴿٥٩﴾ ای نبي (صلی الله علیه وعلی آله واصحابه وسلم)! خپلو مېرمنو (پاکو مېرمنو)، خپلو لوڼو او د مؤمنانو ښځو ته ووایئ چې د خپلو څادرونو یوه برخه دې پر ځان (له پاسه) راځوړنده کړي؛ دا (پرده) دې ته ډېره نږدې ده چې وپېژندل شي (چې شریفې ښځې دي) او ونه ځورول شي؛ او الله ډېر بښونکی، بې‌حده مهربان دی.
p ۞ لَّئِن لَّمْ يَنتَهِ ٱلْمُنَٰفِقُونَ وَٱلَّذِينَ فِى قُلُوبِهِم مَّرَضٌۭ وَٱلْمُرْجِفُونَ فِى ٱلْمَدِينَةِ لَنُغْرِيَنَّكَ بِهِمْ ثُمَّ لَا يُجَاوِرُونَكَ فِيهَآ إِلَّا قَلِيلًۭا ﴿٦٠﴾ اگر منافقین باز نہ آئے اور وہ لوگ جن کے دلوں میں (شک کی) بیماری ہے اور مدینہ میں افواہیں پھیلانے والے ہم آپ (صلی اللہ علیہ وعلیٰ آلہٖ واصحابہٖ وسلم) کو ضرور اُن پر مسلط کر دیں گے پھر وہ آپ کے ساتھ اس شہر (مدینہ) میں تھوڑے ہی دن رہ پائیں گے۔ || ۞ لَّئِن لَّمْ يَنتَهِ ٱلْمُنَٰفِقُونَ وَٱلَّذِينَ فِى قُلُوبِهِم مَّرَضٌۭ وَٱلْمُرْجِفُونَ فِى ٱلْمَدِينَةِ لَنُغْرِيَنَّكَ بِهِمْ ثُمَّ لَا يُجَاوِرُونَكَ فِيهَآ إِلَّا قَلِيلًۭا ﴿٦٠﴾ If the hypocrites, and those in whose hearts is a disease, and the alarmists in the city do not cease, We verily shall urge thee on against them, then they will be your neighbours in it but a little while. (33:60) || ۞ لَّئِن لَّمْ يَنتَهِ ٱلْمُنَٰفِقُونَ وَٱلَّذِينَ فِى قُلُوبِهِم مَّرَضٌۭ وَٱلْمُرْجِفُونَ فِى ٱلْمَدِينَةِ لَنُغْرِيَنَّكَ بِهِمْ ثُمَّ لَا يُجَاوِرُونَكَ فِيهَآ إِلَّا قَلِيلًۭا ﴿٦٠﴾ که منافقان، او هغه کسان چې په زړونو کې یې د (شک) ناروغي ده، او په مدینه کې د اوازو خپروونکي منع نه شول، نو موږ به خامخا تاسو (صلی الله علیه وعلی آله واصحابه وسلم) پر هغوی مسلط کړو، بیا به هغوی له تاسو سره په دې ښار (مدینه) کې یوازې لږې ورځې پاتې شي.
p مَّلْعُونِينَ ۖ أَيْنَمَا ثُقِفُوٓا۟ أُخِذُوا۟ وَقُتِّلُوا۟ تَقْتِيلًۭا ﴿٦١﴾ (اس حال میں کہ) لعنت کیے ہوئے جہاں کہیں وہ پائے جائیں پکڑ لیے جائیں اور بری طرح ٹکڑے ٹکڑے کر کے قتل کر دیے جائیں۔ || مَّلْعُونِينَ ۖ أَيْنَمَا ثُقِفُوٓا۟ أُخِذُوا۟ وَقُتِّلُوا۟ تَقْتِيلًۭا ﴿٦١﴾ Accursed, they will be seized wherever found and slain with a (fierce) slaughter. (33:61) || مَّلْعُونِينَ ۖ أَيْنَمَا ثُقِفُوٓا۟ أُخِذُوا۟ وَقُتِّلُوا۟ تَقْتِيلًۭا ﴿٦١﴾ (په داسې حال کې چې) لعنت شوي، هر چېرې چې ومومل شي ونیول شي او په بده توګه ټوټې ټوټې ووژل شي.
