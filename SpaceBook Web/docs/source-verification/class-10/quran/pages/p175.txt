p ٱلَّذِينَ كَذَّبُوا۟ بِٱلْكِتَٰبِ وَبِمَآ أَرْسَلْنَا بِهِۦ رُسُلَنَا ۖ فَسَوْفَ يَعْلَمُونَ ﴿٧٠﴾ — جن لوگوں نے (اللہ کی) کتاب کو جھٹلایا اور اس چیز کو بھی (جھٹلایا) جس کے ساتھ ہم نے اپنے رسولوں کو بھیجا تو عنقریب وہ جان لیں گے۔ || ٱلَّذِينَ كَذَّبُوا۟ بِٱلْكِتَٰبِ وَبِمَآ أَرْسَلْنَا بِهِۦ رُسُلَنَا ۖ فَسَوْفَ يَعْلَمُونَ ﴿٧٠﴾ — Those who deny the Book (of Allah) and what We sent Our messengers with — they will soon come to know, || ٱلَّذِينَ كَذَّبُوا۟ بِٱلْكِتَٰبِ وَبِمَآ أَرْسَلْنَا بِهِۦ رُسُلَنَا ۖ فَسَوْفَ يَعْلَمُونَ ﴿٧٠﴾ — هغه کسان چې د (الله) کتاب یې درواغ وګاڼه او هغه شی یې هم (درواغ وګاڼه) چې موږ پرې خپل رسولان لېږلي، نو ژر به پوه شي،

p ثُمَّ قِيلَ لَهُمْ أَيْنَ مَا كُنتُمْ تُشْرِكُونَ ﴿٧٣﴾ — پھر ان سے کہا جائے گا کہاں ہیں وہ جنھیں تم شریک ٹھہراتے تھے۔ || ثُمَّ قِيلَ لَهُمْ أَيْنَ مَا كُنتُمْ تُشْرِكُونَ ﴿٧٣﴾ — Then it will be said to them: "Where are those you used to associate || ثُمَّ قِيلَ لَهُمْ أَيْنَ مَا كُنتُمْ تُشْرِكُونَ ﴿٧٣﴾ — بیا به ورته وویل شي: "هغه چېرې دي چې تاسو به شریکول
p مِن دُونِ ٱللَّهِ ۖ قَالُوا۟ ضَلُّوا۟ عَنَّا بَل لَّمْ نَكُن نَّدْعُوا۟ مِن قَبْلُ شَيْـًۭٔا ۚ كَذَٰلِكَ يُضِلُّ ٱللَّهُ ٱلْكَٰفِرِينَ ﴿٧٤﴾ — اللہ کے سوا وہ جواب دیں گے کہ وہ ہم سے گم ہو گئے بلکہ ہم تو اس سے پہلے کسی کو بھی نہیں پوجتے تھے اسی طرح اللہ کافروں کو گم راہ کرتا ہے۔ || مِن دُونِ ٱللَّهِ ۖ قَالُوا۟ ضَلُّوا۟ عَنَّا بَل لَّمْ نَكُن نَّدْعُوا۟ مِن قَبْلُ شَيْـًۭٔا ۚ كَذَٰلِكَ يُضِلُّ ٱللَّهُ ٱلْكَٰفِرِينَ ﴿٧٤﴾ — besides Allah?" They will reply: "They have vanished from us; rather, we did not call upon anything before." Thus Allah leaves the disbelievers astray. || مِن دُونِ ٱللَّهِ ۖ قَالُوا۟ ضَلُّوا۟ عَنَّا بَل لَّمْ نَكُن نَّدْعُوا۟ مِن قَبْلُ شَيْـًۭٔا ۚ كَذَٰلِكَ يُضِلُّ ٱللَّهُ ٱلْكَٰفِرِينَ ﴿٧٤﴾ — پرته له الله؟" هغوی به ځواب ورکړي: "هغه له موږ ورک شول، بلکې موږ خو له دې مخکې د هېچا عبادت نه کاوه"؛ همداسې الله کافران ګمراه کوي.
p ذَٰلِكُم بِمَا كُنتُمْ تَفْرَحُونَ فِى ٱلْأَرْضِ بِغَيْرِ ٱلْحَقِّ وَبِمَا كُنتُمْ تَمْرَحُونَ ﴿٧٥﴾ — یہ اس کا بدلہ ہے جو تم زمین میں خوشی مناتے تھے ناحق اور اس لیے کہ تم اتراتے تھے۔ || ذَٰلِكُم بِمَا كُنتُمْ تَفْرَحُونَ فِى ٱلْأَرْضِ بِغَيْرِ ٱلْحَقِّ وَبِمَا كُنتُمْ تَمْرَحُونَ ﴿٧٥﴾ — "That is because you used to rejoice on earth without right, and because you used to exult. || ذَٰلِكُم بِمَا كُنتُمْ تَفْرَحُونَ فِى ٱلْأَرْضِ بِغَيْرِ ٱلْحَقِّ وَبِمَا كُنتُمْ تَمْرَحُونَ ﴿٧٥﴾ — "دا د هغه بدله ده چې تاسو په ځمکه کې په ناحقه خوشحالي کوله، او ځکه چې تاسو به غرور کاوه.
p ٱدْخُلُوٓا۟ أَبْوَٰبَ جَهَنَّمَ خَٰلِدِينَ فِيهَا ۖ فَبِئْسَ مَثْوَى ٱلْمُتَكَبِّرِينَ ﴿٧٦﴾ — جہنم کے دروازوں سے داخل ہو جاؤ اس میں ہمیشہ رہنے والے ہو تو یقیناً تکبر کرنے والوں کا بہت برا ٹھکانا ہے۔ || ٱدْخُلُوٓا۟ أَبْوَٰبَ جَهَنَّمَ خَٰلِدِينَ فِيهَا ۖ فَبِئْسَ مَثْوَى ٱلْمُتَكَبِّرِينَ ﴿٧٦﴾ — Enter the gates of Hell, to abide therein forever." Evil indeed is the abode of the arrogant. || ٱدْخُلُوٓا۟ أَبْوَٰبَ جَهَنَّمَ خَٰلِدِينَ فِيهَا ۖ فَبِئْسَ مَثْوَى ٱلْمُتَكَبِّرِينَ ﴿٧٦﴾ — د دوزخ له دروازو ننوځئ، تل به پکې اوسئ"؛ نو یقیناً د تکبر کوونکو ځای ډېر بد دی.
p فَٱصْبِرْ إِنَّ وَعْدَ ٱللَّهِ حَقٌّۭ ۚ فَإِمَّا نُرِيَنَّكَ بَعْضَ ٱلَّذِى نَعِدُهُمْ أَوْ نَتَوَفَّيَنَّكَ فَإِلَيْنَا يُرْجَعُونَ ﴿٧٧﴾ — تو آپ (صلی اللہ علیہ وآلہ وسلم) صبر کیجیے بے شک اللہ کا وعدہ سچا ہے تو اگر ہم (عذاب کا) کچھ حصہ آپ کو دکھا دیں جس کا ہم ان سے وعدہ کرتے ہیں یا (اس سے پہلے) ہم آپ کو وفات دے دیں تو وہ ہماری ہی طرف لوٹائے جائیں گے۔ || فَٱصْبِرْ إِنَّ وَعْدَ ٱللَّهِ حَقٌّۭ ۚ فَإِمَّا نُرِيَنَّكَ بَعْضَ ٱلَّذِى نَعِدُهُمْ أَوْ نَتَوَفَّيَنَّكَ فَإِلَيْنَا يُرْجَعُونَ ﴿٧٧﴾ — So be patient; Allah's promise is surely true. Whether We show you some of what We promise them (of punishment) or cause you to die (before that), to Us they will be returned. || فَٱصْبِرْ إِنَّ وَعْدَ ٱللَّهِ حَقٌّۭ ۚ فَإِمَّا نُرِيَنَّكَ بَعْضَ ٱلَّذِى نَعِدُهُمْ أَوْ نَتَوَفَّيَنَّكَ فَإِلَيْنَا يُرْجَعُونَ ﴿٧٧﴾ — نو ته صبر وکړه، بې شکه د الله ژمنه رښتیا ده؛ نو که موږ تا ته د هغه (عذاب) څه برخه وښیو چې ورسره ژمنه یې کوو، یا (له هغه مخکې) تا وفات کړو، نو هغوی به یوازې موږ ته ورګرځول کېږي.
p وَلَقَدْ أَرْسَلْنَا رُسُلًۭا مِّن قَبْلِكَ مِنْهُم مَّن قَصَصْنَا عَلَيْكَ وَمِنْهُم مَّن لَّمْ نَقْصُصْ عَلَيْكَ ۗ وَمَا كَانَ لِرَسُولٍ أَن يَأْتِىَ بِـَٔايَةٍ إِلَّا بِإِذْنِ ٱللَّهِ ۚ فَإِذَا جَآءَ أَمْرُ ٱللَّهِ قُضِىَ بِٱلْحَقِّ وَخَسِرَ هُنَالِكَ ٱلْمُبْطِلُونَ ﴿٧٨﴾ — اور یقیناً ہم نے آپ (صلی اللہ علیہ وآلہ وسلم) سے پہلے کئی رسول بھیجے ان میں کچھ تو ایسے ہیں جن کے حالات آپ سے بیان کر دیے ہیں اور کچھ ایسے ہیں جن کے حالات بیان نہیں کیے اور کسی بھی رسول کے لیے ایسا (ممکن) نہ تھا کہ اللہ کی اجازت کے بغیر کوئی نشانی لے آئیں پھر جب اللہ کا حکم آ گیا تو حق کے ساتھ فیصلہ کر دیا گیا اور اس وقت اہلِ باطل بہت خسارے میں رہے۔ || وَلَقَدْ أَرْسَلْنَا رُسُلًۭا مِّن قَبْلِكَ مِنْهُم مَّن قَصَصْنَا عَلَيْكَ وَمِنْهُم مَّن لَّمْ نَقْصُصْ عَلَيْكَ ۗ وَمَا كَانَ لِرَسُولٍ أَن يَأْتِىَ بِـَٔايَةٍ إِلَّا بِإِذْنِ ٱللَّهِ ۚ فَإِذَا جَآءَ أَمْرُ ٱللَّهِ قُضِىَ بِٱلْحَقِّ وَخَسِرَ هُنَالِكَ ٱلْمُبْطِلُونَ ﴿٧٨﴾ — We certainly sent messengers before you; of some We have told you, and of some We have not told you. It was not (possible) for any messenger to bring a sign except by Allah's leave. Then when Allah's command came, judgement was made with truth, and the followers of falsehood were then the losers. || وَلَقَدْ أَرْسَلْنَا رُسُلًۭا مِّن قَبْلِكَ مِنْهُم مَّن قَصَصْنَا عَلَيْكَ وَمِنْهُم مَّن لَّمْ نَقْصُصْ عَلَيْكَ ۗ وَمَا كَانَ لِرَسُولٍ أَن يَأْتِىَ بِـَٔايَةٍ إِلَّا بِإِذْنِ ٱللَّهِ ۚ فَإِذَا جَآءَ أَمْرُ ٱللَّهِ قُضِىَ بِٱلْحَقِّ وَخَسِرَ هُنَالِكَ ٱلْمُبْطِلُونَ ﴿٧٨﴾ — او یقیناً موږ له تا مخکې ډېر رسولان ولېږل؛ ځینې یې داسې دي چې حالات یې تا ته بیان کړي، او ځینې داسې دي چې حالات یې نه دي بیان کړي؛ او هېڅ رسول ته (ممکن) نه و چې پرته د الله له اجازې کومه نښه راوړي؛ بیا کله چې د الله حکم راغی، نو په حق پرېکړه وشوه، او په هغه وخت د باطل خاوندان ډېر په تاوان کې پاتې شول.
p ٱللَّهُ ٱلَّذِى جَعَلَ لَكُمُ ٱلْأَنْعَٰمَ لِتَرْكَبُوا۟ مِنْهَا وَمِنْهَا تَأْكُلُونَ ﴿٧٩﴾ — اللہ ہی ہے جس نے تمھارے لیے چوپائے بنائے تاکہ ان میں سے بعض پر تم سواری کرو اور بعض (کے گوشت) کو تم کھاؤ۔ || ٱللَّهُ ٱلَّذِى جَعَلَ لَكُمُ ٱلْأَنْعَٰمَ لِتَرْكَبُوا۟ مِنْهَا وَمِنْهَا تَأْكُلُونَ ﴿٧٩﴾ — It is Allah who made cattle for you, that you may ride some of them and eat (the meat of) some of them. || ٱللَّهُ ٱلَّذِى جَعَلَ لَكُمُ ٱلْأَنْعَٰمَ لِتَرْكَبُوا۟ مِنْهَا وَمِنْهَا تَأْكُلُونَ ﴿٧٩﴾ — یوازې الله دی چې ستاسو لپاره یې څاروي جوړ کړل څو پر ځینو یې سپاره شئ او د ځینو (غوښه) یې وخورئ.
p وَلَكُمْ فِيهَا مَنَٰفِعُ وَلِتَبْلُغُوا۟ عَلَيْهَا حَاجَةًۭ فِى صُدُورِكُمْ وَعَلَيْهَا وَعَلَى ٱلْفُلْكِ تُحْمَلُونَ ﴿٨٠﴾ — اور تمھارے لیے ان میں کئی فائدے ہیں اور تاکہ تم ان کے ذریعے (اپنی) ان ضروریات کو پورا کرو جو تمھارے دلوں میں ہیں اور ان (چوپایوں) پر اور کشتیوں پر تم سوار کیے جاتے ہو۔ || وَلَكُمْ فِيهَا مَنَٰفِعُ وَلِتَبْلُغُوا۟ عَلَيْهَا حَاجَةًۭ فِى صُدُورِكُمْ وَعَلَيْهَا وَعَلَى ٱلْفُلْكِ تُحْمَلُونَ ﴿٨٠﴾ — And there are many benefits in them for you, and that by them you may fulfil the needs in your hearts; and on them (cattle) and on ships you are carried. || وَلَكُمْ فِيهَا مَنَٰفِعُ وَلِتَبْلُغُوا۟ عَلَيْهَا حَاجَةًۭ فِى صُدُورِكُمْ وَعَلَيْهَا وَعَلَى ٱلْفُلْكِ تُحْمَلُونَ ﴿٨٠﴾ — او ستاسو لپاره په هغو کې ډېرې ګټې دي، او څو په هغو (خپلې) هغه اړتیاوې پوره کړئ چې ستاسو په زړونو کې دي؛ او پر هغو (څارویو) او کښتیو سپارول کېږئ.
