p ذَٰلِكُمُ ٱللَّهُ رَبُّكُمْ خَٰلِقُ كُلِّ شَىْءٍۢ لَّآ إِلَٰهَ إِلَّا هُوَ ۖ فَأَنَّىٰ تُؤْفَكُونَ ﴿٦٢﴾ — یہی اللہ تمھارا رب ہے (وہی) ہر شے کا پیدا فرمانے والا ہے اس کے سوا کوئی معبود نہیں پھر تم کہاں بہکے چلے جا رہے ہو؟ || ذَٰلِكُمُ ٱللَّهُ رَبُّكُمْ خَٰلِقُ كُلِّ شَىْءٍۢ لَّآ إِلَٰهَ إِلَّا هُوَ ۖ فَأَنَّىٰ تُؤْفَكُونَ ﴿٦٢﴾ — That is Allah, your Lord, the Creator of all things; there is no god but Him. How then are you deluded? || ذَٰلِكُمُ ٱللَّهُ رَبُّكُمْ خَٰلِقُ كُلِّ شَىْءٍۢ لَّآ إِلَٰهَ إِلَّا هُوَ ۖ فَأَنَّىٰ تُؤْفَكُونَ ﴿٦٢﴾ — همدا الله ستاسو رب دی، (همغه) د هر شي پیدا کوونکی دی، پرته له هغه هېڅ معبود نشته؛ نو بیا چېرته اړول کېږئ؟
p كَذَٰلِكَ يُؤْفَكُ ٱلَّذِينَ كَانُوا۟ بِـَٔايَٰتِ ٱللَّهِ يَجْحَدُونَ ﴿٦٣﴾ — اسی طرح وہ لوگ بھی بہکائے جاتے رہے جو اللہ کی آیات کا انکار کرتے تھے۔ || كَذَٰلِكَ يُؤْفَكُ ٱلَّذِينَ كَانُوا۟ بِـَٔايَٰتِ ٱللَّهِ يَجْحَدُونَ ﴿٦٣﴾ — Thus were deluded those who used to deny Allah's verses. || كَذَٰلِكَ يُؤْفَكُ ٱلَّذِينَ كَانُوا۟ بِـَٔايَٰتِ ٱللَّهِ يَجْحَدُونَ ﴿٦٣﴾ — همداسې هغه خلک هم اړول کېدل چې د الله له آیتونو یې انکار کاوه.
p ٱللَّهُ ٱلَّذِى جَعَلَ لَكُمُ ٱلْأَرْضَ قَرَارًۭا وَٱلسَّمَآءَ بِنَآءًۭ وَصَوَّرَكُمْ فَأَحْسَنَ صُوَرَكُمْ وَرَزَقَكُم مِّنَ ٱلطَّيِّبَٰتِ ۚ ذَٰلِكُمُ ٱللَّهُ رَبُّكُمْ ۖ فَتَبَارَكَ ٱللَّهُ رَبُّ ٱلْعَٰلَمِينَ ﴿٦٤﴾ — اللہ ہی ہے جس نے زمین کو ٹھہرنے کی جگہ بنایا اور آسمان کو چھت بنایا اور تمھاری صورتیں بنائیں تو (کیا ہی) اچھی تمھاری صورتیں بنائیں اور پاکیزہ چیزوں سے تمھیں رزق عطا فرمایا یہی اللہ تمھارا رب ہے پس بہت بابرکت ہے اللہ جو تمام جہانوں کا رب ہے۔ || ٱللَّهُ ٱلَّذِى جَعَلَ لَكُمُ ٱلْأَرْضَ قَرَارًۭا وَٱلسَّمَآءَ بِنَآءًۭ وَصَوَّرَكُمْ فَأَحْسَنَ صُوَرَكُمْ وَرَزَقَكُم مِّنَ ٱلطَّيِّبَٰتِ ۚ ذَٰلِكُمُ ٱللَّهُ رَبُّكُمْ ۖ فَتَبَارَكَ ٱللَّهُ رَبُّ ٱلْعَٰلَمِينَ ﴿٦٤﴾ — It is Allah who made the earth a place of settlement for you and the sky a canopy, and shaped you and made your shapes good, and provided you with good things. That is Allah, your Lord; so blessed is Allah, Lord of all the worlds. || ٱللَّهُ ٱلَّذِى جَعَلَ لَكُمُ ٱلْأَرْضَ قَرَارًۭا وَٱلسَّمَآءَ بِنَآءًۭ وَصَوَّرَكُمْ فَأَحْسَنَ صُوَرَكُمْ وَرَزَقَكُم مِّنَ ٱلطَّيِّبَٰتِ ۚ ذَٰلِكُمُ ٱللَّهُ رَبُّكُمْ ۖ فَتَبَارَكَ ٱللَّهُ رَبُّ ٱلْعَٰلَمِينَ ﴿٦٤﴾ — یوازې الله دی چې ځمکه یې د اوسېدو ځای وګرځوله او اسمان یې چت کړ، او ستاسو بڼې یې جوړې کړې نو (څومره) ښې بڼې یې جوړې کړې، او له پاکو شیانو یې رزق درکړ؛ همدا الله ستاسو رب دی؛ نو ډېر برکتناک دی الله چې د ټولو جهانونو رب دی.
p هُوَ ٱلْحَىُّ لَآ إِلَٰهَ إِلَّا هُوَ فَٱدْعُوهُ مُخْلِصِينَ لَهُ ٱلدِّينَ ۗ ٱلْحَمْدُ لِلَّهِ رَبِّ ٱلْعَٰلَمِينَ ﴿٦٥﴾ — وہی زندہ ہے اس کے سوا کوئی معبود نہیں لہٰذا اسی کی عبادت کرو اس کے لیے دین کو خالص کرتے ہوئے تمام تعریفیں اللہ کے لیے ہیں جو تمام جہانوں کا رب ہے۔ || هُوَ ٱلْحَىُّ لَآ إِلَٰهَ إِلَّا هُوَ فَٱدْعُوهُ مُخْلِصِينَ لَهُ ٱلدِّينَ ۗ ٱلْحَمْدُ لِلَّهِ رَبِّ ٱلْعَٰلَمِينَ ﴿٦٥﴾ — He is the Ever-Living; there is no god but Him. So call upon Him, making religion purely His. All praise is for Allah, Lord of all the worlds. || هُوَ ٱلْحَىُّ لَآ إِلَٰهَ إِلَّا هُوَ فَٱدْعُوهُ مُخْلِصِينَ لَهُ ٱلدِّينَ ۗ ٱلْحَمْدُ لِلَّهِ رَبِّ ٱلْعَٰلَمِينَ ﴿٦٥﴾ — همغه ژوندی دی، پرته له هغه هېڅ معبود نشته؛ نو د همغه عبادت وکړئ، د هغه لپاره دین خالص کړئ؛ ټول ستاینه د الله لپاره ده چې د ټولو جهانونو رب دی.
p ۞ قُلْ إِنِّى نُهِيتُ أَنْ أَعْبُدَ ٱلَّذِينَ تَدْعُونَ مِن دُونِ ٱللَّهِ لَمَّا جَآءَنِىَ ٱلْبَيِّنَٰتُ مِن رَّبِّى وَأُمِرْتُ أَنْ أُسْلِمَ لِرَبِّ ٱلْعَٰلَمِينَ ﴿٦٦﴾ — آپ (صلی اللہ علیہ وآلہ وسلم) فرما دیجیے بے شک مجھے منع کیا گیا ہے کہ میں ان کی عبادت کروں جنھیں تم اللہ کے سوا پوجتے ہو جب کہ میرے پاس میرے رب کی طرف سے واضح نشانیاں آ چکی ہیں اور مجھے حکم دیا گیا ہے کہ میں تمام جہانوں کے رب کا فرماں بردار بن کر رہوں۔ || ۞ قُلْ إِنِّى نُهِيتُ أَنْ أَعْبُدَ ٱلَّذِينَ تَدْعُونَ مِن دُونِ ٱللَّهِ لَمَّا جَآءَنِىَ ٱلْبَيِّنَٰتُ مِن رَّبِّى وَأُمِرْتُ أَنْ أُسْلِمَ لِرَبِّ ٱلْعَٰلَمِينَ ﴿٦٦﴾ — Say: "I have been forbidden to worship those you call upon besides Allah, since clear signs have come to me from my Lord, and I have been commanded to submit to the Lord of all the worlds." || ۞ قُلْ إِنِّى نُهِيتُ أَنْ أَعْبُدَ ٱلَّذِينَ تَدْعُونَ مِن دُونِ ٱللَّهِ لَمَّا جَآءَنِىَ ٱلْبَيِّنَٰتُ مِن رَّبِّى وَأُمِرْتُ أَنْ أُسْلِمَ لِرَبِّ ٱلْعَٰلَمِينَ ﴿٦٦﴾ — ووایه: "بې شکه ما منع کړای شوی یم چې د هغو عبادت وکړم چې تاسو پرته له الله یې عبادت کوئ، کله چې ما ته زما د رب له لوري روښانه نښې راغلې، او ما ته حکم شوی چې د ټولو جهانونو د رب فرمانبردار واوسم."
p هُوَ ٱلَّذِى خَلَقَكُم مِّن تُرَابٍۢ ثُمَّ مِن نُّطْفَةٍۢ ثُمَّ مِنْ عَلَقَةٍۢ ثُمَّ يُخْرِجُكُمْ طِفْلًۭا ثُمَّ لِتَبْلُغُوٓا۟ أَشُدَّكُمْ ثُمَّ لِتَكُونُوا۟ شُيُوخًۭا ۚ وَمِنكُم مَّن يُتَوَفَّىٰ مِن قَبْلُ ۖ وَلِتَبْلُغُوٓا۟ أَجَلًۭا مُّسَمًّۭى وَلَعَلَّكُمْ تَعْقِلُونَ ﴿٦٧﴾ — وہی ہے جس نے تمھیں مٹی سے پیدا فرمایا پھر ایک نطفہ (قطرے) سے پھر جمے ہوئے خون سے پھر وہ تمھیں بچے کی صورت میں نکالتا ہے پھر تم جوانی کو پہنچتے ہو پھر تم بوڑھے ہو جاتے ہو اور تم میں وہ بھی ہیں جو اس سے پہلے وفات دے دیے جاتے ہیں اور تاکہ تم مقررہ وقت تک پہنچ جاؤ اور تاکہ تم (حقیقت کو) سمجھو۔ || هُوَ ٱلَّذِى خَلَقَكُم مِّن تُرَابٍۢ ثُمَّ مِن نُّطْفَةٍۢ ثُمَّ مِنْ عَلَقَةٍۢ ثُمَّ يُخْرِجُكُمْ طِفْلًۭا ثُمَّ لِتَبْلُغُوٓا۟ أَشُدَّكُمْ ثُمَّ لِتَكُونُوا۟ شُيُوخًۭا ۚ وَمِنكُم مَّن يُتَوَفَّىٰ مِن قَبْلُ ۖ وَلِتَبْلُغُوٓا۟ أَجَلًۭا مُّسَمًّۭى وَلَعَلَّكُمْ تَعْقِلُونَ ﴿٦٧﴾ — It is He who created you from dust, then from a drop of fluid, then from a clinging clot; then He brings you forth as a child, then (lets you) reach your full strength, then become old — and among you is one who is taken in death before — and (lets you) reach an appointed term, so that you may understand (the truth). || هُوَ ٱلَّذِى خَلَقَكُم مِّن تُرَابٍۢ ثُمَّ مِن نُّطْفَةٍۢ ثُمَّ مِنْ عَلَقَةٍۢ ثُمَّ يُخْرِجُكُمْ طِفْلًۭا ثُمَّ لِتَبْلُغُوٓا۟ أَشُدَّكُمْ ثُمَّ لِتَكُونُوا۟ شُيُوخًۭا ۚ وَمِنكُم مَّن يُتَوَفَّىٰ مِن قَبْلُ ۖ وَلِتَبْلُغُوٓا۟ أَجَلًۭا مُّسَمًّۭى وَلَعَلَّكُمْ تَعْقِلُونَ ﴿٦٧﴾ — همغه دی چې تاسو یې له خاورې پیدا کړئ، بیا له یوې نطفې (څاڅکي)، بیا له کلکې وینې، بیا تاسو د ماشوم په بڼه راوباسي، بیا ځوانۍ ته رسېږئ، بیا زاړه کېږئ؛ او په تاسو کې هغه هم شته چې له دې مخکې مړه کړای شي؛ او څو تر ټاکلي وخته ورسېږئ، او څو (په حقیقت) پوه شئ.
p هُوَ ٱلَّذِى يُحْىِۦ وَيُمِيتُ ۖ فَإِذَا قَضَىٰٓ أَمْرًۭا فَإِنَّمَا يَقُولُ لَهُۥ كُن فَيَكُونُ ﴿٦٨﴾ — وہی ہے جو زندگی دیتا اور موت دیتا ہے پس جب وہ کسی کام کا فیصلہ فرماتا ہے تو اس سے یہی فرماتا ہے کہ ہو جا تو وہ ہو جاتا ہے۔ || هُوَ ٱلَّذِى يُحْىِۦ وَيُمِيتُ ۖ فَإِذَا قَضَىٰٓ أَمْرًۭا فَإِنَّمَا يَقُولُ لَهُۥ كُن فَيَكُونُ ﴿٦٨﴾ — It is He who gives life and causes death; and when He decrees a matter, He only says to it "Be", and it is. || هُوَ ٱلَّذِى يُحْىِۦ وَيُمِيتُ ۖ فَإِذَا قَضَىٰٓ أَمْرًۭا فَإِنَّمَا يَقُولُ لَهُۥ كُن فَيَكُونُ ﴿٦٨﴾ — همغه دی چې ژوند ورکوي او مرګ ورکوي؛ نو کله چې د کوم کار پرېکړه وکړي، نو یوازې ورته وايي "شه"، نو هغه کېږي.
p أَلَمْ تَرَ إِلَى ٱلَّذِينَ يُجَٰدِلُونَ فِىٓ ءَايَٰتِ ٱللَّهِ أَنَّىٰ يُصْرَفُونَ ﴿٦٩﴾ — کیا آپ نے ان لوگوں کو نہیں دیکھا (جو) اللہ کی آیتوں میں جھگڑتے ہیں وہ کہاں پھیرے جاتے ہیں۔ || أَلَمْ تَرَ إِلَى ٱلَّذِينَ يُجَٰدِلُونَ فِىٓ ءَايَٰتِ ٱللَّهِ أَنَّىٰ يُصْرَفُونَ ﴿٦٩﴾ — Have you not seen those who dispute about Allah's verses — how they are turned away? || أَلَمْ تَرَ إِلَى ٱلَّذِينَ يُجَٰدِلُونَ فِىٓ ءَايَٰتِ ٱللَّهِ أَنَّىٰ يُصْرَفُونَ ﴿٦٩﴾ — آیا تا هغه خلک نه دي لیدلي (چې) د الله په آیتونو کې جګړه کوي، چېرته اړول کېږي؟
