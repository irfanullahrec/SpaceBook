p ذَٰلِكُم بِأَنَّهُۥٓ إِذَا دُعِىَ ٱللَّهُ وَحْدَهُۥ كَفَرْتُمْ ۖ وَإِن يُشْرَكْ بِهِۦ تُؤْمِنُوا۟ ۚ فَٱلْحُكْمُ لِلَّهِ ٱلْعَلِىِّ ٱلْكَبِيرِ ﴿١٢﴾ — یہ اس لیے ہے کہ جب اکیلے اللہ کو پکارا جاتا تھا تو تم انکار کرتے تھے اور اگر اس کے ساتھ کسی اور کو شریک ٹھہرایا جاتا تھا تو تم مان لیتے تھے تو حکم کا اختیار اللہ ہی کے لیے ہے (جو) بلند مرتبہ بڑائی والا ہے۔ || ذَٰلِكُم بِأَنَّهُۥٓ إِذَا دُعِىَ ٱللَّهُ وَحْدَهُۥ كَفَرْتُمْ ۖ وَإِن يُشْرَكْ بِهِۦ تُؤْمِنُوا۟ ۚ فَٱلْحُكْمُ لِلَّهِ ٱلْعَلِىِّ ٱلْكَبِيرِ ﴿١٢﴾ — That is because when Allah alone was called upon, you disbelieved; but if others were associated with Him, you believed. So judgement belongs to Allah, the Most High, the Most Great. || ذَٰلِكُم بِأَنَّهُۥٓ إِذَا دُعِىَ ٱللَّهُ وَحْدَهُۥ كَفَرْتُمْ ۖ وَإِن يُشْرَكْ بِهِۦ تُؤْمِنُوا۟ ۚ فَٱلْحُكْمُ لِلَّهِ ٱلْعَلِىِّ ٱلْكَبِيرِ ﴿١٢﴾ — دا ځکه چې کله به یوازې الله ته غږ کېده، نو تاسو به انکار کاوه، او که له هغه سره به بل څوک شریک نیول کېده، نو منل به مو؛ نو د حکم واک یوازې د الله دی (چې) لوړ مرتبه، لوی دی.
p هُوَ ٱلَّذِى يُرِيكُمْ ءَايَٰتِهِۦ وَيُنَزِّلُ لَكُم مِّنَ ٱلسَّمَآءِ رِزْقًۭا ۚ وَمَا يَتَذَكَّرُ إِلَّا مَن يُنِيبُ ﴿١٣﴾ — وہی ہے جو تمھیں اپنی نشانیاں دکھاتا ہے اور تمھارے لیے آسمان سے رزق نازل فرماتا ہے اور نصیحت تو صرف وہی حاصل کرتا ہے جو رجوع کرتا ہے۔ || هُوَ ٱلَّذِى يُرِيكُمْ ءَايَٰتِهِۦ وَيُنَزِّلُ لَكُم مِّنَ ٱلسَّمَآءِ رِزْقًۭا ۚ وَمَا يَتَذَكَّرُ إِلَّا مَن يُنِيبُ ﴿١٣﴾ — It is He who shows you His signs and sends down provision for you from the sky. None takes heed except one who turns (to Him). || هُوَ ٱلَّذِى يُرِيكُمْ ءَايَٰتِهِۦ وَيُنَزِّلُ لَكُم مِّنَ ٱلسَّمَآءِ رِزْقًۭا ۚ وَمَا يَتَذَكَّرُ إِلَّا مَن يُنِيبُ ﴿١٣﴾ — همغه دی چې تاسو ته خپلې نښې ښيي او ستاسو لپاره له اسمانه رزق نازلوي؛ او نصیحت یوازې هغه اخلي چې رجوع کوي.
p فَٱدْعُوا۟ ٱللَّهَ مُخْلِصِينَ لَهُ ٱلدِّينَ وَلَوْ كَرِهَ ٱلْكَٰفِرُونَ ﴿١٤﴾ — پس تم اللہ کی عبادت کرو اس کے لیے دین (عبادت) کو خالص کرتے ہوئے خواہ کافروں کو کتنا ہی ناگوار ہو۔ || فَٱدْعُوا۟ ٱللَّهَ مُخْلِصِينَ لَهُ ٱلدِّينَ وَلَوْ كَرِهَ ٱلْكَٰفِرُونَ ﴿١٤﴾ — So call upon Allah, making religion (worship) purely His, however much the disbelievers dislike it. || فَٱدْعُوا۟ ٱللَّهَ مُخْلِصِينَ لَهُ ٱلدِّينَ وَلَوْ كَرِهَ ٱلْكَٰفِرُونَ ﴿١٤﴾ — نو د الله عبادت وکړئ، د هغه لپاره دین (عبادت) خالص کړئ، که څه هم کافرانو ته هر څومره بد ښکاري.
p رَفِيعُ ٱلدَّرَجَٰتِ ذُو ٱلْعَرْشِ يُلْقِى ٱلرُّوحَ مِنْ أَمْرِهِۦ عَلَىٰ مَن يَشَآءُ مِنْ عِبَادِهِۦ لِيُنذِرَ يَوْمَ ٱلتَّلَاقِ ﴿١٥﴾ — (وہ) بلند درجوں والا (اور) عرش کا مالک ہے وہ اپنے حکم سے روح (وحی) کو نازل فرماتا ہے اپنے بندوں میں سے جس پر چاہتا ہے تاکہ وہ ملاقات کے دن سے ڈرائے۔ || رَفِيعُ ٱلدَّرَجَٰتِ ذُو ٱلْعَرْشِ يُلْقِى ٱلرُّوحَ مِنْ أَمْرِهِۦ عَلَىٰ مَن يَشَآءُ مِنْ عِبَادِهِۦ لِيُنذِرَ يَوْمَ ٱلتَّلَاقِ ﴿١٥﴾ — (He is) Exalted in ranks, Owner of the Throne. He sends the Spirit (revelation) by His command upon whomever He wills of His servants, to warn of the Day of Meeting — || رَفِيعُ ٱلدَّرَجَٰتِ ذُو ٱلْعَرْشِ يُلْقِى ٱلرُّوحَ مِنْ أَمْرِهِۦ عَلَىٰ مَن يَشَآءُ مِنْ عِبَادِهِۦ لِيُنذِرَ يَوْمَ ٱلتَّلَاقِ ﴿١٥﴾ — (هغه) د لوړو درجو خاوند (او) د عرش مالک دی؛ هغه په خپل حکم روح (وحې) پر خپلو بندګانو کې پر چا چې وغواړي نازلوي، څو د ملاقات له ورځې ووېروي —
p يَوْمَ هُم بَٰرِزُونَ ۖ لَا يَخْفَىٰ عَلَى ٱللَّهِ مِنْهُمْ شَىْءٌۭ ۚ لِّمَنِ ٱلْمُلْكُ ٱلْيَوْمَ ۖ لِلَّهِ ٱلْوَٰحِدِ ٱلْقَهَّارِ ﴿١٦﴾ — جس دن وہ سب نکل پڑیں گے (قبروں سے) ان کی کوئی چیز اللہ سے چھپی نہ ہوگی (ارشاد ہوگا) آج کس کی بادشاہی ہے؟ صرف اللہ کی جو اکیلا ہے سب پر غالب ہے۔ || يَوْمَ هُم بَٰرِزُونَ ۖ لَا يَخْفَىٰ عَلَى ٱللَّهِ مِنْهُمْ شَىْءٌۭ ۚ لِّمَنِ ٱلْمُلْكُ ٱلْيَوْمَ ۖ لِلَّهِ ٱلْوَٰحِدِ ٱلْقَهَّارِ ﴿١٦﴾ — the Day they will come forth (from the graves), nothing of them hidden from Allah. (It will be asked:) "Whose is the sovereignty today?" "Allah's alone, the One, the Irresistible." || يَوْمَ هُم بَٰرِزُونَ ۖ لَا يَخْفَىٰ عَلَى ٱللَّهِ مِنْهُمْ شَىْءٌۭ ۚ لِّمَنِ ٱلْمُلْكُ ٱلْيَوْمَ ۖ لِلَّهِ ٱلْوَٰحِدِ ٱلْقَهَّارِ ﴿١٦﴾ — هغه ورځ چې ټول به (له قبرونو) راووځي، د هغوی هېڅ شی به له الله پټ نه وي؛ (ویل به شي:) "نن پاچاهي د چا ده؟" "یوازې د الله چې یو دی، پر ټولو غالب دی."
p ٱلْيَوْمَ تُجْزَىٰ كُلُّ نَفْسٍۭ بِمَا كَسَبَتْ ۚ لَا ظُلْمَ ٱلْيَوْمَ ۚ إِنَّ ٱللَّهَ سَرِيعُ ٱلْحِسَابِ ﴿١٧﴾ — آج کے دن ہر شخص کو اس کا بدلہ دیا جائے گا جو اس نے کمایا آج کوئی ظلم نہیں ہوگا بے شک اللہ بہت جلد حساب لینے والا ہے۔ || ٱلْيَوْمَ تُجْزَىٰ كُلُّ نَفْسٍۭ بِمَا كَسَبَتْ ۚ لَا ظُلْمَ ٱلْيَوْمَ ۚ إِنَّ ٱللَّهَ سَرِيعُ ٱلْحِسَابِ ﴿١٧﴾ — This Day every soul will be repaid for what it earned; there is no injustice today. Surely Allah is swift in reckoning. || ٱلْيَوْمَ تُجْزَىٰ كُلُّ نَفْسٍۭ بِمَا كَسَبَتْ ۚ لَا ظُلْمَ ٱلْيَوْمَ ۚ إِنَّ ٱللَّهَ سَرِيعُ ٱلْحِسَابِ ﴿١٧﴾ — نن ورځ به هر چا ته د هغه څه بدله ورکړل شي چې ګټلي یې دي؛ نن هېڅ ظلم نشته؛ بې شکه الله ډېر ژر حساب اخیستونکی دی.
p وَأَنذِرْهُمْ يَوْمَ ٱلْءَازِفَةِ إِذِ ٱلْقُلُوبُ لَدَى ٱلْحَنَاجِرِ كَٰظِمِينَ ۚ مَا لِلظَّٰلِمِينَ مِنْ حَمِيمٍۢ وَلَا شَفِيعٍۢ يُطَاعُ ﴿١٨﴾ — اور آپ ان کو قریب آنے والے دن سے ڈرائیے جب دل غم سے بھرے حلق کو آ رہے ہوں گے ظالموں کا نہ کوئی دوست ہوگا اور نہ کوئی ایسا سفارشی جس کی بات مانی جائے۔ || وَأَنذِرْهُمْ يَوْمَ ٱلْءَازِفَةِ إِذِ ٱلْقُلُوبُ لَدَى ٱلْحَنَاجِرِ كَٰظِمِينَ ۚ مَا لِلظَّٰلِمِينَ مِنْ حَمِيمٍۢ وَلَا شَفِيعٍۢ يُطَاعُ ﴿١٨﴾ — And warn them of the approaching Day, when hearts, choked with grief, will rise to the throats. The wrongdoers will have no friend, nor any intercessor who will be heeded. || وَأَنذِرْهُمْ يَوْمَ ٱلْءَازِفَةِ إِذِ ٱلْقُلُوبُ لَدَى ٱلْحَنَاجِرِ كَٰظِمِينَ ۚ مَا لِلظَّٰلِمِينَ مِنْ حَمِيمٍۢ وَلَا شَفِيعٍۢ يُطَاعُ ﴿١٨﴾ — او هغوی د نږدې راتلونکې ورځې څخه ووېروه، کله چې زړونه به له غمه ډک ستوني ته راځي؛ د ظالمانو به نه کوم دوست وي او نه داسې شفاعت کوونکی چې خبره یې ومنل شي.
p يَعْلَمُ خَآئِنَةَ ٱلْأَعْيُنِ وَمَا تُخْفِى ٱلصُّدُورُ ﴿١٩﴾ — وہ (اللہ) آنکھوں کی خیانت کو جانتا ہے اور ان باتوں کو بھی جن کو سینے چھپاتے ہیں۔ || يَعْلَمُ خَآئِنَةَ ٱلْأَعْيُنِ وَمَا تُخْفِى ٱلصُّدُورُ ﴿١٩﴾ — He (Allah) knows the treachery of the eyes and what the breasts conceal. || يَعْلَمُ خَآئِنَةَ ٱلْأَعْيُنِ وَمَا تُخْفِى ٱلصُّدُورُ ﴿١٩﴾ — هغه (الله) د سترګو په خیانت پوهېږي او په هغو خبرو هم چې سینې یې پټوي.
p وَٱللَّهُ يَقْضِى بِٱلْحَقِّ ۖ وَٱلَّذِينَ يَدْعُونَ مِن دُونِهِۦ لَا يَقْضُونَ بِشَىْءٍ ۗ إِنَّ ٱللَّهَ هُوَ ٱلسَّمِيعُ ٱلْبَصِيرُ ﴿٢٠﴾ — اور اللہ حق کے ساتھ فیصلہ فرماتا ہے اور جنھیں یہ لوگ اس کے سوا پوجتے ہیں وہ کسی چیز کا فیصلہ نہیں کر سکتے بے شک اللہ ہی خوب سننے والا خوب دیکھنے والا ہے۔ || وَٱللَّهُ يَقْضِى بِٱلْحَقِّ ۖ وَٱلَّذِينَ يَدْعُونَ مِن دُونِهِۦ لَا يَقْضُونَ بِشَىْءٍ ۗ إِنَّ ٱللَّهَ هُوَ ٱلسَّمِيعُ ٱلْبَصِيرُ ﴿٢٠﴾ — And Allah judges with truth, while those they call upon besides Him judge nothing.
