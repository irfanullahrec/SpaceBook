p قُلِ ٱللَّهُ يُنَجِّيكُم مِّنْهَا وَمِن كُلِّ كَرْبٍۢ ثُمَّ أَنتُمْ تُشْرِكُونَ ﴿٦٤﴾ — آپ (صلی اللہ علیہ وآلہ وسلم) فرما دیجیے اللہ ہی تمھیں اس (مصیبت) سے اور ہر غم سے نجات دیتا ہے پھر بھی تم شرک کرتے ہو۔ || قُلِ ٱللَّهُ يُنَجِّيكُم مِّنْهَا وَمِن كُلِّ كَرْبٍۢ ثُمَّ أَنتُمْ تُشْرِكُونَ ﴿٦٤﴾ — Say: "Allah saves you from it (the distress) and from every grief, yet you associate partners (with Him)." || قُلِ ٱللَّهُ يُنَجِّيكُم مِّنْهَا وَمِن كُلِّ كَرْبٍۢ ثُمَّ أَنتُمْ تُشْرِكُونَ ﴿٦٤﴾ — ووایه: "یوازې الله تاسو له دې (مصیبت) او له هر غم ژغوري، بیا هم تاسو شرک کوئ."
p قُلْ هُوَ ٱلْقَادِرُ عَلَىٰٓ أَن يَبْعَثَ عَلَيْكُمْ عَذَابًۭا مِّن فَوْقِكُمْ أَوْ مِن تَحْتِ أَرْجُلِكُمْ أَوْ يَلْبِسَكُمْ شِيَعًۭا وَيُذِيقَ بَعْضَكُم بَأْسَ بَعْضٍ ۗ ٱنظُرْ كَيْفَ نُصَرِّفُ ٱلْءَايَٰتِ لَعَلَّهُمْ يَفْقَهُونَ ﴿٦٥﴾ — آپ (صلی اللہ علیہ وآلہ وسلم) فرما دیجیے وہ اس پر قادر ہے کہ تم پر عذاب بھیج دے تمھارے اوپر سے یا تمھارے پاؤں کے نیچے سے یا تمھیں گروہوں میں تقسیم کر کے بھڑا دے اور ایک کو دوسرے کی طاقت کا مزہ چکھائے آپ دیکھیے ہم کس کس طرح اپنی آیتیں بیان کرتے ہیں تاکہ وہ سمجھیں۔ || قُلْ هُوَ ٱلْقَادِرُ عَلَىٰٓ أَن يَبْعَثَ عَلَيْكُمْ عَذَابًۭا مِّن فَوْقِكُمْ أَوْ مِن تَحْتِ أَرْجُلِكُمْ أَوْ يَلْبِسَكُمْ شِيَعًۭا وَيُذِيقَ بَعْضَكُم بَأْسَ بَعْضٍ ۗ ٱنظُرْ كَيْفَ نُصَرِّفُ ٱلْءَايَٰتِ لَعَلَّهُمْ يَفْقَهُونَ ﴿٦٥﴾ — Say: "He is able to send punishment on you from above you or from beneath your feet, or to divide you into factions and make you taste one another's violence." See how We set out the verses in various ways, that they may understand. || قُلْ هُوَ ٱلْقَادِرُ عَلَىٰٓ أَن يَبْعَثَ عَلَيْكُمْ عَذَابًۭا مِّن فَوْقِكُمْ أَوْ مِن تَحْتِ أَرْجُلِكُمْ أَوْ يَلْبِسَكُمْ شِيَعًۭا وَيُذِيقَ بَعْضَكُم بَأْسَ بَعْضٍ ۗ ٱنظُرْ كَيْفَ نُصَرِّفُ ٱلْءَايَٰتِ لَعَلَّهُمْ يَفْقَهُونَ ﴿٦٥﴾ — ووایه: "هغه پر دې قادر دی چې پر تاسو له پاسه یا ستاسو د پښو لاندې عذاب ولېږي، یا مو په ډلو ووېشي او سره مو وجنګوي او یو ته د بل د زور خوند وڅکوي." وګوره چې موږ څنګه بېلابېل ډول خپل آیتونه بیانوو څو پوه شي.
p وَكَذَّبَ بِهِۦ قَوْمُكَ وَهُوَ ٱلْحَقُّ ۚ قُل لَّسْتُ عَلَيْكُم بِوَكِيلٍۢ ﴿٦٦﴾ — اور آپ کی قوم نے اس (قرآن) کو جھٹلا دیا حالاں کہ وہ حق ہے آپ (صلی اللہ علیہ وآلہ وسلم) فرما دیجیے میں تم پر نگہبان نہیں ہوں۔ || وَكَذَّبَ بِهِۦ قَوْمُكَ وَهُوَ ٱلْحَقُّ ۚ قُل لَّسْتُ عَلَيْكُم بِوَكِيلٍۢ ﴿٦٦﴾ — And your people have denied it (the Qur'an), though it is the truth. Say: "I am not a guardian over you." || وَكَذَّبَ بِهِۦ قَوْمُكَ وَهُوَ ٱلْحَقُّ ۚ قُل لَّسْتُ عَلَيْكُم بِوَكِيلٍۢ ﴿٦٦﴾ — او ستا قوم دا (قرآن) دروغ وګاڼه حال دا چې حق دی. ووایه: "زه پر تاسو ساتونکی نه یم."
p لِّكُلِّ نَبَإٍۢ مُّسْتَقَرٌّۭ ۚ وَسَوْفَ تَعْلَمُونَ ﴿٦٧﴾ — ہر خبر (کے واقع ہونے) کا ایک وقت مقرر ہے اور تم عنقریب جان لوگے۔ || لِّكُلِّ نَبَإٍۢ مُّسْتَقَرٌّۭ ۚ وَسَوْفَ تَعْلَمُونَ ﴿٦٧﴾ — For every news there is a fixed time (of its happening), and you will soon know. || لِّكُلِّ نَبَإٍۢ مُّسْتَقَرٌّۭ ۚ وَسَوْفَ تَعْلَمُونَ ﴿٦٧﴾ — د هرې خبرې (د پېښېدو) لپاره یو ټاکلی وخت دی، او ژر به پوه شئ.
p وَإِذَا رَأَيْتَ ٱلَّذِينَ يَخُوضُونَ فِىٓ ءَايَٰتِنَا فَأَعْرِضْ عَنْهُمْ حَتَّىٰ يَخُوضُوا۟ فِى حَدِيثٍ غَيْرِهِۦ ۚ وَإِمَّا يُنسِيَنَّكَ ٱلشَّيْطَٰنُ فَلَا تَقْعُدْ بَعْدَ ٱلذِّكْرَىٰ مَعَ ٱلْقَوْمِ ٱلظَّٰلِمِينَ ﴿٦٨﴾ — اور جب آپ ان لوگوں کو دیکھیں جو ہماری آیات کے بارے میں (فضول) بحث کرتے ہیں پس آپ ان سے منھ پھیر لیجیے یہاں تک کہ وہ کسی دوسری بات میں لگ جائیں اور (اے مخاطب!) اگر شیطان تجھے (یہ بات) بھلا دے تو یاد آنے کے بعد ظالم لوگوں کے ساتھ مت بیٹھ۔ || وَإِذَا رَأَيْتَ ٱلَّذِينَ يَخُوضُونَ فِىٓ ءَايَٰتِنَا فَأَعْرِضْ عَنْهُمْ حَتَّىٰ يَخُوضُوا۟ فِى حَدِيثٍ غَيْرِهِۦ ۚ وَإِمَّا يُنسِيَنَّكَ ٱلشَّيْطَٰنُ فَلَا تَقْعُدْ بَعْدَ ٱلذِّكْرَىٰ مَعَ ٱلْقَوْمِ ٱلظَّٰلِمِينَ ﴿٦٨﴾ — And when you see those who engage in (idle) talk about Our verses, turn away from them until they take up some other talk. And (O addressee!) if Satan makes you forget (this), then after remembering do not sit with the wrongdoing people. || وَإِذَا رَأَيْتَ ٱلَّذِينَ يَخُوضُونَ فِىٓ ءَايَٰتِنَا فَأَعْرِضْ عَنْهُمْ حَتَّىٰ يَخُوضُوا۟ فِى حَدِيثٍ غَيْرِهِۦ ۚ وَإِمَّا يُنسِيَنَّكَ ٱلشَّيْطَٰنُ فَلَا تَقْعُدْ بَعْدَ ٱلذِّكْرَىٰ مَعَ ٱلْقَوْمِ ٱلظَّٰلِمِينَ ﴿٦٨﴾ — او کله چې هغه خلک ووینې چې زموږ د آیتونو په اړه (بې ځایه) بحث کوي، نو ترې مخ واړوه تر څو په بله خبره بوخت شي؛ او (ای مخاطبه!) که شیطان دا (خبره) درنه هېره کړي، نو له یادېدو وروسته له ظالمو خلکو سره مه کېنه.
p وَمَا عَلَى ٱلَّذِينَ يَتَّقُونَ مِنْ حِسَابِهِم مِّن شَىْءٍۢ وَلَٰكِن ذِكْرَىٰ لَعَلَّهُمْ يَتَّقُونَ ﴿٦٩﴾ — اور پرہیزگاروں پر ان (نافرمان) لوگوں کے حساب میں سے کسی چیز کی کوئی ذمہ داری نہیں لیکن نصیحت کر دینا (ان کے ذمہ) ہے تاکہ وہ پرہیزگاری اختیار کریں۔ || وَمَا عَلَى ٱلَّذِينَ يَتَّقُونَ مِنْ حِسَابِهِم مِّن شَىْءٍۢ وَلَٰكِن ذِكْرَىٰ لَعَلَّهُمْ يَتَّقُونَ ﴿٦٩﴾ — Those who are God-fearing bear no responsibility for anything of their (the disobedient's) account, but (it is on them) to give a reminder, that they may become God-fearing. || وَمَا عَلَى ٱلَّذِينَ يَتَّقُونَ مِنْ حِسَابِهِم مِّن شَىْءٍۢ وَلَٰكِن ذِكْرَىٰ لَعَلَّهُمْ يَتَّقُونَ ﴿٦٩﴾ — پر پرهېزګارانو د هغو (نافرمانو) خلکو له حساب څخه د هېڅ شي مسؤلیت نشته، خو نصیحت کول (د هغوی پر غاړه) دي څو پرهېزګاري غوره کړي.
p وَذَرِ ٱلَّذِينَ ٱتَّخَذُوا۟ دِينَهُمْ لَعِبًۭا وَلَهْوًۭا وَغَرَّتْهُمُ ٱلْحَيَوٰةُ ٱلدُّنْيَا ۚ وَذَكِّرْ بِهِۦٓ أَن تُبْسَلَ نَفْسٌۢ بِمَا كَسَبَتْ لَيْسَ لَهَا مِن دُونِ ٱللَّهِ وَلِىٌّۭ وَلَا شَفِيعٌۭ وَإِن تَعْدِلْ كُلَّ عَدْلٍۢ لَّا يُؤْخَذْ مِنْهَآ ۗ أُو۟لَٰٓئِكَ ٱلَّذِينَ أُبْسِلُوا۟ بِمَا كَسَبُوا۟ ۖ لَهُمْ شَرَابٌۭ مِّنْ حَمِيمٍۢ وَعَذَابٌ أَلِيمٌۢ بِمَا كَانُوا۟ يَكْفُرُونَ ﴿٧٠﴾ — اور ان لوگوں کو چھوڑ دیں جنھوں نے اپنے دین کو کھیل اور تماشا بنا لیا ہے اور دنیا کی زندگی نے انھیں دھوکے میں ڈال رکھا ہے اور اس (قرآن) کے ذریعے نصیحت کرتے رہیے تاکہ کوئی شخص اپنے اعمال کی وجہ سے پکڑا نہ جائے (اس دن) اللہ کے سوا نہ تو کوئی اس کا کارساز ہوگا اور نہ سفارش کرنے والا اور اگر وہ فدیہ میں سب کچھ دینا چاہے تو بھی قبول نہیں کیا جائے گا یہ وہ لوگ ہیں جو اپنے اعمال کی وجہ سے پکڑے گئے ان کے لیے پینے کو کھولتا ہوا پانی ہوگا اور (انھیں) دردناک عذاب ہوگا اس وجہ سے کہ وہ کفر کرتے تھے۔ || وَذَرِ ٱلَّذِينَ ٱتَّخَذُوا۟ دِينَهُمْ لَعِبًۭا وَلَهْوًۭا وَغَرَّتْهُمُ ٱلْحَيَوٰةُ ٱلدُّنْيَا ۚ وَذَكِّرْ بِهِۦٓ أَن تُبْسَلَ نَفْسٌۢ بِمَا كَسَبَتْ لَيْسَ لَهَا مِن دُونِ ٱللَّهِ وَلِىٌّۭ وَلَا شَفِيعٌۭ وَإِن تَعْدِلْ كُلَّ عَدْلٍۢ لَّا يُؤْخَذْ مِنْهَآ ۗ أُو۟لَٰٓئِكَ ٱلَّذِينَ أُبْسِلُوا۟ بِمَا كَسَبُوا۟ ۖ لَهُمْ شَرَابٌۭ مِّنْ حَمِيمٍۢ وَعَذَابٌ أَلِيمٌۢ بِمَا كَانُوا۟ يَكْفُرُونَ ﴿٧٠﴾ — And leave those who have taken their religion as play and amusement, and whom the life of this world has deceived; and keep reminding by it (the Qur'an), lest a soul be seized for what it has earned — it will have (on that Day) no protector or intercessor besides Allah, and even if it offered every ransom, it would not be accepted. These are the ones seized for what they earned; for them is a drink of boiling water and a painful punishment because they used to disbelieve. || وَذَرِ ٱلَّذِينَ ٱتَّخَذُوا۟ دِينَهُمْ لَعِبًۭا وَلَهْوًۭا وَغَرَّتْهُمُ ٱلْحَيَوٰةُ ٱلدُّنْيَا ۚ وَذَكِّرْ بِهِۦٓ أَن تُبْسَلَ نَفْسٌۢ بِمَا كَسَبَتْ لَيْسَ لَهَا مِن دُونِ ٱللَّهِ وَلِىٌّۭ وَلَا شَفِيعٌۭ وَإِن تَعْدِلْ كُلَّ عَدْلٍۢ لَّا يُؤْخَذْ مِنْهَآ ۗ أُو۟لَٰٓئِكَ ٱلَّذِينَ أُبْسِلُوا۟ بِمَا كَسَبُوا۟ ۖ لَهُمْ شَرَابٌۭ مِّنْ حَمِيمٍۢ وَعَذَابٌ أَلِيمٌۢ بِمَا كَانُوا۟ يَكْفُرُونَ ﴿٧٠﴾ — او هغه کسان پرېږده چې خپل دین یې لوبې او تماشا ګرځولی او دنیوي ژوند غولولي؛ او په دې (قرآن) نصیحت کوه څو هېڅوک د خپلو عملونو له امله ونه نیول شي؛ (په هغه ورځ) له الله پرته به یې نه کارساز وي او نه شفاعت کوونکی، او که د فدیې په توګه هر څه ورکړي، هم به ونه منل شي. دا هغه کسان دي چې د خپلو عملونو له امله ونیول شول؛ د هغوی لپاره به د څښلو لپاره ایشېدلې اوبه وي او (هغوی ته به) دردناک عذاب وي ځکه چې کفر یې کاوه.
