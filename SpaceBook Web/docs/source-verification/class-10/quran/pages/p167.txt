p ذَٰلِكَ بِأَنَّهُمْ كَانَت تَّأْتِيهِمْ رُسُلُهُم بِٱلْبَيِّنَٰتِ فَكَفَرُوا۟ فَأَخَذَهُمُ ٱللَّهُ ۚ إِنَّهُۥ قَوِىٌّۭ شَدِيدُ ٱلْعِقَابِ ﴿٢٢﴾ — یہ اس لیے کہ ان کے پاس ان کے رسول کھلی نشانیاں لے کر آئے تھے پھر انھوں نے کفر کیا تو اللہ نے انھیں پکڑ لیا بے شک وہ بہت قوت والا سخت سزا دینے والا ہے۔ || ذَٰلِكَ بِأَنَّهُمْ كَانَت تَّأْتِيهِمْ رُسُلُهُم بِٱلْبَيِّنَٰتِ فَكَفَرُوا۟ فَأَخَذَهُمُ ٱللَّهُ ۚ إِنَّهُۥ قَوِىٌّۭ شَدِيدُ ٱلْعِقَابِ ﴿٢٢﴾ — That was because their messengers came to them with clear signs, but they disbelieved; so Allah seized them. He is surely Strong, Severe in punishment. || ذَٰلِكَ بِأَنَّهُمْ كَانَت تَّأْتِيهِمْ رُسُلُهُم بِٱلْبَيِّنَٰتِ فَكَفَرُوا۟ فَأَخَذَهُمُ ٱللَّهُ ۚ إِنَّهُۥ قَوِىٌّۭ شَدِيدُ ٱلْعِقَابِ ﴿٢٢﴾ — دا ځکه چې د هغوی رسولان ورته له ښکاره نښو سره راغلي وو، بیا هغوی کفر وکړ، نو الله ونیول؛ بې شکه هغه ډېر ځواکمن، سخت سزا ورکوونکی دی.
p وَلَقَدْ أَرْسَلْنَا مُوسَىٰ بِـَٔايَٰتِنَا وَسُلْطَٰنٍۢ مُّبِينٍ ﴿٢٣﴾ — اور یقیناً ہم نے بھیجا موسیٰ (علیہ السلام) کو اپنی نشانیوں کے ساتھ اور واضح دلیل کے ساتھ۔ || وَلَقَدْ أَرْسَلْنَا مُوسَىٰ بِـَٔايَٰتِنَا وَسُلْطَٰنٍۢ مُّبِينٍ ﴿٢٣﴾ — And We certainly sent Musa with Our signs and a clear authority || وَلَقَدْ أَرْسَلْنَا مُوسَىٰ بِـَٔايَٰتِنَا وَسُلْطَٰنٍۢ مُّبِينٍ ﴿٢٣﴾ — او یقیناً موږ موسی (علیه السلام) له خپلو نښو او روښانه دلیل سره ولېږه
p إِلَىٰ فِرْعَوْنَ وَهَٰمَٰنَ وَقَٰرُونَ فَقَالُوا۟ سَٰحِرٌۭ كَذَّابٌۭ ﴿٢٤﴾ — فرعون اور ہامان اور قارون کی طرف انھوں نے کہا کہ (یہ تو) بڑا جھوٹا جادوگر ہے۔ || إِلَىٰ فِرْعَوْنَ وَهَٰمَٰنَ وَقَٰرُونَ فَقَالُوا۟ سَٰحِرٌۭ كَذَّابٌۭ ﴿٢٤﴾ — to Pharaoh, Haman and Qarun; but they said: "(He is) a lying magician." || إِلَىٰ فِرْعَوْنَ وَهَٰمَٰنَ وَقَٰرُونَ فَقَالُوا۟ سَٰحِرٌۭ كَذَّابٌۭ ﴿٢٤﴾ — فرعون، هامان او قارون ته؛ هغوی وویل: "(دا خو) لوی درواغجن کوډګر دی."
p فَلَمَّا جَآءَهُم بِٱلْحَقِّ مِنْ عِندِنَا قَالُوا۟ ٱقْتُلُوٓا۟ أَبْنَآءَ ٱلَّذِينَ ءَامَنُوا۟ مَعَهُۥ وَٱسْتَحْيُوا۟ نِسَآءَهُمْ ۚ وَمَا كَيْدُ ٱلْكَٰفِرِينَ إِلَّا فِى ضَلَٰلٍۢ ﴿٢٥﴾ — پھر جب (موسیٰ علیہ السلام) ان کے پاس ہماری طرف سے حق لے کر آئے (تو) انھوں نے کہا جو لوگ ان کے ساتھ ایمان لائے ان کے بیٹوں کو قتل کر دو اور ان کی عورتوں کو زندہ چھوڑ دو اور کافروں کی چال بالکل بے کار ہوئی۔ || فَلَمَّا جَآءَهُم بِٱلْحَقِّ مِنْ عِندِنَا قَالُوا۟ ٱقْتُلُوٓا۟ أَبْنَآءَ ٱلَّذِينَ ءَامَنُوا۟ مَعَهُۥ وَٱسْتَحْيُوا۟ نِسَآءَهُمْ ۚ وَمَا كَيْدُ ٱلْكَٰفِرِينَ إِلَّا فِى ضَلَٰلٍۢ ﴿٢٥﴾ — Then when he (Musa) brought them the truth from Us, they said: "Kill the sons of those who believe with him and spare their women." But the plot of the disbelievers is only in error (comes to nothing). || فَلَمَّا جَآءَهُم بِٱلْحَقِّ مِنْ عِندِنَا قَالُوا۟ ٱقْتُلُوٓا۟ أَبْنَآءَ ٱلَّذِينَ ءَامَنُوا۟ مَعَهُۥ وَٱسْتَحْيُوا۟ نِسَآءَهُمْ ۚ وَمَا كَيْدُ ٱلْكَٰفِرِينَ إِلَّا فِى ضَلَٰلٍۢ ﴿٢٥﴾ — بیا کله چې (موسی علیه السلام) ورته زموږ له لوري حق راوړ، (نو) هغوی وویل: "د هغو کسانو زامن ووژنئ چې ورسره یې ایمان راوړی او ښځې یې ژوندۍ پرېږدئ"؛ او د کافرانو چل بالکل بې ګټې شو.
p وَقَالَ فِرْعَوْنُ ذَرُونِىٓ أَقْتُلْ مُوسَىٰ وَلْيَدْعُ رَبَّهُۥٓ ۖ إِنِّىٓ أَخَافُ أَن يُبَدِّلَ دِينَكُمْ أَوْ أَن يُظْهِرَ فِى ٱلْأَرْضِ ٱلْفَسَادَ ﴿٢٦﴾ — اور فرعون نے کہا مجھے چھوڑ دو کہ موسیٰ کو قتل کر دوں اور اسے چاہیے کہ وہ اپنے رب کو پکارے بے شک میں ڈرتا ہوں کہیں وہ تمھارا دین بدل دے یا زمین میں فساد برپا کر دے۔ || وَقَالَ فِرْعَوْنُ ذَرُونِىٓ أَقْتُلْ مُوسَىٰ وَلْيَدْعُ رَبَّهُۥٓ ۖ إِنِّىٓ أَخَافُ أَن يُبَدِّلَ دِينَكُمْ أَوْ أَن يُظْهِرَ فِى ٱلْأَرْضِ ٱلْفَسَادَ ﴿٢٦﴾ — And Pharaoh said: "Leave me to kill Musa, and let him call upon his Lord. I fear he will change your religion or cause corruption in the land." || وَقَالَ فِرْعَوْنُ ذَرُونِىٓ أَقْتُلْ مُوسَىٰ وَلْيَدْعُ رَبَّهُۥٓ ۖ إِنِّىٓ أَخَافُ أَن يُبَدِّلَ دِينَكُمْ أَوْ أَن يُظْهِرَ فِى ٱلْأَرْضِ ٱلْفَسَادَ ﴿٢٦﴾ — او فرعون وویل: "پرېږدئ مې چې موسی ووژنم، او هغه دې خپل رب ته غږ وکړي؛ بې شکه زه وېرېږم چې ستاسو دین بدل نه کړي یا په ځمکه کې فساد ونه کړي."
p وَقَالَ مُوسَىٰٓ إِنِّى عُذْتُ بِرَبِّى وَرَبِّكُم مِّن كُلِّ مُتَكَبِّرٍۢ لَّا يُؤْمِنُ بِيَوْمِ ٱلْحِسَابِ ﴿٢٧﴾ — اور موسیٰ (علیہ السلام) نے فرمایا بے شک میں نے اپنے رب اور تمھارے رب کی پناہ لے لی ہے ہر متکبر سے جو حساب کے دن پر ایمان نہیں رکھتا۔ || وَقَالَ مُوسَىٰٓ إِنِّى عُذْتُ بِرَبِّى وَرَبِّكُم مِّن كُلِّ مُتَكَبِّرٍۢ لَّا يُؤْمِنُ بِيَوْمِ ٱلْحِسَابِ ﴿٢٧﴾ — And Musa said: "I seek refuge in my Lord and your Lord from every arrogant one who does not believe in the Day of Reckoning." || وَقَالَ مُوسَىٰٓ إِنِّى عُذْتُ بِرَبِّى وَرَبِّكُم مِّن كُلِّ مُتَكَبِّرٍۢ لَّا يُؤْمِنُ بِيَوْمِ ٱلْحِسَابِ ﴿٢٧﴾ — او موسی (علیه السلام) وفرمایل: "بې شکه ما د خپل رب او ستاسو د رب پناه اخیستې له هر متکبر څخه چې د حساب پر ورځ ایمان نه لري."
p وَقَالَ رَجُلٌۭ مُّؤْمِنٌۭ مِّنْ ءَالِ فِرْعَوْنَ يَكْتُمُ إِيمَٰنَهُۥٓ أَتَقْتُلُونَ رَجُلًا أَن يَقُولَ رَبِّىَ ٱللَّهُ وَقَدْ جَآءَكُم بِٱلْبَيِّنَٰتِ مِن رَّبِّكُمْ ۖ وَإِن يَكُ كَٰذِبًۭا فَعَلَيْهِ كَذِبُهُۥ ۖ وَإِن يَكُ صَادِقًۭا يُصِبْكُم بَعْضُ ٱلَّذِى يَعِدُكُمْ ۖ إِنَّ ٱللَّهَ لَا يَهْدِى مَنْ هُوَ مُسْرِفٌۭ كَذَّابٌۭ ﴿٢٨﴾ — اور آلِ فرعون کے ایک مومن مرد نے کہا جو اپنا ایمان چھپائے ہوئے تھا کیا تم ایک شخص کو صرف اس لیے قتل کر رہے ہو کہ وہ کہتا ہے میرا رب اللہ ہے حالاں کہ وہ تمھارے پاس واضح نشانیاں لے کر آئے ہیں تمھارے رب کی طرف سے (بالفرض) اگر وہ جھوٹے ہیں تو ان کا جھوٹ انھی پر ہے اور اگر وہ سچے ہیں تو وہ جو تم سے وعدہ کر رہے ہیں اس کا کچھ (عذاب) تم پر (ضرور) پہنچے گا بے شک اللہ (اس کو) ہدایت نہیں دیتا جو حد سے بڑھنے والا سخت جھوٹا ہو۔ || وَقَالَ رَجُلٌۭ مُّؤْمِنٌۭ مِّنْ ءَالِ فِرْعَوْنَ يَكْتُمُ إِيمَٰنَهُۥٓ أَتَقْتُلُونَ رَجُلًا أَن يَقُولَ رَبِّىَ ٱللَّهُ وَقَدْ جَآءَكُم بِٱلْبَيِّنَٰتِ مِن رَّبِّكُمْ ۖ وَإِن يَكُ كَٰذِبًۭا فَعَلَيْهِ كَذِبُهُۥ ۖ وَإِن يَكُ صَادِقًۭا يُصِبْكُم بَعْضُ ٱلَّذِى يَعِدُكُمْ ۖ إِنَّ ٱللَّهَ لَا يَهْدِى مَنْ هُوَ مُسْرِفٌۭ كَذَّابٌۭ ﴿٢٨﴾ — And a believing man of Pharaoh's people, who concealed his faith, said: "Will you kill a man merely for saying 'My Lord is Allah', when he has brought you clear signs from your Lord? If he is a liar, his lie is upon him; and if he is truthful, some of what he promises you will (surely) strike you. Surely Allah does not guide one who is a transgressor, a great liar. || وَقَالَ رَجُلٌۭ مُّؤْمِنٌۭ مِّنْ ءَالِ فِرْعَوْنَ يَكْتُمُ إِيمَٰنَهُۥٓ أَتَقْتُلُونَ رَجُلًا أَن يَقُولَ رَبِّىَ ٱللَّهُ وَقَدْ جَآءَكُم بِٱلْبَيِّنَٰتِ مِن رَّبِّكُمْ ۖ وَإِن يَكُ كَٰذِبًۭا فَعَلَيْهِ كَذِبُهُۥ ۖ وَإِن يَكُ صَادِقًۭا يُصِبْكُم بَعْضُ ٱلَّذِى يَعِدُكُمْ ۖ إِنَّ ٱللَّهَ لَا يَهْدِى مَنْ هُوَ مُسْرِفٌۭ كَذَّابٌۭ ﴿٢٨﴾ — او د فرعون د کورنۍ یوه مومن سړي چې خپل ایمان یې پټ ساتلی و وویل: "آیا تاسو یو سړی یوازې ځکه وژنئ چې وایي زما رب الله دی، په داسې حال کې چې هغه ستاسو د رب له لوري تاسو ته روښانه نښې راوړې؟ که (فرضاً) درواغجن وي، نو درواغ یې پر همغه دي؛ او که رښتینی وي، نو د هغه څه برخه چې درسره یې ژمنه کوي (خامخا) درته رسېږي؛ بې شکه الله هغه چا ته هدایت نه کوي چې له حده تېرېدونکی، سخت درواغجن وي.
