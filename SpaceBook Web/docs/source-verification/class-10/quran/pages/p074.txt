p وَيَقُولُ ٱلَّذِينَ كَفَرُوا۟ لَوْلَآ أُنزِلَ عَلَيْهِ ءَايَةٌۭ مِّن رَّبِّهِۦ ۗ قُلْ إِنَّ ٱللَّهَ يُضِلُّ مَن يَشَآءُ وَيَهْدِىٓ إِلَيْهِ مَنْ أَنَابَ ﴿٢٧﴾ — اور جنھوں نے کفر کیا وہ کہتے ہیں اُن پر کوئی نشانی کیوں نہ نازل کی گئی اُن کے رب کی طرف سے آپ (صلی اللہ علیہ وآلہ وسلم) فرما دیجیے بے شک اللہ گم راہ کر دیتا ہے جسے چاہتا ہے اور وہ اسے ہدایت عطا فرماتا ہے جو اس کی طرف رجوع کرتا ہے۔ || وَيَقُولُ ٱلَّذِينَ كَفَرُوا۟ لَوْلَآ أُنزِلَ عَلَيْهِ ءَايَةٌۭ مِّن رَّبِّهِۦ ۗ قُلْ إِنَّ ٱللَّهَ يُضِلُّ مَن يَشَآءُ وَيَهْدِىٓ إِلَيْهِ مَنْ أَنَابَ ﴿٢٧﴾ — Those who disbelieve say: "Why has no sign been sent down to him from his Lord?" Say: "Allah leaves astray whom He wills, and guides to Himself whoever turns to Him" — || وَيَقُولُ ٱلَّذِينَ كَفَرُوا۟ لَوْلَآ أُنزِلَ عَلَيْهِ ءَايَةٌۭ مِّن رَّبِّهِۦ ۗ قُلْ إِنَّ ٱللَّهَ يُضِلُّ مَن يَشَآءُ وَيَهْدِىٓ إِلَيْهِ مَنْ أَنَابَ ﴿٢٧﴾ — او هغه کسان چې کافران شول، وايي: "پر ده ولې د هغه د رب له لوري کومه نښه نه ده نازله شوې؟" ووایه: "بې شکه الله چې چا ته وغواړي ګمراه کوي یې، او هغه ته هدایت کوي چې هغه ته رجوع کوي" —
p ٱلَّذِينَ ءَامَنُوا۟ وَتَطْمَئِنُّ قُلُوبُهُم بِذِكْرِ ٱللَّهِ ۗ أَلَا بِذِكْرِ ٱللَّهِ تَطْمَئِنُّ ٱلْقُلُوبُ ﴿٢٨﴾ — (یہ وہ لوگ ہیں) جو ایمان لائے اور ان کے دل اللہ کی یاد سے مطمئن ہوتے ہیں آگاہ ہو جاؤ! اللہ کی یاد ہی سے دل مطمئن ہوتے ہیں۔ || ٱلَّذِينَ ءَامَنُوا۟ وَتَطْمَئِنُّ قُلُوبُهُم بِذِكْرِ ٱللَّهِ ۗ أَلَا بِذِكْرِ ٱللَّهِ تَطْمَئِنُّ ٱلْقُلُوبُ ﴿٢٨﴾ — (these are) those who believe and whose hearts find rest in the remembrance of Allah. Beware! It is in the remembrance of Allah that hearts find rest. || ٱلَّذِينَ ءَامَنُوا۟ وَتَطْمَئِنُّ قُلُوبُهُم بِذِكْرِ ٱللَّهِ ۗ أَلَا بِذِكْرِ ٱللَّهِ تَطْمَئِنُّ ٱلْقُلُوبُ ﴿٢٨﴾ — (دا هغه خلک دي) چې ایمان یې راوړی او زړونه یې د الله په یاد ډاډه کېږي. خبر اوسئ! یوازې د الله په یاد زړونه ډاډه کېږي.
p ٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّٰلِحَٰتِ طُوبَىٰ لَهُمْ وَحُسْنُ مَـَٔابٍۢ ﴿٢٩﴾ — وہ لوگ جو ایمان لائے اور انھوں نے نیک اعمال کیے اُن کے لیے خوش حالی ہے اور اچھا ٹھکانا ہے۔ || ٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّٰلِحَٰتِ طُوبَىٰ لَهُمْ وَحُسْنُ مَـَٔابٍۢ ﴿٢٩﴾ — Those who believe and do righteous deeds — for them is happiness and a good return. || ٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّٰلِحَٰتِ طُوبَىٰ لَهُمْ وَحُسْنُ مَـَٔابٍۢ ﴿٢٩﴾ — هغه کسان چې ایمان یې راوړ او نېک عملونه یې وکړل، د هغوی لپاره خوشحالي او ښه ځای دی.
p كَذَٰلِكَ أَرْسَلْنَٰكَ فِىٓ أُمَّةٍۢ قَدْ خَلَتْ مِن قَبْلِهَآ أُمَمٌۭ لِّتَتْلُوَا۟ عَلَيْهِمُ ٱلَّذِىٓ أَوْحَيْنَآ إِلَيْكَ وَهُمْ يَكْفُرُونَ بِٱلرَّحْمَٰنِ ۚ قُلْ هُوَ رَبِّى لَآ إِلَٰهَ إِلَّا هُوَ عَلَيْهِ تَوَكَّلْتُ وَإِلَيْهِ مَتَابِ ﴿٣٠﴾ — اسی طرح ہم نے آپ (صلی اللہ علیہ وآلہ وسلم) کو ایک ایسی اُمت میں (رسول بنا کر) بھیجا جس سے پہلے یقیناً کئی اُمتیں گزر چکی ہیں تاکہ آپ انھیں وہ (کتاب) پڑھ کر سنائیں جو ہم نے آپ (صلی اللہ علیہ وآلہ وسلم) کی طرف وحی فرمائی ہے اور وہ رحمٰن کا انکار کر رہے ہیں آپ (صلی اللہ علیہ وآلہ وسلم) فرما دیجیے وہی میرا رب ہے اس کے سوا کوئی معبود نہیں میں نے اسی پر بھروسہ کیا اور اسی کی طرف میں رجوع کرتا ہوں۔ || كَذَٰلِكَ أَرْسَلْنَٰكَ فِىٓ أُمَّةٍۢ قَدْ خَلَتْ مِن قَبْلِهَآ أُمَمٌۭ لِّتَتْلُوَا۟ عَلَيْهِمُ ٱلَّذِىٓ أَوْحَيْنَآ إِلَيْكَ وَهُمْ يَكْفُرُونَ بِٱلرَّحْمَٰنِ ۚ قُلْ هُوَ رَبِّى لَآ إِلَٰهَ إِلَّا هُوَ عَلَيْهِ تَوَكَّلْتُ وَإِلَيْهِ مَتَابِ ﴿٣٠﴾ — Thus We have sent you (as messenger) to a community before which other communities have passed away, that you may recite to them what We have revealed to you, while they disbelieve in the Most Merciful. Say: "He is my Lord; there is no god but Him. In Him I put my trust, and to Him I turn." || كَذَٰلِكَ أَرْسَلْنَٰكَ فِىٓ أُمَّةٍۢ قَدْ خَلَتْ مِن قَبْلِهَآ أُمَمٌۭ لِّتَتْلُوَا۟ عَلَيْهِمُ ٱلَّذِىٓ أَوْحَيْنَآ إِلَيْكَ وَهُمْ يَكْفُرُونَ بِٱلرَّحْمَٰنِ ۚ قُلْ هُوَ رَبِّى لَآ إِلَٰهَ إِلَّا هُوَ عَلَيْهِ تَوَكَّلْتُ وَإِلَيْهِ مَتَابِ ﴿٣٠﴾ — همداسې موږ ته په داسې امت کې (د رسول په توګه) ولېږلې چې له هغه مخکې یقیناً ډېر امتونه تېر شوي، څو ته ورته هغه (کتاب) ولولې چې موږ تا ته وحي کړی، او هغوی له رحمن انکار کوي. ووایه: "همغه زما رب دی؛ له هغه پرته بل معبود نشته؛ ما پر همده توکل کړی او همده ته رجوع کوم."
p وَلَوْ أَنَّ قُرْءَانًۭا سُيِّرَتْ بِهِ ٱلْجِبَالُ أَوْ قُطِّعَتْ بِهِ ٱلْأَرْضُ أَوْ كُلِّمَ بِهِ ٱلْمَوْتَىٰ ۗ بَل لِّلَّهِ ٱلْأَمْرُ جَمِيعًا ۗ أَفَلَمْ يَا۟يْـَٔسِ ٱلَّذِينَ ءَامَنُوٓا۟ أَن لَّوْ يَشَآءُ ٱللَّهُ لَهَدَى ٱلنَّاسَ جَمِيعًۭا ۗ وَلَا يَزَالُ ٱلَّذِينَ كَفَرُوا۟ تُصِيبُهُم بِمَا صَنَعُوا۟ قَارِعَةٌ أَوْ تَحُلُّ قَرِيبًۭا مِّن دَارِهِمْ حَتَّىٰ يَأْتِىَ وَعْدُ ٱللَّهِ ۚ إِنَّ ٱللَّهَ لَا يُخْلِفُ ٱلْمِيعَادَ ﴿٣١﴾ — اور اگر کوئی ایسا قرآن ہوتا کہ جس سے پہاڑ چلنے لگتے یا اس کے ذریعہ زمین کے ٹکڑے ٹکڑے کر دیے جاتے یا اس کے ذریعے مُردوں سے بات کی جا سکتی (تو پھر بھی کافر نہ مانتے) بلکہ سب کام اللہ ہی کے اختیار میں ہے تو کیا جو ایمان لائے ہیں انھیں اطمینان نہ ہوا کہ اگر اللہ چاہتا (تو) یقیناً سب لوگوں کو ہدایت عطا فرما دیتا اور کافروں کو پہنچتی رہے گی ان کے کرتوتوں کی وجہ سے کوئی سخت آفت یا ان کے گھر کے قریب (کوئی مصیبت) اترتی رہے گی یہاں تک کہ اللہ کا وعدہ (عذاب) آ جائے گا بے شک اللہ وعدے کے خلاف نہیں کرتا۔ || وَلَوْ أَنَّ قُرْءَانًۭا سُيِّرَتْ بِهِ ٱلْجِبَالُ أَوْ قُطِّعَتْ بِهِ ٱلْأَرْضُ أَوْ كُلِّمَ بِهِ ٱلْمَوْتَىٰ ۗ بَل لِّلَّهِ ٱلْأَمْرُ جَمِيعًا ۗ أَفَلَمْ يَا۟يْـَٔسِ ٱلَّذِينَ ءَامَنُوٓا۟ أَن لَّوْ يَشَآءُ ٱللَّهُ لَهَدَى ٱلنَّاسَ جَمِيعًۭا ۗ وَلَا يَزَالُ ٱلَّذِينَ كَفَرُوا۟ تُصِيبُهُم بِمَا صَنَعُوا۟ قَارِعَةٌ أَوْ تَحُلُّ قَرِيبًۭا مِّن دَارِهِمْ حَتَّىٰ يَأْتِىَ وَعْدُ ٱللَّهِ ۚ إِنَّ ٱللَّهَ لَا يُخْلِفُ ٱلْمِيعَادَ ﴿٣١﴾ — If there were a Qur'an by which mountains were moved, or the earth split apart, or the dead spoken to (still the disbelievers would not believe). Rather, the whole matter belongs to Allah. Have the believers not been assured that had Allah willed He would have guided all people? Calamity will continue to strike the disbelievers for what they have done, or settle near their homes, until Allah's promise (punishment) comes. Surely Allah does not break His promise. || وَلَوْ أَنَّ قُرْءَانًۭا سُيِّرَتْ بِهِ ٱلْجِبَالُ أَوْ قُطِّعَتْ بِهِ ٱلْأَرْضُ أَوْ كُلِّمَ بِهِ ٱلْمَوْتَىٰ ۗ بَل لِّلَّهِ ٱلْأَمْرُ جَمِيعًا ۗ أَفَلَمْ يَا۟يْـَٔسِ ٱلَّذِينَ ءَامَنُوٓا۟ أَن لَّوْ يَشَآءُ ٱللَّهُ لَهَدَى ٱلنَّاسَ جَمِيعًۭا ۗ وَلَا يَزَالُ ٱلَّذِينَ كَفَرُوا۟ تُصِيبُهُم بِمَا صَنَعُوا۟ قَارِعَةٌ أَوْ تَحُلُّ قَرِيبًۭا مِّن دَارِهِمْ حَتَّىٰ يَأْتِىَ وَعْدُ ٱللَّهِ ۚ إِنَّ ٱللَّهَ لَا يُخْلِفُ ٱلْمِيعَادَ ﴿٣١﴾ — او که داسې کوم قرآن وای چې غرونه پرې روان شوي وای، یا پرې ځمکه ټوټې ټوټې شوې وای، یا پرې له مړو سره خبرې شوې وای (بیا هم به کافرانو نه منل). بلکې ټول کار د الله په واک کې دی. نو آیا مومنانو ته ډاډ نه دی ورکړل شوی چې که الله غوښتي وای، (نو) یقیناً به یې ټولو خلکو ته هدایت کړی وای؟ او کافرانو ته به د خپلو کړنو له امله سخته افت رسېږي یا به د هغوی کور ته نږدې (مصیبت) راکوزېږي، تر څو د الله ژمنه (عذاب) راشي. بې شکه الله له ژمنې نه اوړي.
p وَلَقَدِ ٱسْتُهْزِئَ بِرُسُلٍۢ مِّن قَبْلِكَ فَأَمْلَيْتُ لِلَّذِينَ كَفَرُوا۟ ثُمَّ أَخَذْتُهُمْ ۖ فَكَيْفَ كَانَ عِقَابِ ﴿٣٢﴾ — اور یقیناً آپ (صلی اللہ علیہ وآلہ وسلم) سے پہلے بھی رسولوں کا مذاق اڑایا گیا تو جنھوں نے کفر کیا میں نے اُن کو مہلت دی پھر میں نے انھیں پکڑ لیا پھر (دیکھ لو کہ) میرا عذاب کیسا تھا؟ || وَلَقَدِ ٱسْتُهْزِئَ بِرُسُلٍۢ مِّن قَبْلِكَ فَأَمْلَيْتُ لِلَّذِينَ كَفَرُوا۟ ثُمَّ أَخَذْتُهُمْ ۖ فَكَيْفَ كَانَ عِقَابِ ﴿٣٢﴾ — Messengers before you were mocked; I gave respite to those who disbelieved, then I seized them — and how (terrible) was My penalty! || وَلَقَدِ ٱسْتُهْزِئَ بِرُسُلٍۢ مِّن قَبْلِكَ فَأَمْلَيْتُ لِلَّذِينَ كَفَرُوا۟ ثُمَّ أَخَذْتُهُمْ ۖ فَكَيْفَ كَانَ عِقَابِ ﴿٣٢﴾ — او یقیناً له تا مخکې هم پر رسولانو ټوکې شوې؛ نو ما کافرانو ته مهلت ورکړ، بیا مې ونیول؛ نو (وګوره) زما سزا څنګه وه!
