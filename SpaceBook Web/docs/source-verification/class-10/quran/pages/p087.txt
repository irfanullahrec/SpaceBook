p رَّبَّنَآ إِنِّىٓ أَسْكَنتُ مِن ذُرِّيَّتِى بِوَادٍ غَيْرِ ذِى زَرْعٍ عِندَ بَيْتِكَ ٱلْمُحَرَّمِ رَبَّنَا لِيُقِيمُوا۟ ٱلصَّلَوٰةَ فَٱجْعَلْ أَفْـِٔدَةًۭ مِّنَ ٱلنَّاسِ تَهْوِىٓ إِلَيْهِمْ وَٱرْزُقْهُم مِّنَ ٱلثَّمَرَٰتِ لَعَلَّهُمْ يَشْكُرُونَ ﴿٣٧﴾ — اے ہمارے رب! بے شک میں نے اپنی کچھ اولاد کو ایک ایسی وادی میں بسایا ہے جو بنجر تھی تیرے حرمت والے گھر کے پاس اے ہمارے رب! تاکہ وہ نماز قائم کریں پس تو لوگوں کے دل ان کی طرف مائل فرما دے اور انھیں پھلوں میں سے رزق عطا فرما تاکہ وہ شکر ادا کریں۔ || رَّبَّنَآ إِنِّىٓ أَسْكَنتُ مِن ذُرِّيَّتِى بِوَادٍ غَيْرِ ذِى زَرْعٍ عِندَ بَيْتِكَ ٱلْمُحَرَّمِ رَبَّنَا لِيُقِيمُوا۟ ٱلصَّلَوٰةَ فَٱجْعَلْ أَفْـِٔدَةًۭ مِّنَ ٱلنَّاسِ تَهْوِىٓ إِلَيْهِمْ وَٱرْزُقْهُم مِّنَ ٱلثَّمَرَٰتِ لَعَلَّهُمْ يَشْكُرُونَ ﴿٣٧﴾ — "Our Lord, I have settled some of my offspring in a barren valley near Your Sacred House — our Lord, that they may establish prayer. So make the hearts of people incline towards them, and provide them with fruits, that they may be grateful. || رَّبَّنَآ إِنِّىٓ أَسْكَنتُ مِن ذُرِّيَّتِى بِوَادٍ غَيْرِ ذِى زَرْعٍ عِندَ بَيْتِكَ ٱلْمُحَرَّمِ رَبَّنَا لِيُقِيمُوا۟ ٱلصَّلَوٰةَ فَٱجْعَلْ أَفْـِٔدَةًۭ مِّنَ ٱلنَّاسِ تَهْوِىٓ إِلَيْهِمْ وَٱرْزُقْهُم مِّنَ ٱلثَّمَرَٰتِ لَعَلَّهُمْ يَشْكُرُونَ ﴿٣٧﴾ — "ای زموږ ربه! بې شکه ما خپل ځینې اولاد په داسې یوه درّه کې مېشت کړل چې شاړه وه، ستا د درناوي وړ کور ته نږدې؛ ای زموږ ربه! څو لمونځ قایم کړي؛ نو د خلکو زړونه هغوی ته مایل کړه او له مېوو رزق ورکړه څو شکر ادا کړي.
p رَبَّنَآ إِنَّكَ تَعْلَمُ مَا نُخْفِى وَمَا نُعْلِنُ ۗ وَمَا يَخْفَىٰ عَلَى ٱللَّهِ مِن شَىْءٍۢ فِى ٱلْأَرْضِ وَلَا فِى ٱلسَّمَآءِ ﴿٣٨﴾ — اے ہمارے رب! بے شک تو جانتا ہے جو (کچھ) ہم چھپاتے اور جو ہم ظاہر کرتے ہیں اور اللہ سے کوئی چیز بھی چھپی ہوئی نہیں زمین میں اور نہ آسمان میں۔ || رَبَّنَآ إِنَّكَ تَعْلَمُ مَا نُخْفِى وَمَا نُعْلِنُ ۗ وَمَا يَخْفَىٰ عَلَى ٱللَّهِ مِن شَىْءٍۢ فِى ٱلْأَرْضِ وَلَا فِى ٱلسَّمَآءِ ﴿٣٨﴾ — Our Lord, You know what we conceal and what we reveal; nothing is hidden from Allah on earth or in the sky. || رَبَّنَآ إِنَّكَ تَعْلَمُ مَا نُخْفِى وَمَا نُعْلِنُ ۗ وَمَا يَخْفَىٰ عَلَى ٱللَّهِ مِن شَىْءٍۢ فِى ٱلْأَرْضِ وَلَا فِى ٱلسَّمَآءِ ﴿٣٨﴾ — ای زموږ ربه! بې شکه ته پوهېږې په هغه (څه) چې موږ یې پټوو او په هغه چې ښکاره کوو؛ او له الله هېڅ شی پټ نه دی، نه په ځمکه کې او نه په اسمان کې.
p ٱلْحَمْدُ لِلَّهِ ٱلَّذِى وَهَبَ لِى عَلَى ٱلْكِبَرِ إِسْمَٰعِيلَ وَإِسْحَٰقَ ۚ إِنَّ رَبِّى لَسَمِيعُ ٱلدُّعَآءِ ﴿٣٩﴾ — تمام تعریفیں اللہ کے لیے ہیں جس نے مجھے بڑھاپے میں اسماعیل (علیہ السلام) اور اسحاق (علیہ السلام) عطا فرمائے یقیناً میرا رب دعائیں ضرور سننے والا ہے۔ || ٱلْحَمْدُ لِلَّهِ ٱلَّذِى وَهَبَ لِى عَلَى ٱلْكِبَرِ إِسْمَٰعِيلَ وَإِسْحَٰقَ ۚ إِنَّ رَبِّى لَسَمِيعُ ٱلدُّعَآءِ ﴿٣٩﴾ — All praise is for Allah, who has granted me in my old age Isma'il and Ishaq. Surely my Lord is the Hearer of prayer. || ٱلْحَمْدُ لِلَّهِ ٱلَّذِى وَهَبَ لِى عَلَى ٱلْكِبَرِ إِسْمَٰعِيلَ وَإِسْحَٰقَ ۚ إِنَّ رَبِّى لَسَمِيعُ ٱلدُّعَآءِ ﴿٣٩﴾ — ټول ستاینه د الله لپاره ده چې ما ته یې په زړښت کې اسماعیل (علیه السلام) او اسحاق (علیه السلام) راکړل؛ یقیناً زما رب دعاګانې خامخا اورېدونکی دی.
p رَبِّ ٱجْعَلْنِى مُقِيمَ ٱلصَّلَوٰةِ وَمِن ذُرِّيَّتِى ۚ رَبَّنَا وَتَقَبَّلْ دُعَآءِ ﴿٤٠﴾ — اے میرے رب! مجھے نماز قائم کرنے والا بنا دیجیے اور میری اولاد میں سے بھی اے ہمارے رب! اور میری دعا قبول فرما۔ || رَبِّ ٱجْعَلْنِى مُقِيمَ ٱلصَّلَوٰةِ وَمِن ذُرِّيَّتِى ۚ رَبَّنَا وَتَقَبَّلْ دُعَآءِ ﴿٤٠﴾ — My Lord, make me one who establishes prayer, and (some) of my offspring too. Our Lord, accept my prayer. || رَبِّ ٱجْعَلْنِى مُقِيمَ ٱلصَّلَوٰةِ وَمِن ذُرِّيَّتِى ۚ رَبَّنَا وَتَقَبَّلْ دُعَآءِ ﴿٤٠﴾ — ای زما ربه! ما د لمانځه قایموونکی وګرځوه او زما له اولاده هم؛ ای زموږ ربه! زما دعا قبوله کړه.
p رَبَّنَا ٱغْفِرْ لِى وَلِوَٰلِدَىَّ وَلِلْمُؤْمِنِينَ يَوْمَ يَقُومُ ٱلْحِسَابُ ﴿٤١﴾ — اے ہمارے رب! مجھے اور میرے والدین کو بخش دیجیے اور سب ایمان والوں کو بھی جس دن حساب قائم ہوگا۔ || رَبَّنَا ٱغْفِرْ لِى وَلِوَٰلِدَىَّ وَلِلْمُؤْمِنِينَ يَوْمَ يَقُومُ ٱلْحِسَابُ ﴿٤١﴾ — Our Lord, forgive me and my parents and the believers on the Day the reckoning is established." || رَبَّنَا ٱغْفِرْ لِى وَلِوَٰلِدَىَّ وَلِلْمُؤْمِنِينَ يَوْمَ يَقُومُ ٱلْحِسَابُ ﴿٤١﴾ — ای زموږ ربه! ما او زما مور او پلار او ټول مومنان هم وبښه په هغه ورځ چې حساب قایم شي."
p وَلَا تَحْسَبَنَّ ٱللَّهَ غَٰفِلًا عَمَّا يَعْمَلُ ٱلظَّٰلِمُونَ ۚ إِنَّمَا يُؤَخِّرُهُمْ لِيَوْمٍۢ تَشْخَصُ فِيهِ ٱلْأَبْصَٰرُ ﴿٤٢﴾ — اور تم اللہ کو ہرگز بے خبر خیال نہ کرنا اُن (اعمال) سے جو ظالم کر رہے ہیں بے شک وہ تو انھیں اُس دن کے لیے مہلت دے رہا ہے جس دن آنکھیں (خوف کی وجہ سے) پھٹی کی پھٹی رہ جائیں گی۔ || وَلَا تَحْسَبَنَّ ٱللَّهَ غَٰفِلًا عَمَّا يَعْمَلُ ٱلظَّٰلِمُونَ ۚ إِنَّمَا يُؤَخِّرُهُمْ لِيَوْمٍۢ تَشْخَصُ فِيهِ ٱلْأَبْصَٰرُ ﴿٤٢﴾ — Never think that Allah is unaware of what the wrongdoers do; He only gives them respite until a Day when eyes will stare (in horror), || وَلَا تَحْسَبَنَّ ٱللَّهَ غَٰفِلًا عَمَّا يَعْمَلُ ٱلظَّٰلِمُونَ ۚ إِنَّمَا يُؤَخِّرُهُمْ لِيَوْمٍۢ تَشْخَصُ فِيهِ ٱلْأَبْصَٰرُ ﴿٤٢﴾ — او هېڅکله دا ګمان مه کوئ چې الله له هغو (عملونو) بې خبره دی چې ظالمان یې کوي؛ بې شکه هغه یې د هغې ورځې لپاره مهلت ورکوي چې پکې به سترګې (د وېرې له امله) پرانیستې پاتې شي.
p مُهْطِعِينَ مُقْنِعِى رُءُوسِهِمْ لَا يَرْتَدُّ إِلَيْهِمْ طَرْفُهُمْ ۖ وَأَفْـِٔدَتُهُمْ هَوَآءٌۭ ﴿٤٣﴾ — وہ اپنے سر اوپر اٹھائے ہوئے دوڑ رہے ہوں گے ان کی نگاہیں ان کی طرف واپس نہ آ سکیں گی اور اُن کے دل (خوف کی وجہ سے) بدحواس ہوں گے۔ || مُهْطِعِينَ مُقْنِعِى رُءُوسِهِمْ لَا يَرْتَدُّ إِلَيْهِمْ طَرْفُهُمْ ۖ وَأَفْـِٔدَتُهُمْ هَوَآءٌۭ ﴿٤٣﴾ — rushing ahead with heads raised, their gaze not returning to them, and their hearts void (with terror). || مُهْطِعِينَ مُقْنِعِى رُءُوسِهِمْ لَا يَرْتَدُّ إِلَيْهِمْ طَرْفُهُمْ ۖ وَأَفْـِٔدَتُهُمْ هَوَآءٌۭ ﴿٤٣﴾ — هغوی به خپل سرونه پورته نیولي منډې وهي، سترګې به یې بېرته هغوی ته نه راګرځي، او زړونه به یې (د وېرې له امله) حواس باختي وي.
p وَأَنذِرِ ٱلنَّاسَ يَوْمَ يَأْتِيهِمُ ٱلْعَذَابُ فَيَقُولُ ٱلَّذِينَ ظَلَمُوا۟ رَبَّنَآ أَخِّرْنَآ إِلَىٰٓ أَجَلٍۢ قَرِيبٍۢ نُّجِبْ دَعْوَتَكَ وَنَتَّبِعِ ٱلرُّسُلَ ۗ أَوَلَمْ تَكُونُوٓا۟ أَقْسَمْتُم مِّن قَبْلُ مَا لَكُم مِّن زَوَالٍۢ ﴿٤٤﴾ — اور آپ لوگوں کو اُس دن سے ڈرائیے جب اُن پر عذاب آئے گا تو جنھوں نے ظلم کیا وہ کہیں گے اے ہمارے رب! ہمیں تھوڑی مدت کے لیے مہلت دے دے ہم تیری دعوت قبول کریں گے اور رسولوں کی پیروی کریں گے (کہا جائے گا) کیا تم اس سے پہلے قسمیں نہیں کھاتے تھے کہ تمھارے لیے کوئی زوال نہیں ہے؟ || وَأَنذِرِ ٱلنَّاسَ يَوْمَ يَأْتِيهِمُ ٱلْعَذَابُ فَيَقُولُ ٱلَّذِينَ ظَلَمُوا۟ رَبَّنَآ أَخِّرْنَآ إِلَىٰٓ أَجَلٍۢ قَرِيبٍۢ نُّجِبْ دَعْوَتَكَ وَنَتَّبِعِ ٱلرُّسُلَ ۗ أَوَلَمْ تَكُونُوٓا۟ أَقْسَمْتُم مِّن قَبْلُ مَا لَكُم مِّن زَوَالٍۢ ﴿٤٤﴾ — And warn people of the Day when the punishment comes to them, when the wrongdoers will say: "Our Lord, give us respite for a short term; we will answer Your call and follow the messengers." (It will be said:) "Did you not swear before that there would be no end for you?" || وَأَنذِرِ ٱلنَّاسَ يَوْمَ يَأْتِيهِمُ ٱلْعَذَابُ فَيَقُولُ ٱلَّذِينَ ظَلَمُوا۟ رَبَّنَآ أَخِّرْنَآ إِلَىٰٓ أَجَلٍۢ قَرِيبٍۢ نُّجِبْ دَعْوَتَكَ وَنَتَّبِعِ ٱلرُّسُلَ ۗ أَوَلَمْ تَكُونُوٓا۟ أَقْسَمْتُم مِّن قَبْلُ مَا لَكُم مِّن زَوَالٍۢ ﴿٤٤﴾ — او خلک د هغې ورځې څخه ووېروه چې پرې عذاب راشي، نو هغه کسان چې ظلم یې کړی وایي به: "ای زموږ ربه! موږ ته لږ وخت مهلت راکړه، موږ به ستا بلنه ومنو او د رسولانو پیروي به وکړو." (ویل به شي:) "آیا تاسو له دې مخکې قسمونه نه خوړل چې ستاسو لپاره هېڅ زوال نشته؟"
