p قُل لَّئِنِ ٱجْتَمَعَتِ ٱلْإِنسُ وَٱلْجِنُّ عَلَىٰٓ أَن يَأْتُوا۟ بِمِثْلِ هَٰذَا ٱلْقُرْءَانِ لَا يَأْتُونَ بِمِثْلِهِۦ وَلَوْ كَانَ بَعْضُهُمْ لِبَعْضٍۢ ظَهِيرًۭا ﴿٨٨﴾ — آپ (صلی اللہ علیہ وآلہ وسلم) فرما دیجیے اگر تمام انسان اور جنات جمع ہو جائیں اس (بات) پر کہ وہ اس قرآن جیسا بنا لائیں (تب بھی) وہ اس جیسا نہیں لا سکیں گے اگرچہ وہ ایک دوسرے کے مددگار بھی بن جائیں۔ || قُل لَّئِنِ ٱجْتَمَعَتِ ٱلْإِنسُ وَٱلْجِنُّ عَلَىٰٓ أَن يَأْتُوا۟ بِمِثْلِ هَٰذَا ٱلْقُرْءَانِ لَا يَأْتُونَ بِمِثْلِهِۦ وَلَوْ كَانَ بَعْضُهُمْ لِبَعْضٍۢ ظَهِيرًۭا ﴿٨٨﴾ — Say: "If all mankind and jinn gathered to produce the like of this Qur'an, they could not produce its like, even if they helped one another." || قُل لَّئِنِ ٱجْتَمَعَتِ ٱلْإِنسُ وَٱلْجِنُّ عَلَىٰٓ أَن يَأْتُوا۟ بِمِثْلِ هَٰذَا ٱلْقُرْءَانِ لَا يَأْتُونَ بِمِثْلِهِۦ وَلَوْ كَانَ بَعْضُهُمْ لِبَعْضٍۢ ظَهِيرًۭا ﴿٨٨﴾ — ووایه: "که ټول انسانان او پېریان په دې (خبره) سره راټول شي چې د دې قرآن په څېر راوړي، (بیا هم) د هغه په څېر نه شي راوړای، که څه هم یو د بل مرستندویان شي."
p وَلَقَدْ صَرَّفْنَا لِلنَّاسِ فِى هَٰذَا ٱلْقُرْءَانِ مِن كُلِّ مَثَلٍۢ فَأَبَىٰٓ أَكْثَرُ ٱلنَّاسِ إِلَّا كُفُورًۭا ﴿٨٩﴾ — اور یقیناً ہم نے اس قرآن میں لوگوں کے (سمجھانے کے) لیے ہر قسم کی مثالیں طرح طرح سے بیان فرمائی ہیں لیکن اکثر لوگ انکار ہی کرتے رہے۔ || وَلَقَدْ صَرَّفْنَا لِلنَّاسِ فِى هَٰذَا ٱلْقُرْءَانِ مِن كُلِّ مَثَلٍۢ فَأَبَىٰٓ أَكْثَرُ ٱلنَّاسِ إِلَّا كُفُورًۭا ﴿٨٩﴾ — And We have certainly set forth for people in this Qur'an every kind of example in various ways, but most people refuse (anything) but disbelief. || وَلَقَدْ صَرَّفْنَا لِلنَّاسِ فِى هَٰذَا ٱلْقُرْءَانِ مِن كُلِّ مَثَلٍۢ فَأَبَىٰٓ أَكْثَرُ ٱلنَّاسِ إِلَّا كُفُورًۭا ﴿٨٩﴾ — او یقیناً موږ په دې قرآن کې د خلکو (د پوهولو) لپاره هر ډول بېلګې په بېلابېلو ډولونو بیان کړې، خو ډېری خلک یوازې انکار کوي.
p وَقَالُوا۟ لَن نُّؤْمِنَ لَكَ حَتَّىٰ تَفْجُرَ لَنَا مِنَ ٱلْأَرْضِ يَنۢبُوعًا ﴿٩٠﴾ — اور انھوں نے کہا ہم ہرگز آپ کی بات نہ مانیں گے یہاں تک کہ آپ ہمارے لیے زمین سے کوئی چشمہ جاری کر دیں۔ || وَقَالُوا۟ لَن نُّؤْمِنَ لَكَ حَتَّىٰ تَفْجُرَ لَنَا مِنَ ٱلْأَرْضِ يَنۢبُوعًا ﴿٩٠﴾ — And they said: "We will never believe you until you make a spring gush forth from the earth for us, || وَقَالُوا۟ لَن نُّؤْمِنَ لَكَ حَتَّىٰ تَفْجُرَ لَنَا مِنَ ٱلْأَرْضِ يَنۢبُوعًا ﴿٩٠﴾ — او هغوی وویل: "موږ به هېڅکله ستا خبره ونه منو تر څو چې زموږ لپاره له ځمکې کومه چینه روانه نه کړې،
p أَوْ تَكُونَ لَكَ جَنَّةٌۭ مِّن نَّخِيلٍۢ وَعِنَبٍۢ فَتُفَجِّرَ ٱلْأَنْهَٰرَ خِلَٰلَهَا تَفْجِيرًا ﴿٩١﴾ — یا آپ کے لیے کھجوروں اور انگوروں کا ایک باغ ہو پھر آپ اس کے درمیان بہت سی نہریں جاری کر دیں۔ || أَوْ تَكُونَ لَكَ جَنَّةٌۭ مِّن نَّخِيلٍۢ وَعِنَبٍۢ فَتُفَجِّرَ ٱلْأَنْهَٰرَ خِلَٰلَهَا تَفْجِيرًا ﴿٩١﴾ — or you have a garden of date-palms and grapes and make rivers gush forth abundantly in its midst, || أَوْ تَكُونَ لَكَ جَنَّةٌۭ مِّن نَّخِيلٍۢ وَعِنَبٍۢ فَتُفَجِّرَ ٱلْأَنْهَٰرَ خِلَٰلَهَا تَفْجِيرًا ﴿٩١﴾ — یا ستا لپاره د کجورو او انګورو یو باغ وي، بیا د هغه په منځ کې ډېرې ویالې روانې کړې،
p أَوْ تُسْقِطَ ٱلسَّمَآءَ كَمَا زَعَمْتَ عَلَيْنَا كِسَفًا أَوْ تَأْتِىَ بِٱللَّهِ وَٱلْمَلَٰٓئِكَةِ قَبِيلًا ﴿٩٢﴾ — یا آپ آسمان کو ہم پر ٹکڑے ٹکڑے کر کے گرا دیں جیسا کہ آپ کا خیال ہے یا آپ اللہ کو اور فرشتوں کو (ہمارے) سامنے لے آئیں۔ || أَوْ تُسْقِطَ ٱلسَّمَآءَ كَمَا زَعَمْتَ عَلَيْنَا كِسَفًا أَوْ تَأْتِىَ بِٱللَّهِ وَٱلْمَلَٰٓئِكَةِ قَبِيلًا ﴿٩٢﴾ — or you cause the sky to fall upon us in pieces, as you claim, or bring Allah and the angels before (us), || أَوْ تُسْقِطَ ٱلسَّمَآءَ كَمَا زَعَمْتَ عَلَيْنَا كِسَفًا أَوْ تَأْتِىَ بِٱللَّهِ وَٱلْمَلَٰٓئِكَةِ قَبِيلًا ﴿٩٢﴾ — یا اسمان پر موږ ټوټې ټوټې راوغورځوې لکه څنګه چې ستا ګمان دی، یا الله او پرښتې (زموږ) مخې ته راولې،
p أَوْ يَكُونَ لَكَ بَيْتٌۭ مِّن زُخْرُفٍ أَوْ تَرْقَىٰ فِى ٱلسَّمَآءِ وَلَن نُّؤْمِنَ لِرُقِيِّكَ حَتَّىٰ تُنَزِّلَ عَلَيْنَا كِتَٰبًۭا نَّقْرَؤُهُۥ ۗ قُلْ سُبْحَانَ رَبِّى هَلْ كُنتُ إِلَّا بَشَرًۭا رَّسُولًۭا ﴿٩٣﴾ — یا آپ کے لیے سونے کا ایک گھر ہو یا آپ آسمان پر چڑھ جائیں اور ہم آپ کے چڑھنے کو بھی نہ مانیں گے یہاں تک کہ آپ ایک کتاب اُتار لائیں جسے ہم خود پڑھیں آپ (صلی اللہ علیہ وآلہ وسلم) فرما دیجیے میرا رب پاک ہے میں تو ایک بشر (اور اللہ کا) رسول ہوں۔ || أَوْ يَكُونَ لَكَ بَيْتٌۭ مِّن زُخْرُفٍ أَوْ تَرْقَىٰ فِى ٱلسَّمَآءِ وَلَن نُّؤْمِنَ لِرُقِيِّكَ حَتَّىٰ تُنَزِّلَ عَلَيْنَا كِتَٰبًۭا نَّقْرَؤُهُۥ ۗ قُلْ سُبْحَانَ رَبِّى هَلْ كُنتُ إِلَّا بَشَرًۭا رَّسُولًۭا ﴿٩٣﴾ — or you have a house of gold, or you ascend into the sky — and we will not believe in your ascension until you bring down to us a book we can read." Say: "Glorified is my Lord! Am I anything but a human being, a messenger (of Allah)?" || أَوْ يَكُونَ لَكَ بَيْتٌۭ مِّن زُخْرُفٍ أَوْ تَرْقَىٰ فِى ٱلسَّمَآءِ وَلَن نُّؤْمِنَ لِرُقِيِّكَ حَتَّىٰ تُنَزِّلَ عَلَيْنَا كِتَٰبًۭا نَّقْرَؤُهُۥ ۗ قُلْ سُبْحَانَ رَبِّى هَلْ كُنتُ إِلَّا بَشَرًۭا رَّسُولًۭا ﴿٩٣﴾ — یا ستا لپاره د سرو زرو یو کور وي، یا اسمان ته وخېژې؛ او موږ به ستا ختل هم ونه منو تر څو چې یو کتاب راکښته کړې چې موږ یې پخپله ولولو." ووایه: "زما رب پاک دی؛ زه خو یو بشر (او د الله) رسول یم."
p وَمَا مَنَعَ ٱلنَّاسَ أَن يُؤْمِنُوٓا۟ إِذْ جَآءَهُمُ ٱلْهُدَىٰٓ إِلَّآ أَن قَالُوٓا۟ أَبَعَثَ ٱللَّهُ بَشَرًۭا رَّسُولًۭا ﴿٩٤﴾ — اور لوگوں کو ایمان لانے سے نہیں روکا جب ان کے پاس ہدایت آ گئی مگر اسی بات نے کہ انھوں نے کہا کیا اللہ نے بشر کو رسول بنا کر بھیجا ہے؟ || وَمَا مَنَعَ ٱلنَّاسَ أَن يُؤْمِنُوٓا۟ إِذْ جَآءَهُمُ ٱلْهُدَىٰٓ إِلَّآ أَن قَالُوٓا۟ أَبَعَثَ ٱللَّهُ بَشَرًۭا رَّسُولًۭا ﴿٩٤﴾ — Nothing prevented people from believing when guidance came to them except that they said: "Has Allah sent a human being as a messenger?" || وَمَا مَنَعَ ٱلنَّاسَ أَن يُؤْمِنُوٓا۟ إِذْ جَآءَهُمُ ٱلْهُدَىٰٓ إِلَّآ أَن قَالُوٓا۟ أَبَعَثَ ٱللَّهُ بَشَرًۭا رَّسُولًۭا ﴿٩٤﴾ — او خلک یې له ایمان راوړلو منع نه کړل کله چې هدایت ورته راغی مګر همدې خبرې چې وې ویل: "آیا الله یو بشر رسول کړی لېږلی؟"
p قُل لَّوْ كَانَ فِى ٱلْأَرْضِ مَلَٰٓئِكَةٌۭ يَمْشُونَ مُطْمَئِنِّينَ لَنَزَّلْنَا عَلَيْهِم مِّنَ ٱلسَّمَآءِ مَلَكًۭا رَّسُولًۭا ﴿٩٥﴾ — آپ (صلی اللہ علیہ وآلہ وسلم) فرما دیجیے کہ اگر زمین میں (انسانوں کے بجائے) فرشتے ہوتے جو اطمینان سے چلتے پھرتے تو ضرور ہم ان پر آسمانوں سے فرشتہ رسول (بنا کر) بھیجتے۔ || قُل لَّوْ كَانَ فِى ٱلْأَرْضِ مَلَٰٓئِكَةٌۭ يَمْشُونَ مُطْمَئِنِّينَ لَنَزَّلْنَا عَلَيْهِم مِّنَ ٱلسَّمَآءِ مَلَكًۭا رَّسُولًۭا ﴿٩٥﴾ — Say: "If there were angels on the earth (instead of humans), walking about in peace, We would surely have sent down to them from the sky an angel as messenger." || قُل لَّوْ كَانَ فِى ٱلْأَرْضِ مَلَٰٓئِكَةٌۭ يَمْشُونَ مُطْمَئِنِّينَ لَنَزَّلْنَا عَلَيْهِم مِّنَ ٱلسَّمَآءِ مَلَكًۭا رَّسُولًۭا ﴿٩٥﴾ — ووایه: "که په ځمکه کې (د انسانانو پر ځای) پرښتې وای چې په ډاډ ګرځېدې، نو موږ به خامخا پرې له اسمانونو یوه پرښته رسول (کړې) لېږلې وای."
p قُلْ كَفَىٰ بِٱللَّهِ شَهِيدًۢا بَيْنِى وَبَيْنَكُمْ ۚ إِنَّهُۥ كَانَ بِعِبَادِهِۦ خَبِيرًۢا بَصِيرًۭا ﴿٩٦﴾ — آپ (صلی اللہ علیہ وآلہ وسلم) فرما دیجیے میرے اور تمھارے درمیان اللہ ہی گواہ کافی ہے بے شک وہ اپنے بندوں سے بڑا باخبر خوب دیکھنے والا ہے۔ || قُلْ كَفَىٰ بِٱللَّهِ شَهِيدًۢا بَيْنِى وَبَيْنَكُمْ ۚ إِنَّهُۥ كَانَ بِعِبَادِهِۦ خَبِيرًۢا بَصِيرًۭا ﴿٩٦﴾ — Say: "Allah suffices as a witness between me and you. Surely He is All-Aware, All-Seeing of His servants." || قُلْ كَفَىٰ بِٱللَّهِ شَهِيدًۢا بَيْنِى وَبَيْنَكُمْ ۚ إِنَّهُۥ كَانَ بِعِبَادِهِۦ خَبِيرًۢا بَصِيرًۭا ﴿٩٦﴾ — ووایه: "زما او ستاسو ترمنځ یوازې الله شاهد بس دی؛ بې شکه هغه له خپلو بندګانو ډېر باخبره، ښه لیدونکی دی."

We will gather them on the Day of Resurrection on their faces, blind, dumb and deaf; their abode is Hell.
