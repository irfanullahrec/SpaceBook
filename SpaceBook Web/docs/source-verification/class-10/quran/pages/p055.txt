p يَٰٓأَيُّهَا ٱلنَّاسُ قَدْ جَآءَتْكُم مَّوْعِظَةٌۭ مِّن رَّبِّكُمْ وَشِفَآءٌۭ لِّمَا فِى ٱلصُّدُورِ وَهُدًۭى وَرَحْمَةٌۭ لِّلْمُؤْمِنِينَ ﴿٥٧﴾ — اے لوگو! یقیناً تمھارے پاس تمھارے رب کی طرف سے نصیحت آ چکی ہے اور اُن (بیماریوں) کی شفا جو سینوں میں ہیں اور مومنوں کے لیے ہدایت اور رحمت ہے۔ || يَٰٓأَيُّهَا ٱلنَّاسُ قَدْ جَآءَتْكُم مَّوْعِظَةٌۭ مِّن رَّبِّكُمْ وَشِفَآءٌۭ لِّمَا فِى ٱلصُّدُورِ وَهُدًۭى وَرَحْمَةٌۭ لِّلْمُؤْمِنِينَ ﴿٥٧﴾ — O people! There has come to you an admonition from your Lord, a healing for the (diseases) in the breasts, and guidance and mercy for the believers. || يَٰٓأَيُّهَا ٱلنَّاسُ قَدْ جَآءَتْكُم مَّوْعِظَةٌۭ مِّن رَّبِّكُمْ وَشِفَآءٌۭ لِّمَا فِى ٱلصُّدُورِ وَهُدًۭى وَرَحْمَةٌۭ لِّلْمُؤْمِنِينَ ﴿٥٧﴾ — ای خلکو! یقیناً تاسو ته ستاسو د رب له لوري نصیحت راغلی، او د هغو (ناروغیو) شفا چې په سینو کې دي، او د مومنانو لپاره هدایت او رحمت.
p قُلْ بِفَضْلِ ٱللَّهِ وَبِرَحْمَتِهِۦ فَبِذَٰلِكَ فَلْيَفْرَحُوا۟ هُوَ خَيْرٌۭ مِّمَّا يَجْمَعُونَ ﴿٥٨﴾ — آپ (صلی اللہ علیہ وآلہ وسلم) فرما دیجیے یہ (سب کچھ) اللہ کے فضل اور اس کی رحمت سے ہے لہٰذا انھیں اس پر خوش ہونا چاہیے یہ ان تمام چیزوں سے بہتر ہے جو وہ جمع کر رہے ہیں۔ || قُلْ بِفَضْلِ ٱللَّهِ وَبِرَحْمَتِهِۦ فَبِذَٰلِكَ فَلْيَفْرَحُوا۟ هُوَ خَيْرٌۭ مِّمَّا يَجْمَعُونَ ﴿٥٨﴾ — Say: "(All this is) by the grace of Allah and His mercy; so in that let them rejoice. It is better than all they gather." || قُلْ بِفَضْلِ ٱللَّهِ وَبِرَحْمَتِهِۦ فَبِذَٰلِكَ فَلْيَفْرَحُوا۟ هُوَ خَيْرٌۭ مِّمَّا يَجْمَعُونَ ﴿٥٨﴾ — ووایه: "دا (ټول) د الله په فضل او رحمت دي؛ نو باید پر همدې خوشحاله شي؛ دا له ټولو هغو شیانو غوره دی چې دوی یې راټولوي."
p قُلْ أَرَءَيْتُم مَّآ أَنزَلَ ٱللَّهُ لَكُم مِّن رِّزْقٍۢ فَجَعَلْتُم مِّنْهُ حَرَامًۭا وَحَلَٰلًۭا قُلْ ءَآللَّهُ أَذِنَ لَكُمْ ۖ أَمْ عَلَى ٱللَّهِ تَفْتَرُونَ ﴿٥٩﴾ — آپ (صلی اللہ علیہ وآلہ وسلم) فرما دیجیے بھلا تم دیکھو جو اللہ نے تمھارے لیے رزق نازل فرمایا ہے تو تم نے اُس میں سے کچھ حرام اور کچھ حلال ٹھہرا لیا ہے آپ (صلی اللہ علیہ وآلہ وسلم) فرما دیجیے کیا اللہ نے تمھیں اجازت دی ہے یا تم اللہ پر جھوٹ باندھ رہے ہو؟ || قُلْ أَرَءَيْتُم مَّآ أَنزَلَ ٱللَّهُ لَكُم مِّن رِّزْقٍۢ فَجَعَلْتُم مِّنْهُ حَرَامًۭا وَحَلَٰلًۭا قُلْ ءَآللَّهُ أَذِنَ لَكُمْ ۖ أَمْ عَلَى ٱللَّهِ تَفْتَرُونَ ﴿٥٩﴾ — Say: "Tell me, the provision Allah has sent down for you — you have made some of it unlawful and some lawful." Say: "Has Allah permitted you, or do you invent lies against Allah?" || قُلْ أَرَءَيْتُم مَّآ أَنزَلَ ٱللَّهُ لَكُم مِّن رِّزْقٍۢ فَجَعَلْتُم مِّنْهُ حَرَامًۭا وَحَلَٰلًۭا قُلْ ءَآللَّهُ أَذِنَ لَكُمْ ۖ أَمْ عَلَى ٱللَّهِ تَفْتَرُونَ ﴿٥٩﴾ — ووایه: "ووایئ، هغه رزق چې الله درته نازل کړی، تاسو ځینې یې حرام او ځینې حلال ګرځولی." ووایه: "آیا الله درته اجازه درکړې، که پر الله دروغ تړئ؟"
p وَمَا ظَنُّ ٱلَّذِينَ يَفْتَرُونَ عَلَى ٱللَّهِ ٱلْكَذِبَ يَوْمَ ٱلْقِيَٰمَةِ ۗ إِنَّ ٱللَّهَ لَذُو فَضْلٍ عَلَى ٱلنَّاسِ وَلَٰكِنَّ أَكْثَرَهُمْ لَا يَشْكُرُونَ ﴿٦٠﴾ — اور ان لوگوں کا کیا گمان ہے جو اللہ پر جھوٹ باندھتے ہیں قیامت کے بارے میں؟ یقیناً اللہ لوگوں پر فضل فرمانے والا ہے لیکن ان میں سے اکثر شکر ادا نہیں کرتے۔ || وَمَا ظَنُّ ٱلَّذِينَ يَفْتَرُونَ عَلَى ٱللَّهِ ٱلْكَذِبَ يَوْمَ ٱلْقِيَٰمَةِ ۗ إِنَّ ٱللَّهَ لَذُو فَضْلٍ عَلَى ٱلنَّاسِ وَلَٰكِنَّ أَكْثَرَهُمْ لَا يَشْكُرُونَ ﴿٦٠﴾ — And what will those who invent lies against Allah think on the Day of Resurrection? Surely Allah is gracious to people, but most of them do not give thanks. || وَمَا ظَنُّ ٱلَّذِينَ يَفْتَرُونَ عَلَى ٱللَّهِ ٱلْكَذِبَ يَوْمَ ٱلْقِيَٰمَةِ ۗ إِنَّ ٱللَّهَ لَذُو فَضْلٍ عَلَى ٱلنَّاسِ وَلَٰكِنَّ أَكْثَرَهُمْ لَا يَشْكُرُونَ ﴿٦٠﴾ — او هغه خلک چې پر الله دروغ تړي، د قیامت په اړه یې څه ګومان دی؟ یقیناً الله پر خلکو فضل کوونکی دی، خو ډېری یې شکر نه ادا کوي.
p وَمَا تَكُونُ فِى شَأْنٍۢ وَمَا تَتْلُوا۟ مِنْهُ مِن قُرْءَانٍۢ وَلَا تَعْمَلُونَ مِنْ عَمَلٍ إِلَّا كُنَّا عَلَيْكُمْ شُهُودًا إِذْ تُفِيضُونَ فِيهِ ۚ وَمَا يَعْزُبُ عَن رَّبِّكَ مِن مِّثْقَالِ ذَرَّةٍۢ فِى ٱلْأَرْضِ وَلَا فِى ٱلسَّمَآءِ وَلَآ أَصْغَرَ مِن ذَٰلِكَ وَلَآ أَكْبَرَ إِلَّا فِى كِتَٰبٍۢ مُّبِينٍ ﴿٦١﴾ — اور آپ جس حال میں بھی ہوتے ہیں اور آپ اُس کی طرف سے (نازل شدہ) قرآن میں سے جو بھی تلاوت کرتے ہیں اور تم لوگ جو بھی عمل کرتے ہو ہم تم پر گواہ ہوتے ہیں جب تم اس میں مشغول ہوتے ہو اور آپ کے رب سے ذرہ برابر بھی کوئی چیز نہیں چھپتی زمین میں اور نہ آسمان میں اور نہ اس (ذرہ) سے کوئی چھوٹی (چیز) اور نہ بڑی مگر واضح کتاب (لوحِ محفوظ) میں (درج) ہے۔ || وَمَا تَكُونُ فِى شَأْنٍۢ وَمَا تَتْلُوا۟ مِنْهُ مِن قُرْءَانٍۢ وَلَا تَعْمَلُونَ مِنْ عَمَلٍ إِلَّا كُنَّا عَلَيْكُمْ شُهُودًا إِذْ تُفِيضُونَ فِيهِ ۚ وَمَا يَعْزُبُ عَن رَّبِّكَ مِن مِّثْقَالِ ذَرَّةٍۢ فِى ٱلْأَرْضِ وَلَا فِى ٱلسَّمَآءِ وَلَآ أَصْغَرَ مِن ذَٰلِكَ وَلَآ أَكْبَرَ إِلَّا فِى كِتَٰبٍۢ مُّبِينٍ ﴿٦١﴾ — In whatever state you are, whatever you recite of the Qur'an (sent down) from Him, and whatever deed you (people) do, We are witness over you when you are engaged in it. Not an atom's weight escapes your Lord on earth or in the sky, nor anything smaller or larger, but it is (recorded) in a clear Book (the Preserved Tablet). || وَمَا تَكُونُ فِى شَأْنٍۢ وَمَا تَتْلُوا۟ مِنْهُ مِن قُرْءَانٍۢ وَلَا تَعْمَلُونَ مِنْ عَمَلٍ إِلَّا كُنَّا عَلَيْكُمْ شُهُودًا إِذْ تُفِيضُونَ فِيهِ ۚ وَمَا يَعْزُبُ عَن رَّبِّكَ مِن مِّثْقَالِ ذَرَّةٍۢ فِى ٱلْأَرْضِ وَلَا فِى ٱلسَّمَآءِ وَلَآ أَصْغَرَ مِن ذَٰلِكَ وَلَآ أَكْبَرَ إِلَّا فِى كِتَٰبٍۢ مُّبِينٍ ﴿٦١﴾ — او ته په هر حال کې چې یې، او د هغه له (نازل شوي) قرآن څخه چې څه لولې، او تاسو خلک چې هر عمل کوئ، موږ پر تاسو شاهدان یو کله چې پکې بوخت یاست؛ او ستا له رب څخه د ذرې په اندازه هم څه نه پټېږي، نه په ځمکه کې او نه په اسمان کې، او نه له هغې (ذرې) کوچنی (شی) او نه لوی، مګر په ښکاره کتاب (لوح محفوظ) کې (لیکل شوی) دی.
p أَلَآ إِنَّ أَوْلِيَآءَ ٱللَّهِ لَا خَوْفٌ عَلَيْهِمْ وَلَا هُمْ يَحْزَنُونَ ﴿٦٢﴾ — آگاہ ہو جاؤ! بے شک جو اللہ کے دوست ہیں اُن پر نہ کوئی خوف ہے اور نہ وہ غمگین ہوں گے۔ || أَلَآ إِنَّ أَوْلِيَآءَ ٱللَّهِ لَا خَوْفٌ عَلَيْهِمْ وَلَا هُمْ يَحْزَنُونَ ﴿٦٢﴾ — Beware! Surely on the friends of Allah there is no fear, nor will they grieve — || أَلَآ إِنَّ أَوْلِيَآءَ ٱللَّهِ لَا خَوْفٌ عَلَيْهِمْ وَلَا هُمْ يَحْزَنُونَ ﴿٦٢﴾ — خبر اوسئ! بې شکه هغه چې د الله دوستان دي، پر هغوی نه وېره ده او نه به غمجن شي —
p ٱلَّذِينَ ءَامَنُوا۟ وَكَانُوا۟ يَتَّقُونَ ﴿٦٣﴾ — جو ایمان لائے اور ڈرتے رہے۔ || ٱلَّذِينَ ءَامَنُوا۟ وَكَانُوا۟ يَتَّقُونَ ﴿٦٣﴾ — those who believe and remain God-fearing. || ٱلَّذِينَ ءَامَنُوا۟ وَكَانُوا۟ يَتَّقُونَ ﴿٦٣﴾ — هغه چې ایمان یې راوړ او وېرېدل.
p لَهُمُ ٱلْبُشْرَىٰ فِى ٱلْحَيَوٰةِ ٱلدُّنْيَا وَفِى ٱلْءَاخِرَةِ ۚ لَا تَبْدِيلَ لِكَلِمَٰتِ ٱللَّهِ ۚ ذَٰلِكَ هُوَ ٱلْفَوْزُ ٱلْعَظِيمُ ﴿٦٤﴾ — اُن کے لیے خوش خبری ہے دنیا کی زندگی میں (بھی) اور آخرت میں (بھی) اللہ کی باتوں میں کوئی تبدیلی نہیں ہو سکتی یہی بہت بڑی کامیابی ہے۔ || لَهُمُ ٱلْبُشْرَىٰ فِى ٱلْحَيَوٰةِ ٱلدُّنْيَا وَفِى ٱلْءَاخِرَةِ ۚ لَا تَبْدِيلَ لِكَلِمَٰتِ ٱللَّهِ ۚ ذَٰلِكَ هُوَ ٱلْفَوْزُ ٱلْعَظِيمُ ﴿٦٤﴾ — For them is good news in the life of this world and in the Hereafter. There is no changing the words of Allah; that is the great success. || لَهُمُ ٱلْبُشْرَىٰ فِى ٱلْحَيَوٰةِ ٱلدُّنْيَا وَفِى ٱلْءَاخِرَةِ ۚ لَا تَبْدِيلَ لِكَلِمَٰتِ ٱللَّهِ ۚ ذَٰلِكَ هُوَ ٱلْفَوْزُ ٱلْعَظِيمُ ﴿٦٤﴾ — د هغوی لپاره په دنیوي ژوند کې (هم) زېری دی او په آخرت کې (هم)؛ د الله په خبرو کې هېڅ بدلون نه شي کېدای؛ همدا ډېره لویه بریا ده.
p وَلَا يَحْزُنكَ قَوْلُهُمْ ۘ إِنَّ ٱلْعِزَّةَ لِلَّهِ جَمِيعًا ۚ هُوَ ٱلسَّمِيعُ ٱلْعَلِيمُ ﴿٦٥﴾ — لہٰذا ان کی باتیں آپ (صلی اللہ علیہ وآلہ وسلم) کو غمگین نہ کریں بے شک ساری عزت اللہ ہی کے لیے ہے وہ خوب سننے والا خوب جاننے والا ہے۔ || وَلَا يَحْزُنكَ قَوْلُهُمْ ۘ إِنَّ ٱلْعِزَّةَ لِلَّهِ جَمِيعًا ۚ هُوَ ٱلسَّمِيعُ ٱلْعَلِيمُ ﴿٦٥﴾ — So let not their words grieve you. Surely all honour belongs to Allah; He is the All-Hearing, the All-Knowing. || وَلَا يَحْزُنكَ قَوْلُهُمْ ۘ إِنَّ ٱلْعِزَّةَ لِلَّهِ جَمِيعًا ۚ هُوَ ٱلسَّمِيعُ ٱلْعَلِيمُ ﴿٦٥﴾ — نو د هغوی خبرې دې تا غمجن نه کړي؛ بې شکه ټول عزت یوازې د الله لپاره دی؛ هغه ښه اورېدونکی، ښه پوه دی.
p أَلَآ إِنَّ لِلَّهِ مَن فِى ٱلسَّمَٰوَٰتِ وَمَن فِى ٱلْأَرْضِ ۗ وَمَا يَتَّبِعُ ٱلَّذِينَ يَدْعُونَ مِن دُونِ ٱللَّهِ شُرَكَآءَ ۚ إِن يَتَّبِعُونَ إِلَّا ٱلظَّنَّ وَإِنْ هُمْ إِلَّا يَخْرُصُونَ ﴿٦٦﴾ — آگاہ ہو جاؤ! بے شک اللہ ہی کے لیے ہے جو کوئی آسمانوں میں ہے اور جو کوئی زمینوں میں ہے اور جو لوگ اللہ کے سوا شریکوں کو پکارتے ہیں (درحقیقت) وہ کسی کی پیروی نہیں کر رہے وہ تو صرف گمان کی پیروی کرتے ہیں اور وہ صرف جھوٹا اندازہ لگاتے ہیں۔ || أَلَآ إِنَّ لِلَّهِ مَن فِى ٱلسَّمَٰوَٰتِ وَمَن فِى ٱلْأَرْضِ ۗ وَمَا يَتَّبِعُ ٱلَّذِينَ يَدْعُونَ مِن دُونِ ٱللَّهِ شُرَكَآءَ ۚ إِن يَتَّبِعُونَ إِلَّا ٱلظَّنَّ وَإِنْ هُمْ إِلَّا يَخْرُصُونَ ﴿٦٦﴾ — Beware! To Allah belongs whoever is in the heavens and whoever is on the earth.
