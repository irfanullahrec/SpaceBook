p قُلْ إِنِّى عَلَىٰ بَيِّنَةٍۢ مِّن رَّبِّى وَكَذَّبْتُم بِهِۦ ۚ مَا عِندِى مَا تَسْتَعْجِلُونَ بِهِۦٓ ۚ إِنِ ٱلْحُكْمُ إِلَّا لِلَّهِ ۖ يَقُصُّ ٱلْحَقَّ ۖ وَهُوَ خَيْرُ ٱلْفَٰصِلِينَ ﴿٥٧﴾ — آپ (صلی اللہ علیہ وآلہ وسلم) فرما دیجیے بے شک میں اپنے رب کی طرف سے ایک واضح دلیل پر (قائم) ہوں اور تم نے اسے جھٹلا دیا ہے میرے پاس وہ نہیں ہے جس کی تم جلدی کرتے ہو حکم کا اختیار صرف اللہ کو ہے وہ حق بیان فرماتا ہے اور وہ سب سے بہتر فیصلہ فرمانے والا ہے۔ || قُلْ إِنِّى عَلَىٰ بَيِّنَةٍۢ مِّن رَّبِّى وَكَذَّبْتُم بِهِۦ ۚ مَا عِندِى مَا تَسْتَعْجِلُونَ بِهِۦٓ ۚ إِنِ ٱلْحُكْمُ إِلَّا لِلَّهِ ۖ يَقُصُّ ٱلْحَقَّ ۖ وَهُوَ خَيْرُ ٱلْفَٰصِلِينَ ﴿٥٧﴾ — Say: "I stand on clear proof from my Lord, and you have denied it. I do not have what you are hastening for. Judgement belongs only to Allah; He relates the truth, and He is the best of judges." || قُلْ إِنِّى عَلَىٰ بَيِّنَةٍۢ مِّن رَّبِّى وَكَذَّبْتُم بِهِۦ ۚ مَا عِندِى مَا تَسْتَعْجِلُونَ بِهِۦٓ ۚ إِنِ ٱلْحُكْمُ إِلَّا لِلَّهِ ۖ يَقُصُّ ٱلْحَقَّ ۖ وَهُوَ خَيْرُ ٱلْفَٰصِلِينَ ﴿٥٧﴾ — ووایه: "زه د خپل رب له لوري پر ښکاره دلیل (ولاړ) یم او تاسو هغه دروغ ګڼلی؛ هغه څه چې تاسو یې بېړه کوئ، زما سره نشته. د حکم واک یوازې د الله دی؛ هغه حق بیانوي او هغه تر ټولو غوره پرېکړه کوونکی دی."
p قُل لَّوْ أَنَّ عِندِى مَا تَسْتَعْجِلُونَ بِهِۦ لَقُضِىَ ٱلْأَمْرُ بَيْنِى وَبَيْنَكُمْ ۗ وَٱللَّهُ أَعْلَمُ بِٱلظَّٰلِمِينَ ﴿٥٨﴾ — آپ (صلی اللہ علیہ وآلہ وسلم) فرما دیجیے اگر میرے پاس وہ (چیز) ہوتی جس کی تم جلدی کر رہے ہو (تو) یقیناً میرے اور تمھارے درمیان فیصلہ ہو چکا ہوتا اور اللہ ظالموں کو خوب جاننے والا ہے۔ || قُل لَّوْ أَنَّ عِندِى مَا تَسْتَعْجِلُونَ بِهِۦ لَقُضِىَ ٱلْأَمْرُ بَيْنِى وَبَيْنَكُمْ ۗ وَٱللَّهُ أَعْلَمُ بِٱلظَّٰلِمِينَ ﴿٥٨﴾ — Say: "If I had what you are hastening for, the matter between me and you would surely have been decided; and Allah knows best the wrongdoers." || قُل لَّوْ أَنَّ عِندِى مَا تَسْتَعْجِلُونَ بِهِۦ لَقُضِىَ ٱلْأَمْرُ بَيْنِى وَبَيْنَكُمْ ۗ وَٱللَّهُ أَعْلَمُ بِٱلظَّٰلِمِينَ ﴿٥٨﴾ — ووایه: "که هغه (شی) چې تاسو یې بېړه کوئ زما سره وای، (نو) یقیناً به زما او ستاسو ترمنځ پرېکړه شوې وای؛ او الله ظالمان ښه پېژني."
p ۞ وَعِندَهُۥ مَفَاتِحُ ٱلْغَيْبِ لَا يَعْلَمُهَآ إِلَّا هُوَ ۚ وَيَعْلَمُ مَا فِى ٱلْبَرِّ وَٱلْبَحْرِ ۚ وَمَا تَسْقُطُ مِن وَرَقَةٍ إِلَّا يَعْلَمُهَا وَلَا حَبَّةٍۢ فِى ظُلُمَٰتِ ٱلْأَرْضِ وَلَا رَطْبٍۢ وَلَا يَابِسٍ إِلَّا فِى كِتَٰبٍۢ مُّبِينٍۢ ﴿٥٩﴾ — اور اسی (اللہ) کے پاس غیب کی چابیاں ہیں انھیں اس کے سوا (از خود) کوئی نہیں جانتا اور وہ جانتا ہے جو کچھ خشکی اور تری میں ہے اور کوئی پتا نہیں گرتا مگر وہ اسے جانتا ہے اور نہ زمین کے اندھیروں میں کوئی (ایسا) دانہ ہے اور نہ کوئی تر چیز ہے اور نہ کوئی خشک چیز مگر واضح کتاب میں (لکھی ہوئی) ہے۔ || ۞ وَعِندَهُۥ مَفَاتِحُ ٱلْغَيْبِ لَا يَعْلَمُهَآ إِلَّا هُوَ ۚ وَيَعْلَمُ مَا فِى ٱلْبَرِّ وَٱلْبَحْرِ ۚ وَمَا تَسْقُطُ مِن وَرَقَةٍ إِلَّا يَعْلَمُهَا وَلَا حَبَّةٍۢ فِى ظُلُمَٰتِ ٱلْأَرْضِ وَلَا رَطْبٍۢ وَلَا يَابِسٍ إِلَّا فِى كِتَٰبٍۢ مُّبِينٍۢ ﴿٥٩﴾ — With Him (Allah) are the keys of the unseen; none knows them (of himself) but He. He knows what is on land and in the sea; not a leaf falls but He knows it, nor is there a grain in the darkness of the earth, nor anything moist or dry, but it is (written) in a clear Book. || ۞ وَعِندَهُۥ مَفَاتِحُ ٱلْغَيْبِ لَا يَعْلَمُهَآ إِلَّا هُوَ ۚ وَيَعْلَمُ مَا فِى ٱلْبَرِّ وَٱلْبَحْرِ ۚ وَمَا تَسْقُطُ مِن وَرَقَةٍ إِلَّا يَعْلَمُهَا وَلَا حَبَّةٍۢ فِى ظُلُمَٰتِ ٱلْأَرْضِ وَلَا رَطْبٍۢ وَلَا يَابِسٍ إِلَّا فِى كِتَٰبٍۢ مُّبِينٍۢ ﴿٥٩﴾ — او د هغه (الله) سره د غیب کلي دي؛ له هغه پرته یې (پخپله) هېڅوک نه پېژني؛ او هغه پوهېږي په هغه څه چې په وچه او لمده کې دي؛ او هېڅ پاڼه نه غورځي مګر هغه پرې پوهېږي، او نه د ځمکې په تیارو کې کومه دانه، او نه کوم لوند او نه کوم وچ شی، مګر په ښکاره کتاب کې (لیکل شوی) دی.
p وَهُوَ ٱلَّذِى يَتَوَفَّىٰكُم بِٱلَّيْلِ وَيَعْلَمُ مَا جَرَحْتُم بِٱلنَّهَارِ ثُمَّ يَبْعَثُكُمْ فِيهِ لِيُقْضَىٰٓ أَجَلٌۭ مُّسَمًّۭى ۖ ثُمَّ إِلَيْهِ مَرْجِعُكُمْ ثُمَّ يُنَبِّئُكُم بِمَا كُنتُمْ تَعْمَلُونَ ﴿٦٠﴾ — اور وہی (اللہ) ہے جو رات میں (نیند کے دوران) تمھاری روحیں قبض فرما لیتا ہے اور وہ جانتا ہے جو کچھ تم دن میں کرتے ہو پھر وہ تمھیں اس (دن) میں اٹھا دیتا ہے تاکہ (تمھاری زندگی کی) مقررہ مدت پوری کر دی جائے پھر اسی کی طرف تمھیں واپس جانا ہے پھر وہ تمھیں وہ بتا دے گا جو کچھ تم کیا کرتے تھے۔ || وَهُوَ ٱلَّذِى يَتَوَفَّىٰكُم بِٱلَّيْلِ وَيَعْلَمُ مَا جَرَحْتُم بِٱلنَّهَارِ ثُمَّ يَبْعَثُكُمْ فِيهِ لِيُقْضَىٰٓ أَجَلٌۭ مُّسَمًّۭى ۖ ثُمَّ إِلَيْهِ مَرْجِعُكُمْ ثُمَّ يُنَبِّئُكُم بِمَا كُنتُمْ تَعْمَلُونَ ﴿٦٠﴾ — It is He (Allah) who takes your souls by night (in sleep) and knows what you do by day; then He raises you up in it (the day) so that the appointed term (of your life) may be completed. Then to Him is your return; then He will inform you of what you used to do. || وَهُوَ ٱلَّذِى يَتَوَفَّىٰكُم بِٱلَّيْلِ وَيَعْلَمُ مَا جَرَحْتُم بِٱلنَّهَارِ ثُمَّ يَبْعَثُكُمْ فِيهِ لِيُقْضَىٰٓ أَجَلٌۭ مُّسَمًّۭى ۖ ثُمَّ إِلَيْهِ مَرْجِعُكُمْ ثُمَّ يُنَبِّئُكُم بِمَا كُنتُمْ تَعْمَلُونَ ﴿٦٠﴾ — او همغه (الله) دی چې د شپې (د خوب پر مهال) ستاسو روحونه قبضوي، او پوهېږي په هغه څه چې تاسو یې د ورځې کوئ؛ بیا مو په هغې (ورځ) کې پاڅوي څو (ستاسو د ژوند) ټاکلې موده پوره شي؛ بیا هغه ته ستاسو ورګرځېدل دي؛ بیا به تاسو ته ووایي چې څه مو کول.
p وَهُوَ ٱلْقَاهِرُ فَوْقَ عِبَادِهِۦ ۖ وَيُرْسِلُ عَلَيْكُمْ حَفَظَةً حَتَّىٰٓ إِذَا جَآءَ أَحَدَكُمُ ٱلْمَوْتُ تَوَفَّتْهُ رُسُلُنَا وَهُمْ لَا يُفَرِّطُونَ ﴿٦١﴾ — اور وہ (اللہ) اپنے بندوں پر غالب ہے اور وہ تم پر نگہبانی کرنے والے (فرشتے) بھیجتا ہے یہاں تک کہ جب تم میں سے کسی کو موت آتی ہے تو ہمارے بھیجے ہوئے (فرشتے) اُس کی روح قبض کر لیتے ہیں اور وہ کوئی کوتاہی نہیں کرتے۔ || وَهُوَ ٱلْقَاهِرُ فَوْقَ عِبَادِهِۦ ۖ وَيُرْسِلُ عَلَيْكُمْ حَفَظَةً حَتَّىٰٓ إِذَا جَآءَ أَحَدَكُمُ ٱلْمَوْتُ تَوَفَّتْهُ رُسُلُنَا وَهُمْ لَا يُفَرِّطُونَ ﴿٦١﴾ — He (Allah) is the Irresistible over His servants, and He sends guardians (angels) over you, until, when death comes to one of you, Our messengers (angels) take his soul, and they do not fail (in their duty). || وَهُوَ ٱلْقَاهِرُ فَوْقَ عِبَادِهِۦ ۖ وَيُرْسِلُ عَلَيْكُمْ حَفَظَةً حَتَّىٰٓ إِذَا جَآءَ أَحَدَكُمُ ٱلْمَوْتُ تَوَفَّتْهُ رُسُلُنَا وَهُمْ لَا يُفَرِّطُونَ ﴿٦١﴾ — او هغه (الله) پر خپلو بندګانو غالب دی، او پر تاسو ساتونکې (پرښتې) لېږي؛ تر دې چې کله په تاسو کې چا ته مرګ راشي، زموږ لېږل شوې (پرښتې) یې روح قبضوي او هېڅ کوتاهي نه کوي.
p ثُمَّ رُدُّوٓا۟ إِلَى ٱللَّهِ مَوْلَىٰهُمُ ٱلْحَقِّ ۚ أَلَا لَهُ ٱلْحُكْمُ وَهُوَ أَسْرَعُ ٱلْحَٰسِبِينَ ﴿٦٢﴾ — پھر وہ (سب) اللہ کی طرف لوٹائے جائیں گے جو اُن کا حقیقی مالک ہے آگاہ ہو جاؤ اسی کا حکم (چلتا) ہے اور وہ بہت جلد حساب لینے والا ہے۔ || ثُمَّ رُدُّوٓا۟ إِلَى ٱللَّهِ مَوْلَىٰهُمُ ٱلْحَقِّ ۚ أَلَا لَهُ ٱلْحُكْمُ وَهُوَ أَسْرَعُ ٱلْحَٰسِبِينَ ﴿٦٢﴾ — Then they (all) will be returned to Allah, their true Master. Beware, His is the judgement, and He is the swiftest of reckoners. || ثُمَّ رُدُّوٓا۟ إِلَى ٱللَّهِ مَوْلَىٰهُمُ ٱلْحَقِّ ۚ أَلَا لَهُ ٱلْحُكْمُ وَهُوَ أَسْرَعُ ٱلْحَٰسِبِينَ ﴿٦٢﴾ — بیا به (ټول) الله ته ورګرځول شي چې د هغوی حقیقي مالک دی. خبر اوسئ، حکم یوازې د هغه (چلېږي)، او هغه ډېر ژر حساب اخیستونکی دی.
p قُلْ مَن يُنَجِّيكُم مِّن ظُلُمَٰتِ ٱلْبَرِّ وَٱلْبَحْرِ تَدْعُونَهُۥ تَضَرُّعًۭا وَخُفْيَةًۭ لَّئِنْ أَنجَىٰنَا مِنْ هَٰذِهِۦ لَنَكُونَنَّ مِنَ ٱلشَّٰكِرِينَ ﴿٦٣﴾ — آپ (صلی اللہ علیہ وآلہ وسلم) فرما دیجیے کون ہے جو تمھیں نجات دیتا ہے خشکی اور سمندر کے اندھیروں سے تم اس کو گڑگڑا کر اور چپکے چپکے پکارتے ہو (اور کہتے ہو) اگر وہ ہمیں اس سے نجات دے دے تو یقیناً ہم شکر کرنے والوں میں سے ہو جائیں گے۔ || قُلْ مَن يُنَجِّيكُم مِّن ظُلُمَٰتِ ٱلْبَرِّ وَٱلْبَحْرِ تَدْعُونَهُۥ تَضَرُّعًۭا وَخُفْيَةًۭ لَّئِنْ أَنجَىٰنَا مِنْ هَٰذِهِۦ لَنَكُونَنَّ مِنَ ٱلشَّٰكِرِينَ ﴿٦٣﴾ — Say: "Who saves you from the darknesses of land and sea, when you call on Him humbly and secretly, (saying): If He saves us from this, we shall surely be among the grateful?" || قُلْ مَن يُنَجِّيكُم مِّن ظُلُمَٰتِ ٱلْبَرِّ وَٱلْبَحْرِ تَدْعُونَهُۥ تَضَرُّعًۭا وَخُفْيَةًۭ لَّئِنْ أَنجَىٰنَا مِنْ هَٰذِهِۦ لَنَكُونَنَّ مِنَ ٱلشَّٰكِرِينَ ﴿٦٣﴾ — ووایه: "څوک دی چې تاسو د وچې او سمندر له تیارو ژغوري، چې په زارۍ او پټه یې رابولئ (او وایئ): که هغه موږ له دې وژغوره، خامخا به له شکر ګزارو شو؟"
