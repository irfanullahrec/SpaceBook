p وَلَقَدِ ٱسْتُهْزِئَ بِرُسُلٍۢ مِّن قَبْلِكَ فَحَاقَ بِٱلَّذِينَ سَخِرُوا۟ مِنْهُم مَّا كَانُوا۟ بِهِۦ يَسْتَهْزِءُونَ ﴿١٠﴾ — اور یقیناً آپ (صلی اللہ علیہ وآلہ وسلم) سے پہلے بھی رسولوں کا مذاق اڑایا گیا پھر جن لوگوں نے مذاق اڑایا تھا اسی (عذاب) نے انھیں گھیر لیا جس کا وہ مذاق اڑایا کرتے تھے۔ || وَلَقَدِ ٱسْتُهْزِئَ بِرُسُلٍۢ مِّن قَبْلِكَ فَحَاقَ بِٱلَّذِينَ سَخِرُوا۟ مِنْهُم مَّا كَانُوا۟ بِهِۦ يَسْتَهْزِءُونَ ﴿١٠﴾ — And messengers before you were certainly mocked; then those who mocked were encircled by the very (punishment) they used to mock. || وَلَقَدِ ٱسْتُهْزِئَ بِرُسُلٍۢ مِّن قَبْلِكَ فَحَاقَ بِٱلَّذِينَ سَخِرُوا۟ مِنْهُم مَّا كَانُوا۟ بِهِۦ يَسْتَهْزِءُونَ ﴿١٠﴾ — او یقیناً له تا مخکې هم پر رسولانو ټوکې شوې، بیا هغه کسان چې ټوکې یې کولې، همغه (عذاب) راګېر کړل چې ټوکې یې پرې کولې.
p قُلْ سِيرُوا۟ فِى ٱلْأَرْضِ ثُمَّ ٱنظُرُوا۟ كَيْفَ كَانَ عَٰقِبَةُ ٱلْمُكَذِّبِينَ ﴿١١﴾ — آپ (صلی اللہ علیہ وآلہ وسلم) فرما دیجیے تم زمین میں چلو پھرو پھر تم دیکھ لو جھٹلانے والوں کا کیسا انجام ہوا۔ || قُلْ سِيرُوا۟ فِى ٱلْأَرْضِ ثُمَّ ٱنظُرُوا۟ كَيْفَ كَانَ عَٰقِبَةُ ٱلْمُكَذِّبِينَ ﴿١١﴾ — Say: "Travel through the land, then see what was the end of those who denied." || قُلْ سِيرُوا۟ فِى ٱلْأَرْضِ ثُمَّ ٱنظُرُوا۟ كَيْفَ كَانَ عَٰقِبَةُ ٱلْمُكَذِّبِينَ ﴿١١﴾ — ووایه: "په ځمکه کې وګرځئ، بیا وګورئ چې د دروغ ګڼونکو انجام څنګه شو."
p قُل لِّمَن مَّا فِى ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ ۖ قُل لِّلَّهِ ۚ كَتَبَ عَلَىٰ نَفْسِهِ ٱلرَّحْمَةَ ۚ لَيَجْمَعَنَّكُمْ إِلَىٰ يَوْمِ ٱلْقِيَٰمَةِ لَا رَيْبَ فِيهِ ۚ ٱلَّذِينَ خَسِرُوٓا۟ أَنفُسَهُمْ فَهُمْ لَا يُؤْمِنُونَ ﴿١٢﴾ — آپ (صلی اللہ علیہ وآلہ وسلم) فرما دیجیے کس کا ہے جو کچھ آسمانوں اور زمین میں ہے آپ (صلی اللہ علیہ وآلہ وسلم) فرما دیجیے سب کچھ اللہ ہی کا ہے اس نے (اپنے کرم سے) رحمت کرنا اپنے اوپر لازم فرما لیا ہے وہ یقیناً قیامت کے دن تمھیں جمع فرمائے گا اس میں کوئی شک نہیں جنھوں نے اپنے آپ کو نقصان میں ڈال دیا پس وہ ایمان نہیں لائیں گے۔ || قُل لِّمَن مَّا فِى ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ ۖ قُل لِّلَّهِ ۚ كَتَبَ عَلَىٰ نَفْسِهِ ٱلرَّحْمَةَ ۚ لَيَجْمَعَنَّكُمْ إِلَىٰ يَوْمِ ٱلْقِيَٰمَةِ لَا رَيْبَ فِيهِ ۚ ٱلَّذِينَ خَسِرُوٓا۟ أَنفُسَهُمْ فَهُمْ لَا يُؤْمِنُونَ ﴿١٢﴾ — Say: "To whom belongs whatever is in the heavens and the earth?" Say: "Everything belongs to Allah." He has (of His grace) made mercy binding on Himself. He will surely gather you on the Day of Resurrection, about which there is no doubt. Those who have put themselves into loss will not believe. || قُل لِّمَن مَّا فِى ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ ۖ قُل لِّلَّهِ ۚ كَتَبَ عَلَىٰ نَفْسِهِ ٱلرَّحْمَةَ ۚ لَيَجْمَعَنَّكُمْ إِلَىٰ يَوْمِ ٱلْقِيَٰمَةِ لَا رَيْبَ فِيهِ ۚ ٱلَّذِينَ خَسِرُوٓا۟ أَنفُسَهُمْ فَهُمْ لَا يُؤْمِنُونَ ﴿١٢﴾ — ووایه: "څه چې په اسمانونو او ځمکه کې دي، د چا دي؟" ووایه: "هر څه د الله دي." هغه (په خپل کرم) رحمت پر ځان لازم کړی. هغه به خامخا د قیامت په ورځ تاسو راټول کړي چې هېڅ شک پکې نشته. هغه کسان چې ځانونه یې په تاوان کې اچولي، ایمان به رانه وړي.
p ۞ وَلَهُۥ مَا سَكَنَ فِى ٱلَّيْلِ وَٱلنَّهَارِ ۚ وَهُوَ ٱلسَّمِيعُ ٱلْعَلِيمُ ﴿١٣﴾ — اور رات اور دن میں جو (مخلوق) بستی ہے سب اسی کی ہے اور وہ خوب سننے والا خوب جاننے والا ہے۔ || ۞ وَلَهُۥ مَا سَكَنَ فِى ٱلَّيْلِ وَٱلنَّهَارِ ۚ وَهُوَ ٱلسَّمِيعُ ٱلْعَلِيمُ ﴿١٣﴾ — And to Him belongs whatever dwells in the night and the day, and He is the All-Hearing, the All-Knowing. || ۞ وَلَهُۥ مَا سَكَنَ فِى ٱلَّيْلِ وَٱلنَّهَارِ ۚ وَهُوَ ٱلسَّمِيعُ ٱلْعَلِيمُ ﴿١٣﴾ — او څه چې په شپه او ورځ کې اوسي، ټول د هغه دي، او هغه ښه اورېدونکی، ښه پوه دی.
p قُلْ أَغَيْرَ ٱللَّهِ أَتَّخِذُ وَلِيًّۭا فَاطِرِ ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ وَهُوَ يُطْعِمُ وَلَا يُطْعَمُ ۗ قُلْ إِنِّىٓ أُمِرْتُ أَنْ أَكُونَ أَوَّلَ مَنْ أَسْلَمَ ۖ وَلَا تَكُونَنَّ مِنَ ٱلْمُشْرِكِينَ ﴿١٤﴾ — آپ (صلی اللہ علیہ وآلہ وسلم) فرما دیجیے کیا میں اللہ کو چھوڑ کر کسی اور کو کارساز (معبود) بناؤں (وہی) آسمانوں اور زمینوں کا پیدا فرمانے والا ہے اور وہ (سب کو) کھلاتا ہے اور اسے کھلایا نہیں جاتا آپ (صلی اللہ علیہ وآلہ وسلم) فرما دیجیے مجھے تو یہی حکم دیا گیا ہے کہ میں سب سے پہلے فرماں بردار ہو جاؤں اور (یہ حکم بھی دیا گیا ہے کہ) تم مشرکوں میں سے ہرگز نہ ہو جانا۔ || قُلْ أَغَيْرَ ٱللَّهِ أَتَّخِذُ وَلِيًّۭا فَاطِرِ ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ وَهُوَ يُطْعِمُ وَلَا يُطْعَمُ ۗ قُلْ إِنِّىٓ أُمِرْتُ أَنْ أَكُونَ أَوَّلَ مَنْ أَسْلَمَ ۖ وَلَا تَكُونَنَّ مِنَ ٱلْمُشْرِكِينَ ﴿١٤﴾ — Say: "Shall I take as protector (and god) anyone other than Allah, the Creator of the heavens and the earth, who feeds (all) and is not fed?" Say: "I have been commanded to be the first to submit, and (told): 'Never be one of the polytheists.'" || قُلْ أَغَيْرَ ٱللَّهِ أَتَّخِذُ وَلِيًّۭا فَاطِرِ ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ وَهُوَ يُطْعِمُ وَلَا يُطْعَمُ ۗ قُلْ إِنِّىٓ أُمِرْتُ أَنْ أَكُونَ أَوَّلَ مَنْ أَسْلَمَ ۖ وَلَا تَكُونَنَّ مِنَ ٱلْمُشْرِكِينَ ﴿١٤﴾ — ووایه: "آیا زه له الله پرته بل څوک کارساز (معبود) ونیسم، (هغه) چې د اسمانونو او ځمکو پیدا کوونکی دی، او (ټولو ته) خواړه ورکوي او هغه ته خواړه نه ورکول کېږي؟" ووایه: "ما ته همدا امر شوی چې تر ټولو لومړی فرمانبردار شم، او (دا امر هم شوی چې) هېڅکله له مشرکانو مه کېږه."
p قُلْ إِنِّىٓ أَخَافُ إِنْ عَصَيْتُ رَبِّى عَذَابَ يَوْمٍ عَظِيمٍۢ ﴿١٥﴾ — آپ (صلی اللہ علیہ وآلہ وسلم) فرما دیجیے اگر میں اپنے رب کی نافرمانی کروں تو بے شک میں ڈرتا ہوں ایک بڑے دن کے عذاب سے۔ || قُلْ إِنِّىٓ أَخَافُ إِنْ عَصَيْتُ رَبِّى عَذَابَ يَوْمٍ عَظِيمٍۢ ﴿١٥﴾ — Say: "Indeed I fear, if I disobey my Lord, the punishment of a mighty Day." || قُلْ إِنِّىٓ أَخَافُ إِنْ عَصَيْتُ رَبِّى عَذَابَ يَوْمٍ عَظِيمٍۢ ﴿١٥﴾ — ووایه: "که زه د خپل رب نافرماني وکړم، نو بې شکه د یوې لویې ورځې له عذابه وېرېږم."
p مَّن يُصْرَفْ عَنْهُ يَوْمَئِذٍۢ فَقَدْ رَحِمَهُۥ ۚ وَذَٰلِكَ ٱلْفَوْزُ ٱلْمُبِينُ ﴿١٦﴾ — اس دن جس شخص سے (عذاب) ہٹا دیا گیا تو یقیناً اس پر اس (اللہ) نے رحم فرمایا اور یہی کھلی کامیابی ہے۔ || مَّن يُصْرَفْ عَنْهُ يَوْمَئِذٍۢ فَقَدْ رَحِمَهُۥ ۚ وَذَٰلِكَ ٱلْفَوْزُ ٱلْمُبِينُ ﴿١٦﴾ — Whoever has (the punishment) averted from him on that Day, He (Allah) has surely had mercy on him; and that is the clear success. || مَّن يُصْرَفْ عَنْهُ يَوْمَئِذٍۢ فَقَدْ رَحِمَهُۥ ۚ وَذَٰلِكَ ٱلْفَوْزُ ٱلْمُبِينُ ﴿١٦﴾ — په هغه ورځ چې له چا (عذاب) لرې شي، نو یقیناً هغه (الله) پرې رحم کړی، او همدا ښکاره بریا ده.
p وَإِن يَمْسَسْكَ ٱللَّهُ بِضُرٍّۢ فَلَا كَاشِفَ لَهُۥٓ إِلَّا هُوَ ۖ وَإِن يَمْسَسْكَ بِخَيْرٍۢ فَهُوَ عَلَىٰ كُلِّ شَىْءٍۢ قَدِيرٌۭ ﴿١٧﴾ — اور اگر اللہ تمھیں کوئی تکلیف پہنچائے تو اُس کے سوا اس (تکلیف) کو دور کرنے والا کوئی نہیں اور اگر (اللہ) تمھیں کوئی بھلائی پہنچائے تو وہ ہر چیز پر بڑی قدرت رکھنے والا ہے۔ || وَإِن يَمْسَسْكَ ٱللَّهُ بِضُرٍّۢ فَلَا كَاشِفَ لَهُۥٓ إِلَّا هُوَ ۖ وَإِن يَمْسَسْكَ بِخَيْرٍۢ فَهُوَ عَلَىٰ كُلِّ شَىْءٍۢ قَدِيرٌۭ ﴿١٧﴾ — And if Allah touches you with harm, none can remove it but He; and if (Allah) touches you with good, then He has power over all things. || وَإِن يَمْسَسْكَ ٱللَّهُ بِضُرٍّۢ فَلَا كَاشِفَ لَهُۥٓ إِلَّا هُوَ ۖ وَإِن يَمْسَسْكَ بِخَيْرٍۢ فَهُوَ عَلَىٰ كُلِّ شَىْءٍۢ قَدِيرٌۭ ﴿١٧﴾ — او که الله تاسو ته کوم تکلیف ورسوي، له هغه پرته یې لرې کوونکی نشته؛ او که (الله) تاسو ته کومه ښېګڼه ورسوي، نو هغه پر هر شي ډېر قادر دی.
p وَهُوَ ٱلْقَاهِرُ فَوْقَ عِبَادِهِۦ ۚ وَهُوَ ٱلْحَكِيمُ ٱلْخَبِيرُ ﴿١٨﴾ — اور وہی اپنے بندوں پر غالب ہے اور وہ بڑی حکمت والا بڑا باخبر ہے۔ || وَهُوَ ٱلْقَاهِرُ فَوْقَ عِبَادِهِۦ ۚ وَهُوَ ٱلْحَكِيمُ ٱلْخَبِيرُ ﴿١٨﴾ — And He is the Irresistible, above His servants, and He is the All-Wise, the All-Aware. || وَهُوَ ٱلْقَاهِرُ فَوْقَ عِبَادِهِۦ ۚ وَهُوَ ٱلْحَكِيمُ ٱلْخَبِيرُ ﴿١٨﴾ — او همغه پر خپلو بندګانو غالب دی، او هغه ډېر حکمت لرونکی، ډېر خبر دی.

Do you really testify that there are other gods besides Allah?" Say: "I do not testify (to that falsehood)."
