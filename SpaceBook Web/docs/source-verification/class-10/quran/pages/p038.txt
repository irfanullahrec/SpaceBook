p وَعَلَى ٱلَّذِينَ هَادُوا۟ حَرَّمْنَا كُلَّ ذِى ظُفُرٍۢ ۖ وَمِنَ ٱلْبَقَرِ وَٱلْغَنَمِ حَرَّمْنَا عَلَيْهِمْ شُحُومَهُمَآ إِلَّا مَا حَمَلَتْ ظُهُورُهُمَآ أَوِ ٱلْحَوَايَآ أَوْ مَا ٱخْتَلَطَ بِعَظْمٍۢ ۚ ذَٰلِكَ جَزَيْنَٰهُم بِبَغْيِهِمْ ۖ وَإِنَّا لَصَٰدِقُونَ ﴿١٤٦﴾ — اور اُن لوگوں پر جو یہودی ہوئے ہم نے ہر ناخن والا جانور حرام کر دیا تھا اور گائے اور بکری میں سے ہم نے ان پر دونوں کی چربی حرام کر دی تھی سوائے اس (چربی) کے جو ان دونوں کی پیٹھ میں ہو یا آنتوں میں لگی ہو یا جو ہڈی کے ساتھ ملی ہو یہ سزا ہم نے انھیں اُن کی سرکشی کی وجہ سے دی اور بے شک ہم ضرور سچے ہیں۔ || وَعَلَى ٱلَّذِينَ هَادُوا۟ حَرَّمْنَا كُلَّ ذِى ظُفُرٍۢ ۖ وَمِنَ ٱلْبَقَرِ وَٱلْغَنَمِ حَرَّمْنَا عَلَيْهِمْ شُحُومَهُمَآ إِلَّا مَا حَمَلَتْ ظُهُورُهُمَآ أَوِ ٱلْحَوَايَآ أَوْ مَا ٱخْتَلَطَ بِعَظْمٍۢ ۚ ذَٰلِكَ جَزَيْنَٰهُم بِبَغْيِهِمْ ۖ وَإِنَّا لَصَٰدِقُونَ ﴿١٤٦﴾ — And to those who were Jews We forbade every animal with claws (undivided hoof), and of cattle and sheep We forbade them their fat, except what is on their backs or in their entrails or mixed with bone. That was the penalty We gave them for their rebellion; and We are surely truthful. || وَعَلَى ٱلَّذِينَ هَادُوا۟ حَرَّمْنَا كُلَّ ذِى ظُفُرٍۢ ۖ وَمِنَ ٱلْبَقَرِ وَٱلْغَنَمِ حَرَّمْنَا عَلَيْهِمْ شُحُومَهُمَآ إِلَّا مَا حَمَلَتْ ظُهُورُهُمَآ أَوِ ٱلْحَوَايَآ أَوْ مَا ٱخْتَلَطَ بِعَظْمٍۢ ۚ ذَٰلِكَ جَزَيْنَٰهُم بِبَغْيِهِمْ ۖ وَإِنَّا لَصَٰدِقُونَ ﴿١٤٦﴾ — او پر هغو خلکو چې یهودان شول، موږ هر نوکي لرونکی څاروی حرام کړی و، او له غوا او وزې یې پرې د دواړو غوړي حرام کړي وو، پرته له هغو (غوړو) چې د دواړو په شا وي، یا په کولمو پورې نښتي وي، یا له هډوکي سره ګډ وي. دا سزا مو د هغوی د سرکشۍ له امله ورکړه، او بې شکه موږ خامخا رښتیني یو.
p فَإِن كَذَّبُوكَ فَقُل رَّبُّكُمْ ذُو رَحْمَةٍۢ وَٰسِعَةٍۢ وَلَا يُرَدُّ بَأْسُهُۥ عَنِ ٱلْقَوْمِ ٱلْمُجْرِمِينَ ﴿١٤٧﴾ — لہٰذا اگر وہ آپ (صلی اللہ علیہ وآلہ وسلم) کو جھٹلائیں تو آپ فرما دیجیے کہ تمھارا رب وسیع رحمت والا ہے اور مجرم قوم سے اُس کا عذاب نہیں ٹالا جا سکتا۔ || فَإِن كَذَّبُوكَ فَقُل رَّبُّكُمْ ذُو رَحْمَةٍۢ وَٰسِعَةٍۢ وَلَا يُرَدُّ بَأْسُهُۥ عَنِ ٱلْقَوْمِ ٱلْمُجْرِمِينَ ﴿١٤٧﴾ — So if they deny you, say: "Your Lord is the Possessor of vast mercy, and His punishment cannot be turned back from the criminal people." || فَإِن كَذَّبُوكَ فَقُل رَّبُّكُمْ ذُو رَحْمَةٍۢ وَٰسِعَةٍۢ وَلَا يُرَدُّ بَأْسُهُۥ عَنِ ٱلْقَوْمِ ٱلْمُجْرِمِينَ ﴿١٤٧﴾ — نو که تا دروغجن وبولي، ووایه: "ستاسو رب د پراخ رحمت خاوند دی، او د هغه عذاب له مجرم قوم نه شي اړول کېدای."
p سَيَقُولُ ٱلَّذِينَ أَشْرَكُوا۟ لَوْ شَآءَ ٱللَّهُ مَآ أَشْرَكْنَا وَلَآ ءَابَآؤُنَا وَلَا حَرَّمْنَا مِن شَىْءٍۢ ۚ كَذَٰلِكَ كَذَّبَ ٱلَّذِينَ مِن قَبْلِهِمْ حَتَّىٰ ذَاقُوا۟ بَأْسَنَا ۗ قُلْ هَلْ عِندَكُم مِّنْ عِلْمٍۢ فَتُخْرِجُوهُ لَنَآ ۖ إِن تَتَّبِعُونَ إِلَّا ٱلظَّنَّ وَإِنْ أَنتُمْ إِلَّا تَخْرُصُونَ ﴿١٤٨﴾ — عنقریب وہ کہیں گے جنھوں نے شرک کیا کہ اگر اللہ چاہتا تو نہ ہم شرک کرتے اور نہ ہمارے باپ دادا اور نہ ہم کسی چیز کو حرام ٹھہراتے اسی طرح ان لوگوں نے جھٹلایا تھا جو ان سے پہلے تھے یہاں تک کہ انھوں نے ہمارا عذاب چکھا آپ (صلی اللہ علیہ وآلہ وسلم) فرما دیجیے کیا تمھارے پاس کوئی علم ہے تو تم اسے ہمارے سامنے پیش کرو تم تو صرف گمان کی پیروی کر رہے ہو اور تم تو صرف (جھوٹے) اندازوں سے باتیں کرتے ہو۔ || سَيَقُولُ ٱلَّذِينَ أَشْرَكُوا۟ لَوْ شَآءَ ٱللَّهُ مَآ أَشْرَكْنَا وَلَآ ءَابَآؤُنَا وَلَا حَرَّمْنَا مِن شَىْءٍۢ ۚ كَذَٰلِكَ كَذَّبَ ٱلَّذِينَ مِن قَبْلِهِمْ حَتَّىٰ ذَاقُوا۟ بَأْسَنَا ۗ قُلْ هَلْ عِندَكُم مِّنْ عِلْمٍۢ فَتُخْرِجُوهُ لَنَآ ۖ إِن تَتَّبِعُونَ إِلَّا ٱلظَّنَّ وَإِنْ أَنتُمْ إِلَّا تَخْرُصُونَ ﴿١٤٨﴾ — Those who associate partners will say: "Had Allah willed, neither we nor our forefathers would have associated (partners), nor would we have forbidden anything." Thus did those before them deny, until they tasted Our punishment. Say: "Have you any knowledge? Then produce it for us. You follow nothing but conjecture, and you only speak by (false) guesswork." || سَيَقُولُ ٱلَّذِينَ أَشْرَكُوا۟ لَوْ شَآءَ ٱللَّهُ مَآ أَشْرَكْنَا وَلَآ ءَابَآؤُنَا وَلَا حَرَّمْنَا مِن شَىْءٍۢ ۚ كَذَٰلِكَ كَذَّبَ ٱلَّذِينَ مِن قَبْلِهِمْ حَتَّىٰ ذَاقُوا۟ بَأْسَنَا ۗ قُلْ هَلْ عِندَكُم مِّنْ عِلْمٍۢ فَتُخْرِجُوهُ لَنَآ ۖ إِن تَتَّبِعُونَ إِلَّا ٱلظَّنَّ وَإِنْ أَنتُمْ إِلَّا تَخْرُصُونَ ﴿١٤٨﴾ — ژر به هغه کسان چې شرک یې کړی ووایي: "که الله غوښتي وای، نه به موږ شرک کړی وای او نه زموږ پلرونو، او نه به مو کوم شی حرام ګرځولی وای." همداسې هغو خلکو دروغ ګڼلي وو چې له دوی مخکې وو، تر دې چې زموږ عذاب یې وڅکه. ووایه: "آیا تاسو سره کوم علم شته؟ نو زموږ مخې ته یې راوړئ. تاسو یوازې د ګومان پیروي کوئ او یوازې په (درواغجنو) اټکلونو خبرې کوئ."
p قُلْ فَلِلَّهِ ٱلْحُجَّةُ ٱلْبَٰلِغَةُ ۖ فَلَوْ شَآءَ لَهَدَىٰكُمْ أَجْمَعِينَ ﴿١٤٩﴾ — آپ (صلی اللہ علیہ وآلہ وسلم) فرما دیجیے بس اللہ ہی کے لیے ہے کامل دلیل لہٰذا اگر اللہ چاہتا تو تم سب کو ہدایت عطا فرما دیتا۔ || قُلْ فَلِلَّهِ ٱلْحُجَّةُ ٱلْبَٰلِغَةُ ۖ فَلَوْ شَآءَ لَهَدَىٰكُمْ أَجْمَعِينَ ﴿١٤٩﴾ — Say: "To Allah belongs the conclusive argument; had He willed, He would have guided you all." || قُلْ فَلِلَّهِ ٱلْحُجَّةُ ٱلْبَٰلِغَةُ ۖ فَلَوْ شَآءَ لَهَدَىٰكُمْ أَجْمَعِينَ ﴿١٤٩﴾ — ووایه: "بشپړ دلیل یوازې د الله دی؛ نو که الله غوښتي وای، تاسو ټولو ته به یې هدایت کړی وای."
p قُلْ هَلُمَّ شُهَدَآءَكُمُ ٱلَّذِينَ يَشْهَدُونَ أَنَّ ٱللَّهَ حَرَّمَ هَٰذَا ۖ فَإِن شَهِدُوا۟ فَلَا تَشْهَدْ مَعَهُمْ ۚ وَلَا تَتَّبِعْ أَهْوَآءَ ٱلَّذِينَ كَذَّبُوا۟ بِـَٔايَٰتِنَا وَٱلَّذِينَ لَا يُؤْمِنُونَ بِٱلْءَاخِرَةِ وَهُم بِرَبِّهِمْ يَعْدِلُونَ ﴿١٥٠﴾ — آپ (صلی اللہ علیہ وآلہ وسلم) فرما دیجیے لاؤ اپنے گواہوں کو جو گواہی دیں کہ اللہ نے ان (چیزوں) کو حرام کیا ہے پھر اگر وہ (جھوٹی) گواہی دے بھی دیں تو آپ اُن کے ساتھ گواہی نہ دیجیے اور اُن کی خواہشات کی پیروی نہ کیجیے جنھوں نے ہماری آیات کو جھٹلایا اور جو لوگ آخرت پر ایمان نہیں رکھتے اور وہ (دوسروں کو) اپنے رب کے برابر ٹھہراتے ہیں۔ || قُلْ هَلُمَّ شُهَدَآءَكُمُ ٱلَّذِينَ يَشْهَدُونَ أَنَّ ٱللَّهَ حَرَّمَ هَٰذَا ۖ فَإِن شَهِدُوا۟ فَلَا تَشْهَدْ مَعَهُمْ ۚ وَلَا تَتَّبِعْ أَهْوَآءَ ٱلَّذِينَ كَذَّبُوا۟ بِـَٔايَٰتِنَا وَٱلَّذِينَ لَا يُؤْمِنُونَ بِٱلْءَاخِرَةِ وَهُم بِرَبِّهِمْ يَعْدِلُونَ ﴿١٥٠﴾ — Say: "Bring your witnesses who testify that Allah forbade these (things)." If they (falsely) testify, do not testify with them, and do not follow the desires of those who deny Our verses and do not believe in the Hereafter and set up (others as) equals to their Lord. || قُلْ هَلُمَّ شُهَدَآءَكُمُ ٱلَّذِينَ يَشْهَدُونَ أَنَّ ٱللَّهَ حَرَّمَ هَٰذَا ۖ فَإِن شَهِدُوا۟ فَلَا تَشْهَدْ مَعَهُمْ ۚ وَلَا تَتَّبِعْ أَهْوَآءَ ٱلَّذِينَ كَذَّبُوا۟ بِـَٔايَٰتِنَا وَٱلَّذِينَ لَا يُؤْمِنُونَ بِٱلْءَاخِرَةِ وَهُم بِرَبِّهِمْ يَعْدِلُونَ ﴿١٥٠﴾ — ووایه: "خپل شاهدان راوړئ چې شاهدي ورکړي چې الله دا (شیان) حرام کړي." بیا که (درواغجنه) شاهدي ورکړي هم، ته ورسره شاهدي مه ورکوه، او د هغو د غوښتنو پیروي مه کوه چې زموږ آیتونه یې دروغ ګڼلي او پر آخرت ایمان نه لري او (نور) له خپل رب سره برابروي.
