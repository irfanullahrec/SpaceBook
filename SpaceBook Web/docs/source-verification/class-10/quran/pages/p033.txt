p ذَٰلِكَ أَن لَّمْ يَكُن رَّبُّكَ مُهْلِكَ ٱلْقُرَىٰ بِظُلْمٍۢ وَأَهْلُهَا غَٰفِلُونَ ﴿١٣١﴾ — یہ اس لیے کہ بے شک آپ کا رب کبھی بستیوں کو ظلم کی وجہ سے ہلاک نہیں کرتا جب کہ اس کے رہنے والے بے خبر ہوں۔ || ذَٰلِكَ أَن لَّمْ يَكُن رَّبُّكَ مُهْلِكَ ٱلْقُرَىٰ بِظُلْمٍۢ وَأَهْلُهَا غَٰفِلُونَ ﴿١٣١﴾ — That is because your Lord would never destroy towns for wrongdoing while their people were unaware. || ذَٰلِكَ أَن لَّمْ يَكُن رَّبُّكَ مُهْلِكَ ٱلْقُرَىٰ بِظُلْمٍۢ وَأَهْلُهَا غَٰفِلُونَ ﴿١٣١﴾ — دا ځکه چې بې شکه ستا رب هېڅکله کلي د ظلم له امله نه هلاکوي په داسې حال کې چې اوسېدونکي یې بې خبره وي.
p وَلِكُلٍّۢ دَرَجَٰتٌۭ مِّمَّا عَمِلُوا۟ ۚ وَمَا رَبُّكَ بِغَٰفِلٍ عَمَّا يَعْمَلُونَ ﴿١٣٢﴾ — اور ہر ایک کے لیے (مختلف) درجے ہیں ان (اعمال) کی وجہ سے جو انھوں نے کیے اور آپ کا رب بے خبر نہیں ان کاموں سے جو وہ کرتے ہیں۔ || وَلِكُلٍّۢ دَرَجَٰتٌۭ مِّمَّا عَمِلُوا۟ ۚ وَمَا رَبُّكَ بِغَٰفِلٍ عَمَّا يَعْمَلُونَ ﴿١٣٢﴾ — For all there are (different) ranks according to what (deeds) they did; and your Lord is not unaware of what they do. || وَلِكُلٍّۢ دَرَجَٰتٌۭ مِّمَّا عَمِلُوا۟ ۚ وَمَا رَبُّكَ بِغَٰفِلٍ عَمَّا يَعْمَلُونَ ﴿١٣٢﴾ — او د هر یوه لپاره د هغو (عملونو) له امله (بېلابېلې) درجې دي چې کړي یې دي؛ او ستا رب د هغو کارونو څخه بې خبره نه دی چې دوی یې کوي.
p وَرَبُّكَ ٱلْغَنِىُّ ذُو ٱلرَّحْمَةِ ۚ إِن يَشَأْ يُذْهِبْكُمْ وَيَسْتَخْلِفْ مِنۢ بَعْدِكُم مَّا يَشَآءُ كَمَآ أَنشَأَكُم مِّن ذُرِّيَّةِ قَوْمٍ ءَاخَرِينَ ﴿١٣٣﴾ — اور تمھارا رب بے نیاز نہایت ہی رحمت والا ہے اگر وہ چاہے تو تمھیں لے جائے اور تمھارے بعد جس کو چاہے تمھاری جگہ آباد فرما دے جیسا کہ اس نے تمھیں دوسری قوم کی نسل سے پیدا فرمایا۔ || وَرَبُّكَ ٱلْغَنِىُّ ذُو ٱلرَّحْمَةِ ۚ إِن يَشَأْ يُذْهِبْكُمْ وَيَسْتَخْلِفْ مِنۢ بَعْدِكُم مَّا يَشَآءُ كَمَآ أَنشَأَكُم مِّن ذُرِّيَّةِ قَوْمٍ ءَاخَرِينَ ﴿١٣٣﴾ — Your Lord is the Self-Sufficient, full of mercy. If He wills, He can take you away and settle in your place whom He wills after you, just as He produced you from the offspring of another people. || وَرَبُّكَ ٱلْغَنِىُّ ذُو ٱلرَّحْمَةِ ۚ إِن يَشَأْ يُذْهِبْكُمْ وَيَسْتَخْلِفْ مِنۢ بَعْدِكُم مَّا يَشَآءُ كَمَآ أَنشَأَكُم مِّن ذُرِّيَّةِ قَوْمٍ ءَاخَرِينَ ﴿١٣٣﴾ — او ستاسو رب بې نیازه، ډېر رحمت لرونکی دی؛ که وغواړي، تاسو یوسي او له تاسو وروسته چې څوک وغواړي ستاسو پر ځای یې مېشت کړي، لکه څنګه چې یې تاسو د بل قوم له نسله پیدا کړئ.
p إِنَّ مَا تُوعَدُونَ لَءَاتٍۢ ۖ وَمَآ أَنتُم بِمُعْجِزِينَ ﴿١٣٤﴾ — بے شک جس بات کا تم سے وعدہ کیا جاتا ہے وہ یقیناً آنے والی ہے اور تم (اللہ کو) عاجز کرنے والے نہیں ہو۔ || إِنَّ مَا تُوعَدُونَ لَءَاتٍۢ ۖ وَمَآ أَنتُم بِمُعْجِزِينَ ﴿١٣٤﴾ — What you are promised will surely come, and you cannot frustrate (Allah). || إِنَّ مَا تُوعَدُونَ لَءَاتٍۢ ۖ وَمَآ أَنتُم بِمُعْجِزِينَ ﴿١٣٤﴾ — بې شکه هغه چې له تاسو سره یې ژمنه کېږي، خامخا راتلونکې ده، او تاسو (الله) عاجز کوونکي نه یاست.
p قُلْ يَٰقَوْمِ ٱعْمَلُوا۟ عَلَىٰ مَكَانَتِكُمْ إِنِّى عَامِلٌۭ ۖ فَسَوْفَ تَعْلَمُونَ مَن تَكُونُ لَهُۥ عَٰقِبَةُ ٱلدَّارِ ۗ إِنَّهُۥ لَا يُفْلِحُ ٱلظَّٰلِمُونَ ﴿١٣٥﴾ — آپ (صلی اللہ علیہ وآلہ وسلم) فرما دیجیے اے میری قوم! تم اپنی جگہ پر عمل کرو بے شک میں (بھی) عمل کرنے والا ہوں تو عنقریب تم جان لو گے کہ آخرت کا انجام کس کے لیے (بہتر) ہے بے شک ظالم لوگ فلاح نہیں پائیں گے۔ || قُلْ يَٰقَوْمِ ٱعْمَلُوا۟ عَلَىٰ مَكَانَتِكُمْ إِنِّى عَامِلٌۭ ۖ فَسَوْفَ تَعْلَمُونَ مَن تَكُونُ لَهُۥ عَٰقِبَةُ ٱلدَّارِ ۗ إِنَّهُۥ لَا يُفْلِحُ ٱلظَّٰلِمُونَ ﴿١٣٥﴾ — Say: "O my people, work according to your position; I (too) am working. You will soon know whose will be the (good) end of the Hereafter. Surely the wrongdoers will not succeed." || قُلْ يَٰقَوْمِ ٱعْمَلُوا۟ عَلَىٰ مَكَانَتِكُمْ إِنِّى عَامِلٌۭ ۖ فَسَوْفَ تَعْلَمُونَ مَن تَكُونُ لَهُۥ عَٰقِبَةُ ٱلدَّارِ ۗ إِنَّهُۥ لَا يُفْلِحُ ٱلظَّٰلِمُونَ ﴿١٣٥﴾ — ووایه: "ای زما قومه! تاسو په خپل ځای عمل وکړئ، بې شکه زه (هم) عمل کوونکی یم؛ نو ژر به پوه شئ چې د آخرت انجام د چا لپاره (غوره) دی؛ بې شکه ظالمان به بریالي نه شي."
p وَجَعَلُوا۟ لِلَّهِ مِمَّا ذَرَأَ مِنَ ٱلْحَرْثِ وَٱلْأَنْعَٰمِ نَصِيبًۭا فَقَالُوا۟ هَٰذَا لِلَّهِ بِزَعْمِهِمْ وَهَٰذَا لِشُرَكَآئِنَا ۖ فَمَا كَانَ لِشُرَكَآئِهِمْ فَلَا يَصِلُ إِلَى ٱللَّهِ ۖ وَمَا كَانَ لِلَّهِ فَهُوَ يَصِلُ إِلَىٰ شُرَكَآئِهِمْ ۗ سَآءَ مَا يَحْكُمُونَ ﴿١٣٦﴾ — اور انھوں نے اللہ کے لیے مقرر کیا ہے ان کھیتی اور چوپایوں میں سے ایک حصہ جو اس نے پیدا فرمائے ہیں پس وہ اپنے خیال سے کہتے ہیں کہ یہ (حصہ) تو اللہ کا ہے اور یہ ہمارے شریکوں کا تو جو (حصہ) ان کے شریکوں کا ہے وہ تو اللہ کی طرف نہیں پہنچتا اور جو (حصہ) اللہ کا ہے وہ ان کے شریکوں کو پہنچ جاتا ہے برا ہے جو وہ فیصلہ کرتے ہیں۔ || وَجَعَلُوا۟ لِلَّهِ مِمَّا ذَرَأَ مِنَ ٱلْحَرْثِ وَٱلْأَنْعَٰمِ نَصِيبًۭا فَقَالُوا۟ هَٰذَا لِلَّهِ بِزَعْمِهِمْ وَهَٰذَا لِشُرَكَآئِنَا ۖ فَمَا كَانَ لِشُرَكَآئِهِمْ فَلَا يَصِلُ إِلَى ٱللَّهِ ۖ وَمَا كَانَ لِلَّهِ فَهُوَ يَصِلُ إِلَىٰ شُرَكَآئِهِمْ ۗ سَآءَ مَا يَحْكُمُونَ ﴿١٣٦﴾ — And they assign to Allah a share of the crops and cattle He has produced, and say, by their claim: "This is for Allah, and this is for our partners." But what is for their partners does not reach Allah, while what is for Allah reaches their partners. Evil is how they judge. || وَجَعَلُوا۟ لِلَّهِ مِمَّا ذَرَأَ مِنَ ٱلْحَرْثِ وَٱلْأَنْعَٰمِ نَصِيبًۭا فَقَالُوا۟ هَٰذَا لِلَّهِ بِزَعْمِهِمْ وَهَٰذَا لِشُرَكَآئِنَا ۖ فَمَا كَانَ لِشُرَكَآئِهِمْ فَلَا يَصِلُ إِلَى ٱللَّهِ ۖ وَمَا كَانَ لِلَّهِ فَهُوَ يَصِلُ إِلَىٰ شُرَكَآئِهِمْ ۗ سَآءَ مَا يَحْكُمُونَ ﴿١٣٦﴾ — او هغوی د الله لپاره د هغو کښتونو او څارویو یوه برخه ټاکلې چې هغه پیدا کړي، او په خپل ګومان وايي: "دا (برخه) د الله ده او دا زموږ د شریکانو." نو هغه (برخه) چې د هغوی د شریکانو ده، الله ته نه رسېږي، او هغه چې د الله ده، د هغوی شریکانو ته رسېږي. بده ده هغه پرېکړه چې کوي یې.
p وَكَذَٰلِكَ زَيَّنَ لِكَثِيرٍۢ مِّنَ ٱلْمُشْرِكِينَ قَتْلَ أَوْلَٰدِهِمْ شُرَكَآؤُهُمْ لِيُرْدُوهُمْ وَلِيَلْبِسُوا۟ عَلَيْهِمْ دِينَهُمْ ۖ وَلَوْ شَآءَ ٱللَّهُ مَا فَعَلُوهُ ۖ فَذَرْهُمْ وَمَا يَفْتَرُونَ ﴿١٣٧﴾ — اور اسی طرح بہت سے مشرکوں کے لیے ان کے شریکوں نے ان کی اولاد کا قتل خوش نما بنا دیا ہے تاکہ انھیں ہلاکت میں ڈالیں اور ان پر ان کا دین مشتبہ کریں اور اگر اللہ چاہتا تو وہ ایسا نہ کرتے تو آپ ان کو چھوڑ دیجیے اور (اسے) جو وہ جھوٹ گھڑا کرتے ہیں۔ || وَكَذَٰلِكَ زَيَّنَ لِكَثِيرٍۢ مِّنَ ٱلْمُشْرِكِينَ قَتْلَ أَوْلَٰدِهِمْ شُرَكَآؤُهُمْ لِيُرْدُوهُمْ وَلِيَلْبِسُوا۟ عَلَيْهِمْ دِينَهُمْ ۖ وَلَوْ شَآءَ ٱللَّهُ مَا فَعَلُوهُ ۖ فَذَرْهُمْ وَمَا يَفْتَرُونَ ﴿١٣٧﴾ — And thus their partners have made the killing of their children seem fair to many of the polytheists, to ruin them and confuse their religion for them. Had Allah willed, they would not have done it; so leave them and what they invent. || وَكَذَٰلِكَ زَيَّنَ لِكَثِيرٍۢ مِّنَ ٱلْمُشْرِكِينَ قَتْلَ أَوْلَٰدِهِمْ شُرَكَآؤُهُمْ لِيُرْدُوهُمْ وَلِيَلْبِسُوا۟ عَلَيْهِمْ دِينَهُمْ ۖ وَلَوْ شَآءَ ٱللَّهُ مَا فَعَلُوهُ ۖ فَذَرْهُمْ وَمَا يَفْتَرُونَ ﴿١٣٧﴾ — او همداسې ډېرو مشرکانو ته د هغوی شریکانو د خپل اولاد وژل ښکلي ښودلي، څو یې هلاکت ته واچوي او دین یې پرې ګډوډ کړي؛ او که الله غوښتي وای، داسې به یې نه وای کړي؛ نو هغوی او هغه دروغ پرېږده چې جوړوي یې.
p وَقَالُوا۟ هَٰذِهِۦٓ أَنْعَٰمٌۭ وَحَرْثٌ حِجْرٌۭ لَّا يَطْعَمُهَآ إِلَّا مَن نَّشَآءُ بِزَعْمِهِمْ وَأَنْعَٰمٌ حُرِّمَتْ ظُهُورُهَا وَأَنْعَٰمٌۭ لَّا يَذْكُرُونَ ٱسْمَ ٱللَّهِ عَلَيْهَا ٱفْتِرَآءً عَلَيْهِ ۚ سَيَجْزِيهِم بِمَا كَانُوا۟ يَفْتَرُونَ ﴿١٣٨﴾ — اور وہ کہتے ہیں یہ مویشی اور کھیتی ممنوع ہیں اسے وہی کھا سکتا ہے جسے ہم اپنے خیال سے چاہیں اور کچھ جانور ہیں کہ ان کی پیٹھ (پر سواری) حرام ہے اور کچھ جانور ایسے ہیں کہ وہ (ذبح کرتے ہوئے) ان پر اللہ کا نام نہیں لیتے اس پر جھوٹ گھڑتے ہوئے عنقریب وہ (اللہ) انھیں اس کا بدلہ دے گا جو وہ جھوٹ گھڑا کرتے تھے۔ || وَقَالُوا۟ هَٰذِهِۦٓ أَنْعَٰمٌۭ وَحَرْثٌ حِجْرٌۭ لَّا يَطْعَمُهَآ إِلَّا مَن نَّشَآءُ بِزَعْمِهِمْ وَأَنْعَٰمٌ حُرِّمَتْ ظُهُورُهَا وَأَنْعَٰمٌۭ لَّا يَذْكُرُونَ ٱسْمَ ٱللَّهِ عَلَيْهَا ٱفْتِرَآءً عَلَيْهِ ۚ سَيَجْزِيهِم بِمَا كَانُوا۟ يَفْتَرُونَ ﴿١٣٨﴾ — And they say: "These cattle and crops are forbidden; none may eat them except whom we will," by their claim; and some cattle whose backs (for riding) are forbidden, and some cattle over which they do not mention Allah's name (at slaughter) — inventing a lie against Him. He (Allah) will soon repay them for what they used to invent. || وَقَالُوا۟ هَٰذِهِۦٓ أَنْعَٰمٌۭ وَحَرْثٌ حِجْرٌۭ لَّا يَطْعَمُهَآ إِلَّا مَن نَّشَآءُ بِزَعْمِهِمْ وَأَنْعَٰمٌ حُرِّمَتْ ظُهُورُهَا وَأَنْعَٰمٌۭ لَّا يَذْكُرُونَ ٱسْمَ ٱللَّهِ عَلَيْهَا ٱفْتِرَآءً عَلَيْهِ ۚ سَيَجْزِيهِم بِمَا كَانُوا۟ يَفْتَرُونَ ﴿١٣٨﴾ — او وايي: "دا څاروي او کښت منع دي؛ یوازې هغه یې خوړلای شي چې موږ یې په خپل ګومان وغواړو"؛ او ځینې څاروي داسې دي چې شا یې (سپرلۍ ته) حرامه ده، او ځینې څاروي داسې دي چې (د حلالولو پر وخت) د الله نوم پرې نه اخلي، پر هغه دروغ تړي؛ ژر به هغه (الله) د هغو دروغو بدله ورکړي چې جوړول یې.
