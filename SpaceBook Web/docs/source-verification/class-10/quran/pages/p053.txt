p أَمْ يَقُولُونَ ٱفْتَرَىٰهُ ۖ قُلْ فَأْتُوا۟ بِسُورَةٍۢ مِّثْلِهِۦ وَٱدْعُوا۟ مَنِ ٱسْتَطَعْتُم مِّن دُونِ ٱللَّهِ إِن كُنتُمْ صَٰدِقِينَ ﴿٣٨﴾ — کیا وہ یہ کہتے ہیں کہ اس (نبی) نے اسے خود گھڑ لیا ہے آپ (صلی اللہ علیہ وآلہ وسلم) فرما دیجیے پھر تم اس جیسی ایک سورت لے آؤ اور اللہ کے سوا جسے بلا سکو بلا لو اگر تم سچے ہو۔ || أَمْ يَقُولُونَ ٱفْتَرَىٰهُ ۖ قُلْ فَأْتُوا۟ بِسُورَةٍۢ مِّثْلِهِۦ وَٱدْعُوا۟ مَنِ ٱسْتَطَعْتُم مِّن دُونِ ٱللَّهِ إِن كُنتُمْ صَٰدِقِينَ ﴿٣٨﴾ — Or do they say: "He (the Prophet) has invented it"? Say: "Then bring a surah like it, and call whomever you can besides Allah, if you are truthful." || أَمْ يَقُولُونَ ٱفْتَرَىٰهُ ۖ قُلْ فَأْتُوا۟ بِسُورَةٍۢ مِّثْلِهِۦ وَٱدْعُوا۟ مَنِ ٱسْتَطَعْتُم مِّن دُونِ ٱللَّهِ إِن كُنتُمْ صَٰدِقِينَ ﴿٣٨﴾ — آیا وايي چې ده (نبي) پخپله جوړ کړی؟ ووایه: "نو د هغه په څېر یو سورت راوړئ او له الله پرته چې څوک رابللای شئ، راوغواړئ، که رښتیني یاست."
p بَلْ كَذَّبُوا۟ بِمَا لَمْ يُحِيطُوا۟ بِعِلْمِهِۦ وَلَمَّا يَأْتِهِمْ تَأْوِيلُهُۥ ۚ كَذَٰلِكَ كَذَّبَ ٱلَّذِينَ مِن قَبْلِهِمْ ۖ فَٱنظُرْ كَيْفَ كَانَ عَٰقِبَةُ ٱلظَّٰلِمِينَ ﴿٣٩﴾ — بلکہ انھوں نے (اسے) جھٹلا دیا جس کا وہ اپنے علم سے احاطہ نہ کر سکے اور ابھی تک اس کی حقیقت ان کے پاس نہیں آئی اسی طرح ان لوگوں نے جھٹلایا تھا جو اِن سے پہلے تھے تو دیکھیے ظالموں کا کیسا انجام ہوا؟ || بَلْ كَذَّبُوا۟ بِمَا لَمْ يُحِيطُوا۟ بِعِلْمِهِۦ وَلَمَّا يَأْتِهِمْ تَأْوِيلُهُۥ ۚ كَذَٰلِكَ كَذَّبَ ٱلَّذِينَ مِن قَبْلِهِمْ ۖ فَٱنظُرْ كَيْفَ كَانَ عَٰقِبَةُ ٱلظَّٰلِمِينَ ﴿٣٩﴾ — Rather, they denied what they could not encompass in their knowledge, and whose interpretation has not yet come to them. Thus did those before them deny; so see what was the end of the wrongdoers. || بَلْ كَذَّبُوا۟ بِمَا لَمْ يُحِيطُوا۟ بِعِلْمِهِۦ وَلَمَّا يَأْتِهِمْ تَأْوِيلُهُۥ ۚ كَذَٰلِكَ كَذَّبَ ٱلَّذِينَ مِن قَبْلِهِمْ ۖ فَٱنظُرْ كَيْفَ كَانَ عَٰقِبَةُ ٱلظَّٰلِمِينَ ﴿٣٩﴾ — بلکې هغوی هغه دروغ وګاڼه چې په خپل علم یې احاطه نه شوه کړای، او لا تر اوسه یې حقیقت ورته نه دی راغلی؛ همداسې هغو خلکو دروغ ګڼلي وو چې له دوی مخکې وو؛ نو وګوره چې د ظالمانو انجام څنګه شو؟
p وَمِنْهُم مَّن يُؤْمِنُ بِهِۦ وَمِنْهُم مَّن لَّا يُؤْمِنُ بِهِۦ ۚ وَرَبُّكَ أَعْلَمُ بِٱلْمُفْسِدِينَ ﴿٤٠﴾ — اور ان میں سے کچھ وہ ہیں جو اس (قرآن) پر ایمان لاتے ہیں اور ان میں سے کچھ اس پر ایمان نہیں لاتے اور آپ (صلی اللہ علیہ وآلہ وسلم) کا رب خوب جانتا ہے فساد کرنے والوں کو۔ || وَمِنْهُم مَّن يُؤْمِنُ بِهِۦ وَمِنْهُم مَّن لَّا يُؤْمِنُ بِهِۦ ۚ وَرَبُّكَ أَعْلَمُ بِٱلْمُفْسِدِينَ ﴿٤٠﴾ — Among them are some who believe in it (the Qur'an), and some who do not believe in it; and your Lord knows best the corrupters. || وَمِنْهُم مَّن يُؤْمِنُ بِهِۦ وَمِنْهُم مَّن لَّا يُؤْمِنُ بِهِۦ ۚ وَرَبُّكَ أَعْلَمُ بِٱلْمُفْسِدِينَ ﴿٤٠﴾ — او په دوی کې ځینې هغه دي چې پر دې (قرآن) ایمان راوړي، او ځینې پرې ایمان نه راوړي؛ او ستا رب فساد کوونکي ښه پېژني.
p وَإِن كَذَّبُوكَ فَقُل لِّى عَمَلِى وَلَكُمْ عَمَلُكُمْ ۖ أَنتُم بَرِيٓـُٔونَ مِمَّآ أَعْمَلُ وَأَنَا۠ بَرِىٓءٌۭ مِّمَّا تَعْمَلُونَ ﴿٤١﴾ — اور اگر وہ آپ کو جھٹلائیں تو آپ (صلی اللہ علیہ وآلہ وسلم) فرما دیجیے میرے لیے میرا عمل ہے اور تمھارے لیے تمھارا عمل تم بری ہو اُس سے جو میں کرتا ہوں اور میں بری ہوں اُس سے جو تم کرتے ہو۔ || وَإِن كَذَّبُوكَ فَقُل لِّى عَمَلِى وَلَكُمْ عَمَلُكُمْ ۖ أَنتُم بَرِيٓـُٔونَ مِمَّآ أَعْمَلُ وَأَنَا۠ بَرِىٓءٌۭ مِّمَّا تَعْمَلُونَ ﴿٤١﴾ — If they deny you, say: "For me are my deeds and for you are your deeds. You are free of what I do, and I am free of what you do." || وَإِن كَذَّبُوكَ فَقُل لِّى عَمَلِى وَلَكُمْ عَمَلُكُمْ ۖ أَنتُم بَرِيٓـُٔونَ مِمَّآ أَعْمَلُ وَأَنَا۠ بَرِىٓءٌۭ مِّمَّا تَعْمَلُونَ ﴿٤١﴾ — او که تا دروغجن وبولي، ووایه: "زما لپاره زما عمل دی او ستاسو لپاره ستاسو عمل؛ تاسو له هغه بري یاست چې زه یې کوم، او زه له هغه بري یم چې تاسو یې کوئ."
p وَمِنْهُم مَّن يَسْتَمِعُونَ إِلَيْكَ ۚ أَفَأَنتَ تُسْمِعُ ٱلصُّمَّ وَلَوْ كَانُوا۟ لَا يَعْقِلُونَ ﴿٤٢﴾ — اور ان میں سے کچھ وہ ہیں جو (بظاہر) آپ کی طرف کان لگاتے ہیں تو کیا آپ بہروں کو سناتے ہیں خواہ وہ کچھ نہ سمجھتے ہوں۔ || وَمِنْهُم مَّن يَسْتَمِعُونَ إِلَيْكَ ۚ أَفَأَنتَ تُسْمِعُ ٱلصُّمَّ وَلَوْ كَانُوا۟ لَا يَعْقِلُونَ ﴿٤٢﴾ — Among them are some who (seemingly) listen to you; but can you make the deaf hear, even though they do not understand? || وَمِنْهُم مَّن يَسْتَمِعُونَ إِلَيْكَ ۚ أَفَأَنتَ تُسْمِعُ ٱلصُّمَّ وَلَوْ كَانُوا۟ لَا يَعْقِلُونَ ﴿٤٢﴾ — او په دوی کې ځینې هغه دي چې (په ظاهره) تا ته غوږ نیسي؛ نو آیا ته کاڼو ته اورول کوې، که څه هم هېڅ نه پوهېږي؟
p وَمِنْهُم مَّن يَنظُرُ إِلَيْكَ ۚ أَفَأَنتَ تَهْدِى ٱلْعُمْىَ وَلَوْ كَانُوا۟ لَا يُبْصِرُونَ ﴿٤٣﴾ — اور اُن میں سے کچھ وہ ہیں جو (بظاہر) آپ کی طرف دیکھتے ہیں تو کیا آپ اندھوں کو (ہدایت کی) راہ دکھاتے ہیں خواہ وہ کچھ نہ دیکھتے ہوں۔ || وَمِنْهُم مَّن يَنظُرُ إِلَيْكَ ۚ أَفَأَنتَ تَهْدِى ٱلْعُمْىَ وَلَوْ كَانُوا۟ لَا يُبْصِرُونَ ﴿٤٣﴾ — And among them are some who (seemingly) look at you; but can you guide the blind (to the path), even though they do not see? || وَمِنْهُم مَّن يَنظُرُ إِلَيْكَ ۚ أَفَأَنتَ تَهْدِى ٱلْعُمْىَ وَلَوْ كَانُوا۟ لَا يُبْصِرُونَ ﴿٤٣﴾ — او په دوی کې ځینې هغه دي چې (په ظاهره) تا ته ګوري؛ نو آیا ته ړندو ته د (هدایت) لاره ښیې، که څه هم هېڅ نه ویني؟
p إِنَّ ٱللَّهَ لَا يَظْلِمُ ٱلنَّاسَ شَيْـًۭٔا وَلَٰكِنَّ ٱلنَّاسَ أَنفُسَهُمْ يَظْلِمُونَ ﴿٤٤﴾ — بے شک اللہ لوگوں پر کچھ بھی ظلم نہیں کرتا لیکن لوگ خود ہی اپنے جانوں پر ظلم کرتے ہیں۔ || إِنَّ ٱللَّهَ لَا يَظْلِمُ ٱلنَّاسَ شَيْـًۭٔا وَلَٰكِنَّ ٱلنَّاسَ أَنفُسَهُمْ يَظْلِمُونَ ﴿٤٤﴾ — Surely Allah does not wrong people at all, but people wrong themselves. || إِنَّ ٱللَّهَ لَا يَظْلِمُ ٱلنَّاسَ شَيْـًۭٔا وَلَٰكِنَّ ٱلنَّاسَ أَنفُسَهُمْ يَظْلِمُونَ ﴿٤٤﴾ — بې شکه الله پر خلکو هېڅ ظلم نه کوي، خو خلک پخپله پر خپلو ځانونو ظلم کوي.
p وَيَوْمَ يَحْشُرُهُمْ كَأَن لَّمْ يَلْبَثُوٓا۟ إِلَّا سَاعَةًۭ مِّنَ ٱلنَّهَارِ يَتَعَارَفُونَ بَيْنَهُمْ ۚ قَدْ خَسِرَ ٱلَّذِينَ كَذَّبُوا۟ بِلِقَآءِ ٱللَّهِ وَمَا كَانُوا۟ مُهْتَدِينَ ﴿٤٥﴾ — اور جس دن وہ انھیں جمع فرمائے گا (تو انھیں ایسا محسوس ہوگا) گویا کہ وہ (دنیا میں) دن کی صرف ایک گھڑی ٹھہرے ہیں وہ آپس میں ایک دوسرے کو پہچانتے ہوں گے یقیناً وہ خسارے میں رہے جنھوں نے اللہ کی ملاقات کو جھٹلایا اور وہ ہدایت پانے والے نہ تھے۔ || وَيَوْمَ يَحْشُرُهُمْ كَأَن لَّمْ يَلْبَثُوٓا۟ إِلَّا سَاعَةًۭ مِّنَ ٱلنَّهَارِ يَتَعَارَفُونَ بَيْنَهُمْ ۚ قَدْ خَسِرَ ٱلَّذِينَ كَذَّبُوا۟ بِلِقَآءِ ٱللَّهِ وَمَا كَانُوا۟ مُهْتَدِينَ ﴿٤٥﴾ — On the Day He gathers them, (they will feel) as if they had remained (in the world) only an hour of a day, recognising one another. Losers indeed are those who denied the meeting with Allah and were not guided. || وَيَوْمَ يَحْشُرُهُمْ كَأَن لَّمْ يَلْبَثُوٓا۟ إِلَّا سَاعَةًۭ مِّنَ ٱلنَّهَارِ يَتَعَارَفُونَ بَيْنَهُمْ ۚ قَدْ خَسِرَ ٱلَّذِينَ كَذَّبُوا۟ بِلِقَآءِ ٱللَّهِ وَمَا كَانُوا۟ مُهْتَدِينَ ﴿٤٥﴾ — او په هغه ورځ چې هغه راټول کړي، (داسې به ورته ښکاري) لکه (په دنیا کې) چې یوازې د ورځې یو ساعت پاتې شوي وي؛ یو بل به پېژني. یقیناً په تاوان کې شول هغه چې د الله ملاقات یې دروغ وګاڼه او هدایت موندونکي نه وو.
p وَإِمَّا نُرِيَنَّكَ بَعْضَ ٱلَّذِى نَعِدُهُمْ أَوْ نَتَوَفَّيَنَّكَ فَإِلَيْنَا مَرْجِعُهُمْ ثُمَّ ٱللَّهُ شَهِيدٌ عَلَىٰ مَا يَفْعَلُونَ ﴿٤٦﴾ — اور اگر ہم (عذاب کا) کچھ حصہ آپ کو دکھا دیں جس کا ہم ان سے وعدہ کرتے ہیں یا ہم آپ کو وفات دے دیں تو انھیں ہماری ہی طرف واپس آنا ہے پھر اللہ گواہ ہے اُس پر جو وہ کر رہے ہیں۔ || وَإِمَّا نُرِيَنَّكَ بَعْضَ ٱلَّذِى نَعِدُهُمْ أَوْ نَتَوَفَّيَنَّكَ فَإِلَيْنَا مَرْجِعُهُمْ ثُمَّ ٱللَّهُ شَهِيدٌ عَلَىٰ مَا يَفْعَلُونَ ﴿٤٦﴾ — Whether We show you part of (the punishment) We promise them, or take you (in death), to Us is their return; then Allah is witness over what they do. || وَإِمَّا نُرِيَنَّكَ بَعْضَ ٱلَّذِى نَعِدُهُمْ أَوْ نَتَوَفَّيَنَّكَ فَإِلَيْنَا مَرْجِعُهُمْ ثُمَّ ٱللَّهُ شَهِيدٌ عَلَىٰ مَا يَفْعَلُونَ ﴿٤٦﴾ — او که موږ تا ته د هغه (عذاب) ځینې برخه وښیو چې ورسره یې ژمنه کوو، یا تا ته وفات درکړو، نو ورګرځېدل یې زموږ لوري ته دي؛ بیا الله پر هغه څه شاهد دی چې دوی یې کوي.
