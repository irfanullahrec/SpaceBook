p ٱلَّذِينَ ءَاتَيْنَٰهُمُ ٱلْكِتَٰبَ يَعْرِفُونَهُۥ كَمَا يَعْرِفُونَ أَبْنَآءَهُمُ ۘ ٱلَّذِينَ خَسِرُوٓا۟ أَنفُسَهُمْ فَهُمْ لَا يُؤْمِنُونَ ﴿٢٠﴾ — جن لوگوں کو ہم نے کتاب عطا فرمائی ہے وہ اس (نبی صلی اللہ علیہ وآلہ وسلم) کو اس طرح پہچانتے ہیں جیسے اپنے بیٹوں کو پہچانتے ہیں جنھوں نے اپنے آپ کو نقصان میں ڈال رکھا ہے تو وہ ایمان نہیں لائیں گے۔ || ٱلَّذِينَ ءَاتَيْنَٰهُمُ ٱلْكِتَٰبَ يَعْرِفُونَهُۥ كَمَا يَعْرِفُونَ أَبْنَآءَهُمُ ۘ ٱلَّذِينَ خَسِرُوٓا۟ أَنفُسَهُمْ فَهُمْ لَا يُؤْمِنُونَ ﴿٢٠﴾ — Those to whom We have given the Book recognise him (the Prophet) as they recognise their own sons. Those who have put themselves into loss will not believe. || ٱلَّذِينَ ءَاتَيْنَٰهُمُ ٱلْكِتَٰبَ يَعْرِفُونَهُۥ كَمَا يَعْرِفُونَ أَبْنَآءَهُمُ ۘ ٱلَّذِينَ خَسِرُوٓا۟ أَنفُسَهُمْ فَهُمْ لَا يُؤْمِنُونَ ﴿٢٠﴾ — هغه کسان چې موږ ورته کتاب ورکړی، هغه (نبي) داسې پېژني لکه خپل زامن چې پېژني. هغه کسان چې ځانونه یې په تاوان کې اچولي، ایمان به رانه وړي.
p وَمَنْ أَظْلَمُ مِمَّنِ ٱفْتَرَىٰ عَلَى ٱللَّهِ كَذِبًا أَوْ كَذَّبَ بِـَٔايَٰتِهِۦٓ ۗ إِنَّهُۥ لَا يُفْلِحُ ٱلظَّٰلِمُونَ ﴿٢١﴾ — اور اس سے بڑا ظالم کون ہوگا جو اللہ پر جھوٹ باندھے یا اس کی آیتوں کو جھٹلائے بے شک ظلم کرنے والے کامیاب نہیں ہوں گے۔ || وَمَنْ أَظْلَمُ مِمَّنِ ٱفْتَرَىٰ عَلَى ٱللَّهِ كَذِبًا أَوْ كَذَّبَ بِـَٔايَٰتِهِۦٓ ۗ إِنَّهُۥ لَا يُفْلِحُ ٱلظَّٰلِمُونَ ﴿٢١﴾ — And who does greater wrong than one who invents a lie against Allah or denies His verses? Surely the wrongdoers will not succeed. || وَمَنْ أَظْلَمُ مِمَّنِ ٱفْتَرَىٰ عَلَى ٱللَّهِ كَذِبًا أَوْ كَذَّبَ بِـَٔايَٰتِهِۦٓ ۗ إِنَّهُۥ لَا يُفْلِحُ ٱلظَّٰلِمُونَ ﴿٢١﴾ — او له هغه به لوی ظالم څوک وي چې پر الله دروغ وتړي یا د هغه آیتونه دروغ وګڼي؟ بې شکه ظالمان به بریالي نه شي.
p وَيَوْمَ نَحْشُرُهُمْ جَمِيعًۭا ثُمَّ نَقُولُ لِلَّذِينَ أَشْرَكُوٓا۟ أَيْنَ شُرَكَآؤُكُمُ ٱلَّذِينَ كُنتُمْ تَزْعُمُونَ ﴿٢٢﴾ — اور جس دن ہم ان سب کو جمع کریں گے پھر ہم کہیں گے ان لوگوں سے جنھوں نے شرک کیا تھا (آج) تمھارے وہ شریک کہاں ہیں جنھیں تم (میرا شریک) خیال کرتے تھے۔ || وَيَوْمَ نَحْشُرُهُمْ جَمِيعًۭا ثُمَّ نَقُولُ لِلَّذِينَ أَشْرَكُوٓا۟ أَيْنَ شُرَكَآؤُكُمُ ٱلَّذِينَ كُنتُمْ تَزْعُمُونَ ﴿٢٢﴾ — And on the Day We gather them all, We will say to those who associated partners: "Where are (today) your partners whom you claimed (to be My partners)?" || وَيَوْمَ نَحْشُرُهُمْ جَمِيعًۭا ثُمَّ نَقُولُ لِلَّذِينَ أَشْرَكُوٓا۟ أَيْنَ شُرَكَآؤُكُمُ ٱلَّذِينَ كُنتُمْ تَزْعُمُونَ ﴿٢٢﴾ — او په هغه ورځ چې موږ ټول راټول کړو، بیا به هغو ته چې شرک یې کړی و ووایو: "(نن) ستاسو هغه شریکان چېرې دي چې تاسو (زما شریکان) ګڼل؟"
p ثُمَّ لَمْ تَكُن فِتْنَتُهُمْ إِلَّآ أَن قَالُوا۟ وَٱللَّهِ رَبِّنَا مَا كُنَّا مُشْرِكِينَ ﴿٢٣﴾ — پھر ان کے پاس کوئی عذر نہ ہوگا سوائے اس کے کہ وہ کہیں گے اللہ کی قسم! جو ہمارا رب ہے ہم شرک کرنے والے نہ تھے۔ || ثُمَّ لَمْ تَكُن فِتْنَتُهُمْ إِلَّآ أَن قَالُوا۟ وَٱللَّهِ رَبِّنَا مَا كُنَّا مُشْرِكِينَ ﴿٢٣﴾ — Then they will have no excuse except that they will say: "By Allah, our Lord, we were not polytheists." || ثُمَّ لَمْ تَكُن فِتْنَتُهُمْ إِلَّآ أَن قَالُوا۟ وَٱللَّهِ رَبِّنَا مَا كُنَّا مُشْرِكِينَ ﴿٢٣﴾ — بیا به یې هېڅ عذر نه وي پرته له دې چې ووایي: "په الله قسم چې زموږ رب دی، موږ شرک کوونکي نه وو."
p ٱنظُرْ كَيْفَ كَذَبُوا۟ عَلَىٰٓ أَنفُسِهِمْ ۚ وَضَلَّ عَنْهُم مَّا كَانُوا۟ يَفْتَرُونَ ﴿٢٤﴾ — دیکھیے انھوں نے کیسا جھوٹ بولا اپنے اوپر اور ان سے گم ہو جائے گا جو وہ جھوٹ باندھا کرتے تھے۔ || ٱنظُرْ كَيْفَ كَذَبُوا۟ عَلَىٰٓ أَنفُسِهِمْ ۚ وَضَلَّ عَنْهُم مَّا كَانُوا۟ يَفْتَرُونَ ﴿٢٤﴾ — See how they lied against themselves, and what they used to invent will be lost from them. || ٱنظُرْ كَيْفَ كَذَبُوا۟ عَلَىٰٓ أَنفُسِهِمْ ۚ وَضَلَّ عَنْهُم مَّا كَانُوا۟ يَفْتَرُونَ ﴿٢٤﴾ — وګوره چې پر ځان یې څنګه دروغ وویل، او له هغوی به ورک شي هغه څه چې دروغ یې تړل.
p وَمِنْهُم مَّن يَسْتَمِعُ إِلَيْكَ ۖ وَجَعَلْنَا عَلَىٰ قُلُوبِهِمْ أَكِنَّةً أَن يَفْقَهُوهُ وَفِىٓ ءَاذَانِهِمْ وَقْرًۭا ۚ وَإِن يَرَوْا۟ كُلَّ ءَايَةٍۢ لَّا يُؤْمِنُوا۟ بِهَا ۚ حَتَّىٰٓ إِذَا جَآءُوكَ يُجَٰدِلُونَكَ يَقُولُ ٱلَّذِينَ كَفَرُوٓا۟ إِنْ هَٰذَآ إِلَّآ أَسَٰطِيرُ ٱلْأَوَّلِينَ ﴿٢٥﴾ — اور ان میں سے کچھ ایسے بھی ہیں جو آپ کی طرف کان لگائے رہتے ہیں اور ہم نے ان کے دلوں پر ایسے پردے ڈال دیے ہیں (کہ ممکن نہیں) کہ وہ اس (قرآن) کو سمجھیں اور ان کے کانوں میں بوجھ ہے اور اگر وہ تمام نشانیاں دیکھ بھی لیں (تب بھی) ان پر ایمان نہیں لائیں گے یہاں تک کہ جب وہ آپ کے پاس آتے ہیں (تو) آپ سے جھگڑا کرتے ہیں وہ لوگ جنھوں نے کفر کیا کہتے ہیں یہ (قرآن) تو صرف پہلے لوگوں کی کہانیاں ہیں۔ || وَمِنْهُم مَّن يَسْتَمِعُ إِلَيْكَ ۖ وَجَعَلْنَا عَلَىٰ قُلُوبِهِمْ أَكِنَّةً أَن يَفْقَهُوهُ وَفِىٓ ءَاذَانِهِمْ وَقْرًۭا ۚ وَإِن يَرَوْا۟ كُلَّ ءَايَةٍۢ لَّا يُؤْمِنُوا۟ بِهَا ۚ حَتَّىٰٓ إِذَا جَآءُوكَ يُجَٰدِلُونَكَ يَقُولُ ٱلَّذِينَ كَفَرُوٓا۟ إِنْ هَٰذَآ إِلَّآ أَسَٰطِيرُ ٱلْأَوَّلِينَ ﴿٢٥﴾ — And among them are some who listen to you, but We have placed such coverings over their hearts (that it is not possible) for them to understand it (the Qur'an), and in their ears is deafness. Even if they saw every sign, they would not believe in it; so much so that when they come to you, they argue with you. Those who disbelieve say: "This (Qur'an) is nothing but tales of the ancients." || وَمِنْهُم مَّن يَسْتَمِعُ إِلَيْكَ ۖ وَجَعَلْنَا عَلَىٰ قُلُوبِهِمْ أَكِنَّةً أَن يَفْقَهُوهُ وَفِىٓ ءَاذَانِهِمْ وَقْرًۭا ۚ وَإِن يَرَوْا۟ كُلَّ ءَايَةٍۢ لَّا يُؤْمِنُوا۟ بِهَا ۚ حَتَّىٰٓ إِذَا جَآءُوكَ يُجَٰدِلُونَكَ يَقُولُ ٱلَّذِينَ كَفَرُوٓا۟ إِنْ هَٰذَآ إِلَّآ أَسَٰطِيرُ ٱلْأَوَّلِينَ ﴿٢٥﴾ — او په دوی کې ځینې داسې هم دي چې تا ته غوږ نیسي، او موږ د هغوی پر زړونو داسې پردې اچولې چې دا (قرآن) ونه پوهېږي، او په غوږونو کې یې درنښت دی؛ او که ټولې نښې هم وویني، (بیا هم) پرې ایمان نه راوړي؛ تر دې چې کله تا ته راشي، له تا سره شخړه کوي. کافران وايي: "دا (قرآن) یوازې د پخوانیو کیسې دي."
p وَهُمْ يَنْهَوْنَ عَنْهُ وَيَنْـَٔوْنَ عَنْهُ ۖ وَإِن يُهْلِكُونَ إِلَّآ أَنفُسَهُمْ وَمَا يَشْعُرُونَ ﴿٢٦﴾ — اور یہ لوگ (دوسروں کو) اس (قرآن) سے روکتے ہیں اور خود بھی اس سے دور بھاگتے ہیں اور (اس طرح) وہ صرف اپنے آپ کو ہی ہلاک کر رہے ہیں اور وہ کچھ خبر نہیں رکھتے۔ || وَهُمْ يَنْهَوْنَ عَنْهُ وَيَنْـَٔوْنَ عَنْهُ ۖ وَإِن يُهْلِكُونَ إِلَّآ أَنفُسَهُمْ وَمَا يَشْعُرُونَ ﴿٢٦﴾ — And they keep (others) from it (the Qur'an) and themselves keep away from it; and (so) they destroy only themselves, though they do not perceive it. || وَهُمْ يَنْهَوْنَ عَنْهُ وَيَنْـَٔوْنَ عَنْهُ ۖ وَإِن يُهْلِكُونَ إِلَّآ أَنفُسَهُمْ وَمَا يَشْعُرُونَ ﴿٢٦﴾ — او دا خلک (نور) له دې (قرآن) منع کوي او پخپله هم ترې لرې تښتي؛ او (په دې ډول) یوازې ځانونه هلاکوي او هېڅ خبر نه دي.
p وَلَوْ تَرَىٰٓ إِذْ وُقِفُوا۟ عَلَى ٱلنَّارِ فَقَالُوا۟ يَٰلَيْتَنَا نُرَدُّ وَلَا نُكَذِّبَ بِـَٔايَٰتِ رَبِّنَا وَنَكُونَ مِنَ ٱلْمُؤْمِنِينَ ﴿٢٧﴾ — اور اگر آپ (اس وقت) دیکھیں جب انھیں آگ (کے کنارے) پر کھڑا کیا جائے گا تو وہ کہیں گے کاش! ہم (دنیا میں) واپس بھیج دیے جائیں اور ہم اپنے رب کی آیات کو نہ جھٹلائیں اور ہم ایمان لانے والوں میں سے ہو جائیں۔ || وَلَوْ تَرَىٰٓ إِذْ وُقِفُوا۟ عَلَى ٱلنَّارِ فَقَالُوا۟ يَٰلَيْتَنَا نُرَدُّ وَلَا نُكَذِّبَ بِـَٔايَٰتِ رَبِّنَا وَنَكُونَ مِنَ ٱلْمُؤْمِنِينَ ﴿٢٧﴾ — And if you could see (them) when they are made to stand at (the edge of) the Fire, and they say: "Would that we were sent back (to the world); then we would not deny the verses of our Lord, and we would be among the believers." || وَلَوْ تَرَىٰٓ إِذْ وُقِفُوا۟ عَلَى ٱلنَّارِ فَقَالُوا۟ يَٰلَيْتَنَا نُرَدُّ وَلَا نُكَذِّبَ بِـَٔايَٰتِ رَبِّنَا وَنَكُونَ مِنَ ٱلْمُؤْمِنِينَ ﴿٢٧﴾ — او که ته (هغه وخت) ووینې چې هغوی د اور (پر غاړه) ودرول شي، نو وايي به: "کاشکې موږ (دنیا ته) بېرته ولېږل شو او د خپل رب آیتونه دروغ ونه ګڼو او له مومنانو شو."
