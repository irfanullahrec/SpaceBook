p وَأَشْرَقَتِ ٱلْأَرْضُ بِنُورِ رَبِّهَا وَوُضِعَ ٱلْكِتَٰبُ وَجِا۟ىٓءَ بِٱلنَّبِيِّۦنَ وَٱلشُّهَدَآءِ وَقُضِىَ بَيْنَهُم بِٱلْحَقِّ وَهُمْ لَا يُظْلَمُونَ ﴿٦٩﴾ — اور زمین اپنے رب کے نور سے چمک اٹھے گی اور (اعمال کی) کتاب رکھ دی جائے گی اور انبیاء (علیہم السلام) اور گواہ لائے جائیں گے اور ان کے درمیان حق کے ساتھ فیصلہ کر دیا جائے گا اور اُن پر ظلم نہیں کیا جائے گا۔ || وَأَشْرَقَتِ ٱلْأَرْضُ بِنُورِ رَبِّهَا وَوُضِعَ ٱلْكِتَٰبُ وَجِا۟ىٓءَ بِٱلنَّبِيِّۦنَ وَٱلشُّهَدَآءِ وَقُضِىَ بَيْنَهُم بِٱلْحَقِّ وَهُمْ لَا يُظْلَمُونَ ﴿٦٩﴾ — And the earth will shine with the light of its Lord, the Record (of deeds) will be placed, the prophets and the witnesses will be brought, and judgement will be made between them with truth, and they will not be wronged. || وَأَشْرَقَتِ ٱلْأَرْضُ بِنُورِ رَبِّهَا وَوُضِعَ ٱلْكِتَٰبُ وَجِا۟ىٓءَ بِٱلنَّبِيِّۦنَ وَٱلشُّهَدَآءِ وَقُضِىَ بَيْنَهُم بِٱلْحَقِّ وَهُمْ لَا يُظْلَمُونَ ﴿٦٩﴾ — او ځمکه به د خپل رب په نور وځلېږي، او (د عملونو) کتاب به کېښودل شي، او انبیاء (علیهم السلام) او شاهدان به راوستل شي، او د هغوی ترمنځ به په حق پرېکړه وشي، او پر هغوی به ظلم ونه شي.
p وَوُفِّيَتْ كُلُّ نَفْسٍۢ مَّا عَمِلَتْ وَهُوَ أَعْلَمُ بِمَا يَفْعَلُونَ ﴿٧٠﴾ — اور ہر شخص کو اس کے اعمال کا پورا پورا بدلہ دیا جائے گا اور وہ خوب جانتا ہے جو کچھ وہ کرتے ہیں۔ || وَوُفِّيَتْ كُلُّ نَفْسٍۢ مَّا عَمِلَتْ وَهُوَ أَعْلَمُ بِمَا يَفْعَلُونَ ﴿٧٠﴾ — And every soul will be paid in full for what it did, and He knows best what they do. || وَوُفِّيَتْ كُلُّ نَفْسٍۢ مَّا عَمِلَتْ وَهُوَ أَعْلَمُ بِمَا يَفْعَلُونَ ﴿٧٠﴾ — او هر چا ته به د هغه د عملونو پوره پوره بدله ورکړل شي؛ او هغه ښه پوهېږي په هغه چې دوی یې کوي.
p وَسِيقَ ٱلَّذِينَ كَفَرُوٓا۟ إِلَىٰ جَهَنَّمَ زُمَرًا ۖ حَتَّىٰٓ إِذَا جَآءُوهَا فُتِحَتْ أَبْوَٰبُهَا وَقَالَ لَهُمْ خَزَنَتُهَآ أَلَمْ يَأْتِكُمْ رُسُلٌۭ مِّنكُمْ يَتْلُونَ عَلَيْكُمْ ءَايَٰتِ رَبِّكُمْ وَيُنذِرُونَكُمْ لِقَآءَ يَوْمِكُمْ هَٰذَا ۚ قَالُوا۟ بَلَىٰ وَلَٰكِنْ حَقَّتْ كَلِمَةُ ٱلْعَذَابِ عَلَى ٱلْكَٰفِرِينَ ﴿٧١﴾ — اور جن لوگوں نے کفر کیا ہے وہ جہنم کی طرف گروہ در گروہ ہانکے جائیں گے یہاں تک کہ جب وہ اس کے پاس آئیں گے تو اس کے دروازے کھولے جائیں گے اور اس کے داروغہ ان سے کہیں گے کیا تمھارے پاس تم ہی میں سے رسول نہیں آئے تھے جو تمھیں تمھارے رب کی آیات پڑھ کر سناتے تھے اور تمھیں اس دن کی ملاقات سے ڈراتے تھے وہ کہیں گے کیوں نہیں! لیکن کافروں پر عذاب کا حکم ثابت ہو گیا۔ || وَسِيقَ ٱلَّذِينَ كَفَرُوٓا۟ إِلَىٰ جَهَنَّمَ زُمَرًا ۖ حَتَّىٰٓ إِذَا جَآءُوهَا فُتِحَتْ أَبْوَٰبُهَا وَقَالَ لَهُمْ خَزَنَتُهَآ أَلَمْ يَأْتِكُمْ رُسُلٌۭ مِّنكُمْ يَتْلُونَ عَلَيْكُمْ ءَايَٰتِ رَبِّكُمْ وَيُنذِرُونَكُمْ لِقَآءَ يَوْمِكُمْ هَٰذَا ۚ قَالُوا۟ بَلَىٰ وَلَٰكِنْ حَقَّتْ كَلِمَةُ ٱلْعَذَابِ عَلَى ٱلْكَٰفِرِينَ ﴿٧١﴾ — Those who disbelieved will be driven to Hell in groups, until, when they reach it, its gates will be opened and its keepers will say to them: "Did not messengers come to you from among yourselves, reciting to you the verses of your Lord and warning you of the meeting of this Day of yours?" They will say: "Yes indeed!" But the word of punishment has been confirmed against the disbelievers. || وَسِيقَ ٱلَّذِينَ كَفَرُوٓا۟ إِلَىٰ جَهَنَّمَ زُمَرًا ۖ حَتَّىٰٓ إِذَا جَآءُوهَا فُتِحَتْ أَبْوَٰبُهَا وَقَالَ لَهُمْ خَزَنَتُهَآ أَلَمْ يَأْتِكُمْ رُسُلٌۭ مِّنكُمْ يَتْلُونَ عَلَيْكُمْ ءَايَٰتِ رَبِّكُمْ وَيُنذِرُونَكُمْ لِقَآءَ يَوْمِكُمْ هَٰذَا ۚ قَالُوا۟ بَلَىٰ وَلَٰكِنْ حَقَّتْ كَلِمَةُ ٱلْعَذَابِ عَلَى ٱلْكَٰفِرِينَ ﴿٧١﴾ — او هغه کسان چې کفر یې کړی، دوزخ ته به ډله ډله وشړل شي، تر دې چې کله ورته ورسېږي، نو دروازې به یې پرانیستل شي، او د هغه ساتونکي به ورته ووایي: "آیا تاسو ته له تاسو څخه رسولان نه وو راغلي چې ستاسو د رب آیتونه به یې درته لوستل او د دې ورځې له ملاقاته به یې وېرولئ؟" وایي به: "ولې نه!" خو پر کافرانو د عذاب حکم ثابت شو.
p قِيلَ ٱدْخُلُوٓا۟ أَبْوَٰبَ جَهَنَّمَ خَٰلِدِينَ فِيهَا ۖ فَبِئْسَ مَثْوَى ٱلْمُتَكَبِّرِينَ ﴿٧٢﴾ — ان سے کہا جائے گا جہنم کے دروازوں میں داخل ہو جاؤ (تم) اس میں ہمیشہ رہو گے لہٰذا تکبر کرنے والوں کا برا ٹھکانا ہے۔ || قِيلَ ٱدْخُلُوٓا۟ أَبْوَٰبَ جَهَنَّمَ خَٰلِدِينَ فِيهَا ۖ فَبِئْسَ مَثْوَى ٱلْمُتَكَبِّرِينَ ﴿٧٢﴾ — It will be said to them: "Enter the gates of Hell, to abide in it forever." Evil indeed is the abode of the arrogant. || قِيلَ ٱدْخُلُوٓا۟ أَبْوَٰبَ جَهَنَّمَ خَٰلِدِينَ فِيهَا ۖ فَبِئْسَ مَثْوَى ٱلْمُتَكَبِّرِينَ ﴿٧٢﴾ — هغوی ته به وویل شي: "د دوزخ له دروازو ننوځئ، (تاسو) به پکې تل اوسئ"؛ نو د تکبر کوونکو ځای ډېر بد دی.
p وَسِيقَ ٱلَّذِينَ ٱتَّقَوْا۟ رَبَّهُمْ إِلَى ٱلْجَنَّةِ زُمَرًا ۖ حَتَّىٰٓ إِذَا جَآءُوهَا وَفُتِحَتْ أَبْوَٰبُهَا وَقَالَ لَهُمْ خَزَنَتُهَا سَلَٰمٌ عَلَيْكُمْ طِبْتُمْ فَٱدْخُلُوهَا خَٰلِدِينَ ﴿٧٣﴾ — اور جو لوگ اپنے رب سے ڈرتے رہے انھیں گروہ در گروہ جنت کی طرف لے جایا جائے گا یہاں تک کہ جب وہ اس کے پاس پہنچیں گے اس کے دروازے (پہلے ہی) کھولے جا چکے ہوں گے اور اس کے محافظ ان سے کہیں گے تم پر سلام ہو تم خوب رہے لہٰذا اس میں ہمیشہ رہنے کے لیے داخل ہو جاؤ۔ || وَسِيقَ ٱلَّذِينَ ٱتَّقَوْا۟ رَبَّهُمْ إِلَى ٱلْجَنَّةِ زُمَرًا ۖ حَتَّىٰٓ إِذَا جَآءُوهَا وَفُتِحَتْ أَبْوَٰبُهَا وَقَالَ لَهُمْ خَزَنَتُهَا سَلَٰمٌ عَلَيْكُمْ طِبْتُمْ فَٱدْخُلُوهَا خَٰلِدِينَ ﴿٧٣﴾ — And those who feared their Lord will be led to Paradise in groups, until, when they reach it, its gates (already) opened, its keepers will say to them: "Peace be upon you; you have done well, so enter it to abide forever." || وَسِيقَ ٱلَّذِينَ ٱتَّقَوْا۟ رَبَّهُمْ إِلَى ٱلْجَنَّةِ زُمَرًا ۖ حَتَّىٰٓ إِذَا جَآءُوهَا وَفُتِحَتْ أَبْوَٰبُهَا وَقَالَ لَهُمْ خَزَنَتُهَا سَلَٰمٌ عَلَيْكُمْ طِبْتُمْ فَٱدْخُلُوهَا خَٰلِدِينَ ﴿٧٣﴾ — او هغه کسان چې له خپل رب وېرېدل، ډله ډله به جنت ته بوتلل شي، تر دې چې کله ورته ورسېږي، دروازې به یې (مخکې) پرانیستل شوې وي، او ساتونکي به یې ورته ووایي: "پر تاسو سلام وي، تاسو ښه پاتې شوئ، نو په دې کې د تل اوسېدو لپاره ننوځئ."
p وَقَالُوا۟ ٱلْحَمْدُ لِلَّهِ ٱلَّذِى صَدَقَنَا وَعْدَهُۥ وَأَوْرَثَنَا ٱلْأَرْضَ نَتَبَوَّأُ مِنَ ٱلْجَنَّةِ حَيْثُ نَشَآءُ ۖ فَنِعْمَ أَجْرُ ٱلْعَٰمِلِينَ ﴿٧٤﴾ — اور وہ کہیں گے تمام تعریفیں اس اللہ کے لیے ہیں جس نے ہم سے اپنا وعدہ سچا کر دکھایا اور ہمیں اس زمین کا وارث بنا دیا اس جنت میں ہم جہاں چاہیں رہیں لہٰذا عمل کرنے والوں کا کیا ہی اچھا بدلہ ہے۔ || وَقَالُوا۟ ٱلْحَمْدُ لِلَّهِ ٱلَّذِى صَدَقَنَا وَعْدَهُۥ وَأَوْرَثَنَا ٱلْأَرْضَ نَتَبَوَّأُ مِنَ ٱلْجَنَّةِ حَيْثُ نَشَآءُ ۖ فَنِعْمَ أَجْرُ ٱلْعَٰمِلِينَ ﴿٧٤﴾ — And they will say: "All praise is for Allah, who has fulfilled His promise to us and made us inherit the land, so that we may settle in Paradise wherever we will." How excellent is the reward of the workers (of good)! || وَقَالُوا۟ ٱلْحَمْدُ لِلَّهِ ٱلَّذِى صَدَقَنَا وَعْدَهُۥ وَأَوْرَثَنَا ٱلْأَرْضَ نَتَبَوَّأُ مِنَ ٱلْجَنَّةِ حَيْثُ نَشَآءُ ۖ فَنِعْمَ أَجْرُ ٱلْعَٰمِلِينَ ﴿٧٤﴾ — او هغوی به ووایي: "ټول ستاینه د هغه الله لپاره ده چې له موږ سره یې خپله ژمنه رښتیا کړه او موږ یې د دې ځمکې وارثان کړو؛ په دې جنت کې چېرته چې وغواړو اوسېږو"؛ نو د عمل کوونکو بدله څومره ښه ده!
p وَتَرَى ٱلْمَلَٰٓئِكَةَ حَآفِّينَ مِنْ حَوْلِ ٱلْعَرْشِ يُسَبِّحُونَ بِحَمْدِ رَبِّهِمْ ۖ وَقُضِىَ بَيْنَهُم بِٱلْحَقِّ وَقِيلَ ٱلْحَمْدُ لِلَّهِ رَبِّ ٱلْعَٰلَمِينَ ﴿٧٥﴾ — اور آپ فرشتوں کو دیکھیں گے عرش کے گرد حلقہ بنائے ہوئے اپنے رب کی حمد کے ساتھ اس کی تسبیح کر رہے ہوں گے اور ان کے درمیان حق کے ساتھ فیصلہ کر دیا جائے گا اور کہا جائے گا تمام تعریفیں اللہ ہی کے لیے ہیں جو تمام جہانوں کا رب ہے۔ || وَتَرَى ٱلْمَلَٰٓئِكَةَ حَآفِّينَ مِنْ حَوْلِ ٱلْعَرْشِ يُسَبِّحُونَ بِحَمْدِ رَبِّهِمْ ۖ وَقُضِىَ بَيْنَهُم بِٱلْحَقِّ وَقِيلَ ٱلْحَمْدُ لِلَّهِ رَبِّ ٱلْعَٰلَمِينَ ﴿٧٥﴾ — And you will see the angels surrounding the Throne, glorifying their Lord with His praise. Judgement will be made between them with truth, and it will be said: "All praise is for Allah, Lord of all the worlds." || وَتَرَى ٱلْمَلَٰٓئِكَةَ حَآفِّينَ مِنْ حَوْلِ ٱلْعَرْشِ يُسَبِّحُونَ بِحَمْدِ رَبِّهِمْ ۖ وَقُضِىَ بَيْنَهُم بِٱلْحَقِّ وَقِيلَ ٱلْحَمْدُ لِلَّهِ رَبِّ ٱلْعَٰلَمِينَ ﴿٧٥﴾ — او ته به پرښتې وګورې چې د عرش شاوخوا یې حلقه جوړه کړې، د خپل رب له حمد سره یې تسبیح وایي؛ او د هغوی ترمنځ به په حق پرېکړه وشي، او وبه ویل شي: "ټول ستاینه د الله لپاره ده چې د ټولو جهانونو رب دی."
