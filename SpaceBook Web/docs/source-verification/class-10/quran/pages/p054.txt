p وَلِكُلِّ أُمَّةٍۢ رَّسُولٌۭ ۖ فَإِذَا جَآءَ رَسُولُهُمْ قُضِىَ بَيْنَهُم بِٱلْقِسْطِ وَهُمْ لَا يُظْلَمُونَ ﴿٤٧﴾ — اور ہر اُمت کے لیے ایک رسول ہے لہٰذا جب ان کا رسول آ جاتا ہے تو اُن کے درمیان انصاف کے ساتھ فیصلہ کر دیا جاتا ہے اور اُن پر ظلم نہیں کیا جاتا۔ || وَلِكُلِّ أُمَّةٍۢ رَّسُولٌۭ ۖ فَإِذَا جَآءَ رَسُولُهُمْ قُضِىَ بَيْنَهُم بِٱلْقِسْطِ وَهُمْ لَا يُظْلَمُونَ ﴿٤٧﴾ — For every nation there is a messenger; when their messenger comes, judgement is made between them with justice, and they are not wronged. || وَلِكُلِّ أُمَّةٍۢ رَّسُولٌۭ ۖ فَإِذَا جَآءَ رَسُولُهُمْ قُضِىَ بَيْنَهُم بِٱلْقِسْطِ وَهُمْ لَا يُظْلَمُونَ ﴿٤٧﴾ — او هر امت ته یو رسول دی؛ نو کله چې یې رسول راشي، د هغوی ترمنځ په انصاف پرېکړه کېږي او پر هغوی ظلم نه کېږي.
p وَيَقُولُونَ مَتَىٰ هَٰذَا ٱلْوَعْدُ إِن كُنتُمْ صَٰدِقِينَ ﴿٤٨﴾ — اور وہ کہتے ہیں یہ (قیامت کا) وعدہ کب (پورا) ہوگا اگر تم سچے ہو؟ || وَيَقُولُونَ مَتَىٰ هَٰذَا ٱلْوَعْدُ إِن كُنتُمْ صَٰدِقِينَ ﴿٤٨﴾ — And they say: "When will this promise (of the Resurrection) be (fulfilled), if you are truthful?" || وَيَقُولُونَ مَتَىٰ هَٰذَا ٱلْوَعْدُ إِن كُنتُمْ صَٰدِقِينَ ﴿٤٨﴾ — او وايي: "دا (د قیامت) ژمنه به کله (پوره) شي، که رښتیني یاست؟"
p قُل لَّآ أَمْلِكُ لِنَفْسِى ضَرًّۭا وَلَا نَفْعًا إِلَّا مَا شَآءَ ٱللَّهُ ۗ لِكُلِّ أُمَّةٍ أَجَلٌ ۚ إِذَا جَآءَ أَجَلُهُمْ فَلَا يَسْتَـْٔخِرُونَ سَاعَةًۭ ۖ وَلَا يَسْتَقْدِمُونَ ﴿٤٩﴾ — آپ (صلی اللہ علیہ وآلہ وسلم) فرما دیجیے میں خود اپنے لیے اختیار نہیں رکھتا (نہ) کسی نقصان کا اور نہ ہی کسی نفع کا مگر جو اللہ چاہے ہر اُمت کے لیے ایک وقت مقرر ہے جب ان کا مقرر وقت آ جاتا ہے تو نہ وہ ایک گھڑی پیچھے ہٹ سکتے ہیں اور نہ وہ آگے بڑھ سکتے ہیں۔ || قُل لَّآ أَمْلِكُ لِنَفْسِى ضَرًّۭا وَلَا نَفْعًا إِلَّا مَا شَآءَ ٱللَّهُ ۗ لِكُلِّ أُمَّةٍ أَجَلٌ ۚ إِذَا جَآءَ أَجَلُهُمْ فَلَا يَسْتَـْٔخِرُونَ سَاعَةًۭ ۖ وَلَا يَسْتَقْدِمُونَ ﴿٤٩﴾ — Say: "I have no power over harm or benefit for myself except as Allah wills. For every nation there is a term; when their term comes, they can neither delay it by an hour nor advance it." || قُل لَّآ أَمْلِكُ لِنَفْسِى ضَرًّۭا وَلَا نَفْعًا إِلَّا مَا شَآءَ ٱللَّهُ ۗ لِكُلِّ أُمَّةٍ أَجَلٌ ۚ إِذَا جَآءَ أَجَلُهُمْ فَلَا يَسْتَـْٔخِرُونَ سَاعَةًۭ ۖ وَلَا يَسْتَقْدِمُونَ ﴿٤٩﴾ — ووایه: "زه پخپله ځان ته (نه) د کوم زیان واک لرم او نه د کومې ګټې، مګر څه چې الله وغواړي. هر امت ته یو ټاکلی وخت دی؛ کله چې یې ټاکلی وخت راشي، نه یو ساعت وروسته کېدای شي او نه مخکې."
p قُلْ أَرَءَيْتُمْ إِنْ أَتَىٰكُمْ عَذَابُهُۥ بَيَٰتًا أَوْ نَهَارًۭا مَّاذَا يَسْتَعْجِلُ مِنْهُ ٱلْمُجْرِمُونَ ﴿٥٠﴾ — آپ (صلی اللہ علیہ وآلہ وسلم) فرما دیجیے بھلا تم دیکھو اگر تم پر اُس (اللہ) کا عذاب رات یا دن کو آ جائے (آخر) وہ کون سی چیز ہے جس کی مجرم جلدی کر رہے ہیں؟ || قُلْ أَرَءَيْتُمْ إِنْ أَتَىٰكُمْ عَذَابُهُۥ بَيَٰتًا أَوْ نَهَارًۭا مَّاذَا يَسْتَعْجِلُ مِنْهُ ٱلْمُجْرِمُونَ ﴿٥٠﴾ — Say: "Tell me, if His (Allah's) punishment came upon you by night or by day — what part of it would the criminals hasten?" || قُلْ أَرَءَيْتُمْ إِنْ أَتَىٰكُمْ عَذَابُهُۥ بَيَٰتًا أَوْ نَهَارًۭا مَّاذَا يَسْتَعْجِلُ مِنْهُ ٱلْمُجْرِمُونَ ﴿٥٠﴾ — ووایه: "ووایئ، که پر تاسو د هغه (الله) عذاب د شپې یا ورځې راشي، (نو) هغه کوم شی دی چې مجرمان یې بېړه کوي؟"
p أَثُمَّ إِذَا مَا وَقَعَ ءَامَنتُم بِهِۦٓ ۚ ءَآلْـَٰٔنَ وَقَدْ كُنتُم بِهِۦ تَسْتَعْجِلُونَ ﴿٥١﴾ — کیا جب (عذاب) واقع ہو جائے گا تب تم اُس پر ایمان لاؤ گے (کہا جائے گا) کیا اب (مانتے ہو) حالاں کہ تم تو اس عذاب کے لیے بڑی جلدی مچا رہے تھے۔ || أَثُمَّ إِذَا مَا وَقَعَ ءَامَنتُم بِهِۦٓ ۚ ءَآلْـَٰٔنَ وَقَدْ كُنتُم بِهِۦ تَسْتَعْجِلُونَ ﴿٥١﴾ — Is it then, when it (the punishment) has befallen, that you will believe in it? (It will be said:) "Now (you believe)? Yet you were hastening it." || أَثُمَّ إِذَا مَا وَقَعَ ءَامَنتُم بِهِۦٓ ۚ ءَآلْـَٰٔنَ وَقَدْ كُنتُم بِهِۦ تَسْتَعْجِلُونَ ﴿٥١﴾ — آیا کله چې (عذاب) واقع شي، هغه وخت به پرې ایمان راوړئ؟ (ویل به شي:) "آیا اوس (منئ)؟ حال دا چې تاسو د دې عذاب لپاره ډېره بېړه کوله."
p ثُمَّ قِيلَ لِلَّذِينَ ظَلَمُوا۟ ذُوقُوا۟ عَذَابَ ٱلْخُلْدِ هَلْ تُجْزَوْنَ إِلَّا بِمَا كُنتُمْ تَكْسِبُونَ ﴿٥٢﴾ — پھر اُن لوگوں سے کہا جائے گا جنھوں نے ظلم کیا ہمیشہ کا عذاب چکھو (کہا جائے گا) کیا تمھیں تمھارے کیے کے سوا کسی اور چیز کا بدلہ دیا جائے گا؟ || ثُمَّ قِيلَ لِلَّذِينَ ظَلَمُوا۟ ذُوقُوا۟ عَذَابَ ٱلْخُلْدِ هَلْ تُجْزَوْنَ إِلَّا بِمَا كُنتُمْ تَكْسِبُونَ ﴿٥٢﴾ — Then it will be said to those who did wrong: "Taste the everlasting punishment. Are you repaid for anything but what you used to earn?" || ثُمَّ قِيلَ لِلَّذِينَ ظَلَمُوا۟ ذُوقُوا۟ عَذَابَ ٱلْخُلْدِ هَلْ تُجْزَوْنَ إِلَّا بِمَا كُنتُمْ تَكْسِبُونَ ﴿٥٢﴾ — بیا به هغو ته چې ظلم یې کړی وویل شي: "د تل عذاب وڅکئ؛ آیا تاسو ته له خپلو کړو پرته د بل شي بدله درکول کېږي؟"
p ۞ وَيَسْتَنۢبِـُٔونَكَ أَحَقٌّ هُوَ ۖ قُلْ إِى وَرَبِّىٓ إِنَّهُۥ لَحَقٌّۭ ۖ وَمَآ أَنتُم بِمُعْجِزِينَ ﴿٥٣﴾ — اور وہ آپ (صلی اللہ علیہ وآلہ وسلم) سے پوچھتے ہیں کیا یہ حق ہے؟ آپ (صلی اللہ علیہ وآلہ وسلم) فرما دیجیے ہاں! میرے رب کی قسم! یقیناً یہی حق ہے اور تم (اللہ کو) عاجز کرنے والے نہیں۔ || ۞ وَيَسْتَنۢبِـُٔونَكَ أَحَقٌّ هُوَ ۖ قُلْ إِى وَرَبِّىٓ إِنَّهُۥ لَحَقٌّۭ ۖ وَمَآ أَنتُم بِمُعْجِزِينَ ﴿٥٣﴾ — They ask you: "Is it true?" Say: "Yes, by my Lord! It is surely the truth, and you cannot frustrate (Allah)." || ۞ وَيَسْتَنۢبِـُٔونَكَ أَحَقٌّ هُوَ ۖ قُلْ إِى وَرَبِّىٓ إِنَّهُۥ لَحَقٌّۭ ۖ وَمَآ أَنتُم بِمُعْجِزِينَ ﴿٥٣﴾ — او له تا پوښتنه کوي: "آیا دا حق دی؟" ووایه: "هو! زما په رب قسم! یقیناً همدا حق دی، او تاسو (الله) عاجز کوونکي نه یاست."
p وَلَوْ أَنَّ لِكُلِّ نَفْسٍۢ ظَلَمَتْ مَا فِى ٱلْأَرْضِ لَٱفْتَدَتْ بِهِۦ ۗ وَأَسَرُّوا۟ ٱلنَّدَامَةَ لَمَّا رَأَوُا۟ ٱلْعَذَابَ ۖ وَقُضِىَ بَيْنَهُم بِٱلْقِسْطِ ۚ وَهُمْ لَا يُظْلَمُونَ ﴿٥٤﴾ — اور اگر ہر اُس شخص کے لیے جس نے ظلم کیا (وہ سب کچھ ہو) جو زمین میں ہے (تو) وہ (عذاب سے بچنے کے لیے) اُسے ضرور فدیے میں دے دے اور وہ ندامت کو چھپائیں گے جب وہ عذاب دیکھ لیں گے اور ان کے درمیان انصاف کے ساتھ فیصلہ کیا جائے گا اور اُن پر ظلم نہیں کیا جائے گا۔ || وَلَوْ أَنَّ لِكُلِّ نَفْسٍۢ ظَلَمَتْ مَا فِى ٱلْأَرْضِ لَٱفْتَدَتْ بِهِۦ ۗ وَأَسَرُّوا۟ ٱلنَّدَامَةَ لَمَّا رَأَوُا۟ ٱلْعَذَابَ ۖ وَقُضِىَ بَيْنَهُم بِٱلْقِسْطِ ۚ وَهُمْ لَا يُظْلَمُونَ ﴿٥٤﴾ — If every soul that did wrong had all that is on earth, it would surely offer it as ransom (to escape the punishment). They will hide their regret when they see the punishment, and judgement will be made between them with justice, and they will not be wronged. || وَلَوْ أَنَّ لِكُلِّ نَفْسٍۢ ظَلَمَتْ مَا فِى ٱلْأَرْضِ لَٱفْتَدَتْ بِهِۦ ۗ وَأَسَرُّوا۟ ٱلنَّدَامَةَ لَمَّا رَأَوُا۟ ٱلْعَذَابَ ۖ وَقُضِىَ بَيْنَهُم بِٱلْقِسْطِ ۚ وَهُمْ لَا يُظْلَمُونَ ﴿٥٤﴾ — او که د هر هغه چا لپاره چې ظلم یې کړی (هر څه وي) چې په ځمکه کې دي، (نو) هغه به یې خامخا (له عذابه د ژغورنې لپاره) فدیه کړي؛ او کله چې عذاب ووینې، پښېماني به پټه کړي، او د هغوی ترمنځ به په انصاف پرېکړه وشي او پر هغوی به ظلم ونه شي.
p أَلَآ إِنَّ لِلَّهِ مَا فِى ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ ۗ أَلَآ إِنَّ وَعْدَ ٱللَّهِ حَقٌّۭ وَلَٰكِنَّ أَكْثَرَهُمْ لَا يَعْلَمُونَ ﴿٥٥﴾ — آگاہ ہو جاؤ! بے شک اللہ ہی کا ہے جو کچھ آسمانوں اور زمینوں میں ہے آگاہ ہو جاؤ! بے شک اللہ کا وعدہ سچا ہے لیکن ان میں سے اکثر لوگ نہیں جانتے۔ || أَلَآ إِنَّ لِلَّهِ مَا فِى ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ ۗ أَلَآ إِنَّ وَعْدَ ٱللَّهِ حَقٌّۭ وَلَٰكِنَّ أَكْثَرَهُمْ لَا يَعْلَمُونَ ﴿٥٥﴾ — Beware! To Allah belongs whatever is in the heavens and the earth. Beware! Allah's promise is true, but most of them do not know. || أَلَآ إِنَّ لِلَّهِ مَا فِى ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ ۗ أَلَآ إِنَّ وَعْدَ ٱللَّهِ حَقٌّۭ وَلَٰكِنَّ أَكْثَرَهُمْ لَا يَعْلَمُونَ ﴿٥٥﴾ — خبر اوسئ! بې شکه هر څه چې په اسمانونو او ځمکو کې دي، د الله دي. خبر اوسئ! بې شکه د الله ژمنه رښتیا ده، خو ډېری یې نه پوهېږي.
p هُوَ يُحْىِۦ وَيُمِيتُ وَإِلَيْهِ تُرْجَعُونَ ﴿٥٦﴾ — وہی زندگی عطا فرماتا ہے اور موت دیتا ہے اور اُسی کی طرف تم لوٹائے جاؤ گے۔ || هُوَ يُحْىِۦ وَيُمِيتُ وَإِلَيْهِ تُرْجَعُونَ ﴿٥٦﴾ — He gives life and causes death, and to Him you will be returned. || هُوَ يُحْىِۦ وَيُمِيتُ وَإِلَيْهِ تُرْجَعُونَ ﴿٥٦﴾ — همغه ژوند ورکوي او مرګ ورکوي، او هغه ته به ورګرځول شئ.
