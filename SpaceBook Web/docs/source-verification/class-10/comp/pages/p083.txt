### پروگرام 3.18 C لینگویج میں استعمال ہونے والے تمام آپریٹرز کی فوقیت واضح کرنے والا پروگرام۔ || Program 3.18 A program that shows the precedence of the operators used in the C language || پروګرام ۳.۱۸ داسې پروګرام چې په C ژبه کې د کارېدونکو عملګرونو لومړیتوب ښيي
m #include <stdio.h>
m void main()
m {
m     int a = 20;
m     int b = 10;
m     int c = 15;
m     int d = 5;
m     int e;
m     e = (a + b) * c / d;      // ( 30 * 15 ) / 5
m     printf("Value of (a + b) * c / d is : %d\n",  e );
m     e = ((a + b) * c) / d;    // (30 * 15 ) / 5
m     printf("Value of ((a + b) * c) / d is  : %d\n" ,  e );
m     e = (a + b) * (c / d);    // (30) * (15/5)
m     printf("Value of (a + b) * (c / d) is  : %d\n",  e );
m     e = a + (b * c) / d;      //  20 + (150/5)
m     printf("Value of a + (b * c) / d is  : %d\n" ,  e );
m }
