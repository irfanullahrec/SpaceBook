### پروگرام 4.7 یہ پروگرام بتائے گا کہ جو کریکٹر اِن پُٹ ہوا ہے وہ Vowel ہے یا نہیں۔ || Program 4.7 This program tells whether the character input is a vowel or not || پروګرام ۴.۷ دا پروګرام ښيي چې داخل شوی توری vowel دی که نه
m #include<stdio.h>
m void main( )
m {
m     char ch;
m     printf("Enter a character   ");
m     scanf("%c",&ch);
m     switch (ch)
m     {
m         case 'a':     printf("vowel a");
m         break ;
m         case 'e':     printf("vowel e");
m         break ;
m         case 'o':     printf("vowel o");
m         break ;
