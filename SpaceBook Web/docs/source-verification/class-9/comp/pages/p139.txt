m void main()
m {
m     int i,j;

### Program 5.14: A program to find factorial of a number. || پروگرام 5.14: کسی عدد کا فیکٹوریل معلوم کرنے کا پروگرام۔ || پروګرام ۵.۱۴: د عدد فکټوریل موندل.
m #include <iostream>
m using namespace std;
m int main() {
m   int i, n, factorial = 1;
m   cout<<"Enter a positive integer: ";
m   cin>>n;
m   for (i = 1; i <= n; ++i) {
m     factorial *= i;   // factorial = factorial * i;
m   }
m   cout<< "Factorial of "<<n<<" = "<<factorial;
m             return 0;
m }
