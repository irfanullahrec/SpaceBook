### Program 3.18: Program to illustrate the operator precedence available in C programming language: || پروگرام 3.18: C پروگرامنگ زبان میں آپریٹر کی ترجیح دکھانے کا پروگرام: || پروګرام ۳.۱۸: په C کې د operator لومړیتوب ښودل:
m #include <stdio.h>
m void main()
m {
m     int a = 20;
m     int b = 10;
m     int c = 15;
m     int d = 5;
m     int e;
m     e = (a + b) * c / d;      // ( 30 * 15 ) / 5
m     printf("Value of (a + b) * c / d is : %d\n",  e );
m
m     e = ((a + b) * c) / d;    // (30 * 15 ) / 5
m     printf("Value of ((a + b) * c) / d is  : %d\n" ,  e );
m     e = (a + b) * (c / d);    // (30) * (15/5)
