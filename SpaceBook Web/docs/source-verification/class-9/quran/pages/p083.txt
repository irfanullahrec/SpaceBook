p … اَمْ كُنْتَ مِنَ الْكٰذِبِيْنَ ﴿٢٧﴾ اِذْهَبْ بِّكِتٰبِيْ هٰذَا فَاَلْقِهْ اِلَيْهِمْ ثُمَّ تَوَلَّ عَنْهُمْ فَانْظُرْ مَاذَا يَرْجِعُوْنَ ﴿٢٨﴾ — … یا تم جھوٹوں میں سے ہو۔ میرا یہ خط لے جاؤ پھر اسے ان کی طرف ڈال دو پھر ان سے (الگ) ہٹ کر دیکھو کہ وہ کیا جواب دیتے ہیں؟ || "… or were of the liars. Take this letter of mine and deliver it to them. Then leave them and see what (answer) they will return." || … یا له دروغجنو یې. دا زما لیک یوسه او هغوی ته یې وغورځوه، بیا ترې (لرې) شه او وګوره چې څه ځواب راکوي؟
p قَالَتْ يٰٓاَيُّهَا الْمَلَؤُا اِنِّىْٓ اُلْقِيَ اِلَيَّ كِتٰبٌ كَرِيْمٌ ﴿٢٩﴾ اِنَّهٗ مِنْ سُلَيْمٰنَ وَاِنَّهٗ بِسْمِ اللّٰهِ الرَّحْمٰنِ الرَّحِيْمِ ﴿٣٠﴾ — اس (ملکہ) نے کہا اے دربار والو! بے شک میرے پاس ایک عزت والا خط بھیجا گیا ہے۔ بے شک وہ سلیمان (علیہ السلام) کی طرف سے ہے اور بے شک وہ اللہ کے نام سے (شروع کیا گیا) ہے جو بڑا مہربان نہایت رحم فرمانے والا ہے۔ || She said: "O eminent ones, indeed, to me has been delivered a noble letter. Indeed, it is from Solomon, and indeed, it reads: 'In the name of Allah, the Most Gracious, the Most Merciful, || هغې (ملکې) وویل: ای درباریانو! بېشکه ماته یو باعزته لیک لېږل شوی. بېشکه هغه د سلیمان (علیه السلام) له لوري دی، او بېشکه د الله په نامه (پیل شوی) چې ډېر مهربان، بې حده رحم کوونکی دی.
p اَلَّا تَعْلُوْا عَلَيَّ وَاْتُوْنِيْ مُسْلِمِيْنَ ﴿٣١﴾ — یہ کہ میرے مقابلے میں سرکشی نہ کرو اور فرماں بردار بن کر میرے پاس آ جاؤ۔ || Be not haughty with me but come to me in submission.'" || چې زما پر وړاندې سرکشي مه کوئ او فرمانبردار شوي ماته راشئ.
p قَالَتْ يٰٓاَيُّهَا الْمَلَؤُا اَفْتُوْنِيْ فِيْٓ اَمْرِيْ ۚ مَا كُنْتُ قَاطِعَةً اَمْرًا حَتّٰى تَشْهَدُوْنِ ﴿٣٢﴾ — اس (ملکہ) نے کہا اے دربار والو! میرے معاملے میں مجھے مشورہ دو میں کسی معاملے میں کوئی قطعی فیصلہ نہیں کرتی جب تک تم میرے پاس موجود نہ ہو۔ || She said: "O eminent ones, advise me in my affair. I would not decide a matter until you witness (for) me." || هغې وویل: ای درباریانو! زما په چاره کې ماته مشوره راکړئ؛ زه په هېڅ چاره کې پرېکړه نه کوم تر څو چې تاسو حاضر نه یاست.
p قَالُوْا نَحْنُ اُولُوْا قُوَّةٍ وَّاُولُوْا بَاْسٍ شَدِيْدٍ ڏ وَّالْاَمْرُ اِلَيْكِ فَانْظُرِيْ مَاذَا تَاْمُرِيْنَ ﴿٣٣﴾ — انھوں نے کہا ہم بڑی قوت والے اور بہت سخت جنگ والے ہیں اور حکم کا اختیار آپ کو ہے آپ خود ہی غور کریں کہ آپ ہمیں کیا حکم دیتی ہیں۔ || They said: "We are men of strength and of great military might, but the command is yours, so see what you will command." || هغوی وویل: موږ د لوی ځواک او سختې جګړې خاوندان یو، او د امر واک ستا دی، نو پخپله فکر وکړه چې څه امر راته کوې.
p قَالَتْ اِنَّ الْمُلُوْكَ اِذَا دَخَلُوْا قَرْيَةً اَفْسَدُوْهَا وَجَعَلُوْٓا اَعِزَّةَ اَهْلِهَآ اَذِلَّةً ۚ وَكَذٰلِكَ يَفْعَلُوْنَ ﴿٣٤﴾ — اس (ملکہ) نے کہا بے شک بادشاہ جب کسی بستی میں داخل ہوتے ہیں تو اسے تباہ کر دیتے ہیں اور وہاں کے عزت والے لوگوں کو ذلیل کر دیتے ہیں اور یہ (بھی) اسی طرح کریں گے۔ || She said: "Indeed, kings — when they enter a city, they ruin it and render the honoured of its people humbled. And thus do they do." || هغې وویل: بېشکه پاچایان چې کوم کلي ته ننوځي، وران یې کړي او د هغه ځای عزتمن خلک ذلیل کړي؛ او دوی به (هم) همداسې وکړي.
p وَاِنِّىْ مُرْسِلَةٌ اِلَيْهِمْ بِهَدِيَّةٍ فَنٰظِرَةٌۢ بِمَ يَرْجِعُ الْمُرْسَلُوْنَ ﴿٣٥﴾ — اور میں ان کی طرف کچھ تحفہ بھیجتی ہوں پھر دیکھتی ہوں قاصد کیا جواب لاتے ہیں؟ || "But indeed, I will send to them a gift and see with what (reply) the messengers will return." || او زه ورته یوه ډالۍ لېږم، بیا ګورم چې استازي څه ځواب راوړي.
p فَلَمَّا جَاۗءَ سُلَيْمٰنَ قَالَ اَتُمِدُّوْنَنِ بِمَالٍ ۡ فَمَآ اٰتٰىنِۦ اللّٰهُ خَيْرٌ مِّمَّآ اٰتٰىكُمْ ۚ بَلْ اَنْتُمْ بِهَدِيَّتِكُمْ تَفْرَحُوْنَ ﴿٣٦﴾ — تو جب وہ (قاصد) سلیمان (علیہ السلام) کے پاس آیا تو انھوں نے فرمایا کیا تم مال سے میری مدد کرنا چاہتے ہو؟ تو جو کچھ اللہ نے مجھے عطا فرمایا ہے وہ اس سے بہتر ہے جو اس نے تمھیں دیا ہے بلکہ تم ہی اپنے تحفے سے خوش رہو۔ || So when they came to Solomon, he said: "Do you provide me with wealth? But what Allah has given me is better than what He has given you. Rather, it is you who rejoice in your gift." || کله چې (استازی) سلیمان (علیه السلام) ته راغی، هغه وویل: ایا په مال زما مرسته کوئ؟ څه چې الله ماته راکړي، له هغه غوره دي چې تاسو ته یې درکړي؛ بلکې تاسو پخپله په خپله ډالۍ خوشحاله اوسئ.
p اِرْجِعْ اِلَيْهِمْ فَلَنَاْتِيَنَّهُمْ بِجُنُوْدٍ لَّا قِبَلَ لَهُمْ بِهَا وَلَنُخْرِجَنَّهُمْ مِّنْهَآ اَذِلَّةً وَّهُمْ صٰغِرُوْنَ ﴿٣٧﴾ — تم ان کے پاس واپس چلے جاؤ ہم ان پر ایسے لشکر لے کر آئیں گے جن کا وہ مقابلہ نہیں کر سکیں گے اور ہم انھیں وہاں سے ایسا ذلیل کر کے نکالیں گے کہ وہ بے عزت ہو کر رہ جائیں گے۔ || "Return to them, for we will surely come to them with soldiers that they will be powerless to encounter, and we will surely expel them therefrom in humiliation, and they will be debased." || هغوی ته بېرته ورشه؛ موږ به پرې داسې لښکرې راوړو چې مقابله به یې ونه شي کړای، او له هغه ځایه به یې په داسې ذلت وباسو چې بې عزته به شي.
p قَالَ يٰٓاَيُّهَا الْمَلَؤُا اَيُّكُمْ يَاْتِيْنِيْ بِعَرْشِهَا قَبْلَ اَنْ يَّاْتُوْنِيْ مُسْلِمِيْنَ ﴿٣٨﴾ — (سلیمان علیہ السلام نے) فرمایا اے دربار والو! تم میں سے کون اس (ملکہ) کا تخت میرے پاس لا سکتا ہے؟ اس سے پہلے کہ وہ فرماں بردار ہو کر میرے پاس آ جائیں۔ || (Solomon) said: "O assembly (of jinn), which of you will bring me her throne before they come to me in submission?" || (سلیمان علیه السلام) وویل: ای درباریانو! په تاسو کې څوک د هغې (ملکې) تخت ماته راوړلی شي، مخکې له دې چې هغوی فرمانبردار شوي ماته راشي؟
p قَالَ عِفْرِيْتٌ مِّنَ الْجِنِّ اَنَا اٰتِيْكَ بِهٖ قَبْلَ اَنْ تَقُوْمَ مِنْ مَّقَامِكَ ۚ وَاِنِّىْ عَلَيْهِ لَقَوِيٌّ اَمِيْنٌ ﴿٣٩﴾ قَالَ الَّذِيْ عِنْدَهٗ عِلْمٌ مِّنَ الْكِتٰبِ اَنَا اٰتِيْكَ بِهٖ … — ایک قوی ہیکل جن نے کہا میں اسے آپ کے پاس لا سکتا ہوں اس سے پہلے کہ آپ اپنے مقام سے اٹھیں اور بے شک میں اس پر پوری قوت رکھتا ہوں (اور) امانت دار بھی ہوں۔ اس نے کہا جس کے پاس کتاب کا علم تھا میں اسے آپ کے پاس لا سکتا ہوں … || A powerful one from among the jinn said: "I will bring it to you before you rise from your place, and indeed, I am for this (task) strong and trustworthy."
